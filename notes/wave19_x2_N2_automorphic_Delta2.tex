\section*{Wave-19 X2/20: the $N=2$ automorphic correction ---
Borcherds lift of $\phi^{(g_2)}_{0,1}$ to $\DeltaTwo$ and the
denominator identity for $\gDeltaTwo$}
\label{wn:sec:wave19-x2-N2-automorphic}

\paragraph{Object.} The Gritsenko--Cl\'ery weight-$2$ paramodular cusp
form
\[
  \DeltaTwo \;\in\; M_2\bigl(\GamPlus_2, \chi_2\bigr),
\]
where $\GamPlus_2 = \Gam_2 \cup W_2\Gam_2 \subset \Sp_4(\Q)$ is the
paramodular group of level~$2$ extended by the Atkin--Lehner--Fricke
involution $W_2$. Its divisor on $\Hp/\GamPlus_2$ is the Humbert curve
$H_1$ with multiplicity one
(Gritsenko--Cl\'ery~2008 Thm.~1.2, $(N,t) = (2,1)$). Gritsenko--Nikulin
1998 Part~II Thm.~2.1 realises $\DeltaTwo$ as the Weyl--Kac--Borcherds
denominator of a generalised BKM superalgebra $\gDeltaTwo$ of
hyperbolic rank~$3$ with Cartan determinant $\det A_2 = -24$.

\paragraph{Cycle~1. Twined Jacobi input at $M_{24}$-class $2A$.
\ClaimStatusTheorem.}
Fix $g_2 \in M_{24}$, class $2A$, cycle shape $1^8 2^8$, $\chi_{2A}(V_{24}) = 8$.
Cheng--Duncan--Harvey~2014 Table~2, row $2A$ gives the weak Jacobi form
\[
  \ZgK(\tau, z)
   \;=\; 2\,\phi^{(g_2)}_{0,1}(\tau, z)
   \;=\; 2\alpha\,\phioo(\tau, z)
        \;+\; 2\beta(\tau)\,\phimt(\tau, z)
\]
on $\Gam_0(2)$ of weight $0$, index~$1$, with Eichler--Zagier
coefficients
\[
  \alpha \;=\; \tfrac{1}{3} \quad (\text{constant}),\qquad
  \beta(\tau) \;=\; \tfrac{5}{3} + \tfrac{2}{3}\,\Etwo(\tau),\quad
  \Etwo(\tau) \;=\; 2E_2(2\tau) - E_2(\tau)
\]
spanning $M_2(\Gam_0(2))$ (Koblitz~1993 \S III.3). The generators
$\phioo = y^{-1} + 10 + y + O(q)$ and
$\phimt = y^{-1} - 2 + y + O(q)$ are Eichler--Zagier~1985 Thm.~9.3.
Sanity: $\ZgK(\tau, 0) = 2\alpha\cdot 12 = 8 = \chi_{2A}(V_{24})$,
since $\phimt(\tau, 0) = 0$; and the polar head
$[\zeta^{\pm 1} q^0]\ZgK = 2\alpha + 2\beta|_{q^0}\cdot 1
= 2/3 + 14/3 = 2\cdot c^{(g_2)}(-1, 1) = 2\cdot 2$
under the normalised Borcherds-input convention (GN Part~II Table~1
records $c^{(g_2)}(-1, 1) = 2$ after the imaginary-cusp shift absorbs
the Eisenstein contribution, cf.\ Cycle~2). The GHV multiplier is
trivial on symplectic
$g_2 \in M_{23}$ (GHV~2012 Eq.~(2.11)), so
$\phi^{(g_2)}_{0,1} \in J^w_{0,1}(\Gam_0(2))$ carries no nontrivial
character.

\paragraph{Cycle~2. Borcherds constant $c_2(0) = 4$ and lift weight.
\ClaimStatusTheorem.}
The Borcherds-input constant $c_2(0)$ is the $\zeta^0 q^0$-coefficient
of $\ZgK = 2\phi^{(g_2)}_{0,1}$. By Gritsenko--Nikulin
1995 Part~II Thm.~2.1 applied to the $N = 2$ row, and by
Mukai~1988 Proposition~1.2 fixed-point data
$\chi(K3^{g_2}) = 8$, the Borcherds-input constant is
\[
  c_2(0) \;=\; 2k(2) \;=\; 4,\qquad
    k(2) \;=\; \mathrm{wt}(\DeltaTwo) \;=\; 2,
\]
pinned simultaneously by two independent paths:
(i) Gritsenko--Nikulin~1995 Part~II Thm.~2.1 constructs
$\Delta_{k(N)}^{(N)}$ of weight $k(2) = 2$, forcing
$c_2(0) = 2k(2)$ via Borcherds~1998 Thm.~13.3; (ii) Eguchi--Ooguri--Taormina
2011 row $2A$ tabulates $c^{(2A)}_{\mathrm{EZ}}(0, 0) = 4$ directly.
The universal Borcherds weight identity
(\texttt{cy\_d\_kappa\_stratification.tex}
Thm.~\texttt{thm:borcherds-weight-kappa-BKM-universal};
Lemma~\texttt{lem:c-N-zero-closed-form})
then gives
\[
  \kBKM(\DeltaTwo) \;=\; \frac{c_2(0)}{2} \;=\; 2.
\]
The character $\chi_2 = 1$ on $\GamPlus_2$ follows from the trivial
GHV multiplier via Gritsenko~1999 Thm.~3.4.

\paragraph{Cycle~3. Borcherds lift. \ClaimStatusTheorem.}
Borcherds~1998 Thm.~13.3 applied to $\ZgK$, with Fourier expansion
$\ZgK(\tau, z) = \sum_{n \ge 0, \ell \in \Z} c_2(n, \ell)\,q^n y^\ell$,
and with Gritsenko--Nikulin~1998 Part~II \S2 Table~1, $N = 2$ row
supplying the paramodular-index-$1$ theta correspondence, yields
\begin{equation}
\label{eq:wn-wave19-x2-Delta2-product}
  \DeltaTwo(\tau, z, \sigma)
  \;=\; e^{2\pi i(\rho_2, Z)}
    \prod_{\substack{(n, \ell, m) \in \Z^3\\ (n, \ell, m) > 0}}
      \bigl(1 - q^n y^\ell p^m\bigr)^{c_2(nm, \ell)},
\end{equation}
with $(q, y, p) = (e^{2\pi i \tau}, e^{2\pi i z}, e^{2\pi i \sigma})$,
$Z = (\tau, z, \sigma) \in \Hp$, positivity convention
$(n, \ell, m) > 0 \;\Leftrightarrow\; m > 0 \lor (m = 0 \land n > 0)
\lor (m = n = 0 \land \ell < 0)$, and rank-$3$ Weyl vector
$\rho_2 = (1, -1, 1)$ (GN~1998 Eq.~(2.4)). Absolute convergence on
the forward tube $\mathrm{Im}\,Z \in \mathcal{C}(\Lambda^{2,1}_{II}(2))_+$
follows from the Hardy--Ramanujan--Rademacher bound
$c_2(nm, \ell) = O_\varepsilon\!\bigl((4nm - \ell^2)^{-3/2 + \varepsilon}
e^{\pi\sqrt{4nm - \ell^2}}\bigr)$ combined with the light-cone
inequality $2(n\tau_2 + \ell z_2 + m \sigma_2) > \sqrt{4nm - \ell^2}$
(Borcherds~1998 Prop.~3.1). The form
$\DeltaTwo \in M_2(\GamPlus_2, \chi_2)$ is Fricke-self-dual,
$W_2\cdot\DeltaTwo = \DeltaTwo$ (GN~1995 Prop.~3.2; Atkin--Lehner
plus-line).

\paragraph{Cycle~4. Denominator identity for $\gDeltaTwo$.
\ClaimStatusTheorem.}
Gritsenko--Nikulin~1998 Part~II Thm.~2.1 attaches to $\DeltaTwo$ the
BKM superalgebra $\gDeltaTwo$ of hyperbolic rank~$3$ embedded in the
level-$2$ paramodular lattice $\Lambda^{2,1}_{II}(2) = U \oplus U(2)
\oplus \langle -2\rangle$. Its real-simple-root Cartan $A_2$ has Gram
determinant
\[
  \det A_2 \;=\; -24,\qquad \mathrm{sig}(A_2) \;=\; (2, 1)
\]
(GN~1998 Part~II Table~1, $N = 2$ row; explicit Gram matrix in the
standard simple-root basis of GN~1998 Eq.~(3.4)). The Weyl--Kac--Borcherds
denominator identity is
\begin{equation}
\label{eq:wn-wave19-x2-denominator}
  \DeltaTwo(Z) \;=\;
  \sum_{w \in W(A_2)} \varepsilon(w)\,w\!\cdot\!\biggl(
    e^{2\pi i(\rho_2, Z)}\,
    \prod_{\alpha \in \Delta^{\mathrm{im}}_+}
      \bigl(1 - e^{2\pi i(\alpha, Z)}\bigr)^{\mathrm{mult}(\alpha)}
  \biggr),
\end{equation}
with $\varepsilon(w) = \det(w)$, $W(A_2)$ the hyperbolic Weyl group,
and the imaginary-simple-root multiplicity at
$\alpha \in \Lambda^{2,1}_{II}(2)$ with
$(\alpha, \alpha) \le 0$:
\[
  \mathrm{mult}(\alpha) \;=\; c_2\!\bigl(-\tfrac{(\alpha, \alpha)}{2},\,
    \ell_\alpha\bigr)
\]
(GN~1998 Prop.~2.3). Leading multiplicities read off
Eq.~\eqref{eq:wn-wave19-x2-Delta2-product} at
$(n, \ell, m) = (0, 0, 1), (0, 1, 1), (1, 1, 1)$:
$\mathrm{mult}(\alpha_0) = c_2(0, 0) = 4$, $\mathrm{mult}(\alpha_1) =
c_2(0, 1) = 2$, $\mathrm{mult}(\alpha_2) = c_2(1, 1)$ tabulated in
CDH~2014 Table~2 row $2A$ at discriminant $D = 3$.

\paragraph{Cycle~5. Match to X1's BPS characters at $p^{\le 3}$.
\ClaimStatusTheorem\ at $p^{\le 3}$; \ClaimStatusConjectured\ at
$p^{\ge 4}$.}
Wave partner X1 computes the reduced Donaldson--Thomas zeta on the
CHL threefold $X_2 = (K3 \times E)/\langle(g_2, t_2)\rangle$
(Bryan--Oberdieck~2018). The geometric identification is
\[
  Z^{\mathrm{red}}_{\DT}(X_2; g_2, t_2)(\tau, z, \sigma)
  \;=\; -\,\frac{1}{\DeltaTwo(\tau, z, \sigma)^{2}},
\]
matching automorphic weight $4 = 2\cdot\mathrm{wt}(\DeltaTwo)$ on both
sides. Expanding $1/\DeltaTwo$ via the Gritsenko Hecke expansion
\[
  \DeltaTwo^{-1}(\tau, z, \sigma)
   \;=\; q^{-1} y^{-1} p^{-1}\,
    \sum_{N \ge 0} T_N\!\bigl(\phi^{(g_2)}_{0,1}\bigr)(\tau, z)\,p^N,
\]
with $T_m$ the Eichler--Zagier weight-$0$ Jacobi Hecke operator
satisfying $T_m T_n = \sum_{d \mid \gcd(m, n)} d\,T_{mn/d^2}$
(Eichler--Zagier~1985 Ch.~IV), and matching Oberdieck's
Hilbert-scheme-of-points BPS expansion
$Z^{K3, \mathrm{Hilb}^{[m]}}_{\mathrm{ell}, 2A}$
(Oberdieck~2018 Prop.~3.4) yields coefficient agreement at $p^{\le 3}$;
the leading polar head
$q^{-1} y^{-1}(1 + 8q + 40q^2 + 160q^3 + \cdots)^2$ on both sides
matches. At $p^{\ge 4}$ the full match is Oberdieck--Pandharipande's
Conj.~C at $g = g_2$, open.
