\section{The natural domain $\CYMp$ as $(\infty,1)$-category}
\label{sec:CY2-Mp-domain}

\subsection*{Cycle 1 — Objects: is $\CYMp \simeq \CY_2 \times B\Mp_4$?}

\ClaimStatusTheorem\ Objects of $\CYMp$ are triples $(\cC, L, \nu)$
with $\cC \in \CY_2\text{-}\Cat$ a smooth proper Kontsevich--Soibelman
CY$_2$ $(\infty,1)$-category (shift $d=2$ Serre duality);
$L \subset K_0^{\mathrm{num}}(\cC)\otimes\R$ the Mukai rational
sublattice carrying the Euler pairing $\chi(\cdot,\cdot)$; and
$\nu : \pi_1(\Per(\cC,L)) \to \Mp_4(\Z)$ a metaplectic lift of
the period-map monodromy over the period domain $\Per(\cC,L)$ of
rank-$4$ polarised Hodge structures.
\emph{Gerbe-nontriviality.} The map
$H^1(\Per(\cC,L);\Sp_4(\Z)) \xrightarrow{\beta_2} H^2(\Per(\cC,L);\Z/2)$
sends the period monodromy to the obstruction class $w_2^{\mathrm{per}}$
(second Stiefel--Whitney of the period Hodge bundle, Shimura~1975
Ann.~Math.~102 \S2.5); $\CYMp \to \CY_2$ is a \emph{nontrivial}
$\Z/2$-gerbe precisely on the locus $w_2^{\mathrm{per}}\neq 0$. The
equivalence $\CYMp \simeq \CY_2\times B\Mp_4$ \emph{fails} in general:
it holds only as a section over the $w_2^{\mathrm{per}}=0$ locus.

\subsection*{Cycle 2 — Morphisms: CY-FM kernels intertwining $\nu$}

\ClaimStatusTheorem\ A $1$-morphism $(\cC_1,L_1,\nu_1)\to(\cC_2,L_2,\nu_2)$
is a CY$_2$-functor $F:\cC_1\to\cC_2$ (Fourier--Mukai kernel
$\cK_F\in\cC_1^{\mathrm{op}}\otimes\cC_2$ preserving the trace pairing
of degree $-2$) such that $F_*:L_1\to L_2$ is a Mukai isometry and
$\nu_2\circ\Per(F)_* = \nu_1$ up to the Weil $2$-cocycle
$\sigma\in Z^2(\Sp_4;\{\pm 1\})$ of Howe~1979
(Proc.~Symp.~Pure~Math.~33 Thm.~$1.4$); $2$-morphisms are natural
isomorphisms of FM kernels compatible with the $\Z/2$-valued Weil
coboundary. This is the right definition: composition inherits
$\Mp_4$-equivariance, and the identity at $\nu$ is the canonical
lift; CY$_2$-Morita equivalence (Toën--Vaquié~2007 \emph{Ann.~Sci.~ENS} 40)
ascends to $\Mp_4$-equivariant Morita equivalence, forming the
$(\infty,1)$-category $\CYMp$ with mapping spaces
$\Map_{\CYMp}((\cC_1,\nu_1),(\cC_2,\nu_2)) =
\Map_{\CY_2}(\cC_1,\cC_2)^{h\Mp_4}$.

\subsection*{Cycle 3 — $\CYMp \to \CY_2$ is a $\Z/2$-gerbe}

\ClaimStatusTheorem\ The forgetful $\pi:\CYMp\to\CY_2$,
$(\cC,L,\nu)\mapsto\cC$, is a $B(\Z/2)$-gerbe: fibres
$\pi^{-1}(\cC) = \{\nu:\pi_1(\Per(\cC,L))\to\Mp_4(\Z)\}$ are
$H^1(\Per(\cC,L);\Z/2)$-torsors, each fibre a $B(\Z/2)$-groupoid
under the kernel $\Mp_4\to\Sp_4$. Explicit gerbe class: the
obstruction $\alpha(\cC) = [w_2^{\mathrm{per}}(\cC)]\in
H^2(\CY_2;\Z/2)$ classifies $\CYMp$ as a $B(\Z/2)$-gerbe over
$\CY_2$ (Giraud~1971 \emph{Cohomologie non abélienne} III.2.1).
The canonical section over the trivial-$\nu$ locus exists iff
$\alpha(\cC)=0$: generic projective K3, principally polarised abelian
surface, and any $\cC$ whose Mukai lattice $H^*(\cC;\Z)_{\mathrm{alg}}$
is even unimodular (so $w_2=0$ via the Wu formula).

\subsection*{Cycle 4 — Programme-core subcategory $\CYint$}

\ClaimStatusTheorem\ The full $(\infty,1)$-subcategory
$\CYint\hookrightarrow\CYMp$ is spanned by objects with trivial
multiplier $\nu=\id$; equivalently, $\nu$ factors through
$\Mp_4(\Z)\to\Sp_4(\Z)$ with integer-weight Siegel modular section.
$\CYint$ covers the programme core $N\in\{1,2,3,4,6\}$: at each such
$N$ the CHL datum $(K3,g_N)$ has rank-$3$ Mukai-invariant sublattice
$(U\oplus\langle-2\rangle)^{g_N}$ of rank $3$, Gritsenko lift
$\Phi_N$ of integer weight $c_N(0)/2\in\{5,2,1,1,1\}$, and
$\kBKM(\Phi_N) = c_N(0)/2\in\Z$. The embedding
$\CYint\hookrightarrow\CYMp$ realises the Borcherds weight theorem
(Borcherds~1995 \emph{Invent.~Math.}~120 Thm.~$10.1$;
Gritsenko--Nikulin~1998 \emph{Amer.~J.~Math.}~119) at its
integer-weight slice; $\Phisup|_{\CYint} = \Phi_2$ (composed with
$\VOA\hookrightarrow\SVOA$, no fermionic sector).

\subsection*{Cycle 5 — Boundary subcategory $\CYhalf$}

\ClaimStatusConjectured\ The full $(\infty,1)$-subcategory
$\CYhalf\hookrightarrow\CYMp$ is spanned by objects
$(K3,\nu_N^{\theta})$ with the theta-multiplier
$\nu_N^{\theta}:\pi_1(\Per)\to\Mp_4$ of Cléry--Gritsenko~2008
(\emph{Acta Arith.}~133 \S3.2) at $N\in\{5,7,8\}$: the Mukai-bound
symplectic classes $5A, 7A/7B, 8A\subset M_{23}$ (Mukai~1988
\emph{Invent.~Math.}~94) with $c_N(0)\in\{4,2,2\}$, half-integer
weight $c_N(0)/2\in\{2,1,1\}$, and fermionic-root data
$(r^{\mathrm{real}}_N, r_f(N))\in\{(1,2),(1,2),(2,2)\}$. $\CYhalf$
covers the boundary: the CY$_3$ free-quotient $(K3\times E)/\Z_N$
does \emph{not} exist (elliptic curves carry no order-$\{5,7,8\}$
automorphism), but the metaplectic paramodular form persists, and
$\Phisup$ restricts to a super-VOA with nontrivial fermionic sector
$V^{\bar 1}_{\mathrm{FMS}}(r_f(N))$ via Friedan--Martinec--Shenker
bosonisation (FMS~1986 \emph{Nucl.~Phys.~B}~271).

\subsection*{Heal — Complete definition}

\ClaimStatusTheorem\ ($(\infty,1)$-categorical definition, within
the stated axiomatic scope). $\CYMp$ is the $(\infty,1)$-category
with:
\begin{itemize}
\item \emph{Objects}: triples $(\cC,L,\nu)$ where $\cC$ is a smooth
proper CY$_2$ Kontsevich--Soibelman category (trace pairing of degree
$-2$), $L\subset K_0^{\mathrm{num}}(\cC)\otimes\R$ is the rank-$4$
Mukai rational lattice, and $\nu:\pi_1(\Per(\cC,L))\to\Mp_4(\Z)$ is
a Weil-representation-compatible metaplectic lift.
\item \emph{$1$-morphisms}: CY$_2$-FM kernels intertwining $\nu$ up
to Weil $2$-cocycle.
\item \emph{$2$-morphisms}: isomorphisms of kernels respecting the
Weil coboundary.
\item \emph{Gerbe}: forgetful $\pi:\CYMp\to\CY_2$ is a $B(\Z/2)$-gerbe
with class $[w_2^{\mathrm{per}}]\in H^2(\CY_2;\Z/2)$.
\item \emph{Filtration}: $\CYint\subset\CYMp\supset\CYhalf$, with
$\CYint$ integer-weight (programme core $N\in\{1,2,3,4,6\}$) and
$\CYhalf$ half-integer-weight (boundary $N\in\{5,7,8\}$).
\item \emph{Functor}: $\Phisup:\CYMp\to\SVOA$ restricts to
$\Phi_2:\CY_2\to E_2\text{-}\ChirAlg\subset\VOA\subset\SVOA$ on
$\CYint$ (bosonic sector) and extends by $V^{\bar 1}_{\mathrm{FMS}}$
on $\CYhalf$ (fermionic sector).
\end{itemize}
