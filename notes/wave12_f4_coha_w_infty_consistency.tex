\author{Raeez Lorgat}

\begin{abstract}
For affine three-space the cohomological Hall algebra is the positive
half $Y^+(\gghone)$. The vertex algebra $\Winf$ appears after the
additional double, centre, completion, and evaluation data. The full
affine Yangian is obtained by Drinfeld doubling, and $\Winf[\lambda]$ is
reached only through Fock or vacuum evaluation. The same typing governs
the conifold and $K3\times E$: for $d\ge 3$ the specialised
CY-to-chiral output is native $\Eone$, while $\Etwo$ lives on a centre,
double, or representation category after the required pairing and
completion data have been supplied.
\end{abstract}

\section{Conventions}
\label{sec:conventions}

The four scalar invariants are
\[
 \kch,\qquad \kcat=\chi(\cO_X),\qquad
 \kBKM=c_N(0)/2,\qquad \kfib.
\]
They are not interchangeable. The non-compact examples used below have
the direct chain-level values
\[
 \kch(\mathrm{conifold})=1,\qquad
 \kch(K_{\P^2})=\frac32.
\]
The first comes from the two-vertex Klebanov-Witten McKay/NCCR
calculation; the second from the Beilinson $\mathbb Z/3$-McKay local
surface calculation. For $d\ge 3$, the specialised CY-to-chiral output
is $\Eone$:
\[
 \Phi_d^{(\Sigma_{d-1},C)}(\cC)
 =\SpCh_{\Sigma_{d-1},C}\bigl(\PhiFA_d(\cC)\bigr)
 \in \Eone\text{-}\mathrm{ChirAlg}(C).
\]
The algebra $A=\Phi_d^{(\Sigma_{d-1},C)}(\cC)$ has native $\Eone$
order.  Braiding is a structure on $\cZ(\Rep^{\Eone}(A))$ or on the
completed Drinfeld double when the pairing, negative half, Cartan
factor, and completion have been constructed.

\section{Affine Three-Space}
\label{sec:c3}

\begin{theorem}[Positive half, double, and evaluation]
\label{thm:c3-dictionary}
Let $Q_{\CC^3}$ be the one-vertex quiver with loops $x,y,z$ and
potential $W_{\CC^3}=\mathrm{tr}(x[y,z])$. Its Jacobi algebra is $\CC[x,y,z]$,
and after the standard equivariant localisation its cohomological Hall
algebra satisfies
\[
 \CoHA(\CC^3)\simeq Y^+(\gghone)
\]
as an associative shuffle algebra. In the additive Schiffmann-Vasserot
normalisation used here, the two-variable kernel is
\[
 \omega(z,w)=
 \frac{(z-w-\epsilon_1)(z-w-\epsilon_2)(z-w-\epsilon_3)}
 {(z-w)^3},
 \qquad \epsilon_1+\epsilon_2+\epsilon_3=0.
\]
The completed Drinfeld double
\[
 Y(\gghone)=D(Y^+(\gghone))=
 Y^-\widehat{\bowtie}Y^0\widehat{\bowtie}Y^+
\]
is the affine Yangian in Drinfeld-current form. The vertex algebra
$\Winf[\lambda]$ is obtained only after representation:
\[
 Y^+(\gghone)\hookrightarrow Y(\gghone)
 \xrightarrow{\ \mathrm{ev}_\lambda\ }
 \End(\Winf[\lambda]\text{-vac}).
\]
The last arrow is a representation-theoretic evaluation, not an
isomorphism of associative algebras.
\end{theorem}

\begin{proof}
Schiffmann and Vasserot identify the localised cohomological Hall
algebra of $\CC^3$ with the positive shuffle algebra. Tsymbaliuk identifies the
Drinfeld double with the affine Yangian. Gaiotto and Rapcak, and
Prochazka and Rapcak, realise $\Winf[\lambda]$ on the vacuum/Fock
representation. The three objects differ by associativity class:
$\CoHA(\CC^3)$ and $Y(\gghone)$ are associative $\Eone$ algebras;
$\Winf[\lambda]$ is a vertex algebra on a curve.
\end{proof}

\begin{remark}
The compact formula is
\[
 \CoHA(\CC^3)=Y^+(\gghone),
 \qquad
 \Winf[\lambda]\ \text{is reached only after double, centre, and evaluation}.
\]
This is the same separation used for $K3\times E$: a Hall positive half
and a chiral or BKM target are connected by extra structure, not by
renaming one object as the other.
\end{remark}

\section{Parameters on the CY Surface}
\label{sec:parameters}

Let $(\epsilon_1,\epsilon_2,\epsilon_3)$ be the equivariant weights of
the torus action on $\CC^3$. The torus-equivariant Calabi-Yau identity
is
\[
 \epsilon_1+\epsilon_2+\epsilon_3=0.
\]
After imposing it, the affine-Yangian and $\Winf[\lambda]$ families are
controlled by one ratio, but two useful slices must be distinguished:
\[
 \epsilon_1+\epsilon_2=0
 \quad\text{and}\quad
 \epsilon_1=\epsilon_2.
\]
The first is the self-dual Heisenberg slice with $\epsilon_3=0$. The
second is the unrefined Miki-symmetric slice with
$\epsilon_3=-2\epsilon_1$. Neither statement fixes a universal numerical
value of $\lambda$ without specifying the Gaiotto-Rapcak convention.

\section{Three Calabi-Yau Conditions on $\CC^3$}
\label{sec:cy3-conditions}

For $\CC^3$ three compatible conditions are often compressed into one
sentence. They are distinct:
\begin{enumerate}[label=\textup{(\roman*)}]
\item The Jacobi algebra of the quiver with potential is a
Calabi-Yau algebra of dimension $3$ in the sense of Ginzburg and
Bocklandt.
\item The variety $\CC^3$ has holomorphic volume form
$dz_1\wedge dz_2\wedge dz_3$.
\item The chosen torus action preserves that volume form exactly when
$t_1t_2t_3=1$, equivalently
$\epsilon_1+\epsilon_2+\epsilon_3=0$.
\end{enumerate}
The third condition is the equivariant realisation of the first two; it
records only the torus-weight part of the algebraic Calabi-Yau datum.

\section{Holomorphic Chern-Simons}
\label{sec:hcs}

\begin{proposition}[Stage-one and stage-two hCS]
\label{prop:hcs-two-stage}
The six-real-dimensional holomorphic Chern-Simons observables on
$\CC^3$ form an $\Ethree$ algebra in the symmetric monoidal category of
Dolbeault sheaves:
\[
 \Obs_{\hCS}(\CC^3)\in \Ethree\text{-}\HolFA(\CC^3).
\]
Specialising along a transverse two-cycle and a curve produces an
$\Eone$ chiral algebra:
\[
 \SpCh_{\Sigma_2,C}\bigl(\Obs_{\hCS}(\CC^3)\bigr)
 \in \Eone\text{-}\mathrm{ChirAlg}(C).
\]
The $\Etwo$ structure is recovered on the centre or double of the
specialised algebra when the required pairing exists.
\end{proposition}

\begin{proof}
Costello and Gwilliam's factorisation algebras supply the holomorphic
local structure. The $\Ethree$ product is the three-direction
holomorphic factorisation product on the Dolbeault-BV observables; its
non-trivial content comes from the BV bracket, the holomorphic volume
form, and the $\Omega$-background deformation.  Simple connectivity
only fixes the affine chart. Restriction to a curve is a
boundary/specialisation operation and lowers the native chiral output to
$\Eone$.
\end{proof}

\section{The Conifold}
\label{sec:conifold}

Let
\[
 X_{\mathrm{con}}=\mathrm{Tot}(\cO(-1)^{\oplus 2}\to\P^1).
\]
The Klebanov-Witten quiver has two vertices $0,1$, arrows
$a_1,a_2:0\to1$ and $b_1,b_2:1\to0$, and quartic potential
\[
 W_{\mathrm{con}}=
 \mathrm{tr}(a_1b_1a_2b_2-a_1b_2a_2b_1).
\]
Its Jacobi algebra is Van den Bergh's NCCR of the conifold coordinate
ring $\CC[x,y,z,w]/(xy-zw)$. This is a two-colour CY$_3$ datum,
distinct from the one-colour $\CC^3$ chart and from local-surface
computations.

The conifold Hall algebra is the positive half
\[
 \cH(X_{\mathrm{con}})
 \cong Y^+_{\hbar}(\widehat{\mathfrak{gl}}(1|1))^{\mathrm{con}}
\]
in the conifold affine-super-Yangian normalisation. The ungraded shadow
may be read as a shifted $\widehat{\mathfrak{sl}}_2$ positive half after
supertrace projection, but the primary two-colour object is the
$\widehat{\mathfrak{gl}}(1|1)$ positive half. The full conifold quantum
group requires the double
\[
 D\bigl(Y^+_{\hbar}(\widehat{\mathfrak{gl}}(1|1))^{\mathrm{con}}\bigr),
\]
 and its vertex target is conifold-specific. The $\Winf(\gghone)$ target
belongs to the one-colour affine three-space double/evaluation lane.

The scalar chiral invariant is
\[
 \kch(X_{\mathrm{con}})=1.
\]
This is the direct McKay/Jacobi/NCCR value from the two vertices and the
single compact curve $\P^1$. The local surface formula gives instead
\[
 \kch(K_{\P^2})=\frac12\chi_{\mathrm{top}}(\P^2)=\frac32,
\]
for the local surface \(K_{\P^2}\).  The conifold value comes from the
two-colour NCCR calculation.

The conifold has two compatible Hall descriptions after the cohomology
theory is fixed:
\[
\begin{array}{p{0.35\textwidth}p{0.55\textwidth}}
\text{cohomological} &
Galakhov, Li, and Yamazaki identify the conifold critical CoHA with a
positive affine super-shuffle algebra over the equivariant parameters.\\[0.8ex]
\text{$K$-theoretic} &
Kapranov and Vasserot identify the $K$-theoretic Hall algebra with the
positive half of the relevant quantum toroidal algebra after
formal-group specialisation.
\end{array}
\]
The two statements are compatible through the shuffle-to-quantum-toroidal
comparison. A theorem identifying them must change cohomology theory and
state the formal-group specialisation.

\section{$K3\times E$}
\label{sec:k3e}

\begin{proposition}[Scope of compact CoHA on $K3\times E$]
\label{prop:k3e-coha-scope}
Davison and Davison-Meinhardt construct cohomological Hall algebras and
BPS Lie algebras for quivers with potential and for geometric settings
with the required global critical or d-critical structure. A direct
application to $K3\times E$ is therefore conditional on a global
critical-chart theorem for the relevant sheaf moduli. Character
agreement with reduced DT or Igusa data leaves that critical CoHA
unconstructed.
\end{proposition}

\begin{remark}[Hall-Borcherds double]
A compact endpoint has the following exact form if the compact positive
half, negative half, Hopf pairing, Cartan factor, and completion are
constructed:
\[
 G(K3\times E)=D\bigl(Y^+_{\mathrm{Hall}}(K3\times E)\bigr).
\]
The BKM side is the Hall-Drinfeld double of the $K3\times E$ positive
half. It is separate from the Mukai self-mirror K3 Yangian branch.
\end{remark}

\begin{proposition}[Primitive reduced DT scope]
\label{prop:primitive-scope}
The comparison
\[
 \Omega(\gamma)=\mathrm{mult}_{\mathrm{BKM}}(\alpha_\gamma)
\]
is proved on the primitive K3 curve-class locus where the reduced DT
partition function is identified with the Igusa cusp form. Extending
this formula to imprimitive classes requires the multiple-cover and
stable-pairs terms; no primitive calculation supplies them for free.
\end{proposition}

\section{Mukai-Equivariant Hall Algebra}
\label{sec:mukai-equivariant}

For a finite symplectic automorphism group $G$ of K3 and a torsion
subgroup $E[N]$ of the elliptic curve, the equivariant cohomology of
$K3\times E$ is defined. The corresponding
$G\times E[N]$-equivariant CoHA of semistable sheaves on $K3\times E$
is a theorem only after the following four data are constructed:
\begin{enumerate}
\item a compatible critical or d-critical atlas;
\item equivariant vanishing cycles;
\item Hall multiplication compatible with the $G\times E[N]$ action;
\item comparison with the Borcherds root multiplicities.
\end{enumerate}

\section{Six Constructions}
\label{sec:six-routes}

The six constructions reaching the $G(K3\times E)$ target are:
Mukai lattice, Igusa $\Phi_{10}$ via Gritsenko $\Delta_5$, BKM
Borcherds weight, K3 fibre rank, Heisenberg-Mukai specialisation, and
Maulik-Okounkov stable-envelope data. They are not six applications of
$\Phi$ to the same object. The two-stage formula
\[
 \Phi_d=\SpCh_{\Sigma_{d-1},C}\circ\PhiFA_d
\]
explains why different choices of $(\Sigma_{d-1},C)$ produce different
$\Eone$ shadows from one stage-one holomorphic factorisation algebra.

\section{Abelian Nekrasov Function}
\label{sec:nekrasov}

The fully equivariant $U(1)$ instanton partition function on
$\CC^2$ is a fixed-point sum over partitions:
\[
 Z^{\mathrm{inst}}_{\Omega,U(1)}(\epsilon_1,\epsilon_2;q)
 =
 \sum_{\mu}q^{|\mu|}
 \prod_{s\in\mu}
 \frac{1}{
 \bigl(a_\mu(s)\epsilon_1-(l_\mu(s)+1)\epsilon_2\bigr)
 \bigl((a_\mu(s)+1)\epsilon_1-l_\mu(s)\epsilon_2\bigr)}.
\]
The Euler-characteristic limit reduces to the partition generating
function
\[
 \prod_{n\ge 1}(1-q^n)^{-1},
\]
up to the usual $\eta$-function vacuum prefactor. The product formula is
therefore the nonequivariant Euler limit. The full equivariant Nekrasov
function retains the weight parameters.

The same grammar applies to other targets after the relevant positive
half, pairing, and completion have been constructed:
\[
 Y^+(X)\quad\longrightarrow\quad D(Y^+(X))\quad\longrightarrow\quad

\]
For $\CC^3$ all three arrows are constructed in the cited literature.
For the conifold the positive half is two-colour and conifold-specific.
For $K3\times E$ the reduced DT/Igusa character is proved on the
primitive locus, while the compact Hall positive half and its double are
unconstructed arrows.

\section{Unconstructed Arrows}
\label{sec:unconstructed-arrows}

\begin{enumerate}
\item Construct the global critical or d-critical atlas for the
$K3\times E$ sheaf moduli needed by compact CoHA.
\item Build the $G\times E[N]$-equivariant Hall algebra and compare its
positive half with the Borcherds multiplicity data.
\item Extend the primitive Oberdieck-Pixton comparison to imprimitive
classes.
\item Produce the compact Hall-Drinfeld double for $K3\times E$ and
separate it from the Mukai self-mirror Yangian branch at every step.
\item Keep the $\CC^3$ formula typed:
$\CoHA(\CC^3)=Y^+(\gghone)$, with $\Winf[\lambda]$ reached only after
doubling, centre, and evaluation.
\end{enumerate}
