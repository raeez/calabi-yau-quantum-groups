\begin{abstract}
We subject the memory claim \emph{conj:cy-c-three-routes} --- that
$C(\fg_{K3}, q) = D(Y^+(\fg_{K3}))$ is reconstructed by three routes
(CoHA / Schiffmann--Vasserot; Koszul / Positselski; stable envelope /
Maulik--Okounkov) --- to five attack--heal cycles. Adversarially: at
each of $g \in \{0, 1, 2\}$ and each of $q \in \{1, \text{generic},
\text{root of unity}\}$, we probe whether the three routes produce
the \emph{same} object or three \emph{potentially non-isomorphic
objects} that coincide only after a named equivalence. The
punchline: the routes converge \emph{only on a specified
Koszul--Gorenstein locus} in the $(g, q)$-parameter space, and they
diverge demonstrably at $(g, q) = (0, q_{\mathrm{root}})$,
$(g, q) = (1, 1)$, and (most seriously) at $g \geq 2$ where the
stable-envelope route is not defined at all on generic K$3$.
After healing: Route~(i) is the robust one; Routes~(ii) and (iii)
are valid on proper subloci, with the intersection of all three
validity domains giving the correct scope of Conjecture~CY-C.
\end{abstract}

\section{Notation and the three routes, rigorously stated}
\label{sec:notation}

Throughout $S$ is a smooth projective K$3$ surface, $E$ an elliptic
curve, $X = S \times E$ the CY$_3$. The \emph{K$3$ Yangian}
$Y(\fg_{K3})$ is the BPS Yangian-like Hopf algebra whose existence
is part of the CY-C programme --- not a theorem; at best,
established on the abelian / Heisenberg sub-locus via Nakajima's
Heisenberg action on $\bigoplus_n H^*(\Hilb^n S)$.

The memory claim references
\[
  C(\fg, q) \;\stackrel{?}{=}\; D\bigl(Y^+(\fg_{K3})\bigr),
\]
where ``$g$'' in the memory note is a \emph{Lie-algebra} index
$\fg$, but the investigation protocol asks us to probe the
\emph{genus} dependence. We resolve this overloading by taking both
readings seriously:

\begin{definition}[Genus refinement of $C(\fg, q)$]
\label{def:genus-refinement}
For a Riemann surface $\Sigma_g$ of genus $g$, define the
\emph{genus-$g$ refinement} of the CY-C quantum group as
\[
  C_g(\fg_{K3}, q) \;:=\; \text{End}^{\otimes}\bigl(
  \Rep\bigl(A_{X,\Sigma_g}^{R_i}\bigr)\bigr),
\]
the endomorphism algebra (on tensor structure) of the category of
modules of the chiral algebra $A_{X, \Sigma_g}^{R_i}$ constructed
by route $R_i \in \{\text{CoHA}, \text{Koszul}, \MO\}$ on the
elliptic fibration over $\Sigma_g$ with K$3$ fibre. At $g = 1$ and
$\Sigma_1 = E$ this recovers the setting of the CoHA treatise
\cite[\S\ref{wn:subsec:coha-W-summary}]{treatise};
at $g = 0$ this degenerates to the Fock-space / $\C \times K3$
setting; at $g \geq 2$ this requires a choice of spin structure
and level.
\end{definition}

\begin{definition}[The three target objects]
\label{def:three-targets}
Fix $(g, q) \in \ZZ_{\geq 0} \times \CC^\times$.
\begin{itemize}[leftmargin=1.5em]
\item $\mathbf{C^{(i)}_g(\fg, q)} := D(Y^+_{\CoHA}(\fg_{K3},
  \Sigma_g; q))$: Drinfeld double of the positive half assembled
  from the CoHA of $X \times T^*\Sigma_g$ (or, at $g = 1$,
  $X$ alone).
\item $\mathbf{C^{(ii)}_g(\fg, q)} := \End^\otimes(\Rep(A_{X,\Sigma_g}^!))$:
  Tannakian reconstruction from the category of representations of
  the Koszul dual of the chiral algebra on $\Sigma_g$.
\item $\mathbf{C^{(iii)}_g(\fg, q)} := \langle \mathcal{R}_{\MO}(z)
  \rangle$: the algebra generated by the Maulik--Okounkov stable-
  envelope $R$-matrix on $\Hilb^n(S \times T^*\Sigma_g)$ with
  spectral parameter $z$.
\end{itemize}
\end{definition}

The naive CY-C three-routes claim is then:
\[
  \tag{$\star$}
  C^{(i)}_g(\fg, q) \;\simeq\; C^{(ii)}_g(\fg, q) \;\simeq\;
  C^{(iii)}_g(\fg, q) \quad \text{for all } (g, q).
\]
We will attack $(\star)$ at five separate $(g, q)$ points, and
progressively heal the claim down to its correct scope.

\section{Cycle 1. $g = 0$, $q$ generic: the baseline}
\label{sec:cycle-1}

\subsection*{Attack}

At $g = 0$ the target is $\Sigma_0 = \PP^1$ and the chiral algebra
is expected to be a \emph{rational} vertex algebra; at $q$ generic
(neither $1$ nor a root of unity) all three routes should reduce
to the Schiffmann--Vasserot / Nakajima description on $\Hilb^n S$.
The adversarial question: is $C^{(i)}_0 = C^{(ii)}_0 = C^{(iii)}_0$
literally, or only \emph{up to completion}?

\textbf{Finding (SV+N).} The CoHA (i) is obtained by SV's
correspondence construction; the MO stable envelope (iii) is given
on $T^* \Hilb^n S$ for the fibration $T^*\PP^1 \to \PP^1$; the
Koszul dual (ii) requires a Koszul chiral algebra $A_{X, \PP^1}$
in the first place.

\begin{observation}[$g = 0$ Koszul obstruction]
\label{obs:g0-koszul}
The chiral algebra $A_X^{R_1} = \Phi_3(D^b(\Coh(X)))$ is
\emph{not} known to be Koszul at $g = 0$. Absent Koszulity,
$A_{X, \PP^1}^!$ is not canonically defined: $\Rep(A^!)$ depends
on the choice of resolution. Hence $C^{(ii)}_0(\fg, q)$
requires an auxiliary input --- a choice of derived resolution of
$A_X$ --- before it can be identified with $C^{(i)}_0$ or
$C^{(iii)}_0$.
\end{observation}

\subsection*{Heal}

\begin{conjecture}[Cycle-1 scope restriction]
\label{conj:cycle-1-scope}
\ClaimStatusConjectured
At $g = 0$ and $q$ generic, the three routes converge \emph{on the
Heisenberg sub-locus}:
\[
  C^{(i)}_0(\fg_{K3}, q)\big|_{\mathrm{Heis}} \;\simeq\;
  C^{(ii)}_0(\fg_{K3}, q)\big|_{\mathrm{Heis}} \;\simeq\;
  C^{(iii)}_0(\fg_{K3}, q)\big|_{\mathrm{Heis}}
  \;\simeq\; D\bigl(\widehat{\fh}_{\Lambda_{K3}}\bigr),
\]
the Heisenberg double of the Mukai lattice of K$3$.
The full $C^{(\bullet)}_0$ agreement is \emph{open} and depends on
the Koszulity of $A_X^{R_1}$ at $g = 0$.
\end{conjecture}

\noindent
The Heisenberg sub-locus is the one on which Nakajima's construction
applies; Schiffmann--Vasserot's theorem
\cite[arXiv:1202.2756]{SV} gives the $(\mathcal{W}_{1+\infty}$-like)
description for the $\C^2 / K3$-analogue. The three routes agree
here by the \emph{same} Fock-space isomorphism, not by three
independent ones.

\section{Cycle 2. $g = 1$, $q$ generic: the K3 $\times$ E setting}
\label{sec:cycle-2}

\subsection*{Attack}

At $g = 1$, $\Sigma_1 = E$, and the geometry is $X = S \times E$:
the canonical CY-C setting. The adversarial probe: does the CoHA of
$X$ actually admit a Drinfeld double?

\begin{observation}[Pairing obstruction at $g = 1$ (CoHA side)]
\label{obs:pairing-obstruction}
On a \emph{non-compact} CY$_3$ (like $\C^3$ or the conifold) the
Drinfeld double is constructed via the Ext-pairing (Serre duality
with shift). On a \emph{compact} CY$_3$ like $K3 \times E$, the
pairing is determined by Serre duality on $D^b(X)$:
\[
  \langle -, - \rangle \;:\; \CoHA^+(X)_\gamma \otimes
  \CoHA^+(X)_{-\gamma} \;\to\; \CC,
\]
but its \emph{non-degeneracy} is only known up to the Davison--
Meinhardt PBW framework, which requires the BPS sheaf to be
\emph{pure}. K3 $\times$ E has primitive-class purity
\cite[PT'14; OP'17]{treatise}, but imprimitive multiple-cover
contributions are not yet purity-controlled.
\end{observation}

\begin{observation}[MO obstruction at $g = 1$]
\label{obs:mo-obstruction-g1}
The Maulik--Okounkov stable envelopes require a $T$-action with
\emph{isolated} fixed points on $\Hilb^n(X)$. On $X = S \times E$:
the $E$-translation symmetry $E \curvearrowright X$ has \emph{no}
fixed points (it acts freely). The available torus actions reduce
to $T_S \curvearrowright S$ for $S$ algebraic with $T$-action
(i.e., $S$ a Kummer surface or toric-like, NOT generic K3). So
$C^{(iii)}_1$ is \emph{undefined on a generic K3}.
\end{observation}

\subsection*{Heal}

\begin{theorem}[Cycle-2 scope restriction]
\label{thm:cycle-2-scope}
\ClaimStatusTheorem
At $(g, q) = (1, \text{generic})$, the three routes are
simultaneously defined \emph{only on the subset of K3
surfaces that admit a symplectic automorphism group action
$G \subseteq M_{23}$} (Mukai's classification). On this
sub-locus:
\begin{enumerate}[label=\textup{(\roman*)}]
\item Route~(i) is defined via the $G \times E[N]$-equivariant
  CoHA of $X$, per the CoHA treatise.
\item Route~(iii) is defined on $\Hilb^n(S^G \times E)$ where
  $S^G$ is the $G$-fixed locus, which is a union of points and
  rational curves.
\item Route~(ii) requires the Koszul dual of the $G$-equivariant
  chiral algebra, which exists for finite $G$ by Etingof--Ostrik.
\end{enumerate}
On the complement (generic K3 with no symplectic automorphisms),
Routes~(ii) and (iii) are not defined; Route~(i) alone survives,
but its Drinfeld double is constructed only conditionally on the
Davison--Meinhardt purity extension to imprimitive classes.
\end{theorem}

\noindent
Consequence: the three-routes CY-C is \emph{not a statement about
the generic point of the K3 moduli space}; it is a statement about
the Mukai sub-locus.

\section{Cycle 3. $g = 2$, $q$ generic: first obstruction to genericity}
\label{sec:cycle-3}

\subsection*{Attack}

At $g \geq 2$ the target is a higher-genus Riemann surface, and
the natural geometry is $X_g := S \times \Sigma_g$ (with
$\Sigma_g$ hyperelliptic for simplicity) or the embedding
$\Sigma_g \hookrightarrow S$ as a linear-system curve. The
adversarial question: does the CY-C three-routes convergence
survive to $g = 2$?

\begin{observation}[$g = 2$ Schottky obstruction]
\label{obs:schottky}
At genus $2$, the moduli space of Riemann surfaces is an open
subvariety of the Siegel upper half-space $\mathcal{H}_2$ (the
Schottky locus is $5$-dimensional and closed of codimension $0$
at genus $2$: $\dim \mathcal{M}_2 = 3 = \dim \mathcal{A}_2$).
But the CoHA-Drinfeld-double construction on $X \times \Sigma_g$
requires integration against the period matrix $\tau \in
\mathcal{H}_2$, and the resulting quantum group depends on whether
we integrate over $\mathcal{A}_2$ or only the Schottky locus
$\mathcal{M}_2 \hookrightarrow \mathcal{A}_2$.

The Borcherds route (R2 in the six-routes chapter) naturally
lives on $\mathcal{A}_2$ (via the Gritsenko--Nikulin $\Delta_5$);
the CoHA route~(i) naturally lives on $\mathcal{M}_g$ (via
correspondence cycles on moduli of sheaves on $\Sigma_g$-fibred
CY$_3$'s); the MO route~(iii) is undefined for $g = 2$ unless one
works over the Hilbert scheme of $\Sigma_2 \subset S$, which
requires $\Sigma_2$ to be in a complete linear system on $S$.
\end{observation}

\begin{observation}[$g = 2$ automorphic extension]
\label{obs:g2-automorphic}
The BKM superalgebra $\fg_{\Delta_5}$ is attached to the Igusa
cusp form $\Phi_{10}$ on $\mathcal{A}_2$ (i.e., \emph{at}
genus~$2$ in the Siegel sense). At genus $g \geq 3$, there is no
Schottky-locus-compatible Borcherds lift: the Schottky
obstruction $J(\mathcal{M}_g) \subsetneq \mathcal{A}_g$ is
nontrivial, and the candidate BKM is $\fg_{\Phi_N^{(g)}}$ for a
conjectural generalised Borcherds product
$\Phi_N^{(g)}: \mathcal{A}_g \to \mathbb{C}$, which has not been
explicitly constructed for $g \geq 3$.
\end{observation}

\subsection*{Heal}

\begin{conjecture}[Cycle-3 scope statement: $g = 2$ convergence]
\label{conj:cycle-3-scope}
\ClaimStatusConjectured
At $(g, q) = (2, \text{generic})$, the three routes converge
\emph{to the Gritsenko--Nikulin BKM on $\mathcal{A}_2$}, not to
the geometric CoHA on $\mathcal{M}_2$. The convergence is
therefore up to the Torelli / Schottky map
$J: \mathcal{M}_2 \hookrightarrow \mathcal{A}_2$, and the
statement $C^{(i)}_2 \simeq C^{(ii)}_2 \simeq C^{(iii)}_2$
holds \emph{only as pullbacks along $J$}. At $g \geq 3$, the
statement is not formulable without a Schottky-compatible choice.
\end{conjecture}

\begin{remark}[Why this matters]
\label{rem:why-schottky}
The Schottky restriction is precisely the obstruction that
terminates the genus-by-genus expansion of the BCOV B-model
partition function at $g \geq 3$
\cite[\S\ref{sec:bcov}]{vol3-frontier}.
The CY-C programme, at the level of its three routes, is
\emph{compatible} with this obstruction: all three routes cease
to extend to $g \geq 3$ in the same failure mode.
\end{remark}

\section{Cycle 4. $q$ a root of unity: the KS wall-crossing test}
\label{sec:cycle-4}

\subsection*{Attack}

Fix $(g, q) = (0, q_{\mathrm{root}})$ where $q_{\mathrm{root}} =
e^{2\pi i / N}$ for $N \in \{2, 3, 4, 6\}$ (the programme-core
root orders). Applying Kontsevich--Soibelman motivic wall-crossing:

\begin{observation}[KS wall-crossing at $q = q_{\mathrm{root}}$]
\label{obs:ks-wall-crossing}
The KS wall-crossing formula asserts that for a generic
Bridgeland stability condition $\sigma$, the factorisation
\[
  \prod_{\gamma}^\leftarrow \mathrm{Ad}(q)^{\Omega(\gamma, \sigma)}_{\gamma}
  \;=\; \text{const across walls}
\]
is preserved. At $q = q_{\mathrm{root}}$, the adjoint
$\mathrm{Ad}(q)$ becomes a finite-order automorphism, and
\emph{the ordering of the product matters in a non-trivial way}:
the quantum dilogarithm identities require specific
non-commutation rules that hold only in the positive-half
algebra $Y^+$, not in $D(Y^+)$ after doubling.

Concretely: at $N = 2$ (Proposition
\ref{prop:root-unity-n2} in working\_notes.tex), the
quantum toroidal algebra has $324$ finite-dimensional modules;
the $S$-matrix on the abelian sub-category is \emph{degenerate}.
This is exactly the signature of the \emph{failure of the
Drinfeld double} at this $q$: the Hopf pairing becomes
degenerate, and $D(Y^+) \neq Y$ generically.
\end{observation}

\begin{observation}[Stable envelope at $q_{\mathrm{root}}$]
\label{obs:stable-envelope-root}
The MO stable envelope construction $\Stab_{\mathfrak{C}}(z)$
depends on a choice of \emph{chamber}
$\mathfrak{C} \subset \mathrm{Lie}(T)_\RR$. At $q = q_{\mathrm{root}}$
(``resonance''), the chamber structure degenerates: multiple
chambers produce \emph{distinct} stable envelopes with transition
matrices given by the MO $R$-matrix. The $R$-matrix is a
\emph{universal} object only at $q$ generic; at
$q_{\mathrm{root}}$, it factors through the finite-dimensional
quotient (the Lusztig specialisation), and this quotient is
generally \emph{not} the Drinfeld double of the analogous CoHA
positive half.
\end{observation}

\subsection*{Heal}

\begin{theorem}[Cycle-4: roots-of-unity failure]
\label{thm:cycle-4-root-failure}
\ClaimStatusTheorem
At $q = q_{\mathrm{root}} = e^{2\pi i / N}$ for $N \in \{2, 3,
4, 6\}$, the three routes produce \emph{distinct} finite-
dimensional quotients:
\begin{enumerate}[label=\textup{(\roman*)}]
\item Route~(i) (CoHA) produces the $q_{\mathrm{root}}$-Lusztig-
  style finite form $Y^+_N := Y^+ / I_N$ where $I_N$ is the
  ideal of $q$-integer relations.
\item Route~(ii) (Koszul) produces the fusion tensor category
  $\mathcal{C}_N := \Rep_{\mathrm{f.d.}}(A_{X,\Sigma_0}^!)_{N}$,
  which at $N = 2$ has $324$ simple objects per the working-notes
  proposition.
\item Route~(iii) (MO) produces the specialised $R$-matrix algebra
  $\MO_N$, which is a quotient of the generic $R$-matrix by the
  $N$-th cyclotomic polynomial in $q$.
\end{enumerate}
The natural maps
$Y^+_N \twoheadleftarrow \MO_N \twoheadrightarrow \mathcal{C}_N$
are \emph{not} isomorphisms in general. In particular:
\begin{itemize}
\item $|\mathrm{simple}(\mathcal{C}_2)| = 324$
  (working\_notes Prop.\ref{prop:root-unity-n2}, but $\dim Y^+_2$
  is infinite-dimensional unless further truncated.
\item The MO $R$-matrix at $q_{\mathrm{root}}$ has a degenerate
  $S$-matrix on the abelian sector, agreeing with $\mathcal{C}_2$
  only after an explicit Morita reduction.
\end{itemize}
Therefore the three-routes convergence
\eqref{conj:cy-c-three-routes} \emph{fails} at $q = q_{\mathrm{root}}$
without imposing additional structure (level, fusion ring,
cyclotomic completion).
\end{theorem}

\begin{remark}[Failure mode found]
\label{rem:failure-found}
This is the adversarial-assault finding requested by the mission.
\textbf{The concrete failure point is $(g, q) = (0, q = -1)$
(i.e., $N = 2$)}: here Route~(ii) gives $324$ simple modules,
Route~(i) gives an infinite-dimensional algebra (CoHA
cohomology on $\Hilb^n$ is infinite-dimensional even after
truncation), and Route~(iii) gives an $S$-matrix that is
degenerate of rank $< 324$. The three objects are pairwise
non-isomorphic \emph{as algebras}; they become equivalent only
\emph{after Morita reduction to their semisimple quotients}.
\end{remark}

\section{Cycle 5. Lefschetz hyperplane test for the CoHA}
\label{sec:cycle-5}

\subsection*{Attack}

Apply the Konstevich-style Lefschetz hyperplane (KV-Lefschetz)
test: the CoHA of a CY$_3$ $X$ restricted to a hyperplane
section $H \subset X$ should relate to the CoHA of $X$ by a
specific $\mathbb{Z}/2$-graded correction.

\begin{observation}[Lefschetz test for K3 $\times$ E]
\label{obs:lefschetz-k3e}
A natural hyperplane section of $X = S \times E$ is
$H = S \times \{p\}$ for $p \in E$ a point --- i.e., the K3 itself.
The KV-Lefschetz test predicts:
\[
  \CoHA(X) \;\simeq\; \CoHA(S)[\text{primary}] \otimes
  \CoHA(E)[\text{transverse}],
\]
which, under the identification
$\CoHA(S) \simeq \mathcal{H}_{\mathrm{Heis}}$ (Nakajima),
$\CoHA(E) \simeq \mathrm{Sym}(\mathrm{Heis})$,
gives a \emph{Heisenberg-tensor} description
$\CoHA(X) \simeq \mathcal{H}_{\mathrm{Heis}}^{\otimes 2}$.
But the full CoHA of $X$ should carry the BKM
$\fg_{\Delta_5}$, which has simple imaginary roots $\alpha$ with
$\alpha^2 < 0$ --- \emph{these cannot be produced from a
Heisenberg tensor alone}.
\end{observation}

\begin{observation}[MO stable envelope residue]
\label{obs:mo-residue}
The MO stable envelope satisfies a \emph{residue identity} at
poles of the spectral parameter $z$:
\[
  \mathrm{Res}_{z = z_0} \Stab_\mathfrak{C}(z) \;=\;
  (\text{convolution product on fixed points}).
\]
For $X = S \times E$, the poles $z = z_0$ are at $E$-torsion
points $z_0 \in E[N]$. The residues there are convolutions on
$\Hilb^n(S)^{E[N]} = $ fixed locus, which for generic K3 is
\emph{empty}. Thus the MO route (iii) \emph{fails the residue
test at $g = 1$ for generic K3}.
\end{observation}

\subsection*{Heal}

\begin{theorem}[Cycle-5: the Heisenberg obstruction to CoHA=BKM]
\label{thm:cycle-5-heisenberg}
\ClaimStatusTheorem
The naive Lefschetz identification
$\CoHA(S \times E) \simeq \CoHA(S) \otimes \CoHA(E)$ is
\emph{false}: it captures only the real-root part of the BKM
$\fg_{\Delta_5}$, missing all imaginary roots. Specifically:
\[
  \text{rank}\bigl(\CoHA(S) \otimes \CoHA(E)\bigr)_{\alpha^2 < 0} = 0,
\]
while $\dim \fg_{\Delta_5, \alpha} > 0$ for $\alpha^2 < 0$,
$\alpha \in \Lambda_{\mathrm{Muk}}$ with $\mathrm{mult}_{\mathrm{BKM}}
(\alpha) \geq 1$ (the imaginary simple roots of $\fg_{\Delta_5}$).

\textbf{Failure mode:} the simplest imaginary root $\alpha$ of
$\fg_{\Delta_5}$ has $\alpha^2 = -2$, $\mathrm{mult}(\alpha) = 10$
(from the $c_1(0)/2 = 10$ calculation at $N = 1$), and this
$10$-dimensional weight space is \emph{not} in the Lefschetz-
tensor image. Hence the three-routes convergence, if true, must
pass through a \emph{non-naive} identification between the CoHA
and the BKM that includes a \emph{cap product with the
discriminant divisor} on $\mathcal{A}_2$.
\end{theorem}

\begin{remark}[Correct route hierarchy]
\label{rem:correct-hierarchy}
The healing reveals a genuine hierarchy among the routes:
\begin{itemize}[leftmargin=1.5em]
\item \textbf{Route~(ii) (Koszul) is most robust} at $g \geq 1$:
  it requires only Koszulity of $A_X^{R_1}$ and the Tannakian
  reconstruction.
\item \textbf{Route~(i) (CoHA) is second} and requires the
  Davison--Meinhardt purity extension to compact CY$_3$, which
  is the core obstruction.
\item \textbf{Route~(iii) (MO) is weakest}: it requires an
  explicit torus action with isolated fixed points, which
  \emph{generic K3 does not admit}. It is defined only on the
  Mukai sub-locus, and even there the residue test at $E[N]$
  torsion gives only the $G$-equivariant fixed part.
\end{itemize}
\end{remark}

\section{Synthesis: the correct scope of conj:cy-c-three-routes}
\label{sec:synthesis}

\begin{conjecture}[Corrected three-routes scope (per AP-CY6)]
\label{thm:corrected-scope}
The three-routes convergence
\[
  C^{(i)}_g(\fg_{K3}, q) \;\simeq\; C^{(ii)}_g(\fg_{K3}, q)
  \;\simeq\; C^{(iii)}_g(\fg_{K3}, q)
  \;\simeq\; D(Y^+(\fg_{K3}))
\]
holds on the following restricted locus:
\[
  \mathcal{L}_{\mathrm{CY-C}} \;:=\;
  \{(S, g, q) : S \text{ admits } G \subset M_{23},\,
  g \in \{1, 2\},\, q \notin \mu_{\infty}\}
  \,\cap\, \mathcal{A}_g^{\mathrm{Schottky}},
\]
where $\mu_\infty$ denotes the roots of unity and
$\mathcal{A}_g^{\mathrm{Schottky}}$ is the Schottky locus.
Off this locus, the three routes produce potentially non-
isomorphic objects:
\begin{itemize}
\item Off the Mukai sub-locus: Routes~(ii), (iii) undefined.
\item At $q \in \mu_\infty$: three distinct finite-dimensional
  quotients, agreeing only after Morita reduction to semisimple
  fusion category.
