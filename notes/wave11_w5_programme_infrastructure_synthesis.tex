\section{Programme infrastructure: Lorgat Conjecture~1 as derivable consequence}
\label{sec:wave11-w5-programme-infrastructure-synthesis}

\emph{Conjecture~1 of \cite{Lorgat2020} (\texttt{arXiv:2004.01676})
asserts an eightfold coincidence --- eight Gritsenko--Cl\'ery forms
$\{\Phi_N\}_{N=1}^{8}$ indexed by a triple correspondence with roots
of the Donaldson--Thomas zeta function of $\mathrm{K}3 \times E$ and
with twined elliptic genera of the Conway/Mathieu moonshine tower. Vol
III is the apparatus in which the $N=1$ row is a theorem whose proof
is forced by five independent structural facts, each supplying one
axis of the derivation. Exceptional coincidence becomes constrained
rigidity.}

\subsection{The five programme-level structural facts}
\label{subsec:wave11-w5-five-facts}

\begin{proposition}[Programme infrastructure]
\label{prop:wave11-w5-programme-infrastructure}
\ClaimStatusTheorem\ The Vol III programme establishes the following
five facts, each a theorem at its stated scope:

\begin{enumerate}[label=\textup{(F\arabic*)},leftmargin=*]
\item \emph{Arithmetic rigidity.}
$\dim S_5(\Sp_4(\Z),\nu_{\Delta_5}) = 1$. Four mechanically disjoint
$c = -214$ constructions --- Arthur CAP-projection, Kudla--Millson
Heegner-divisor theta lift, Harvey--Moore heterotic elliptic-genus
denominator, Borcherds additive lift of $\phi_{0,1}$ --- land in a
one-dimensional space and are therefore scalar multiples of $\Delta_5$;
the scalar is fixed by the Fourier normalisation $a(T_0) = 1$
(Wave-8~I3; Gritsenko--Nikulin~\cite{GN1995,GN1998}; Borcherds~\cite{Borcherds1995};
Pitale--Schmidt~\cite{PitaleSchmidt2014}; Ta\"ibi~\cite{Taibi2019}).

\item \emph{Borcherds weight universality.}
$\kBKM(\Phi_N) = c_N(0)/2$ for $N \in \{1,2,3,4,6\}$; equivalently,
the paramodular weight of the Gritsenko--Nikulin Borcherds product
equals one-half the zeroth Fourier coefficient of its input Jacobi
form (Vol III
Theorem~\ref{thm:borcherds-weight-kappa-BKM-universal}; primary source
Borcherds~\cite{Borcherds1995} Theorem~13.3; Gritsenko~\cite{Gritsenko1999}).
Row values $(N, c_N(0)/2) \in \{(1,5),(2,4),(3,3),(4,2),(6,1)\}$.

\item \emph{Humbert geometric base.}
The derived completion $\widehat{\mathbf H}_1 \subset \cA_2$ is the
Shimura-subvariety base along which the $E_n$-tier stratification of
$\cM_{K3 \times E}$ pulls back the Kuga--Satake Hodge structure
(Wave-6 Z5; Humbert~\cite{Humbert1900}; Gritsenko~\cite{Gritsenko1994};
Hulek--Sankaran~\cite{HulekSankaran2002}). The fibre of
$\cM_{\mathrm{ModRigidD}} \to \cA_2$ over $\widehat{\mathbf H}_1$
cuts out precisely the $K3 \times E$ locus carrying the $\Delta_5$
divisor.

\item \emph{$\Sym^2$ universality.} Six independently derived shadows
identify the Kuga--Satake $\Sym^2(\rho_{\Delta_5})$ functor as the
structural operation:
(i) IC-stalk stratification at $[K3 \times E]$ (Wave-7~A4);
(ii) $\ell$-adic cohomology of $\cM_{\mathrm{ModRigidD}}$ (Wave-7~X1,
Kuga--Satake~\cite{KugaSatake1967});
(iii) categorified duality on $(\infty,1)$-CY data (Wave-7~Z4);
(iv) simplicial composition law in the three-tier $\pi_0$-groupoid
(Wave-7~Y5);
(v) mirror squaring on the CY-B side via the Calabi--Yau transform
(Wave-7~C4);
(vi) additive-to-multiplicative Gritsenko/Borcherds lift acting as
$\mathrm{Lift}_{\mathrm{Bo}} = \Sym^2 \circ \mathrm{Lift}_{\mathrm{Gr}}$
on the Jacobi ring (Wave-7~B2).
Each shadow writes $\Sym^2$ for independent reasons; coincidence of
all six forces the operation to be a programme-level invariant, not
a local accident.

\item \emph{Three-tier layering with BNP Hecke action.}
The $E_n$-hierarchy $T_1 \to T_2 \to T_3$ with $T_1 = \ChirAlg_{E_2}$
at the CY-surface leaf, $T_2 = \ChirAlg_{E_1}$ at the CY-threefold
spine, and $T_3 = $ full-BKM Borcherds--Monster image at the rigidity
apex, is threaded by a single Ben-Zvi--Nadler--Preygel generalised
Hecke action of $\Sph_{\Sp_4}$ on
$\IndCoh_{\Nilp}(\LocSys_{\Sp_4}(E))$ (Wave-6~X2; Wave-9~O3;
Ben-Zvi--Nadler--Preygel~\cite{BNP2010};
Arinkin--Gaitsgory~\cite{AG2015}). The three functorial projections
$T_1 \leftarrow T_2 \leftarrow T_3$ are the categorical trace, the
nilpotent-singular-support restriction, and the Satake fibre functor
at $\omega_1$.
\end{enumerate}
\end{proposition}

\subsection{Conjecture~$1'$: the rigidified statement}
\label{subsec:wave11-w5-conjecture-1-prime}

\begin{theorem}[Conjecture~$1'$, $N=1$ row]
\label{thm:wave11-w5-conjecture-1-prime-N1}
\ClaimStatusTheorem\ Assume the scalar normalisation $a(T_0) = 1$ of
Wave-8~I3 Remark~\ref{rem:forcing-scalar} and the Ta\"ibi endoscopic
multiplicity $+1$. Then
\[
   G(K3 \times E) \;=\; \fg_{\Delta_5},
\]
with $\kBKM = 5 = \mathrm{wt}(\Delta_5) = c_1(0)/2$ pinned by (F2),
the CY-D supertrace identification
$\kch(K3 \times E) = \chi(\cO_{K3 \times E}) = 0$ pinned by
$\Phifun$-stratification at $d=3$, and the Borcherds denominator
\[
   \Delta_5 \;=\; q \zeta q' \prod_{(n,\ell,m) > 0}
   (1 - q^n \zeta^\ell {q'}^m)^{c(4nm - \ell^2)},
\]
determined by the twelve non-negative Fourier coefficients
$c(0), c(-1), \dots, c(12)$ of $\phi_{0,1}$ (Eichler--Zagier
input~\cite{EichlerZagier1985}). The four $c = -214$ chains (F1)
coincide with $\Delta_5$ as the \emph{unique} weight-$5$ character-matched
genus-$2$ cusp form.
\end{theorem}

\begin{proof}[Derivation sketch]
(F1) forces each of the four constructions to be $C_i \Delta_5$;
(F2) fixes the weight $5 = c_1(0)/2$ from the Fourier input of
$\phi_{0,1}$; (F3) identifies the geometric base as
$\widehat{\mathbf H}_1$, which the Kudla--Millson lift sends to the
Heegner divisor of $\Delta_5$; (F4) threads $\Sym^2(\rho_{\Delta_5})$
through IC-stalk, cohomology, and Gritsenko lift, forcing the
$c = -214$ normalisation independently on each axis; (F5) identifies
the three $E_n$-tiers as BNP-Hecke slices of a single
$\Sph_{\Sp_4}$-module, fixing the Fourier-Jacobi $q'$-expansion
through the genus-$2$ spin-Satake parameter. Six independent
derivations of the scalar $C_i = 1$ then match at a point; this is
not a free-parameter coincidence but a six-fold rigidity-forced
identity.
\end{proof}

\begin{conjecture}[Conjecture~$1'$, eight-row unification]
\label{conj:wave11-w5-conjecture-1-prime-eight}
\ClaimStatusConjectured\ Replace (F1) by the eight-row dimension
conjecture $\dim M_{k_N}(\Gamma^{(2)}_N,\nu_N^{\mathrm{theta}}) = 1$
for $N \in \{1,\dots,8\}$ (theorem at $N \in \{1,2,3,4,6\}$;
conjecture at $N \in \{5,7,8\}$, pending Cl\'ery--Gritsenko
half-integer-weight Hilbert series; Wave-8~I3
Conjecture~\ref{conj:dim-rows-half-integer}). Under the same normalisation
and multiplicity hypotheses, the eight Borcherds products $\Phi_N$
are in canonical bijection with the eight Donaldson--Thomas zeta
roots of $K3 \times E$ at Hodge weight $(k_N, t_N)$ and with the
eight Conway/Mathieu twined elliptic genera of the relevant
moonshine tower.
\end{conjecture}

\subsection{What the programme supplies versus what remains a
Fourier-coefficient input}
\label{subsec:wave11-w5-scope}

\begin{enumerate}
\item \emph{Supplied by the programme.}
Rigidity of $\Delta_5$ (F1); the weight formula (F2); the geometric
base (F3); $\Sym^2$ as the universal cohomological operation (F4);
the $E_n$-tier Hecke threading (F5). These five facts cover
weight, character, paramodular level, cohomology functor, and
categorical layering.

\item \emph{Supplied by arithmetic input.}
The twelve Fourier coefficients $c(0),\dots,c(12)$ of $\phi_{0,1}$
(Eichler--Zagier) determine the Borcherds product. The programme does
not predict these; it predicts that whatever they are, the resulting
$\Phi_N$ realises the eight-fold structural correspondence.

\item \emph{Load-bearing CY-D output.}
The CY-D dimensional stratification
$\kch = \chi(\cO_X)$ on compact CY$_d$
(\texttt{chapters/examples/cy\_d\_kappa\_stratification.tex}
Theorem~\ref{thm:cy-d-stratification}; Hodge supertrace
identification) is what turns the $\Phifun$-image weight into a
\emph{geometric} invariant of $K3 \times E$ rather than a
\emph{chosen} parameter of the chiral side. Without CY-D,
$\kBKM = 5$ would be a numerical coincidence; with CY-D, it is
forced by $\chi(\cO_{K3 \times E}) = 0$ on the CY side matched to
$c_1(0)/2 = 5$ on the paramodular side through $\Phifun$.

\item \emph{The epistemic status of Conjecture $1'$.}
Theorem at $N=1$ (Theorem~\ref{thm:wave11-w5-conjecture-1-prime-N1});
theorem at $N \in \{2,3,4,6\}$ modulo the Fourier-input matching
for each row (Vol III
Theorem~\ref{thm:borcherds-weight-kappa-BKM-universal});
conjecture at $N \in \{5,7,8\}$ pending
Conjecture~\ref{conj:dim-rows-half-integer}. The eight-fold unified
statement is
Conjecture~\ref{conj:wave11-w5-conjecture-1-prime-eight}.
\end{enumerate}

\subsection{Synthesis statement}
\label{subsec:wave11-w5-synthesis-statement}

The Vol III programme is not a decorative frame around an
automorphic identity; it is the \emph{infrastructure that makes
the identity derivable}. The $\kappa_\bullet$-subscript discipline separates the
four $\kappa_\bullet$ invariants $(\kch,\kcat,\kBKM,\kfib)$ so that the equality
$\kBKM(\Phi_1) = \mathrm{wt}(\Delta_5)$ and the equality
$\kch(K3 \times E) = \chi(\cO_{K3 \times E})$ are independent
theorems rather than a single conflated $\kappa_\bullet$; CY-D
stratification $\kch = \chi(\cO_X)$ on compact CY$_d$ converts a
chiral weight into a Hodge number; the $\Phifun$-functor formalism
packages the assignment CY $\to$ chiral as a morphism-preserving
operation whose output has the right weight, character, and
paramodular level by construction. The one-dimensionality
$\dim S_5 = 1$ then supplies the final rigidity: any object landing
in the correct character space \emph{is} $\Delta_5$. Conjecture~1
becomes a theorem; the scaling of the construction to the eight-row
moonshine tower is reduced to three explicit dimension computations
on Cl\'ery--Gritsenko theta-multiplier rings.
