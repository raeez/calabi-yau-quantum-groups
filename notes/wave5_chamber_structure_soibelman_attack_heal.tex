\section*{Chamber structure (Wave-5 attack-heal expansion, Soibelman channel)}

\begin{remark}[Bridgeland stability space; dimension clarification]
\label{rem:wave5-bridgeland-stab}
The Klebanov--Witten Jacobi algebra $A = J(Q_{\mathrm{con}},
W_{\mathrm{con}})$ has two simple modules $S_0, S_1$ at the vertices;
its numerical Grothendieck group $K_0^{\mathrm{num}}(A\text{-mod})$ is
free of rank two, generated by $[S_0], [S_1]$. Bridgeland's theorem on
stability conditions (Bridgeland \texttt{arXiv:math/0212237} Thm.~1.2;
refined for projective $K3$ surfaces in \texttt{arXiv:1002.4372})
applies to the triangulated category $D^b(A\text{-mod}) \simeq
D^b(\mathrm{Coh}\,\bY)$ through the Van den Bergh tilting: the space
of locally-finite stability conditions
\[
  \mathrm{Stab}(D^b(A\text{-mod})) \;\simeq\;
  \mathrm{Stab}(D^b(\mathrm{Coh}\,\bY))
\]
is a complex manifold whose local dimension equals the rank of
$K_0^{\mathrm{num}} = 2$; the forgetful map
$\mathcal{Z} : \mathrm{Stab} \to \Hom(K_0^{\mathrm{num}}, \bC) = \bC^2$,
$\sigma \mapsto Z_\sigma$, is a local biholomorphism. The $\bC$-action
$z \cdot (Z, \mathcal{P}) = (e^{i\pi z} Z, \mathcal{P}(\mathrm{Re}\, z))$
and the $\mathrm{Aut}(D^b)$-action commute with $\mathcal{Z}$; the
quotient $\mathrm{Stab}/\bC$ inherits the chamber decomposition from
the stratification by walls $W_\alpha = \{\sigma : Z_\sigma(\alpha) \in
\bR_{\leq 0}\}$, $\alpha \in K_0^{\mathrm{num}}$ effective primitive.

The dimensional clarification is load-bearing. ``$\dim_\bC \mathrm{Stab}
= 2$'' here refers to the rank of the numerical $K$-group, \emph{not}
to the Calabi--Yau dimension $d = 3$. Bridgeland's
\texttt{arXiv:1002.4372} is executed on a $K3$ surface ($d = 2$),
whereas the conifold $\bY$ is a $d = 3$ local model; the transport
to our setting proceeds via Nagao--Nakajima's dim-$3$ construction
(\texttt{arXiv:0809.2992}) on the tilted $D^b(A\text{-mod})$ at a
single shift in the heart ladder. Central charges in the reduced slice
are $Z_\sigma(S_i) = m_i e^{i\pi \varphi_i}$, $\varphi_i \in (0, 1]$,
with named chambers:
\begin{itemize}[leftmargin=2em]
\item Non-commutative (Szendrői) $(\varphi_0, \varphi_1) =
  (\tfrac{1}{3}, \tfrac{1}{3})$; both simples simultaneously stable.
  Shift $\mathbf{s} = (0, 0)$.
\item DT large-volume: $\varphi_0 < \varphi_1$ with $\varphi_0 +
  \varphi_1 < 1$; $S_0$ stable, $S_1$ unstable relative to its shift.
  Shift $\mathbf{s} = (0, -1)$ (or $(-1, 0)$ under flopping orientation).
\item PT stable pairs: both shifts $-1$; chamber boundary $\varphi_0,
  \varphi_1 \to 0^+$ while $\varphi_0 / \varphi_1$ bounded. Shift
  $\mathbf{s} = (-1, -1)$.
\item Empty limit: $\varphi_0, \varphi_1 \to 0^+$ with $\mathbf{s}
  \to (+\infty, +\infty)$.
\end{itemize}
\end{remark}

\begin{remark}[DT / PT / NCDT are chambers; GV is a Kähler limit]
\label{rem:wave5-chambers-vs-kahler}
The distinction DT chamber versus GV invariants is a category
separation. A chamber is a connected component of
$\mathrm{Stab} \setminus \bigcup_\alpha W_\alpha$; GV invariants
$n_g^\beta(\bY)$ of Gopakumar--Vafa are integers attached to the
large-radius point of the Kähler moduli $M_K(\bY)$ through the
Maulik--Nekrasov--Okounkov--Pandharipande and Pandharipande--Thomas
correspondences. The three stability chambers NCDT, DT, PT sit over
the DT/PT regime of $M_K$ under the central-charge period map; the GV
generating function is reconstructed from DT (equivalently PT) data
via
\[
  Z_{\mathrm{PT}}(\bY; q, Q) \;=\; \prod_{n \geq 1} (1 - (-q)^n Q)^{n},
\]
with $n_0^\beta = 1$ for $\beta = \beta_0$ the primitive curve class
and all higher-$g$ GV invariants vanishing on the local conifold
(Gopakumar--Vafa \texttt{hep-th/9812127}; Bryan--Morrison
\texttt{arXiv:0812.5007} Thm.~2). GV is a datum extracted at the
large-radius limit, not a chamber of $\mathrm{Stab}$.
\end{remark}

\begin{theorem}[KS wall-crossing as a \v{C}ech $H^1$-cocycle on the chamber cover]
\label{thm:wave5-ks-h1-cocycle}
\ClaimStatusProvedElsewhere\ \emph{(Kontsevich--Soibelman
\texttt{arXiv:0811.2435} §2.5, §6.3; structural re-statement.)}
Fix the open cover $\{U_\alpha\}_{\alpha \in
\pi_0(\mathrm{Stab}_{\mathrm{gen}})}$ of the generic locus of
$\mathrm{Stab}$ by chambers and the refinement $\{U_{\alpha\beta}\}$
by codimension-one walls. Let $G_{\mathrm{KS}}$ denote the pro-nilpotent
group of motivic symplectomorphisms of the quantum torus
$\bT_{Q_{\mathrm{con}}}$ associated to the charge lattice
$K_0^{\mathrm{num}}(A)$. The wall-crossing factors
$\mathbf{U}_{\alpha\beta} \in G_{\mathrm{KS}}$ across
$U_{\alpha\beta}$ constitute a \v{C}ech $1$-cochain with values in
$G_{\mathrm{KS}}$, and the KS wall-crossing formula
(\texttt{arXiv:0811.2435} Thm.~6.3.1) is precisely the $1$-cocycle
condition
\[
  \prod^\curvearrowleft_{W \cap \gamma}\, \mathbf{U}_W \;=\; \mathbf{1}
\]
for every contractible loop $\gamma$ in $\mathrm{Stab}$; equivalently,
on each codimension-two stratum the product of wall operators around
the stratum equals the identity. The cocycle class
$[\mathbf{U}] \in H^1(\mathrm{Stab}; G_{\mathrm{KS}})$ is trivial
precisely because every loop in $\mathrm{Stab}_{\mathrm{gen}}$ bounds
a disk meeting walls transversely; cocycle-triviality is the
wall-crossing formula.
\end{theorem}

\begin{proof}[Derivation]
On a generic chamber $U_\alpha$ the BPS spectrum
$\Omega_\alpha(\gamma)$ is constant; the motivic automorphism
\[
  \mathbf{U}_\alpha
  \;=\; \prod_{\gamma : Z_\sigma(\gamma) \in \bH^-,\,
    \mathrm{arg}\,\nearrow}
      \mathbf{U}_\gamma^{\Omega_\alpha(\gamma)},
  \qquad
  \mathbf{U}_\gamma(x_{\gamma'})
  \;=\; (1 - x_\gamma)^{\langle \gamma, \gamma'\rangle}\, x_{\gamma'},
\]
is the formal symplectomorphism recording BPS states ordered by the
phase of $Z_\sigma$ on the lower half-plane. Across a wall
$U_{\alpha\beta}$ two collinear BPS rays merge and the KS identity
states $\mathbf{U}_\alpha = \mathbf{U}_\beta$; equivalently
$\mathbf{T}_{\alpha\beta} = \mathbf{U}_\alpha \mathbf{U}_\beta^{-1}$
is a word in the quantum-dilogarithm operators $\mathbf{U}_\gamma$
supported on the shared rays. On a codimension-two stratum
$U_{\alpha\beta\gamma}$ the \v{C}ech cocycle condition
$\mathbf{T}_{\alpha\beta} \mathbf{T}_{\beta\gamma}
\mathbf{T}_{\gamma\alpha} = \mathbf{1}$ becomes the pentagon identity
of Theorem~\ref{thm:wave5-pentagon} below. Since
$\mathrm{Stab}_{\mathrm{gen}}$ is simply connected (Bridgeland
\texttt{arXiv:math/0212237} Thm.~7.1, local contractibility), every
loop bounds and the cocycle is a coboundary.
\end{proof}

\begin{theorem}[Pentagon identity for the conifold (Faddeev--Kashaev form)]
\label{thm:wave5-pentagon}
\ClaimStatusProvedElsewhere\ \emph{(Faddeev--Kashaev
\texttt{hep-th/9310070}; specialised to the conifold via KS
\texttt{arXiv:0811.2435} §6.4.)}
Let $x_0, x_1$ be the formal torus generators associated to the charge
classes $[S_0], [S_1] \in K_0^{\mathrm{num}}(A)$, satisfying the
quantum-torus commutation
\[
  x_1 x_0 \;=\; q^{\langle [S_0], [S_1]\rangle_E}\, x_0 x_1
  \;=\; q^{-1} x_0 x_1,
\]
where $\langle \cdot, \cdot\rangle_E$ is the antisymmetrised Euler
form of $(Q_{\mathrm{con}}, W_{\mathrm{con}})$ and
$\langle [S_0], [S_1]\rangle_E = -1$ on the conifold quiver. Let
$\Psi(x) = E_q(x)^{-1} = \prod_{k \geq 0} (1 - q^{k+1/2} x)$ be the
refined Faddeev quantum dilogarithm, so the wall operator
$\mathbf{U}_\gamma$ of a hypermultiplet of charge $\gamma$ acts as
conjugation by $\Psi(x_\gamma)$. Across the codimension-two wall
intersection corresponding to the Seiberg duality
$S_0 \leftrightarrow S_0 \oplus S_1$,
\begin{equation}\label{eq:wave5-pentagon}
  \Psi(x_0)\, \Psi(x_1)
  \;=\; \Psi(x_1)\, \Psi(q^{-1/2} x_0 x_1)\, \Psi(x_0).
\end{equation}
\end{theorem}

\begin{proof}[Derivation]
Two inputs combine.

\emph{(i)} The Faddeev--Kashaev pentagon for the quantum dilogarithm
on two generators $u, v$ with $v u = q u v$ (Faddeev--Kashaev
\texttt{hep-th/9310070} Eq.~(15); refined form Kashaev--Nakanishi
\texttt{arXiv:1104.4630} Eq.~(2.7)):
\[
  \Psi(u)\, \Psi(v) \;=\; \Psi(v)\, \Psi(q^{-1/2} u v)\, \Psi(u),
\]
valid whenever $[u, v]_q = (q - 1) u v$.

\emph{(ii)} Invariance of the BPS spectrum across the wall: the BPS
index $\Omega(\gamma) = 1$ for $\gamma \in \{[S_0], [S_1], [S_0] +
[S_1]\}$ is a KS wall-crossing invariant (MMNS
\texttt{arXiv:1107.5017} Thm.~1; Bryan--Morrison
\texttt{arXiv:0812.5007} Prop.~1).

Setting $u = x_0$, $v = x_1$: the Euler form $\langle [S_0], [S_1]
\rangle_E = -1$ gives $x_1 x_0 = q^{-1} x_0 x_1$, i.e.\ $v u = q^{-1}
u v$; absorbing the sign into $q \mapsto q^{-1}$ reduces to the
Faddeev--Kashaev form. The class $[S_0] + [S_1]$ on the RHS is the
wall-produced bound state (the $\bP^1$-wrapping D-brane), carrying
BPS index $1$ by the MMNS real-root count, and $q^{-1/2} x_0 x_1$ is
the quantum-shifted torus class.

Equivalently by KS: on the triple-wall stratum at the semiclassical
point $Z(S_0) = Z(S_1) = i\epsilon$, $\epsilon \to 0^+$, three
collinear rays $\gamma \in \{[S_0], [S_0]+[S_1], [S_1]\}$ converge,
and the wall-crossing formula
\[
  \mathbf{U}_{[S_0]}\, \mathbf{U}_{[S_1]}
  \;=\; \mathbf{U}_{[S_1]}\, \mathbf{U}_{[S_0]+[S_1]}\, \mathbf{U}_{[S_0]}
\]
translates under the $\Psi(x_\gamma)$-representation to precisely
\eqref{eq:wave5-pentagon}. The $q$-binomial identity
\[
  \sum_{k=0}^n \binom{n}{k}_{\!q} (-1)^k q^{\binom{k}{2}} x^k
  \;=\; \prod_{j=0}^{n-1}(1 - q^j x)
\]
is the finite-dimensional shadow of both sides truncated at the top
weight, discharging the identity order-by-order in $q$.
\end{proof}

\begin{remark}[Pentagon as KS 3-cocycle and as bound-state mechanism]
\label{rem:wave5-pentagon-cocycle}
The pentagon~\eqref{eq:wave5-pentagon} discharges two roles
simultaneously. As a KS cocycle: the triple-wall condition in the
\v{C}ech $H^1$ of Theorem~\ref{thm:wave5-ks-h1-cocycle},
instantiating $\mathbf{T}_{\alpha\beta} \mathbf{T}_{\beta\gamma}
\mathbf{T}_{\gamma\alpha} = \mathbf{1}$ at the conifold's fundamental
codimension-two stratum. As BPS-index invariance: the Denef
multi-centre attractor flow-tree identity, with the middle factor
$\Psi(q^{-1/2} x_0 x_1)$ recording the bound state $(S_0, S_1)$ that
appears on one side of the wall and disappears on the other with BPS
index preserved.
\end{remark}

\begin{theorem}[CoHA chamber-independence via Davison--Meinhardt Sym-decomposition]
\label{thm:wave5-coha-chamber-inv}
\ClaimStatusProvedElsewhere\ \emph{(Davison--Meinhardt
\texttt{arXiv:1601.02479} Thm.~A, Thm.~B; super-extension
Davison--Hennecart--Schlegel-Mejia \texttt{arXiv:2303.12592}.)}
The critical cohomological Hall algebra $\CoHA(Q_{\mathrm{con}},
W_{\mathrm{con}})$ is chamber-independent: for any two generic
$\sigma, \sigma' \in \mathrm{Stab}$ there is a canonical Hopf-algebra
isomorphism $\CoHA_\sigma \xrightarrow{\sim} \CoHA_{\sigma'}$ induced
by the Davison--Meinhardt Sym-decomposition
\[
  \CoHA_\sigma(Q_{\mathrm{con}}, W_{\mathrm{con}})
  \;\cong\;
  \mathrm{Sym}^{\boxtimes_+}_{\mathrm{super}}\!\bigl(
    H(B\bC^\times)_{\mathrm{vir}} \otimes
    \fg_{\mathrm{BPS}}^{\mathrm{con}}
  \bigr),
\]
whose right-hand side depends only on the super-BPS Lie algebra
$\fg_{\mathrm{BPS}}^{\mathrm{con}}$ and the polynomial ring
$H(B\bC^\times)_{\mathrm{vir}} = \bQ[u]$, both chamber-invariants.
\end{theorem}

\begin{proof}[Derivation]
Davison--Meinhardt \texttt{arXiv:1601.02479} Thm.~B (integrality):
for any symmetric quiver with potential satisfying the
Mozgovoy--Reineke optimality condition $\mathrm{HH}^{<0}(J) = 0$, the
motivic BPS integers $\Omega^{\mathrm{mot}}(\gamma) \in
\bZ[\bL^{\pm 1/2}]$ are wall-crossing-invariants. On the conifold
quiver, $W_{\mathrm{con}}$ is Mozgovoy--Reineke optimal
(\texttt{arXiv:0708.1259} Prop.~5.2), and the BPS indices on
$\Delta_+^{\mathrm{re}} \cup \Delta_+^{\mathrm{im}}$ are
chamber-invariant. The Sym-decomposition of
\texttt{arXiv:1601.02479} Thm.~A presents $\CoHA_\sigma$ functorially
in the BPS data, so $(\fg_{\mathrm{BPS}}^{\mathrm{con}})_\sigma =
(\fg_{\mathrm{BPS}}^{\mathrm{con}})_{\sigma'}$ promotes to a
Hopf-algebra isomorphism on $\CoHA$. The super-extension of
Davison--Hennecart--Schlegel-Mejia \texttt{arXiv:2303.12592} confirms
the identification at the $\bZ/2$-graded level, the relevant scope
for the conifold's super root system.
\end{proof}

\begin{theorem}[Chamber-independence of the Hopf pairing (reformulated)]
\label{thm:wave5-chamber-inv-pairing}
\ClaimStatusProvedElsewhere\ \emph{(Joyce--Song
\texttt{arXiv:0810.5645} Thm.~5.11 + KS wall-crossing
\texttt{arXiv:0811.2435} §6.3.)}
The Joyce--Song integration map
$I : \mathrm{SF}^{\mathrm{ind}}(\mathfrak{M}) \to
\widehat{U}(\fg_{\mathrm{BPS}})[[\hbar]]$
is KS-equivariant, and wall-crossing corrections
$\alpha_W = \sum_{\gamma = k\gamma_W} \Omega(\gamma)\,[\gamma]/k^2$
land in orthogonal $K_0$-graded pieces. Hence the Hopf-pairing matrix
entries are literally equal across chambers, not merely projectively
equivariant; combined with Theorem~\ref{thm:wave5-coha-chamber-inv},
the entire Hopf-algebraic structure of $\CoHA(Q_{\mathrm{con}},
W_{\mathrm{con}})$ (product, coproduct, antipode, pairing) is a
chamber-invariant of $(Q_{\mathrm{con}}, W_{\mathrm{con}})$.
\end{theorem}

\begin{remark}[Synthesis: three instantiations of one chamber-invariance]
\label{rem:wave5-three-chamber-invariances}
Theorems~\ref{thm:wave5-ks-h1-cocycle},
\ref{thm:wave5-coha-chamber-inv}, and
\ref{thm:wave5-chamber-inv-pairing} are three faces of a single
invariance. KS cocycle-triviality says the transition data between
chambers, viewed in $G_{\mathrm{KS}}$, lifts to a trivial $H^1$
class. Sym-decomposition says that at the level of the CoHA as a
Hopf algebra the transition is canonical. Integration-map
equivariance is the pairing-level shadow. Each is a theorem at the
stated scope; none is a re-statement of the others. What unifies them
is Davison--Meinhardt integrality: the BPS indices
$\Omega^{\mathrm{mot}}(\gamma)$ are the sole wall-invariant data, and
every chamber-invariance statement above factors through them.
\end{remark}
