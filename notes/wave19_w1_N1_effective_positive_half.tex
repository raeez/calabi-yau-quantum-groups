\section*{The $N=1$ effective positive half: $\Meff(\K\times E)$}
\label{wn:wave19-w1-N1-effective-positive-half}

\subsection*{Attack--heal 1: the moduli stack}

On $X=\K\times E$ the positive cone of the numerical lattice
$N^{1}(X)$ is generated by primitive Mukai vectors
$\gamma=(r,\beta_{h},n)\in H^{0}\oplus H^{2}(\K,\Z)\oplus H^{4}$ with
$\beta_{h}$ primitive on $\K$, $\beta_{h}^{2}=2h-2$, and an elliptic
degree $\ell\in H_{2}(E,\Z)$ attached by the K\"unneth summand
$H_{2}(\K)\otimes H_{0}(E)\oplus H_{0}(\K)\otimes H_{2}(E)$.
\textbf{Flaw.} Standard DT moduli
$\mathcal{M}_{\mathrm{DT}}(X;\gamma)$ produce a vanishing
virtual class: the holomorphic volume
$\Omega_{X}=\sigma_{\K}\wedge dz_{E}$ lifts the K3 symplectic
anomaly to $X$, and
$\mathrm{Ob}\twoheadrightarrow \mathcal{O}$ kills the integration.
\textbf{Heal (Maulik--Pandharipande--Thomas 2010, arXiv:1001.2719).}
On $\K$-primitive classes the reduced obstruction theory
$\mathrm{Ob}^{\mathrm{red}}=\ker(\mathrm{Ob}\twoheadrightarrow\mathcal{O})$
exists and has virtual dimension
$\mathrm{vd}^{\mathrm{red}}(\gamma)=\mathrm{vd}(\gamma)+1$; on $X$
the $E$-translation direction absorbs the extra one, yielding
$\mathrm{vd}^{\mathrm{red}}_{X}(\gamma)=0$ after quotienting by
$\Cstar_{E}$. \textbf{Resolution.} Write
\[
  \Mred(\K\times E;\gamma)
  \;:=\;
  \bigl\{\,I^{\bullet}\in \Db(X)\;\big|\;
         \mathrm{ch}(I^{\bullet})=\gamma,\;\beta_{h}\text{ primitive on }\K\,\bigr\}
\]
with reduced perfect obstruction theory, equipped with
$\virred\in A_{0}^{\Cstar_{E}}(\Mred)$. Kawai--Yoshioka 2000 supply
the seed
$\sum_{n}\chi(\mathrm{Hilb}^{n}\K)q^{n}=q/\eta^{24}(q)$ governing
the integration of $\virred$.

\subsection*{Attack--heal 2: equivariance and positivity}

The continuous symmetry of $X$ is $\Cstar_{E}\simeq\mathrm{Aut}^{0}(E)$;
there is no global torus acting on $\K$. The discrete symplectic
symmetry $\Auts(\K)$ acts through Mukai's eleven maximal subgroups of
$M_{23}$; at $N=1$ it is the trivial subgroup (``identity twine''),
and the equivariance reduces to $\Cstar_{E}$.
\textbf{Flaw.} $\Cstar_{E}$ acts freely on $E$; na\"ive Atiyah--Bott
localisation is vacuous.
\textbf{Heal.} The equivariant cohomology $\Heq_{\Cstar_{E}}(\Mred)$
is the residual weight-tracking device: the $E$-translation
$\tau\in\Cstar_{E}$ carries $\K$-fibred stable pairs
$I^{\bullet}_{\gamma,n}$ to isomorphic pairs, and the induced
$\Cstar_{E}$-weight on fibrewise normal data produces the elliptic
grading variable $t=e^{2\pi i z}$ of the Oberdieck--Pixton expansion.
The quotient stack $\Mred/\Cstar_{E}$ carries the reduced zero-cycle
whose integration is the partition function.
\textbf{Resolution.}
\[
  \boxed{\;
  \Meff(\K\times E)\big|_{N=1}
  \;=\;
  \Mred(\K\times E;\gamma\text{ prim on }\K)\big/\Cstar_{E},
  \quad
  \Yplus(\K\times E)\big|_{N=1}
  \;=\;
  \Heq_{\Cstar_{E}}(\Mred,\phi_{W})
  \;}
\]
where $\phi_{W}$ is the Kontsevich--Soibelman vanishing cycle
attached to the analytic potential on $\K\times E$; the reduction
$\Cstar_{E}$-residual integration matches the universal grammar
$\Yplus(X)=\Heq(\Meff(X),\phi_{W})$ of Wave-18 H5.

\subsection*{Attack--heal 3: the character formula}

\ClaimStatusTheorem\ \emph{Oberdieck--Pixton 2017.}
\begin{equation}
\label{eq:N1-character}
  Z^{\mathrm{red}}_{\mathrm{DT}}(\K\times E;\gamma\text{ prim})(q,t,p)
  \;=\; -\frac{1}{\PhiTen(q,t,p)},
\end{equation}
with $q=e^{2\pi i\tau}$, $t=e^{2\pi i z}$, $p=e^{2\pi i\sigma}$, and
$\PhiTen$ the Igusa cusp form of weight $10$ on
$\mathrm{Sp}_{4}(\Z)$ (arXiv:1706.10100, Invent.\ Math.\ 222, 2020,
Theorem~1).
\textbf{Flaw.} Equation~\eqref{eq:N1-character} is a generating
function; it does not individuate the moduli-geometric source of
each Fourier exponent. \textbf{Heal.} Expand the Borcherds product
\[
  \PhiTen(q,t,p)
  \;=\; q\,t\,p\,\prod_{\substack{(n,\ell,m)>0\\4nm-\ell^{2}\ge -1}}
        \bigl(1-q^{n}t^{\ell}p^{m}\bigr)^{\,c(4nm-\ell^{2})},
\]
with $c(k)$ the Fourier coefficients of $2\phi_{10,1}$
(equivalently $\phi_{0,1}$ of weight $0$ index $1$; Gritsenko--Nikulin
1998 alg-geom/9504006). The exponent $c(4nm-\ell^{2})$ at each
primitive $(n,\ell,m)$ is the root multiplicity of $\gDeltaFive$ at
$\alpha=(n,\ell,m)$ in the Weyl--Kac--Borcherds denominator identity
for $\DeltaFive$. \textbf{Resolution.} The character formula
\eqref{eq:N1-character} identifies the $\Cstar_{E}$-graded dimensions
of $\Yplus(\K\times E)|_{N=1}$ with the root multiplicities of
$\gDeltaFive$ positive-root by positive-root.

\subsection*{Attack--heal 4: the Lie-algebraic bridge}

Let $\CoHAred(\K\times E)$ denote the reduced cohomological Hall
algebra on $\Meff(\K\times E)|_{N=1}$, taken in the Davison--Meinhardt
formalism (arXiv:1601.02479, Invent.\ Math.\ 221, 2020) on the
d-critical atlas of Brav--Bussi--Dupont--Joyce--Szendr\H{o}i
(Publ.~IHES 129, 2019) with algebraic orientation data (Joyce--Upmeier,
Bojko--Kuhn--Liu). Write
$\gBPS(\K\times E):=\mathrm{Prim}\,\CoHAred(\K\times E)$
for the primitive Lie subalgebra supplied by the PBW filtration.
\textbf{Flaw.} Global applicability of the Davison--Meinhardt PBW on
compact $\K\times E$ is conditional (see \S\S3--5 of the
Wave-12 C1 primer): $\K\times E$ is not a quiver moduli.
\textbf{Heal.} Two routes align.
(i) \emph{Character route.} The graded Hilbert series of
$\gBPS(\K\times E)$ on $\K$-primitive classes equals, by the
Davison integrality template, the logarithmic derivative of
$Z^{\mathrm{red}}_{\mathrm{DT}}$; combined with
\eqref{eq:N1-character} and the Borcherds-product expansion,
\[
  \sum_{\gamma}\dim\gBPS(\K\times E)_{\gamma}\,q^{n}t^{\ell}p^{m}
  \;=\;
  -\log \PhiTen(q,t,p)\big|_{\mathrm{Lie-primitive}}.
\]
(ii) \emph{GBKM route.} Gritsenko--Nikulin 1998 plus Lorgat 2020
(arXiv:2007.14218) construct $\gDeltaFive$ as a generalised
Borcherds--Kac--Moody superalgebra of rank $3$ on
$\Lambda^{2,1}_{\mathrm{II}}=\Lambda^{1,1}\oplus[2]$ with real simple
roots $\{\delta_{1},\delta_{2},\delta_{3}\}$ of Gram diagonal $2$,
off-diagonal $-2$; odd imaginary simple roots at lightlike
$(a,a)=0$ with $\tau(a_{0})=9$ and at timelike $(a,a)<0$ with
$m(a)<0$; Weyl vector $\rho=f_{2}-\tfrac{1}{2}f_{3}+f_{-2}$. The
denominator identity
\[
  \DeltaFive(Z)
  \;=\; e^{-2\pi i(\rho,Z)}
       \prod_{\alpha\in\Delta_{+}}
       \bigl(1-e^{-2\pi i(\alpha,Z)}\bigr)^{\mathrm{mult}_{\gDeltaFive}(\alpha)}
\]
reads off
$\mathrm{mult}_{\gDeltaFive}(n,\ell,m)=c(4nm-\ell^{2})$.
\textbf{Resolution.}
\begin{equation}
\label{eq:N1-lie-bridge}
  \dim\gBPS(\K\times E)_{(\beta_{nm},\ell)}
  \;=\; \mathrm{mult}_{\gDeltaFive}(n,\ell,m)
  \;=\; c(4nm-\ell^{2})
\end{equation}
on $\K$-primitive classes. Bracket-level
$\gBPS(\K\times E)\cong\gDeltaFive$ remains conditional (AP-CY34;
Wave-12 C1 \S5); charge-by-charge multiplicities are unconditional
at $N=1$ through \eqref{eq:N1-character} and the Borcherds product.

\subsection*{Attack--heal 5: automorphic closure}

\textbf{Flaw.} A charge-by-charge numerical identity does not yet
present $\PhiTen$ as an automorphic object on the BPS side.
\textbf{Heal (Borcherds singular theta lift).} Borcherds 1998
(alg-geom/9609022, Theorem~13.3) lifts $2\phi_{0,1}$ to a weight-$10$
Siegel cusp form via singular theta integration against the Siegel
theta kernel on $\mathrm{II}_{2,2}$; uniqueness of Igusa's
rank-$3$ cusp form forces
\[
  \PhiTen \;=\; \mathrm{Bor}(2\phi_{0,1}), \qquad \DeltaFive^{2}=\PhiTen.
\]
The constant-term identity $c_{1}(0)=10$ on $\phi_{0,1}$ gives
\[
  \kBKM(\DeltaFive) \;=\; \tfrac{1}{2}\,c_{1}(0) \;=\; 5,
\]
matching Vol III's Borcherds weight theorem
$\kBKM(\Phi_{N})=c_{N}(0)/2$ at $N=1$.
\textbf{Resolution.} The theorem circle closes: reduced DT moduli
$\rightarrow$ equivariant cohomology $\rightarrow$ partition function
$-1/\PhiTen$ $\rightarrow$ Borcherds product $\rightarrow$
$\gDeltaFive$ denominator $\rightarrow$ singular theta lift of
$\phi_{0,1}$ $\rightarrow$ $\PhiTen=\DeltaFive^{2}$.

\subsection*{Theorem-complete statement}

\begin{theorem}[$N=1$ effective positive half]
\label{thm:wave19-N1-effective-positive-half}
\ClaimStatusTheorem\ At $N=1$ on $X=\K\times E$ with identity twine
$(g_{1},h_{1})=(\mathrm{id},\mathrm{id})$:
\begin{enumerate}
\item $\Meff(X)|_{N=1}=\Mred(X;\gamma\text{ prim on }\K)/\Cstar_{E}$,
      with $\Mred$ the Pandharipande--Oberdieck reduced DT moduli
      (MPT 2010) and $\Cstar_{E}$ the $E$-translation torus.
\item $\Yplus(X)|_{N=1}
      =\Heq_{\Cstar_{E}}\bigl(\Mred,\phi_{W}\bigr)$,
      with character
      $\chi_{\Yplus}(q,t,p)=1/\PhiTen(q,t,p)$
      (OP 2017 sign convention
      $Z^{\mathrm{red}}_{\mathrm{DT}}=-1/\PhiTen$).
\item On $\K$-primitive classes,
      $\dim\gBPS(X)_{(\beta_{nm},\ell)}
      =\mathrm{mult}_{\gDeltaFive}(n,\ell,m)
      =c(4nm-\ell^{2})$
      with $c(\cdot)$ the Fourier coefficients of $\phi_{0,1}$
      (Gritsenko--Nikulin 1998; Lorgat 2020).
\item $\PhiTen=\DeltaFive^{2}=\mathrm{Bor}(2\phi_{0,1})$ and
      $\kBKM(\DeltaFive)=c_{1}(0)/2=5$.
\end{enumerate}
\end{theorem}

\ClaimStatusConjectured\ \emph{Bracket-level.} The Lie-algebra
isomorphism $\gBPS(\K\times E)\cong\gDeltaFive$ at the level of
brackets is conditional on algebraic orientation data on $\Db(\K\times E)$
(Bojko--Kuhn--Liu) and on the strong-orientation PBW theorem for
compact 3-CY categories (Kinjo--Park--Safronov 2024,
arXiv:2406.12838).

\paragraph{Subscripted invariants.}
$\kcat(\K\times E)=\chi(\mathcal{O}_{\K\times E})=0$
(K\"unneth: $\chi(\mathcal{O}_{\K})=2$, $\chi(\mathcal{O}_{E})=0$);
$\kch(\K\times E)=\sum_{q}(-1)^{q}h^{0,q}=0$ (compact CY$_{3}$);
$\kBKM(\DeltaFive)=5$; $\kfib(\K)=24$ (Mukai rank). The
$\{2,3,5,24\}$ spectrum arises from four distinct constructions.
