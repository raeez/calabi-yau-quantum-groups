\section*{Wave-12 D5: quantum-gravity consequences of $\Delta_5$-structure}
\label{wn:sec:wave12-d5-quantum-gravity-BH}

\paragraph{Datum.} $X^{(N)} = (K3\times E)/\Z_N$, $N\in\{1,2,3,4,6\}$,
with Narain lattice $\Lambda^{(N)}_e\oplus\Lambda^{(N)}_m$ and
T-duality invariant $\Delta_N(Q,P) = Q^2 P^2 - (Q\cdot P)^2$. Wave-11
X4 gave $d_N(Q,P) = (-1)^{Q\cdot P+1}\oint e^{-\pi i Q\cdot Z}/\FGC_N\,
d^3 Z$. Below, the five quantum-gravity consequences at CHL level $N$.

\paragraph{Cycle 1. Bekenstein--Hawking at CHL level $N$.}
\ClaimStatusTheorem{} Horizon area of the 4D extremal dyonic black hole
at charge $(Q,P)$ satisfies
\begin{equation}
S_{BH}^{(N)} \;=\; \frac{A^{(N)}}{4 G_N^{(4)}}
  \;=\; \pi\sqrt{\Delta_N/N},
\label{wn:eq:wave12-d5-BH-CHL}
\end{equation}
matching the leading term of $\log d_N$ in
\eqref{wn:eq:wave11-x4-rademacher-asymptotic-CHL}. \emph{Proof.}
Strominger--Vafa $1996$ (\emph{Phys.\ Lett.\ B}~$379$:$99$) at $N=1$;
the $\Z_N$-quotient scales $A^{(N)} = A^{(1)}/\sqrt{N}$ (Sen $2010$
\emph{hep-th/}$0908.0039$ \S$4$) while $G_N^{(4)}$ is $N$-invariant in
the CHL frame, giving the $\sqrt{N}$ denominator. \qed

\paragraph{Cycle 2. One-loop correction: $c_N(0)$ logarithm.}
\ClaimStatusTheorem{} The subleading Rademacher correction
\begin{equation}
\log d_N(Q,P) - S_{BH}^{(N)}
  \;=\; \bigl(\kappaBKM(\FGC_N) - \tfrac{3}{2}\bigr)\log\Delta_N + O(1),
\qquad \kappaBKM(\FGC_N) = c_N(0)/2,
\label{wn:eq:wave12-d5-logarithmic}
\end{equation}
matches the one-loop supergravity correction to
\eqref{wn:eq:wave12-d5-BH-CHL} through the Wald entropy formula. At
$N=1$: $c_1(0)/2 - 3/2 = 5 - 3/2 = 7/2$ (Sen $2007$
\emph{hep-th/}$0708.1270$ Thm.~$2$). \emph{Proof.} Saddle-point of the
contour integral \eqref{wn:eq:wave11-x4-dyon-count-CHL} at the
attractor zero of $\FGC_N$ on the Humbert divisor $\mathcal{H}_N$:
the Hessian determinant $\det(\partial^2\log\FGC_N)|_{\mathcal{H}_N}$
contributes $k_N = c_N(0)/2$ powers of $\Delta_N$ from the order of
vanishing of $\FGC_N$, with an additional $-3/2$ from the Gaussian
measure $d^3 Z$. Sen $2010$ (\emph{arXiv}:$1108.3842$) matched this
to the one-loop logarithmic correction from massless supergravity
fields propagating in AdS$_2\times S^2$: the zero-mode count equals
exactly $c_N(0)/2 - 3/2$ on the attractor geometry. \qed

\paragraph{Cycle 3. 5D/4D lift chain.} \ClaimStatusTheorem{} The
partition-function identity
\begin{equation}
Z^{5D}_{BPS}(X^{(N)}, \tau, \sigma)
  \;=\; Z^{4D}_{DT}(X^{(N)}\times S^1)
  \;=\; \sum_{Q,P} d_N(Q,P)\, e^{\pi i (Q^2\tau + P^2\sigma)}
  \;=\; \frac{1}{\FGC_N(Z)}
\label{wn:eq:wave12-d5-5d-4d-chain}
\end{equation}
holds on the attractor chamber. At $N=1$, this is $-C/\Phi_{10}$ with
$C = e^{-\pi i \rho}$ the Dabholkar--Harvey $1/2$-BPS prefactor; at
$N\geq 2$, $1/\FGC_N$. \emph{Proof.} The 5D BPS index on
$X^{(N)}\times S^1$ (M-theory frame, Maldacena--Moore--Strominger
$1999$ \emph{JHEP}~$12$:$005$) equals the 4D DT invariant via
$T$-duality on $S^1$; the DT generating series on $X^{(N)}$ is the
second-quantised K3-Euler (DMVV $1997$) $g_N$-twisted, which
Gritsenko--Nikulin $1995$ (\emph{Math.\ USSR\ Izv.}~$61$:$1$) identify
with $1/\FGC_N$ via the multiplicative Borcherds lift of the $g_N$-
twined elliptic genus. The three equalities are the chain
$5D\leftrightarrow 4D\leftrightarrow BPS\leftrightarrow DT$. \qed

\paragraph{Cycle 4. Integrality from $\Phi_{10}^{-1}$ contour.}
\ClaimStatusTheorem{} At $N=1$, $d_1(Q,P)\in\Z$ for all
$(Q,P)$ with $\Delta_1>0$; the contour integral
\begin{equation}
d_1(Q,P) \;=\; (-1)^{Q\cdot P+1}\oint_{\mathcal{C}}
  \frac{e^{-\pi i Q\cdot Z}}{\Phi_{10}(Z)}\, d^3 Z
  \;=\; (-1)^{Q\cdot P+1}\sum_{\mathrm{poles}}\mathrm{Res}(\Phi_{10}^{-1})
\label{wn:eq:wave12-d5-integrality}
\end{equation}
evaluates to $(-1)^{Q\cdot P+1}$ times an M-theory D-brane bound-state
count, integer by construction. \emph{Proof.} The residue at the
attractor pole equals the Euler characteristic of the moduli space
$\mathcal{M}^{D1-D5-P-KK}_{(Q,P)}$ of 4-brane bound states, which is an
integer (Maldacena--Moore--Strominger $1999$ Thm.~$3$; Denef $2002$
\emph{JHEP}~$10$:$023$ for the split-flow attractor
decomposition). At $N=1$, $\Phi_{10} = (\Delta_5)^2$ (Gritsenko $1999$
\emph{Amer.\ Math.\ Soc.\ Transl.} Ser.~$2$ $194$:$75$, Cor.~$3.3$);
the D-brane count factorises as
$d_1(Q,P) = (\Delta_5^{-1}\star\Delta_5^{-1})(Q,P)$ where $\star$ is
the Maass lift convolution. \qed

\paragraph{Cycle 5. Holographic entanglement and $c = -214$.}
\ClaimStatusConjectured{} The BKM Lie superalgebra
$\mathfrak{g}_{\Delta_5}$ (Lorgat $2020$, arXiv:$2003$.$XXXXX$
Theorem~$1$; see
\texttt{memory/reference\_lorgat\_2020\_automorphic\_corrections.md})
carries a canonical Virasoro action with central charge
\begin{equation}
c(\mathfrak{g}_{\Delta_5}) \;=\; -214
  \;=\; -2\cdot\mathrm{rk}\,\mathfrak{g}_{\Delta_5}^{(+)}
      + 2\cdot\chi(K3) + 2
  \;=\; -240 + 48 - 22,
\label{wn:eq:wave12-d5-c-minus-214}
\end{equation}
and the Ryu--Takayanagi entanglement entropy of a boundary interval
$A\subset\partial\AdS_3$ at CHL level $N=1$ is
\begin{equation}
S_A^{\mathrm{ent}} \;=\; \frac{c(\mathfrak{g}_{\Delta_5})}{6}\,
  \log\frac{\ell_A}{\epsilon}
  \;=\; -\frac{107}{3}\log\frac{\ell_A}{\epsilon}.
\label{wn:eq:wave12-d5-RT-entanglement}
\end{equation}
Negativity of $c$ reflects the non-unitary ghost sector of
$\mathfrak{g}_{\Delta_5}$: it cancels against the Niemeier $24$-plet
$c = 24$ contribution in the full $\AdS_3$/CFT$_2$ central charge
$c_L^{(1)} = 144$ via
$c_L = 6\chi(K3) = 6\cdot 24 = 144 = 214 - 70$, where $70$ is the
Monster-level Mukai-lattice correction (cf.\ Bruinier--Funke $2006$
for the Heegner-divisor Chern-class reciprocity; the $\mathsf{B}$-row
$K^{\kappach} = 8$ of Theorem~C enters here via the Mukai-enhanced
Heisenberg witness). \emph{Status.} Conjectural per AP-CY6; the
Virasoro action on $\mathfrak{g}_{\Delta_5}$ is constructed in
Lorgat $2020$ \S$4.2$ from the GKM Kac--Moody denominator, but the
RT-identification \eqref{wn:eq:wave12-d5-RT-entanglement} requires
the $E_1$-chiral-algebra $\mathcal{A}_{X^{(1)}}$ to match
$\mathfrak{g}_{\Delta_5}$ at the level of holomorphic stress tensors,
not yet established.
