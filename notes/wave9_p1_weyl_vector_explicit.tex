\section*{The Weyl Vector of \texorpdfstring{$\mathfrak g_{\Delta_5}$}{g Delta5} at \texorpdfstring{$N=1$}{N=1}}

All assertions are in the $N=1$ primitive-denominator normalisation.
The only CHL constant used here is $c_1(0)=10$; CHL constants at
higher level and the Gritsenko-Clery diagonal-divisor constants live in
separate normalisations.

The ambient lattice has basis $f_2,f_3,f_{-2}$ and bilinear form
\[
  (f_2,f_{-2})=-1,\qquad (f_3,f_3)=2,
\]
with all other pairings zero.  The type-II hyperbolic sublattice is
\[
  \LII
  =
  \{a f_2+b f_3+c f_{-2}\mid a,c\in 2\mathbb Z,\ b\in\mathbb Z\}
  =
  \mathbb Z\delta_1\oplus\mathbb Z\delta_2\oplus\mathbb Z\delta_3
\]
with real simple roots
\[
  \delta_1=2f_2-f_3,\qquad
  \delta_2=2f_{-2}-f_3,\qquad
  \delta_3=f_3.
\]
The real-root cone is
$\mathbb R_{\geq0}\delta_1+\mathbb R_{\geq0}\delta_2+
\mathbb R_{\geq0}\delta_3$, and the chamber is cut out by
$(x,\delta_i)\leq0$.  Directly from the ambient form,
\[
  G=\bigl((\delta_i,\delta_j)\bigr)_{1\leq i,j\leq 3}
  =
  \begin{pmatrix}
  2&-2&-2\\
  -2&2&-2\\
  -2&-2&2
  \end{pmatrix}
  =
  2(2I-J_3).
\]

\begin{proposition}[Weyl vector]
\label{prop:weyl-vector-delta5-N1}
The real-root Weyl vector of the Gritsenko-Nikulin denominator datum
for $\gDelta$ is
\[
  \rho=\tfrac12(\delta_1+\delta_2+\delta_3)
       =f_2-\tfrac12 f_3+f_{-2}.
\]
It is the unique vector in $\LII\otimes\mathbb Q$ satisfying
$(\rho,\delta_i)=-1$ for $i=1,2,3$, and it has norm
$(\rho,\rho)=-3/2$.
\end{proposition}

\begin{proof}
The Weyl-vector equations are $Gx=-\mathbf 1$.  Since
$G=2(2I-J_3)$, one has
$G(1/2,1/2,1/2)^{\mathsf T}=(-1,-1,-1)^{\mathsf T}$ in the
simple-root basis.  Hence
\[
  \rho=\tfrac12(\delta_1+\delta_2+\delta_3)
       =\tfrac12\bigl((2f_2-f_3)+(2f_{-2}-f_3)+f_3\bigr)
       =f_2-\tfrac12 f_3+f_{-2}.
\]
The determinant of $G$ is $-32$, so the solution is unique.  Finally
\[
  (\rho,\rho)
  =
  \rho^{\mathsf T}G\rho
  =
  -\mathbf 1^{\mathsf T}(1/2,1/2,1/2)^{\mathsf T}
  =
  -3/2.
\]
\end{proof}

\paragraph{Denominator datum.}
For $\gamma=(n,\ell,m)$ put
\[
  \alpha(\gamma)=2nf_2-\ell f_3+2mf_{-2}.
\]
Then $(\alpha(\gamma),\alpha(\gamma))=-2(4nm-\ell^2)$, and the chamber
order on positive triples is the denominator-product order attached to
$\rho$.  The reflection character satisfies
\[
  \nu_{\Delta_5}(s_{\delta_i})=-1,\qquad i=1,2,3,
\]
while the $S_3$ chamber automorphisms permute the three real walls.

With $\phi_{0,1}=\sum f(n,\ell)q^nr^\ell$ and
$\phi_{0,1}=(y^{-1}+10+y)+O(q)$, the monic denominator product is
\[
  \Dfive(Z)=64^{-1}\Delta_5(Z)
  =
  q^{1/2}r^{1/2}s^{1/2}
  \prod_{(n,\ell,m)>0}(1-q^nr^\ell s^m)^{f(nm,\ell)}.
\]
Borcherds' weight formula gives
\[
  \kBKM(\gDelta)=\mathrm{wt}(\Delta_5)
  =\left.c_N(0)/2\right|_{N=1}=10/2=5.
\]
The product exponents are signed multiplicities of the denominator
identity; a parity-resolved presentation requires the even and odd
simple-root fixture in addition to these exponents.

\paragraph{Scalar square.}
The squared Oberdieck-Pixton scalar is a separate normalisation:
\[
  \Phi_{10}^{\mathrm{OP}}=\Dfive^2=4096^{-1}\Delta_5^2,\qquad
  Z_{\mathrm{OP}}=-4096\,\Delta_5^{-2}.
\]
