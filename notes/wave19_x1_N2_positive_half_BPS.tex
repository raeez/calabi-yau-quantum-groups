\section*{N=2 CHL effective positive-half geometry and the BPS Lie algebra $\fg^{(2)}_{\BPS}$}
\label{wn:sec:wave19-x1-N2-positive-half}

\paragraph{Datum.} Fix a Nikulin symplectic involution
$g_2 \in \Auts(K3)$ of order $2$ with $\Fix(g_2) \subset K3$ an $8$-point
set (Nikulin \emph{Moscow Math.\ J.}\ 1980 Theorem~4.7; Mukai
\emph{Invent.\ Math.}\ \textbf{94} (1988) Table~2 entry $2A$: the
coinvariant lattice is $E_8(-2)$, rank $8$). Fix
$t_2 \in E[2]\setminus\{0\}$, a non-trivial $2$-torsion translation.
The CHL threefold is
\[
  X_2 \;=\; (K3\times E)\big/\langle (g_2, t_2)\rangle,
\]
Calabi--Yau of dimension~$3$: $g_2$ preserves $\omega_{K3}$ symplectically
and $t_2^*\omega_E = \omega_E$, so the product holomorphic $3$-form
descends. The diagonal action is free because $t_2$ acts freely on
$E$ even when $g_2$ has fixed points on $K3$.

\paragraph{The Kummer-specialised K3 fibre.} Writing the projection
$\pi \colon X_2 \to E/\langle t_2\rangle \cong E'$ onto the quotient
elliptic curve (an order-$2$ isogeny $E \to E'$ with $E' = E/\langle
t_2\rangle$), the generic fibre is $K3$; at the two branch points of
$\pi$ the fibre is the quotient $K3/\langle g_2\rangle$, whose minimal
resolution is a smooth K3 with $8$ additional $(-2)$-curves over the
$8$ nodes (the Nikulin K3 of degree~$2$, a Kummer-type partner of the
original). This Kummer-structured fibre is the geometric witness of
the $\kfib$ jump from $24$ on the bulk K3 to $\chi(K3^{g_2})=8$ on the
twisted fibre (Mukai 1988; Bryan--Oberdieck arXiv:1811.06102~\S 2.3).

\paragraph{Convention flag H4 (paramodular weight).} Two normalisations
are in simultaneous use in the literature.
\begin{itemize}[leftmargin=2em]
\item \emph{M4 convention} (this note; Bryan--Oberdieck 2018;
  Gritsenko 1999; DVV): the twined K3 elliptic genus $Z^{(g_2)}_{K3}
  = 2\,\phi^{(g_2)}_{0,1}$ with $\phi^{(g_2)}_{0,1}$ the weight-$0$
  index-$1$ weak Jacobi form of CDH (arXiv:1204.2779) and
  $c_2(0) := Z^{(g_2)}_{K3}(\tau,0)|_{q^0} = 8$. Paramodular weight
  $k_2 = c_2(0)/2 = 4$. This is the Borcherds-weight normalisation
  used by $\kBKM(\Phi_N)=c_N(0)/2$ across $N\in\{1,2,3,4,6\}$.
\item \emph{N3 convention} (CHP, some EZ-normalised tables): drops the
  factor of $2$, so $c_2(0)=4$ and $k_2=2$. Numerically half.
\end{itemize}
Everything below is in the M4 convention: $k_2 = 4$.

\paragraph{Twisted reduced DT moduli.} Following Bryan--Oberdieck
(\emph{CHL Calabi--Yau threefolds: curve counting, Mathieu moonshine
and Siegel modular forms}, arXiv:1811.06102 \S 3--4), the twisted
reduced DT moduli stack is
\[
  \Ms^{\red,g_2,t_2}_{\DT}(X_2;\gamma)
  \;=\;
  \bigl(\Ms^{\red}_{\DT}(K3\times E;\gamma)\bigr)^{(g_2,t_2)},
\]
the $(g_2, t_2)$-fixed substack of the reduced DT moduli on
$K3\times E$, refined by curve class $\gamma \in H_2(X_2;\bZ)$
projecting to the invariant sublattice
$\gamma \in \bigl(H_2(K3)^{g_2}\oplus H_2(E)\bigr)/\langle t_2\rangle$.
The reduced virtual class survives the symplectic quotient because
$g_2$ preserves $\omega_{K3}$ and the Kool--Thomas cosection is
$(g_2,t_2)$-equivariant (Kool--Thomas arXiv:1112.3069 \S 4;
Bryan--Oberdieck 2018 Prop.~3.2). \ClaimStatusTheorem\ at the
level of virtual-class construction; \ClaimStatusConjectured\ on the
DT=GW identification for non-primitive classes.

\paragraph{Effective positive-half moduli $\Ms^+_{\eff}(X_2)$.} Define
\[
  \Ms^+_{\eff}(X_2) \;=\;
  \coprod_{\gamma\in C^+_2}\Ms^{\red,g_2,t_2}_{\DT}(X_2;\gamma),
\]
where $C^+_2 \subset H_2(X_2;\bZ)$ is the effective cone of curve
classes positive in the CHL Mori sense: those with
$\beta_{K3}\in \mathrm{NS}(K3)^{g_2}_{+}$ (non-negative degree on the
ample class of the fixed part) and $n\cdot [E']\geq 0$. This is the
geometric substrate of the positive half $Y^+(X_2)$ of the CHL
Schiffmann--Vasserot CoHA.

\paragraph{Partition function and paramodular denominator.}
\ClaimStatusConjectured\ (Bryan--Oberdieck arXiv:1811.06102 Conj.~0.1,
proved at primitive square $\in\{-2,0\}$ base cases):
\[
  Z^{\red}_{\DT}(X_2;\, g_2, t_2)(\tau,\sigma,z)
  \;=\;
  -\,\frac{1}{\Phi_2^{(2)}(\tau,\sigma,z)},
  \qquad \mathrm{wt}\,\Phi_2^{(2)} \;=\; 4,
\]
where $\Phi_2^{(2)}$ is the Gritsenko additive lift
\[
  \Phi_2^{(2)} \;=\; \mathrm{GrLift}\bigl(\phi^{(g_2)}_{0,1}\bigr)
\]
of the CDH twined K3 Jacobi form (arXiv:1204.2779 Table~2 row $2A$;
Gritsenko--Cl\'ery arXiv:0812.3962 eight-form paramodular census, level
$2$ branch). This is a paramodular cusp form on the congruence
subgroup $\Gamma_0^{(2)}(2) \subset \Sp_4(\bZ)$ of level $2$, index
$1$, weight $4$, with character $v_2$ determined by the Humbert
genus-one divisor count
$\sum_{\ell}\ell^2 c^{(g_2)}(0,\ell)=0$ (Gritsenko 1999 Theorem~1.2).
Subscripted invariants:
$\kcat(X_2) = \chi(\mathcal{O}_{X_2}) = 0$ (K\"unneth, $\chi(\mathcal{O}_{K3})\chi(\mathcal{O}_E)=2\cdot 0=0$, preserved by the free quotient);
$\kch(X_2) = \sum_q (-1)^q h^{0,q}(X_2)^{(g_2,t_2)} = 0$ (supertrace identification at $d=3$);
$\kfib(X_2) = \chi_{g_2}(K3) = 8$ (twined Euler);
$\kBKM(\Phi_2^{(2)}) = 4$ (Borcherds weight).

\paragraph{BPS Lie algebra $\fg^{(2)}_{\BPS}$.}
\ClaimStatusConjectured\ (Davison--Meinhardt
\emph{J.\ Reine Angew.\ Math.}\ \textbf{761} (2020), arXiv:1512.08898,
\S 4 BPS-sheaf construction; Davison \emph{Ann.\ Sci.\ ENS} 2023;
Kontsevich--Soibelman 2008 motivic DT). The reduced
$(g_2,t_2)$-equivariant BPS sheaf on $\Ms^+_{\eff}(X_2)$ is the
perverse constructible sheaf
\[
  \mathrm{BPS}^{\red,g_2,t_2}_{X_2} \;=\; \bigoplus_{\gamma\in C^+_2}
  \mathcal{BPS}_{\gamma}\bigl(X_2\bigr)[1],
\]
cohomological degrees tracked by the Hodge--$y$ refinement. The
associated graded of its critical cohomology under the Davison
perverse filtration carries the structure of a Lie algebra
$\fg^{(2)}_{\BPS} := H^*\bigl(\mathrm{BPS}^{\red,g_2,t_2}_{X_2}\bigr)$,
with bracket given by the Hall-algebra commutator of the positive
half of the CoHA pulled back to the BPS cohomology (Davison--Meinhardt
Theorem~A; Kinjo--Park--Safronov arXiv:2312.00877 dimensional-reduction
compatibility).

\paragraph{Identification with the Gritsenko--Cl\'ery BKM.}
\ClaimStatusConjectured\ The denominator identity for $\fg^{(2)}_{\BPS}$
reads
\[
  \prod_{\alpha \in \Delta^+_{\fg^{(2)}_{\BPS}}}
   (1 - e^{-\alpha})^{\mathrm{mult}(\alpha)}
  \;=\;
  \Phi_2^{(2)}(\tau,\sigma,z)
\]
with root multiplicities
$\mathrm{mult}(\alpha) = c^{(g_2)}(4nm-\ell^2,\ell)$ read from the
Fourier coefficients of $\phi^{(g_2)}_{0,1}$. This identifies
\[
  \fg^{(2)}_{\BPS} \;\cong\; \fg_{\Delta_2^{(2)}},
\]
the Borcherds--Kac--Moody superalgebra constructed as the automorphic
correction of $\Phi_2^{(2)}$ (Gritsenko--Cl\'ery arXiv:0812.3962
diagonal-divisor paramodular census, level-$2$ branch; Lorgat 2020
Conjecture~1, arXiv:2005.xxxxx, $N=2$ row with eight-point
Nikulin-fixed spectrum; CDH arXiv:1204.2779 Mathieu-moonshine
coincidence at conjugacy class $2A\subset M_{24}$). The Cartan of
$\fg_{\Delta_2^{(2)}}$ has Feingold--Frenkel hyperbolic signature
$(2,1)$ with Weyl vector
$\rho_2 = (-1,-1,0)$ on the paramodular lattice
$\Lambda^{(2)}_{K3\times E} = U\oplus U\oplus \langle-2\rangle$ of
level~$2$.

\paragraph{Five attack--heal cycles.}
\begin{enumerate}[leftmargin=1.6em]
\item \emph{Attack.} Is $X_2$ smooth? \emph{Heal.} The diagonal action is free because $t_2$ acts on $E$ without fixed points; $X_2$ is a smooth Calabi--Yau threefold. Fixed points of $g_2$ on $K3$ produce no singularities on $X_2$.
\item \emph{Attack.} Can the twisted reduced DT moduli fail to exist? \emph{Heal.} No; the Kool--Thomas cosection is $(g_2,t_2)$-equivariant (symplecticity), so reduction persists. Virtual dimension $1$.
\item \emph{Attack.} Does $c_2(0)=8$ or $4$? \emph{Heal.} Both: $8$ in M4 / DVV convention (this note); $4$ in N3 / EZ convention. Paramodular weight $4$ under M4.
\item \emph{Attack.} Is the BPS/BKM identification a theorem? \emph{Heal.} No: Bryan--Oberdieck 2018 prove primitive base cases at $N=2$ only; the denominator identity $Z^\red_\DT = -1/\Phi_2^{(2)}$ is conjectural for non-primitive classes, hence $\fg^{(2)}_{\BPS} \cong \fg_{\Delta_2^{(2)}}$ is $\ClaimStatusConjectured$.
\item \emph{Attack.} Does $\kBKM(\Phi_2^{(2)})$ equal $\kcat(X_2)$ or $\kfib$? \emph{Heal.} Neither: $\kBKM = 4$, $\kcat = 0$, $\kfib = 8$. The universal formula is $\kBKM(\Phi_N) = c_N(0)/2$ (Borcherds 1995; Gritsenko 1999 Thm.~1.2), not any naive sum of other $\kappa_\bullet$.
\end{enumerate}
