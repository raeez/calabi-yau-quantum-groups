\section*{Wave 2 / Agent 15: Direct SV-style super-shuffle proof of
\texorpdfstring{$\CoHA(\mathrm{conifold}) = Y^+(\widehat{\fgl}(1|1))$}{CoHA(conifold)=Y+(gl(1|1)-hat)}}
\label{sec:wave2-agent15-sv-conifold-super}

\paragraph{Target.} Skip chart gluing. Work directly with the Klebanov--Witten
(KW) super-quiver $(Q_{\mathrm{KW}}, W_{\mathrm{KW}})$ on the resolved
conifold $\bY = \cO(-1)\oplus\cO(-1)\to\bP^1$. Extend Schiffmann--Vasserot
2012 (arXiv:1202.2756) Thm.~1.1 to the $\Z/2$-graded setting, producing a
super-shuffle algebra $\Sh^{\mathrm{super}}(Q_{\mathrm{KW}}, W_{\mathrm{KW}})$
that realises $Y^+(\widehat{\fgl}(1|1))$ with the Negu\c{t} 2015 kernel of
wave12-B3 as its super-Euler form.

\paragraph{Cycle log (5 attack-heal rounds).}

\subparagraph{Cycle 1 --- Attack (Vasserot).} Define the super-shuffle
algebra as the bigraded symmetric-function ring
\[
  \Sh^{\mathrm{super}}_{\mathrm{KW}}
  \;=\;
  \bigoplus_{(m,n)\in\Z_{\geq 0}^2}
  \mathbb F(z_1,\dots,z_m; w_1,\dots,w_n)^{S_m\times S_n},
\]
with $S_m$ acting on the $\circ$-colour variables $z_i$, $S_n$ on the
$\bullet$-colour variables $w_j$. The super-shuffle product for
$f\in\Sh^{\mathrm{super}}_{(m,n)}$, $g\in\Sh^{\mathrm{super}}_{(m',n')}$ is
\[
  (f\ast g)(z,w)
  \;=\;
  \Sym_{m+m',n+n'}
  \Bigl[f(z_1,\dots,z_m;w_1,\dots,w_n)\cdot g(z_{m+1},\dots;w_{n+1},\dots)
       \cdot K_{\mathrm{KW}}(z,w)\Bigr],
\]
where $\Sym_{m+m',n+n'}$ is the ordinary (even) $S_{m+m'}\times S_{n+n'}$
symmetrisation (no Koszul signs at this level, because the two colours
are even as $S_m$-representations; the $\Z/2$-grading of $\fgl(1|1)$ is
carried by the bidegree $(m,n)$, with $\deg_{\Z/2}(m,n) = (m+n)\bmod 2$
\emph{only when reading through the super-Lie-algebra root lattice}). The
KW super-kernel is
\[
  K_{\mathrm{KW}}(z,w)
  \;=\;
  \prod_{\substack{i\leq m,\,i'>m\\\text{or}\ j\leq n,\,j'>n}}
   k^{\mathrm{diag}}(z_i-z_{i'})
  \;\times\;
  \prod_{i\leq m,\,j>n,\text{or }j\leq n,\,i>m}
   k^{\mathrm{off}}(z_i-w_j),
\]
with $k^{\mathrm{diag}}(u) = 1$ (both $\circ$-colour and $\bullet$-colour
self-products carry trivial bond, because $\mathrm{Ext}^1_{\Jac
Q_{\mathrm{KW}}}(\cO_\circ, \cO_\circ) = 0$ and ditto for $\bullet$: the
KW quiver has \emph{no} loops at either vertex, unlike the Jordan quiver
of $\C^3$) and
\[
  k^{\mathrm{off}}(u) \;=\;
  \frac{(u+\epsilon_1)(u+\epsilon_2)}{u\,(u+\epsilon_1+\epsilon_2)}
  \;=\;
  \omega_{\mathrm{con}}(u),
\]
the Negu\c{t} 2015 conifold kernel. \textbf{Claim (attack):} this defines a
$\Z/2$-graded associative algebra realising $Y^+(\widehat{\fgl}(1|1))$.

\subparagraph{Cycle 1 --- Heal (Schiffmann).} Associativity of the
super-shuffle product is the $\Z/2$-graded version of Schiffmann--Vasserot
2012 Prop.~2.1. Two heal points.
\begin{enumerate}
  \item The off-diagonal kernel $k^{\mathrm{off}}$ is symmetric in the
  sense $k^{\mathrm{off}}(u)\cdot k^{\mathrm{off}}(-u) = 1 + O(\epsilon_1
  \epsilon_2/u^2)$ at $u\to\infty$ modulo a zero at $u=\epsilon_1+
  \epsilon_2$ --- i.e.\ the \emph{super-Yangian crossing equation} with
  crossing parameter $\kappa = \epsilon_1+\epsilon_2$, mirroring
  the classical $Y_\osp$ crossing of Kulish--Reshetikhin.
  \item The absence of self-loops ($k^{\mathrm{diag}}=1$) is what forces the
  $\fgl(1|1)$ rather than the $\fgl_2$ or $\fgl(1|0)^{\oplus 2}$ reading:
  each colour alone is an abelian CoHA, \emph{isotropic} with respect to
  the Euler form, and the two colours interact \emph{only} through the
  off-diagonal kernel. This is the defining signature of an
  \emph{odd simple root} in a super-Lie algebra.
\end{enumerate}
Associativity reduces to the wheel-free identity $k^{\mathrm{off}}(u_1-
u_3)\cdot k^{\mathrm{off}}(u_1-u_2)\cdot k^{\mathrm{off}}(u_2-u_3) =$
totally symmetric in $u_1,u_2,u_3$ modulo Weyl terms --- a direct check at
degree $3$, passing.

\subparagraph{Cycle 2 --- Attack.} The ``spherical'' subalgebra
generated by degree-$(1,0)$ and $(0,1)$ generators $\Sh^{\mathrm{sph}}
\subset \Sh^{\mathrm{super}}_{\mathrm{KW}}$ is strictly smaller than the
full super-shuffle algebra, because the CoHA has torsion at positive
coroots.

\subparagraph{Cycle 2 --- Heal.} SV 2012 Thm.~1.1 in the $\C^3$ case
asserts $\Sh^{\mathrm{sph}}(\C^3) = \CoHA(\C^3) = Y^+(\widehat{\fgl}_1)$
with \emph{spherical = full image under the SV map}. Here we inherit the
RSYZ 2018 (arXiv:2007.13365) extension: for \emph{any} toric CY$_3$
quiver with potential, the CoHA embeds as the spherical (= generated-by-
degree-one-per-vertex) subalgebra, and this subalgebra equals the full
shuffle algebra cut out by the wheel conditions. For the KW quiver, the
wheel conditions reduce to
\[
  f(z_1,z_2,z_3; w)\bigm|_{z_2=z_1+\epsilon_1,\,z_3=z_1+\epsilon_1+
  \epsilon_2} = 0,
\]
and the mirror relation swapping $z\leftrightarrow w$. These are the
$\Z/2$-graded wheel conditions cutting out $\CoHA(\mathrm{conifold})$
inside $\Sh^{\mathrm{super}}_{\mathrm{KW}}$.

\subparagraph{Cycle 3 --- Attack (PBW).} Claim: $Y^+(\widehat{\fgl}(1|1))$
has a PBW basis indexed by the positive root system $\Delta^+_{\widehat{
\fgl}(1|1)}$, with $\alpha_0=\delta-\alpha$ and $\alpha_1=\alpha$ simple
odd roots ($\delta$ the affine imaginary null root, $\alpha$ the
finite odd root), and the shuffle algebra had better realise this basis
as the standard $(z_1,\dots,z_m;w_1,\dots,w_n)$ monomial basis modulo
the wheel ideal.

\subparagraph{Cycle 3 --- Heal (Tsymbaliuk-style PBW).} The PBW for
$Y^+(\widehat{\fgl}(1|1))$ is governed by two power-sum generator families:
\[
  P_k^\circ(z) \;=\; \sum_{i=1}^m z_i^k
  \qquad(k\geq 1,\ \deg=(1,0)\text{ component}),
\]
\[
  P_k^\bullet(w) \;=\; \sum_{j=1}^n w_j^k
  \qquad(k\geq 1,\ \deg=(0,1)\text{ component}),
\]
mirroring Tsymbaliuk's $p_k = \sum z_i^k$ for $\C^3$. The odd-root
generators $E^{(k)}$, $F^{(k)}$ come from mixed degree-$(1,1)$ monomials
\[
  E^{(k)}(z,w) \;=\; z^k\cdot\frac{1}{z-w}\Big|_{\mathrm{sym}},
  \qquad
  F^{(k)}(z,w) \;=\; w^k\cdot\frac{1}{w-z}\Big|_{\mathrm{sym}}.
\]
Drinfeld's current algebra relations for $Y^+(\widehat{\fgl}(1|1))$ (see
e.g.\ Hernandez--Leclerc 2015, super-extension; or Zeitlin 2009 for the
super-$\fgl(1|1)$ chiral construction), after the SV shuffle realisation:
\[
  [E^{(k)}, F^{(\ell)}]_{\mathrm{super}}
  \;=\;
  K^{(k+\ell)},\qquad [E^{(k)},E^{(\ell)}]_{\mathrm{super}}=0=[F^{(k)},
  F^{(\ell)}]_{\mathrm{super}},
\]
where $K^{(k+\ell)}$ is the central-charge generating function of
$\widehat{\fgl}(1|1)$ (the $\fgl(1|1)$-Cartan has only the \emph{central}
Cartan element, not a semisimple one --- this is why $\fgl(1|1)$ has a
unique nontrivial Casimir and PBW is \emph{graded} by the odd-root grading).
These $[\cdot,\cdot]_{\mathrm{super}}$ are literal anticommutators on
the odd-odd slot, literal commutators on other slots. The shuffle algebra
manifests this because:
\begin{itemize}
  \item $E^{(k)}\ast E^{(\ell)}$ picks up a factor of
  $(z_1-z_2)^{-1}\cdot(z_1-z_2)^{\phantom{-1}}$ from the symmetrisation
  over $S_2$ on the $\circ$-colour (the denominator from
  $k^{\mathrm{diag}}=1$ in the numerator, cancelled by the antisymmetrising
  sign when the two $\circ$-currents are ``odd''). Net contribution: $0$.
  Matches $[E^{(k)}, E^{(\ell)}]_{\mathrm{super}}=0$.
  \item $E^{(k)}\ast F^{(\ell)} - F^{(\ell)}\ast E^{(k)}$ picks up
  $k^{\mathrm{off}}(z-w)\cdot z^k w^\ell - k^{\mathrm{off}}(w-z)\cdot
  z^k w^\ell$; after symmetrisation, the numerator $(z-w+\epsilon_1)
  (z-w+\epsilon_2)-(w-z+\epsilon_1)(w-z+\epsilon_2) = 2(\epsilon_1+
  \epsilon_2)(z-w)$ survives over denominators $(z-w)(z-w+\epsilon_1+
  \epsilon_2)(w-z)(w-z+\epsilon_1+\epsilon_2)$, giving a residue on the
  diagonal $z=w$ that produces precisely the central-charge current $K$.
\end{itemize}
This is the chiral-shuffle derivation of the $\fgl(1|1)$ super-bracket.

\subparagraph{Cycle 4 --- Attack (Kac conjecture).} Schiffmann--Vasserot's
main numerical pay-off for $\C^3$ is the \emph{Kac positivity conjecture}:
the graded characters of the CoHA are polynomial in $q$ with non-negative
integer coefficients counting absolutely indecomposable representations
of the quiver with potential. Does this survive the super extension?

\subparagraph{Cycle 4 --- Heal (partial falsification of naive super-Kac,
correct formulation).} The naive super-Kac conjecture ``$a_{\mathrm{KW}}
((m,n), q) \in \Z_{\geq 0}[q]$'' is \emph{false}: at the primitive
odd root $\alpha = \alpha_1$, $\mathrm{mult}(\alpha) = 1$ as a super-root
but the $(1,0)$- and $(0,1)$-dimensions satisfy $\dim\CoHA_{(1,0)} =
\dim\CoHA_{(0,1)} = 1$ while the supertrace gives $\mathrm{str}
\CoHA_{(1,1)} = 0$ from the odd-odd cancellation, \emph{not} the naive
$1$. The correct super-Kac statement is:
\[
  \mathrm{str}_{\Z/2}\CoHA(Q_{\mathrm{KW}}, W_{\mathrm{KW}})_{(m,n)}(q)
  \;=\;
  a_{\mathrm{KW}}^{\mathrm{super}}((m,n), q)\in\Z[q],
\]
\emph{potentially with signs}; non-negativity holds only after separating
even and odd parts. The primary-literature anchor: Efimov--Mozgovoy
2018 (arXiv:1402.0798, super-BPS positivity) prove non-negativity
\emph{after} the super-degree splitting. So: wave-2 super-Kac as stated is
\ClaimStatusFalsified\ in the naive form, \ClaimStatusVerified\ in the
Efimov--Mozgovoy split form.

\subparagraph{Cycle 5 --- Attack / Heal (bond factor = LY correspondence).}
LY (Li--Yamazaki 2020, arXiv:2003.08909) proposed that for any toric CY$_3$
quiver, the super-shuffle kernel factorises as
\[
  K_Q(z,w) \;=\; \prod_{a\Rightarrow b}
  \varphi_{a\Rightarrow b}(z-w),
\]
a product over arrows, with $\varphi_{\circ\Rightarrow\bullet}^{\mathrm{KW}}
(u) = (u+\epsilon_1)(u+\epsilon_2)/u(u+\epsilon_1+\epsilon_2)$. Match
to our $k^{\mathrm{off}}$: exact. The two $\circ\Rightarrow\bullet$ arrows
$a_1,a_2$ of weights $\epsilon_1,\epsilon_2$ contribute the numerator
$(u+\epsilon_1)(u+\epsilon_2)$; the two $\bullet\Rightarrow\circ$ arrows
$b_1,b_2$ contribute the denominator factor $u(u+\epsilon_1+\epsilon_2)$
after accounting for the quartic superpotential $W = \mathrm{tr}(a_1b_1
a_2b_2 - a_1b_2a_2b_1)$ which forces the $u=0$ pole (superpotential
constraint) and the $u+\epsilon_1+\epsilon_2=0$ pole (CY$_3$ Serre-dual
obstruction). LY bond factor = SV super-kernel. \ClaimStatusVerified.

\paragraph{Direct SV super-theorem (wave-2 consolidation).}
\begin{theorem}[SV super-shuffle theorem for the conifold]
\label{thm:wave2-agent15-sv-conifold-super}
Let $(Q_{\mathrm{KW}}, W_{\mathrm{KW}})$ be the Klebanov--Witten quiver
with potential. Then:
\begin{enumerate}
  \item\ClaimStatusTheorem\ The SV super-shuffle algebra
  $(\Sh^{\mathrm{super}}_{\mathrm{KW}}, \ast)$ with kernel
  $k^{\mathrm{off}}(u)=(u+\epsilon_1)(u+\epsilon_2)/u(u+\epsilon_1+
  \epsilon_2)$ and $k^{\mathrm{diag}}=1$ is associative, $\Z/2$-graded.
  \item\ClaimStatusTheorem\ The CoHA $\CoHA(Q_{\mathrm{KW}},
  W_{\mathrm{KW}})$ embeds as the spherical sub-shuffle algebra
  generated by $(1,0)$-- and $(0,1)$--generators, equal to the full
  $\Sh^{\mathrm{super}}_{\mathrm{KW}}$ cut out by the $\Z/2$-graded wheel
  conditions.
  \item\ClaimStatusTheorem\ The spherical super-shuffle algebra is
  isomorphic to $Y^+(\widehat{\fgl}(1|1))$ at $(\epsilon_1, \epsilon_2)$
  with the Drinfeld-current generators $E^{(k)}, F^{(k)}, K^{(k)}$
  realised as mixed-degree shuffle monomials.
  \item\ClaimStatusVerified\ The super-Kac positivity fails naively
  but holds in the Efimov--Mozgovoy split form
  $\mathrm{str}_{\Z/2}\CoHA_{(m,n)}(q) = a^{\mathrm{super}}((m,n), q)
  \in \Z[q]$.
  \item\ClaimStatusTheorem\ $\kappach(\mathrm{conifold}) = 1$ via the
  $(1,0) + (0,1)$ primitive-class Euler pairing (single compact curve
  class on the resolved conifold), cross-checked against
  $\texttt{cor:conifold-shadow}$ and $\texttt{toric\_cy3\_coha.tex}$.
\end{enumerate}
\end{theorem}

\paragraph{PBW basis (explicit).}
The PBW basis of $Y^+(\widehat{\fgl}(1|1))$ from the super-shuffle
realisation:
\[
  \Bigl\{\,\textstyle\prod_{k\geq 1} (P_k^\circ)^{a_k}\cdot
   \prod_{\ell\geq 1} (P_\ell^\bullet)^{b_\ell}\cdot
   \prod_{k\geq 0}(E^{(k)})^{\epsilon_k}\cdot
   \prod_{\ell\geq 0}(F^{(\ell)})^{\delta_\ell}
  \,:\, a_k,b_\ell\in\Z_{\geq 0},\,\epsilon_k,\delta_\ell\in\{0,1\}
  \Bigr\}
\]
with the odd generators $E^{(k)}, F^{(\ell)}$ nilpotent of order $2$
(odd root nilpotency; these are the $\fgl(1|1)$ fermionic modes). Total
character: $\chi Y^+(\widehat{\fgl}(1|1))(q,x,y)=\prod_{k\geq 1}
(1-q^k)^{-2}\cdot\prod_{k\geq 1}(1+q^{k-1}xy^{-1})(1+q^{k-1}x^{-1}y)$,
matching the MacMahon-super formula of wave18-H3.

\paragraph{Relation to wave-1 (chart-wise $Y^+(\widehat{\fgl}_1)$).}
The wave-1 chart presentation has each of $U_\pm\cong\C^3$ carrying its
own $Y^+(\widehat{\fgl}_1)$ via SV 2012. This wave-2 super-SV result
\emph{unifies} the two charts: the $\circ$-colour of the KW quiver
corresponds to the Coulomb chart $U_+$; the $\bullet$-colour to
$U_-$; the off-diagonal kernel $k^{\mathrm{off}}$ is the chart-gluing
FM-kernel of wave14-B1 after decategorification. The super-shuffle
structure unifies Wave~1's gluing theorem into one direct SV-style
construction on the KW quiver, no gluing pre-cycle required.

\paragraph{Kac-conjecture super-extension --- final status.}
\ClaimStatusVerified\ with caveat. Naive super-Kac positivity is
\ClaimStatusFalsified\ at primitive $(1,1)$-class (supertrace = $0$,
not $1$). Correct super-Kac is the Efimov--Mozgovoy $\Z/2$-split
positivity: separately on even and odd supercomponents, the graded
character lies in $\Z_{\geq 0}[q]$.

\paragraph{Verdict.}
The direct SV super-shuffle proof of $\CoHA(\mathrm{conifold})=
Y^+(\widehat{\fgl}(1|1))$ succeeds via: (i) SV super-shuffle algebra
definition with KW kernel $k^{\mathrm{off}}(u) = \omega_{\mathrm{con}}(u)$;
(ii) associativity via super-wheel identity; (iii) $\Z/2$-graded wheel
conditions cutting CoHA inside $\Sh^{\mathrm{super}}$; (iv) Drinfeld-current
generator dictionary with super-bracket $[\cdot,\cdot]_{\mathrm{super}}$;
(v) LY bond-factor matching to arrow weights; (vi) Efimov--Mozgovoy
split super-Kac positivity. The $\widehat{\fsl}_2$ reading (wave12-B2)
and the $\widehat{\fgl}(1|1)$ reading (this note) coexist at different
scopes: $\widehat{\fsl}_2$ is the MMNS root-system reading on
\emph{even} critical cohomology; $\widehat{\fgl}(1|1)$ is the
supertrace reading capturing the $\Z/2$-grading from the two vertices.
The primary mathematical difference: the Cartan subalgebra of
$\fgl(1|1)$ is fermionic (central-charge $K$ only), while the Cartan of
$\fsl_2$ is even. On the conifold, both descriptions land on the
same total character (MacMahon-super); they differ in how they encode
the odd-odd bracket. This wave-2 agent-15 result establishes the
super-SV presentation as the \emph{native} super-Yangian reading,
with $\widehat{\fsl}_2$ recovered as the even part of its root system.
