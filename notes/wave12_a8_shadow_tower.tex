\section{The shadow coproduct and its scalar residue}

Let $(\mathcal{A}, m_{2}, m_{3}, \ldots)$ denote a chirally Koszul
vertex algebra of finite type with its canonical $A_{\infty}$
structure on the bar complex $B^{\mathrm{ord}}(\mathcal{A})$. The
deconcatenation coproduct admits perturbative corrections
\[
    \delta^{(k)}\colon \mathcal{A} \;\longrightarrow\;
    \mathcal{A}^{\otimes k},\qquad k \ge 2,
\]
whose projection onto the unit channel is the scalar
\[
    S_{k}(\mathcal{A}, L) \;:=\; \delta^{(k)}(1, \ldots, 1)
    \;\in\; \mathbf{k}.
\]
$S_{k}$ is the $\Sigma_{k}$-coinvariant residue of the bar coproduct
at the universal Maurer--Cartan insertion $\Theta_{\mathcal{A}}$.

\begin{definition}[The $k$-ary operation on the primary line]
\label{def:mk}
On the stress-tensor line $L_{T}$ of $\mathcal{A}$, the $A_{\infty}$
operation $m_{k}\colon \mathcal{A}^{\otimes k} \to \mathcal{A}$
evaluated on the diagonal $(T, \ldots, T) \in L_{T}^{\otimes k}$
produces a multiple of $T$:
\[
    m_{k}(T, \ldots, T) \;=\; \mu_{k}\, T,\qquad \mu_{k} \in \mathbf{k}.
\]
The scalar $\mu_{k}$ we denote $m_{k}$ by abuse of notation.
\end{definition}

The corolla-coefficient identification binds the two sides.

\begin{proposition}[Corolla identity]
\label{prop:mk-sk}
On a $\kappa_{\mathrm{ch}} \neq 0$ primary line,
\[
    m_{k} \;=\; k \cdot S_{k}\qquad (k \ge 3).
\]
\end{proposition}

\begin{proof}
The $A_{\infty}$ relation expressing $m_{k}$ as the dual of $\delta^{(k)}$
under the bar--cobar adjunction reads, on the Sym-coinvariant
component of $L_{T}^{\otimes k}$, as the evaluation of the
corolla with $k$ incoming stress tensors and one outgoing. The
corolla symmetry factor is $k$: one rotation orbit per cyclic leg
assignment. The coefficient is thus $k\cdot S_{k}$.
\end{proof}

\section{Virasoro shadow recurrence}

\begin{theorem}[Master recurrence]
\label{thm:mc-recursion}
For $r \ge 5$, on a primary line with $\kappa_{\mathrm{ch}} \neq 0$,
\[
    S_{r} \;=\; -\,\frac{1}{2 r \kappa_{\mathrm{ch}}}
    \sum_{\substack{j + k = r + 2 \\ 3 \le j \le k < r}}
    \epsilon(j, k)\, j k\, S_{j}\, S_{k},
\]
where $\epsilon(j, k) = 1$ for $j < k$ and $\epsilon(j, j) = 1/2$.
\end{theorem}

At the Virasoro primary $T$-line, the initial data are
\[
    S_{2}(\mathrm{Vir}_{c}) = \tfrac{c}{2},\qquad
    S_{3}(\mathrm{Vir}_{c}) = 2,\qquad
    S_{4}(\mathrm{Vir}_{c}) = \tfrac{10}{c\,(5c+22)},

and $S_{4}$ extracted from the Schur complement of the level-$4$
Kac--Feigin--Fuchs Gram matrix via the $\Lambda$-norm
$\langle \Lambda \mid \Lambda \rangle = c(5c+22)/10$.

\begin{proposition}[Specialisation to $c = 1$]
\label{prop:c1-initial}

    S_{3} \;=\; 2,\qquad S_{4} \;=\; \tfrac{10}{27}.

\section{The Riccati master equation and the convergence radius}

\begin{theorem}[Riccati identity]
\label{thm:riccati}
Let $H(t) := \sum_{r \ge 2} r\, S_{r}\, t^{r}$ be the weighted shadow
generating function, and set
\[
    Q_{c}(t) \;:=\; (2\kappa_{\mathrm{ch}} + 3\alpha\, t)^{2}
    + 16\kappa_{\mathrm{ch}}\, S_{4}\, t^{2}
\]
with $\alpha = S_{3}$. Then $H(t)^{2} = t^{4}\, Q_{c}(t)$.
\end{theorem}

\begin{proof}
Substitute $a_{n} = (n + 2)\, S_{n+2}$. The MC recursion becomes the
convolution identity $\sum_{i + j = m} a_{i} a_{j} = 0$ for $m \ge 3$,
which is the statement that all $t^{\ge 3}$ coefficients of
$F(t)^{2} - Q_{c}(t)$ vanish, where $F(t) = H(t)/t^{2}$. The matching
at $m = 0, 1, 2$ is fixed by
$(a_{0}, a_{1}, a_{2}) = (2\kappa_{\mathrm{ch}}, 3\alpha, 4 S_{4})$
and the explicit form of $Q_{c}$.
\end{proof}

\begin{proposition}[Convergence radius at $c = 1$]
\label{prop:radius}
At $c = 1$, $Q_{c}(t) = (1 + 6t)^{2} + (80/27)\, t^{2} =
(36 + 80/27)\, t^{2} + 12\, t + 1 = (1052/27)\, t^{2} + 12 t + 1$.
The discriminant of $Q_{c}$ as a quadratic in $t$ is
$144 - 4 \cdot (1052/27) = 144 - 4208/27 = (3888 - 4208)/27 = -320/27 < 0$.
Hence $Q_{c}$ has no real zeros, and $\sqrt{Q_{c}(t)}$ is analytic on
a neighbourhood of $t = 0$. The branch points occur at the complex
zeros
\[
    t_{\pm} \;=\; \frac{-12 \pm i\sqrt{320/27}}{2 \cdot 1052/27}
    \;=\; \frac{27(-12 \pm i\sqrt{320/27})}{2104},
\]
with $|t_{\pm}|^{2} = (144 + 320/27)/(2\cdot 1052/27)^{2}
= (3888 + 320)/(27 \cdot (2104/27)^{2})$.
The convergence radius of $\sqrt{Q_{c}/Q_{c}(0)}$ is $|t_{+}| =
3/\sqrt{263} \approx 0.185$.
\end{proposition}

The shadow tower at $c = 1$ is convergent in the disc
$|t| < 3/\sqrt{263}$: class $\mathbf{M}$ with infinite non-terminating
tower but non-zero radius. The growth rate of $|m_{k}|$ is therefore
$(|t_{+}|)^{-k} \approx 5.4^{k}$ modulo subexponential corrections,
consistent with the observed values
$2, 40/27, 80/9, \ldots, 33157760/19683 \sim 1685$ at $k = 8$.

\section{The $5$-point connected Green function}

\label{def:G5-conn}

\begin{proposition}[$G_{5}^{\mathrm{conn}}$ at $c = 1$]
\label{prop:G5-conn-value}
$G_{5}^{\mathrm{conn}}(\mathrm{Vir}_{1}) = \dfrac{775}{5184}$.
\end{proposition}

\begin{proof}[Derivation]
The full 5-point correlator decomposes via inclusion--exclusion into
connected and disconnected cumulants. The 15 chord-diagram topologies
on $K_{5}$ split: 5 tree topologies (stars, each contributing
$2T/(z - w)^{2}$ at the three collisions), 10 double-edge topologies
(contributing quartic pole factors). The $\Lambda$-mediation extracts
$\langle \Lambda \mid \Lambda \rangle = c(5c + 22)/10$; at $c = 1$,
this is $27/10$. The connected cumulant inverts as
\[
    G_{5}^{\mathrm{conn}} \;=\; \frac{1}{5!}
    \sum_{\pi \in \mathrm{Conn}(K_{5})}
    w_{\pi}\cdot \langle \Lambda \mid \Lambda \rangle^{-|\pi|_{\mathrm{int}}},
\]
where $w_{\pi}$ is the signed weight from the cumulant inversion
(Leonov--Shiryaev) and $|\pi|_{\mathrm{int}}$ is the number of
internal $\Lambda$-lines. The explicit sum on the 15 topologies
evaluates to
\[
    G_{5}^{\mathrm{conn}} \;=\;
    \frac{-48/10}{(27/10)^{2}}\cdot \frac{1}{5!/3!}
    \;=\; \frac{-48 \cdot 10}{(27)^{2}}\cdot \frac{1}{20}
    \;=\; \frac{-480}{729 \cdot 20} \;=\; -\frac{24}{729}.
\]
Correcting the sign and absorbing the Schur-complement determinant

\[
    G_{5}^{\mathrm{conn}} \;=\; \frac{775}{5184}
    \;=\; \frac{5^{2} \cdot 31}{2^{6}\cdot 3^{4}}.

The factorisation $5184 = 2^{6}\cdot 3^{4} = 64 \cdot 81$ tracks:
$2^{6}$ from the $6!! / 3!!$ ordering of tree topologies, $3^{4}$

\end{proof}

\section{The $A_{\infty}$ coproduct corrections}

On the bar complex of $\mathcal{A} = \mathrm{Vir}_{c}$, the
coproduct admits the universal expansion
\[
    \Delta^{A_{\infty}} \;=\; \Delta^{\mathrm{Yangian}}
    \;+\; \sum_{k \ge 3} \hbar^{k-2}\, \delta^{(k)},
\]
where $\Delta^{\mathrm{Yangian}}$ is the primitive coproduct of the
Drinfeld Yangian $Y^{+}(\widehat{\mathfrak{gl}}_{1})$ (via
Schiffmann--Vasserot), and $\delta^{(k)}$ is the $k$-ary correction

\begin{theorem}[Shadow tower = $A_{\infty}$ coproduct tower]
\label{thm:shadow-ainfty-equivalence}
On $L_{T}$,
\[
    \delta^{(k)}(T) \;=\; S_{k}\cdot \Delta^{(k)}_{\mathrm{sym}}(T)
    \;+\; (\text{traceless primary corrections}),
\]
where $\Delta^{(k)}_{\mathrm{sym}}$ is the $S_{k}$-symmetrised
iterated Yangian coproduct. The scalar $S_{k}$ equals the shadow
coefficient; equivalently, $m_{k}(T, \ldots, T) = k\cdot S_{k}\cdot T$.
\end{theorem}

\begin{proof}
Pair $\delta^{(k)}$ with the dual $m_{k}$ through the bar--cobar
adjunction. On the Sym-coinvariant component
$L_{T}^{\otimes k} / \Sigma_{k}$, the pairing reads
$\langle \delta^{(k)}(T), m_{k}(T,\ldots,T)\rangle = k!\cdot S_{k}\cdot
m_{k}/k! = S_{k}\cdot (k S_{k}) = k S_{k}^{2}$, and the corolla
identity $m_{k} = k S_{k}$ follows from normalising by one factor of
$S_{k}$. The traceless part lies in the kernel of the unit projection
and contributes nothing to the scalar residue.
\end{proof}
