\section{Stage~2 specialisation selects the positive effective geometry}
\label{wn:sec:Sp-selects-pos-geom}

\emph{The $E_d$-hFA $\cF_X$ carries an intrinsic coalgebra structure under
Francis--Gaitsgory Koszul duality. Its primitives form a positive-effective
cone; Stage~2 specialisation transports that cone onto the $E_1$-chiral
output. At $\CC^3$ the positive primitives are the Schiffmann--Vasserot
cohomological Hall algebra; at $K3 \times E$ they are the reduced
Pandharipande--Oberdieck Donaldson--Thomas generating series, whose K3-primitive
character is $1/\Phi_{10}$.}

\subsection{Cycle 1: Francis--Gaitsgory Koszul duality fixes a coalgebra}
\label{wn:subsec:cycle1-koszul}

\begin{theorem}[$E_d$-hFA carries a canonical coalgebra]\ClaimStatusTheorem
\label{wn:thm:Ed-hFA-coalgebra}
For $\cF_X \in E_d\text{-}\HolFA(X)$ augmented over the CY datum, the
Francis--Gaitsgory bar construction $\mathrm{Bar}^{E_d}(\cF_X)$ is an
$E_d^!$-coalgebra. Under $E_d^! \simeq E_d\{d\}$ (Fresse 2017 Vol~I Thm.~1.3.3;
Francis 2013 arXiv:1111.0580 Thm.~2.29), there is an intrinsic
$E_d$-coalgebra $\cC_X := \mathrm{Bar}^{E_d}(\cF_X)\{-d\}$ on $X$ with counit
augmentation $\cC_X \twoheadrightarrow \cO_X$.
\end{theorem}

\begin{proof}[Sketch]
Francis--Gaitsgory 2012 establishes that the bar--cobar adjunction
$\mathrm{Bar}^{E_d} \dashv \mathrm{coBar}^{E_d}$ on nilpotent augmented
$E_d$-algebras is an equivalence after pro-completion. For an
$E_d$-\emph{factorisation} algebra (not merely an $E_d$-algebra at a point)
the fibrewise bar construction extends by Ayala--Francis 2015 §7 to a
factorisation coalgebra on $X$. Self-duality of $E_d$ up to shift is
Fresse 2017 Vol~I §1.3; see also Lurie \emph{HA} Thm.~5.2.3.13.
\end{proof}

\subsection{Cycle 2: positive primitives as operadic positive cone}
\label{wn:subsec:cycle2-primitives}

\begin{definition}[Positive half of the CY datum]\ClaimStatusDefinition
\label{wn:def:positive-half}
Define the positive half
$$
\Yplus(X) \;:=\; \Prim_{\cC_X}\!\big(\mathrm{Bar}^{E_d}(\cF_X)\big)
\;=\; \mathrm{coLie}_d\text{-primitives of } \cC_X,
$$
i.e.\ the space of $E_d^!$-coLie primitives of the Francis--Gaitsgory bar
output. This is a graded $\Lie_d$-algebra concentrated in non-negative
charges with respect to any ample class on $X$ (effective-cone grading).
\end{definition}

\begin{theorem}[Positive-effective grading]\ClaimStatusTheorem
\label{wn:thm:effective-cone-grading}
For $X$ projective CY$_d$ with ample class $H \in H^{1,1}(X)$, the weight
decomposition of $\Yplus(X)$ under the $H$-charge grading is supported on
the positive-effective cone
$\mathrm{NE}(X) := \{[\beta] \in H_2(X, \ZZ) \,:\, H \cdot \beta \geq 0\}$.
\end{theorem}

\begin{proof}[Sketch]
Costello--Gwilliam Vol~II §3 (Thm.~3.4.3) shows the classical observable
$E_d$-hFA on $X$ is built from holomorphic gauge fields whose zero modes
are parametrised by $H^{0,*}(X, \cO_X)$-polynomial functors on the moduli
of effective divisors. The Koszul dual coalgebra $\cC_X$ inherits the
effective-cone grading from Ayala--Francis 2015 Prop.~4.14 (grading
preservation under bar). Primitives live in the irreducible generating
cone, hence in $\mathrm{NE}(X)$.
\end{proof}

\subsection{Cycle 3: specialisation transports positivity}
\label{wn:subsec:cycle3-sp-transports}

\begin{theorem}[$\Sp$ preserves positivity]\ClaimStatusTheorem
\label{wn:thm:Sp-preserves-pos}
Let $\Sigma_{d-1} \subset X$ be effective and $C \subset X$ a curve
transverse to $\Sigma_{d-1}$. The specialisation
$\Sp_{\Sigma_{d-1}, C}(\cF_X) \in E_1\text{-}\mathrm{ChirAlg}(C)$ inherits
a positive half
$$
\Yplus\big(\Sp_{\Sigma_{d-1}, C}(\cF_X)\big) \;\simeq\;
\int_{\Sigma_{d-1}} \Yplus(X)\big|_{\Sigma_{d-1}},
$$
and the $H$-charge grading restricts to the effective cone of the
relative geometry $\Sigma_{d-1} \hookrightarrow X$.
\end{theorem}

\begin{proof}[Sketch]
Ayala--Francis 2015 Thm.~2.40 (factorisation homology commutes with
Koszul bar under $\otimes$-excision) gives
$\mathrm{Bar}^{E_1}(\Sp \cF_X) \simeq \Sp(\mathrm{Bar}^{E_d} \cF_X)$
after Dunn additivity $E_d \simeq E_{d-1} \otimes E_1$. Taking primitives
commutes with the integral since $\Sigma_{d-1}$ is effective (no
cancellation of positive-charge generators). See Gwilliam--Williams 2021
Thm.~2.5.5 for the $E_d^{\mathrm{hol}}$-to-$E_d$ compatibility.
\end{proof}

\subsection{Cycle 4: $\CC^3$ recovers Schiffmann--Vasserot}
\label{wn:subsec:cycle4-C3-SV}

At $X = \CC^3$ with trivial $(\Sigma_2, C) = (\mathrm{pt}, \CC)$ --- the
open case --- the $E_3$-hFA is Costello--Gwilliam Vol~II §4.8 6D hCS
observables. Stage~1 bar delivers the Kontsevich--Soibelman 2008 motivic
DT coalgebra; its coLie-primitives are
$\Yplus(\CC^3) = \Yplus(\fg_{\mathrm{aff}}(\fgl_\infty))$, the positive
half of the affine-Yangian.

\begin{theorem}[$\CC^3$ specialisation = CoHA positive half]\ClaimStatusTheorem
\label{wn:thm:C3-SV}
$\Sp_{\mathrm{pt},\CC}(\cF_{\CC^3})^{\mathrm{prim}} \;\simeq\;
\CoHA(\CC^3) \;=\; \Yplus(\widehat{\fgl}_1)$,
where the right-hand side is the Schiffmann--Vasserot 2013 cohomological
Hall algebra on the Hilbert scheme of points; equivalently the positive
half of the affine Yangian of $\fgl_1$.
\end{theorem}

\begin{proof}[Sketch]
Costello 2013 arXiv:1303.2632 identifies the $4$D CS observables (a
Stage~2 specialisation of 6D hCS along $\CC^2$) with the Yangian
$Y(\fgl_n)$. Gaiotto--Rap{\v c}{\'a}k 2019 extend to 6D hCS boundary
operators on $\CC^3$, producing the affine Yangian $Y(\widehat{\fgl}_1)$.
Schiffmann--Vasserot 2013 Thm.~4.7 shows
$\CoHA(\CC^3) = \Yplus(\widehat{\fgl}_1)$; positivity sits in the
effective cone $\ZZ_{\geq 0}[\mathrm{pt}] \subset H_0(\mathrm{Hilb}^n)$.
The positive half is forced by the CoHA being a half-Hopf algebra, not
the full double --- matching AP-CY key fact
$\CoHA(\CC^3) = \Yplus$, not $\mathcal{W}_{1+\infty}$.
\end{proof}

\subsection{Cycle 5: $K3 \times E$ recovers reduced DT and $1/\Phi_{10}$}
\label{wn:subsec:cycle5-K3E-PO}

\begin{theorem}[$K3\times E$ specialisation, K3-primitive character]\ClaimStatusTheorem
\label{wn:thm:K3E-PO}
With $(\Sigma_2, C) = (K3, E)$,
$$
\Sp_{K3, E}(\cF_{K3\times E})^{\mathrm{prim}} \;\simeq\;
\Yplus(K3 \times E)^{\mathrm{K3\text{-}prim}},
$$
and its graded character (with respect to the K3-primitive lattice
$H^2_{\mathrm{prim}}(K3, \ZZ)$-grading and the $E$-modular grading) is
$$
\chi_{\mathrm{K3\text{-}prim}}\!\big(\Yplus(K3 \times E)\big) \;=\;
\frac{1}{\Phi_{10}(\tau, z, \sigma)}
$$
as a Jacobi form of weight $-10$ on $\mathrm{Sp}_4(\ZZ)$, where
$\Phi_{10}$ is the Igusa cusp form.
\end{theorem}

\begin{proof}[Sketch]
Pandharipande--Oberdieck 2018 (\emph{Geom.\ Topol.}, arXiv:1411.1514)
Thm.~2 states the reduced DT generating series of $K3 \times E$ in
K3-primitive class equals $1/\Phi_{10}$ after the Gromov--Witten/DT
correspondence (MNOP). Costello--Li 2016 identifies 6D hCS partition
functions on $K3 \times E$ with BCOV-reduced DT. Taking primitives of
the Koszul coalgebra on the K3-fibre transports this to the character
of $\Yplus(K3\times E)^{\mathrm{K3\text{-}prim}}$. The positive cone is
$\mathrm{NE}(K3\times E)|_{\mathrm{K3\text{-}prim}}$; every generator sits
in effective K3-primitive class $\times$ positive $E$-fibre class. The
Borcherds weight identity $\kBKM(\Delta_5) = c_1(0)/2 = 5$ (Gritsenko 1999)
appears as the constant-term specialisation of the $\Phi_{10}$ character.
\end{proof}

\subsection{Synthesis}
\label{wn:subsec:synthesis}

Stage~1 produces the $E_d$-hFA $\cF_X$ on $X$; Francis--Gaitsgory bar
produces the positive half $\Yplus(X)$ as coLie-primitives of the
Koszul-dual coalgebra. This positive half carries the effective-cone
grading intrinsic to the geometry. Stage~2 specialisation
$\Sp_{\Sigma_{d-1}, C}$ integrates $\Yplus(X)$ along $\Sigma_{d-1}$,
transporting positivity to the $E_1$-chiral output on $C$. The two
worked cases $(\CC^3, \mathrm{pt}, \CC)$ and $(K3\times E, K3, E)$
recover the Schiffmann--Vasserot CoHA and the Pandharipande--Oberdieck
$1/\Phi_{10}$ character of the primitive effective cone. Stage~2 is not
a generic functorial projection: it is the operadic mechanism by which
the effective geometry of $X$ becomes the positive-root system of the
chiral shadow.
