\section{Specialisation and Positive Effective Geometry}
\label{wn:sec:Sp-selects-pos-geom}

\emph{An augmented $E_d$ holomorphic factorisation algebra determines a
Koszul bar coalgebra. An effective charge monoid cuts out its positive
primitive sector. A witnessed factorisation-homology specialisation with
degree-zero kernel preserves that sector. For $d \geq 3$ the output on
the reference curve is native $E_1$; braided structure belongs to a
centre, a completed Hall-Drinfeld double, or a module category after
the pairing and completion data have been supplied. The toric model is
$\CoHA(\CC^3)=\Yplus(\widehat{\fgl}_1)$ after the standard equivariant
localisation and completion. The compact model $K3\times E$ gives the
Oberdieck-Pandharipande reduced-DT scalar; compact Hall-Borcherds
recognition is an additional theorem with its own hypotheses.}

\subsection{Koszul Bar Coalgebra}
\label{wn:subsec:koszul-coalgebra}

\begin{proposition}[Bar coalgebra of an augmented $E_d$ holomorphic factorisation algebra]
\label{wn:prop:Ed-hFA-bar-coalgebra}
Let $\cF_X\in E_d\text{-}\HolFA(X)$ be an augmented holomorphic
factorisation algebra on a CY$_d$ target, valued in a pro-nilpotent
complete category where Francis-Gaitsgory Koszul duality applies. Then
\[
 \mathrm{Bar}^{E_d}(\cF_X)
\]
is an $E_d^!$ coalgebra. Using the standard operadic self-duality
$E_d^!\simeq E_d\{d\}$, the shifted object
\[
 \cC_X:=\mathrm{Bar}^{E_d}(\cF_X)\{-d\}
\]
is an $E_d$ coalgebra with counit induced by the augmentation of
$\cF_X$.
\end{proposition}

\begin{proof}
Francis and Gaitsgory, \emph{Chiral Koszul duality} (2012), construct
Koszul duality for augmented chiral algebras under nilpotence and
completion hypotheses. On a disk the construction is the usual reduced
$E_d$ bar construction. Factorisation descent transports the coalgebra
structure across the cosheaf. The displayed shift is exactly the
operadic suspension in the Koszul self-duality
$E_d^!\simeq E_d\{d\}$, as in Fresse,
\emph{Homotopy of Operads and Grothendieck-Teichm\"uller Groups},
Vol.~I, Theorem~1.3.3, and Lurie, \emph{Higher Algebra},
Theorem~5.2.3.13.
\end{proof}

\subsection{Effective Positive Half}
\label{wn:subsec:positive-primitives}

\begin{definition}[Positive half from an effective charge monoid]
\label{wn:def:positive-half}
Let $M_X$ be an effective charge monoid for the CY datum. For a
projective target take the monoid generated by effective curve classes
and zero-dimensional charge. For a toric local target take the
non-negative dimension-vector monoid. Put
$M_X^\circ=M_X\setminus\{0\}$ and suppose the reduced bar coalgebra is
$M_X$-graded,
\[
 \cC_X=\bigoplus_{\gamma\in M_X}\cC_{X,\gamma},
 \qquad
 \Delta(\cC_{X,\gamma})
 \subset
 \bigoplus_{\gamma_1+\gamma_2=\gamma}
 \cC_{X,\gamma_1}\otimes\cC_{X,\gamma_2}.
\]
The positive primitive complex is
\[
 \Yplus_M(X)
 :=
 \bigoplus_{\gamma\in M_X^\circ}
 \Prim(\cC_X)_\gamma,
 \qquad
 \Prim(\cC_X)
 :=
 \ker\!\left(
   \bar\Delta\colon \bar\cC_X\to \bar\cC_X\otimes\bar\cC_X
 \right).
\]
When the coalgebra is realised by Hall correspondences, the shifted
Lie operations on primitives are the positive Hall Lie operations.
\end{definition}

\begin{lemma}[Effective support]
\label{wn:lem:effective-support}
For a projective CY$_d$ target with ample class $H$, every charge
$\gamma=(\beta,n)$ in the effective Hall or Donaldson-Thomas subcategory
generated by sheaves, ideal sheaves, or stable pairs has
$H\cdot\beta\geq0$. Consequently the primitive complex defined from the
effective Hall coalgebra is supported on the positive-effective cone.
\end{lemma}

\begin{proof}
The curve component $\beta$ of a stable sheaf, ideal sheaf, or stable
pair is represented by an effective one-cycle. Positivity of an ample
class gives $H\cdot\beta\geq0$. Hall multiplication and comultiplication
are defined by extensions inside the chosen effective subcategory, so
the charge of an extension is the sum of the two charges. The coproduct
is therefore graded by the effective monoid, and the primitive kernel is
a graded subcomplex of that monoid.
\end{proof}

\subsection{Specialisation Kernel}
\label{wn:subsec:sp-transports-positivity}

\begin{theorem}[Specialisation preserves the positive charge filtration]
\label{wn:thm:Sp-preserves-pos}
Let $(\Sigma_{d-1},C)$ be a witnessed admissible specialisation datum
for $X$, and let
\[
 \SpCh_{\Sigma_{d-1},C}\colon E_d\text{-}\HolFA(X)\longrightarrow
 E_{n(d)}\text{-}\HolFA(C),
 \qquad
 n(d)=
 \begin{cases}
 \infty,&d=1,\\
 2,&d=2,\\
 1,&d\geq3,
 \end{cases}
\]
be the factorisation-homology specialisation. Assume the witnessed kernel
$\cK_{\Sigma_{d-1},C}$ is degree zero for the effective charge
grading and the Beck-Chevalley, Fubini, properness, Tor-independence,
and completion cells of the witnessed datum hold. Then
$\SpCh_{\Sigma_{d-1},C}$ is filtered by effective charge and induces
\[
 \SpCh_{\Sigma_{d-1},C}\bigl(\Yplus_M(X)\bigr)
 \subset
 \Yplus_{M_C}\!\bigl(\SpCh_{\Sigma_{d-1},C}(\cF_X)\bigr),
\]
where $M_C$ is the image charge monoid on the reference curve. If the
specialisation is conservative on primitives and primitive extraction is
exact on the relevant graded conilpotent coalgebras, this inclusion is
an equality.
\end{theorem}

\begin{proof}
The witnessed kernel formula identifies specialisation with
\[
 \SpCh_{\Sigma_{d-1},C}(\cF)
 =
 \bigl((\pi_C)_*(\pi_X^*\cF\otimes^{\mathbf L}
 \cK_{\Sigma_{d-1},C})\bigr)^{E_{n(d)}\text{-}\mathrm{structure}}.
\]
This is part of the witnessed specialisation datum. The degree-zero
condition on $\cK_{\Sigma_{d-1},C}$ makes the push-pull
charge-preserving. The projection formula and Fubini cell identify
coproducts before and after pushforward, so the primitive kernel maps to
the primitive kernel after specialisation. Exactness and conservativity
on primitives upgrade the filtered map to equality. The
PTVV/Dunn-Lurie level calculation gives $n(d)=1$ for all $d\geq3$;
centre and Drinfeld-double constructions supply braided $E_2$ data once
their Hopf pairing and completion hypotheses are available.
\end{proof}

\subsection{The Toric Model $\CC^3$}
\label{wn:subsec:C3-SV-positive-half}

Let $Q$ be the one-vertex quiver with loops $X,Y,Z$ and potential
$W=\operatorname{tr}(X[Y,Z])$. Its Jacobi algebra is
\[
 \CC\langle X,Y,Z\rangle/
 ([X,Y],[Y,Z],[Z,X])
 \cong \CC[X,Y,Z],
\]
so $(Q,W)$ is the standard CY$_3$ quiver presentation of $\CC^3$.

\begin{theorem}[$\CC^3$ Hall positive half]
\label{wn:thm:C3-SV}
Let $\mathcal M_n$ be the stack of $n$-dimensional representations of
the Jordan three-loop quiver. The torus-equivariant vanishing-cycle
cohomological Hall algebra
\[
 \CoHA(\CC^3)
 :=
 \bigoplus_{n\geq0}
 H^\bullet_T(\mathcal M_n,\phi_W)
\]
is, after the standard equivariant localisation and shuffle completion,
the Schiffmann-Vasserot positive half,
\[
 \CoHA(\CC^3)
 \cong
 \Yplus(\widehat{\fgl}_1).
\]
On the toric hCS-Hall comparison locus, this is the Hall positive half
of the chiral specialisation of the $\CC^3$ CY$_3$ datum. The full
affine Yangian and the $\mathcal W_{1+\infty}$ vertex realisation are
obtained from the Drinfeld double or derived centre together with the
vacuum realisation.
\end{theorem}

\begin{proof}
Kontsevich-Soibelman identify the cohomological Hall algebra of a
quiver with potential from the vanishing-cycle cohomology of its
representation stacks. For the Jordan three-loop quiver the critical
locus is the commuting-triple locus, and the $T=(\CC^\times)^3$
equivariant localisation is governed by plane partitions, equivalently
the effective monoid $\ZZ_{\geq0}[\mathrm{pt}]$. Schiffmann and Vasserot
(2013, Theorem~4.7, in the affine-Yangian normalisation
$h_1+h_2+h_3=0$) identify the localised CoHA with the positive shuffle
half of the affine Yangian of $\fgl_1$. Tsymbaliuk's
Drinfeld-current presentation identifies the completed double with the
full affine Yangian. The vertex algebra $\mathcal W_{1+\infty}$ is a
vacuum-module realisation of the full algebra, so the CoHA itself is the
positive half.
\end{proof}

\subsection{\texorpdfstring{$K3\times E$}{K3 x E}: Reduced DT Character and Hall Recognition}
\label{wn:subsec:K3E-PO-character}

\begin{proposition}[\texorpdfstring{$K3\times E$}{K3 x E} scalar test and primitive denominator]
\label{wn:prop:K3E-PO}
Let $X=K3\times E$ and use the K3-fibre specialisation
$(\Sigma_2,C)=(K3,E)$. On the framed object-level locus of the
CY$_3$ construction,
\[
 A_{K3,E}:=
 \SpCh_{K3,E}\bigl(\PhiFA_3(\mathrm{Perf}(K3\times E))\bigr)
 \in E_1\text{-}\mathrm{ChirAlg}(E).
\]
Oberdieck and Pandharipande compute the reduced primitive
Donaldson-Thomas scalar as
\[
 Z^{\mathrm{red}}_{\DT}(K3\times E)
 =
 -(\PhiTenOP)^{-1}
 =
 -\Dfive^{-2}
 =
 -4096\,\DeltaFive^{-2},
\]
with
\[
 \Dfive=64^{-1}\DeltaFive,\qquad
 \PhiTenOP=\Dfive^2=4096^{-1}\DeltaFive^2,\qquad
 \PhiTenUn=\DeltaFive^2.
\]
The primitive Borcherds denominator has
\[
 \kBKM(\DeltaFive)=c_1(0)/2=10/2=5.
\]
The scalar identity is a character-level arithmetic test for the target
denominator. A compact Hall recognition theorem identifying
$A_{K3,E}$ with $U_{\mathrm{ch}}(\fg_{\DeltaFive})$ additionally
requires a compact critical Hall source, orientation and parity data, a
Hopf pairing, radical quotient, PBW theorem, relation-exhaustion check,
and Mittag-Leffler compatibility.
\end{proposition}

\begin{proof}
Oberdieck and Pandharipande, \emph{Curve counting on $K3\times E$, the
Igusa cusp form $\chi_{10}$, and descendent integration}, compute the
reduced DT partition function in primitive K3 class and identify it
with the reciprocal of the OP-normalised Igusa square. The normalisation
$\Dfive=64^{-1}\DeltaFive$ is the monic denominator convention used in
the $K3\times E$ BKM chapter, so $\PhiTenOP=\Dfive^2$ and
$\PhiTenUn=\DeltaFive^2$ are scalar-square conventions of weight $10$.
Borcherds' singular theta lift weight formula gives
$\wt(\DeltaFive)=c_1(0)/2=5$, with
$\phi_{0,1}|_{q^0}=\zeta^{-1}+10+\zeta$. Character agreement supplies
the denominator test. Recognition of
$\SpCh_{K3,E}(\PhiFA_3(\mathrm{Perf}(K3\times E)))$ as a compact
Hall-Borcherds chiral envelope needs the additional Hall source,
orientation, pairing, quotient, PBW, and completion data listed above.
\end{proof}

\begin{corollary}[Numerical invariants of \texorpdfstring{$K3\times E$}{K3 x E}]
\label{wn:cor:K3E-four-invariants}
The standard $K3\times E$ values on the compact, Heisenberg, Borcherds,
and fibre readings are
\[
 \kcat(K3\times E)=0,\qquad
 \kch(K3\times E)=0,\qquad
 \kHeis(K3\times E)=3,\qquad
 \kBKM(\DeltaFive)=5,\qquad
 \kfib(K3)=24.
\]
They arise from distinct constructions: K\"unneth multiplicativity of
$\chi(\cO)$ on the total space, the compact Hodge supertrace, the
Heisenberg-Cartan rank of the K3-fibre specialisation, the Borcherds
weight of the primitive denominator, and the K3 Mukai/fibre rank.
\end{corollary}

\begin{proof}
K\"unneth gives
$\chi(\cO_{K3\times E})=\chi(\cO_{K3})\chi(\cO_E)=2\cdot0=0$, hence
$\kcat(K3\times E)=0$. The compact chiral Hodge supertrace is
$\sum_q(-1)^q h^{0,q}(K3\times E)=1-1+1-1=0$, hence
$\kch(K3\times E)=0$. The Heisenberg-Cartan specialisation has rank
$2+1=3$. The Borcherds value is the weight calculation of
Proposition~\ref{wn:prop:K3E-PO}. The fibre value is the Mukai rank, or
equivalently the topological Euler number
$\chi_{\mathrm{top}}(K3)=24$. These values come from distinct
constructions.
\end{proof}

\subsection{CHL and Gritsenko-Cl\'ery Constants}
\label{wn:subsec:constants}

The primitive CHL denominator ladder on the $K3\times E$ slice is
\[
\begin{array}{c|ccccc}
N&1&2&3&4&6\\ \hline
c_N(0)&10&8&6&4&2\\
\kBKM(\Phi_N)=c_N(0)/2&5&4&3&2&1
\end{array}
\]
by Borcherds 1998 Theorem~13.3, Gritsenko-Nikulin 1995 Part~II, and
Gritsenko 1999.

The Nikulin symplectic orders outside primitive compact $K3\times E$
CHL are
\[
\begin{array}{c|ccc}
N&5&7&8\\ \hline
c_N(0)&4&2&2\\
\kBKM(\Phi_N)=c_N(0)/2&2&1&1
\end{array}
\]
They are Borcherds-weight values for the corresponding twined K3
Jacobi inputs. Compact $K3\times E$ CHL statements at these orders
require a source package with the quotient, orientation, reduced DT
theory, and Hall recognition data.

The diagonal-divisor Gritsenko-Cl\'ery catalogue is a separate
triple-indexed atlas. Its triple order is
\[
(t,N)=(1,1),(2,1),(1,2),(3,1),(1,3),(4,1),(1,4),(2,2),
\]
and its weight tuple is
\[
 \wt=(5,2,3,1,2,\tfrac12,\tfrac32,1).
\]
Equivalently $2\wt=(10,4,6,2,4,1,3,2)$. These diagonal-divisor
weights are distinct from the primitive CHL constant table. Comparison
with a CHL denominator requires a row map and normalisation theorem
identifying the two inputs.

\subsection{Positive Root System}
\label{wn:subsec:positive-root-system}

The construction separates three layers. The bar coalgebra supplies
primitives. The effective charge monoid selects the positive half. The
specialisation kernel transports that positive half to the native
$E_1$ chiral output when $d\geq3$. At $\CC^3$ this gives the
Schiffmann-Vasserot CoHA positive half
$\Yplus(\widehat{\fgl}_1)$. At $K3\times E$ the reduced-DT scalar
fixes the primitive denominator test. Compact Hall-Borcherds recognition
must still supply the algebraic source, orientation character, pairing,
radical quotient, and PBW theorem. The Hall-Drinfeld/BKM branch and the
K3 self-mirror Yangian branch are distinct comparison problems.
