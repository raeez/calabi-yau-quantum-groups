\section*{Wave-18 H5/20: positive-geometry gluing, reduced DT, and
                 chiral-hFA/CoHA equivalence}
\label{wn:sec:wave18-h5-synthesis}

Three strands close into one grammar. The first is Wave-17: the global
positive half is glued from affine charts along a cluster cocycle and
obeys the universal rule $\Ypluscy = \Heq(\Meff(X), \phi_W)$. The second
is Wave-18 F: $K3\times E$ does not admit a global NCCR, so the
toric-quiver shortcut is unavailable; what survives is a reduced-DT
positive effective geometry with $\Cstar_E$ equivariance. The third is
Wave-18 G: holomorphic-Chern--Simons observables on $\C^3$ and the
cohomological Hall algebra are two presentations of the same
$E_3$-factorisation algebra, matched at the level of the
Maulik--Okounkov $R$-matrix. Five attack-heal cycles assemble these
into a single statement.

\subsection*{Attack-heal 1 --- universal: $G(X)=D(\Ypluscy)$ as the
            Drinfeld double of a chart-glued positive half}

\ClaimStatusDefinition\ \emph{Universal grammar.} For a Calabi--Yau
$\infty$-category $X$ of dimension $d$, the positive half is
$\Ypluscy = \Heq(\Mbps(X), \phi_W)$, the equivariant cohomology of the
effective moduli stack carrying the Kontsevich--Soibelman vanishing
cycle structure (KS 2008, arXiv:0811.2435, \S\S 6--7; Schiffmann--
Vasserot 2013, Thm.~1.1 for $X=\C^3$; Davison--Meinhardt 2020
Invent.~Math.\ 221, cohomological integration map).

\ClaimStatusDerivation\ \emph{Drinfeld double.} Given a non-degenerate
Hopf pairing between $\Ypluscy$ and the opposite half $Y^-(X)$ obtained
by Serre duality on $\Db(X)$ (Ext-shifted by $d$), the Drinfeld double
is $G(X) = \Ypluscy \bowtie Y^0(X) \bowtie Y^-(X)$. The pairing is the
Atiyah--Bott equivariant residue at toric fixed loci and the
Davison--Meinhardt PBW-integration in the compact stratum; the two
agree on the intersection. \emph{Six routes to} $G(K3\times E)$ are six
presentations of this one double, not six applications of $\Phi$.

\subsection*{Attack-heal 2 --- case stratification: toric, orbifold,
             compact non-toric, lattice-polarised}

\ClaimStatusDerivation\ Four strata share one grammar; only the
coefficient ring $\Heq(\mathrm{pt})$ differs.

\begin{enumerate}
\item \emph{Toric CY$_3$.} NCCR exact: Van den Bergh 2004 (Duke 122)
      provides a tilting bundle whose endomorphism algebra is the
      Jacobi algebra of a quiver with potential $(Q_X, W_X)$;
      $\Heq = H^{*}_{T}$ with $T=(\Cstar)^3$, and the MO stable
      envelope computes characters via $T$-fixed-point localisation.
      Examples: $\C^3$ Jordan quiver, resolved conifold, local $\bP^2$,
      local del Pezzo $\dP_n$ with $n\le 3$.

\item \emph{Orbifold / finite-symplectic.} $X=[Y/G]$ with $G$ finite
      symplectic. NCCR + inertia: Wemyss 2018 on ADE chart atlases,
      plus orbifold equivariance $\Heq = H^{*}_{G}$ on the inertia
      stack. Nikulin--Mukai $M_{23}$ classification for $K3$
      symplectic automorphisms; CHL $(K3\times E)/(\Z/N)$ for
      $N\in\{1,2,3,4,6\}$.

\item \emph{Compact non-toric CY$_3$.} Quasi-NCCR substitute + reduced
      DT: no global tilting by Serre-duality obstruction
      ($\omega_X \cong \cO_X$ kills finite global dimension;
      Proposition on no-tilting-on-K3, Bridgeland 2008 Duke 141).
      Replace by reduced virtual class on K3-primitive classes
      (Maulik--Pandharipande--Thomas 2010, arXiv:1001.2719) with
      residual $\Cstar_E$ from $E$-translation. Character:
      Oberdieck--Pixton 2017 (arXiv:1706.10100, Invent.~Math.\ 222)
      $Z^{\mathrm{red}}_{\DT}(K3\times E;\gamma\,\text{prim on }K3)
      = -1/\PhiTen(q,t,p)$.

\item \emph{Lattice-polarised.} Hocolim over $\mathcal{M}_L$: when $X$
      is polarised by $L\subset H^2(X,\Z)$, the Hall stack fibres over
      the period domain $\mathrm{O}(L)\backslash \mathbb{H}_{g}$;
      $\Heq = H^{*}_{\mathrm{O}(L)}$, and the character of $\Ypluscy$
      is an automorphic form. Gritsenko--Nikulin 1998 Borcherds lift
      of $\phi_{0,1}$ to $\Delta_5$ anchors $\kBKM(\PhiTen)
      = c_1(0)/2 = 5$.
\end{enumerate}

The K3$\times E$ NCCR obstruction (Wave-18 F1): the Seidel--Thomas
spherical twists around $(-2)$-classes generate $\Db(\Coh\,K3)$ without
presenting it as $\mathrm{mod}\text{-}\Lambda$ for finite-dimensional
$\Lambda$; the stratum-3 quasi-NCCR is the operative substitute, and
no $T^3$-action forces the replacement of the toric triple
(fixed points, integration, vertex) by the reduced triple (reduced
virtual class, $\Cstar_E$-residual integration, $\Auts(K3)$-orbit
averaging).

\subsection*{Attack-heal 3 --- chain-level compatibility: chiral
             factorisation $\mathcal{F}_X$ matches CoHA sheaf
             $\mathcal{Y}^+_X$}

\ClaimStatusTheorem\ \emph{hCS--CoHA match on $\C^3$.} The abelian
holomorphic-Chern--Simons observables $\Obs_{\hCS}^{\gl_1}(\C^3)$ in
the Costello--Gwilliam BV formalism (Costello--Li 2016 \S\S 3--4) and
the shuffle algebra $\bigoplus_n H^{*}_T(\Hilb^n(\C^3))$ of
Schiffmann--Vasserot 2013 are quasi-isomorphic as $E_3$-factorisation
algebras in $\mathrm{Ch}(\mathrm{Dolb})$:
\[
   \Obs_{\hCS}^{\gl_1}(\C^3)
   \;\simeq\;
   \bigoplus_{n\ge 0} H^{*}_{T}\bigl(\Hilb^n(\C^3)\bigr)
   \;=\;
   \Yplus(\glhat_1).
\]
$E_3$-structure on the hCS side is Dunn additivity
$E_3 = E_1\otimes E_1 \otimes E_1$ along $\C^3=\C\times\C\times\C$; on
the CoHA side it is the three-fold Feigin--Odesskii shuffle. Both
generate MacMahon $M(q) = \prod_{k\ge 1}(1-q^k)^{-k}$ at the free level,
and the non-abelian extension to $\gl_N$ via Wave-15 N1 matches term
by term against the Maulik--Okounkov $R$-matrix of $Y(\glhat_N)$ at
the Calabi--Yau torus $\epsilon_1+\epsilon_2+\epsilon_3 = 0$.

\ClaimStatusTheorem\ \emph{MO $R$-matrix as spectral residue of the
cocycle.} The Wave-17 B4 cluster gluing cocycle $\phi_{UV}(u)$,
rational in the spectral parameter $u$ with poles on the Jacobi-ideal
hyperplane arrangement, has the MO $R$-matrix at its wall residue:
$R^{\mathrm{MO}}_{C_+ C_-}(u) = \mathrm{Res}_{u=u_\star}
\phi^+_{UV}(u) \circ (\Stab^U_{C_+})^{-1}$ (Wave-18 G4). Yang-Baxter on
$R^{\mathrm{MO}}$ descends from Koszul-homotopy closure of $\phi$ on
triple overlaps; the dynamical twist is the regular part of $\phi^+_{UV}$
away from walls. On the Jordan $\C^3$ chart this recovers the Yang
rational $R$-matrix; on the conifold, the Klebanov--Witten instanton
$R$-matrix of Smirnov.

\subsection*{Attack-heal 4 --- preface-grade statement}

\ClaimStatusSynthesis\ \emph{Grammar.}
$\Ypluscy = \Heq(\Mbps(X), \phi_W)$; the effective moduli $\Mbps(X)$
is one universal Hall site, cut by the positive cone of the numerical
$K$-theory lattice. \emph{Two-stage factorisation.}
$\Phi_d = \Sp_{\Sigma_{d-1}, C} \circ \PhiFA_d$: Stage~1 produces the
$E_d$-hFA $\mathcal{F}_X$ on $X$ (Costello--Gwilliam BV quantisation of
6D holomorphic Chern--Simons on a CY$_3$ datum); Stage~2 specialises
along a curve $C$ transverse to a positive cycle $\Sigma_{d-1}$,
transporting the positive cone onto the $E_1$-chiral output.
\emph{Positive half as moduli cohomology.} Francis--Gaitsgory Koszul
duality on $\mathcal{F}_X$ supplies an intrinsic $E_d^!$-coalgebra
$\mathcal{C}_X = \mathrm{Bar}^{E_d}(\mathcal{F}_X)\{-d\}$; its
coLie-primitives, graded by the effective cone
$\mathrm{NE}(X)\subset H_2(X,\Z)$, are $\Ypluscy$. In the two worked
cases:
\begin{itemize}
\item \emph{$\C^3$, $(\Sigma_2, C)=(\mathrm{pt}, \C)$.}
      $\Sp_{\mathrm{pt},\C}(\mathcal{F}_{\C^3})^{\mathrm{prim}}
       \simeq \CoHA(\C^3) = \Yplus(\glhat_1)$;
      positivity sits in $\Z_{\ge 0}[\mathrm{pt}]\subset
      H_0(\Hilb^n)$. Chain-level identity with
      MacMahon $M(q)$ and $Z_{\mathrm{Nek}}(\C^3) = M(q)$ confirms the
      match.
\item \emph{$K3\times E$, $(\Sigma_2, C)=(K3, E)$.}
      $\Sp_{K3,E}(\mathcal{F}_{K3\times E})^{\mathrm{prim}} \simeq
      \Yplus(K3\times E)^{K3\text{-prim}}$, with graded character
      $1/\PhiTen$ on the K3-primitive lattice (Oberdieck--Pixton 2017
      Thm.~1); the Borcherds weight constant term $\kBKM(\PhiTen)
      = c_1(0)/2 = 5$ from Gritsenko 1999.
\end{itemize}
The subscripted invariants read off $K3\times E$:
$\kcat(K3\times E)=\chi(\cO_{K3\times E})=0$ (K\"unneth against
$\chi(\cO_E)=0$, total space, not fibre),
$\kch(K3\times E) = 0$ (compact CY$_3$ Hodge supertrace),
$\kBKM(\PhiTen) = 5$, $\kfib(K3) = 24$ from Mukai lattice rank. The
spectrum $\{2, 3, 5, 24\}$ on $G(K3\times E)$ indexes four distinct
constructions (Mukai lattice, Igusa $\PhiTen$ via Gritsenko $\Delta_5$,
BKM Borcherds weight, K3 fibre rank), not four $\Phi$-applications.

\subsection*{Attack-heal 5 --- residual open problems}

\ClaimStatusOpen\ \emph{Global NCCR for $K3\times E$.} Seidel--Thomas
spherical content of $\Db(\Coh\,K3)$ is the obstruction; an ADE-chart
atlas with $A_\infty$-bimodule transitions supplies a quasi-NCCR, but
no global quiver-with-potential single-chart reduction exists
(Wave-18 F1 Corollary on NCCR obstruction). The universal grammar
survives: $\Yplus(K3\times E)$ is an $(\infty, 1)$-categorical object
organised by $\Phi$, not by a single quiver.

\ClaimStatusOpen\ \emph{Fake-Monster $d=5$ witness.} On
$K3_1\times K3_2\times E$ with the doubly-reduced DT stack and
rank-$5$ equivariance $\Cstar_E\times\Auts(K3_1)\times\Auts(K3_2)$,
the character $1/\Phi_{12}^{\mathrm{FM}}$ after Niemeier projection
onto the Leech slice $\Lambda_{24}\subset\mathrm{II}_{3,27}$ is
conjectural; the weight match $\kBKM(\Phi_{12}^{\mathrm{FM}})
= c(0)/2 = 12$ is secure (Borcherds 1990 Invent.~Math.\ 99,
Thm.~10.4), as is the Niemeier-no-roots selection principle from
Conway--Sloane Ch.~18 and Mukai 1988. The
bracket-level Lie-algebra identification
$\mathfrak{g}_{\mathrm{BPS}}(K3^2\times E)\cong \mathfrak{g}^{\mathrm{FM}}$
and the doubly-reduced DT integrand matching $\Phi_{12}^{-1}$ in closed
form are open.

\ClaimStatusOpen\ \emph{Bracket-level $K3\times E$.}
$\mathfrak{g}_{\mathrm{BPS}}(K3\times E)\cong \mathfrak{g}_{\Delta_5}$
at charge-by-charge dimensions via KS wall-crossing; the bracket-level
identification is open (AP-CY34 on $[m_3, B^{(2)}]$ cyclic
invariance).

\ClaimStatusOpen\ \emph{Polarised-irrational CY$_3$.} Hodge-graded
$\Heq$ vs.\ Joyce--Song motivic Hall for smooth projective CY$_3$
without torus and without Nikulin orbifold; existence of $\Meff(X)$
for compact non-toric $X$ beyond the Oberdieck--Pixton reduced
template is open.
