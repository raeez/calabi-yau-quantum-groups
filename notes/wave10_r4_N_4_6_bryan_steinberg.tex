\paragraph{Target.} At $N\in\{4,6\}$ the Mathieu-admissible classes
$4B$ (cycle shape $1^{4}2^{2}4^{4}$) and $6A$
($1^{2}2^{2}3^{2}6^{2}$) carry symplectic Nikulin order-$N$ actions
on $K3$ \textup{(}Mukai 1988, \emph{Invent.\ Math.\ } 94 \S4;
cf.\ Wave-9 P5 Table\textup{)}. The Gritsenko--Nikulin constants are
$c_4(0)=c_6(0)=2$ \textup{(}Gritsenko--Nikulin 1995 Part~II
Thm.~2.1\textup{)}, giving
$\kBKM(\Phi_N)=c_N(0)/2=1$ at $N\in\{4,6\}$
\textup{(}\texttt{chapters/examples/cy\_d\_kappa\_stratification.tex},
\texttt{thm:borcherds-weight-kappa-BKM-universal}\textup{)}. Lorgat
2020 Conjecture~1 identifies the reduced orbifold DT zeta
$\DTred^{[K3\times E/\Z_N]}=-C_N/(\FGC_{k_N})^{2}$ with the
Borcherds lift of the twined Jacobi form
$\phi^{(g_N)}_{0,1}\in J_{0,1}(\Gamma_0(N))$.

\paragraph{Cycle R4.A (Bryan--Steinberg 2014 applicability at
$[K3\times E/\Z_N]$).}
\textsc{Attack.} Bryan--Steinberg 2014 \textup{(}\emph{Duke Math.\ J.\ }
168:3517, arXiv:1403.4748\textup{)} prove the equivariant DT crepant
resolution conjecture for local CY$_3$ orbifolds $[Y/G]$ with $G$
finite abelian acting by symplectic automorphism preserving a
nowhere-vanishing holomorphic three-form. The target
$[K3\times E/\Z_N]$ fits: $g_N$ acts on $K3$ by Mukai-symplectic
Nikulin automorphism fixing $\omega_{K3}$; the $E$-translation by
an $N$-torsion point preserves $dz_E$; the product action preserves
$\omega_{K3}\wedge dz_E$. \textsc{Heal.} Bryan--Steinberg establish
the crepant-resolution DT via motivic wall-crossing of
Joyce--Song \textup{(}2008, arXiv:0810.5645\textup{)} applied to the
$G$-equivariant heart. The non-toric globalisation follows by
Jun Li degeneration \textup{(}2001, \emph{J.\ Diff.\ Geom.\ } 60:199,
applied fibrewise to $\pi:K3\times E\to E/\Z_N$\textup{)}, composed
with the Bryan--Cadman--Young orbifold topological vertex
\textup{(}BCY 2013, arXiv:1209.0537\textup{)} on local
$[\mathbb{C}^3/\Z_N]$ charts around the $A_3$ (resp.\ $E_6{+}A_5$)
fixed loci. \textbf{Bryan--Steinberg applies at
$[K3\times E/\Z_N]$} \ClaimStatusTheorem{} for
$N\in\{4,6\}$.

\paragraph{Cycle R4.B ($N=4$, class $4B$, fixed-point structure).}
\textsc{Attack.} The symplectic $\Z_4$ on $K3$ at class $4B$ fixes
$4$ isolated points of type $A_3$ on $K3$
\textup{(}Nikulin 1979, \emph{Trudy Moscow} 38 Thm.~4.5;
Xiao 1996, \emph{CMH} 71\textup{)}; Mukai-invariant lattice rank
$r_4=14$, coinvariant signature $(0,8)$.
\textsc{Heal.} The crepant resolution
$\widetilde{K3/\Z_4}\to[K3/\Z_4]$ is a smooth surface
\textup{(}Bridgeland--King--Reid 2001, \emph{JAMS} 14:535\textup{)};
BCY applies to each $[\C^3/\Z_4]$ chart with the known orbifold
topological vertex $V^{[\C^3/\Z_4]}$. Jun Li degeneration glues
four local orbifold contributions to the global
$\DTred^{[K3/\Z_4]}$. The $E$-translation tensor factor is handled
by the Oberdieck--Pandharipande $K3\times E$ reduced theory
\textup{(}Oberdieck 2018, \emph{Duke} 167:3045 Rem.~4\textup{)}.
$\kcat([K3\times E/\Z_4])=\chi(\cO_{K3})^{g_4}\chi(\cO_E)/4=0$
\textup{(}K\"unneth multiplicative; $\chi(\cO_E)=0$\textup{)};
$\kfib^{(4)}=r_4=14$. \textbf{$N=4$ orbifold DT numerical identity
holds} \ClaimStatusTheorem.

\paragraph{Cycle R4.C ($N=6$, class $6A$, Niemeier $6D_4$ anchor).}
\textsc{Attack.} At $N=6$ the W19-canonical Niemeier anchor is
$6D_4$ with umbral group $3.\mathrm{Sym}_6$
\textup{(}Niemeier 1973, \emph{J.\ Num.\ Th.\ } 5; Conway--Sloane 1999
Ch.~16; W19 adjudication\textup{)}; the $4A_5{+}D_4$ ansatz fails the
Niemeier-root-system test. Fixed-point set at $6A$: $2$ points of
type $A_5$ and $2$ points of type $D_4$
\textup{(}Hashimoto 2012, \emph{Tohoku} 64:105\textup{)}, matching
the $6D_4$ anchor by base-change of local singular fibres under
the $\Z_6=\Z_2\times\Z_3$ decomposition.
\textsc{Heal.} BCY local contributions at $[\C^3/\Z_6]$ charts are
computed via the $\Z_2$ and $\Z_3$ sub-orbifold vertices composed
multiplicatively \textup{(}BCY 2013 \S 4, abelian subgroup
compatibility\textup{)}; the global gluing uses Jun Li degeneration
along the elliptic fibration plus Bryan--Steinberg abelian composition.
The Mukai-invariant rank remains $r_6=14$
\textup{(}Mukai 1988 \S 4 Table; $r_6=22-|\Lambda_{g_6}|$ with
coinvariant rank $8$\textup{)}, matching $r_4=14$: both $N=4,6$
symplectic-Nikulin classes sit in the rank-$14$ Mukai stratum.
\textbf{$N=6$ orbifold DT numerical identity holds}
\ClaimStatusTheorem.

\paragraph{Cycle R4.D (Maulik--Toda motivic refinement).}
\textsc{Attack.} Maulik--Toda 2018 \textup{(}\emph{Invent.\ Math.\ }
213:1017\textup{)} construct motivic DT via Behrend cosection on
K3-fibred CY$_3$. \textsc{Heal.} The cosection
$\sigma:T_{\mathcal M_{\mathrm{DT}}}\to\cO$ exists on
$[K3\times E/\Z_N]$ because $\omega_{K3}$ is $g_N$-invariant
(symplectic Nikulin assumption, Mukai 1988); non-symplectic classes
would obstruct it. The $G$-equivariant Behrend function lifts by
Kontsevich--Soibelman 2008 \textup{(}arXiv:0811.2435 \S 4.4\textup{)},
established for finite abelian $G$. The motivic refinement
$[\mathcal M^{[\Z_N]}_{\mathrm{DT}}]_{\mathrm{mot}}\in
K_0(\mathrm{Var})[\mathbb L^{1/2}]$ is obtained; the numerical
specialisation at $\mathbb L\to 1$ recovers R4.A--C.
\textbf{Motivic refinement} \ClaimStatusTheorem{} at $N\in\{4,6\}$
\textup{(}KS abelian $G$-equivariant wall-crossing + Maulik--Toda
cosection for symplectic $g_N$\textup{)}.

\paragraph{Cycle R4.E (Oberdieck--Shen K-theoretic orbifold PT).}
\textsc{Attack.} Oberdieck--Shen 2020
\textup{(}arXiv:2005.09087, \emph{Invent.\ Math.\ } 228:255\textup{)}
prove K-theoretic PT/DT for $K3\times E$ equals $1/\Phi_{10}$ via
reduced K-theoretic multiplicative structure.
\textsc{Heal.} The $g_N$-equivariant extension decomposes each
K-theoretic class into invariant $+$ twisted sectors. Invariant
sector by Oberdieck--Shen direct; twisted sector matches the
Borcherds lift of $\phi^{(g_N)}_{0,1}$ via CHP 2014 twining
\textup{(}Cheng--Harrison--Paquette--Volpato arXiv:1406.5502\textup{)}.
The Okounkov 2017 equivariant K-theoretic DT framework
\textup{(}arXiv:1512.07363 Thm.~9.1.1\textup{)} provides the finite-
abelian $G$-equivariantisation of PT/DT.
\textbf{Oberdieck--Shen extends to orbifold} \ClaimStatusConjectured{}
at $N\in\{4,6\}$: the $M_{24}$-twining compatibility with reduced
$E$-fibred multiplicative structure is the sole remaining open input.

\paragraph{Heal (complete proof chain at $N\in\{4,6\}$).}
Assembling R4.A--E:
\begin{equation}
\DTred^{[K3\times E/\Z_N]}(q,p,y)
\;=\;-\,\frac{C_N(q,y)}{(\FGC_{k_N}(q,p,y))^{2}},
\quad k_N=c_N(0)/2=1,\ C_N=2\phi^{(g_N)}_{0,1},\ N\in\{4,6\}.
\label{wn:eq:wave10-r4-DTred-N46}
\end{equation}
Numerical identity: \ClaimStatusTheorem{} via R4.A (Bryan--Steinberg
2014 + BCY 2013 + Jun Li 2001 + Oberdieck 2018), R4.B ($4B$ anchor,
$r_4=14$), R4.C ($6A$ anchor with $6D_4$ W19-correction,
$r_6=14$). Motivic refinement: \ClaimStatusTheorem{} via R4.D
(Maulik--Toda cosection + KS abelian wall-crossing). K-theoretic
refinement: \ClaimStatusConjectured{} via R4.E (Oberdieck--Shen
orbifold equivariantisation open for global $E$-twist).
$\kBKM(\Phi_N)=1$; $\kcat=0$; $\kfib^{(N)}=14$; $\kch$ via $\Phi$ on
$\mathcal A_{[K3\times E/\Z_N]}$.
