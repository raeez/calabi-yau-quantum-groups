\begin{abstract}
Let $X$ be a smooth projective Calabi--Yau threefold with a normalised
holomorphic volume form, and let $C$ be a homologically trivial algebraic
$1$-cycle on $X$. The chiral volume conjecture predicts that a
level-$N$ chiral partition function attached to the framed
CY-to-chiral output at $d=3$ has exponential growth
\[
  \lim_{N \to \infty} \frac{1}{N}
  \log \bigl|Z_N(\cA_X;C)\bigr|
  =
  \frac{1}{2\pi}\,\bigl|\AJ_X(C)\bigr|.
\]
The statement is conditional. It requires the chain-level CY-A$_3$
construction for the chosen input, an admissible curve specialisation,
a finite trace package defining $Z_N$, and a semiclassical analytic
argument excluding larger saddles. At $d=3$ the chiral algebra is
natively $E_1$; any $E_2$ braiding belongs to the Drinfeld centre of
its representation category, not to $\cA_X$ itself.
\end{abstract}

\section{Calabi--Yau and Abel--Jacobi normalisation}
\label{sec:aj-normalisation}

Fix a smooth projective Calabi--Yau threefold $X$ over $\bC$ and a
holomorphic volume form $\Omega_X\in H^0(X,K_X)$. The Hodge norm is
normalised by
\[
  \|\Omega_X\|^2_H
  :=
  i^9\int_X \Omega_X\wedge \overline{\Omega_X}
  =
  1.
\]
No Tate factor is absorbed into period integrals in this note; all
factors of $2\pi$ are displayed in the conjectural formula.

The Abel--Jacobi map used here is the codimension-$2$ Griffiths map.
Its source is the group of homologically trivial algebraic $1$-cycles,
with arbitrary embedded curves excluded by homological degree. Thus a
single effective curve $C\subset X$ is admissible only if $[C]=0$ in
$H_2(X,\bZ)$; otherwise one must replace it by a homologically trivial
integral combination
$\sum_i n_i C_i$.

\begin{definition}[Intermediate Jacobian and norm]
\label{def:aj-norm}
The intermediate Jacobian is written in homological form
\[
  J^2(X)
  :=
  (F^2H^3(X,\bC))^\vee/H_3(X,\bZ),
\]
where a cycle $\gamma\in H_3(X,\bZ)$ acts on $F^2H^3(X,\bC)$ by
$\omega\mapsto \int_\gamma\omega$.  If $C\in Z^2_{\mathrm{hom}}(X)$
and $\partial\Gamma=C$, then
\[
  \AJ_X^{\mathrm{full}}(C)
  =
  \left[
    \omega\longmapsto \int_\Gamma\omega
  \right]
  \in J^2(X).
\]
The symbol $|\AJ_X(C)|$ denotes the quotient Hodge norm of this class:
the minimal Hodge norm of a representative of
$\AJ_X^{\mathrm{full}}(C)$ in $(F^2H^3(X,\bC))^\vee$.  If the normal
function lies on the line detected by $\Omega_X$, this becomes
\[
  |\AJ_X(C)|
  =
  \left|\int_\Gamma \Omega_X\right|
\]
under the normalisation $\|\Omega_X\|_H=1$.
\end{definition}

The absolute value on the right side of the conjecture is therefore a
real norm on the intermediate Jacobian. It is a complex-structure
invariant of the normal function of the codimension-$2$ cycle, distinct
from the K\"ahler volume \(\int_X\omega^3/3!\) and from the norm of a
Bridgeland central charge.

\section{The \texorpdfstring{$d=3$}{d=3} chiral input}
\label{sec:d3-input}

The CY-to-chiral construction at $d=3$ is the framed object-level
assignment
\[
  \Phi_3^{(\Sigma_2,C_0)}(\cC)
  =
  \operatorname{SpCh}_{\Sigma_2,C_0}\bigl(\Phi^{\mathrm{FA}}_3(\cC)\bigr)
  =
  A_\cC^{(\Sigma_2,C_0)}
  \in E_1\text{-}\mathrm{ChirAlg}(C_0),
\]
under the hypotheses H1--H4 of CY-A$_3$: smoothness, properness,
a negative-cyclic CY$_3$ class, and a chain-level $\mathbb S^3$-framing
with the Costello correction and filtered Hochschild comparison. For a
smooth projective variety $X$, the input category is
$\operatorname{Perf}^{\mathrm{dg}}(X)$, equivalently
$D^b_{\mathrm{dg}}\operatorname{Coh}(X)$.

Compact non-formal threefolds require their own strictification and
analytic-completion witnesses. Arbitrary CY$_3$ morphism functoriality,
a global quantum vertex group, and a Borcherds denominator identity are
additional constructions. The theorem supplies an $E_1$ chiral algebra
on the chosen specialisation locus when the hypotheses are witnessed.

The $E_2$ object at $d=3$ is a centre-level object:
\[
  \mathcal Z\bigl(\mathrm{Rep}^{E_1}(A_\cC)\bigr)
  \simeq
  \mathrm{Rep}^{E_2}\bigl(Z^{\mathrm{der}}_{\mathrm{ch}}(A_\cC)\bigr)
\]
on the Ben-Zvi--Francis--Nadler/Lurie comparison surface. Thus any
braiding, modular tensor structure, or $R$-matrix belongs to the
Drinfeld-centre layer. The partition function below is attached first
to the $E_1$ factorisation algebra and only secondarily to centre data
when such data have been constructed.

\section{Level-\texorpdfstring{$N$}{N} trace datum}
\label{sec:trace}

\begin{definition}[Level-$N$ trace datum]
\label{def:trace-datum}
For the specialised $E_1$ chiral algebra $\cA_X$ on the curve locus
$C_0$, a level-$N$ trace datum consists of:
\begin{enumerate}[label=\textup{(T\arabic*)}]
\item a finite or trace-class $C_0$-graded vector space
  $\cH_N(\cA_X;C)=\bigoplus_m \cH_{N,m}$;
\item an operator $L_0^{(C_0)}$ acting by the grade $m$ on
  $\cH_{N,m}$;
\item a scalar shift $c_{C_0}/24$ when a Virasoro or conformal
  lift is supplied, and $c_{C_0}=0$ otherwise;
\item an Abel--Jacobi evaluation parameter $q_C$ with
  \[
    -\log |q_C|
    =
    \frac{1}{2\pi}\,|\AJ_X(C)|.
  \]
\end{enumerate}
The evaluated level-$N$ partition function is
\[
  Z_N(\cA_X;C)
  :=
  \Tr_{\cH_N(\cA_X;C)}
  q_C^{\,L_0^{(C_0)}-c_{C_0}/24},
\]
whenever the trace converges and is nonzero.
\end{definition}

The integer $N$ is the level or colour. It is the variable tending to
infinity. The Abel--Jacobi parameter $q_C$ is fixed as $N$ varies.
The limit is a level-asymptotic limit with fixed \(q_C\). The absolute
value on the left side of the conjecture is the ordinary complex modulus
of the evaluated trace.

Definition~\ref{def:trace-datum} is part of the required analytic
package. For a merely $E_1$ chiral algebra there is no automatic
Virasoro operator, no automatic central charge, and no automatic
finite-dimensional conformal-block space. The notation $L_0^{(C_0)}$
records the weaker $C_0$-grading; it agrees with the usual $L_0$ only
after a conformal or centre-level lift has been constructed.

\section{The conjecture}
\label{sec:conjecture}

\begin{conjecture}[Chiral volume conjecture]
\label{conj:chiral-volume}
Let $X$ be a smooth projective CY$_3$ with $\Omega_X$ normalised as in
Section~\ref{sec:aj-normalisation}. Let
$C\in Z^2_{\mathrm{hom}}(X)$ be a homologically trivial algebraic
$1$-cycle. Assume:
\begin{enumerate}[label=\textup{(A\arabic*)}]
\item the framed CY-A$_3$ hypotheses H1--H4 are witnessed for
  $\operatorname{Perf}^{\mathrm{dg}}(X)$;
\item an admissible specialisation datum $(\Sigma_2,C_0)$ has been
  chosen, producing the $E_1$ chiral algebra
  $\cA_X=\Phi_3^{(\Sigma_2,C_0)}(\operatorname{Perf}^{\mathrm{dg}}(X))$;
\item level-$N$ trace data in the sense of
  Definition~\ref{def:trace-datum} exist for all sufficiently large $N$.
\end{enumerate}
Then the evaluated traces are eventually nonzero, the limit exists, and
\[
  \lim_{N\to\infty}\frac{1}{N}
  \log\bigl|Z_N(\cA_X;C)\bigr|
  =
  \frac{1}{2\pi}\,|\AJ_X(C)|.
\]
\end{conjecture}

The conjecture is a large-$N$ statement for a family of chiral trace
spaces. CY-A$_3$ supplies only the chiral input. The remaining
mathematical content is the construction of the trace datum and the
proof of the displayed asymptotic.

\section{The coefficient}
\label{sec:coefficient}

Kashaev's volume conjecture is equivalently written
\[
  \frac{1}{N}\log |J_N(K;e^{2\pi i/N})|
  \longrightarrow
  \frac{1}{2\pi}\operatorname{Vol}(S^3\setminus K).
\]
The chiral formula uses the same $N^{-1}$ scaling and the same
coefficient convention. There is no inversion of the Kashaev
normalisation.

The formal identity \(q=e^{2\pi i\tau}\) only defines a modular
parameter; the coefficient requires the large-\(N\) saddle
normalisation. The convention here is:
\[
  -\log |q_C|
  =
  \frac{1}{2\pi}|\AJ_X(C)|.
\]
Equivalently, the holomorphic Chern--Simons action is normalised so
that the conjectural source saddle has real action
$|\AJ_X(C)|/(2\pi)$. A proof must derive this normalisation from the
field theory and show that no competing saddle has larger real part.

\section{Holomorphic Chern--Simons heuristic}
\label{sec:hcs}

The relevant holomorphic Chern--Simons theory lives on the CY$_3$
manifold $X$ itself, whose real dimension is $6$. The total space of
\(K_X\) has complex dimension \(4\) and belongs to a different hCS
problem.

For a $(0,1)$ connection $\mathcal A$ on a holomorphic bundle over $X$,
the classical functional is normalised as
\[
  S_{\mathrm{hCS}}(\mathcal A)
  =
  \frac{1}{2\pi}
  \int_X \Omega_X\wedge
  \Tr\left(
    \frac{1}{2}\mathcal A\wedge\bar\partial\mathcal A
    +
    \frac{1}{3}\mathcal A\wedge\mathcal A\wedge\mathcal A
  \right).
\]
In the heuristic, a curve source enters as a codimension-$2$ defect
current. If the defect equation is solved by a Green current whose
boundary is a 3-chain $\Gamma$ with $\partial\Gamma=C$, the conjectural
on-shell source term pairs with the normal function of $C$:
\[
  \operatorname{Re} S_{\mathrm{cl}}(C)
  =
  \frac{1}{2\pi}|\AJ_X(C)|.
\]
This is a heuristic mechanism. A proof requires compact gauge fixing,
determinant regularisation, a trace-class comparison between hCS states
and the chiral spaces $\cH_N$, and large-$N$ saddle dominance.

\section{Admissible tests}
\label{sec:tests}

\begin{proposition}[Admissibility Failures]
\label{prop:admissibility-failures}
The following examples fall outside Conjecture~\ref{conj:chiral-volume}.
\begin{enumerate}[label=\textup{(\roman*)}]
\item A single fibre $\{\mathrm{pt}\}\times E\subset K3\times E$ has a
  nonzero homology class, so Definition~\ref{def:aj-norm} has no
  Abel--Jacobi input for that single curve.
\item A single rational curve on the quintic normally has nonzero
  degree in $H_2(X,\bZ)$; the admissible object is a homologically
  trivial difference of homologous curves.
\item Local $\mathbb P^2$ is non-compact. It requires a relative or
  meromorphic Abel--Jacobi theory and is outside the compact statement.
\end{enumerate}
\end{proposition}

\begin{proposition}[Valid first tests]
\label{prop:valid-tests}
The first meaningful tests are:
\begin{enumerate}[label=\textup{(\roman*)}]
\item a normal function on the quintic attached to a difference
  $C_1-C_2$ of homologous algebraic curves;
\item a K3-fibred CY$_3$ cycle whose homology class vanishes after
  taking an integral difference of fibres or sections and whose
  $\Phi_3^{(\Sigma_2,C_0)}$ specialisation has been constructed;
\item a toric local model only after replacing
  Definition~\ref{def:aj-norm} by a relative compactification with a
  declared boundary condition.
\end{enumerate}
\end{proposition}

\section{Borcherds constants}
\label{sec:borcherds}

The Borcherds weight identity
\[
  \kappa_{\mathrm{BKM}}(\Phi_N)=c_N(0)/2
\]
belongs to the denominator-product specialisation of the K3-fibred
branch. At $N=1$, Gritsenko's $\Delta_5=\Phi_1$ has
$c_1(0)=10$ and $\kappa_{\mathrm{BKM}}(\Phi_1)=5$.

These constants are Borcherds denominator constants, separate from the
Abel--Jacobi term in Conjecture~\ref{conj:chiral-volume}. They can enter
only through subleading terms or through a separate Borcherds
specialisation of the $E_1$ chiral algebra. In particular,
\[
  \kappa_{\mathrm{BKM}}
  \ne
  \kch+\chi(\cO_{\mathrm{fiber}})
\]
as a universal identity; for $K3\times E$ at $N=1$ the literal
right-hand side in the total-space convention is $0+0$, whereas
$\kappa_{\mathrm{BKM}}=5$.

\section{Conditional consequences}
\label{sec:consequences}

\begin{proposition}[Vanishing Abel--Jacobi class]
\label{prop:vanishing}
Assume Conjecture~\ref{conj:chiral-volume}. If
$\AJ_X(C)=0$, then
\[
  \lim_{N\to\infty}\frac{1}{N}
  \log |Z_N(\cA_X;C)|
  =
  0.
\]
Thus the evaluated partition functions have subexponential growth in
the level variable $N$.
\end{proposition}

\begin{remark}[Torsion and Vanishing]
\label{rem:torsion}
A nonzero torsion point of $J^2(X)$ need not have zero Hodge quotient
norm. Torsion of the Abel--Jacobi class supplies subexponential growth
only after the class itself is zero.
\end{remark}

\section{Proof obligations}
\label{sec:obligations}

The conjecture becomes a theorem only after the following points are
settled for the chosen $X$ and $C$.
\begin{enumerate}[label=\textup{(O\arabic*)}]
\item Construct the chain-level CY-A$_3$ witnesses H1--H4 for
  $\operatorname{Perf}^{\mathrm{dg}}(X)$.
\item Construct an admissible specialisation
  $\operatorname{SpCh}_{\Sigma_2,C_0}$ and prove that the output is the
  intended $E_1$ chiral algebra.
\item Construct the level-$N$ trace spaces $\cH_N$, the $C_0$-grading
  operator $L_0^{(C_0)}$, and the trace-class convergence of $Z_N$.
\item Derive the Abel--Jacobi evaluation parameter and the coefficient
  $(2\pi)^{-1}$ from the hCS normalisation, not from notation.
\item Prove eventual nonvanishing and saddle dominance for
  $Z_N(\cA_X;C)$.
\item Test the formula on homologically trivial cycles, for example
  normal functions on the quintic and constructed K3-fibred
  differences, not on single non-null-homologous curves.
\end{enumerate}
