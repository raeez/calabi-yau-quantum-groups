\title[Non-Abelian hCS on the Quintic]{Non-Abelian Holomorphic Chern--Simons Theory on the Quintic}
\author{Raeez Lorgat}
\date{2026-04-30}

\begin{document}

\begin{abstract}
For a smooth quintic threefold $X_5\subset\PP^4$,
$\int_{X_5}H^3=5$, $\chi_{\mathrm{top}}(X_5)=-200$,
$\kcat(X_5)=0$, and
$\kchHodge(X_5)=1-0+0-1=0$. The BCOV genus-one Euler coefficient is
the separate scalar
\[
  \bBCOV(X_5)=\chi_{\mathrm{top}}(X_5)/24=-200/24=-25/3.
\]
The compact non-abelian holomorphic Chern--Simons problem is a quantum
master equation problem. In complex dimension three its local
cohomological gauge obstruction is descended from a quartic invariant
polynomial, while the quadratic Casimir governs field-strength
renormalisation. A compact chiral output on a curve requires anomaly
cancellation, orientation and framing, descent and sewing, counterterms,
analytic completion, and a comparison map to the $E_1$ chiral
deformation complex.
\end{abstract}

\section*{Holomorphic Chern--Simons data}
\label{sec:hcs-quintic-nonabelian}

Let $X_5 \subset \PP^4$ be a smooth quintic threefold,
let $H=c_1(\cO_{\PP^4}(1))|_{X_5}$, and choose a holomorphic volume
form $\Omega_{X_5}\in H^0(X_5,\Omega^3_{X_5})$. For a Lie algebra
$\fg$ with invariant bilinear form, the classical holomorphic
Chern--Simons field is
\[
  \cA \in \Omega^{0,\bullet}(X_5,\fg)[1],
\]
with action
\[
S_{\mathrm{cl}}[\cA]
 =
\int_{X_5}\Omega_{X_5}\wedge
\left(
  \frac{1}{2}\langle \cA,\bar\partial \cA\rangle
  +\frac{1}{6}\langle \cA,[\cA,\cA]\rangle
\right).
\]
The compact non-abelian problem is the construction of a quantum BV
theory whose Costello--Gwilliam observables descend to the
$d=3$ CY-to-chiral output. At $d=3$ the native output is $E_1$ on a
chosen curve $C\subset X_5$. The $E_2$ operations live on centre,
double, module, or boundary-enhancement layers.

\section*{Chern, Hodge, and BCOV scalars}
\label{sec:hcs-quintic-scalars}

\begin{proposition}[Chern and Hodge data of $X_5$]
\label{prop:hcs-quintic-data}
For a smooth quintic threefold,
\[
  \int_{X_5}H^3=5,\qquad
  c(T_{X_5})=\frac{(1+H)^5}{1+5H}
  =1+10H^2-40H^3.
\]
Consequently
\[
  \chi_{\mathrm{top}}(X_5)=\int_{X_5}c_3(T_{X_5})
  =-40\int_{X_5}H^3=-200.
\]
The relevant Hodge numbers are
\[
  h^{0,0}=h^{0,3}=1,\qquad h^{0,1}=h^{0,2}=0,\qquad
  h^{1,1}=1,\qquad h^{2,1}=101.
\]
Thus
\[
  \kcat(X_5)=\chi(\cO_{X_5})=1-0+0-1=0.
\]
\end{proposition}

\begin{proof}
Adjunction gives
$c(T_{X_5})=c(T_{\PP^4}|_{X_5})/c(\cO_{X_5}(5))=(1+H)^5/(1+5H)$.
Expanding through degree three gives $c_2=10H^2$ and $c_3=-40H^3$.
The hypersurface has degree $5$, so $\int_{X_5}H^3=5$, and
Chern--Gauss--Bonnet gives the Euler number. The Hodge numbers are the
standard Lefschetz and Griffiths calculation for the smooth quintic.
\end{proof}

The two compact scalars occupy different slots:
\[
  \kchHodge(X_5)=\sum_{q=0}^3(-1)^q h^{0,q}(X_5)
  =1-0+0-1=0,
  \qquad
  \bBCOV(X_5)=\frac{\chi_{\mathrm{top}}(X_5)}{24}=-\frac{25}{3}.
\]
The first scalar is the Hodge supertrace of the framed $\Phi_3$
construction once the compact object and completion have been supplied.
The second scalar is the Euler term in the genus-one BCOV determinant
line. The vanishing $H^{0,1}(X_5)=0$ removes the proposed
$H^{0,1}$ source for this number. The absence of a Borcherds
denominator for $X_5$ leaves $\kBKM(X_5)$ undefined.

\section*{Quartic BV anomaly}
\label{sec:hcs-quintic-anomaly}

\begin{proposition}[One-loop obstruction data]
\label{prop:hcs-quintic-anomaly-data}
For holomorphic Chern--Simons theory in complex dimension $3$, the
cohomological holomorphic gauge anomaly is the descent class of a
degree-four invariant polynomial
\[
  I_4^\rho \in \Sym^4(\fg^\vee)^\fg
\]
in the chosen representation or trace convention $\rho$. The quadratic
Casimir coefficient $C_2(\fg)$ governs logarithmic field-strength
renormalisation of the two-point function. Cubic tensors $d^{abc}$
belong to a different anomaly degree.
\end{proposition}

\begin{proof}
The local holomorphic gauge anomaly in complex dimension $d$ is
obtained by descent from an invariant polynomial of degree $d+1$ in
the gauge field strength. Dimension $3$ gives degree $4$. The two-point
graph has a separate coefficient controlled by the quadratic Casimir;
it rescales the kinetic term
$\int\Omega\wedge\langle\cA,\bar\partial\cA\rangle$ after a regulator
choice. The quantum master equation sees the local cohomology class
represented by the degree-four descent datum.
\end{proof}

The expression
\[
  -\,\frac{400N\hbar}{(2\pi)^3}
  \qquad(\fg=\fsl_N)
\]
fails as a compact quintic QME anomaly. It combines the Euler scalar
$\int_{X_5}c_3(T_{X_5})=-200$ with a two-point field-strength
coefficient and omits the quartic descent representative. An exact
non-abelian anomaly number must specify $I_4^\rho$, trace
normalisation, regulator, and descent cocycle.

\section*{Free abelian sector}
\label{sec:hcs-quintic-abelian}

\begin{proposition}[Free abelian holomorphic Chern--Simons]
\label{prop:hcs-quintic-abelian}
For $\fg=\fu(1)$ the interaction term vanishes and the pure
holomorphic Chern--Simons BV theory on $X_5$ is the free Gaussian
theory
\[
  S_{\mathrm{cl}}^{\fu(1)}[\cA]
  =
  \frac{1}{2}\int_{X_5}\Omega_{X_5}\wedge
  \cA\wedge\bar\partial\cA .
\]
Its free Costello--Gwilliam quantisation gives a factorisation algebra
of observables. With CY$_3$ specialisation data $(\Sigma_2,C)$ and the
compact completion, the CY-to-chiral output is an $E_1$ chiral object
on $C$ with Hodge supertrace $\kchHodge(X_5)=0$.
\end{proposition}

\begin{proof}
The bracket on $\fu(1)$ is zero, so the cubic vertex
$\langle\cA,[\cA,\cA]\rangle$ vanishes. The BV Laplacian is the
standard one for a free elliptic complex, and the Costello--Gwilliam
construction quantises free BV theories by the heat-kernel propagator.
The Hodge supertrace is the alternating sum of the $(0,q)$ harmonic
fields on $X_5$, equal to $1-0+0-1=0$.
\end{proof}

The free abelian statement supplies a BV factorisation algebra. The
compact TCFT/BCOV background is extra determinant-line data and fixes
the genus-one scalar $\bBCOV(X_5)=-25/3$.

\section*{Compact non-abelian closure data}
\label{sec:hcs-quintic-required-data}

\begin{theorem}[Necessary data for compact non-abelian closure]
\label{thm:hcs-quintic-necessary-data}
Let $\fg$ be non-abelian and let $E\to X_5$ be the holomorphic
principal bundle or associated vector bundle in which the field
strength $F_\cA$ is evaluated. A compact non-abelian quantum
holomorphic Chern--Simons construction on $X_5$ requires the following
witnesses:
\begin{enumerate}
\item a representative of the local BV anomaly
$\kanom(X_5,\fg,\rho)$ obtained from the quartic descent polynomial
$I_4^\rho$;
\item a QME cancellation datum, either a vanishing proof for this class
or a local counterterm $S_{\mathrm{ct}}$ satisfying
$Q_{\mathrm{BV}}S_{\mathrm{ct}}=-\kanom(X_5,\fg,\rho)$ in the
Costello local functional complex;
\item higher-loop compatibility for the chosen regulator and
renormalisation scheme;
\item an orientation of the determinant line, a CY negative-cyclic
trace, and the compact framing of the bundle sector;
\item descent and sewing data for the Costello--Gwilliam factorisation
algebra on compact opens and for the TCFT/BCOV open--closed background;
\item the CY$_3$ specialisation datum $(\Sigma_2,C,V)$ producing the
$E_1$ chiral output on $C$;
\item an analytic OPE completion and a filtration compatible with the
Costello renormalisation group, compactness of $X_5$, and the chosen
trace convention;
\item a chain map from the local BV functional complex to the
$E_1$ chiral deformation complex, with Hall comparison data if a
CoHA input is used.
\end{enumerate}
These witnesses constitute the closure theorem. In their absence the
compact non-abelian chiral algebra remains an open construction.
\end{theorem}

\begin{proof}
The classical action solves the classical master equation. The quantum
master equation adds local obstruction classes in the Costello complex
of local functionals. Proposition~\ref{prop:hcs-quintic-anomaly-data}
identifies the first cohomological gauge class in dimension $3$ as the
quartic descent class. The scalar $\int c_3(T_{X_5})=-200$ is a
numerical characteristic; a primitive for the QME class must live in the
local functional complex. Compactness adds determinant-line
orientation, sewing, completion, and counterterm choices. The passage
to the CY-to-chiral side adds the curve specialisation and the map from
BV observables to chiral Hochschild cochains. Each item in the list is
therefore a separate construction datum.
\end{proof}

\section*{Counterterms and Green--Schwarz coupling}
\label{sec:hcs-quintic-counterterms}

A local density of the form
\[
  \int_{X_5}\Omega_{X_5}\wedge \mathrm{CS}_3(\cA)
\]
becomes a primitive after the anomaly class has been identified in the
same BV local complex. The descent representative of $I_4^\rho$ and
the trace normalisation fix its coefficient; $\chi_{\mathrm{top}}(X_5)$
alone leaves it undetermined. If the local class
represents nonzero cohomology, this density changes the coupling or
field-strength normalisation while the QME obstruction remains.

A Green--Schwarz construction changes the field content. It introduces
a $B$-field with field strength $H_B$ and a Bianchi identity of the
form
\[
  dH_B =
  \Tr_\rho(F_\cA\wedge F_\cA)-\Tr(R_{T_{X_5}}\wedge R_{T_{X_5}})
\]
or its holomorphic analogue. The Candelas--Horowitz--Strominger--Witten
standard embedding imposes $F_\cA=R_{T_{X_5}}$ in this coupled
heterotic sector. This cancels the corresponding field-strength
polynomial in the coupled theory. Pure non-abelian holomorphic
Chern--Simons theory on $X_5$ still requires its own QME data.

\section*{Hall input and the local terminal chart}
\label{sec:hcs-quintic-hall-input}

The Hall lane starts from oriented critical CoHA data. The Dolbeault BV
complex enters through a comparison map. On the local terminal chart
one has the Schiffmann--Vasserot/Kontsevich--Soibelman equality
\[
  \CoHA(\mathbb C^3)\cong Y^+(\widehat{\mathfrak{gl}}_1),
\]
an $E_1$ associative positive half. The full vertex-algebra comparison
passes through additional layers:
\[
  D(Y^+)=Y^-\otimes Y^0\otimes Y^+,
\]
followed by completion, Hopf pairing, and an evaluation representation
\[
  Y(\widehat{\mathfrak{gl}}_1)\longrightarrow
  \End\!\left(\mathcal W_{1+\infty}[\lambda]\text{-vac}\right).
\]
For the compact quintic, the required bridge is an oriented comparison
\[
  \Theta_{\hCS\to\mathrm{Hall}}^{\mathrm{or}}\colon
  \mathrm{Obs}_{\hCS}^{q}(-,\fg)
  \longrightarrow
  \CoHA_{\mathrm{crit}}^{\mathrm{or}}(-),
\]
compatible with orientation, descent, product, bracket, compact gluing,
and the local $\mathbb C^3$ reduction. The local equality supplies the
terminal Hall chart; the compact quintic construction additionally
requires the comparison map and the global Hall orientation.

\section*{Yukawa obstruction to strictification}
\label{sec:hcs-quintic-yukawa}

\begin{proposition}[Large-volume cubic component]
\label{prop:hcs-quintic-yukawa}
On the mirror large-volume normalisation of the quintic, the transferred
Kodaira--Spencer cubic
\[
  m_3^{\mathrm{KS}}:
  \Sym^3 H^1(T_{X_5})\longrightarrow H^3(\cO_{X_5})\simeq\CC
\]
has a hyperplane component with classical coefficient
\[
  Y_{HHH}^{\mathrm{cl}}(X_5)=\int_{X_5}H^3=5.
\]
Thus a strict uncurved chain-level model for the compact $\Phi_3$
output requires a cyclic null-homotopy for the relevant Yukawa
component. The alternative is a curved cyclic $A_\infty$ or
$L_\infty$ model whose curvature records it.
\end{proposition}

\begin{proof}
The Barannikov--Kontsevich transfer of the Kodaira--Spencer dg Lie
algebra records the genus-zero Yukawa tensor. Mirror symmetry identifies
the large-volume hyperplane direction with $H\in H^2(X_5)$, and the
classical triple product in that direction is
$\int_{X_5}H^3=5$ before Gromov--Witten corrections. The target
$H^3(\cO_{X_5})$ is one-dimensional, so this component is nonzero. A
cyclic null-homotopy for this component is extra data beyond the Hodge
supertrace, the negative-cyclic CY orientation, and the scalar
$\bBCOV(X_5)=-25/3$.
\end{proof}

\section*{Chiral deformation target}
\label{sec:hcs-quintic-chiral-comparison}

When the framed compact object
\[
  A_{X_5}^{(\Sigma_2,C,V)}
  =
  \Phi_3\bigl(D^b_{\mathrm{dg}}\Coh(X_5);\Sigma_2,C,V\bigr)
\]
has been constructed, the non-abelian gauge anomaly has a comparison
target in the deformation complex of this $E_1$ chiral algebra:
\[
  H^\bullet\!\left(
  C^\bullet_{\mathrm{ch},E_1}
  (A_{X_5}^{(\Sigma_2,C,V)},A_{X_5}^{(\Sigma_2,C,V)})
  \right)^\fg .
\]
The comparison map starts in the Costello local functional complex and
lands in the chiral Hochschild complex. The centre
\[
  \cZ\!\left(\Rep^{E_1}(A_{X_5}^{(\Sigma_2,C,V)})\right)
\]
is the natural home for the braided $E_2$ enhancement. The native
$\Phi_3$ output remains the $E_1$ chiral object on $C$.

\section*{Consequences}
\label{sec:hcs-quintic-conclusions}

\begin{enumerate}
\item The stable quintic constants are
$\int_{X_5}H^3=5$, $\chi_{\mathrm{top}}(X_5)=-200$,
$\kcat(X_5)=0$, $\kchHodge(X_5)=1-0+0-1=0$, and
$\bBCOV(X_5)=-25/3$; the large-volume hyperplane Yukawa coefficient is
$Y_{HHH}^{\mathrm{cl}}(X_5)=5$.
\item The compact non-abelian anomaly is a local BV class governed in
complex dimension $3$ by a quartic invariant polynomial in the gauge
field strength. The quadratic Casimir controls field-strength
renormalisation.
\item The pure abelian theory is free and quantises as a BV
factorisation algebra; a completed $\Phi_3$ specialisation is $E_1$ on
a chosen curve.
\item A compact non-abelian quintic chiral algebra requires QME
cancellation, counterterms, orientation and framing, descent and
sewing, TCFT/BCOV background data, analytic completion, and filtration
witnesses.
\item The Hall input begins with the local equality
$\CoHA(\mathbb C^3)\cong Y^+(\widehat{\mathfrak{gl}}_1)$. The full
$\mathcal W_{1+\infty}$ layer requires double, centre, completion, and
evaluation data.
\item The CHSW standard embedding belongs to a coupled
Green--Schwarz/heterotic field-strength cancellation mechanism. Pure
non-abelian holomorphic Chern--Simons theory on $X_5$ has a separate
QME problem.
\end{enumerate}
