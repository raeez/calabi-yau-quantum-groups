\title[Six-Dimensional hCS Observables]{Six-Dimensional Holomorphic Chern-Simons Observables: BV Fields, Local CE Cochains, the Holomorphic E3 Stage, and PBW}
\author{}
\date{}

\begin{document}

\begin{abstract}
Holomorphic Chern-Simons theory on a complex Calabi-Yau threefold is a
BV theory after fixing a holomorphic volume form, a quadratic gauge
coefficient, the odd symplectic pairing, an elliptic gauge fixing,
heat-kernel regularisation, completed local functionals,
renormalisation counterterms, orientation and determinant-line framing,
and factorisation sewing. Its local classical observables are the
continuous Chevalley cochains of the Dolbeault dg Lie algebra
\(\Omega^{0,\bullet}_c(-,\fg)\). On a Stein polydisc in \(\C^3\), with
boundary conditions specified, the generated holomorphic factorisation
algebra is an \(\Ethree^{\mathrm{hol}}\)-enveloping object; an ordinary
\(\Ethree\)-algebra appears only after a locally constant reduction.
Francis and Knudsen PBW gives the associated graded symmetric algebra
on the shifted local Lie object. Quantum hCS exists precisely after the
scale-\(L\) quantum master equation is solved. In complex dimension
three the first nonabelian gauge obstruction is the quartic class
\(I_4^\rho\in\Sym^4(\fg^\vee)^\fg\), while the quadratic Casimir
\(C_2^\rho\) is the coefficient of a two-point field-strength
renormalisation. The \(d=3\) CY-to-chiral specialisation lands in
\(\Eone\)-chiral algebras; braided \(\Etwo\) structure belongs to
centre, double, or module layers.
\end{abstract}

\section{BV Fields}
\label{sec:bv-fields}

\begin{construction}[Holomorphic Chern-Simons datum]
\label{cons:hcs-datum}
Let \(X\) be a complex threefold equipped with a nowhere-vanishing
holomorphic volume form \(\Omega_X\). Let \(\fg\) be a finite-dimensional
complex Lie algebra with a nondegenerate invariant symmetric pairing
\(\langle-,-\rangle\). For an open set \(U\subset X\), put
\[
  \cL_{\hCS}(U)=\Omega^{0,\bullet}_c(U,\fg),
  \qquad
  d=\dbar,
  \qquad
  [\alpha,\beta]=[\alpha\wedge\beta]_\fg .
\]
The BV field complex is
\[
  \cE_U=\cL_{\hCS}(U)[1].
\]
In unshifted Dolbeault degree the \((0,0)\)-component is the ghost, the
\((0,1)\)-component is the physical gauge field, and the \((0,2)\) and
\((0,3)\)-components are BV antifields.
\end{construction}

\begin{definition}[Odd symplectic pairing]
\label{def:bv-pairing}
The BV pairing is
\[
  \omega_{\BV}(\alpha,\beta)
  =
  \int_U\Omega_X\wedge\langle\alpha,\beta\rangle .
\]
It pairs compactly supported \((0,p)\)-forms with \((0,3-p)\)-forms by
Serre duality. After the shift by \([1]\) the pairing has degree \(-1\).
\end{definition}

\begin{definition}[Classical action]
\label{def:classical-action}
For \(a\in\Omega^{0,\bullet}(U,\fg)[1]\), the classical hCS functional is
\[
  S_{\mathrm{cl}}(a)=
  \int_U\Omega_X\wedge
  \left(
    \frac12\langle a,\dbar a\rangle
    +\frac16\langle a,[a,a]\rangle
  \right).
\]
The bracket combines exterior product of Dolbeault forms with the Lie
bracket on \(\fg\).
\end{definition}

\begin{proposition}[Classical master equation]
\label{prop:cme}
The functional \(S_{\mathrm{cl}}\) satisfies
\[
  \{S_{\mathrm{cl}},S_{\mathrm{cl}}\}_{\BV}=0 .
\]
\end{proposition}

\begin{proof}
The Hamiltonian vector field of \(S_{\mathrm{cl}}\) is the Chevalley
differential of the dg Lie algebra
\[
  \bigl(\Omega^{0,\bullet}(U,\fg),\dbar,[-,-]\bigr).
\]
Its square is zero because \(\dbar^2=0\), \(\dbar\) is a derivation of
the wedge-Lie bracket, the Lie bracket satisfies Jacobi, and the pairing
\(\langle-,-\rangle\) is invariant. These four identities are precisely
the kinetic-kinetic, kinetic-cubic, and cubic-cubic terms in the BV
bracket expansion.
\end{proof}

\section{Local CE Observables}
\label{sec:local-ce-observables}

\begin{definition}[Classical observables]
\label{def:classical-observables}
The completed classical observables on \(U\) are the continuous
Chevalley cochains
\[
  \Obs^{\mathrm{cl}}_{\hCS}(U)
  =
  C^\bullet_{\CE,\mathrm{cont}}\bigl(\cL_{\hCS}(U)\bigr)
  =
  \widehat{\Sym}\bigl(\cL_{\hCS}(U)^\vee[-1]\bigr),
\]
with differential \(d_{\dbar}+d_{\CE}\). The completion is by polynomial
degree in continuous linear functionals on compactly supported fields.
\end{definition}

\begin{proposition}[Factorisation maps]
\label{prop:factorisation-maps}
For pairwise disjoint opens \(U_1,\ldots,U_k\subset V\), extension by
zero on compactly supported fields gives a product
\[
  \mu_{U_1,\ldots,U_k;V}\colon
  \bigotimes_{i=1}^k
  \Obs^{\mathrm{cl}}_{\hCS}(U_i)
  \longrightarrow
  \Obs^{\mathrm{cl}}_{\hCS}(V).
\]
These maps make \(U\mapsto\Obs^{\mathrm{cl}}_{\hCS}(U)\) the classical
Costello-Gwilliam factorisation algebra of the local dg Lie algebra
\(\cL_{\hCS}\).
\end{proposition}

\begin{proof}
Extension by zero is covariant for compactly supported fields. Dualising
and completing gives multiplication of polynomial functions on the field
space. The Chevalley differential is a derivation because \(\dbar\) is a
derivation of the bracket and because the bracket on \(\fg\) satisfies
Jacobi. Costello-Gwilliam's construction of observables for a local dg
Lie algebra supplies Weiss descent for elliptic local moduli problems.
\end{proof}

\begin{remark}[Three different layers]
\label{rem:three-layers}
The BV field complex is \(\cE_U=\Omega_c^{0,\bullet}(U,\fg)[1]\). The
classical observable algebra is the continuous CE cochain algebra
\(C^\bullet_{\CE,\mathrm{cont}}(\Omega_c^{0,\bullet}(U,\fg))\). The
factorisation algebra is the cosheaf-like assignment \(U\mapsto
\Obs^{\mathrm{cl}}_{\hCS}(U)\) with products for disjoint opens. A
statement about one of these layers becomes a statement about another
only after applying the stated functor: shift, continuous duality, CE
cochains, or factorisation envelope.
\end{remark}

\section{Holomorphic E3 Stage}
\label{sec:holomorphic-e3-stage}

\begin{definition}[Holomorphic little-disks action]
\label{def:hol-e3}
On a Stein polydisc \(U\subset\C^3\), configurations of \(k\) points act
on \(\Obs^{\mathrm{cl}}_{\hCS}\) through Dolbeault chains on
\(\operatorname{Conf}_k(\C^3)\). The operadic structure is the
holomorphic little-\(3\)-disks structure
\[
  \Ethree^{\mathrm{hol}}\quad\text{in}\quad \Ch(\Dolb).
\]
After choosing translation-invariant Dolbeault contraction data, its
locally constant reduction is an ordinary \(\Ethree\)-algebra.
\end{definition}

\begin{proposition}[Configuration identity]
\label{prop:configuration-identity}
Let \(m_k(z_1,\ldots,z_k)\) be the \(k\)-ary product obtained by placing
compactly supported observables in disjoint polydisks centred at
\((z_1,\ldots,z_k)\in\operatorname{Conf}_k(\C^3)\). On the
Fulton-MacPherson compactification,
\[
  \dbar_{\operatorname{Conf}}m_k
  +
  \sum_{\substack{S\subset\{1,\ldots,k\}\\2\leq |S|<k}}
  \pm\,
  m_{k-|S|+1}
  \bigl(m_{|S|},\operatorname{id},\ldots,\operatorname{id}\bigr)
  =0 .
\]
\end{proposition}

\begin{proof}
The boundary strata of the compactified configuration space are indexed
by collisions of nonempty proper subsets \(S\). Stokes' formula converts
the Dolbeault boundary term into the displayed sum of inserted
operations. The signs are the Koszul signs from Dolbeault degrees.
\end{proof}

\begin{theorem}[Local \(E_3\) PBW]
\label{thm:e3-pbw}
Let \(U\subset\C^3\) be a Stein polydisc, or let \(U=\C^3\) with
polynomial boundary conditions. The holomorphic factorisation algebra
generated by \(\cL_{\hCS}(U)\) is the
\(\Ethree^{\mathrm{hol}}\)-enveloping factorisation algebra. Its
locally constant reduction has a PBW filtration \(F_\bullet\) with
\[
  \operatorname{gr}_F
  \mathrm U^{\mathrm{fact}}_{\Ethree}(\cL_{\hCS})
  \simeq
  \Sym_{\Ch(\Dolb)}\bigl(\cL_{\hCS}[2]\bigr)
\]
as an \(\Ethree\)-factorisation coalgebra. Dually,
\[
  \operatorname{gr}_F\Obs^{\mathrm{cl}}_{\hCS}(U)
  \simeq
  \widehat{\Sym}\bigl(\cL_{\hCS}(U)^\vee[-1]\bigr)
\]
with the linear differential induced by \(\dbar\).
\end{theorem}

\begin{proof}
Francis' \(E_n\) PBW theorem identifies the associated graded of the
\(E_n\)-enveloping filtration with the shifted symmetric coalgebra on
the generating Lie object. Knudsen's higher enveloping theorem gives
the factorisation-algebra form. For \(n=3\) the shift is \([2]\), the
degree of the Browder operation coming from
\[
  H^\bullet(\operatorname{Conf}_2(\R^3);\Q)
  \cong H^\bullet(S^2;\Q).
\]
The Dolbeault local Lie algebra \(\cL_{\hCS}\) is the generating Lie
object for the Costello-Gwilliam hCS observables. The completed
observable statement is obtained by continuous duality and CE cochains.
\end{proof}

\begin{corollary}[Formal point]
\label{cor:formal-point}
At a formal point \(x\in\C^3\), the Dolbeault resolution contracts to
\(\C\). With polynomial boundary conditions,
\[
  \cL_{\hCS,x}\simeq \fg,
  \qquad
  \operatorname{gr}_F
  \mathrm U^{\mathrm{fact}}_{\Ethree}(\cL_{\hCS,x})
  \simeq
  \Sym(\fg[2]).
\]
For a basis \(\{T_a\}\) of \(\fg\), the primitive generators of the
enveloping factorisation algebra are \(t_a\) of cohomological degree
\(2\). The Browder bracket has degree \(1-3=-2\), hence
\[
  [t_a,t_b]_{\mathrm{Br}}
  =
  f_{ab}^{\ \ c}t_c
\]
has degree \(2+2-2=2\). The defining relations are those generated by
\(\dbar\), the Lie bracket, Jacobi, and the invariant pairing entering
the BV action. PBW states that the associated graded has exactly the
symmetric relations displayed above.
\end{corollary}

\begin{remark}[Products and generators]
\label{rem:products-generators}
Secondary monomials such as \(t_at_bt_c\) lie in
\(\Sym^3(\fg[2])\) in the PBW associated graded. They are products of
primitive generators. Axiswise strict products depend on a model of the
little-disks operad; the invariant operation is the coherent
\(\Ethree\) multiplication together with the Browder bracket.
\end{remark}

\section{Heat-Kernel Regularisation}
\label{sec:heat-kernel}

\begin{definition}[Regularised propagator]
\label{def:regularised-propagator}
On \(\C^3\) with the Euclidean Hermitian metric and
\(Q^{\mathrm{GF}}=\dbarstar\), let \(K_t(z,w)\) be the flat Dolbeault
heat kernel. The Costello propagator at scales \(0<\varepsilon<L<\infty\)
is
\[
  P_{\varepsilon,L}(z,w)
  =
  \int_\varepsilon^L(\dbarstar\otimes1)K_t(z,w)\,dt .
\]
\end{definition}

\begin{proposition}[Heat-kernel homotopy]
\label{prop:heat-homotopy}
The regularised propagator is smooth on
\(\C^3\times\C^3\) and satisfies
\[
  \dbar P_{\varepsilon,L}=K_L-K_\varepsilon .
\]
Off the diagonal, the limit \(\varepsilon\to0\), \(L\to\infty\) is the
Bochner-Martinelli kernel
\[
  P_{\mathrm{BM}}(z,w)
  =
  \frac{2!}{(2\pi i)^3}
  \sum_{k=1}^{3}(-1)^{k-1}
  \frac{\overline{z_k-w_k}}{\|z-w\|^6}\,
  d\bar z_1\wedge\cdots\widehat{d\bar z_k}\cdots\wedge d\bar z_3
  \wedge dw_1\wedge dw_2\wedge dw_3 ,
\]
with
\[
  \dbar_w P_{\mathrm{BM}}=\delta_\Delta
\]
as a distribution.
\end{proposition}

\begin{proof}
The scalar flat heat kernel is
\((4\pi t)^{-3}\exp(-\|z-w\|^2/(4t))\). Applying \(\dbarstar\) in one
variable differentiates the Gaussian and supplies the numerator
\(\overline{z_k-w_k}\). The heat equation
\(\partial_tK_t=-\Delta_{\dbar}K_t\), with
\(\Delta_{\dbar}=\dbar\dbarstar+\dbarstar\dbar\), gives the telescoping
identity after integration over \([\varepsilon,L]\). Removing both
cutoffs off the diagonal gives the standard Bochner-Martinelli
\(\dbar\)-homotopy kernel.
\end{proof}

\section{Quantum Master Equation}
\label{sec:qme}

\begin{hypothesis}[Quantum hCS package]
\label{hyp:quantum-hcs-package}
A quantum hCS factorisation algebra on a Calabi-Yau threefold consists
of the following data.
\begin{enumerate}
\item A holomorphic volume form \(\Omega_X\), an orientation, and a
determinant-line framing compatible with the BV pairing.
\item A quadratic gauge Lie algebra or dg Lie algebra
\((\fg,\langle-,-\rangle)\), and a representation
\(\rho:\fg\to\End(V)\) whenever trace invariants are used.
\item An elliptic Costello gauge fixing \(Q^{\mathrm{GF}}\), heat kernels
\(K_t\), regularised propagators \(P_{\varepsilon,L}\), and an
RG-compatible counterterm scheme.
\item Completed spaces of local functionals and observables on which
\(\Delta_L\), \(\{-,-\}_L\), and renormalisation-group flow are defined.
\item Sewing or factorisation descent from local polydisks to \(X\).
\item A trivialisation of the anomaly class \(\kanom(X,\fg,\rho)\) by
vanishing in local BV cohomology, by an explicit local counterterm, or
by an open-closed Green-Schwarz term.
\end{enumerate}
\end{hypothesis}

\begin{definition}[Scale-\(L\) quantum observables]
\label{def:quantum-observables}
Given Hypothesis~\ref{hyp:quantum-hcs-package} and effective
interactions \(I[L]\), the scale-\(L\) QME is
\[
  QI[L]+\frac12\{I[L],I[L]\}_L+\hbar\Delta_L I[L]=0 .
\]
When this equation holds, the quantum observables are
\[
  \Obs^q_{\hCS,L}(U)
  =
  \left(
    \Obs^{\mathrm{cl}}_{\hCS}(U)[[\hbar]],
    d_{\Obs}^{\mathrm{cl}}+\{I[L],-\}_L+\hbar\Delta_L
  \right).
\]
\end{definition}

\begin{proposition}[Abelian theory]
\label{prop:abelian-theory}
For \(\fg=\C\), the cubic interaction vanishes and the perturbative
theory is Gaussian. The two-point function is
\[
  \langle a_{0,1}(z)a^\ast_{0,2}(w)\rangle_\hbar
  =
  \hbar\,P_{\mathrm{BM}}(z,w).
\]
Connected higher correlation functions are obtained by Wick pairings
of the quadratic kernel alone.
\end{proposition}

\begin{proof}
The Lie bracket on \(\C\) is zero, so \(S_{\mathrm{cl}}\) is quadratic.
Costello's BV quantisation of a free elliptic theory gives the star
product
\[
  \phi\star\psi
  =
  \mu\circ\exp(\hbar P\,\partial_\phi\otimes\partial_\psi)
  (\phi\otimes\psi),
\]
with \(P\) the regularised BV kernel. The displayed two-point function
is the corresponding Wick contraction.
\end{proof}

\section{Quartic Anomaly and Two-Point Renormalisation}
\label{sec:quartic-anomaly}

\begin{definition}[Quartic invariant]
\label{def:quartic-invariant}
For a representation \(\rho:\fg\to\End(V)\), define
\[
  I_4^\rho(x_1,x_2,x_3,x_4)
  =
  \frac1{4!}\sum_{\sigma\in S_4}
  \Tr_V\!\left(
    \rho(x_{\sigma(1)})\rho(x_{\sigma(2)})
    \rho(x_{\sigma(3)})\rho(x_{\sigma(4)})
  \right).
\]
Then \(I_4^\rho\in\Sym^4(\fg^\vee)^\fg\). The adjoint trace gives
\(I_4^{\ad}\).
\end{definition}

\begin{theorem}[One-loop QME obstruction]
\label{thm:quartic-obstruction}
In complex dimension three, the first nonabelian gauge obstruction to
the QME is represented in the local Chevalley complex by
\[
  \Theta^\rho_U(\alpha_0,\alpha_1,\alpha_2,\alpha_3)
  =
  \int_U
  \left[
  I_4^\rho\bigl(
    \alpha_0,\partial\alpha_1,\partial\alpha_2,\partial\alpha_3
  \bigr)
  \right]_{(3,3)},
  \qquad
  \alpha_i\in\Omega^{0,\bullet}_c(U,\fg).
\]
Its obstruction class is
\[
  \kanom(X,\fg,\rho)
  =
  [\,\hbar\,\Theta^\rho\,]
  \in
  H^1_{\mathrm{loc}}
  \bigl(\Omega^{0,\bullet}(X,\fg)[1],\cO_{\mathrm{loc}}\bigr).
\]
For \(\rho=\ad\), this is the Costello-Li descent class represented in
ghost-field notation by \(\int_X\Tr_{\ad}(cF_A^3)\), equivalently the
descent of \(\Tr_{\ad}(F_A^4)\).
\end{theorem}

\begin{proof}
The one-loop wheel with four external gauge insertions is the first
degree-one local functional in complex dimension three. Its Lie factor
is the symmetrised trace of four representation matrices, hence
\(I_4^\rho\). Its holomorphic factor contains three \(\partial\)
derivatives; extracting the \((3,3)\)-component produces a local
density. Costello's obstruction theory places this degree-one local
functional in the obstruction complex for the scale-\(L\) QME.
\end{proof}

\begin{definition}[Quadratic Casimir]
\label{def:quadratic-casimir}
Let \(\{T_a\}\) be a basis of \(\fg\), and let \(\{T^a\}\) be the dual
basis for the chosen invariant pairing. The representation-theoretic
quadratic Casimir coefficient is
\[
  C_2^\rho=\sum_a\rho(T_a)\rho(T^a).
\]
For the adjoint representation of a simple Lie algebra in the usual
normalisation, \(C_2^{\ad}=2h^\vee\).
\end{definition}

\begin{proposition}[Two-point field-strength counterterm]
\label{prop:field-strength-counterterm}
With the flat heat-kernel normalisation of
Proposition~\ref{prop:heat-homotopy}, the two-point graph contributes
the degree-zero local counterterm
\[
  S_{\mathrm{ct}}^{(2)}
  =
  -\hbar\,\frac{C_2^\rho}{(4\pi)^3}
  \log(L/\varepsilon)
  \int_X\Omega_X\wedge\Tr_\rho(a\,\dbar a).
\]
It renormalises the kinetic pairing and the field. The degree-one QME
obstruction is the quartic class of Theorem~\ref{thm:quartic-obstruction}.
\end{proposition}

\begin{proof}
Closing two legs in the cubic interaction gives the colour contraction
\(\sum_a\rho(T_a)\rho(T^a)=C_2^\rho\). The contraction leaves two
external fields with the tensor type of the kinetic functional
\(\int_X\Omega_X\wedge\Tr_\rho(a\dbar a)\). The short-distance
six-real-dimensional heat-kernel integral gives the logarithmic factor.
The resulting local functional has cohomological degree zero and arity
two. The quartic wheel has degree one and arity four.
\end{proof}

\begin{corollary}[Quantisation criterion through one loop]
\label{cor:one-loop-criterion}
If \(\fg\) is abelian, then \(\Theta^\rho=0\). More generally, if the
class \(\kanom(X,\fg,\rho)\) vanishes in Costello local BV cohomology
after the specified counterterms or closed-sector terms are included,
then the effective interactions can be chosen to satisfy the QME through
one loop. The higher-loop extension is obtained by the same obstruction
induction in the completed local functional complex.
\end{corollary}

\section{Quantum E3 Object}
\label{sec:quantum-e3-object}

\begin{proposition}[Filtered quantum deformation]
\label{prop:filtered-quantum-deformation}
Assume Hypothesis~\ref{hyp:quantum-hcs-package} and a solution of the
QME. On a Stein polydisc in \(\C^3\), the renormalised observables form
a quantum factorisation algebra over \(\C[[\hbar]]\). Its
\(\hbar\)-adic associated graded is the classical factorisation algebra
of Section~\ref{sec:local-ce-observables}. After a locally constant
reduction, the stalk is a filtered \(\Ethree\)-algebra whose classical
associated graded is governed by Theorem~\ref{thm:e3-pbw}.
\end{proposition}

\begin{proof}
Costello renormalisation constructs the scale-\(L\) bracket,
BV Laplacian, and renormalised factorisation maps from the heat-kernel
propagator and counterterms. The QME is exactly the compatibility
condition for the differential
\[
  d_{\Obs}^{\mathrm{cl}}+\{I[L],-\}_L+\hbar\Delta_L
\]
with these products. Setting \(\hbar=0\) recovers the classical
observable factorisation algebra, and the PBW filtration is inherited
from the classical local Lie input.
\end{proof}

\begin{remark}[Formal-point enveloping object]
\label{rem:ue3-records}
The expression \(U_{\Ethree}(\fg[2])\) records the locally constant
formal-point enveloping object after the Dolbeault contraction
\(\cL_{\hCS,x}\simeq\fg\). It is a statement about the enveloping
factorisation algebra generated by primitives. The completed observable
algebra is the continuous dual CE presentation. Passing between them
requires augmentation, completion, conilpotence, and duality hypotheses.
\end{remark}

\section{CY-to-Chiral Specialisation}
\label{sec:cy-to-chiral-specialisation}

\begin{proposition}[\(d=3\) output level]
\label{prop:d3-output-level}
Let \(\cC\) be a CY\(_3\) category on a witnessed hCS or Hochschild
comparison locus, and let \((\Sigma_2,C)\) be an admissible
specialisation datum. The two-stage construction is
\[
  \Phi_3^{(\Sigma_2,C)}(\cC)
  =
  \SpCh_{\Sigma_2,C}\bigl(\PhiFA_3(\cC)\bigr)
  =
  \left(\int_{\Sigma_2}^{\mathrm{hol}}\PhiFA_3(\cC)\right)\Big|_C
  \in
  \Alg^{\mathrm{ch}}_{\Eone}(C).
\]
The \(\Ethree^{\mathrm{hol}}\) structure belongs to the stage-one
factorisation algebra. The braided \(\Etwo\) structure belongs to
centre, double, or module layers, for example
\[
  \cZ\bigl(\Rep^{\Eone}(A_C)\bigr)
  \simeq
  \Rep^{\Etwo}\bigl(\Zchder(A_C)\bigr)
\]
on the verified Drinfeld-centre locus.
\end{proposition}

\begin{proof}
Stage one produces an \(\Ethree\)-holomorphic factorisation algebra only
after the formality, framing, QME, completion, and comparison witnesses
have been fixed. Factorisation homology over the complex surface
\(\Sigma_2\) lowers the complex operadic level by two. Restriction to
the reference curve \(C\) leaves an \(\Eone\)-chiral algebra. The
Ben-Zvi-Francis-Nadler centre mechanism recovers \(\Etwo\)-braiding on
the representation centre when its hypotheses are satisfied.
\end{proof}

\begin{remark}[\(\C^3\) Hall comparison]
\label{rem:c3-hall-comparison}
For the toric Hall algebra of \(\C^3\),
\[
  \CoHA(\C^3)\simeq \Yp(\widehat{\mathfrak{gl}}_1),
\]
the positive half in the Kontsevich-Soibelman and
Schiffmann-Vasserot normalisation. The full affine Yangian is obtained
by the Drinfeld double
\[
  D\bigl(\Yp(\widehat{\mathfrak{gl}}_1)\bigr)
  =
  \widehat Y(\widehat{\mathfrak{gl}}_1).
\]
The vertex algebra \(\Woneinf\) appears after double or centre passage,
completion, and evaluation on the relevant Fock or vacuum module. PBW
for hCS observables and PBW for \(\Yp(\widehat{\mathfrak{gl}}_1)\)
agree on a constructed toric comparison locus.
\end{remark}

\section{Separated Algebraic Objects}
\label{sec:separated-objects}

\begin{remark}[Fields and Hochschild polyvectors]
\label{rem:fields-polyvectors}
The hCS fields are Dolbeault forms
\(\Omega^{0,\bullet}(X,\operatorname{ad}P)[1]\). Their BV antifields are
the complementary Dolbeault components paired by
\(\int_X\Omega_X\wedge\Tr_\rho(-\, -)\). Hochschild polyvectors of a CY
category form a different complex:
\[
  \HH^\bullet(\Perf(\C^3))
  \simeq
  \C[x,y,z]\otimes
  \bigwedge\nolimits^\bullet
  \langle\partial_x,\partial_y,\partial_z\rangle .
\]
This complex controls intrinsic categorical deformations and the
derived centre. It enters hCS only through an explicit comparison map or
through a chosen framing object.
\end{remark}

\begin{remark}[Five categorical objects]
\label{rem:five-categorical-objects}
For an \(\Eone\)-chiral algebra \(A\), the objects
\[
  A,\qquad B(A),\qquad A^i,\qquad A^!,\qquad \Zchder(A)
\]
have different definitions. The equivalence \(\Omega B(A)\simeq A\) is
bar-cobar inversion. The object \(A^i\) is the Koszul dual coalgebra.
The object \(A^!\) is the Verdier or Koszul dual algebra, depending on
the duality context. The bulk or braided layer at \(d\geq3\) is
\(\Zchder(A)\), the derived chiral centre on the constructed loci.
\end{remark}
