\section{The CY$_4$ Curve Shadow of $\Phi_4$}
\label{sec:cy4-kapranov-output}

\paragraph{The curve-level obstruction.}
Let \(X\) be a compact projective Calabi--Yau fourfold and
\(\cC=D^b\Coh(X)\). The native stage of the CY-to-chiral construction is
the conditional CY-A\(_4\) object
\[
  \PhiFA_4(\cC)\in \Efour\text{-}\HolFA(X),
\]
assembled from the degree \(1-4=-3\) Gerstenhaber bracket on
\(\HH^\bullet(\cC)\), Kontsevich--Tamarkin formality, and
Costello--Gwilliam--Li holomorphic locality. The completed curve shadow
requires an admissible specialisation datum \((\Sigma_3,C)\):
\[
  \Phi_4^{(\Sigma_3,C)}(\cC)
  =
  \SpCh_{\Sigma_3,C}\bigl(\PhiFA_4(\cC)\bigr).
\]
The Hodge computations below are unconditional. Their interpretation as
\(\kappa_{\mathrm{ch}}\) belongs to the constructed CY-A\(_4\) locus of
Theorem~\ref{thm:kappa-hodge-supertrace-identification}. The full
chain-level CY-A\(_4\) functor and its non-product specialisations remain
conditional on the framing, formality, and completion witnesses named in
Theorem~\ref{thm:phi-two-stage-factorisation}.

\begin{proposition}[Native curve output at \(d=4\)]
\label{prop:cy4-native-e1-output}
On every constructed CY-A\(_4\) locus and for every admissible
\((\Sigma_3,C)\), the curve shadow
\(\Phi_4^{(\Sigma_3,C)}(\cC)\) is an \(\Eone\)-chiral algebra on
\(\Ran(C)\). The \(\Etwo\)-braided object attached to it is
\[
  \cZ\bigl(\RepE(\Phi_4^{(\Sigma_3,C)}(\cC))\bigr)
  \simeq
  \mathrm{Rep}^{\Etwo}\bigl(\Zder(\Phi_4^{(\Sigma_3,C)}(\cC))\bigr),
\]
the Drinfeld-centre or derived-centre layer.
\end{proposition}

\begin{proof}
The stage-one object is \(\Efour\)-factorisation on the fourfold. Holomorphic
factorisation homology over the complex threefold \(\Sigma_3\) reduces the
operadic level by three:
\[
  \int_{\Sigma_3}^{\mathrm{hol}}:
  \Efour\text{-}\HolFA(X)\longrightarrow
  \Eone\text{-}\HolFA(C).
\]
This is Proposition~\ref{prop:native-en-level} and the \(d=4\) row of
Proposition~\ref{prop:ptvv-shift-law}. The second multiplication is supplied
by the centre construction, by the Deligne--Ben-Zvi--Francis--Nadler
centre theorem quoted in Remark~\ref{rem:e2-on-drinfeld-centre}.
\end{proof}

\paragraph{Product fourfolds.}
For \(X=X_1\times X_2\) with \(\dim_{\C}X_1+\dim_{\C}X_2=4\), the
product statement is proved only under the split hypotheses of
Proposition~\ref{prop:phi-d-kunneth-split-product}: split framing,
factorwise anomaly cancellation, and product-stable Bridgeland slicing.
Under these hypotheses,
\[
  \Phi_4(X_1\times X_2)
  \simeq
  \Phi_{d_1}(X_1)\boxtimes\Phi_{d_2}(X_2)
\]
as an \(\Eone\)-chiral algebra after curve specialisation. Dunn additivity
acts at the stage-one factorisation level; the final curve object remains
\(\Eone\).

\begin{theorem}[\(K3\times K3\)]
\label{thm:cy4-k3-k3-product}
Let \(X=S_1\times S_2\) be a product of projective \(K3\) surfaces and
\(\cC=D^b\Coh(X)\). On the product-stable locus
\(\StabOp(S_1)\times\StabOp(S_2)\subset\StabOp(X)\), the \(d=4\) curve
shadow is
\[
  \Phi_4(X)
  \simeq
  \cH_{\mathrm{Muk}}(S_1)\boxtimes\cH_{\mathrm{Muk}}(S_2)
\]
as an \(\Eone\)-chiral algebra on \(\Ran(C)\). Each factor
\(\cH_{\mathrm{Muk}}(S_i)=\Phi_2(S_i)\) is the Mukai--Heisenberg chiral
algebra of Theorem~\ref{thm:phi-k3-explicit}.
\end{theorem}

\begin{proof}
The factors are strict compact K\"ahler CY\(_2\) manifolds with Ricci-flat
metrics by Yau's theorem. The \(d\leq2\) BCOV anomaly is absent, so the
split product hypotheses hold on the stated stability locus. Applying
Proposition~\ref{prop:phi-d-kunneth-split-product} with \(d_1=d_2=2\)
gives the displayed external tensor product. Proposition
\ref{prop:cy4-native-e1-output} fixes the curve-level operad as \(\Eone\).
\end{proof}

\begin{remark}[Four \(K3\times K3\) scalars]
\label{rem:cy4-k3-k3-scalars}
The product carries four construction-distinct scalar readings:
\[
\begin{aligned}
  \kappa_{\mathrm{ch}}(K3\times K3)
    &= \Xi(K3\times K3)=2\cdot2=4,\\
  \kappa_{\mathrm{cat}}(K3\times K3)
    &= \chi(\cO_{K3\times K3})=4,\\
  \kappa_{\mathrm{ch}}^{\mathrm{Heis}}(K3\times K3)
    &= 24+24=48,\\
  \kappa_{\mathrm{fiber}}(K3\times K3)
    &= \dim_{\C}H^\bullet(K3\times K3,\C)=24\cdot24=576.
\end{aligned}
\]
Here
\[
  h^{0,\bullet}(K3\times K3)=(1,0,2,0,1),
\]
so the Hodge supertrace is \(1+2+1=4\). The Heisenberg scalar is
the sum of the two Mukai-lattice ranks; the fibre scalar is
K\"unneth-multiplicative on total cohomology. A Borcherds weight for a
product fourfold would require a separately constructed Borcherds lift.
\end{remark}

\paragraph{Irreducible fourfolds.}
Let \(X_6\subset\bP^5\) be the Fermat sextic. It is a smooth compact
CY\(_4\) with
\[
  (h^{0,0},h^{0,1},h^{0,2},h^{0,3},h^{0,4})=(1,0,0,0,1).
\]
Klemm--Pandharipande give the fourfold data
\[
  h^{1,1}=1,\qquad h^{3,1}=426,\qquad h^{2,2}=1752,\qquad
  h^{2,2}_{\mathrm{prim}}=1750,\qquad \chi_{\mathrm{top}}=2610.
\]
Adjunction gives \(p_1(T_{X_6})=-30H^2\), hence
\[
  \ell(X_6,H^2)=\int_{X_6}p_1(T_{X_6})H^2=-180,\qquad
  N(X_6)=\int_{X_6}p_1(T_{X_6})^2=5400.
\]

\begin{construction}[Conditional \(\bP^1\)-family for a CY\(_4\)]
\label{constr:cy4-p1-family-clean}
Assume the CY-A\(_4\) stage-one construction of \(\PhiFA_4\), the
BCOV Maurer--Cartan model of Construction~\ref{constr:phi-4-p1-family},
and the required analytic completion. For a compact projective CY\(_4\)
\(X\), the model assigns to
\([\sigma_3:\sigma_4]\in\bP^1\) an \(\Eone\)-chiral algebra
\[
  A_X^{(\sigma_3,\sigma_4)}
  =
  U^{\mathrm{ch}}_{\Eone}\bigl(L_X^{(\sigma_3,\sigma_4)}\bigr),
\]
where \(L_X^{(\sigma_3,\sigma_4)}\) is obtained from the HKR polyvector
dg Lie algebra \(\PV^\bullet(X)[u]\), negative cyclic refinement, and
the CY\(_4\) BCOV twist
\[
  \bar\partial\mu+\frac12[\mu,\mu]
  +\sigma_3\wedge\mu^2+\sigma_4\wedge\mu^3=0.
\]
The parameter \(\sigma_3\) lies in \(H^{3,1}(X)\). The parameter
\(\sigma_4\) lies in \(H^{2,2}_{\mathrm{prim}}(X)\) and records the
primitive middle-dimensional BCOV direction.
\end{construction}

\begin{remark}[Sextic numerical consequences]
\label{rem:cy4-sextic-consequences}
For \(X_6\), the unconditional Hodge supertrace is
\[
  \Xi(X_6)=1-0+0-0+1=2.
\]
On the conditional family model, Example~\ref{ex:sextic-cy4-associator}
computes the Mukai-style Hodge sum
\[
  1+426+1752+426+1=2606,
\]
then applies the Pontryagin shift \(\ell(X_6,H^2)/12=-15\). The effective
central charge at the \(\sigma_4\to0\) limit is therefore \(2591\), while
the \(\sigma_3\to0\) limit adds
\(\frac12 h^{2,2}_{\mathrm{prim}}=875\), giving \(3466\). These central
charges are computed consequences of the conditional family model, separate
from \(\kappa_{\mathrm{ch}}(X_6)=2\).
\end{remark}

\begin{proposition}[The sextic lies outside \(\RWk{4}\)]
\label{prop:cy4-sextic-outside-rw4}
The Fermat sextic \(X_6\) carries the ordinary Kapranov \(L_\infty\)
structure on \(T_{X_6}[-1]\) determined by the Atiyah class, and lies
outside the weight-\(2\) Rozansky--Witten locus of
Theorem~\ref{thm:weight-k-kapranov-obstruction}.
\end{proposition}

\begin{proof}
The weight-\(2\) Rozansky--Witten condition uses a holomorphic
symplectic form \(\sigma\in H^0(X,\Omega_X^{2,0})\) with the stated rank
condition. A strict CY\(_4\) of sextic type has \(h^{0,2}(X_6)=0\).
Thus the required \((2,0)\)-form is absent. The Atiyah class still
defines Kapranov brackets on \(T_{X_6}[-1]\), but the extra
holomorphic-symplectic contraction that defines \(\RWk{4}\) belongs to
hyperk\"ahler fourfolds such as \(K3^{[2]}\) and Beauville--Donagi
fourfolds.
\end{proof}

\paragraph{\(\kappa_{\mathrm{ch}}\) at \(d=4\).}
For compact CY\(_4\), the Hodge supertrace is
\[
  \Xi(X)=\sum_{q=0}^4(-1)^q h^{0,q}(X).
\]
On the constructed CY-A\(_4\) locus this is
\(\kappa_{\mathrm{ch}}(\cA_X)\) by
Theorem~\ref{thm:kappa-hodge-supertrace-identification}. The following
values are direct Hodge computations:
\[
\begin{array}{lcc}
\toprule
X & h^{0,\bullet}(X) & \Xi(X) \\
\midrule
X_6\subset\bP^5\ \text{sextic} & (1,0,0,0,1) & 2\\
X_8\ \text{octic double cover} & (1,0,149,0,1) & 151\\
X_{10}\subset\bP(1^5,5)\ \text{decic} & (1,0,0,0,1) & 2\\
K3^{[2]} & (1,0,1,0,1) & 3\\
F(Y)\ \text{Beauville--Donagi} & (1,0,1,0,1) & 3\\
K3\times K3 & (1,0,2,0,1) & 4\\
\bottomrule
\end{array}
\]
For strict compact CY\(_4\) with holonomy \(\mathrm{SU}(4)\), the only
nonzero entries in the \(h^{0,\bullet}\) column are \(h^{0,0}=h^{0,4}=1\),
so \(\Xi(X)=2\). Hyperk\"ahler fourfolds have the additional
holomorphic-symplectic class in \(H^{0,2}\), giving \(\Xi=3\). The octic
double cover is a separate middle-column example with \(h^{0,2}=149\).

\begin{remark}[BCOV \(F_2\) scope]
\label{rem:cy4-f2-scope-clean}
Theorem~\ref{thm:bcov-f2-zero-correction-d4} states that the anomalous
\(F_2\)-correction to \(\kappa_{\mathrm{ch}}\) vanishes at \(d=4\). The
constant-map piece of \(F_2\) is classical and already absorbed into the
Hodge-supertrace channel. Thus the \(d=4\) \(\kappa_{\mathrm{ch}}\) values
above are Hodge-supertrace values on the constructed locus, with no
additional genus-\(2\) quantum correction.
\end{remark}

\paragraph{The \(p_1\)-twist.}
The effective pointwise section of the conditional family is controlled by
the first Pontryagin class. The criterion of
Theorem~\ref{thm:phi-4-eff-p1-vanishing} is conditional on the
identification of the obstruction class with \(p_1(T_X)\):
\[
  \Phi_4^{\mathrm{eff}}(X)\ \text{exists as a single framed fibre}
  \quad\Longleftrightarrow\quad
  [p_1(T_X)]=0\in H^4(X;\Z).
\]
When \([p_1(T_X)]\) is nonzero, the family carries integer levels
\[
  \ell(X,\alpha)=\int_X p_1(T_X)\cup\alpha,\qquad \alpha\in H^4(X;\Z).
\]
The weaker number \(N(X)=\int_Xp_1(T_X)^2\) controls the associator
stratum in Conjecture~\ref{conj:cy4-p1-family-associator-sextic}; it is a
different invariant from the cohomology class \([p_1(T_X)]\).

\paragraph{Dimensional grammar.}
The dimension table has three layers:
\[
\begin{array}{ccccc}
\toprule
d & \text{stage-one object} & \text{curve shadow} & \text{centre layer} &
\kappa_{\mathrm{ch}}\text{ mechanism}\\
\midrule
1 & E_1\text{-HolFA} & E_\infty\text{-chiral} & \text{same} &
h^{1,0}\text{ lattice}\\
2 & E_2\text{-HolFA} & \Etwo\text{-chiral} & \text{same} &
\sum_q(-1)^q h^{0,q}\\
3 & E_3\text{-HolFA} & \Eone\text{-chiral} &
\Etwo\text{ on }\cZ(\RepE(A)) &
\text{chain-level shadow}\\
4 & E_4\text{-HolFA} & \Eone\text{-chiral} &
\Etwo\text{ on }\cZ(\RepE(A)) &
\sum_q(-1)^q h^{0,q}\\
5 & E_5\text{-HolFA} & \Eone\text{-chiral} &
\Etwo\text{ on }\cZ(\RepE(A)) &
0\text{ on compact strict CY}\\
\bottomrule
\end{array}
\]
For every \(d\geq3\), the native curve output is \(\Eone\)-chiral. Higher
operadic structures at these dimensions must be read either before
specialisation, or on the derived centre, Drinfeld double, module category,
or a named layer.

\begin{conjecture}[Even-dimensional Kapranov pattern]
\label{conj:cy4-higher-kapranov-clean}
Let \(d=2k\geq4\). The stage-one construction \(\PhiFA_{2k}\) is expected
to carry the \(E_{2k}\)-factorisation structure forced by the Hochschild
bracket degree and the PTVV shift. After specialisation along an admissible
\((2k-1)\)-cycle, the native curve shadow remains \(\Eone\)-chiral. On
Rozansky--Witten loci \(\RWk{2k}\), additional \(E_{k-1}\)-type structure
may appear on centre, module, or factorisation-homology layers once the
weight-\(k\) Kapranov brackets and the required completion data are
constructed. Away from those loci, the expected output is a
\((\bP^1)^{k-1}\)-family of \(\Eone\)-chiral algebras with twist data
controlled by characteristic classes.
\end{conjecture}

\paragraph{Local and automorphic boundaries.}
The local threefold identity
\[
  \mathrm{CoHA}(\C^3)=Y^+(\widehat{\mathfrak{gl}}_1)
\]
belongs to the CY\(_3\) positive-half Hall branch. The full
\(\mathcal{W}_{1+\infty}\) object appears only after Drinfeld double,
centre, completion, and representation data. This identity supplies a
warning against importing \(\C^3\) or \(\mathcal{W}\)-algebra conclusions
into CY\(_4\) without an explicit construction.

For \(K3\times E\), the standard constants remain separated:
\[
\begin{aligned}
 \kappa_{\mathrm{cat}}(K3\times E)&=0,\\
 \kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(K3\times E)&=0,\\
 \kappa_{\mathrm{ch}}^{\mathrm{Heis}}(K3\times E)&=3,\\
 \kappa_{\mathrm{BKM}}(\Delta_5)&=c_1(0)/2=5,\\
 \kappa_{\mathrm{fiber}}(K3)&=24.
\end{aligned}
\]
The Hall--Borcherds double branch and the Mukai Yangian branch are
separate constructions. Borcherds weights are read from
\(\kappa_{\mathrm{BKM}}=c(0)/2\) for the chosen automorphic input.
