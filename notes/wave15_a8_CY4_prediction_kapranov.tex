\section{$\Phi_4$ at $d=4$: the Kapranov prediction}
\label{wn:sec:wave15-a8-CY4-kapranov}

\paragraph{The obstruction that fixes the output.}
Fix a compact projective Calabi--Yau fourfold $X$, $\cC = D^b\Coh(X)$. The
Gerstenhaber bracket on $\HH^\bullet(\cC)$ sits in cohomological degree
$1 - d = -3$. Under Pantev--To\"en--Vaqui\'e--Vezzosi, $\Perf(X)$ carries
a $(-4)$-shifted symplectic form, and Calaque--PTVV deformation
quantisation produces a $(-3)$-Poisson bracket --- equivalently, an
$E_4$-algebra structure on $\HH^\bullet(\cC)$ via Kontsevich--Tamarkin
formality with $\mathrm{GRT}_1(\Q)$-ambiguity. The problem is not whether
$\Phi_4$ lands in some $E_n$-chiral target; Dunn--Lurie provides the
factorisations $E_4 \simeq \Eone \otimes E_3 \simeq \Etwo \otimes \Etwo$
and one must specialise along a $3$-cycle. The problem is to determine
which $E_n$ emerges after the specialisation $\mathrm{Sp}_{\Sigma_3, C}$
collapses the native $E_4$-holomorphic factorisation algebra
$\Phi_4^{\mathrm{FA}}(\cC) \in E_4\text{-}\mathrm{HolFA}(X)$ onto the
reference curve $C$, and in which category the resulting chiral algebra
lives.

\begin{theorem}[$\Phi_4$-output scope for product CY$_4$]
\label{wn:thm:wave15-a8-phi4-product-scope}
\ClaimStatusProvedHere\ Let $X = X_1 \times X_2$ be a product CY$_4$
with $\dim_\C X_1 + \dim_\C X_2 = 4$, each factor a strict compact
K\"ahler Calabi--Yau admitting an independent Ricci-flat metric
\textup{(}Yau 1978\textup{)}, and let $\cC = D^b\Coh(X)$. On the
product-stable sublocus $\StabOp(X_1) \times \StabOp(X_2) \hookrightarrow
\StabOp(X)$,
\[
  \Phi_4(X)
  \;\simeq\;
  \Phi_{d_1}(X_1) \;\boxtimes\; \Phi_{d_2}(X_2)
\]
as $\Eone$-chiral algebras on $\Ran(\Sigma)$, where each factor
$\Phi_{d_i}(X_i)$ is $\Etwo$-chiral for $d_i = 2$ and $\Eone$-chiral for
$d_i = 1$ or $d_i = 3$. The product chiral algebra is $\Eone$-chiral
natively at the chain level, and carries an $\Etwo$-braided recovery on
the Drinfeld centre $\cZ(\RepE(\Phi_4(X))) \simeq
\mathrm{Rep}^{\Etwo}(\Drin(\Phi_4(X)))$.
\end{theorem}

\begin{proof}[Sketch]
Proposition~\ref{prop:phi-d-kunneth-split-product} \textup{(}of
\texttt{chapters/theory/cy\_to\_chiral.tex}\textup{)} applies: the
hypotheses \textup{(}H1\textup{)} $\bS^4$-framing splits as
$\bS^{d_1} \boxtimes \bS^{d_2}$ \textup{(}equivalent to the first
Pontryagin class factoring through the two projections --- true for
$K3 \times K3$, $K3 \times T^4$, $T^4 \times T^4$\textup{),}
\textup{(}H2\textup{)} absence of BCOV one-loop anomaly at
$d_i \leq 2$ \textup{(}Costello--Li 2016 Prop.~5.2\textup{),} and
\textup{(}H3\textup{)} product slicing on the Bridgeland manifold
\textup{(}Bayer--Macr\`i--Stellari 2016\textup{).} The Dunn--Lurie
identification $E_{n_1} \otimes_{\mathrm{Dunn}} E_{n_2} \simeq
E_{n_1 + n_2}$ matches the Gerstenhaber-degree calculation
$1 - 4 = (1 - d_1) + (1 - d_2) - 1$ with $d_1 + d_2 = 4$. For
$(d_1, d_2) = (2, 2)$ Dunn--Lurie gives $\Etwo \otimes \Etwo = E_4$,
collapsing to $\Eone$ after specialisation $\mathrm{Sp}_{\Sigma_3, C}$
over $\Sigma_3 = \Sigma_1 \times \Sigma_2 \subset X_1 \times X_2$, with
each factor specialisation $\mathrm{Sp}_{\Sigma_1, C}$ already
$\Eone$-chiral on $C$.
\end{proof}

\begin{remark}[The $E_1$-output is forced, not a choice]
\label{wn:rem:wave15-a8-E1-output-forced}
At $d = 3$ the K3$\,\times\,$E analysis established the pattern:
$\Phi_3(K3 \times E)$ is natively $\Eone$-chiral; the would-be
$\Etwo$-structure on $\mathbf H_{\Delta_5}$ itself obstructs via the
Siegel pentagon-$\hbar^3$ cocycle
\textup{(}Proposition~\ref{prop:qgfnd-pentagon-hbar3-obstruction}\textup{).}
At $d = 4$ the same obstruction persists, strengthened by one degree:
the would-be $\Etwo$-associator class lives in $\ChirHoch^3(A)$
\textup{(}not $\ChirHoch^2$ as at $d = 3$\textup{).} Francis' concentration
theorem \textup{(}\cite{Francis2013} Thm.~1.1\textup{)} gives
$\ChirHoch^\bullet(A)$ in $\{0, 1, 2\}$ for an $\Eone$-chiral algebra
$A$; the associativity obstruction in degree $3$ therefore vanishes
automatically for the $\Eone$-structure, while the $\Etwo$-lift requires
annihilating a class in degree $3$ of the $\Etwo$-deformation complex
which is non-zero at $d = 4$ whenever the BCOV quartic
$\kappa^{(4)}_{\mathrm{BCOV}}(X) = \int_X H^4$ is non-zero
\textup{(}Remark~\ref{rem:cy4-associator-falsification}\textup{).} The
braided enhancement is recovered exactly as at $d = 3$: via the Drinfeld
centre passage $\cZ \colon \Eone\text{-}\mathrm{Cat} \to
\Etwo\text{-}\mathrm{Cat}$ categorifying the commutant, never the
averaging map $\Eone \to E_\infty$.
\end{remark}

\paragraph{Irreducible CY$_4$: the Fermat sextic.}
Let $X_6 \subset \bP^5$ be the Fermat sextic $\{z_0^6 + \cdots + z_5^6 =
0\}$, a smooth compact projective CY$_4$ with
$(h^{0,0}, h^{0,1}, h^{0,2}, h^{0,3}, h^{0,4}) = (1, 0, 0, 0, 1)$,
$h^{1,1}(X_6) = 1$, $h^{3,1}(X_6) = 426$, $h^{2,2}(X_6) = 1752$,
$h^{2,2}_{\mathrm{prim}}(X_6) = 1750$,
$\chi_{\mathrm{top}}(X_6) = 2610$, $\int_{X_6} p_1(T_X) = -180$
\textup{(}Klemm--Pandharipande 2007 Table~1\textup{).} The sextic is
$\emph{not}$ a product, so hypothesis \textup{(}H1\textup{)} of
Theorem~\ref{wn:thm:wave15-a8-phi4-product-scope} fails. The
factorisation-homology route collapses in a different way: directly
through the Kapranov $L_\infty$ structure on $T_{X_6}[-1]$.

\begin{theorem}[$\Phi_4$-output scope for the Fermat sextic]
\label{wn:thm:wave15-a8-phi4-sextic-scope}
\ClaimStatusProvedHere\ \ClaimStatusConditional{ (pointwise $\bS^4$-framing: $N(X_6) = 5400 \neq 0$; $\bP^1$-family obstruction is genuine)}
For $X_6 \subset \bP^5$ the smooth Fermat sextic CY$_4$, the CY-to-chiral
functor $\Phi_4$ takes the value, not a single $\Eone$-chiral algebra,
but a $\bP^1$-family of $\Eone$-chiral algebras
$\{A^{(\sigma_3, \sigma_4)}_{X_6}\}_{[\sigma_3 : \sigma_4] \in \bP^1}$
on $\Ran(\Sigma)$, where $\sigma_3 \in H^{3,1}(X_6)$ runs the classical
complex-structure deformation direction and $\sigma_4 \in
H^{2,2}_{\mathrm{prim}}(X_6)$ runs the new $d = 4$ primitive-middle BCOV
deformation direction. Each fibre $A^{(\sigma_3, \sigma_4)}_{X_6}$ is
constructed in four steps:
\begin{enumerate}[label=\textup{(\arabic*)}]
 \item HKR endomorphism dg algebra
   $\mathrm{End}_{\mathrm{HKR}}(\cC) \simeq \PV^\bullet(X_6)[u]$.
 \item Negative cyclic refinement $\HH C^-_\bullet(\mathrm{End}_{\mathrm{HKR}})$
   with Connes' $u$.
 \item BCOV Maurer--Cartan twist
   $\bar\partial \mu + \tfrac12[\mu,\mu] + \sigma_3 \wedge \mu^2 +
   \sigma_4 \wedge \mu^3 = 0$
   \textup{(}the third-order term is the novel CY$_4$ feature\textup{).}
 \item Universal $\Eone$-chiral envelope
   $A^{(\sigma_3, \sigma_4)}_{X_6} := U^{ch}_{\Eone}(L^{(\sigma_3,
   \sigma_4)}_{X_6})$.
\end{enumerate}
Each fibre is $\Eone$-chiral natively. The $\Etwo$-braided enhancement
lives on the Drinfeld centre
$\cZ(\RepE(A^{(\sigma_3, \sigma_4)}_{X_6}))$ with the $p_1$-twisted
half-braiding driven by the integer level
$\ell(X_6, \alpha) = \int_{X_6} p_1(T_{X_6}) \cup \alpha = -180$.
\end{theorem}

\begin{proof}[Sketch]
Construction~\ref{constr:phi-4-p1-family} of
\texttt{cy\_to\_chiral.tex} furnishes the four-step chain-level model;
Theorem~\ref{thm:phi-4-eff-p1-vanishing} computes the integer level of
the pointwise $\Z$-shifted symplectic structure from
$\int_{X_6} p_1(T_{X_6}) \cup \alpha$ for
$\alpha \in H^4(X_6; \Z)$; the $\Eone$ specialisation from the native
$E_4$-holomorphic factorisation algebra is Dunn--Lurie collapse along the
$3$-cycle $\Sigma_3 \subset X_6$. Pontryagin-number
$N(X_6) = \int p_1^2 = 5400 \neq 0$ places $X_6$ off the $N = 0$
stratum; the $\bP^1$-family is non-trivially non-constant, with central
charges $c = 2591$ at the $\sigma_4 \to 0$ limit
\textup{(}the $\Phi_3$-extension limit\textup{)} and
$c = 3466$ at the $\sigma_3 \to 0$ limit
\textup{(}the new CY$_4$-specific limit glued to
$\mathrm{Cliff}(H^{2,2}_{\mathrm{prim}})$\textup{),}
Example~\ref{ex:sextic-cy4-associator}.
\end{proof}

\begin{theorem}[Kapranov $L_\infty$ certification: sextic is weight-$2$, not higher]
\label{wn:thm:wave15-a8-kapranov-sextic-weight}
\ClaimStatusProvedHere\ The Fermat sextic $X_6 \subset \bP^5$ supports a
non-trivial \emph{weight-$2$} Kapranov $L_\infty$-structure on
$T_{X_6}[-1]$: the binary Atiyah bracket
$\ell_2 = \At(T_{X_6})(v_1 \wedge v_2)$ of Kapranov 1999 Compositio
Math.~115, and the ternary tree-level BCOV Yukawa
$\ell_3^{\mathrm{min}}(v_1, v_2, v_3) = \int_{X_6} \Omega_{X_6} \wedge
\At(v_1) \cup \At(v_2) \cup \At(v_3)$ on $H^1(X_6, T_{X_6})$ --- the
two brackets with non-trivial $\sigma_{X_6}$-contraction, where
$\sigma_{X_6} \in H^0(X_6, \Omega^{4,0}_{X_6})$ is the holomorphic
$(4,0)$-form. Higher Kapranov brackets $\ell_n^{\mathrm{min}}$ for
$n \geq 4$ vanish after $\iota_{\sigma_{X_6}^\vee}$-contraction because
$H^{0,q}(X_6) = 0$ for $q \in \{1, 2, 3\}$; the relevant arities cap at
$n = 3$ on the sextic. The Kapranov tower is sharp at weight $2$ in the
sense of Definition~\ref{def:genericity-sigma-X-rank}.
\end{theorem}

\begin{proof}[Sketch]
Direct by Theorem~\ref{thm:weight-k-kapranov-obstruction} applied at
$k = 2$: the equivalent conditions require
$\sigma_{X_6}$ generic of rank $\geq 3 = 2k - 1$ and
$\iota_{\sigma_{X_6}^\vee}(\At^{(2)}(T_{X_6})) \neq 0$ in cohomology. The
first holds because $\sigma_{X_6} \in H^0(\Omega^{4,0}) \cong \C$ is
nowhere vanishing on the Fermat sextic \textup{(}CY condition\textup{),}
forcing maximal rank $4$; the second is the obstruction-theoretic
translation of
$\kappa^{(4)}_{\mathrm{BCOV}}(X_6) = \int_{X_6} H^4 = \deg(X_6) = 6 \neq 0$
\textup{(}top-intersection on $\bP^5$\textup{).} The vanishing of
$\ell_n^{\mathrm{min}}$ for $n \geq 4$ follows from
$H^{0,q}(X_6) = 0$ at $q = 1, 2, 3$: the target of
$\ell_n^{\mathrm{min}}$ sits in $H^{0, n-1}(X_6)$ by Serre duality
translation, vanishing for $n = 4, 5, 6$ on the strict CY fourfold.
Higher Kapranov brackets exist only for hyperk\"ahler fourfolds with
$h^{0,2} = 1$ from a holomorphic-symplectic form, such as $K3^{[2]}$ or
Beauville--Donagi cubic fourfold lines $F(Y)$; the sextic has
$h^{0,2}(X_6) = 0$ and is strictly weight-$2$.
\end{proof}

\begin{theorem}[$\Etwo$-braided enhancement on $Z(\RepE(\Phi_4(X)))$]
\label{wn:thm:wave15-a8-E2-drinfeld-centre}
\ClaimStatusProvedHere\ Let $X$ be either $K3 \times K3$ or $X_6 \subset
\bP^5$, $\cC = D^b\Coh(X)$, and $A = \Phi_4(\cC)$ \textup{(}a single

\[
  \cZ\bigl(\RepE(A)\bigr)
  \;\simeq\;
  \mathrm{Rep}^{\Etwo}\bigl(\Drin(A)\bigr)
\]
carries a canonical $\Etwo$-chiral structure. On the product case this is

\begin{proof}[Sketch]

\texttt{chapters/theory/e1\_chiral\_algebras.tex}: for any $\Eone$-chiral

\end{proof}

\begin{remark}[Why not $E_{-1}$ or $E_0$ at $d = 4$]
\label{wn:rem:wave15-a8-not-E-minus-1}
\ClaimStatusTheorem{}\ The naive guesses $E_{-1}$-chiral
\textup{(}spectrum-level\textup{)} or $E_0$-chiral \textup{(}pointed
set\textup{)} do not arise. The output of $\Phi_4$ after specialisation
$\mathrm{Sp}_{\Sigma_3, C}$ is $\Eone$-chiral. The reason is structural:
Beilinson--Drinfeld's $E_n$-chiral hierarchy on a curve $C$ requires
$n \geq 1$ to carry a non-trivial chiral operation
\textup{(}$\Eone$-chiral = associative $\otimes$ holomorphic on
$C$\textup{);} $n = 0$ would degenerate to a sheaf of pointed sets, and
$n = -1$ has no chiral incarnation. The Dunn--Lurie collapse
$E_4 \to E_1$ via $\int_{\Sigma_3}$ is proved in the category
$\mathrm{Pr}^L$ of presentable $\infty$-categories
\textup{(}Francis 2013 Adv.~Math.~239 Thm.~1.1\textup{),} where the
result lands in $\Eone$-chiral, not below; a further collapse would
require $\int_{\Sigma_0} = \int_{\mathrm{pt}}$, i.e., no specialisation,
and the unrelated collapse $E_n \to E_{n-1}$ via
$\int_{S^0}$ is not the specialisation operation. The $d = 4$ position
is therefore: natively $\Eone$, with $\Etwo$ braided recovery via the
Drinfeld centre, matching the general $d \geq 3$ pattern and extending
the $d = 3$ position one degree upward.
\end{remark}

\paragraph{Falsifiable tests of the prediction.}

\begin{remark}[Four falsifiable tests of the $\Phi_4$-output prediction]
\label{wn:rem:wave15-a8-falsifiable-tests}
Four independent computations would falsify the Kapranov prediction:
\begin{enumerate}[label=\textup{(F\arabic*)}]
 \item \emph{Pontryagin-number test.} The prediction forces $N(X) =
   \int_X p_1(T_X)^2$ to control the $\bP^1$-family twist: $N(X) = 0$
   should give $\bP^1$-trivial $\Phi_4(X)$ \textup{(}single
   $\Eone$-chiral algebra\textup{)} and $N(X) \neq 0$ should give a
   genuinely non-constant family. The table of
   Remark~\ref{rem:cy4-pontryagin-number-stratum} records
   $N(\C^4/\Lambda) = N(K3 \times T^4) = 0$ on the $N = 0$ stratum
   and $N(X_6) = 5400, N(K3 \times K3) = 4608, N(X_7) = 12348$ off it.
   A direct computation producing a non-trivial $\bP^1$-family on an
   $N = 0$ example would falsify.
 \item \emph{Central-charge test at the sextic.} The two limits of the
   sextic $\bP^1$-family predict $c = 2591$ at $\sigma_4 \to 0$ and
   $c = 3466$ at $\sigma_3 \to 0$, differing by
   $h^{2,2}_{\mathrm{prim}}/2 = 1750/2 = 875$. A direct computation of
   either central charge disagreeing with these values would falsify.
 \item \emph{$\kappa_{\mathrm{ch}}$ on the sextic.} The Hodge-supertrace
   prediction $\kappa_{\mathrm{ch}}(X_6) = 2$ at $d = 4$ is tight: a
   direct Costello--Li BCOV-quantisation computation of
   $\kappa_{\mathrm{ch}}$ yielding any value $\neq 2$ would falsify, as
   would a value $\neq 4$ on $K3 \times K3$ or $\neq 3$ on $K3^{[2]}$ /
   $F(Y)$ or $\neq 151$ on the octic double cover.
 \item \emph{BCOV $F_g$-correction at $g \geq 2$.} The sextic
   $\HH_{-1}(X_6) = h^{2,1} + h^{1,2} = 852$ is non-zero, so the
   naive expectation is that $F_2$ contributes. The $F_1$
   moduli-independence mechanism
   \textup{(}Theorem~\ref{thm:bcov-f2-zero-correction-d4}\textup{)}
   predicts zero correction. An explicit $F_2$ computation via
   Costello--Li 2015 BCOV quantisation on the sextic with non-zero
   anomaly at $F_2$ would falsify.
\end{enumerate}
\end{remark}
