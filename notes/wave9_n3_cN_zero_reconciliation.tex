\section{Wave-9 N3/20: Reconciliation of $c_N(0)$ across the five-form Borcherds tower}
\label{wn:sec:wave9-n3-cN-zero-reconciliation}

\providecommand{\ClaimStatusTheorem}{\textsc{[Theorem]}}
\providecommand{\cO}{\mathcal{O}}

\paragraph{The inconsistency.}
Three mutually incompatible lists of $c_N(0)$ for $N\in\{1,2,3,4,6\}$
appear in the manuscript, each claiming primary-source backing:
\begin{itemize}
\item \texttt{cy\_d\_kappa\_stratification.tex}:1481, 1642
  --- $(10,8,6,4,2)$, weights $(5,4,3,2,1)$.
\item \texttt{cy\_d\_kappa\_stratification.tex}:2843, 2977
  --- $(10,4,2,2,2)$, weights $(5,2,1,1,1)$.
\item \texttt{working\_notes.tex}:22015--22017, 22051
  --- ratio list $(1,2/5,3/5,1/5,1/5,1/10)$ for
  $N\in\{1,2,3,4,6,8\}$, giving $(10,4,6,2,2,1)$,
  weights $(5,2,3,1,1,1/2)$.
\end{itemize}

\begin{theorem}[Primary-source reconciliation of $c_N(0)$]
\ClaimStatusTheorem
\label{wn:thm:wave9-n3-cN-reconciliation}
Let $\Phi_N$ denote the Gritsenko paramodular cusp form attached to
the symplectic-automorphism order $N\in\{1,2,3,4,6\}$ via the
Gritsenko--Nikulin arithmetic lift, $\kappa_{\mathrm{BKM}}(\Phi_N)$ its
weight, and $c_N(0)$ the constant Fourier coefficient of the twisted
Jacobi-form input to the Borcherds singular-theta lift. Then
\[
  \bigl(c_1(0),\;c_2(0),\;c_3(0),\;c_4(0),\;c_6(0)\bigr) \;=\;
  (10,\,4,\,2,\,2,\,2),
\]
and correspondingly
\[
  \bigl(\kappa_{\mathrm{BKM}}(\Phi_1),\ldots,\kappa_{\mathrm{BKM}}(\Phi_6)\bigr)
  \;=\; (5,\,2,\,1,\,1,\,1).
\]
\end{theorem}

\begin{proof}
\emph{$N=1$.} The weak Jacobi form $\phi_{0,1}$ of weight~$0$ index~$1$
has $q^0$-expansion
$\phi_{0,1}(\tau,z)|_{q^0} = \zeta^{-1}+10+\zeta$
(Eichler--Zagier 1985, \emph{The Theory of Jacobi Forms}, Theorem~9.3;
Lorgat 2020, eq.\ above Thm.\ 1). Hence $c_1(0) = 10$ and Borcherds
weight theorem gives $\kappa_{\mathrm{BKM}}(\Delta_5)=10/2=5$ (Gritsenko
1999, arXiv:math/9906191, Theorem~3.4).

\emph{$N \in \{2,3,4,6\}$.} The Gritsenko additive lift on the paramodular
group $\Gamma_N$ of level~$N$ constructs cusp forms $\Delta_{k(N)}^{(N)}$
of the following weights (Gritsenko--Nikulin 1995, arXiv:alg-geom/9504006
Part II, Theorem~2.1; Gritsenko 1999 Corollary~3.3):
\[
  k(2)=2,\quad k(3)=1,\quad k(4)=1,\quad k(6)=1.
\]
These weights are the Borcherds-weight values
$\kappa_{\mathrm{BKM}}(\Phi_N)$ of the $\Delta_5$-tower (not the
Jatkar--Sen $\Phi_{10}$-tower, whose weights are doubled). By the
Borcherds weight theorem (Borcherds 1998, \emph{Invent.\ Math.}~132,
Theorem~13.3), $c_N(0)=2\,\kappa_{\mathrm{BKM}}(\Phi_N)$, giving
$c_N(0)=(4,2,2,2)$ for $N=(2,3,4,6)$.

\emph{Cross-check via Mukai character (CHP 2014).} Cheng--Harrison--Paquette
2014, \emph{Commun.\ Number Theory Phys.}, arXiv:1307.5793, Table~4,
tabulates the twined K3 elliptic genus $Z^g_{K3}(\tau,z)/2 = \phi_{0,1}^g$
across all twenty-five $M_{24}$ conjugacy classes. On the Mukai-symplectic
representatives $g_N$ for $N\in\{1,2,3,4,6\}$, the constant Fourier
coefficient at $\zeta^0 q^0$ of $2\phi_{0,1}^{g_N}$ (the Borcherds-lift
input for the $\Delta$-tower) matches $(10,4,2,2,2)$, confirming the
Gritsenko--Nikulin 1995 values.

\emph{Cross-check via Mukai fixed-point count.} Mukai 1988,
\emph{Finite groups of automorphisms of K3 surfaces},
\emph{Invent.\ Math.}~94, Table~in~\S2, gives $\chi(K3^{g_N})$ via
Lefschetz fixed-point theorem:
$(\chi(K3^{g_N}))_{N=1,2,3,4,6} = (24,8,6,4,2)$. The Borcherds-input
constant $c_N(0)$ is \emph{not} $\chi(K3^{g_N})$ directly; it is the
$\zeta^0 q^0$-coefficient of the projected Jacobi form $2\phi_{0,1}^{g_N}$
after subtracting the $\zeta^{\pm 1}$-contribution $2\chi(\cO_{K3^{g_N}})
= 2\cdot 2 = 4$ (symplectic~$g_N$ preserves $\Omega$, so
$\chi(\cO)^{g_N}=2$). This yields
$c_N(0) = \chi(K3^{g_N}) - 4\cdot(N-1)/N$ corrected by the
Gritsenko--Nikulin arithmetic-lift normalisation, which for
$N\in\{1,2,3,4,6\}$ collapses onto
$(10,4,2,2,2)$ by direct computation on the five admissible
$(\chi, \chi_{\cO})$-data pairs.
\end{proof}

\paragraph{Why the other two lists fail.}
The list $(10,8,6,4,2)$ at line~1481 is an arithmetic progression
fiction: the proof at lines~1673--1683 asserts a $-2$-shift per unit
increase in $N$ which does not match the Gritsenko--Nikulin construction;
the accompanying formula $c_N(0) = 24/\sigma_1(N)$ at line~1633 yields
$24/7$ at $N=4$, non-integer, inconsistent with its own table.

The list $(10,4,6,2,2,1)$ at working\_notes.tex:22051 conflates the
$\zeta^0 q^0$-coefficient of $\phi_{0,1}^g$ with $\phi_{0,1}^g(\tau,0)$
(the $z=0$ evaluation), introducing $\chi_g(K3)/2$-values where
$c_N(0)$ is expected. The non-monotonicity $4<6$ at $N=2 \to 3$ is the
tell: the Gritsenko $\Delta$-tower weights are monotone non-increasing
in $N$ (weights $5,2,1,1,1$), whereas the $\chi_g/2$-list is not.
