\section{Chiral cochains on the Ran space}
Let $X$ be a smooth complex manifold. A holomorphic chiral Lie algebroid
on $X$ consists of a right $\cD_X$-module, equivalently a holomorphic
factorisation object on $\Ran(X)$ in the sense of Beilinson--Drinfeld,
with a chiral bracket
\[
  \mu:j_*j^*(\cL\boxtimes\cL)\longrightarrow\Delta_*\cL
\]
and an anchor $\rho:\cL\to\mathrm{Der}(\cO_X)$ satisfying skew-symmetry,
the chiral Jacobi identity on $\Ran^{\leq3}(X)$, and the Leibniz rule
against $\rho$. Here $j:X^2\setminus\Delta\hookrightarrow X^2$ is the
open complement of the diagonal.

\begin{definition}[Chiral CE cochains]\label{def:chiral-ce-cochains}
For a chiral $\cL$-module $\cM$, define
\[
 \CE^n_{\chir}(\cL,\cM)
 :=
 \Hom_{\cD_{X^n}}\!\bigl(\Sym^n_{\chir}\cL,\Delta^{(n)}_*\cM\bigr),
\]
where $\Delta^{(n)}:X\hookrightarrow X^n$ is the small diagonal and
$\Sym^n_{\chir}\cL$ is the Beilinson--Drinfeld chiral symmetric power.
The chiral differential $d_{\chir}=d_\mu+d_\rho$ is the alternating sum
of bracket insertions and module-action insertions along the diagonal
strata of $\Ran^{\leq n+1}(X)$.
\end{definition}

The identity $d_{\chir}^2=0$ follows from the chiral Jacobi identity,
the anchor Leibniz rule, and the module axiom.
The proof is the Ran-space analogue of the classical
Chevalley--Eilenberg cancellation: the two bracket insertions cancel by
Jacobi, and the action terms cancel by the module identity.

\section{Dolbeault refinement}
Let $\cL_{\mathrm{hol}}$ be a holomorphic Lie algebroid and set
\[
  \cL=\Omega^{0,\bullet}(X,\cL_{\mathrm{hol}}).
\]
The Dolbeault operator and the chiral Lie structure are compatible when
\[
\begin{aligned}
  \dbar[\alpha,\beta]
  &=
  [\dbar\alpha,\beta]+(-1)^{|\alpha|}[\alpha,\dbar\beta],\\
  \dbar(\rho(\alpha)m)
  &=
  \rho(\dbar\alpha)m+(-1)^{|\alpha|}\rho(\alpha)\dbar m,
\end{aligned}
\]
and the chiral operation $\mu$ is $\dbar$-closed as a distributional
operation along the diagonal. These hypotheses are part of the datum.

\begin{definition}[Dolbeault--chiral CE complex]
\label{def:dolbeault-chiral-ce}
For a Dolbeault chiral module $\cM$, set
\[
 \CE^*_{\dbar,\chir}(\cL,\cM)
 :=
 \left(
   \prod_{n\geq0}
   \Omega^{0,\bullet}\!\bigl(
     X^n,\Hom(\Sym^n_{\chir}\cL,\Delta^{(n)}_*\cM)
   \bigr)^{\Sigma_n},
   D=\dbar+d_{\chir}
 \right).
\]
\end{definition}

\begin{lemma}[Square-zero criterion]\label{lem:square-zero-criterion}
Under the compatibility hypotheses above, $D^2=0$.
\end{lemma}

\begin{proof}
The Dolbeault complex gives $\dbar^2=0$. The chiral Lie and module
axioms give $d_{\chir}^2=0$. The displayed derivation identities and
the $\dbar$-closedness of the diagonal operation give
$[\dbar,d_{\chir}]=0$.
\end{proof}

\section{Current and tangent algebroids on \texorpdfstring{$\C^3$}{C3}}
Let $\g$ be a finite-dimensional Lie algebra with invariant pairing.
The current algebroid
\[
  \cE_{\g}:=\Omega^{0,\bullet}(\C^3,\g)
\]
has trivial anchor and chiral bracket
\[
  \mu(\alpha,\beta)=[\alpha,\beta]_{\g}\,\delta_\Delta .
\]
The diagonal distribution is $\dbar$-closed and the coefficient bracket
is a graded Dolbeault derivation, so Lemma~\ref{lem:square-zero-criterion}
applies.

\begin{proposition}[Classical hCS observables]\label{prop:hcs-observables}
For affine holomorphic Chern--Simons theory on $\C^3$ with structure
Lie algebra $\g$,
\[
  \Obs_{\hCS}^{\cl}(\C^3)
  \simeq
  \CE^*_{\dbar,\chir}\!\bigl(\cE_{\g},\cO_{\C^3}\bigr)
\]
as completed classical factorisation algebras on $\C^3$.
\end{proposition}

\begin{proof}
The BV fields are $\Omega^{0,\bullet}(\C^3,\g)[1]$. The quadratic part
of the hCS action contributes the Dolbeault differential, and the cubic
vertex contributes the CE insertion induced by $[-,-]_{\g}$. Locality of
the action identifies multiplication over disjoint opens with the
factorisation product.
\end{proof}

\begin{construction}[Jet tangent algebroid]\label{con:jet-tangent}
Let
\[
  \cJ_{\cT}:=\Omega^{0,\bullet}(\C^3,J^\infty T^{1,0}_{\C^3}).
\]
The first-order Schouten bracket of holomorphic vector fields prolongs
to a Beilinson--Drinfeld chiral operation on the jet algebroid, supported
on the formal diagonal. The natural anchor is the prolonged action on
jets of $\cO_{\C^3}$.
\end{construction}

\begin{proposition}[Infinitesimal holomorphic diffeomorphisms]
\label{prop:tangent-jet-ce}
The complex
\[
  \CE^*_{\dbar,\chir}(\cJ_{\cT},\cO_{\C^3})
\]
is the chain-level factorisation algebra of infinitesimal holomorphic
diffeomorphism observables. On a Stein polydisc, its linear compactly
supported part is the continuous dual of
$\Omega^{0,\bullet}_c(D^3,J^\infty T^{1,0}_{D^3})[-1]$.
\end{proposition}

\begin{proof}
The Schouten Jacobi identity gives the chiral Jacobi identity after jet
prolongation. Holomorphicity of the bracket gives the Dolbeault
derivation identity. The CE construction then applies
Lemma~\ref{lem:square-zero-criterion}; compactly supported linear
observables are continuous duals of compactly supported fields shifted by
the BV convention.
\end{proof}

\section{Koszul duality and the \texorpdfstring{$d=3$}{d=3} operadic level}
Let $U^{\chir}_{E_3}(\cE_{\g})$ be the augmented completed holomorphic
$E_3$ chiral envelope of the current algebroid. Under the standard
augmentation, conilpotence, and completion hypotheses, Francis'
$E_n$ bar--cobar duality gives
\[
  U^{\chir}_{E_3}(\cE_{\g})
  \simeq
  \Omega_{E_3}\CE^*_{\dbar,\chir}(\cE_{\g},\C),
  \qquad
  \CE^*_{\dbar,\chir}(\cE_{\g},\C)
  \simeq
  B_{E_3}U^{\chir}_{E_3}(\cE_{\g}).
\]
This statement has two equal readings. Chain-level, it is the completed
Dolbeault double complex with the diagonal CE differential. In the
$(\infty,1)$-categorical lane, it is bar--cobar duality in the completed
holomorphic factorisation category.

For a CY$_3$ target, the local holomorphic factorisation algebra belongs
to the $\PhiFA_3$ stage. A curve-valued CY-to-chiral specialisation has
the form
\[
  A_X^{(\Sigma_2,C)}
  =
  \SpCh_{\Sigma_2,C}\bigl(\PhiFA_3(D^b\Coh(X))\bigr),
\]
and the specialised output is $E_1$-chiral. The $E_2$ structure, when
available, belongs to $\cZ(\Rep^{E_1}(A_X^{(\Sigma_2,C)}))$ or to a
completed Drinfeld double after a Hall pairing and completion datum have
been fixed.

\section{The \texorpdfstring{$\C^3$}{C3} CoHA comparison}
For the affine toric CY$_3$, the cohomological Hall algebra is the
positive half $Y^+$. Let $\gBPS(\C^3)$ denote the primitive BPS Lie
algebra. After the augmentation $Y^+\to\C$, the expected chiral
Koszul comparison is
\[
  \int_{\C^3} Y^+
  \simeq
  \CE^*_{\chir}\!\bigl(\gBPS(\C^3),\C\bigr),
\]
in the completed chiral Hall category. The comparison uses the CoHA
algebra; the full $\mathcal{W}_{1+\infty}$ object is attached to the
double or evaluation layer.

\section{\texorpdfstring{$K3 \times E$}{K3 x E} and the BKM layer}
Let $Y=K3\times E$. Where the cohomological Hall algebra is constructed,
write
\[
  \gBPS(Y):=\operatorname{Prim}\CoHA(Y)
\]
for its primitive $E_1$ Lie object. The identification of this primitive
object with the Borcherds--Kac--Moody algebra
$\mathfrak g_{\Delta_5}$ is conjectural. The Dolbeault Lie algebroid
\[
  \cE_Y:=\Omega^{0,\bullet}(Y,\gBPS(Y))
\]
has trivial anchor, and its chiral CE factorisation algebra is
\[
  \cF^{\chir}_{\mathrm{BKM}}(Y)
  :=
  \CE^*_{\dbar,\chir}(\cE_Y,\cO_Y).
\]

\begin{conjecture}[Chiral BKM CE model]\label{conj:chiral-bkm-ce-model}
Assume the primitive Hall Lie object $\gBPS(Y)$ exists with the required
completion and carries the diagonal chiral operation compatible with
$\dbar$. If $\gBPS(Y)\simeq\mathfrak g_{\Delta_5}$, then
$\cF^{\chir}_{\mathrm{BKM}}(Y)$ is the chiral CE factorisation algebra
of the $K3\times E$ BKM layer.
\end{conjecture}

The subscripted invariants attached to $Y$ remain separate:
\[
\begin{array}{ccl}
 \kcat(Y) &=& \chi(\cO_{K3})\chi(\cO_E)=2\cdot0=0,\\[2mm]
 \kch(\PhiFA_3(Y)) &=&
   \displaystyle\sum_{q=0}^3(-1)^q h^{0,q}(Y)
   =1-1+1-1=0,\\[2mm]
 \kch^{\mathrm{Heis}}(Y) &=& 3,\\[1mm]
 \kBKM(\Delta_5) &=& c_1(0)/2=10/2=5,\\[1mm]
 \kfiber(Y) &=& \operatorname{rk}\widetilde H(K3,\Z)
   =1+22+1=24.
\end{array}
\]
The K3 fibre separately has $\kcat(K3)=2$; the invariant menu for
$K3\times E$ is $\{0,3,5,24\}$, indexed by the constructions displayed
above. A renormalised Euler characteristic of
$\cF^{\chir}_{\mathrm{BKM}}(Y)$ requires a trace, completion, and
coefficient-sector convention, so no additional scalar is assigned to it
here.
