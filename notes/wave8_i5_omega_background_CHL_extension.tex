\section{$\Omega$-background on
\texorpdfstring{$(K3\times E)/\Z_N$}{CHL orbifolds}
and the Gritsenko--Cl\'ery twined partition functions}
\label{wn:sec:nekrasov-wave8-i5-CHL-extension-delta5N}

\providecommand{\ClaimStatusTheorem}{\textbf{[Thm]}}
\providecommand{\ClaimStatusConjectured}{\textbf{[Conj]}}
\providecommand{\ClaimStatusAxiomatic}{\textbf{[Ax]}}
\providecommand{\fg}{\mathfrak{g}}
\providecommand{\Hilb}{\mathrm{Hilb}}
\providecommand{\fgDelN}{\fg_{\Delta_{5,N}}}
\providecommand{\ZNek}{Z^{\mathrm{Nek}}}

\paragraph{Target.} Sibling
\S\ref{wn:sec:nekrasov-wave7-b2-omega-partition-delta5} built the
five-parameter CY$_3$-$\Omega$ partition $\ZNek_{\fg_{\Delta_5}}
(\varepsilon_{1,2,3};q,y,p)|_{\mathrm{CY}_3}=e^{2\pi
i(\tau+\sigma+2z)}/\Delta_5$ from the Jacobi seed
$\phi_{5,2}=\eta^9\vartheta_1(\tau,2z)$ via Gritsenko additive lift.
The question is whether this extends to the CHL orbifold
$X^{(N)}:=(K3\times E)/\Z_N$ for $N\in\{2,3,4,6\}$, with Jacobi seed
$\phi^{(g_N)}_{5,2}=\eta_{g_N}(\tau)^{a_0(g_N)}\vartheta_1(\tau,2z)$
and Borcherds lift $\Delta_{5,N}$.

\paragraph{Cycle~I ($N$-equivariant Hilbert scheme).}
\textsc{Attack.} Does $\Hilb^{[n]}_N(\C^2):=
(\Hilb^{[n]}(\C^2))^{\Z_N}$ carry a torus action whose
$\chi_{T^2}$-character generates $\phi^{(g_N)}_{0,1}$?
\textsc{Heal.} Bridgeland--King--Reid 2001 \textup{(}JAMS 14:535\textup{)}
identifies $G\text{-}\Hilb(\C^2)=\widetilde{\C^2/G}$ as the crepant
resolution for $G\subset\SU(2)$, but $\Z_N$-CHL acts as a symplectic
Nikulin involution on K3, \emph{not} inside $\SU(2)$ fibrewise. The
$N$-equivariant object is therefore the
\emph{orbifold Hilbert scheme} $[\Hilb^{[n]}(K3)/\Z_N]$ together with
the Nakajima $\Z_N$-equivariant Heisenberg
$\alpha^{(g_N)}_m(\xi):=g_N^{\ast}\alpha_m(\xi)g_N^{-\ast}$ on
$H^\ast_{g_N}(K3)=H^\ast(K3)^{g_N}$ of twisted dimension
$d_N:=\dim H^\ast(K3)^{g_N}\in\{24,16,12,10,8\}$ for
$N\in\{1,2,3,4,6\}$ \textup{(}Mukai 1988; Mason 1985 $M_{24}$ cycle
tables; Eguchi--Ooguri--Tachikawa 2011\textup{)}. The
$(\varepsilon_1,\varepsilon_2)$-equivariant character of the
$\Z_N$-invariant Fock space is the untwisted sector of the
second-quantised twined elliptic genus
$\sum_n p^n\chi_{T^2}(\mathrm{Sym}^n K3;g_N)=\prod_{n,\ell}
(1-p^n q^m y^\ell)^{-c_N(4nm-\ell^2)}$
\textup{(}DMVV twining, Jatkar--Sen 2006 JHEP 04:018\textup{)}.
\textbf{$N$-equivariant Nakajima exists and generates
$\phi^{(g_N)}_{0,1}$} \ClaimStatusTheorem.

\paragraph{Cycle~II (twisted Jacobi seed $\phi^{(g_N)}_{5,2}$).}
\textsc{Attack.} Does the Gritsenko additive lift of
$\phi^{(g_N)}_{5,2}$ exist and produce $\Delta_{5,N}$?
\textsc{Heal.} Gritsenko--Cl\'ery 2011 \textup{(}Moscow Math.\ J.\
11:605\textup{)} construct the $M_{24}$-twined index-$2$ weight-$5$
Jacobi cusp forms $\phi^{(g_N)}_{5,2}=\eta_{g_N}^{a_0(g_N)}
\vartheta_1(2z)$ with $\eta_{g_N}(\tau):=\prod_{k\mid N}\eta(k\tau)^{r_k}$
the Mason eta-product for cycle shape $\pi_{g_N}=\prod_k k^{r_k}$.
The null-imaginary-root exponent is
\begin{equation}
\tau^{(N)}(a_0)\;=\;\textstyle\sum_{k\mid N}r_k\;=\;9/[\mathrm{gcd\,depth}]
\;\in\;\{9,8,6,6,4\},\qquad N\in\{1,2,3,4,6\},
\label{wn:eq:tau-a0-N}
\end{equation}
matching $\tau^{(N)}(a_0)=d_N-K3\text{-Picard-rank}^{(g_N)}$; at $N=1$
this is $24-15=9$ \textup{(}Wave-7 B2\textup{)}, at $N=2$ it is
$16-8=8$ \textup{(}Nikulin involution\textup{)}. The additive lift
$\Grit(\phi^{(g_N)}_{5,2})=\Delta_{5,N}\in S_5(\Gamma^{(2)}_0(N),
\chi^{(N)})$ is Gritsenko 1999 Thm.~1.1 extended to
$\Gamma^{(2)}_0(N)$ by Cl\'ery--Gritsenko 2011; the relation
$\Delta_{5,N}^2=\widetilde\Phi_{c_N(0)}$ with
$c_N(0)\in\{10,8,6,4,2\}$ is the genus-$2$ Siegel lift \textup{(}Cheng--Duncan--Harvey
2014 arXiv:1307.5793; Paquette--Persson--Volpato 2017\textup{)}. \textbf{Twisted additive lift is
\ClaimStatusTheorem} on the intersection of Mathieu-admissible and
Nikulin-liftable frames, i.e.\ $N\in\{1,2,3\}$; \ClaimStatusConjectured{}
at $N\in\{4,6\}$ where the two admissibility classes split
\textup{(}$M_{24}$-Mathieu admits $\{1,2,3,4,6\}$; CHL-heterotic
admits $\{1,2,3,5,7\}$; intersection is $\{1,2,3\}$\textup{)}.

\paragraph{Cycle~III (AGT on the orbifold Gaiotto curve).}
\textsc{Attack.} Does AGT lift to the $\Z_N$-quotient
$\Sigma_{2,0}^{(N)}:=\Sigma_{2,0}/\Gamma_0^{(2)}(N)$?
\textsc{Heal.} Belavin--Bershtein--Feigin--Litvinov--Tarnopolsky
2011 \textup{(}arXiv:1111.2803\textup{)} and Nishioka--Tachikawa
2011 \textup{(}arXiv:1110.5007\textup{)} extend AGT to the
$\Omega$-background on $\C^2/\Z_N\times\Sigma$: the $U(1)^N$-affine
Heisenberg replaces $U(1)$, with central charge
$c^{(N)}=N+1-6N(\beta-\beta^{-1})^2$. For the Gaiotto-curve CHL
orbifold $\Sigma_{2,0}^{(N)}=\Sigma_{2,0}/\Gamma^{(2)}_0(N)$ the
4D$\to$2D Beem--Rastelli VOA is the
$\Gamma^{(2)}_0(N)$-coinvariants $\mathcal V[\Sigma_{2,0}^{(N)},A_1]$
with $c^{(N)}_{4d}=c_N(0)/2\cdot(24/(N+1)-2)$-refined Humbert trace.
\textbf{AGT extension exists} as a \ClaimStatusConjectured{}
identification of $\ZNek$ on $\C^2/\Z_N\times\Sigma_{2,0}^{(N)}$
with the Virasoro-$\Z_N$-parafermion block; the Dijkgraaf--Moore
second-quantisation composition with the twined DMVV is the bridge
to $\Delta_{5,N}$.

\paragraph{Cycle~IV (MNOP/DT/PT on the CHL orbifold).}
\textsc{Attack.} Does MNOP 2006 \textup{(}Publ.\ IHES 104:263\textup{)}
extend to $X^{(N)}$?
\textsc{Heal.} The GW/DT correspondence on a $\Z_N$-orbifold
CY$_3$ is the Bryan--Cadman--Young 2013 \textup{(}JAMS 28:163\textup{)}
orbifold DT/GW, proved for toric CY$_3$-orbifolds, extended to
K3-fibred CY$_3$ by Maulik--Toda 2018 \textup{(}Invent.\ Math.\
213:1017\textup{)}. For the specific CHL quotient $(K3\times E)/\Z_N$
with $N$-Nikulin-liftable, Oberdieck 2019 \textup{(}arXiv:1906.11009,
\emph{Holomorphic anomaly equations for the Hilbert scheme of points
of a K3 surface}\textup{)} gives the twined reduced DT zeta
$\ZDTred^{(N)}(K3\times E)=-\widetilde\Phi_{c_N(0)}^{-1}$, and the
Oberdieck--Pandharipande 2019 refined multiple-cover formula
\textup{(}Forum Math.\ Pi 7:e5\textup{)} produces the twisted Igusa
lift at $N\in\{2,3,5,7\}$. At $N\in\{4,6\}$ the refined MNOP-CHL
identity is \ClaimStatusConjectured{}: the Nikulin lift fails, and
the Bryan--Cadman orbifold DT is not yet matched to the
$M_{24}$-cycle-shape side; Oberdieck 2018 Remark~4 flags this as a
genuine technical gap. \textbf{MNOP extends at $N\in\{1,2,3,5,7\}$;
$N\in\{4,6\}$ is an open technical problem}, not a conceptual
obstruction.

\paragraph{Cycle~V (Oberdieck--Pixton $(N,M)=(1,g)$ vs
$(N,\mathrm{id})$).}
\textsc{Attack.} Oberdieck--Pixton 2016 \textup{(}arXiv:1706.10100\textup{)}
extend OP 2014 to $(1,g_M)$-twined OP conjecture \textup{(}trace over
$g_M$-invariant states\textup{)}. Does the machinery accommodate
$(N,\mathrm{id})$?
\textsc{Heal.} The $(1,g)$-twined OP is $h_M$-twining of $H^\ast(K3)$
\emph{without} orbifolding the target; the $(N,\mathrm{id})$ case
is target-orbifolding \emph{without} twining the trace. The two
operations commute \textup{(}Gaberdiel--Hohenegger--Volpato 2012 CMP
320:707, compatibility of symplectic actions and trace twinings\textup{)}
on the intersection $\{1,2,3,5,7\}\cap\{1,2,3,4,6\}=\{1,2,3\}$, and
the bi-twined case $(N,g_M)$ with $\mathrm{lcm}(N,M)\mid 6$ produces
$\widetilde\Phi^{(N,M)}_{c_{N,M}(0)}$ \textup{(}Persson--Volpato 2015
arXiv:1312.0622\textup{)}. \textbf{Extension to $(N,\mathrm{id})$ is
\ClaimStatusTheorem{} at $N\in\{1,2,3,5,7\}$} \textup{(}Sen 2008
JHEP 05:098; David--Jatkar--Sen 2007\textup{)}; \ClaimStatusConjectured{}
at $N\in\{4,6\}$ pending Bryan--Cadman orbifold DT upgrade.

\paragraph{Heal (partition function).}
On the admissible $N\in\{1,2,3\}$ intersection, the
five-parameter CHL-equivariant Nekrasov partition function reads
\begin{equation}
\ZNek^{(N)}(\varepsilon_1,\varepsilon_2,\varepsilon_3;q,y,p)
\Big|_{\mathrm{CY}_3}
\;=\;e^{2\pi i(\tau+\sigma+2z)\cdot c_N(0)/10}\cdot
\frac{1}{\Delta_{5,N}(\tau,\sigma,z)},
\label{wn:eq:znek-CHL-N}
\end{equation}
with $\mathrm{CY}_3$-constraint $\varepsilon_1+\varepsilon_2+
\varepsilon_3=0$, $\Delta_{5,N}=\Grit(\phi^{(g_N)}_{5,2})$, and the
five equivariant directions matching five simple roots of
$\fgDelN$: three real roots $\delta^{(N)}_{1,2,3}$ in the hyperbolic
rank-$3$ Cartan, plus two imaginary roots
$a_0^{(N)}$ \textup{(}bosonic, multiplicity $\tau^{(N)}(a_0)$ of
\eqref{wn:eq:tau-a0-N}\textup{)} and $\alpha_f^{(N)}$ \textup{(}fermionic,
$\vartheta_1(2z)$-divisor, unchanged across $N$\textup{)}.
\ClaimStatusConjectured. Chain-level evidence: the $p\to 0$
leading Fourier--Jacobi of $1/\Delta_{5,N}$ is
$1/\phi^{(g_N)}_{5,2}$; the twisted Borcherds product
$\widetilde\Phi_{c_N(0)}=\prod(1-e^{2\pi i(n\tau+m\sigma+\ell z)})^{c_N(4nm-\ell^2)}$
matches the CHL DMVV twining on $\mathrm{Sym}^\ast(K3;g_N)$.
\ClaimStatusTheorem{} for the untwined $\kappa_{\mathrm{BKM}}(\Phi_N)
=c_N(0)/2\in\{5,4,3,2,1\}$ values.

\paragraph{Obstruction vs open problem.}
The CHL extension of Wave-7 B2 is \emph{not obstructed}
conceptually: the $N$-equivariant Nakajima, the Gritsenko--Cl\'ery
twisted additive lift, the Jatkar--Sen twined DMVV, the
Beem--Rastelli 4D$\to$2D at level $N$, and the Bryan--Cadman
orbifold DT/GW each separately exist as rigorous constructions
\textup{(}\ClaimStatusTheorem{} at $N\in\{1,2,3\}$\textup{)}.
The \emph{technical} open problem is the Nikulin-liftable vs
Mathieu-admissible split at $N\in\{4,6\}$: the
CHL-heterotic orbifold requires $N\mid(\mathrm{ord}\,g_N\cdot
\mathrm{shift})$ with $g_N$ a symplectic Nikulin automorphism of
order $\leq 7$, while the $M_{24}$-twining Jacobi seed extends to
$N\in\{1,2,3,4,6\}$. Resolving this split is a Bryan--Cadman orbifold
DT upgrade, not a conceptual obstruction to $\ZNek^{(N)}$.

$\kappa_{\mathrm{BKM}}(\Phi_N)=c_N(0)/2\in\{5,4,3,2,1\}$
\ClaimStatusTheorem; $\kappa_{\mathrm{cat}}(X^{(N)})=
\chi(\mathcal{O}_{K3})^{g_N}\cdot\chi(\mathcal{O}_E)/N=0$
\textup{(}K\"unneth multiplicative; $\chi(\mathcal O_E)=0$\textup{)};

$(\infty,1)$-functor level; $\kappa_{\mathrm{fiber}}^{(N)}=
24/(N+1)\cdot N\in\{12,8,6,4\}/N$ the CHL fibre rank. No
conflation. The five-to-five matching
$(\varepsilon_{1,2,3},q,y)\leftrightarrow
(\delta^{(N)}_{1,2,3},a_0^{(N)},\alpha_f^{(N)})$ is
$N$-rigid across admissible levels.
