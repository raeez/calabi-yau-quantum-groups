\section{The $E_2$-factorisation attack on
$\mathcal F_{\mathfrak g_{\Delta_5}}$ and the three-tier
$E_n$-layering}
\label{sec:wave6-e2-fact-attack-heal-x2}

\emph{Francis--Lurie channel, Wave~6 X2/20.} V1's coherence
remark \Cref{rem:delta5-humbert-coherence-functor-v1} posits a
BKM factorisation structure
$\mathcal F_{\mathfrak g_{\Delta_5}} \in
\Fact_{\mathrm E_2}(\widehat{\mathbf H}_1)$
on the derived formal completion of the Humbert divisor. The
CY-to-chiral discipline at $d = 3$ gives an $\Eone$-chiral
algebra for $K3 \times E$ (Theorem~CY-A$_3$,
\Cref{subsec:c3-drinfeld-center},
Warning~\ref{warn:gx-not-constructed}), not an $\Etwo$-chiral
algebra. The two statements live on three different layers;
conflating them is an instance of AP-CY$12$
($\Phi$-output-scope confusion).

\subsection{Attack: three distinct layers, three distinct $E_n$'s}
\label{subsec:wave6-x2-attack}

\begin{warning}[Three $E_n$'s, three different carriers]
\label{warn:three-en-layers}
The pair
$(\widehat{\mathbf H}_1, \mathcal F_{\mathfrak g_{\Delta_5}})$
of \Cref{rem:delta5-humbert-coherence-functor-v1} carries
$\Etwo$ as a factorisation covariance on the moduli-theoretic
carrier $\widehat{\mathbf H}_1 \subset \mathcal A_2$. This is
categorically distinct from the algebra-level covariance of
$\mathbf H_{\Delta_5} := \Phi_3(\Perf(K3 \times E))$, which is
$\Eone$. Three layers:
\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{llll}
\toprule
Layer & Carrier & $E_n$ & Source \\
\midrule
(L1) Algebra & $\mathbf H_{\Delta_5}$ & $\Eone$ &
CY-A$_3$ at $d = 3$ \\
(L2) Centre &
$\cZ(\Rep^{\Eone}(\mathbf H_{\Delta_5}))$ & $\Etwo$ &
Dunn--Lurie \\
(L3) Moduli & $\widehat{\mathbf H}_1 \subset \mathcal A_2$ &
$\Etwo$ & Fact.\ on $\cA_2$ \\
\bottomrule
\end{tabular}
\end{center}
The algebra-level $\Eone$ of (L1) and the moduli-level $\Etwo$
of (L3) are not the same structure; (L2) is the intermediate
centre restoring algebra-internal $\Etwo$-braiding in the
representation category of (L1), while (L3) is an external
$\Etwo$-factorisation covariance on the Siegel moduli itself.
\end{warning}

\begin{proposition}[Three attacks and their resolutions]
\label{prop:wave6-x2-three-attacks-resolved}
Three potential objections against
$\mathcal F_{\mathfrak g_{\Delta_5}} \in
\Fact_{\mathrm E_2}(\widehat{\mathbf H}_1)$ resolve:
\begin{enumerate}
\item[(A1)] \emph{$\Phi$ lands in $\Eone$-chiral at $d = 3$,
so $\Etwo$-factorisation is forbidden by AP-CY$12$.} The
discipline bars $\Etwo$-covariance on $\mathbf H_{\Delta_5}$
itself (L1), not on the moduli-theoretic carrier
$\widehat{\mathbf H}_1 \subset \mathcal A_2$: the Siegel upper
half-plane $\mathbb H_2$ is four-real-dimensional and carries
a canonical $\Etwo$-factorisation structure (Ran space of
$\cA_2$ over $\Spec \Z$; Lurie \emph{HA} $5.5.4$). The object
in $\Fact_{\Etwo}$ is the constructible cosheaf of
$\Lambda^{2, 1}_{II}$-graded vacua on $\widehat{\mathbf H}_1$,
whose vertical stalk at the generic Humbert point is
$\mathbf H_{\Delta_5}$ and whose horizontal factorisation data
lives on $\cA_2$.

\item[(A2)] \emph{$\Fact_{\Etwo}$ and $\Etwo$-chiral are the
same structure.} False. $\Etwo$-chiral at $d = 3$ is
factorisation homology of the Ran space of the \emph{target}
curve or surface (locally $\C$ or $\C^2$);
$\Fact_{\Etwo}(\widehat{\mathbf H}_1)$ is factorisation on the
\emph{moduli} $\cA_2$. The first is obstructed at $d = 3$ by
CY-A$_3$ and the framing defect $\bS^3 \to \bS^2$; the second
is unobstructed, because $\cA_2$ has trivial framing
obstructions and the BKM generators act by Hecke
correspondences automatically factorisation-local on $\cA_2$
(Ben-Zvi--Nadler--Preygel $2010$, \S$2.3$).

\item[(A3)] \emph{The Drinfeld centre already gave $\Etwo$ on
$\cZ(\Rep^{\Eone}(\mathbf H_{\Delta_5}))$, so the moduli-layer
$\Etwo$ is redundant.} The two $\Etwo$-structures are
categorically different and both load-bearing. (L2) is an
algebraic $\Etwo$ (Lurie--Dunn) on the representation-category
centre of (L1); (L3) is a geometric $\Etwo$
(Beilinson--Drinfeld factorisation) on the Siegel moduli. The
compatibility is the theorem of
\S\ref{subsec:wave6-x2-hidden-bnp}, realised by the
Ben-Zvi--Nadler--Preygel generalised Hecke action of
$\fg_{\Delta_5}$ on
$\IndCoh_{\Nilp}(\LocSys_{\Sp_4}(E))$.
\end{enumerate}
\end{proposition}

\begin{proof}[Proof sketch]
(A1) follows from Gaitsgory's $\infty$-categorical formulation
of factorisation on schemes $S$ over $\Spec \Z$: the cosheaf
condition is dimensional in the source, not in the target
algebra. (A2) follows from the distinction (Lurie \emph{HA}
$5.3$ vs.\ $5.5$) between $\Etwo$ as an operadic covariance of
algebras and $\Fact_{\Etwo}$ as a Ran-space cosheaf of
algebras; the former is (L1), the latter (L3). (A3) is
\S\ref{subsec:wave6-x2-hidden-bnp}.
\end{proof}

\subsection{Heal: the three-tier layering, explicit}
\label{subsec:wave6-x2-heal-layering}

\begin{definition}[Three-tier $E_n$-carrier data at
$(g, d, N) = (2, 3, \infty)$]
\label{def:three-tier-en-carrier}
For $K3 \times E$ the following three data, carrying three
distinct $E_n$-structures, are well-defined:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{ll}
\toprule
\textbf{Tier} & \textbf{Description} \\
\midrule
$\mathrm T_1$ (algebra) & $\mathbf H_{\Delta_5} =
\Phi_3(\Perf(K3 \times E))$, $\Eone$-chiral on $E$. \\
 & Character:
$\ch(\mathbf H_{\Delta_5}; \tau) = \phi_{0, 1}(\tau, z)$. \\
$\mathrm T_2$ (centre) &
$\cZ(\Rep^{\Eone}(\mathbf H_{\Delta_5}))$,
$\Etwo$-monoidal $\infty$-category. \\
 & Character: Verlinde-enhanced elliptic genus. \\
$\mathrm T_3$ (moduli) &
$\mathcal F_{\mathfrak g_{\Delta_5}} \in
\Fact_{\Etwo}(\widehat{\mathbf H}_1)$, factorisation cosheaf. \\
 & Character:
$\ch_{\cA_2}(\mathcal F) = \Delta_5(\Omega_2)$. \\
\bottomrule
\end{tabular}
\end{center}
\end{definition}

The three tiers are related by two functors:
\begin{equation*}
\mathrm T_1 \;\xrightarrow{\;\cZ \circ \Rep^{\Eone}\;}\;
\mathrm T_2 \;\xrightarrow{\;\Fact\text{-Hecke}\;}\;
\mathrm T_3,
\end{equation*}
the first Dunn--Lurie (centre-of-representations), the second
the factorisation-valued generalised Hecke functor of
Ben-Zvi--Nadler--Preygel. Neither is an equivalence: the first
adds a dimension of commutativity; the second projects to the
moduli-level cosheaf, retaining only Siegel-equivariant data.

\begin{proposition}[Character compatibility across three tiers]
\label{prop:wave6-x2-character-compat}
The graded-character images fit into a Saito--Kurokawa-type
commutative square:
\begin{equation*}
\begin{array}{ccc}
\phi_{0, 1}(\tau, z) &
\xrightarrow{\;\mathrm{Maass}\;} &
\Delta_5(\Omega_2) \\
\downarrow\scriptstyle{\mathrm{Verlinde}} & &
\downarrow\scriptstyle{\mathrm{Gritsenko}\ V_2} \\
\ch(\mathrm T_2) &
\xrightarrow{\;\text{genus-}2\ \text{assembly}\;} &
\Delta_{10}(\Omega_2).
\end{array}
\end{equation*}
Upper row: Maa{\ss} lift,
$\kappa_{\mathrm{BKM}}(\Delta_5) = c_1(0) / 2 = 5$. Lower row:
Gritsenko $V_2$-doubling,
$\kappa_{\mathrm{BKM}}(\Delta_{10}) = 10 =
\kappa_{\mathrm{BKM}}(\Delta_5 \cdot V_2 \Delta_5)$. Verticals:
Verlinde is the Drinfeld centre on $\Rep^{\Eone}$; genus-$2$
assembly is $\int_{\Sigma_2} \mathbf H_{\Delta_5} = \Delta_5^2$
via pants-decomposition factorisation homology.
\end{proposition}

\begin{proof}[Proof sketch]
Commutativity follows from the Saito--Kurokawa image criterion
(Eichler $1984$, Zagier $1981$): the Maa{\ss} lift of
$\phi_{0, 1}$ lands in $S_5^{\mathrm{SK}}(\Sp_4(\Z),
\nu_{\Delta_5})$ and commutes with Hecke operators on both
rows. Verlinde and $V_2$ agree on Fourier coefficients by
Theorem~\ref{wn:thm:bcov-siegel-k3-e}. Check at $c(0)$:
upper-right $c_1(0) = 10 \Rightarrow \kappa_{\mathrm{BKM}} =
5$; lower-right $c_{10}(0) = 20 \Rightarrow
\kappa_{\mathrm{BKM}} = 10$; ratio $10 / 5 = 2 = $ Gritsenko
$V_2$-multiplier.
\end{proof}

\subsection{Hidden: BNP generalised Hecke threads all three
tiers}
\label{subsec:wave6-x2-hidden-bnp}

\begin{theorem}[Hidden coherence: BNP threads the three tiers]
\label{thm:wave6-x2-bnp-threads-three-tiers}
Let $\mathcal H^{\BNP}_{\fg_{\Delta_5}} \colon
\fg_{\Delta_5} \otimes
\IndCoh_{\Nilp}(\LocSys_{\Sp_4}(E)) \to
\IndCoh_{\Nilp}(\LocSys_{\Sp_4}(E))$ denote the BNP generalised
Hecke action. The three $E_n$-tiers of
\Cref{def:three-tier-en-carrier} are images of a single
$\mathcal H^{\BNP}$-module under three functors:
\begin{equation*}
\begin{array}{lcl}
(\mathrm T_1)\ \mathbf H_{\Delta_5} & = &
R\Gamma(E;\ \text{local stalk of}\ \mathcal H^{\BNP}), \\
(\mathrm T_2)\ \cZ(\Rep^{\Eone}(\mathbf H_{\Delta_5})) & = &
\End_{\mathcal H^{\BNP}}(\IndCoh_{\Nilp}(\LocSys_{\Sp_4}(E))), \\
(\mathrm T_3)\ \mathcal F_{\fg_{\Delta_5}} & = &
\pi_!\ \mathcal H^{\BNP}|_{\widehat{\mathbf H}_1},
\end{array}
\end{equation*}
with $\pi \colon \LocSys_{\Sp_4}(E) \to \cA_2$ the moduli
projection (Narasimhan--Seshadri--Simpson, noncompact). The
commutator of $\mathcal H^{\BNP}$ is $r_{\mathrm{CY}}$, whose
genus-$2$ iteration is $\Delta_5 \cdot \Delta_5 = \Phi_{10}$;
hence
\begin{equation*}
\mathrm T_1 \otimes_{\mathcal H^{\BNP}} \mathrm T_1 \;\simeq\;
\mathrm T_2 \otimes_{\Fact} \mathrm T_3 \;\simeq\;
\Delta_{10}\text{-module}.
\end{equation*}
\end{theorem}

\begin{proof}[Proof sketch]
$(T_1)$ is the chiral-differential-operator presentation
$\mathbf H_{\Delta_5} = R\Gamma^{Q_{\BRST}}
(\Omega^{\mathrm{ch}}_{K3 \times E})$ (Ben-Zvi--Heluani
--Szczesny $2008$). $(T_2)$ is Lurie--Dunn plus
Bezrukavnikov--Finkelberg identification of the centre of a
Hecke category with endomorphisms of the Whittaker model; the
noncompact case for $E$ is Ben-Zvi--Nadler $2018$. $(T_3)$ is
Gaitsgory proper-base-change for the factorisation-valued
pushforward along
$\pi \colon \LocSys_{\Sp_4}(E) \to \cA_2$, carrying the
$\fg_{\Delta_5}$-action to a
$\Fact_{\Etwo}(\widehat{\mathbf H}_1)$-cosheaf (BNP $2010$,
\S$5.2$). The commutator relation
$[\mathcal H^{\BNP}_\alpha, \mathcal H^{\BNP}_\beta] =
r_{\mathrm{CY}}(\alpha, \beta)\ \mathcal H^{\BNP}_{\alpha +
\beta}$ is the Hecke-shadow of the classical Yang--Baxter
equation; iteration gives $\Delta_5^2 = \Phi_{10}$ at the
genus-$2$ character level.
\end{proof}

\begin{corollary}[The $E_2$-factorisation of V1 is consistent]
\label{cor:wave6-x2-v1-e2-fact-consistent}
$\mathcal F_{\mathfrak g_{\Delta_5}} \in
\Fact_{\Etwo}(\widehat{\mathbf H}_1)$ is consistent with
$\Phi_3(\Perf(K3 \times E)) \in \Eone\text{-}\ChirAlg$ of
Theorem~CY-A$_3$. The two $E_n$-structures live on different
tiers of \Cref{def:three-tier-en-carrier}, both threaded by
the single BNP generalised Hecke action of
\Cref{thm:wave6-x2-bnp-threads-three-tiers}. AP-CY$12$ is
not violated: the $\Etwo$ at $\mathrm T_3$ is a factorisation
covariance on the moduli $\cA_2$, not an algebra-internal
$\Etwo$-chiral structure on $\mathbf H_{\Delta_5}$.
\end{corollary}
