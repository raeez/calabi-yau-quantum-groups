\section*{$\CE^*_{\dbar}(\cE_{\hCS}) \;=\; E_3$-algebra of observables
  (Wave-13 G5/20)}

\paragraph{Theorem (main).}
Let $\cE_{\hCS} := \Omega^{0,\bullet}(\C^3, \g)$ with $\dbar$-differential
and pointwise bracket. The Dolbeault--chiral Chevalley--Eilenberg
cochain complex (E4 Def.~\ref{def:dolbeault-chiral-ce})
\[
  \CE^*_{\dbar, \chir}\!\bigl(\cE_{\hCS},\, \cO_{\C^3}\bigr)
\]
is an $E_3$-algebra in $\Ch(\Dolb)$, and the identification
\[
  \Obs_{\hCS}(\C^3) \;\simeq\; \CE^*_{\dbar, \chir}\!\bigl(\cE_{\hCS},\, \cO_{\C^3}\bigr)
  \qquad \text{in } \mathrm{Alg}_{E_3}\bigl(\Ch(\Dolb)\bigr)
\]
is an equivalence of $E_3$-algebras, with $E_3$-product implemented on
cochains by \emph{Čech-wedge of $\CE$-cochains along Dolbeault pairing}.

\paragraph{Cycle 1: Kontsevich's conjecture as formal input.}
\emph{Attack.} Kontsevich 1999 (``Operads and motives in deformation
quantization,'' Lett.\ Math.\ Phys.\ 48, \S4) conjectures: the
deformation complex of an $E_n$-algebra $A$ is an $E_{n+1}$-algebra.
Tamarkin 2003 (\emph{``Deformation complex of a d-algebra is a
(d+1)-algebra,''} arXiv:math/0010072) proves it for $A$ an $E_n$-algebra
in chain complexes via the $E_n$-operad's formality over $\mathbb{Q}$.
Francis 2013 (\emph{Compos.\ Math.} 149, Thm~1.1) refines:
$\HHch_{E_n}(A, A)[n] \simeq T_A \mathrm{Alg}_{E_n}$ is itself
$E_{n+1}$, by Koszul duality $E_n \leftrightarrow E_n^!$ with
$E_n^! \simeq E_n[n]$.
\emph{Heal.} Applied to $A = \cO_{\C^3}$ as $E_3$-algebra
(trivial, since $\cO$ is commutative hence $E_\infty$), the deformation
complex $\CE^*_{E_3}(\cO, \cO)$ is an $E_4$-algebra. This is \emph{not}
our target: hCS observables are $E_3$, not $E_4$. The operadic shift
$n \to n+1$ belongs to a different identification; here the
$E_3$-structure on $\CE^*_{\dbar,\chir}(\cE, \cO)$ is \emph{native},
inherited from $\cE$ being a holomorphic Lie algebroid on CY$_3$
rather than from iterated deformation theory.

\paragraph{Cycle 2: the specific correspondence.}
\emph{Attack.} For 6D hCS on $\C^3$, the classical BV complex is
\[
  (\Obs_{\hCS}^{cl}, d) = \bigl(\Sym^\bullet(\cE[1])^\vee, \; \dbar + \{S,-\}\bigr),
\]
and the Maurer--Cartan action $S = \int \Omega \wedge \langle \cA, \dbar \cA +
\tfrac{1}{3}[\cA,\cA]\rangle$ produces the classical observables as the
$\dbar$-enhanced CE cochains:
$\Obs_{\hCS}^{cl}(\C^3) \simeq \CE^*_{\dbar,\chir}(\cE_{\hCS}, \cO)$
(E4 Prop.~\ref{prop:hcs-obs-chiral-ce}).
The $E_3$-structure on the left is from E3 (Gwilliam--Williams 2021
Thm~2.5.5: holomorphic factorisation algebra on $\C^3 \leftrightarrow$
$E_3^{\mathrm{hol}}$-algebra in $\Ch(\Dolb)$).
\emph{Heal.} The $E_3$-structure transports across the quasi-isomorphism:
both sides are holomorphic factorisation algebras on $\C^3$, and the
quasi-isomorphism of E4 Prop.~\ref{prop:hcs-obs-chiral-ce} respects
disjoint-union structure maps because the Maurer--Cartan element $S$ is
local. Locality is checked directly: $S$ is an integral over $\C^3$ of
a local density, so for disjoint $U_1, U_2 \subset \C^3$,
$S|_{U_1 \sqcup U_2} = S|_{U_1} + S|_{U_2}$, and the CE differential
$d_{\chir}$ (E4 Def.~\ref{def:chiral-ce-cochains}) is built from the
chiral bracket supported on diagonals --- hence acts separately on
$U_1, U_2$.

\paragraph{Cycle 3: explicit $E_3$-product on $\CE$-cochains.}
\emph{Attack.} Need an explicit map
$\mu^{E_3}_{D_1, D_2}:
  \CE^*_{\dbar,\chir}(\cE|_{D_1}, \cO|_{D_1}) \otimes
  \CE^*_{\dbar,\chir}(\cE|_{D_2}, \cO|_{D_2})
  \to \CE^*_{\dbar,\chir}(\cE|_{D}, \cO|_{D})$
for disjoint polydisks $D_1, D_2 \subset D \subset \C^3$.
\emph{Construction.} Given $f_i \in \Hom_{\cD_{X^{n_i}}}
(\Sym^{n_i}_{\chir}\cE|_{D_i}, \Delta_*\cO|_{D_i})$ of Dolbeault type
$(0, q_i)$, define
\[
  (\mu^{E_3}_{D_1, D_2}(f_1 \otimes f_2))
     (l_1, \ldots, l_{n_1+n_2})
  \;:=\;
  \sum_{\substack{S \sqcup T = \{1,\ldots,n\} \\ |S| = n_1,\,|T| = n_2}}
     \epsilon(S,T)\,
     f_1\bigl(l_S|_{D_1}\bigr) \cdot_{\cO} f_2\bigl(l_T|_{D_2}\bigr),
\]
where $l_S|_{D_1}$ is the restriction of the $|S|$-tuple of chiral
sections to the configuration space $\Conf_{|S|}(D_1)$, the product
$\cdot_\cO$ is pointwise multiplication in $\cO_{\C^3}$ after pulling
back to $D$ along the disjoint-union inclusion
$D_1 \sqcup D_2 \hookrightarrow D$, and $\epsilon(S,T)$ is the
Koszul sign from reordering $\Lambda^{n_1+n_2} \to
\Lambda^{n_1} \otimes \Lambda^{n_2}$. The Dolbeault bidegrees add:
type $(0, q_1) \otimes (0, q_2) \to (0, q_1 + q_2)$.
\emph{Heal.} Associativity up to $\dbar$-exact homotopy:
for three disjoint $D_1, D_2, D_3$, the two bracketings
$(\mu(f_1, f_2), f_3)$ and $(f_1, \mu(f_2, f_3))$ differ by a
Čech coboundary on the nerve of $\{D_1, D_2, D_3\} \subset D$,
which is $\dbar$-exact by the Dolbeault Mayer--Vietoris sequence
on $\C^3$ (partitions-of-unity are $C^\infty$, so
$\dbar$-soft). Commutativity up to the $E_3$-twist:
$\mu^{E_3}_{D_1, D_2}(f_1, f_2) = (-1)^{|f_1||f_2|}
 \mu^{E_3}_{D_2, D_1}(f_2, f_1)$ after swap of configuration labels,
which is an $E_3$-operation of degree zero because $\Conf_2(\C^3)$
is simply connected (its fundamental group is the pure braid group
$P_2(\C^3) = \pi_1(\Conf_2(\C^3)) \cong \pi_1(S^5) = 0$; this is
the $n = 3$ Fadell--Neuwirth fibre). Higher homotopies: the
$m_k$-operations from E3 Cycle 5 act on the CE side by the
$\dbar$-flow on $\Conf_k(\C^3)$ composed with the CE cup product,
giving $E_3$-coherence to all orders.

\paragraph{Cycle 4: pairing with factorisation.}
\emph{Attack.} Verify the explicit $\mu^{E_3}$ agrees with the
factorisation structure map on $\Obs_{\hCS}$ under E4
Prop.~\ref{prop:hcs-obs-chiral-ce}. For disjoint $p_1, p_2 \in \C^3$
and observables $\cO_1 \in \Obs(U_1), \cO_2 \in \Obs(U_2)$ supported
near $p_1, p_2$, the BV convolution (E3 Cycle 4) is
$\mu^{\hCS}_{U_1, U_2}(\cO_1 \otimes \cO_2) = \cO_1 \cdot \cO_2$ with
Costello propagator smooth on $U_1 \times U_2$.
\emph{Heal.} Under the isomorphism $\Obs \simeq \CE^*_{\dbar,\chir}$,
an observable $\cO_i$ of degree $(n_i, q_i)$ corresponds to a CE
cochain $f_i$ of the same bidegree (Sym-dualisation of
polynomial observables in $\cE[1]$). The BV convolution
on $\cO_1 \cdot \cO_2$ becomes, on the CE side, exactly the
sum-over-shuffles formula of Cycle 3: each shuffle $S \sqcup T$
encodes one Wick contraction pattern in the BV integral, and the
Koszul sign is the fermion sign from reordering. The Costello
regularisation at coincident points is absent here because
$D_1 \cap D_2 = \emptyset$, so the cup product $\cdot_\cO$ is the
classical (unrenormalised) pointwise product.
\emph{Conclusion:} $\mu^{E_3} = \mu^{\hCS}$ under the identification.

\paragraph{Cycle 5: Koszul duality inside.}
\emph{Attack.} The $E_3$-Koszul dual of $\CE^*_{\dbar,\chir}(\cE, \cO)$
is predicted to be the higher enveloping algebra $U_{E_3}(\cE)$,
i.e.\ the chiral envelope in Beilinson--Drinfeld's sense lifted to the
holomorphic setting of $\C^3$.
\emph{Heal.} By Francis 2013 Thm~3.4 and Francis--Gaitsgory 2012
(Selecta Math.\ 18, Thm~6.2.1), the $E_n$-Koszul duality pairs
$E_n$-enveloping algebras $U_{E_n}(\cL)$ of $E_n$-Lie algebras $\cL$
with their CE-cochain algebras $\CE^*_{E_n}(\cL, k)$, via the bar--cobar
adjunction on the $E_n$-operad:
\[
  U_{E_n}(\cL) \;=\; \Omega_{E_n}\!\bigl(\CE^*_{E_n}(\cL, k)\bigr),
  \qquad
  \CE^*_{E_n}(\cL, k) \;=\; B_{E_n}\!\bigl(U_{E_n}(\cL)\bigr).
\]
Specialising to $n = 3$ and $\cL = \cE_{\hCS}$, the chiral envelope
$U^{\chir}(\cE_{\hCS})$ is the $E_3$-Koszul dual of our CE complex:
\[
  U^{\chir}(\cE_{\hCS}) \;\simeq\; \Omega_{E_3}\!\bigl(
    \CE^*_{\dbar,\chir}(\cE_{\hCS}, \cO)
  \bigr).
\]
This is the chain-level image of E4 Prop.~\ref{prop:koszul-factorisation}
(Kapranov--Vasserot chiral Hall extension). The factorisation-homology
side gives the $(\infty,1)$-categorical statement:
$\int_{\C^3} U^{\chir}(\cE_{\hCS}) \simeq
\CE^*_{\dbar,\chir}(\cE_{\hCS}, \C)^\vee$ as $E_3$-algebras.

\paragraph{Consequence for $\Phi$ and $\kch$.}
The identification $\Obs_{\hCS} = \CE^*_{\dbar,\chir}(\cE_{\hCS}, \cO)$
as $E_3$-algebras realises $\Phi(\mathrm{CY}_3) \to E_3$-chiral at the
pre-quotient level. CY-D stratification (CLAUDE.md): $\Phi$ outputs
$E_1$-chiral at $d \geq 3$ \emph{for the algebra}; the $E_3$ structure
lives on the factorisation algebra before passing to $A = H^0 \Obs$.
For compact CY$_3$ $X$:
\[
  \kch(A_X) \;=\; \chi\!\bigl(\CE^*_{\dbar,\chir}(\cE_X, \cO_X)\bigr)
   \;=\; \sum_q (-1)^q h^{0,q}(X) \cdot \dim \CE^*(\g)_{\text{hor}},
\]
the Hodge-supertrace identification (E4 \S7, matching the CLAUDE.md
essential constant $\kch(A_X) = \sum_q (-1)^q h^{0,q}(X)$ for the
abelian case $\dim\CE^*(\g)_{\mathrm{hor}} = 1$).

\paragraph{Scope.}
\emph{Chain-level}: explicit sum-over-shuffles for $\mu^{E_3}$ with
Koszul signs; $\dbar$-flow on $\Conf_k(\C^3)$ as $m_k$; Costello
propagator on disjoint polydisks. \emph{$(\infty,1)$-categorical}:
equivalence in $\mathrm{Alg}_{E_3}(\Ch(\Dolb))$ via
Gwilliam--Williams 2021 Thm~2.5.5 and Francis 2013 Thm~1.1.
Both lanes load-bearing (CLAUDE.md equal-status rule).
