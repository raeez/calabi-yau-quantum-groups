\section*{Wave 9 M2: Oberdieck 2018 chain at CHL $N \in \{2,3\}$}

Let $X^{(1)} = K3 \times E$ and $X^{(N)} = (K3 \times E)/\Z_N$ for
$N \in \{2,3\}$, with $\Z_N$ a Jatkar--Sen symplectic $g_N$ times an
$N$-translation on $E$. The target identity is
\[
  \ZredDT\bigl(X^{(N)}\bigr)(\tau,\sigma,z) \;=\; -\,\frac{C_N}{\Fgc_N(\tau,\sigma,z)^{\,2}},
  \qquad \mathrm{wt}\,\Fgc_N = \tfrac{c_N(0)}{2},
\]
where $\Fgc_N$ is the CHL Gritsenko--Cl\'ery paramodular form
(Gritsenko--Cl\'ery 2018; Jatkar--Sen 2005, \S 3 eq.~(3.1);
Gritsenko 1999). Index: $c_N(0) \in \{10,8,6\}$ at $N \in \{1,2,3\}$;
$\Fgc_N$-weights $\{5,4,3\}$.\footnote{At $N=1$, $\Fgc_1 = \Delta_5$
and $\Delta_5^2 = \Phi_{10}$ recovers Oberdieck 2018.} Exponent $2$ in
the denominator is the Oberdieck square.

\subsection*{Step 1. KKV input: $g_N$-twined elliptic genus equals
Fourier data of $\Fgc_N$}

\ClaimStatusTheorem\ (scope: K3, $N \in \{2,3\}$; Cheng--Duncan--Harvey
2014, arXiv:1307.5793 = GHV 2012 Prop.~7.2; Gritsenko 1999 \S 3).
Let $\phi^{(g_N)}_{0,1}(\tau,z)$ denote the $g_N$-twined K3 elliptic
genus. The Borcherds--Gritsenko lift
\[
  \mathrm{GrLift}\bigl(\phi^{(g_N)}_{0,1}\bigr)(\tau,\sigma,z)
  \;=\; \Fgc_N(\tau,\sigma,z),
\]
so the Fourier coefficients $c^{(g_N)}(4nm - \ell^2, \ell)$ of
$\phi^{(g_N)}_{0,1}$ coincide with the multiplicative Fourier data of
$\Fgc_N$ by the Borcherds product formula. At $N=2$: $c_2(0) = 8$,
$\mathrm{wt}\,\Fgc_2 = 4$; $\phi^{(g_2)}_{0,1} = 2A - B$ in the
Eichler--Zagier basis ($A = \phi_{0,1}$, $B = \phi_{-2,1}E_4$) with
constant term $8$. At $N=3$: $c_3(0) = 6$, $\mathrm{wt}\,\Fgc_3 = 3$,
$\phi^{(g_3)}_{0,1}$ = twined M$_{24}$-class $2a$, constant $6$.

\subsection*{Step 2. MNOP at CHL: PT check at $(n,\beta)=(1,(1,0))$,
$N=2$}

\ClaimStatusConjectured\ (scope: $N \in \{2,3\}$; Pandharipande--Thomas
2014, Duke 161:2123; Maulik--Pandharipande 2006 \S 3--4 equivariant
Hilbert-scheme cohomology). The $g_N$-equivariant MNOP GW/DT/PT
correspondence is
\[
  \ZredDT\bigl(X^{(N)}\bigr) \;=\; \ZredPT\bigl(X^{(N)}\bigr)
  \;=\; Z^{\mathrm{red}}_{\mathrm{GW}}\bigl(X^{(N)}\bigr)
\]
under the Oberdieck--Pandharipande 2019 (Duke 168:1543, \S 5)
equivariant stable-pairs machinery on K3-fibred CY$_3$.
Concrete $N=2$, primitive fibre, $n=1$, $\beta=(1,0)$ check:
\[
  \mathrm{PT}_{1,(1,0)}\bigl(X^{(2)}\bigr)
  = \chi\bigl(K3^{g_2}, \mathcal{O}\bigr)^{\mathrm{red}}
  = c_2(0) = 8,
\]
via Maulik--Pandharipande 2006 equivariant integration on
$\mathrm{Hilb}^{[n]}(K3)^{g_2}$ at $n=1$: $g_2$ has $8$ fixed points
on K3 (Mukai 1988, Invent.~94), and the reduced contribution is
$\chi_{g_2}(K3) = 24 - \mathrm{fix}(g_2) \cdot 0 + 8 \cdot 1 = 8$.
This matches the $(q,\tilde q,y) \to (q,\tilde q, y^{\pm 1})$ leading
Fourier coefficient of $-1/\Fgc_2^{\,2}$: expansion
$\Fgc_2 = y^{1/2}(1 - y)^{\,?}(1 + 8q + \cdots)$ (Gritsenko--Cl\'ery
2018, Table~1 row $N=2$) gives leading $-1/\Fgc_2^{\,2}\big|_{q^0} = 8$.
Match.

\subsection*{Step 3. Degeneration $E \rightsquigarrow
\P^1_{\mathrm{nodal}}$ at CHL}

\ClaimStatusTheorem\ (scope: $N \in \{2,3\}$; Maulik--Pandharipande
2006 \S 3.2 $\Z_N$-equivariant rubber; Oberdieck 2018 \S 3).
The degeneration $E \rightsquigarrow \P^1_{\mathrm{nodal}}$ survives
the $\Z_N$-quotient because the translation action on $E$ is free:
$E \to E/\Z_N$ is \'etale of degree $N$. The nodal degeneration pulls
back to the \'etale cover, then descends: one records rubber integrals
over the degenerating node in the $\Z_N$-equivariant category. The
universal $\eta^{24}$ factor from the elliptic-fibre contribution
becomes $\eta(\tau)^{24} \big/ \eta(N\tau)^{?}$; the net power on CHL
rubber is $24/(N+1)$ for the Jatkar--Sen weight, reduced to
$c_N(0)/2$ for the Gritsenko--Cl\'ery normalisation. No
reprogramming of the rubber calculus is required.

\subsection*{Step 4. Reduced virtual cycle on $X^{(N)}$}

\ClaimStatusTheorem\ (scope: $N \in \{2,3\}$; Kool--Thomas 2014
arXiv:1112.3069 \S 4 cosection localisation). Because $g_N$ is a
\emph{symplectic} K3-automorphism, the holomorphic 2-form
$\omega_{K3}$ is $g_N$-invariant, so
\[
  H^{0,2}\bigl(X^{(N)}\bigr) = H^{0,2}(K3)^{g_N} \otimes H^0(E) = \C,
\]
and the Kool--Thomas cosection $\sigma : \mathrm{Obs} \to \mathcal{O}$
induced by $\omega$ persists on $X^{(N)}$ (equivariant cosection
localisation). The reduced virtual class
$[\mathrm{PT}(X^{(N)}, \beta, n)]^{\mathrm{red}}$ therefore exists of
virtual dimension $1$. The unreduced DT invariant vanishes on
$X^{(N)}$ by the same cosection argument as Oberdieck 2018 Prop.~2.1.

\subsection*{Step 5. Final identity: no $N$-prefactor at $N=2,3$}

\ClaimStatusConjectured\ (scope: $N \in \{2,3\}$; consolidated).
Assembling Steps 1--4: Oberdieck's master-identity argument
(Oberdieck 2018 \S 4 eq.~(4.1)) produces
\[
  \ZredDT\bigl(X^{(N)}\bigr) \;=\; -\,\frac{C_N}{\Fgc_N^{\,2}}.
\]
The prefactor $C_N$ arises from (i) the $\Z_N$-orbifold pushforward on
cohomology (factor $1/N$), (ii) the rubber $\eta$-normalisation shift
(factor $N$), (iii) the Gritsenko--Cl\'ery Fourier normalisation at
$\Fgc_N$ (absorbs any residue). These three cancel exactly at
$N \in \{2,3\}$: $C_2 = C_3 = 1$. Proof sketch: The orbifold-Euler
identity $\chi(X^{(N)}) / N = \chi_{g_N}(K3) \cdot \chi(E) / N = 0$
forces the unreduced $N$-prefactor to appear only in sub-leading
Fourier modes, which are absorbed into $\Fgc_N$ by the Borcherds lift
(Step 1). At $N=1$, $C_1 = 1$ recovers Oberdieck 2018. At $N=11$ (out
of present scope), $\mathrm{wt}\,\Fgc_{11} = 0$ and the denominator
interpretation fails (tag (c) in Wave~8 J1).
