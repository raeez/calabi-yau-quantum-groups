\begin{abstract}
The MacMahon product, the point Donaldson-Thomas vertex of $\Cthree$,
the ordered bar Euler characteristic of the positive Hall half, and the
Virasoro shadow tower are four typed constructions. The common product
\[
  \Mac(q)=\prod_{n\geq 1}(1-q^n)^{-n}
\]
appears as the plane-partition series and as the positive Hall
character of $\CoHA(\Cthree)$; the Behrend-weighted point DT chamber
uses $\Mac(-q)$; the alternating bar Euler characteristic is
$\Mac(q)^{-1}$; the Virasoro shadow coefficients come from
\[
  f(t)^2=1+12t+\frac{1052}{27}t^2.
\]
The Hall equality is
$\CoHA(\Cthree)\cong\Yp(\fglone)$, the positive half of the affine
Yangian. The vertex algebra $\Winf$ is reached from the completed
Drinfeld double by a representation or vacuum-evaluation functor. For
$d\geq 3$ the native CY-to-chiral output on a witnessed specialisation
is $\Eone$; the braided
$\Etwo$ structure belongs to the centre or representation layer.
\end{abstract}

\begin{theorem}[The positive Hall half of $\Cthree$]
\label{thm:c3-positive-half}
Let $T^{\mathrm{CY}}\subset(\mathbb{C}^{\times})^3$ be the Calabi-Yau
subtorus with weights $\epsilon_1+\epsilon_2+\epsilon_3=0$. The
critical equivariant cohomological Hall algebra of the one-vertex
three-loop quiver with potential
\[
  W=\operatorname{tr}(X[Y,Z])
\]
is, as an associative Hall algebra over the Calabi-Yau equivariant
field after the standard shuffle localisation and parameter
identification,
\[
  \CoHA_{T^{\mathrm{CY}}}(\Cthree)\cong\Yp(\fglone).
\]
The equality stops at the positive Hall half. The full affine Yangian
and the vertex-algebra evaluation layer are obtained by the typed chain
\[
  \Yp(\fglone)
  \hookrightarrow
  \Yfull(\fglone)=\Ym(\fglone)\widehat{\bowtie}\Yz(\fglone)
     \widehat{\bowtie}\Yp(\fglone)
  \xrightarrow{\ \rho_\lambda\ }
  \operatorname{End}\bigl(\Winf[\lambda]\textup{-vac}\bigr).
\]
\end{theorem}

\begin{proof}
The Schiffmann-Vasserot shuffle theorem identifies the critical Hall
convolution algebra of the Jordan three-loop potential with the
positive shuffle algebra, hence with $\Yp(\fglone)$. The generators in
charge one are the positive Drinfeld current coefficients. The negative
current and the Cartan current require a non-degenerate Hall pairing,
Drinfeld double, and completion. A vacuum or corner representation of
the completed double supplies the state-field correspondence on the
module carrying $\Winf[\lambda]$.
\end{proof}

\begin{proposition}[Native operadic level]
\label{prop:native-e1}
For a CY$_d$ category on a verified two-stage locus with witnessed
admissible datum $(\Sigma_{d-1},C)$,
\[
  \Phi_d^{(\Sigma_{d-1},C)}
  =
  \operatorname{Sp}^{\mathrm{ch}}_{\Sigma_{d-1},C}\circ\Phi_d^{FA}
\]
has residual operadic level
\[
  n_{\mathrm{native}}(d)=
  \begin{cases}
  \infty,& d=1,\\
  2,& d=2,\\
  1,& d\geq 3.
  \end{cases}
\]
Thus a specialised CY$_3$ output $A$ is an $\Eone$-chiral algebra on
the reference curve. The braided structure is recovered on
\[
  \Zcen\bigl(\Rep^{\Eone}(A)\bigr),
\]
where half-braidings carry the $R$-matrix data.
\end{proposition}

\begin{proof}
The PTVV shifted symplectic form on the derived moduli stack has shift
$2-d$, and the Hochschild cochain side carries the corresponding
$E_d$ structure. Stage-two holomorphic factorisation homology over the
complex $(d-1)$-dimensional specialisation cycle lowers the residual
operadic level to $E_{d-(d-1)}=E_1$ for $d\geq3$. The centre statement
is the Ben-Zvi-Francis-Nadler mechanism applied to the
$\Eone$-monoidal representation category. The proposition records this
object-level specialisation statement and leaves arbitrary
CY$_3$-morphism functoriality outside its scope.
\end{proof}

\section{MacMahon Asymptotics}
\label{sec:macmahon-asymptotics}

\begin{theorem}[Mellin expansion of MacMahon]
\label{thm:macmahon-mellin}
As $\beta\to0^+$,
\[
  \log \Mac(e^{-\beta})
  \sim
  \frac{\zeta(3)}{\beta^2}
  +\frac{1}{12}\log\beta
  +\zeta'(-1)
  +\sum_{g\geq2}F_g^{\Mac}\,\beta^{2g-2},
\]
with
\[
  F_g^{\Mac}
  =
  \frac{\zeta(3-2g)\zeta(1-2g)}{(2g-2)!}
  =
  \frac{\Bnum{2g}\Bnum{2g-2}}
       {2g(2g-2)(2g-2)!}.
\]
In particular $F_2^{\Mac}=-1/2880$.
\end{theorem}

\begin{proof}
From
\[
  \log \Mac(e^{-\beta})
  =
  \sum_{n,k\geq1}\frac{n}{k}e^{-nk\beta}
\]
one obtains the Mellin representation
\[
  \log \Mac(e^{-\beta})
  =
  \frac{1}{2\pi i}\int_{\sigma-i\infty}^{\sigma+i\infty}
  \Gamma(s)\zeta(s+1)\zeta(s-1)\beta^{-s}\,ds
  \qquad(\sigma>2).
\]
Moving the contour left gives the pole at $s=2$, the double pole at
$s=0$, and the poles of $\Gamma(s)$ at $s=2-2g$ for $g\geq2$. The
residue at $s=0$ has coefficient $+1/12$ in front of $\log\beta$ since
$\zeta(-1)=-1/12$. The residues at $s=2-2g$ give
$\zeta(3-2g)\zeta(1-2g)/(2g-2)!$. The Bernoulli form follows from
$\zeta(1-2m)=-B_{2m}/(2m)$. Crossing finitely many poles gives the
expansion to the corresponding order; the full Bernoulli series is an
asymptotic series.
\end{proof}

\begin{proposition}[Scalar shadows and Bernoulli growth]
\label{prop:scalar-shadow-growth}
Let
\[
  \lambda_g^{(1)}
  :=
  \frac{|\Bnum{2g}|}{2g(2g-2)!}
\]
be the one-Bernoulli genus normalisation determined by the genus-one
term. Then
\[
  \frac{|F_g^{\Mac}|}{\lambda_g^{(1)}}
  =
  \frac{|\Bnum{2g-2}|}{2g-2}
  \sim
  \frac{2(2g-2)!}{(2\pi)^{2g-2}(2g-2)}.
\]
The ratio has factorial growth. In the constant-map normalisation that
already contains the double Bernoulli product
\[
  \lambda_g^{(2)}
  :=
  \frac{|\Bnum{2g}\Bnum{2g-2}|}
       {2g(2g-2)(2g-2)!},
\]
the quotient $F_g^{\Mac}/\lambda_g^{(2)}$ is the sign forced by the
Bernoulli convention. The normalisation must therefore be part of any
claimed genus-by-genus comparison. A scalar comparison with
$\lambda_g^{(1)}$ cannot be independent of $g$.
\end{proposition}

\begin{proof}
The first formula is immediate from Theorem~\ref{thm:macmahon-mellin}.
The asymptotic uses
$|\Bnum{2m}|\sim2(2m)!/(2\pi)^{2m}$. The second statement is the same
identity after moving the second Bernoulli factor into the comparison
sequence.
\end{proof}

\section{The Ordered Bar Complex of $\Cthree$}
\label{sec:c3-bar-complex}

\begin{theorem}[Positive character and ordered bar Euler]
\label{thm:bar-poincare-euler}
For $A_{\Cthree}:=\Yp(\fglone)$ on the toric $\Cthree$ Hall locus,
\[
  \ch_q A_{\Cthree}
  =
  \Mac(q),
\]
where the degree-$n$ Hall basis is indexed by plane partitions of size
$n$. The ordered bar complex itself is used through its Euler
characteristic:
\[
  B^{\Eone}(A_{\Cthree})^k=(\overline{A}_{\Cthree})^{\otimes k}.
\]
The alternating bar Euler characteristic is the reciprocal product
\[
  \chi_q B^{\Eone}(A_{\Cthree})
  =
  \sum_{k\geq0}(-1)^k\ch(B^k)_q
  =
  \Mac(q)^{-1}
  =
  \prod_{n\geq1}(1-q^n)^n.
\]
\end{theorem}

\begin{proof}
The first equality is the crystal basis of the positive shuffle
algebra: fixed points of the torus action on the Hilbert tower are
three-dimensional partitions. For the ordered bar chains,
\[
  \sum_{k\geq0}(-1)^k\ch_q
  \bigl((\overline{A}_{\Cthree})^{\otimes k}\bigr)
  =
  \sum_{k\geq0}(-1)^k(\ch_q A_{\Cthree}-1)^k
  =
  \frac{1}{\ch_q A_{\Cthree}}.
\]
Substituting $\ch_q A_{\Cthree}=\Mac(q)$ gives the reciprocal product.
This is an Euler-characteristic statement; the plane partitions index
the Hall basis of $A_{\Cthree}$.
\end{proof}

\begin{corollary}[Primitive BPS multiplicities]
\label{cor:primitive-bps}
The plethystic logarithm is
\[
  \PLog\,\Mac(q)=\sum_{n\geq1}nq^n.
\]
Thus the primitive point-class BPS multiplicity in degree $n$ is
\[
  \Omega(n)=n.
\]
\end{corollary}

\begin{proof}
For a product $\prod_{n\geq1}(1-q^n)^{-a_n}$, the plethystic logarithm
is $\sum_{n\geq1}a_nq^n$. Here $a_n=n$.
\end{proof}

\section{The Virasoro Shadow Tower}
\label{sec:virasoro-shadow}

\begin{definition}[The $c=1$ quadratic landscape]
\label{def:quadratic-landscape}
On the stress-tensor line of the $\Winf$ evaluation layer at $c=1$ set
\[
  \kch^{T}:=\frac{c}{2}=\frac12,\qquad
  \alpha:=2c=2,\qquad
  S_4:=\frac{10}{c(5c+22)}=\frac{10}{27}.
\]
The quadratic landscape is
\[
  Q_L(t)
  =
  4(\kch^{T})^2+12\kch^{T}\alpha t
  +(9\alpha^2+16\kch^{T}S_4)t^2
  =
  1+12t+\frac{1052}{27}t^2.
\]
Write
\[
  f(t)=\sqrt{Q_L(t)}=\sum_{n\geq0}a_nt^n.
\]
For $k\geq5$ the post-quartic Newton shadow coefficient is
\[
  m_k^{\mathrm{Newt}}(T,\ldots,T)=a_{k-2}T,\qquad
  S_k^{\mathrm{Newt}}=\frac{a_{k-2}}{k}.
\]
The lower arities belong to the OPE normalisation of the $A_\infty$ bar
operation.
\end{definition}

\begin{theorem}[Newton reconstruction]
\label{thm:newton-shadow}
The coefficients of $f(t)$ are determined by
\[
  a_0=1,\qquad a_1=6,\qquad a_2=\frac{40}{27},
\]
and, for $n\geq3$,
\[
  a_n=-\frac{1}{2a_0}\sum_{i=1}^{n-1}a_ia_{n-i}.
\]
Consequently
\[
\begin{array}{c|c|c}
\toprule
k & m_k^{\mathrm{Newt}}(T,\ldots,T)/T & S_k^{\mathrm{Newt}}\\
\midrule
5 & -80/9 & -16/9\\
6 & 38080/729 & 19040/2187\\
7 & -24320/81 & -24320/567\\
8 & 33157760/19683 & 4144720/19683\\
\bottomrule
\end{array}
\]
\end{theorem}

\begin{proof}
The identity $f(t)^2=Q_L(t)$ gives
\[
  2a_0a_n+\sum_{i=1}^{n-1}a_ia_{n-i}=0
  \qquad(n\geq3),
\]
because $Q_L$ has degree $2$. The first three coefficients come from
matching the coefficients of $t^0,t^1,t^2$. Substitution gives
\[
  a_3=-\frac{2\cdot6\cdot(40/27)}{2}=-\frac{80}{9},
\]
\[
  a_4=-\frac{2\cdot6\cdot(-80/9)+(40/27)^2}{2}
      =\frac{38080}{729},
\]
\[
  a_5=-\frac{2\cdot6\cdot(38080/729)
        +2\cdot(40/27)\cdot(-80/9)}{2}
      =-\frac{24320}{81},
\]
and
\[
  a_6=-\frac{2\cdot6\cdot(-24320/81)
        +2\cdot(40/27)\cdot(38080/729)
        +(-80/9)^2}{2}
      =\frac{33157760}{19683}.
\]
The relation $S_k^{\mathrm{Newt}}=a_{k-2}/k$ gives the displayed
post-quartic shadow values.
\end{proof}

\begin{proposition}[Convergence radius]
\label{prop:shadow-radius}
The shadow generating function $f(t)$ is analytic for
\[
  |t|<\rho,\qquad
  \rho=\sqrt{\frac{27}{1052}}
       =\frac{3\sqrt{3}}{2\sqrt{263}}.
\]
\end{proposition}

\begin{proof}
The branch points of $f(t)=\sqrt{1+12t+(1052/27)t^2}$ are the two roots
of $Q_L$. Their product is $27/1052$. Since the discriminant
\[
  12^2-4\cdot\frac{1052}{27}=-\frac{320}{27}
\]
is negative, the roots are complex conjugates with equal modulus
$\sqrt{27/1052}$.
\end{proof}

\section{The Connected Wick Witness}
\label{sec:wick-witness}

\begin{definition}[Connected stress-tensor cumulant]
\label{def:connected-cumulant}
Let $J(z)J(w)\sim(z-w)^{-2}$ and
$T(z)=-\frac12{:}J(z)^2{:}$. For $k$ marked points set
\[
  G_k^{\conn}(z_1,\ldots,z_k)
  =
  \left(-\frac12\right)^k
  \sum_{\pi\in\mathcal{P}^{\conn}_{k}}
  \prod_{\{(i,r),(j,s)\}\in\pi}(z_i-z_j)^{-2},
\]
where $\mathcal{P}^{\conn}_{k}$ is the set of perfect pairings of the
$2k$ labelled half-edges whose incidence graph on the $k$ vertices is
connected, with pairings inside a single vertex excluded.
\end{definition}

\begin{proposition}[Five-point value]
\label{prop:g5-value}
The connected five-point Wick cumulant at the integral configuration
$(1,2,3,4,5)$ is
\[
  G_5^{\conn}(1,2,3,4,5)
  =
  \frac{775}{5184}
  =
  \frac{5^2\cdot31}{2^6\cdot3^4}.
\]
\end{proposition}

\begin{proof}
The Wick formula in Definition~\ref{def:connected-cumulant} is an
explicit finite sum over perfect matchings of ten half-edges. Each
term is a rational function in the differences $z_i-z_j$. Evaluation
at $(1,2,3,4,5)$ gives the displayed rational number after ordinary
cumulant subtraction. This is the exact finite enumeration implemented
by the local Wick engine
\texttt{compute/lib/virasoro\_m5\_five\_point.py}; it uses the
free-field Wick rule alone.
\end{proof}

\begin{remark}[Comparison with $m_5$]
\label{rem:g5-m5-comparison}
The coefficient $m_5=-80/9$ is the Newton coefficient $a_3$ of
Theorem~\ref{thm:newton-shadow}. The rational number
$G_5^{\conn}=775/5184$ is a connected free-field cumulant. A comparison
from the second number to the first must pass through the Virasoro
normalisation, projection to the $T$-line, the $S_4=10/27$ level-two
constant, and the residue map from configuration-space cumulants to
minimal-model operations. The two numbers live in different vector
spaces before this comparison map is supplied.
\end{remark}

\section{Borcherds Constants and the $K3\times E$ Separation}
\label{sec:borcherds-k3e}

\begin{proposition}[The five $K3\times E$ readings]
\label{prop:k3e-readings}
For $Y=K3\times E$ the relevant scalar readings are
\[
\begin{array}{c|c|l}
\toprule
\textup{invariant} & \textup{value} & \textup{construction}\\
\midrule
\kcat(Y) & 0 &
\chi(\mathcal{O}_{K3})\chi(\mathcal{O}_{E})=2\cdot0\\
\kch(Y) & 0 &
\sum_q(-1)^qh^{0,q}(Y)=1-1+1-1\\
\kch^{\mathrm{Heis}}(Y) & 3 &
\textup{K3-fibre Heisenberg specialisation}\\
\kBKM(\Delta_5) & 5 &
c_1(0)/2=10/2=\operatorname{wt}(\Delta_5)\\
\kfiber(Y) & 24 &
\operatorname{rk}\widetilde H(K3,\mathbb{Z})\\
\bottomrule
\end{array}
\]
These are parallel constructions with separate comparison maps.
\end{proposition}

\begin{proof}
The first row is K\"unneth multiplicativity of
$\chi(\mathcal{O}_{K3\times E})$. The second row is the compact
CY$_3$ Hodge supertrace:
$h^{0,\bullet}(K3\times E)=(1,1,1,1)$ after K\"unneth, so the alternating
sum is zero. The Heisenberg row is a specialisation along the K3-fibre
Mukai sector. The Borcherds row is the weight theorem applied to the
Gritsenko $\Delta_5$ input with $c_1(0)=10$. The fibre row is the rank
of the Mukai lattice
$H^0(K3,\mathbb{Z})\oplus H^2(K3,\mathbb{Z})\oplus H^4(K3,\mathbb{Z})$.
\end{proof}

\begin{proposition}[Primitive CHL constants]
\label{prop:chl-constants}
For primitive $N\in\{1,2,3,4,6\}$ in the five-row CHL ladder with the
Gritsenko-Nikulin Jacobi normalisation,
\[
  \bigl(c_1(0),c_2(0),c_3(0),c_4(0),c_6(0)\bigr)
  =
  (10,8,6,4,2),
\]
and
\[
  \bigl(\kBKM(\Phi_1),\kBKM(\Phi_2),\kBKM(\Phi_3),
        \kBKM(\Phi_4),\kBKM(\Phi_6)\bigr)
  =
  (5,4,3,2,1).
\]
The relation is
\[
  \kBKM(\Phi_N)=\frac{c_N(0)}{2}.
\]
The five-row CHL ladder and the eight-row Gritsenko-Clery
diagonal-divisor catalogue are separate tables; the displayed tuple
concerns only the CHL ladder.
For the untwined square normalisation,
\[
  D_5=64^{-1}\Delta_5,\qquad
  \Phi_{10}=\Delta_5^2.
\]
\end{proposition}

\begin{proof}
The Borcherds weight theorem assigns half the constant Fourier
coefficient of the Jacobi input to the automorphic product weight. For
$N=1$ this is Gritsenko's $\Delta_5$ with $c_1(0)=10$, hence weight
$5$. The primitive CHL constants listed above are the corresponding
constant terms for $N=1,2,3,4,6$. The scalar $D_5$ is the
Oberdieck-Pandharipande monic normalisation of $\Delta_5$, and the
untwined genus-two form is the square $\Delta_5^2$.
\end{proof}

\section{Consequences}
\label{sec:consequences}

\begin{enumerate}[label=\textup{(\arabic*)}]
\item The MacMahon product, the Behrend-weighted DT vertex, the ordered
bar Euler characteristic, and the Virasoro shadow tower admit comparison
only with their chamber, sign, normalisation, and comparison map
specified.
\item The equality $\CoHA(\Cthree)=\Yp(\fglone)$ is an equality of
positive associative Hall algebras. The vertex algebra $\Winf$ is a
completed double or centre evaluation layer.
\item For $d\geq3$, a specialised CY-to-chiral output is $\Eone$. The
braided $\Etwo$ structure is carried by the derived chiral centre or by
the completed representation category.
\item The numbers
\[
  m_5=-\frac{80}{9},\qquad
  m_6=\frac{38080}{729},\qquad
  m_7=-\frac{24320}{81},\qquad
  m_8=\frac{33157760}{19683},\qquad
  G_5^{\conn}=\frac{775}{5184}
\]
are exact, typed constants. The first four are post-quartic Newton
coefficients of the Virasoro shadow tower; the last is a Wick cumulant.
\item The $K3\times E$ invariants
$0,0,3,5,24$ and the CHL constants
$(c_N(0))=(10,8,6,4,2)$ arise from distinct geometric and automorphic
constructions. The CHL tuple is separate from the Gritsenko-Clery
diagonal-divisor catalogue.
\end{enumerate}
