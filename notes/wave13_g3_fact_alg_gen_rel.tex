\section*{Generators-and-relations presentation of $\Obs_{\hCS}^{\mathrm{Ab}}(\C^3)$
(Wave-13 G3/20)}

\paragraph{Theorem (main).}
The 6D holomorphic Chern--Simons factorisation algebra on $(\C^3, \Omega)$
with Abelian gauge $\g = \C$ admits the presentation
\[
  \Obs_{\hCS}^{\mathrm{Ab}}
  \;=\;
  \Sym\!\bigl(\, \cE^{\vee}[1] \,\bigr)\bigl\llbracket \hb \bigr\rrbracket
  \;\Big/\;
  \bigl(\, Q_{\dbar} + \hb\,\Delta_{\BM} \,\bigr),
\]
generated by Dolbeault modes $\cA_{\bar{I}, \vec{\alpha}}(p)$ and ghost
modes $c_{\vec{\alpha}}(p)$, constrained by the single Bochner--Martinelli
OPE relation of Cycle~2 below. The object is $E_1$-chiral (CY-D
dimensional stratification at $d = 3$) with $\kch(\C^3) = \chi(\cO_{\C^3}) = 1$.

\paragraph{Cycle 1: generators.}
\emph{Attack.} The BV field space is
$\cE = \Omega^{0,\bullet}(\C^3, \C[1]) = \cA \oplus c[1]$ with
$\cA \in \Omega^{0,1}$ the gauge field and $c \in \Omega^{0,0}$ the ghost
(BV degree $-1$ after the shift). Linear observables at a point
$p \in \C^3$ are the continuous dual, generated by the derivative-evaluation
functionals
\[
  \cA_{\bar{I}, \vec{\alpha}}(p)\!: \cA \mapsto
    \bigl(\partial_z^{\vec{\alpha}} \cA\bigr)_{\bar{I}}(p),
  \qquad
  c_{\vec{\alpha}}(p)\!: c \mapsto
    \bigl(\partial_z^{\vec{\alpha}} c\bigr)(p),
\]
where $\vec{\alpha} \in \N^3$ is a holomorphic multi-index and
$\bar{I} \in \{\bar{1}, \bar{2}, \bar{3}\}$ the antiholomorphic leg.
\emph{Heal.} Antiholomorphic derivatives are redundant: $\dbar \cA$ is
$Q$-exact, so $\partial_{\bar{z}}^{\vec{\beta}} \cA$ vanishes on-shell for
$\vec{\beta} \neq 0$. The generator set is finite-per-jet-order:
at jet order $N$, the generators span
$\bigoplus_{|\vec{\alpha}| \leq N}(\C^3 \oplus \C)$.

\paragraph{Cycle 2: OPE relation (Bochner--Martinelli).}
\emph{Attack.} The only non-trivial relation arises from the BV bracket
$\{\cA(z), \cA(w)\}$ mediated by the Costello--Gwilliam Vol~II Prop~5.4.2
propagator, the Bochner--Martinelli kernel
\[
  P_{\BM}(z, w) \;=\;
  \frac{2}{(2\pi i)^3}
  \sum_{k=1}^{3}(-1)^{k-1}
  \frac{\overline{z_k - w_k}}{\lvert z - w \rvert^6}\,
  d\bar{z}_1 \wedge \widehat{d\bar{z}_k} \wedge d\bar{z}_2 \wedge d\bar{z}_3,
\]
so that the double-contraction OPE, modulo $\dbar$-exact terms, reads
\begin{equation}
\label{eq:bm-ope}
  \cA(z)\,\cA(w)
  \;\sim\;
  \hb\,\frac{2}{(2\pi i)^3}\,
  \frac{\overline{(z - w)}\cdot d\bar{z}\big|_{\widehat{k}}}
       {\lvert z - w \rvert^6}
  \cdot \mathbf{1}
  \;+\; Q\text{-exact},
\end{equation}
i.e.\ the pole is degree $-5$ in $|z - w|$ (not $|z-w|^{-6}$ as the naive
scalar would suggest: BM carries one factor of $\overline{(z-w)}$ in the
numerator, giving effective $|z-w|^{-5}$ behaviour).
\emph{Heal.} On-shell, $Q\cA = \dbar \cA$, so \eqref{eq:bm-ope} is an
equality of cohomology classes; the $(0,2)$-form content fixes the
tensorial structure. In the Abelian case the cubic vertex vanishes
($[\cA, \cA]_{\g} = 0$ for $\g = \C$) and \eqref{eq:bm-ope} is the
\emph{only} relation: Costello 2013 Thm~9.2.1 gives no level-shift,
and the Costello--Li 2016 (arXiv:1606.00365) Thm~1.1 $\beta\gamma$-VOA
identification along a line defect is a quotient of the free algebra
presentation here.

\paragraph{Cycle 3: point-support observables.}
\emph{Attack.} For a single point $p \in \C^3$,
Costello--Gwilliam Vol~II \S5.6 identifies
\[
  \Obs_{\hCS}^{\mathrm{Ab}}(\{p\})
  \;\cong\;
  \C\!\bigl\llbracket
    \partial_z^{\vec{\alpha}} \cA_{\bar{I}}(p),\;
    \partial_z^{\vec{\alpha}} c(p)
  \bigr\rrbracket\!\bigl\llbracket \hb \bigr\rrbracket,
\]
the (graded-)symmetric polynomial algebra on jet-evaluation generators,
with degree count $|\cA_{\bar{I}, \vec{\alpha}}| = 0$ and $|c_{\vec{\alpha}}| = -1$.
\emph{Heal.} The completion $\llbracket - \rrbracket$ is in the polynomial
jet variables; higher-derivative generators contribute at successively
positive filtration weights. At $p$, the BM propagator degenerates, so
the OPE relation \eqref{eq:bm-ope} reduces to the trivial relation
$\cA(p) \cdot \cA(p) = 0$ (nilpotent fermionic square) only in the ghost
sector: $c(p)^2 = 0$. The bosonic $\cA(p) \cdot \cA(p)$ is unconstrained
at a single point and lives as a polynomial generator.

\paragraph{Cycle 4: disk-to-sphere transport via the factorisation structure.}
\emph{Attack.} Let $D_r(p) \subset \C^3$ be an open polydisk of radius $r$
at $p$. Factorisation restriction gives a map
\[
  \Obs_{\hCS}^{\mathrm{Ab}}(\{p\})
  \;\xrightarrow{\;\iota_p^{D_r}\;}\;
  \Obs_{\hCS}^{\mathrm{Ab}}(D_r(p)),
\]
which at the level of generators extends a jet at $p$ to a holomorphic
germ on $D_r$. Across two disjoint disks $D_1 \sqcup D_2 \subset V$,
\[
  \Obs_{\hCS}^{\mathrm{Ab}}(D_1) \otimes \Obs_{\hCS}^{\mathrm{Ab}}(D_2)
  \;\xrightarrow{\;m_V\;}\;
  \Obs_{\hCS}^{\mathrm{Ab}}(V),
  \qquad
  m_V(\cO_1 \otimes \cO_2) = \cO_1 \cdot \cO_2 \mod \Delta_{\BM},
\]
the symmetric-algebra product modulo the BV Laplacian built from $P_{\BM}$.
\emph{Heal.} For $V = \C^3$ (the ``sphere'' in the sense of compactifying
$\C^3$ to $\mathbb{CP}^3$ with boundary $\mathbb{CP}^2_\infty$),
Costello--Gwilliam Vol~II Thm~5.5.4 gives
$\Obs_{\hCS}^{\mathrm{Ab}}(\C^3) = \mathrm{colim}_{\{D_i\}} \Obs(D_i)$
over Weiss covers. Generators at global level are \emph{holomorphic polynomials}
$\cA_{\bar{I}}[\vec{z}]$ and $c[\vec{z}]$ (jets truncated to polynomial
order), with relations \eqref{eq:bm-ope} enforced as integrated pair
contractions
$\oint_{|z-w| = \epsilon} P_{\BM}(z, w) \, \cA(z) \otimes \cA(w)$ for
every $\epsilon > 0$, independent of $\epsilon$ on-shell by the homotopy
invariance of Dolbeault cohomology.

\paragraph{Cycle 5: Koszul-duality completeness.}
\emph{Attack.} The Abelian free $L_\infty$-algebra $\cE = \Omega^{0,\bullet}(\C^3, \C)$
with trivial bracket and linear differential $\dbar$ is Koszul-dual, by
Loday--Vallette \emph{Algebraic Operads} Thm~10.4.5 lifted to local
$L_\infty$-algebras (Costello--Gwilliam Vol~II Prop~5.5.9), to the
completed cofree coalgebra $\Sym(\cE^{\vee}[1])$ with CE differential.
Hence
\[
  \Obs_{\hCS}^{\mathrm{Ab}}
  \;\simeq\;
  \CE^{\bullet}_{\dbar}(\cE)\bigl\llbracket \hb \bigr\rrbracket,
\]
recovering the identification of the E2 inscription
(\texttt{wave13\_e2} Thm~5.1) with the generators-and-relations side.
\emph{Heal.} Completeness: the generators $\{\cA_{\bar{I}, \vec{\alpha}}(p),
c_{\vec{\alpha}}(p)\}_{p \in \C^3, \vec{\alpha} \in \N^3, \bar{I}}$ span
the underlying graded space; the single relation \eqref{eq:bm-ope} is the
quadratic part of the Maurer--Cartan / CE differential applied to
$\cE^{\vee}[1]$; no cubic or higher relations appear because $\g = \C$ kills
the trilinear CE bracket $\ell_2$. The Koszul-dual Lie algebra recovers
$\cE$ in the free-resolution sense: $\Omega(\mathrm{coBar})(\Obs) \simeq \cE$.

\paragraph{Explicit presentation.}
In summary, as a filtered complete dg-algebra over $\C\llbracket \hb \rrbracket$:
\begin{align*}
  \text{Generators:} \quad
  & \{\cA_{\bar{I}, \vec{\alpha}}(p)\}_{\bar{I} \in \{\bar{1},\bar{2},\bar{3}\},\,
      \vec{\alpha} \in \N^3,\, p \in \C^3}
    && \text{(BV degree } 0\text{)} \\
  & \{c_{\vec{\alpha}}(p)\}_{\vec{\alpha} \in \N^3,\, p \in \C^3}
    && \text{(BV degree } -1\text{)} \\
  \text{Relations:} \quad
  & \cA(z) \, \cA(w) - \hb \cdot P_{\BM}(z, w)\cdot \mathbf{1}
    \equiv 0 \pmod{Q\text{-exact}}
    && \text{(Bochner--Martinelli OPE)} \\
  & c(p)^2 = 0
    && \text{(ghost nilpotency)} \\
  \text{Differential:} \quad
  & Q = \dbar + \hb \Delta_{\BM},
    \quad \dbar c = 0, \quad \dbar \cA = (\text{free})
    && \text{(CE differential)}.
\end{align*}

\paragraph{Consequence for $\Phi$.}
The Abelian case is the linearisation of $\Phi$ at the trivial
$\g = 0$ locus of the CY-to-chiral functor. Along a line defect
$\C_{z_1} \subset \C^3$ it reduces to the $\beta\gamma$ system of
central charge $c = 2\kch = 2$ (Costello--Li 2016 Thm~1.1), and the
Bochner--Martinelli pole \eqref{eq:bm-ope} descends to the standard
$\beta(z)\gamma(w) \sim 1/(z - w)$ OPE after $\dbar$-integration over
the transverse $\C^2$. For $\g$ simple non-Abelian the cubic vertex
adds a Lie-bracket OPE term and the level-shift of Costello 2013
Thm~9.2.1; that extension is the target of Wave-13 H3/20.
