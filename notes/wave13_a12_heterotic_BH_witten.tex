\section*{Heterotic dyons, M5 spectra, and the primitive
\texorpdfstring{$\Delta_5$}{Delta5} denominator}
\label{sec:heterotic-BH-witten}

\paragraph{Normalisation.}
Let $\Delta_5$ denote Gritsenko's weight-$5$ Siegel paramodular cusp
form with order-$2$ multiplier on $\Sp_4(\Z)$, and let
$\Phi_{10}$ denote the scalar Igusa cusp form used in the
Dijkgraaf--Verlinde--Verlinde dyon formula. The convention in this note is
\[
 \Phi_{10}(Z)=\Delta_5(Z)^2 .
\]
The denominator identity of the primitive Borcherds--Kac--Moody
superalgebra $\fg_{\Delta_5}$ is written, up to the standard
Gritsenko--Nikulin scalar and argument normalisation, as
\[
 \operatorname{den}(\fg_{\Delta_5})(Z)=64^{-1}\Delta_5(2Z).
\]
The scalar four-dimensional dyon partition function is instead
$1/\Phi_{10}=1/\Delta_5^2$. Thus the primitive BKM denominator and the
physical scalar dyon denominator are related by squaring but live in
different structures.

\begin{theorem}[Primitive Borcherds weight and Igusa square]
\label{thm:delta5-phi10-primitive}
For the Borcherds product attached to the weak Jacobi form
$\phi_{0,1}$ of weight $0$ and index $1$,
\[
 \kBKM(\Delta_5)=\mathrm{wt}(\Delta_5)
 =\frac{c_{\phi_{0,1}}(0,0)}{2}
 =\frac{10}{2}=5.
\]
The physical scalar denominator for heterotic $1/4$-BPS dyons on
$K3\times T^2$ is $\Phi_{10}=\Delta_5^2$, of weight $10$.
\end{theorem}

\begin{proof}
Borcherds' singular theta lift gives a product whose automorphic weight
is half the constant term of the Jacobi input. For
$\phi_{0,1}$, $c_{\phi_{0,1}}(0,0)=10$, hence the primitive product has
weight $5$. Gritsenko--Nikulin identify this product with the
weight-$5$ form $\Delta_5$ carrying the non-trivial multiplier. Squaring
kills the multiplier and gives the scalar Igusa cusp form
$\Phi_{10}$ of weight $10$. The DVV integral uses this scalar square:
\[
 d(Q,P)=\int_{\mathcal C(Q,P)}
 \frac{\exp\!\left(-2\pi i(\rho P^2+\sigma Q^2+2v\,Q\cdot P)\right)}
 {\Phi_{10}(\rho,\sigma,v)}\,d\rho\,d\sigma\,dv ,
\]
with contour determined by the chamber of the charge vector. This proves
both the primitive BKM value $\kBKM(\Delta_5)=5$ and the scalar dyon
identity $\Phi_{10}=\Delta_5^2$.
\end{proof}

\begin{remark}[No scalar-square-root dyon count]
\label{rem:no-scalar-square-root-dyon-count}
The reciprocal $1/\Delta_5$ is the character-level denominator of the
primitive Borcherds algebra. It is not the scalar four-dimensional
$1/4$-BPS dyon partition function. The scalar physical index is governed
by $1/\Phi_{10}=1/\Delta_5^2$. Any chiral-half reading of
$\Delta_5$ must be stated as a denominator-algebra statement, not as a
replacement for the DVV scalar integral.
\end{remark}

\subsection*{The \texorpdfstring{$K3\times E$}{K3 x E} invariants}
\label{subsec:k3e-invariants}

\begin{proposition}[Four non-interchangeable invariants on
\texorpdfstring{$K3\times E$}{K3 x E}]
\label{prop:k3e-four-invariants-heterotic-note}
For $X=K3\times E$,
\[
 \kcat(X)=\chi(\cO_X)=0,\qquad
 \kch(A_X)=\sum_q(-1)^q h^{0,q}(X)=0,
\]
\[
 \kch^{\mathrm{Heis}}(K3\times E)=3,\qquad
 \kBKM(\Delta_5)=5,\qquad
 \kfib(K3)=24 .
\]
The value $2=\chi(\cO_{K3})$ is a fibre witness, not the categorical
Euler invariant of the total space $K3\times E$.
\end{proposition}

\begin{proof}
Kunneth multiplicativity gives
\[
 \chi(\cO_{K3\times E})=\chi(\cO_{K3})\chi(\cO_E)=2\cdot 0=0.
\]
The same Hodge column gives
\[
 \sum_q(-1)^qh^{0,q}(K3\times E)=(1-1)(1+1)=0.
\]
The rank-$3$ value is a different construction: it is the
Heisenberg--Cartan rank of the $K3$-fibre specialisation to the elliptic
curve, whose three surviving Mukai modes form the Cartan of
$\fg_{\Delta_5}$. The BKM value is the Borcherds weight from
Theorem~\ref{thm:delta5-phi10-primitive}. The fibre value $24$ is the
rank of $H^\bullet(K3;\Z)$, or equivalently the Euler characteristic of
the K3 fibre. These four numbers are attached to four different
constructions.
\end{proof}

\begin{proposition}[\texorpdfstring{$d=3$}{d=3} chiral output]
\label{prop:d3-e1-output-heterotic-note}
For the $d=3$ input $\Perf(K3\times E)$, the specialised
CY-to-chiral output on the elliptic curve is $E_1$-chiral. An $E_2$
structure appears only after passing to a derived centre, representation
centre, or Hall--Drinfeld double, and only on the hypotheses required to
construct that centre or double.
\end{proposition}

\begin{proof}
The two-stage construction sends the CY$_3$ datum first to a
factorisation object and then specialises along a surface fibre to a
curve. Factorisation on a curve is $E_1$-chiral. The $E_2$ operation is
not native to this curve-level algebra; it is the operation of the
centre or double of the $E_1$ object. Thus
\[
 \Phi_3(\Perf(K3\times E))_{\Sigma_2,E}
 \in E_1\text{-}\mathrm{ChirAlg}(E),
\]
while
\[
 Z_{\mathrm{ch}}^{\mathrm{der}}(\Phi_3(\Perf(K3\times E)))\quad
\text{or}\quad
\mathcal D_\hbar(\CoHA_{K3\times E})
\]
is the $E_2$-level object after the additional centre or double data are
supplied.
\end{proof}

\subsection*{Heterotic dyons and the scalar denominator}
\label{subsec:heterotic-dyons}

\begin{theorem}[DVV scalar partition function]
\label{thm:dvv-scalar-denominator}
In the heterotic frame on $K3\times T^2$, or equivalently in the
type-II dual frame on $K3\times T^2$, the indexed degeneracies of
four-dimensional $\mathcal N=4$ $1/4$-BPS dyons are Fourier coefficients
of $1/\Phi_{10}$. Since $\Phi_{10}=\Delta_5^2$, the scalar denominator is
the square of the primitive BKM denominator.
\end{theorem}

\begin{proof}
The electric and magnetic charges lie in the Narain lattice
$\Gamma^{6,22}\oplus\Gamma^{6,22}$, with T-duality invariants
$Q^2$, $P^2$, and $Q\cdot P$. DVV package the protected index into the
genus-$2$ Siegel modular form
\[
 \frac{1}{\Phi_{10}(\rho,\sigma,v)}
 =\sum_{Q,P} d(Q,P)\,
 e^{2\pi i(\rho P^2+\sigma Q^2+2v\,Q\cdot P)} .
\]
Wall crossing changes the contour and residue prescription, not the
scalar modular form. Theorem~\ref{thm:delta5-phi10-primitive} identifies
this scalar form as $\Delta_5^2$ and leaves the primitive BKM denominator
as $\Delta_5$.
\end{proof}

\begin{remark}[The rank-$24$ phrase]
\label{rem:rank-24-phrase-heterotic-note}
At a generic heterotic point, the left-moving internal current algebra
and compactification vectors give a rank-$24$ abelian gauge sector inside
the full $\mathcal N=4$ vector system. This rank is not the rank-$3$
Lorentzian lattice supporting $\fg_{\Delta_5}$ and not the
$24$-dimensional Mukai cohomology of the K3 fibre. The same integer
appears in three separate roles.
\end{remark}

\subsection*{M5-branes and Cardy growth}
\label{subsec:m5-cardy-growth}

\begin{theorem}[MSW central charges for the K3 fibre divisor]
\label{thm:msw-k3-fibre-central-charge}
Let $P=[K3]\subset K3\times E$ be the K3 fibre divisor. The MSW CFT
obtained from an M5-brane wrapped on $P$ has
\[
 c_L=P^3+c_2(K3\times E)\cdot P=24,\qquad
 c_R=P^3+\frac12 c_2(K3\times E)\cdot P=12 .
\]
The associated Cardy growth is
\[
 S_{\mathrm{Cardy}}(n_L)=2\pi\sqrt{\frac{c_L n_L}{6}}
 =4\pi\sqrt{n_L}.
\]
\end{theorem}

\begin{proof}
The fibre has self-intersection $P^3=0$. Since $c_2(E)=0$ and
$\int_{K3}c_2(K3)=24$, one obtains
\[
 c_2(K3\times E)\cdot P=24.
\]
The MSW anomaly formula gives the stated pair $(c_L,c_R)=(24,12)$.
Substitution into Cardy's formula gives $4\pi\sqrt{n_L}$.
\end{proof}

\begin{remark}[No determination of MSW charge from
\texorpdfstring{$\kch^{\mathrm{Heis}}$}{chiral Heisenberg characteristic}]
\label{rem:no-msw-from-kch-heis}
The equality $c_L=24$ is an M5 anomaly calculation. It is not determined
by $\kch^{\mathrm{Heis}}(K3\times E)=3$. The number $3$ records the
rank-$3$ Heisenberg--Cartan shadow of the $\Delta_5$ denominator
algebra. Matching a scalar expression of the form
$\pi\sqrt{2\kch^{\mathrm{Heis}}D}$ to the MSW entropy requires an
explicit definition of $D$ in terms of the physical level $n_L$ or the
quartic charge invariant. Without that normalisation it is not a
black-hole entropy theorem.
\end{remark}

\begin{theorem}[BPS scalar partition function in the MSW frame]
\label{thm:msw-phi10}
The scalar protected partition function of the $K3\times T^2$ MSW
system is governed by
\[
 Z_{\mathrm{MSW}}(Z)=\frac{1}{\Phi_{10}(Z)}
 =\frac{1}{\Delta_5(Z)^2}.
\]
The primitive algebra $\fg_{\Delta_5}$ has denominator $\Delta_5$; the
MSW scalar index has denominator $\Delta_5^2$.
\end{theorem}

\begin{proof}
The MSW reduction and the heterotic/type-II duality identify the
protected four-dimensional dyonic index with the DVV index. Apply
Theorem~\ref{thm:dvv-scalar-denominator}.
\end{proof}

\subsection*{Asymptotic symmetry and the BKM algebra}
\label{subsec:asymptotic-bkm}

\begin{theorem}[Boundary symmetry is not the primitive BKM algebra]
\label{thm:boundary-not-bkm}
The classical asymptotic symmetry algebra of the
$\AdS_3\times S^2\times K3$ near-horizon supergravity background is the
Brown--Henneaux Virasoro algebra, together with the affine current
algebra forced by the $S^2$ R-symmetry and the internal K3 cohomology.
It is not $\fg_{\Delta_5}$.
\end{theorem}

\begin{proof}
Brown--Henneaux boundary conditions for $\AdS_3$ produce
$\mathrm{Vir}_L\oplus\mathrm{Vir}_R$ with central charge fixed by the
three-dimensional Newton constant. The $S^2$ factor contributes the
standard affine R-symmetry current algebra. These are symmetries of the
classical boundary problem. The Borcherds algebra
$\fg_{\Delta_5}$ is instead defined by the automorphic denominator
$64^{-1}\Delta_5(2Z)$ and the corresponding Lorentzian root system. Its
root multiplicities encode protected BPS arithmetic; they are not
classical boundary gauge generators.
\end{proof}

\begin{proposition}[Spectrum-generating interpretation]
\label{prop:bkm-spectrum-generating-scoped}
The physically meaningful role of $\fg_{\Delta_5}$ in the
$K3\times T^2$ dyon problem is spectrum-generating: its denominator
organises the primitive automorphic BPS data whose scalar square is
seen by the DVV partition function. This is compatible with, and
distinct from, the Brown--Henneaux boundary algebra.
\end{proposition}

\begin{proof}
The denominator formula gives the signed root multiplicities of
$\fg_{\Delta_5}$ from the Fourier coefficients of $\phi_{0,1}$ and
packages them in $\Delta_5$. The scalar dyon index packages physical
degeneracies in $1/\Phi_{10}=1/\Delta_5^2$. The two packages are linked
by the Igusa square but have different algebraic meanings. Therefore
the BKM algebra can organise BPS data without being an asymptotic
supergravity symmetry.
\end{proof}

\subsection*{Duality frames and diagonal-divisor forms}
\label{subsec:duality-frames-diagonal-divisor}

\begin{proposition}[Four frames see the scalar square]
\label{prop:four-frames-scalar-square}
The heterotic, IIA, IIB, and M-theory descriptions of the
$K3\times T^2$ dyon problem all see the scalar form
$\Phi_{10}=\Delta_5^2$ in protected indices or threshold functions. The
primitive form $\Delta_5$ appears as the Borcherds denominator before
the scalar square is formed.
\end{proposition}

\begin{proof}
In the heterotic frame, the DVV integral is written with
$1/\Phi_{10}$. In type-II frames, duality transports the same protected
index to D-brane or MSW variables. In the M-theory frame, the M5
worldvolume CFT gives the same Cardy growth and protected index after
duality. In threshold-correction language one meets logarithms such as
$\log|\Phi_{10}|=2\log|\Delta_5|$. None of these scalar appearances
identifies $1/\Delta_5$ with the physical dyon partition function.
\end{proof}

\begin{conjecture}[Diagonal-divisor DT realisations]
\label{conj:diagonal-divisor-dt-realisations}
The remaining diagonal-divisor paramodular forms in the
Gritsenko--Clery catalogue arise from twisted Donaldson--Thomas theories
of $(S\times E)/(\Z/N\Z)$ after the correct multiplier, group,
orientation, and chamber data are fixed. The $N=1$ scalar
$K3\times E$ identity is the proved base case; the higher rows require
their own DT realisation data.
\end{conjecture}

\begin{proof}[Evidence]
The CHL orbifold dyons supply physical scalar Siegel forms with weights
matching the corresponding reduced-rank arithmetic. The Vol III
Borcherds-weight theorem gives
\[
 \kBKM(\Phi_N)=\frac{c_N(0)}{2}
\]
for the BKM-denominator side. These two pieces of evidence agree at the
level of weights and duality-frame assignments. A theorem for each
non-$N=1$ row still needs the actual DT chamber, orientation,
multiplier, and Hall recognition data.
\end{proof}

\subsection*{Logarithmic corrections}
\label{subsec:log-corrections-delta5}

\begin{theorem}[The \texorpdfstring{$\mathcal N=4$}{N=4} logarithmic
coefficient uses the scalar square]
\label{thm:log-correction-scalar-square}
For the $K3\times T^2$ $\mathcal N=4$ dyon problem, the logarithmic
correction to the macroscopic entropy is computed from the
near-horizon zero-mode spectrum together with the scalar modular form
$\Phi_{10}=\Delta_5^2$. The coefficient is not determined by the
primitive weight $5$ alone.
\end{theorem}

\begin{proof}
The microscopic Rademacher expansion uses the modular weight and polar
data of the inverse scalar form $1/\Phi_{10}$, hence the relevant weight
is $10$, not $5$. The macroscopic computation uses the zero modes of
the $\AdS_2\times S^2$ near-horizon supergravity background. The
standard $\mathcal N=4$ coefficient, written in the area convention of
the entropy-function calculation, is the result of combining these
scalar-modular and zero-mode inputs. Replacing $\Phi_{10}$ by
$\Delta_5$ would halve the modular weight and give the wrong scalar
input.
\end{proof}
