\section*{Mukai signature and the super-Yangian completion problem}

\subsection*{The form preserved by the K3 lattice}

For a complex K3 surface \(S\), the Mukai lattice
\[
  \widetilde H(S,\Z)
  =
  H^0(S,\Z)\oplus H^2(S,\Z)\oplus H^4(S,\Z)
  \simeq U^{\oplus 4}\oplus E_8(-1)^{\oplus 2}
\]
carries the symmetric pairing
\[
  \langle (r,D,s),(r',D',s')\rangle_{\mathrm{Muk}}
  =
  D\cdot D' - r s' - r' s .
\]
Its real signature is \((4,20)\).  The classical real Lie algebra
preserving this form is
\[
  \fso(4,20).
\]
This is the form-theoretic boundary for every non-abelian K3 Yangian
claim built from \(\widetilde H(S,\Z)\).

\begin{proposition}[Mukai form versus Kac orthosymplectic form]
\label{prop:mukai-form-orthogonal-boundary}
The Mukai pairing determines an ordinary orthogonal real form
\(\fso(4,20)\).  A Kac orthosymplectic Lie superalgebra
\(\osp(4\mid 20)\) requires a graded bilinear form whose even part is
symmetric and whose odd part is skew-symmetric.  Therefore the label
\(\osp(4\mid 20)\) requires an additional odd symplectic form on a
parity-shifted \(20\)-dimensional summand.  It is extra structure
beyond the Mukai lattice.
\end{proposition}

\begin{proof}
The Mukai form is symmetric on the whole rank-\(24\) lattice.  After
choosing a positive four-plane \(P\subset \widetilde H(S,\R)\), one has
an orthogonal decomposition
\[
  \widetilde H(S,\R)=P\oplus P^\perp,
  \qquad
  \dim P=4,\quad \dim P^\perp=20,
\]
with the Mukai form symmetric on both summands.  The stabiliser of the
ungraded form is \(\mathrm{O}(4,20)\), and its Lie algebra is
\(\fso(4,20)\).  By Kac's definition, \(\osp(m\mid 2n)\) preserves a
super-bilinear form whose restriction to the odd part is symplectic.
The symmetric restriction of \(\langle-,-\rangle_{\mathrm{Muk}}\) to
\(P^\perp\) supplies no such symplectic form.  A map from the Mukai
datum to a Kac \(\osp(4\mid 20)\) datum must therefore choose a
symplectic form on the parity-shifted negative summand.
\end{proof}

\subsection*{The ambient RTT superalgebra}

The super-vector space
\[
  V=V_{\bar 0}\oplus V_{\bar 1},
  \qquad
  V_{\bar 0}=\C^4,\qquad V_{\bar 1}=\C^{20},
\]
with homogeneous basis \(e_1,\ldots,e_{24}\) and parity
\[
  |i|=
  \begin{cases}
  \bar 0, &1\leq i\leq 4,\\
  \bar 1, &5\leq i\leq 24
  \end{cases}
\]
supports the standard general-linear RTT Yangian.  Let \(E_{ij}\) be
the elementary endomorphism \(e_j\mapsto e_i\), of parity
\(|i|+|j|\).

\begin{definition}[RTT Yangian \(Y_\hbar(\fgl(4\mid 20))\)]
\label{def:rtt-yangian-gl420}
Let \(P_s\in \End(V\otimes V)\) be the super-permutation
\[
  P_s(e_i\otimes e_j)=(-1)^{|i||j|}e_j\otimes e_i .
\]
Set
\[
  R(u)=\Id_{V\otimes V}+\frac{\hbar}{u}P_s .
\]
The RTT Yangian \(Y_\hbar(\fgl(4\mid 20))\) is the associative
superalgebra over \(\C[[\hbar]]\) generated by
\[
  t_{ij}^{(r)},\qquad 1\leq i,j\leq 24,\qquad r\geq 1,
  \qquad |t_{ij}^{(r)}|=|i|+|j|,
\]
with \(t_{ij}^{(0)}=\delta_{ij}\), \(T(u)=\sum_{i,j}\sum_{r\geq 0}
t_{ij}^{(r)}u^{-r}E_{ij}\), and relation
\[
  R_{12}(u-v)T_1(u)T_2(v)
  =
  T_2(v)T_1(u)R_{12}(u-v)
\]
in
\[
  Y_\hbar(\fgl(4\mid 20))
  \otimes \End(V)^{\otimes 2}[[u^{-1},v^{-1}]] .
\]
\end{definition}

\begin{lemma}[Graded Yang--Baxter equation]
\label{lem:gl420-rtt-ybe}
The operator \(R(u)=\Id+\hbar u^{-1}P_s\) satisfies
\[
  R_{12}(u)R_{13}(u+v)R_{23}(v)
  =
  R_{23}(v)R_{13}(u+v)R_{12}(u)
\]
on \(V^{\otimes 3}\).
\end{lemma}

\begin{proof}
The operators \(P_{12},P_{23}\) induced by \(P_s\) give the standard
Koszul-signed action of the transpositions \((12)\) and \((23)\) on
\(V^{\otimes 3}\).  Hence
\[
  P_{12}^2=P_{23}^2=\Id,
  \qquad
  P_{12}P_{23}P_{12}=P_{23}P_{12}P_{23}.
\]
The usual Baxterisation of this graded symmetric-group action gives
the displayed rational Yang--Baxter equation.  Expanding both sides in
\(\hbar\) reduces all coefficients to the two identities above.
\end{proof}

\begin{proposition}[Hopf superalgebra structure]
\label{prop:gl420-hopf}
The RTT algebra \(Y_\hbar(\fgl(4\mid 20))\) is a Hopf superalgebra with
\[
  \Delta(t_{ij}(u))=\sum_{k=1}^{24}t_{ik}(u)\otimes t_{kj}(u),
  \qquad
  \epsilon(t_{ij}(u))=\delta_{ij},
  \qquad
  S(T(u))=T(u)^{-1}.
\]
Multiplication in the tensor square uses the Koszul rule
\[
  (a\otimes b)(c\otimes d)=(-1)^{|b||c|}ac\otimes bd .
\]
\end{proposition}

\begin{proof}
The coproduct is matrix multiplication for the monodromy matrix
\(T(u)\).  Coassociativity is associativity of the triple product
\(T(u)\otimes T(u)\otimes T(u)\), with the displayed Koszul sign in
the tensor product.  The counit sends \(T(u)\) to the identity matrix.
The antipode is the inverse matrix in the completed RTT algebra.
\end{proof}

\begin{proposition}[Ambient PBW theorem]
\label{prop:gl420-pbw}
With the filtration \(\deg t_{ij}^{(r)}=r-1\), the associated graded
superalgebra of \(Y_\hbar(\fgl(4\mid 20))\) is
\[
  \operatorname{gr}Y_\hbar(\fgl(4\mid 20))
  \simeq
  U(\fgl(4\mid 20)[s]).
\]
The symbol of \(t_{ij}^{(r)}\) maps to \(E_{ij}s^{r-1}\).
\end{proposition}

\begin{proof}
The RTT relation gives, in the leading filtered degree, the current
Lie superalgebra bracket
\[
  [E_{ij}s^a,E_{kl}s^b]
  =
  \delta_{jk}E_{il}s^{a+b}
  -
  (-1)^{(|i|+|j|)(|k|+|l|)}
  \delta_{il}E_{kj}s^{a+b}.
\]
The Nazarov--Gow PBW theorem for \(Y_\hbar(\fgl(m\mid n))\) identifies
the ordered RTT monomials with the ordered PBW monomials of the current
Lie superalgebra.  Specialising \(m=4,n=20\) gives the displayed
associated graded algebra.
\end{proof}

\subsection*{Determinant data and the degree claim}

\begin{lemma}[RTT determinant exponents]
\label{lem:rtt-determinant-exponents}
For \(V=\C^{4\mid 20}\),
\[
  \det R(u)
  =
  (1+\hbar u^{-1})^{280}(1-\hbar u^{-1})^{296}.
\]
The exponents are the dimensions of the graded symmetric and graded
antisymmetric squares:
\[
  \dim \mathrm{Sym}^2_s(V)=
  \binom{4+1}{2}+4\cdot 20+\binom{20}{2}=280,
\]
\[
  \dim \Lambda^2_s(V)=
  \binom{4}{2}+4\cdot 20+\binom{20+1}{2}=296 .
\]
\end{lemma}

\begin{proof}
The super-permutation \(P_s\) has eigenvalue \(+1\) on
\(\mathrm{Sym}^2_s(V)\) and eigenvalue \(-1\) on
\(\Lambda^2_s(V)\).  Since \(R(u)=\Id+\hbar u^{-1}P_s\), the determinant
is the product of the two scalar eigenvalues with the displayed
multiplicities.
\end{proof}

The intrinsic RTT datum is the rational matrix
\[
  R(u)=\Id+\hbar u^{-1}P_s
\]
on a \(576\)-dimensional tensor square, together with the determinant
exponents \((280,296)\).  A degree-\((24,24)\) structure function
belongs to the Maulik--Okounkov geometric \(R\)-matrix on
\(\Hilb^n(K3)\) after a chosen embedding
\[
  E_8(-1)^{\oplus 2}\hookrightarrow \widetilde H(S,\Z)
\]
and stable-envelope normalisation.  For the intrinsic
\(E_8\oplus E_8\) Cartan block the rank degree is \((16,16)\).  The
numbers \(16\), \(24\), and \(576\) count different objects:
\[
  \mathrm{rk}(E_8^{\oplus 2})=16,\qquad
  \mathrm{rk}\,\widetilde H(S,\Z)=24,\qquad
  \dim(V\otimes V)=576 .
\]

\subsection*{Drinfeld currents as an induced presentation}

The distinguished Borel of \(\fgl(4\mid 20)\) has \(23\) simple roots:
three even roots in the \(A_3\) block, one odd isotropic root, and
nineteen even roots in the \(A_{19}\) block.  The Gauss decomposition
of \(T(u)\) gives Drinfeld currents
\[
  E_i(u),\quad F_i(u),\quad H_i(u),
  \qquad 1\leq i\leq 23.
\]
Their relations are the Nazarov--Gow RTT relations transported through
Gauss decomposition, with Yamane's extra odd-isotropic Serre relation
at the single odd node.  This current presentation is a presentation
of the ambient algebra \(Y_\hbar(\fgl(4\mid 20))\).  Its transfer to
the Mukai lattice or to compact Hall data requires an additional
representation functor.

\subsection*{The conjectural K3 super-Yangian}

\begin{conjecture}[Mukai super-Yangian completion]
\label{conj:mukai-super-yangian-completion}
There exists a completed Hopf superalgebra
\[
  \mathcal Y_{\mathrm{Muk}}(S)
\]
whose classical limit is a form-preserving current algebra over
\(\fso(4,20)\), whose abelian associated graded recovers the
Mukai-Heisenberg Yangian of \(\widetilde H(S,\Z)\), and whose
geometric \(R\)-matrix on \(\bigoplus_n H^\bullet(\Hilb^n(S))\) is the
Maulik--Okounkov \(R\)-matrix in Mukai rank \(24\).
\end{conjecture}

The ambient construction above supplies generators, RTT relations,
Hopf structure, rational \(R\)-matrix, Drinfeld currents, and PBW for
\(Y_\hbar(\fgl(4\mid 20))\).  The K3 completion becomes a theorem once
the following data are given in compatible form:
\begin{enumerate}
\item generators, defining relations, and an RTT or Drinfeld-current
presentation for \(\mathcal Y_{\mathrm{Muk}}(S)\);
\item coproduct, counit, antipode, and completed tensor category;
\item a PBW theorem identifying the associated graded algebra;
\item universal or dynamical \(R\)-matrix satisfying the graded
Yang--Baxter equation and the required hexagon identities;
\item representation theory, including the tensor category of modules
acting on the K3 Hilbert-scheme or compact Hall category;
\item a comparison functor from the compact Hall or stable-envelope
construction to the algebra in the first item.
\end{enumerate}

A Kac \(Y_\hbar(\osp(4\mid 20))\) can appear only after choosing an odd
symplectic form on the parity-shifted \(20\)-dimensional summand and
constructing the corresponding reflection-equation or RTT object.  The
Mukai lattice supplies the orthogonal form; the symplectic odd form is
separate input.

\subsection*{Borcherds and Mukai branches}

The \(K3\times E\) Borcherds branch is organised by the algebra
\(\fg_{\Delta_5}\) and by compact Hall or Hall--Drinfeld constructions
under their orientation, pairing, and completion hypotheses.  The
historical K3 Yangian branch is the Mukai self-mirror branch.  These
are different structures:
\[
\begin{array}{c|c|c}
\text{object} & \text{input} & \text{algebraic layer}\\ \hline
Y_\hbar^{\mathrm{Heis}}(\widetilde H(S,\Z)) &
\text{Mukai lattice} &
\text{abelian Yangian, Hopf deformation}\\
\mathcal Y_{\mathrm{Muk}}(S) &
\text{Mukai form plus completion data} &
\text{conjectural non-abelian completion}\\
\fg_{\Delta_5} &
\phi_{0,1}^{\mathrm{EZ}}\xrightarrow{\mathrm{Borch}}\Delta_5 &
\text{Borcherds--Kac--Moody Lie superalgebra}\\
\mathbf H_{\Delta_5} &
\CoHA(K3\times E)\ \text{with pairing and completion} &
\text{Hall--Drinfeld or compact Hall object}
\end{array}
\]
The Borcherds row is controlled by Borcherds products and Siegel
modular forms.  The Mukai rows are controlled by rational or geometric
\(R\)-matrices.  A comparison between the rows requires an explicit
functor preserving products, coproducts, \(R\)-matrices, and
representations.

\subsection*{Borcherds constants and \(K3\times E\) invariants}

For the primitive CHL denominator levels
\[
  N\in\{1,2,3,4,6\},
\]
the Gritsenko--Nikulin normalisation gives
\[
  c_N(0)=(10,8,6,4,2),
  \qquad
  \kBKM(\Phi_N)=\frac{c_N(0)}{2}=(5,4,3,2,1).
\]
In particular,
\[
  c_1(0)=10,\qquad
  \kBKM(\Phi_1)=5,
  \qquad
  \Delta_5=\mathrm{Borch}\bigl(\phi_{0,1}^{\mathrm{EZ}}\bigr).
\]
In Eichler--Zagier normalisation,
\[
  \mathrm{Borch}\bigl(\phi_{0,1}^{\mathrm{EZ}}\bigr)=\Delta_5,
  \qquad
  \mathrm{Borch}\bigl(2\phi_{0,1}^{\mathrm{EZ}}\bigr)
  =
  \Phi_{10}
  =
  \Delta_5^2 .
\]
The primitive denominator algebra \(\fg_{\Delta_5}\) reads the
coefficients of \(\phi_{0,1}^{\mathrm{EZ}}\).  The K3 elliptic genus
\(Z_{K3}=2\phi_{0,1}^{\mathrm{EZ}}\) reads the scalar square.

The invariants remain separate:
\[
  \kch,\qquad
  \kcat=\chi(\mathcal O_X),\qquad
  \kBKM=\frac{c_N(0)}{2},\qquad
  \kfiber .
\]
For \(K3\times E\),
\[
  \begin{aligned}
  \kcat(K3\times E)
  &=
  \chi(\mathcal O_{K3})\chi(\mathcal O_E)
  =
  2\cdot 0
  =
  0,\\
  \kch(K3\times E)
  &=
  \sum_{q=0}^{3}(-1)^q h^{0,q}(K3\times E)
  =
  1-1+1-1
  =
  0,\\
  \kch^{\mathrm{Heis}}(K3\times E)
  &=
  3,\\
  \kBKM(\fg_{\Delta_5})
  &=
  \mathrm{wt}(\Delta_5)
  =
  5,\\
  \kfiber(K3\times E)
  &=
  \mathrm{rk}\,\widetilde H(K3,\Z)
  =
  24 .
  \end{aligned}
\]
The compact \(\kch\)-row is the Hodge supertrace.  The Heisenberg row
is the rank-\(3\) Mukai--Cartan specialisation.  The value \(5\) is
the Borcherds weight of \(\Delta_5\), hence a \(\kBKM\)-value.

\subsection*{\(E_1\), \(E_2\), and the positive half}

For Calabi--Yau dimension \(d\geq 3\), the specialised
CY-to-chiral output satisfies
\[
  \Phi_d(\mathcal C)\in E_1\text{-}\mathrm{ChirAlg}.
\]
The \(E_2\) structure lives on the centre, double, module category, or
enhancement layer.  This convention keeps compact Hall objects, Mukai
Yangian objects, and Borcherds objects in the correct operadic place.

For the local model \(\C^3\),
\[
  \CoHA(\C^3)\simeq \Yplus(\widehat{\fgl}_1)
\]
as the positive half.  The full vertex algebra
\(\mathcal W_{1+\infty}\) is obtained after Drinfeld doubling, centre
or completion, and evaluation on the appropriate representation.
