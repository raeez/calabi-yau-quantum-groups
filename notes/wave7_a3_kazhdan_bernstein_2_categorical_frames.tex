\section{Kazhdan--Bernstein channel: 2-categorical
coherence of the four-frame Fourier--Mukai web}
\label{wn:sec:kazhdan-bernstein-wave7-a3-2-categorical-frames}

The Wave-$6$ Z$4$ construction
(Theorem~\ref{thm:categorified-q5-v5-duality-web-wave6-z4})
assembles the four Fourier--Mukai equivalences
$\mathcal{C}_{\mathrm{Het}} \simeq
\mathcal{C}_{\mathrm{F}} \simeq
\mathcal{C}_{\mathrm{IIA}} \simeq
\mathcal{C}_{\mathrm{M}}$
into a $2$-groupoid $\mathfrak{U}$ and invokes Schur's lemma on the
universal module $\mathcal{M}_{\Delta_5}$ to collapse each frame
auto-equivalence to a scalar $C_\star \in \C^\times$. That collapse
is legitimate only after the $2$-morphism data --- pentagonators,
triangulators, and Kontsevich--Soibelman wall-crossing $2$-cells --- have
been verified coherent. The $2$-categorical layer is the present target.

\subsection*{(i) Attack: pentagonator for the spectral-cover
composition}

Schur's argument assumes each composable triple of frame arrows
$(\mathrm{Het} \to \mathrm{F} \to \mathrm{IIA} \to \mathrm{M})$
associates up to an identity $2$-cell. The genuine pentagonator
\begin{equation}
\label{eq:wave7-a3-pentagonator}
\pi_{\star\bullet\circ\diamond}:
(K_{\star\bullet} \circ K_{\bullet\circ}) \circ K_{\circ\diamond}
\;\Rightarrow\;
K_{\star\bullet} \circ (K_{\bullet\circ} \circ K_{\circ\diamond})
\end{equation}
is a natural isomorphism of Fourier--Mukai kernels, not the identity.
Donagi--Grassi--Witten spectral-cover composition $K_{\mathrm{Het}\to\mathrm{F}}$
composed with the adiabatic-shrinking kernel $K_{\mathrm{F}\to\mathrm{IIA}}$
inherits a non-trivial associator from the dependence of the
Higgs-bundle spectral parameter on the $K3$-fibre volume.

\subsection*{(ii) Heal: the coherent pentagonator from
Bridgeland--Maciocia}

\begin{theorem}[Coherent pentagonator]
\label{thm:wave7-a3-coherent-pentagonator}
\ClaimStatusTheorem
For any ordered quadruple
$(\star, \bullet, \circ, \diamond) \subset
\{\mathrm{Het}, \mathrm{F}, \mathrm{IIA}, \mathrm{M}\}$,
the pentagonator~\eqref{eq:wave7-a3-pentagonator} exists and is
unique up to unique $3$-isomorphism. Its value on a BPS object
$\mathcal{O}_Z \in \mathcal{C}_\star$ is the Bridgeland--Maciocia
compatibility iso
$R\pi_{13,*}(p_{12}^* K \otimes p_{23}^* L) \otimes M
\simeq R\pi_{13,*}(p_{12}^* K \otimes p_{23}^*(L \otimes M))$
projected to the triple-fibre product. The pentagon coherence
identity (Mac Lane) among the five orderings of a $4$-fold
composition reduces to the cocycle condition for the derived
projection formula on the triple fibre product
$S_\star \times S_\bullet \times S_\circ \times S_\diamond$.
\end{theorem}

\begin{proof}
Derived projection formula and base change yield existence;
Bridgeland--Maciocia functoriality of Fourier--Mukai kernels
\cite{BridgelandMaciocia2002} plus uniqueness of the derived
push-forward supply uniqueness. The $2$-pentagon identity among
five bracketings becomes the $5$-term iterated projection-formula
cocycle on the $5$-fold fibre product, which holds at the level
of derived categories because each face is the derived tensor of
kernels with the associator of $- \otimes^{\mathbb{L}} -$ in
$D^b$-of-a-triple-fibre-product.
\end{proof}

\subsection*{(iii) Attack: triangulator for the adiabatic-K3
shrinking adjunction}

The $\mathrm{F}\to\mathrm{IIA}$ equivalence is realised by shrinking
the second-factor $K3$ fibre to a point; its pseudo-inverse
reconstructs the $K3$ via eight-dimensional $F$-theory lift. The
unit $\eta$ and counit $\varepsilon$ satisfy the triangle identities
only modulo a coherent $2$-cell. The naive Schur collapse sees only
the identity component of that $2$-cell.

\subsection*{(iv) Heal: triangulator from Toën derived $\infty$-stacks}

\begin{theorem}[Triangulator coherence]
\label{thm:wave7-a3-triangulator}
\ClaimStatusTheorem
The adiabatic-shrinking adjunction
$L \dashv R : \mathcal{C}_{\mathrm{F}} \rightleftarrows \mathcal{C}_{\mathrm{IIA}}$
admits a coherent triangulator: the two composites
$L \xrightarrow{L\eta} LRL \xrightarrow{\varepsilon L} L$ and
$R \xrightarrow{\eta R} RLR \xrightarrow{R\varepsilon} R$
are connected to the identity via a pair of invertible $2$-cells
$\tau_L, \tau_R$ satisfying $\mathrm{Sym}^2$-compatibility with the
pentagonator of Theorem~\ref{thm:wave7-a3-coherent-pentagonator}.
\end{theorem}

\begin{proof}
Toën \cite{Toen2005} classifies adjunctions in derived $\infty$-stacks
via the $\infty$-categorical $\mathrm{Adj}$ walking-adjunction object.
The data of a coherent adjunction in $\mathrm{Cat}_\infty$ is a
functor $\mathrm{Adj} \to \mathrm{Cat}_\infty$; the triangulator is
the image of the unique nonidentity $2$-cell in $\mathrm{Adj}$.
Existence and uniqueness follow from contractibility of the space
of coherent extensions (Riehl--Verity). The $\mathrm{Sym}^2$-compatibility
is the image of the standard Mac Lane triangle-pentagon compatibility.
\end{proof}

\begin{proof}

\subsection*{(vi) Attack--heal: $\mathfrak{U}$ is a $(2,2)$-groupoid}

\begin{theorem}[$\mathfrak{U}$ is $(2,2)$-groupoid]
\label{thm:wave7-a3-22-groupoid}
\ClaimStatusTheorem
The $2$-groupoid $\frakU$ of frame Fourier--Mukai equivalences
is a $(2,2)$-groupoid: every $1$-morphism is invertible up to
invertible $2$-morphism, and every $2$-morphism is strictly
invertible.
\end{theorem}

\begin{proof}
Each Fourier--Mukai kernel $K : \mathcal{C}_\star \to \mathcal{C}_\bullet$
admits a pseudo-inverse $K^\vee \otimes \omega[\dim]$ (Mukai), so
$1$-morphisms are invertible up to
Theorem~\ref{thm:wave7-a3-triangulator}. Each pentagonator
(Theorem~\ref{thm:wave7-a3-coherent-pentagonator}) is an iso of
Fourier--Mukai kernels, hence $2$-invertible. KS $2$-cells
(Theorem~\ref{thm:wave7-a3-ks-2-cells}) are unipotent in
$\widehat{G}_{\mathrm{KS}}$, hence invertible. Thus $\frakU$ is
$(2,2)$.
\end{proof}

\subsection*{(vii) $2$-Schur on $\mathrm{Mod}^{\mathrm{2}}_{\mathcal{A}_{1/2}}$}

\begin{theorem}[$2$-Schur for $\mathcal{M}_{\Delta_5}$]
\label{thm:wave7-a3-2-schur}
\ClaimStatusTheorem
Let $\mathrm{Mod}^{\mathrm{2}}_{\mathcal{A}_{1/2}}$ be the
$(2,1)$-category of $\mathcal{A}_{1/2}$-$2$-modules. The
endomorphism $2$-category
$\mathrm{End}^{\mathrm{2}}(\mathcal{M}_{\Delta_5})$
is equivalent to $B\C^\times$ (one object, $\C^\times$ automorphisms,
identity $2$-cells). Consequently every $\mathcal{A}_{1/2}$-$2$-module
auto-$2$-equivalence of $\mathcal{M}_{\Delta_5}$ is a scalar
$C_\star \in \C^\times$, and every $2$-cell between two such
auto-equivalences is the identity on $C_\star$.
\end{theorem}

\begin{proof}
$\mathcal{M}_{\Delta_5}$ is one-dimensional irreducible, so
$\mathrm{End}(\mathcal{M}_{\Delta_5}) = \C$ by ordinary Schur.
The $2$-morphisms in
$\mathrm{Mod}^{\mathrm{2}}_{\mathcal{A}_{1/2}}$
between two auto-equivalences $C_\star, C_{\star'} : \mathcal{M}_{\Delta_5}
\to \mathcal{M}_{\Delta_5}$ are elements of
$\mathrm{Hom}_{\mathcal{A}_{1/2}}(\mathcal{M}_{\Delta_5}, \mathcal{M}_{\Delta_5}) = \C$;
if $C_\star = C_{\star'}$, this element is forced to $1$ by
unitality. Hence $\mathrm{End}^{\mathrm{2}}(\mathcal{M}_{\Delta_5}) = B\C^\times$.
\end{proof}
