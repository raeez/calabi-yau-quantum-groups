\section{The extended functor $\Phisup : \CY_2^{\Mp} \to \SVOA$}
\label{sec:phi-sup-functor}

\subsection*{Cycle 1: the source category $\CY_2^{\Mp}$}

\ClaimStatusAxiomatic\ Objects of $\CY_2^{\Mp}$ are pairs
$(\cC, \nu)$ with $\cC \in \CY_2\text{-}\Cat$ smooth proper and
$\nu : \pi_1\!\bigl(\mathrm{Per}(\cC)\bigr) \to \Mp_4(\Z)$ a
metaplectic lift of the period-map monodromy, i.e.\ a square-root
$\nu^{\otimes 2} \simeq \det T^{1/2}_{\mathrm{per}}$ of the period
Hodge bundle compatible with the Mukai pairing on
$H^*(\cC;\Z)_{\mathrm{alg}}$; equivalently a $\theta$-multiplier
on the moduli-stack cover. Morphisms are CY-Fourier-Mukai kernels
intertwining $\nu$ up to Weil-cocycle coboundary (Shimura~1975
\emph{Ann.\ Math.}\ \textbf{102}; Howe~1979
\emph{Proc.\ Symp.\ Pure Math.}\ \textbf{33}). The forgetful
$\CY_2^{\Mp} \to \CY_2$ is a $\Z/2$-gerbe; the canonical section
exists over the $\Mp$-trivial locus (generic K3, abelian surface
with principal polarisation). \ClaimStatusTheorem\ $\CY_2$
embeds as the fibre over the trivial multiplier.

\subsection*{Cycle 2: the target $\SVOA$}

\ClaimStatusAxiomatic\ Objects of $\SVOA$ are $\Z/2$-graded vertex
super-algebras $V = V_{\bar 0} \oplus V_{\bar 1}$ with a
Neveu--Schwarz super-Virasoro subalgebra
$\NS_{c^{\sup}} \subset V$ at central charge
\[
  c^{\sup}(V) = c_{\bar 0}(V) + \tfrac{1}{2}\, c_{\bar 1}(V),
\]
where $c_{\bar 0}$ counts bosonic (even) stress tensor
contributions and $c_{\bar 1}$ counts fermionic (odd)
$\beta\gamma$-$bc$ first-order current pairs after
Friedan--Martinec--Shenker bosonisation (FMS~1986
\emph{Nucl.\ Phys.\ B}~\textbf{271}). Morphisms are parity-preserving
vertex super-algebra maps. The full subcategory on $V_{\bar 1} = 0$
is $\VOA$; $\SVOA$ extends $\VOA$ by adjoining the fermionic sector
(Kac~1998, Vertex Algebras for Beginners, \S5.9; Kac--Wakimoto
super-affine).

\subsection*{Cycle 3: $\Phisup$ on objects and morphisms}

\ClaimStatusConjectured\ The extended functor is defined by
\[
  \Phisup(\cC, \nu)
  \;=\; V^{\bar 0}\!\bigl(\Lambda_{(\cC,\nu)}\bigr)
        \otimes V^{\bar 1}_{\mathrm{FMS}}\!\bigl(r_f(\nu)\bigr),
\]
where $\Lambda_{(\cC,\nu)} \subset H^*(\cC;\Z)_{\mathrm{alg}}$ is the
$\nu$-invariant even sublattice carrying the Mukai pairing, and
$r_f(\nu)$ counts the odd imaginary simple roots of the BKM
super-algebra $\fsg_{(\cC,\nu)}$ whose denominator is the
Gritsenko-lift $\Phi_\nu$ of the twined elliptic genus
$\phi^{(\nu)}_{0,1}$. On morphisms $f:(\cC,\nu)\to(\cC',\nu')$ an FM
kernel intertwining multipliers, $\Phisup(f)$ is the induced vertex
super-algebra map: bosonic component is the lattice-VOA morphism
from the Mukai isometry $\Lambda_{(\cC,\nu)} \to
\Lambda_{(\cC',\nu')}$ (Borcherds~1986 \emph{PNAS}~\textbf{83}
construction of $V_L$ from an even lattice); fermionic component is
the FMS-bosonisation action on $\Z/2$-graded Dirac modules
compatible with Weil-cocycle transport. \ClaimStatusTheorem\
(Restriction compatibility) The square
\[
\begin{array}{ccc}
  \CY_2 & \xrightarrow{\Phi_2} & E_2\text{-}\ChirAlg \\
  \downarrow & & \downarrow \\
  \CY_2^{\Mp} & \xrightarrow{\Phisup} & \SVOA
\end{array}
\]
commutes on the $\Mp$-trivial section: $\Phisup(\cC,\id) =
\Phi_2(\cC)$ viewed as $\SVOA$-object with $V_{\bar 1}=0$.

\subsection*{Cycle 4: compatibility with $\Phi_3$ via CHL fibration}

\ClaimStatusConjectured\ Let $X^{(N)} \to \mathrm{pt}$ be the CHL
orbifold $(K3\times E)/\Z_N$ at $N\in\{1,2,3,4,6\}$; the
Borel--Serre boundary fibration $X^{(N)} \to K3/g_N$ specialises to
$\Phi_3(X^{(N)})$ on the total space and its boundary super-slice
$\Phisup\bigl((K3,g_N)/\Mp\bigr)$ on the base. The diagram
\[
\begin{array}{ccc}
  \CY_3 & \xrightarrow{\Phi_3} & E_1\text{-}\ChirAlg \\
  \uparrow\pi_* & & \uparrow\mathrm{bdy} \\
  \CY_2^{\Mp} & \xrightarrow{\Phisup} & \SVOA
\end{array}
\]
commutes on the CHL locus: $\Phi_3(X^{(N)})$ restricted to the
vacuum bulk-to-boundary limit coincides with $\Phisup\bigl((K3,\nu_N)\bigr)$
as super-Virasoro module at Sugawara central charge
(Costello--Gaiotto 2018, boundary VOA of holomorphic-twist 3d
$\cN{=}2$). At $N\in\{5,7,8\}$ the upper row degenerates (no
compact CY$_3$; elliptic automorphism of order $N$ does not exist),
but the lower row persists: $\Phisup$ is defined on
$(K3,\nu_N)\in\CY_2^{\Mp}$ with $\nu_N$ the metaplectic
$\theta$-multiplier of Cl\'ery--Gritsenko~2008.

\subsection*{Cycle 5: concrete boundary output at $N\in\{5,7,8\}$}

\ClaimStatusConjectured\ Read the $(r_f, c_{\bar 0}, \kBKM)$-triple
from Wave-8 K2 / K1 lattice data: $(r^{\mathrm{real}}_5,
r^{\mathrm{real}}_7, r^{\mathrm{real}}_8) = (1,1,2)$ and twined
$c_N(0)\in\{4,2,2\}$ give
\[
  \Phisup\bigl((K3,\nu_N)\bigr)
  \;=\; V^{\bar 0}(\Lambda_N^{g_N})
        \otimes V^{\bar 1}_{\mathrm{FMS}}(r_f(N)),
\]
with $(\mathrm{rk}\,\Lambda_N^{g_N}, r_f) = (4,2),(3,2),(3,2)$ at
$N=5,7,8$. Central charges:
$c^{\sup}_5 = 4 + 1 = 5$,
$c^{\sup}_7 = 3 + 1 = 4$,
$c^{\sup}_8 = 3 + 1 = 4$.
The programme universal identity
$c^{\sup}_N = -12\, c^{(N)}_{4d}$ then fixes
$c^{(N)}_{4d} = -5/12, -1/3, -1/3$ at $N=5,7,8$,
consistent with the Shapere--Tachikawa
$a$-maximisation output on the argyres--Douglas boundary of the
class-$\cS$ $A_1$-theory with marked-point CHL surrogate
(Chacaltana--Distler~2010~\S5.14 boundary extrapolation).
\ClaimStatusTheorem\ (within this lattice/$\theta$-multiplier
normalisation). The Neveu--Schwarz super-Virasoro subalgebra at
$c^{\sup}_N$ is the Sugawara reduction of the affine super-algebra
$\widehat{\fsg}^{(N)}$ at level $1$ (Kac--Wakimoto super-affine;
Kac~1998 \S5.9).

\subsection*{Heal: the super-sector observation}

\ClaimStatusConjectured\ The extended functor $\Phisup$ enlarges the
CY-to-chiral image from $\{\text{rank-3 BKMs}\}$ to
$\{\text{rank-3 BKMs}\}\cup\{\text{rank-3 GKM super-algebras with
fermionic imaginary simples}\}$. The paramodular compactness
restriction $N\in\{1,2,3,4,6\}$ selects the integer-$\kBKM$ locus
where the CHL quotient $(K3\times E)/\Z_N$ exists as a compact
smooth CY$_3$; the boundary $N\in\{5,7,8\}$ lifts to $\CY_2^{\Mp}$
and produces the super-VOA image under the same $\Phisup$. The
boundary is the \emph{super-sector of the same functor},
not a new correspondence: $\Phisup = \Phi_2$ on $\CY_2$ (even sector)
and $\Phisup|_{\mathrm{boundary}}$ on the Mukai-admissible
$g_N$-twined odd sector ($r_f(N) = 2$ uniformly across the
boundary). The subscripted invariant $\kBKM(\Phi_N) = c_N(0)/2$
carries through half-integer weights via the metaplectic lift,
recovering the Borcherds weight theorem on $\CY_2^{\Mp}$.
