\subsection*{Cycle I. The three Macdonald parameters: origin and CY$_3$ locus}
% =====================================================================

\ClaimStatusAttack\ (\emph{Three parameters from thin air.})\
The three-parameter Nekrasov family $(q_1, q_2, q_3)$ or additively
$(\epsilon_1, \epsilon_2, \epsilon_3)$ is often introduced without deriving
\emph{why} they obey $q_1 q_2 q_3 = 1$ / $\epsilon_1 + \epsilon_2 + \epsilon_3 = 0$.
The identity looks like an ansatz.

\ClaimStatusHeal\ (\emph{Derivation from the anti-canonical action.})\

$T = (\C^\times)^3 \curvearrowright \C^3$ by
$(t_1, t_2, t_3) \cdot (x, y, z) = (t_1 x, t_2 y, t_3 z)$. The characters

$X^\bullet(T) \cong \Z^3$. The holomorphic top form
$\Omega_{\C^3} = dX \wedge dY \wedge dZ$ has $T$-weight

anti-canonical kernel
\[
T^{\mathrm{CY}} := \ker\bigl(\epsilon_1 + \epsilon_2 + \epsilon_3 \colon T \to \C^\times\bigr)
\;\subset\; T.
\]
Hence the \emph{two-parameter} torus $T^{\mathrm{CY}} \cong (\C^\times)^2$

by $\epsilon_3 = -\epsilon_1 - \epsilon_2$. The ``three parameters with

\ClaimStatusHeal\ (\emph{Shuffle-side generator + Yangian-mode dictionary.})\

Let $\Lambda = \bigoplus_{n \geq 0} \Lambda_n$ with
$\Lambda_n := \mathbb{F}[z_1, \ldots, z_n]^{S_n}$ equipped with the

\[
(F \star G)(x_1, \ldots, x_{m+n}) =
\frac{1}{m! n!} \sum_{\sigma \in S_{m+n}} \sigma \cdot
\Big[F(x_1, \ldots, x_m)\, G(x_{m+1}, \ldots, x_{m+n})

elementary sums $f_{n,k}(z_1, \ldots, z_n) := \sum_{i=1}^{n} z_i^k \in \Lambda_n$

\begin{enumerate}
\item \emph{Primitivity.} $f_{n, k}$ for $n \geq 2$ is \emph{not} a primitive

$k \in \Z_{\geq 0}$, so $\Yp$ is generated as a shuffle algebra by
$\{f_{1, k} = z^k : k \geq 0\} \subset \Lambda_1$.
\item \emph{Half only.} The full Drinfeld currents involve negative modes
$E_n$ with $n < 0$ (equivalently $F_n$ for $n \geq 0$), which do not live
on $\bigoplus_n \Lambda_n$ at all. They require Hall-Drinfeld doubling
(Cycle V).
\end{enumerate}

\subsection*{Cycle V. Maulik--Okounkov stable envelopes: restoring the negative half geometrically}

\ClaimStatusAttack\ (\emph{``Stable envelopes give $E$ and $F$ automatically.''})\
The Maulik--Okounkov 2019 (Ast.~408) stable-envelope construction on

\ClaimStatusHeal\ (\emph{Stable envelopes generate $\YFull$, not $\Yp$.})\

\paragraph{The MO fixed-point basis.}\
For $T = (\C^\times)^2$ acting on $\Hilb^n(\C^2)$, the fixed points are
labelled by partitions $\lambda \vdash n$. MO stable envelopes
$\mathrm{Stab}_\mathfrak{C}: H^*_T(\Hilb^n(\C^2)^T) \to H^*_T(\Hilb^n(\C^2))$
depend on a chamber $\mathfrak{C}$.

\paragraph{The universal $R$-matrix.}\
For the positive/negative chambers $\mathfrak{C}_{\pm}$,
\[
R_{\MO}(u) := \mathrm{Stab}_{-}^{-1}\, \mathrm{Stab}_{+}
\;\in\; \mathrm{End}\bigl(H^*_T(\Hilb^n(\C^2)^T)\bigr)(u),
\]
diagonal in the partition basis with entries
\[
R_{\lambda, \mu}(u) = \prod_{s \in \lambda, t \in \mu} g(u + c(s) - c(t)),
\qquad g(u) = \frac{(u - \epsilon_1)(u - \epsilon_2)(u - \epsilon_3)}{(u + \epsilon_1)(u + \epsilon_2)(u + \epsilon_3)}.

This matches \emph{exactly} the affine Yangian structure function

\paragraph{Why MO gives the full $\YFull$.}\
The MO construction takes $\bigoplus_n H^*_T(\Hilb^n(\C^2))$ and acts by
stable-envelope correspondence operators \emph{both} increasing and
decreasing $n$. The increasing direction generates $\Yp$ (SV's shuffle
side); the decreasing direction generates $\Ym$; together they give
$\YFull$ with non-degenerate Hopf pairing
\[
\langle \alpha, \beta \rangle_{\MO} :=
\int^{T}_{\Hilb^n(\C^2)} \mathrm{Stab}_{+}(\alpha) \cup \mathrm{Stab}_{-}(\beta),
\]
the geometric incarnation of the Drinfeld pairing $\Yp \otimes \Ym \to \mathbb{F}$.

\paragraph{Comparison to $\C^3$ CoHA.}\
For $\C^3$ one restricts $\C^2 \hookrightarrow \C^3$ and pulls back the
$T^3$-action to $(\C^\times)^2$ via $T^{\mathrm{CY}}$. The MO construction
on $\Hilb^n(\C^2)$ therefore produces a $T^{\mathrm{CY}}$-equivariant
Drinfeld double containing the $\C^3$-CoHA as its positive half.

\begin{theorem}[Corrected CoHA = positive-half-Yangian, with $\Wilpha$ extension]
\label{thm:wave12-a7-coha-drinfeld-heal}
Let $\mathbb{F} = \Q(\epsilon_1, \epsilon_2, \epsilon_3)/(\epsilon_1 + \epsilon_2 + \epsilon_3)$,
$T^{\mathrm{CY}} = \ker(\sum \epsilon_i)$ acting on $\C^3$ preserving $\Omega_{\C^3}$.
\begin{enumerate}
\item[\upshape(a)]\ (Schiffmann--Vasserot 2013 \texttt{arXiv:1202.2756} Thm~1.1;
Schiffmann--Vasserot 2017 \emph{Publ.\ IHES} 118.)\
$\CoHA_{T^{\mathrm{CY}}}(\C^3)$ is isomorphic as an associative algebra to
the positive half $\Yp(\gglone)$ of the affine Yangian of $\gglone$, via the
Feigin--Odesskii shuffle presentation with kernel
$\omega(z, w) = \prod_{i=1}^3 (z-w-\epsilon_i)/(z-w)^3$.
\item[\upshape(b)]\ (Negut 2014 \emph{Publ.\ IHES}; MO 2019 Ast.~408.)\
Hall-Drinfeld doubling with respect to the Euler / MO stable-envelope pairing
restores $\Ym$ and $\Yz$, giving
$\Yp \bowtie \Yz \bowtie \Ym = \YFull(\gglone)$.
\item[\upshape(c)]\ (Miki 2007; Tsymbaliuk 2017 Thm~1.1;
Arbesfeld--Schiffmann 2013 \texttt{arXiv:1209.0429};
Gaiotto--Rap\v{c}\'ak 2017 \texttt{arXiv:1703.00982}.)\
$\YFull(\gglone) \cong \mathrm{modes}(\Wilpha[\lambda])$ via the Miki
$T \in SL_2(\Z)$ automorphism of the quantum toroidal algebra
$\mathcal{E}_{q_1, q_2, q_3}(\gglone)$; specifically
$E_n \leftrightarrow J^{(2)}_{-n-1}$, $F_n \leftrightarrow J^{(2)}_{n+1}$,
$\psi^\pm_n \leftrightarrow J^{(1)}$-mode series.
\item[\upshape(d)]\ (Costello 2013 \texttt{arXiv:1303.2632}; Costello--Paquette 2020
\texttt{arXiv:2009.04834}.)\
$\Phi_3(\mathrm{Perf}(\C^3)) \cong \Wilpha[\lambda]$ as vertex algebras
(equivalently, as holomorphic factorisation algebras on the spectral
line $\C_z$). $\Phi_3$ is the state-field correspondence applied to
$\mathcal{E}^{\mathrm{hor}}$; $\lambda$ is determined by the
transverse $\Omega$-weights.
\item[\upshape(e)]\ Five objects, never conflated:
\[
\CoHA(\C^3) = \Yp \;\;\subsetneq\;\; \YFull \;\;\simeq\;\; \mathcal{E}^{\mathrm{rat}}
\;\;\stackrel{T}{\simeq}\;\; \mathcal{E}^{\mathrm{hor}}
\;\;\stackrel{\text{state-field}}{\Longrightarrow}\;\; \Wilpha[\lambda].
\]
The first is an $E_1$-associative algebra (no Hopf, no chiral); the second
and third are $E_1$-associative Hopf algebras (no vertex); the fourth is
a quantum toroidal Hopf algebra with $SL_2(\Z)$; the fifth is a VOA
(vertex algebra; $E_1$-chiral).
\end{enumerate}
\end{theorem}

\begin{proof}[Proof sketch.]
\emph{(a)} SV 2013: identify critical cohomology of the Jordan-triple loop
quiver $(Q = \bullet \circlearrowleft X, Y, Z,\, W = \mathrm{tr}(X[Y, Z]))$
with the shuffle algebra $\Lambda$ via $T^{\mathrm{CY}}$-localisation on
plane-partition fixed points in $(\Hilb^n(\C^2))^T$ (Cycle I).
SV 2013 Thm~1.1 gives $\Lambda \cong \Yp$.
\emph{(b)} Hall-Drinfeld doubling with the Euler pairing produces
$\YFull$ (Negut 2014 Thm~1.4). MO 2019 \S14 identifies the stable-envelope
pairing with the Euler pairing.
\emph{(c)} Miki 2007 \S3 and Tsymbaliuk 2017 Thm~1.1 on the
$\gglone$ case; $\lambda$ identification via Gaiotto--Rap\v{c}\'ak 2017 \S4.
\emph{(d)} Costello 2013 and Costello--Paquette 2020: the holomorphic-twist
factorisation algebra on $\C_z$ is $\Wilpha[\lambda]$; low-order OPE
verified, full identification relies on Gaiotto--Rap\v{c}\'ak / Costello--Paquette
framework.
\emph{(e)} Categorical: $\Yp, \YFull, \mathcal{E}^{\mathrm{rat}}, \mathcal{E}^{\mathrm{hor}}$
are associative algebras (no state-field operation, only one multiplication);
$\Wilpha$ is a vertex algebra (locality, state-field, infinitely many OPE modes).
They live in distinct categories.
\end{proof}
