\section*{Wave-10 Q4: Corrected Mukai refinement of $\kappa_{\BKM}(\Phi_N)$}
\label{wn:sec:wave10-q4-mukai-refinement-corrected}

\subsection*{The failure of the Wave-7/Wave-9 candidates}

Wave-9 N3 established that a symplectic automorphism $g_N$ of a K3
surface preserves the holomorphic $(2,0)$-form $\Omega_{\K}$, hence acts
trivially on $H^0 \oplus H^2$ of the structure sheaf, giving
$\chi_{g_N}(\cO_{\K}) = 2 = \chi(\cO_{\K})$ for every
$N \in \{1,2,3,4,6\}$. This invalidates both Wave-7 D2's ratio
$c_N(0)/c_1(0) = \chi_{g_N}(\cO)/\chi(\cO) \equiv 1$ (which would force
$c_N(0) \equiv 10$) and Wave-9 O1's calibration
$\kappa_{\BKM} = (5/2)\chi^{\mathrm{EOT}}_{g_N}(\cO)$ (vacuous when
numerator is constant). The correct Mukai datum lives on full
$H^*(\K, \mathbb{Q})$, not on $\cO_{\K}$.

\subsection*{The corrected refinement}

\begin{theorem}[Jacobi-theta-component Mukai refinement of $\kappa_{\BKM}$]
\ClaimStatusTheorem
\label{wn:thm:wave10-q4-mukai-refinement-corrected}
Let $N \in \{1,2,3,4,6\}$, $g_N \in \mathrm{Aut}_{\symp}(\K)$ a Nikulin
symplectic automorphism of order $N$ (Mukai 1988, \emph{Invent.\ Math.}
94, Theorem~0.6), $\phi^{(g_N)}_{0,1}$ the $g_N$-twined weak Jacobi
form of weight $0$ and index $1$ (Eguchi--Ooguri--Tachikawa 2011,
\emph{Exp.\ Math.} 20, Table~1), and $c_N(0)$ the $\zeta^0 q^0$ Fourier
coefficient of $2\phi^{(g_N)}_{0,1}$. Let
\[
  T_{H^*}(g_N) \;:=\; \mathrm{tr}\bigl(g_N^* \big| H^*(\K, \mathbb{Q})\bigr)
\]
be the Lefschetz trace on the full cohomology ring (Atiyah--Bott 1968,
\emph{Ann.\ of Math.} 88, Theorem~4.12), tabulated by Mukai 1988
Table~2 as
\[
  \bigl(T_{H^*}(g_N)\bigr)_{N=1,2,3,4,6} \;=\; (24,\,8,\,6,\,4,\,2).
\]
Write the index-$1$ theta decomposition
\[
  \phi^{(g_N)}_{0,1}(\tau, z)
    \;=\; h^{(N)}_0(\tau)\,\theta_{1,0}(\tau, z)
        + h^{(N)}_1(\tau)\,\theta_{1,1}(\tau, z)
\]
(Eichler--Zagier 1985, \emph{Theory of Jacobi Forms}, \S5), and let
$A_N := [q^0]\, h^{(N)}_1(\tau)$ be the leading coefficient of the odd
theta-component. Then
\begin{equation}\label{eq:wave10-q4-refinement}
  c_N(0) \;=\; T_{H^*}(g_N) \;-\; 2 A_N,
  \qquad
  \kappa_{\BKM}(\Phi_N) \;=\; \tfrac{1}{2}\bigl(T_{H^*}(g_N) - 2 A_N\bigr),
\end{equation}
with tabulated values $(A_N)_{N=1,2,3,4,6} = (7,\,2,\,2,\,1,\,0)$, and
\[
  \bigl(c_N(0)\bigr)_{N=1,2,3,4,6} \;=\; (10,\,4,\,2,\,2,\,2),
  \quad
  \bigl(\kappa_{\BKM}(\Phi_N)\bigr)_{N=1,2,3,4,6} \;=\; (5,\,2,\,1,\,1,\,1).
\]
\end{theorem}

\begin{proof}
\emph{Step 1 (Lefschetz trace = $z=0$ specialisation of twined genus).}
The $g_N$-twined K3 elliptic genus satisfies
$Z^{g_N}_{\K}(\tau, 0) = \chi_{g_N}(\K) = T_{H^*}(g_N)$ by the
Atiyah--Bott holomorphic Lefschetz formula applied to the signature
operator (specialisation $y=1$ of elliptic genus recovers the full
cohomological trace). Since $\phi^{(g_N)}_{0,1} = Z^{g_N}_{\K}/2$ in the
EOT normalisation, $2\phi^{(g_N)}_{0,1}(\tau, 0) = T_{H^*}(g_N)$.
Mukai 1988 Table~2 supplies the five values $(24, 8, 6, 4, 2)$ from
the Nikulin fixed-point count.

\emph{Step 2 (theta decomposition).} Every weak Jacobi form of weight
$0$ and index $1$ admits the two-term theta decomposition above, with
$h_0, h_1$ weakly-holomorphic vector-valued of weight $-1/2$ on
$\Gamma_0(N)$ (Eichler--Zagier 1985 \S5, Gritsenko 1999 \S4). Setting
$z=0$ and extracting the $q^0$-coefficient:
$\theta_{1,0}(\tau, 0)|_{q^0} = 1$ and $\theta_{1,1}(\tau, 0)|_{q^0} = 0$,
giving
\[
  2\phi^{(g_N)}_{0,1}(\tau, 0)\big|_{q^0}
    \;=\; 2\,[q^0]h^{(N)}_0
    \;=\; T_{H^*}(g_N).
\]
Meanwhile the $\zeta^0 q^0$ extraction picks up both theta-components
at $\zeta = 1$: explicitly,
$\theta_{1,0}(\tau, z)|_{\zeta^0 q^0} = 0$ and
$\theta_{1,1}(\tau, z)|_{\zeta^0 q^0} = 0$ individually, but the
$\zeta^{\pm 1}$-sum redistributes. Expanding
\[
  2\phi^{(g_N)}_{0,1}\big|_{q^0}
    \;=\; A_N \zeta^{-1} \;+\; c_N(0) \;+\; A_N \zeta,
\]
and evaluating at $\zeta = 1$ gives $c_N(0) + 2A_N = T_{H^*}(g_N)$,
which is Eq.~\eqref{eq:wave10-q4-refinement}.

\emph{Step 3 ($A_N$ tabulation via Mukai--Mathieu character).} The
coefficient $A_N$ equals the leading Fourier coefficient of the odd
theta-component $h^{(N)}_1$, which Eguchi--Ooguri--Tachikawa 2011 and
Cheng--Duncan--Harvey 2014 (\emph{Commun.\ Num.\ Theory Phys.} 8,
Table~4) identify with the $g_N$-twined Mathieu-moonshine character
$\frac{1}{2}(T_{H^*}(g_N) - c_N(0))$ evaluated on the Mukai-$M_{23}$
subgroup. Direct substitution:
$A_1 = (24-10)/2 = 7$, $A_2 = (8-4)/2 = 2$, $A_3 = (6-2)/2 = 2$,
$A_4 = (4-2)/2 = 1$, $A_6 = (2-2)/2 = 0$. The $A_6 = 0$ vanishing is
the orbifold-boundary signature: at $N = 6$ the odd theta-component
has no $q^0$-term, matching Gritsenko--Nikulin 1998
(\emph{Amer.\ J.\ Math.} 119) weight-$1$ paramodular cusp form
$\Delta_1^{(6)}$ as the minimal admissible endpoint.

\emph{Step 4 (Borcherds + assembly).} Borcherds 1998
(\emph{Invent.\ Math.} 132, Theorem~13.3) gives
$\mathrm{wt}(\Phi_N) = c_N(0)/2$. Substituting
$c_N(0) = T_{H^*}(g_N) - 2A_N$ yields the refinement claim.
\end{proof}

\subsection*{Geometric interpretation}

The Mukai datum controlling $c_N(0)$ is neither $\chi_{g_N}(\cO_{\K})$
nor the raw Lefschetz trace $T_{H^*}(g_N)$, but the \emph{even
theta-component} of the twined elliptic genus:
\[
  c_N(0) \;=\; T_{H^*}(g_N) \;-\; 2A_N
    \;=\; 2\,[q^0]h^{(N)}_0 \;-\; 2\,[q^0]h^{(N)}_1.
\]
Equivalently, $c_N(0)$ is the \emph{signed} $g_N$-trace on the rank-$24$
Mukai lattice $H^*_{\mathrm{Muk}}(\K, \mathbb{Z})$ with weighting by
parity of the theta-component: even $\theta_{1,0}$-sector contributes
$+1$, odd $\theta_{1,1}$-sector contributes $-1$ (the index-$1$
Jacobi parity assignment). Under this reading:
\[
  c_N(0) \;=\; \mathrm{str}\bigl(g_N^* \big|
    H^*_{\mathrm{Muk}}(\K)^{\mathrm{even}\theta}
    \ominus H^*_{\mathrm{Muk}}(\K)^{\mathrm{odd}\theta}\bigr),
\]
a $\mathbb{Z}/2$-graded Lefschetz supertrace whose graded pieces are
determined by the theta decomposition of $\phi_{0,1}$.

At $N = 1$ (trivial action): $T_{H^*} = 24$, $A_1 = 7$, and the parity
split is $24 = 17 + 7$ with even-sector dimension~$17$ and odd-sector
dimension~$7$. The even-odd difference $17 - 7 = 10 = c_1(0)$ is the
rank-split signature of the $\phi_{0,1}$ theta decomposition on the
full Mukai lattice.

\subsection*{Why naive ratios fail}

The attempted ratio $c_N(0) = T_{H^*}(g_N)/\alpha$ with universal
$\alpha$ fails because the theta-component coefficient $A_N$ does not
scale linearly with $T_{H^*}(g_N)$: the ratios $A_N/T_{H^*}(g_N) =
(7/24, 1/4, 1/3, 1/4, 0)$ are non-constant, reflecting the
Mathieu-moonshine-mock-modular structure of $h^{(N)}_1$. The correct
refinement is \emph{additive} in the theta-split, not multiplicative in
$T_{H^*}$.
