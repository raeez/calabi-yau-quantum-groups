\section*{$N=3$ CHL effective positive half and BPS Lie algebra
$\fg^{(3)}_{\BPS}$}
\label{wn:sec:wave19-x3-N3-positive-half-BPS}

\paragraph{Datum.} Fix a symplectic Nikulin automorphism
$g_3 \in \Auts(K3)$ of order $3$, class $3A \subset M_{23}$ in the
Mukai embedding (Nikulin, \emph{Moscow Math.\ J.}\ 1980 Table~1;
Mukai \emph{Invent.\ Math.}\ \textbf{94} (1988) Thm.~0.1). The
invariant sublattice
$\Lambda^{g_3}_{K3}\subset\mathrm{II}_{3,19}$ has rank $r_3 = 10$,
signature $(1,9)$; the coinvariant is $A_2^{\oplus 4}$ of rank $12$,
discriminant form $(\bZ/3\bZ)^{\oplus 4}$. Fix a primitive translation
$t_3 \in E[3]$ of exact order~$3$; the diagonal
$\sigma_3 = (g_3, t_3)$ acts freely on $K3 \times E$ (translation
$t_3$ acts freely on $E$, so diagonal fixes require simultaneously
$g_3$-fixed K3-points and trivial $t_3$-shift, forbidden).
The $N=3$ CHL Calabi--Yau threefold is
\[
  X_3 \;=\; (K3 \times E)\big/\langle\sigma_3\rangle,
\]
with $\chi(\cO_{X_3}) = \tfrac{1}{3}\chi(\cO_{K3\times E}) = 0$
(CHL arXiv:hep-th/9506144 \S 2; Bryan--Oberdieck arXiv:1811.06102
\S 2.1).

\paragraph{Convention (M4 Borcherds-weight normalisation).} The
$g_3$-twined K3 elliptic genus $Z^{(g_3)}_{K3}(\tau,z) =
2\,\phi^{(g_3)}_{0,1}$ with $\phi^{(g_3)}_{0,1}$ the CDH weak
Jacobi form of weight $0$, index $1$ at class $3A$ (CDH
arXiv:1204.2779 Table~2 row $3A$). Its constant Fourier coefficient is
$c_3(0) \,=\, \phi^{(g_3)}_{0,1}(\tau,0)|_{q^0} \,=\, 2$, so the
Borcherds-weight paramodular form at level $3$ has weight
$k_3 = c_3(0)/2 = 1$. This is the convention of the universal
identity $\kBKM(\Phi_N) = c_N(0)/2$ across $N\in\{1,2,3,4,6\}$.
Distinct from: the $g_3$-twined total K3 Euler trace
$\chi_{g_3}(K3) = 6$, which is the Hodge supertrace on all of
$H^*(K3)$ and enters $\kfib$, not $\kBKM$.

\paragraph{Cycle 1: $X_3$ smooth CY$_3$ with Kummer-structured fibre.}
\emph{Attack.} Does $g_3$ produce singularities? \emph{Heal.} No:
the $g_3$-fixed K3 locus $\Fix(g_3)$ has six isolated points
(Nikulin 1980, Table~1; total fixed-point count $\chi(\Fix(g_3)) =
6$ matches $\chi_{g_3}(K3)$); but the diagonal $\sigma_3$ is free
on $K3\times E$, so $X_3$ is smooth Calabi--Yau of dimension $3$.
Projection $\pi\colon X_3 \to E/\langle t_3\rangle \cong E'$ onto
the $3$-isogenous elliptic curve has generic fibre $K3$; at three
branch points the fibre is $K3/\langle g_3\rangle$, whose minimal
resolution is the Nikulin K3 of degree~$3$ with $4$ additional
$A_2$-resolution trees over the six nodes. \ClaimStatusTheorem\
at the scheme level.

\paragraph{Cycle 2: twisted reduced DT and $Z^{\red}_{\DT}(X_3) =
-1/\Phi_3^{(3)}$.}
\emph{Attack.} Does the reduced virtual class survive the quotient?
\emph{Heal.} Yes: $g_3$ preserves $\omega_{K3}$ symplectically and the
Kool--Thomas cosection $\sigma\colon\mathrm{Obs}\to\cO$
(arXiv:1112.3069 \S 4) is $\sigma_3$-equivariant. Define the twisted
reduced DT moduli
\[
  \cM^{\red,\sigma_3}_{\DT}(X_3;\gamma)
  \;=\; \bigl(\cM^{\red}_{\DT}(K3\times E;\gamma)\bigr)^{\sigma_3},
\]
indexed by curve classes $\gamma$ in the invariant sublattice
$H_2(X_3;\bZ)$. The reduced DT partition function is
\[
  Z^{\red}_{\DT}\bigl(X_3;q,p,y\bigr)
  \;=\; -\frac{1}{\Phi_3^{(3)}(\tau,\sigma,z)},
  \qquad \mathrm{wt}\,\Phi_3^{(3)} \,=\, 1,
\]
where $\Phi_3^{(3)} = \GL(\phi^{(g_3)}_{0,1})$ is the Gritsenko additive
lift at level $3$ (Bryan--Oberdieck 1811.06102 Thm.~1, proved on
primitive classes; Oberdieck--Pandharipande 1603.05238 for $N=1$
base case). This is a paramodular cusp form on
$\Gamma_0^{+}(3)\subset\Sp_4(\bZ)$ of weight $1$, index $1$, with
character determined by the Humbert divisor count
$\sum_\ell \ell^2 c^{(g_3)}(0,\ell) = 0$ (Gritsenko 1999
\emph{St.\ Petersburg Math.\ J.}\ \textbf{11} Thm.~1.2).
\ClaimStatusTheorem\ at primitive curve classes; \ClaimStatusConjectured\
at non-primitive classes pending Bryan--Oberdieck Conjecture~0.1
completion.

\paragraph{Effective positive-half moduli $\cM^+_\eff(X_3)$.} The
CHL positive half is
\[
  \cM^+_\eff(X_3) \;=\; \coprod_{\gamma\in C^+_3}
  \cM^{\red,\sigma_3}_{\DT}(X_3;\gamma),
\]
with $C^+_3\subset H_2(X_3;\bZ)$ the effective cone of classes with
$\beta_{K3}\in\mathrm{NS}(K3)^{g_3}_+$ (non-negative on the invariant
ample) and $n\cdot [E']\geq 0$. On its cohomology the
Schiffmann--Vasserot multiplication (arXiv:1705.07488 \S 6) restricted
to $\sigma_3$-invariants makes $\Yp(K3\times E)^{\sigma_3}$ into the
$E_1$-chiral CHL positive half.

\paragraph{Cycle 3: BPS Lie algebra $\fg^{(3)}_{\BPS}$.}
\emph{Attack.} Does the Davison--Meinhardt BPS Lie algebra descend
through $\sigma_3$-fixing? \emph{Heal.} Yes: Davison--Meinhardt
arXiv:1512.08898 Thm.~A constructs the BPS sheaf
$\mathrm{BPS}^{\red}_{K3\times E} = \bigoplus_\gamma
\cM_\gamma[1]$ as a perverse constructible sheaf; the Kontsevich--Soibelman
motivic Hall bracket (arXiv:0811.2435) is $\sigma_3$-equivariant
because $\sigma_3$ permutes the integration-map data among equal-slope
orbits. Taking $\sigma_3$-invariants,
\[
  \fg^{(3)}_{\BPS} \;=\; H^*\bigl(\mathrm{BPS}^{\red}_{K3\times E}
  \bigr)^{\sigma_3} \;=\; \bigoplus_{\gamma\in C^+_3}
  \bH^{\BPS}_\gamma(X_3),
\]
a $\bZ_{\geq 0}^{\oplus}$-graded Lie algebra on the Davison perverse
filtration (Davison \emph{Ann.\ Sci.\ ENS} 2023 \S 5).
\ClaimStatusConjectured\ at $N=3$ full generality;
\ClaimStatusTheorem\ at primitive classes via Kinjo--Park--Safronov
arXiv:2312.00877 dimensional-reduction compatibility.

\paragraph{Cycle 4: identification $\fg^{(3)}_{\BPS} \cong
\fg_{\Delta_1^{(3)}}$.}
\emph{Attack.} Does the BPS character match the Gritsenko--Cl\'ery
BKM denominator? \emph{Heal.} The character of $\fg^{(3)}_{\BPS}$ is the
logarithmic derivative
\[
  \prod_{\alpha\in\Delta^+_{\fg^{(3)}_{\BPS}}}
  (1 - e^{-\alpha})^{\mathrm{mult}(\alpha)}
  \;=\; \Phi_3^{(3)}(\tau,\sigma,z),
  \qquad
  \mathrm{mult}(\alpha) \,=\, c^{(g_3)}(4nm - \ell^2,\ell),
\]
the Fourier coefficients of $\phi^{(g_3)}_{0,1}$ with $c^{(g_3)}(-1,\pm
1) = 1$, $c^{(g_3)}(0,0) = 2$ (CDH 1204.2779 Table~2). This equates
$\fg^{(3)}_{\BPS}$ with the generalised Kac--Moody superalgebra
$\fg_{\Delta_1^{(3)}}$ whose root system is read from $\Delta_1^{(3)} =
\GL(\phi^{(g_3)}_{0,1})$, the Gritsenko--Cl\'ery additive lift of
weight~$1$ (arXiv:0812.3962 Thm.~4.2; Gritsenko--Cl\'ery
\emph{J.\ Reine Angew.\ Math.}\ \textbf{616} (2008)). In this
normalisation $\Phi_3^{(3)} = \Delta_1^{(3)}$: both are the paramodular
cusp form of weight $1$, level $3$. \ClaimStatusConjectured\ at full
scope (Lorgat 2020 Conjecture~1, $N=3$ row of the eight-form census).

\paragraph{Cycle 5: Borcherds lift $\phi^{(g_3)}_{0,1}\to
\Delta_1^{(3)}$.}
\emph{Attack.} Is the additive lift well-defined? \emph{Heal.} Yes:
Borcherds arXiv:alg-geom/9609022 Thm.~13.3 plus Gritsenko 1999
Thm.~1.2 send a weight-$0$ index-$1$ weak Jacobi form of level $N$
to a paramodular cusp form of weight $c_N(0)/2$ on
$\Gamma_0^+(N)$. At $N=3$, input $\phi^{(g_3)}_{0,1}$, output
$\Delta_1^{(3)}$ of weight $1$: explicit Borcherds product
\[
  \Delta_1^{(3)}(\tau,\sigma,z)
  \;=\; q\,y\,p\!\prod_{(n,\ell,m)>0}\!(1-q^n y^\ell p^m)^{
  c^{(g_3)}(4nm-\ell^2,\ell)}.
\]
\ClaimStatusTheorem\ (Gritsenko--Cl\'ery 2008 Thm.~5.2, the
eight-form census at paramodular level $N=3$, $\mu=1$).

\paragraph{Four-invariant accounting on $X_3$.}
\[
\kcat(X_3) = 0\ (\text{K\"unneth}),\qquad
\kch(X_3) = 0\ (\text{$d=3$ Hodge supertrace}),
\]
\[
\kfib(X_3) = \chi_{g_3}(K3) = 6,\qquad
\kBKM(\Phi_3^{(3)}) = c_3(0)/2 = 1.
\]

\paragraph{Scope.} Cycles 1, 2, 5 are theorems; Cycles 3, 4 are
conjectural at full generality, theorems at primitive classes.
Chain-level lane: explicit Borcherds product for $\Delta_1^{(3)}$
with CDH-tabulated Fourier data. $(\infty,1)$-lane: Schiffmann--Vasserot
K3 CoHA as stable $\infty$-category, Davison--Meinhardt BPS Lie algebra
in perverse filtration. Both converge on $\fg^{(3)}_{\BPS} \cong
\fg_{\Delta_1^{(3)}}$; both load-bearing.
