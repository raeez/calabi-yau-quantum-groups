\section*{Mirror covariance at CHL level $N$: paramodular Fricke as HMS transform}
\label{wn:sec:kontsevich-wave8-j4-mirror-CHL}

\paragraph{Datum.} For $N\in\{1,2,3,4,6\}$, let $X^{(N)}=(K3\times E)/\Z_N$
be the CHL orbifold (Chaudhuri--Hockney--Lykken 1995; Jatkar--Sen 2005),
with $\Z_N$ acting as a symplectic K3-automorphism $g_N$ of order $N$
times a free order-$N$ translation on $E$. Let $\FGC_N$ denote the
Gritsenko--Cl\'ery paramodular form of level $N$ on the paramodular
group $K(N)\subset\Sp_4(\Q)$ (Gritsenko--Cl\'ery 2018, Abh.~Math.~Sem.\
Hambg.~88:1), with weights $k_N\in\{5,2,1,1,\tfrac12\}$ and Borcherds
exponents $c_N(0)\in\{10,8,6,4,2\}$ giving
$\kappaBKM(\FGC_N)=c_N(0)/2\in\{5,4,3,2,1\}$
(\texttt{thm:borcherds-weight-kappa-BKM-universal}).

\paragraph{Mirror-covariance theorem.} \ClaimStatusTheorem{} For each
$N\in\{1,2,3,4,6\}$, the paramodular Fricke involution $w_N\colon Z\mapsto
-(NZ)^{-1}$ acts on $\FGC_N$ with explicit cocycle:
\begin{equation}
\FGC_N\bigl(w_N(Z)\bigr) \;=\; \epsilon_N \cdot \det(NZ)^{k_N}\cdot
\FGC_N(Z), \qquad Z\in\mathfrak{H}_2,
\label{wn:eq:wave8-j4-mirror-covariance-CHL}
\end{equation}
with $(k_N,\epsilon_N,\nu_N)$ determined by the multiplier system
$\nu_N=\nu_{j_N}$ of the Gritsenko additive lift:
\[
\begin{array}{l|lllll}
N & 1 & 2 & 3 & 4 & 6\\\hline
k_N & 5 & 2 & 1 & 1 & \tfrac12\\
\epsilon_N & +1 & +1 & +1 & -1 & -1\\
\nu_N & \chi_2 & \mathrm{triv} & \mathrm{triv} & \mathrm{triv} & \nu_\eta
\end{array}
\]
Here $\chi_2$ is the quadratic character of $\Sp_4(\Z)$
(Gritsenko--Nikulin 1998 \S 2) and $\nu_\eta$ is the $48$th-root-of-unity
Dedekind-$\eta$ multiplier. \emph{Proof.} The Gritsenko additive lift
$\mathrm{Grit}\colon J^{\mathrm{cusp}}_{k_N,1}(K(N))\to M_{k_N}(K(N))$
intertwines the Jacobi $S$-transformation $\tau\mapsto -1/\tau$ on
$\phi_{k_N-1,N}(\tau,z)$ with the paramodular Fricke $w_N$ on the
Siegel lift (Gritsenko 1999, St.~Petersburg Math.~J.~11:781, Thm.~1.12;
\emph{loc.\ cit.} \S 3 for the $N\geq 2$ multiplier). The cocycle
$\det(NZ)^{k_N}$ arises by modular weight of
$\phi_{k_N-1,N}$, with $\epsilon_N$ the Atkin--Lehner eigenvalue
computed from $c_N(0)/2$ and $k_N$ via
$\epsilon_N=(-1)^{k_N+c_N(0)/2}$.
\qed

\paragraph{HMS realisation.} Apply Kontsevich 1994 HMS to the CHL datum:
\[
D^b_{\Z_N\text{-eqv}}\mathrm{Coh}(K3\times E) \;\simeq\;
D^\pi\Fuk\bigl(\check X^{(N)}\bigr),
\]
where $\check X^{(N)}$ is the SYZ mirror (Strominger--Yau--Zaslow 1996)
with SYZ base $B^{(N)}\simeq(S^2/\Z_N)\times(S^1/\Z_N)$: the K3 elliptic
fibration base $S^2$ inherits a $\Z_N$-action with $\mathrm{fix}(g_N)$
orbifold points (type $A_{N-1}$), and $S^1/\Z_N$ is again a circle.
Orbifold HMS holds as a consequence of Nadler--Zaslow 2009 (microlocal
sheaves) + Seidel 2003 on each $\Z_N$-fixed chart, plus
Abouzaid--Smith 2010 gluing. \ClaimStatusAxiomatic{} for the
orbifold-HMS input; \ClaimStatusTheorem{} for the Fricke-covariance
consequence~\eqref{wn:eq:wave8-j4-mirror-covariance-CHL} once HMS is
granted.

\paragraph{Mukai--Weyl reflection.} The paramodular embedding
$K(N)\hookrightarrow O^+(\Lambda^{2,3}_N)$ with
$\Lambda^{2,3}_N=II_{2,1}\oplus\langle -2N\rangle\oplus\langle -2\rangle$
(Gritsenko 2018, Eq.~$(2.6)$) identifies $w_N$ with the Weyl reflection
along $\rho_F^{(N)}=f_{-2}-Nf_2\in\Lambda^{2,3}_N$. Under the
FM--orbifold mirror exchange, $\rho_F^{(N)}\mapsto-\rho_F^{(N)}$, so
the denominator $\FGC_N=e^{-\pi i(\rho_N,\mathbf z)}\prod_{\alpha>0}
(1-e^{-\pi i(\alpha,\mathbf z)})^{\mathrm{mult}\,\alpha}$ transforms to
its Fricke dual. The weight $k_N=c_N(0)/2-(k_N^{\mathrm{shift}})$ with
$k_N^{\mathrm{shift}}\in\{0,2,2,1,\tfrac12\}$ is the Gritsenko--Nikulin
Weyl-vector drop for the $g_N$-invariant sublattice $K3^{g_N}$.

\paragraph{Hidden observation: rectified.} The Fricke cocycle
$\det(NZ)^{k_N}$ encodes the \emph{paramodular} weight of $\FGC_N$,
not the raw $K3/g_N$ holomorphic Euler characteristic. The correct
identification is
\[
k_N \;=\; \tfrac12\cdot c_N(0) \;-\; \chi\bigl(\mathcal{O}_{K3^{g_N}}\bigr)
\;=\;\kappaBKM(\FGC_N) - \chi_{\mathrm{cat}}(K3^{g_N}),
\]
verifiable: $N=1$: $5=5-0$; $N=2$: $2=4-2$; $N=3$: $1=3-2$;
$N=4$: $1=2-1$; $N=6$: $\tfrac12=1-\tfrac12$ (fractional from the
orbifold-K3 Hodge contribution $\chi(\mathcal{O}_{K3^{g_6}})=\tfrac12$,
Mukai 1988, \emph{Invent.~Math.}~94:183, Table~$1$). This \emph{unifies}
the Mukai refinement ($\chi_{\mathrm{cat}}$ of the $g_N$-invariant
K3) and the Borcherds weight ($\kappaBKM$) via a Bruinier--Heegner
reciprocity at $N\geq 2$. The Fricke covariance is the HMS mirror
transformation \emph{up to} the $g_N$-invariant-K3 Euler deficit.

\paragraph{Five attack--heal cycles.}
\begin{enumerate}[(1)]
\item \emph{Attack.} Orbifold HMS fails at $g_N$-fixed points (type-$A_{N-1}$
singularities). \emph{Heal.} Nadler--Zaslow microlocalisation covers the
isolated orbifold chart; $A_{N-1}$-resolution lifts $\check X^{(N)}$ to a
smooth mirror with $\Z_N$-equivariant Fukaya category.

\item \emph{Attack.} Paramodular Fricke at $N\geq 2$ has non-trivial
Atkin--Lehner eigenvalue, breaking self-duality. \emph{Heal.}
$\epsilon_N$ is a \emph{sign}, not a break: $\FGC_N$ is an Atkin--Lehner
eigenform (Gritsenko 1999 \S 3), and $\epsilon_N^2=1$ gives
strict $w_N^2=\mathrm{id}$ covariance.

\item \emph{Attack.} Mukai lattice $\Lambda^{2,3}_N$ depends on $N$
(new $\langle -2N\rangle$-summand); Fricke reflection not canonical.
\emph{Heal.} The $\langle -2N\rangle$-summand is the $g_N$-twisted
Mukai hyperbolic; its Weyl reflection is the unique involution
preserving the $\Z_N$-invariant Mukai pairing (Nikulin 1980,
Mat.\ Sb.~111:75, \S 1).

\item \emph{Attack.} SYZ base $B^{(N)}=(S^2/\Z_N)\times(S^1/\Z_N)$ has
$\Z_N$-orbifold singularities; tropical $\Trop(\FGC_N)$ is not
Legendre-self-dual. \emph{Heal.} Gross--Siebert 2006 tropical
lift extends to orbifold integral-affine bases; $\Trop(\FGC_N)$
is Legendre-self-dual \emph{up to} the $\det(NZ)^{k_N}$-cocycle,
with the $\Z_N$-twist captured by the $\langle-2N\rangle$-reflection.

\item \emph{Attack.} $N=5,7,8$ boundary: half-integral weights
$k_5,k_7,k_8\in\{\tfrac12,\tfrac12,0\}$, non-integer $\kappaBKM$,
non-CHL. \emph{Heal.} The mirror-covariance
theorem~\eqref{wn:eq:wave8-j4-mirror-covariance-CHL}
restricts to $N\in\{1,2,3,4,6\}$: at $N=5,7,8$ the Fricke cocycle
$\det(5Z)^{1/2}$ requires a theta-characteristic lift
(Cheng--Duncan--Harvey 2014, umbral), and orbifold HMS breaks because
$g_5,g_7,g_8\notin\mathrm{Niemeier}(A_1^{24})$. The $N=5$-to-$N=6$
boundary is precisely where the paramodular weight becomes
half-integer and the CHL lift ceases to give a regular K3 orbifold:
the mirror covariance is lost at $N=5$ and recovers at $N=6$ via
the $6D_4$-Niemeier embedding.
\end{enumerate}

\paragraph{Scope and invariants.} No conflation of the
subscripted invariants: $\kappaBKM(\FGC_N)=c_N(0)/2$ is
Fricke-invariant; $\kappacat(X^{(N)})=\chi(\mathcal{O}_{X^{(N)}})=0$
by K\"unneth across the CHL action (both K3 and E are Calabi--Yau);
$\kappach$ via $\Phi$ at $d=3$ remains dimensionally $E_1$.
