\section{Seven $\rCY$ faces: explicit $(\Sigma_2, C)$-witnesses on $K3\times E$}
\label{wn:sec:seven-faces-enumerate}

\emph{S5 partitioned the seven faces into three tiers. V3 nails the
concrete $(\Sigma_2, C)$-witness for each face, grounding the
enumeration in Mukai 1988, Borcherds 1995/1998, Harvey--Moore 1996,
and Lorgat 2020. The witness data is the identification card of the
face: it names the cycle pair, the restricted lattice, the surviving
Dolbeault summand, and the automorphic form reading $\rCY$.}

\subsection{Face $r_1 = r_{\Muk}$: Mukai lattice over $(K3, E)$}
\label{wn:subsec:r1-Mukai}

\begin{proposition}[$r_{\Muk}$ witness]\ClaimStatusProposition
\label{wn:prop:r1-Mukai-witness}
$(\Sigma_2, C) = (K3, E)$; restricted lattice
$\Lambda_{\Muk} = H^{2\ast}(K3, \ZZ)$ of signature $(4,20)$, rank $24$,
with pairing $\langle v_1, v_2\rangle = -\chi(v_1, v_2)$ (Mukai 1988
Invent.~77 \S2). The face reads
$$
r_{\Muk} \;=\; \text{the bilinear form on }\ K_0(D^b(K3))\otimes\QQ\ \text{induced from }\Lambda_{\Muk}.
$$
Under $\Sp_{K3, E}$, the abelian sub-$E_3$-hFA
$\cF^{\mathrm{ab}}_{K3\times E}$ projects to $\Uchir(\heis_{\Muk})$ on
$E$ (S1 Thm.~\texttt{wn:thm:sp-abelian}); the pairing on
$\Lambda_{\Muk}$ is the level-$1$ Heisenberg OPE. The face is the
\emph{restriction} of the Mukai pairing to the K3 fibre --- a tier-(i)
CY intrinsic that survives $\Sp$ as the quadratic form of the output
Fock space.
\end{proposition}

\subsection{Face $r_2 = r_{\mathrm{BKM}}$: Borcherds weight over $(K3, E_\tau)$}
\label{wn:subsec:r2-BKM}

\begin{proposition}[$r_{\mathrm{BKM}}$ witness]\ClaimStatusProposition
\label{wn:prop:r2-BKM-witness}
$(\Sigma_2, C) = (K3, E_\tau)$, the elliptic curve of modulus $\tau\in\HH$.
Stage~1 hands over $\cF_{K3\times E_\tau}\in E_3\text{-}\HolFA$; Stage~2
integrates over K3. The output on $E_\tau$ is the $E_1$-chiral bialgebra
$\bH_{\Delta_5}$ with denominator
$$
\Delta_5(\tau, z, \sigma) \;=\; \Phi_{10}(\tau,z,\sigma)^{1/2}
\;=\; \mathrm{Grit}(\phi_{0,1})(\tau, z, \sigma),
$$
the Gritsenko--Nikulin lift of $\phi_{0,1}$ (Gritsenko--Nikulin
alg-geom/9504006, Borcherds 1998 Thm.~13.3). The face reads
$$
\kBKM(r_{\mathrm{BKM}}) \;=\; c_1(0)/2 \;=\; 10/2 \;=\; 5,
$$
$c_1(0) = 10$ the constant Fourier coefficient of $\phi_{0,1}$ (Lorgat
2020 \S2). The Harvey--Moore 1996 threshold-integral identity
$\mathcal{I}_{\mathrm{het}}(T,U) = -\log|\Phi_{10}(T,U,V)|^2$ realises
$r_{\mathrm{BKM}}$ as the 1-loop heterotic $\mathcal{N}=2$ threshold on
$K3\times T^2$ --- the same $\Delta_5$ via a different $C$-fibre reading
(heterotic vs.\ IIA).
\end{proposition}

\subsection{Face $r_3 = r_{\mathrm{fiber}}$: reverse factorisation over $(E, \PP^1)$}
\label{wn:subsec:r3-fiber}

\begin{proposition}[$r_{\mathrm{fiber}}$ witness]\ClaimStatusProposition
\label{wn:prop:r3-fiber-witness}
$(\Sigma_2, C) = (E_\tau \times T^2_u, \PP^1)$ under the Shioda--Inose
elliptic-surface presentation $K3\to\PP^1$: the K3 fibres as an
elliptic surface over $\PP^1$ with $24$ singular fibres, so
$K3\times E$ acquires a \emph{second} factorisation
$(K3\times E)\to\PP^1$ with generic fibre $E_\tau\times T^2_u$ of
rank $\mathrm{rk}\,H^2 = 24 = \kfib$. The face reads
$$
\kfib(r_{\mathrm{fiber}}) \;=\; \mathrm{rk}\,H^{2\ast}(\Sigma_2, \ZZ)_{\mathrm{transv}} \;=\; 24,
$$
the total Mukai rank of the K3 fibre transverse to $\PP^1$, or
equivalently the Euler number $e(K3) = 24$ counted by Kodaira's
$I_1^{24}$-configuration. Reverse-factorisation: the $E_3$-hFA
$\cF_{K3\times E}$ admits the alternative fibration over
$C = \PP^1$ (cf.\ S2 Cycle~2, \texttt{wave16\_s2}); along this
$\PP^1$-base the specialisation reads $\kfib$ as the generic fibre
rank rather than $\kBKM$ as the automorphic weight. Tier-(ii): invariant
of $\cF_X$ upstream of Stage~2 but realised by the $\PP^1$-reversal.
\end{proposition}

\subsection{Face $r_4 = r_{\mathrm{Hodge}}$: canonical Stage-$1$ invariant}
\label{wn:subsec:r4-Hodge}

\begin{proposition}[$r_{\mathrm{Hodge}}$ witness]\ClaimStatusProposition
\label{wn:prop:r4-Hodge-witness}
No $(\Sigma_2, C)$ is consumed. The Hodge-supertrace
$$
\kch(K3\times E) \;=\; \chi(\cO_{K3\times E}) \;=\; \chi(\cO_{K3})\cdot\chi(\cO_E) \;=\; 2\cdot 0 \;=\; 0
$$
is K\"unneth-multiplicative on the total space (cache Key Fact;
never $2$, the fibre). The face reads $\chi(\cO_X)$ directly off the
CY datum; $\Sp$ does not touch it. Tier-(i) intrinsic.
\end{proposition}

\subsection{Face $r_5 = r_{\Niem}$: Niemeier over $(K3^{(i)}, E)$}
\label{wn:subsec:r5-Niemeier}

\begin{proposition}[$r_{\Niem}$ witness]\ClaimStatusProposition
\label{wn:prop:r5-Niemeier-witness}
$(\Sigma_2^{(i)}, C) = (K3^{(i)}, E)$ for $i = 1,\dots,23$, where
$K3^{(i)}$ is a Shioda--Inose K3 with transcendental lattice realising
$N^{(i)} = \Lambda_{\Niem}^{(i)}\oplus U$ for $\Lambda_{\Niem}^{(i)}$
one of the 23 Niemeier lattices with roots (Niemeier 1973; Conway--Sloane
\emph{SPLAG} Ch.~16). The face reads
$$
r_{\Niem}^{(i)} \;=\; \text{Borcherds lift of the Niemeier denominator}\ \eta_{\Lambda_{\Niem}^{(i)}},
$$
an automorphic form on $O(2, 2+\mathrm{rk}\,R^{(i)})$ indexed by the
Niemeier root system $R^{(i)}$ (Borcherds 1995 Invent.~132 Thm.~10.1).
$\Sp_{K3^{(i)}, E}(\cF_{X^{(i)}})$ outputs the $E_1$-chiral bialgebra
$\bH^{(i)}$ with denominator $\eta_{\Lambda_{\Niem}^{(i)}}$. The 23
witnesses index \emph{$\Sigma_2$-isogeny-class} choices, not $\Phi$
applications.
\end{proposition}

\subsection{Face $r_6 = r_{\Hum}$: Humbert-divisor monodromy over $(K3, E)$ near $H_n$}
\label{wn:subsec:r6-Humbert}

\begin{proposition}[$r_{\Hum}$ witness]\ClaimStatusProposition
\label{wn:prop:r6-Humbert-witness}
$(\Sigma_2, C) = (K3, E)$ with $C = E\to E_{H_n}$ approaching the
Humbert divisor $H_n \subset \overline{\cA_2}$ ($n = 1$: $E_1\times E_2$;
$n = 4$: CM-product). The face reads
$$
r_{\Hum}^{(n)} \;=\; \mathrm{Mon}_{H_n}\bigl(\Sp_{K3, E}(\cF_X)\bigr)
\;\in\; \{\mathrm{ord}\ 8\ \text{at}\ H_1,\ \mathrm{ord}\ 2\ \text{at}\ H_4\},
$$
the local monodromy of the nearby-cycles $E_1$-chiral as $(K3, E)$
approaches $H_n$ (Bruinier 2002 LNM~1780 Ch.~5;
\texttt{modular\_koszul\_bridge.tex} Prop.~\texttt{prop:humbert-monodromy-K-kappa}).
Witness: $r_{\Hum}$ is the \emph{discriminant locus} of the
$(\Sigma_2, C)$-moduli-space stratum within $\cA_2$, read off $\Delta_5$
via its vanishing orders along $H_1\cup H_4$ (Gritsenko--Nikulin 1995
\S3, divisor of $\Delta_5$).
\end{proposition}

\subsection{Face $r_7 = r_{\mathrm{tw}}$: CHL-twined family over $(K3/g_N, E/\ZZ_N)$}
\label{wn:subsec:r7-twined}

\begin{proposition}[$r_{\mathrm{tw}}$ witness]\ClaimStatusProposition
\label{wn:prop:r7-twined-witness}
For $N\in\{1,2,3,4,6\}$, CHL quotient $X_N = (K3\times E)/(\ZZ/N\ZZ)$
with $\ZZ/N\ZZ$ acting symplectically on K3 via Nikulin involution
$g_N$ and by $N$-torsion translation on $E$. Take
$(\Sigma_2^{(N)}, C^{(N)}) = (K3/\langle g_N\rangle, E/\ZZ_N)$. The
face reads
$$
\kBKM(r_{\mathrm{tw}}^{(N)}) \;=\; c_N(0)/2 \;\in\; \{5, 4, 3, 2, 1\}\ \text{at}\ N\in\{1,2,3,4,6\},
$$
with Gritsenko--Cl\'ery paramodular form $\Delta_{k_N}$ as the
denominator of $\Sp_{\Sigma_2^{(N)}, C^{(N)}}(\cF_{X_N})$ (Gritsenko
1999; Lorgat 2020 Conjecture~1). The five values $c_N(0)\in\{10,8,6,4,2\}$
are Fourier coefficients of the twined Jacobi forms
$\phi_{0,1}^{(N)} = \mathrm{Tw}_{g_N}(\phi_{0,1})$. Harvey--Moore 1996
\S4 reads the $N=2,3,4,6$ entries as heterotic $\ZZ_N$-orbifold
threshold integrals; the Borcherds lift converts each to a paramodular
form on $\mathrm{Sp}_4(\ZZ)^{+}_{N}$.
\end{proposition}

\subsection{Consequence: the face-count mystery dissolves}
\label{wn:subsec:consequence-enumerate}

The seven faces crystallise as: two CY-datum intrinsics
($r_{\Muk}, r_{\mathrm{Hodge}}$, tier~(i)); one Stage-$1$ invariant of
$\cF_X$ on a reverse factorisation ($r_{\mathrm{fiber}}$, tier~(ii));
four $(\Sigma_2, C)$-specialisations of $\Sp$ ($r_{\mathrm{BKM}},
r_{\Niem}, r_{\Hum}, r_{\mathrm{tw}}$, tier~(iii)). Three-tier
witnessing shows the seven-count is a numerology of the arithmetic
literature refined by the moduli of $(\Sigma_2, C)$-choices; each face
is identified by a concrete cycle-pair, restricted lattice, or
automorphic form. Harvey--Moore's duality-frame quartet (heterotic,
IIB, IIA, M-theory) re-reads $r_{\mathrm{BKM}}$ and $r_{\mathrm{tw}}$
across $C$-choices; Borcherds' Grassmannian-singular automorphic forms
re-read $r_{\Niem}$ across $\Sigma_2$-choices. No new $\rCY$ values,
one $\cF_X$.
