\begin{abstract}
The K3 elliptic genus is the weak Jacobi form
\[
  Z_{K3}(\tau,z)=2\,\phi_{0,1}^{\mathrm{EZ}}(\tau,z),
  \qquad
  \phi_{0,1}^{\mathrm{EZ}}=y^{-1}+10+y+O(q).
\]
Its small $\mathcal N=4$ character expansion contains a weight
$1/2$ mock modular form $H(\tau)$ whose harmonic Maass completion
has shadow $24\,\eta(\tau)^3$ in the standard $\xi_{1/2}$ normalisation.
The Mathieu statement is the existence of an $M_{24}$-graded module
whose graded traces give the twined mock modular forms. The
CY-to-chiral reading is an object-level interpretation of these
theorems. A functorial theorem consists of the additional bar-complex
and equivariance comparison stated below.
\end{abstract}

\section{The K3 elliptic genus}
\label{sec:k3-elliptic-genus}

Let $S$ be a smooth projective K3 surface and let
$\mathcal H_{\mathrm{RR}}$ be the Ramond-Ramond state space of the
$c=6$ $\mathcal N=(4,4)$ K3 sigma-model. Its elliptic genus is
\[
  Z_{K3}(\tau,z)=
  \str_{\mathcal H_{\mathrm{RR}}}
  \bigl((-1)^F y^{J_0^3}q^{L_0-c/24}\bigr),
  \qquad
  q=e^{2\pi i\tau},\quad y=e^{2\pi iz}.
\]
The Eichler-Zagier generator of $J^{\mathrm{weak}}_{0,1}$ is
normalised by
\[
  \phi_{0,1}^{\mathrm{EZ}}(\tau,z)
  =y^{-1}+10+y+q(10y^{-2}-64y^{-1}+108-64y+10y^2)+O(q^2).
\]
The K3 genus is
\[
  Z_{K3}(\tau,z)=2\,\phi_{0,1}^{\mathrm{EZ}}(\tau,z)
  =2y^{-1}+20+2y+O(q).
\]
The factor $2$ belongs to the elliptic genus. The primitive
Borcherds input used for $\Delta_5$ below is
$\phi_{0,1}^{\mathrm{EZ}}$ with constant coefficient $10$.

\begin{theorem}[EOT character decomposition and mock modularity]
\label{thm:eot-mock-modularity}
There is a holomorphic $q$-series
\[
  H(\tau)
  =2q^{-1/8}\bigl(-1+45q+231q^2+770q^3+2277q^4+\cdots\bigr)
\]
such that
\[
  Z_{K3}(\tau,z)
  =
  24\,\mu(\tau,z)\frac{\theta_1(\tau,z)^2}{\eta(\tau)^3}
  +
  H(\tau)\frac{\theta_1(\tau,z)^2}{\eta(\tau)^3},
\]
where $\mu$ is Zwegers's Appell-Lerch sum. The function $H$ is the
holomorphic part of a harmonic Maass form $\widehat H$ of weight
$1/2$. With
\[
  \xiop(F)(\tau):=2iv^{1/2}\,
  \overline{\frac{\partial F}{\partial \bar\tau}(\tau)},
  \qquad \tau=u+iv,
\]
the completion satisfies
\[
  \xiop(\widehat H)=24\,\eta(\tau)^3 .
\]
\end{theorem}

\begin{proof}
By Eichler-Zagier~\cite{EichlerZagier1985}, the space
$J^{\mathrm{weak}}_{0,1}$ is one-dimensional over the
weight-zero scalar modular forms, and the $q^0$ term of the K3
elliptic genus is fixed by the Hodge numbers of K3:
$2y^{-1}+20+2y$. Hence $Z_{K3}=2\phi_{0,1}^{\mathrm{EZ}}$.
Eguchi-Ooguri-Tachikawa~\cite{EguchiOoguriTachikawa2011} decompose
this weak Jacobi form into
small $\mathcal N=4$ Ramond characters at central charge $6$; the
massive-character multiplicities are encoded by the displayed
series $H$. Zwegers's completion of the Appell-Lerch sum gives the
non-holomorphic correction, and the resulting harmonic Maass form
has $\xi_{1/2}$-shadow $24\eta^3$.
\end{proof}

\section{Mathieu moonshine}
\label{sec:mathieu}

\begin{theorem}[Mathieu module and twinings]
\label{thm:mathieu-module}
There is a graded $M_{24}$-module
\[
  K^\natural=\bigoplus_{n\geq 1}K_n^\natural
\]
such that
\[
  H(\tau)=-2q^{-1/8}
  +\sum_{n\geq 1}\dim(K_n^\natural)\,q^{n-1/8}.
\]
Let $[g]$ range over the $26$ ordinary ATLAS conjugacy classes of
$M_{24}$:
\[
\begin{gathered}
1A,\ 2A,\ 2B,\ 3A,\ 3B,\ 4A,\ 4B,\ 4C,\ 5A,\ 6A,\ 6B,\ 7A,\ 7B,\\
8A,\ 10A,\ 11A,\ 12A,\ 12B,\ 14A,\ 14B,\ 15A,\ 15B,\ 21A,\ 21B,\
23A,\ 23B .
\end{gathered}
\]
For each such class the graded trace
\[
  H_g(\tau)=-2q^{-1/8}
  +\sum_{n\geq 1}\operatorname{tr}(g\mid K_n^\natural)\,q^{n-1/8}
\]
is the mock modular summand of the corresponding twined K3 elliptic
genus
\[
  Z_g(\tau,z)=
  \str_{\mathcal H_{\mathrm{RR}}}
  \bigl(g(-1)^F y^{J_0^3}q^{L_0-c/24}\bigr).
\]
The completion $\widehat H_g$ has shadow
\[
  \xiop(\widehat H_g)=\chi_g\,\eta(\tau)^3,
  \qquad
  \chi_g=\operatorname{tr}(g\mid \rho_{24}),
\]
with the multiplier and congruence subgroup determined by the
conjugacy class of $g$. Here $\rho_{24}$ is the $24$-dimensional
Mathieu character governing the twined K3 genus; for geometric
symplectic automorphisms it agrees with the trace on the Mukai
cohomology of $S$. The indexing is by ordinary conjugacy class; frame
shape is auxiliary data and can coincide for paired classes.
\end{theorem}

\begin{proof}
The character decomposition of Theorem~\ref{thm:eot-mock-modularity}
is the Eguchi-Ooguri-Tachikawa formula. Gaberdiel-Hohenegger-Volpato
\cite{GaberdielHoheneggerVolpato2010} construct the twined K3
elliptic genera and identify their $\mathcal N=4$ mock summands.
Gannon~\cite{Gannon2016} proves that the Fourier coefficients are
characters of genuine finite-dimensional $M_{24}$-representations.
Cheng-Duncan-Harvey~\cite{ChengDuncanHarvey2014} place the
$M_{24}$ case in the umbral family attached to the Niemeier root
system $24A_1$.
\end{proof}

\section{The \texorpdfstring{$\Delta_5$}{Delta5} normalization}
\label{sec:delta5}

The primitive Siegel modular form attached to the K3 Jacobi form is
Gritsenko's weight-$5$ cusp form $\Delta_5$. Write
\[
  \phi_{0,1}^{\mathrm{EZ}}(\tau,z)
  =\sum_{n,\ell}c_1(4n-\ell^2)\,q^n y^\ell .
\]
The $q^0y^0$ coefficient is
\[
  c_1(0)=10.
\]
The monic Borcherds product is
\[
  D_5(Z)
  =
  q^{1/2}r^{1/2}s^{1/2}
  \prod_{(n,\ell,m)>0}
  \bigl(1-q^n r^\ell s^m\bigr)^{c_1(4nm-\ell^2)},
  \qquad Z=\begin{pmatrix}\tau&z\\ z&\sigma\end{pmatrix},
\]
and Gritsenko's theta-normalised form satisfies
\[
  \Delta_5=64D_5 .
\]
The scalar $64$ changes the coefficient of
$q^{1/2}r^{1/2}s^{1/2}$ from $1$ to $64$ and leaves the Borcherds
weight unchanged:
\[
  \kBKM(\Delta_5)=\frac{c_1(0)}{2}=5.
\]

\begin{proposition}[Primitive CHL constants]
\label{prop:primitive-chl-constants}
For the primitive CHL denominator levels
$N\in\{1,2,3,4,6\}$, the single-cusp Borcherds inputs have
\[
\begin{array}{c|ccccc}
N & 1 & 2 & 3 & 4 & 6\\
\hline
c_N(0) & 10 & 8 & 6 & 4 & 2\\
\kBKM(\Phi_N) & 5 & 4 & 3 & 2 & 1 .
\end{array}
\]
This is a Borcherds-input table, separate from the ordinary Mathieu
twining table. Ordinary twined K3 genera carry class constants,
multipliers, and congruence subgroups; a Borcherds weight is defined
only after a primitive CHL denominator input has been fixed.
\end{proposition}

\begin{proof}
Borcherds's singular theta lift~\cite{Borcherds1998} assigns to a
weight-zero Jacobi input an automorphic product of weight one half of
the input's zeroth coefficient. Gritsenko's CHL inputs
\cite{Gritsenko1999} at the five primitive levels have the displayed
zeroth coefficients. The identity row is the K3 Eichler-Zagier
generator, so $c_1(0)=10$ and the resulting primitive form has
weight $5$.
\end{proof}

\begin{proposition}[Primitive and square normalisations]
\label{prop:delta5-square-normalisation}
The monic denominator product is
\[
  D_5=64^{-1}\Delta_5 .
\]
The scalar Igusa square data are
\[
  \Phi_{10}^{\mathrm{un}}=\Delta_5^2,\qquad
  \Phi_{10}^{\mathrm{OP}}=D_5^2=4096^{-1}\Delta_5^2 .
\]
Thus $\Delta_5$ carries the primitive BKM value
$\kBKM=5$, while each $\Phi_{10}$ convention records a square of the
primitive denominator and has weight $10$.
\end{proposition}

\section{The four invariants}
\label{sec:invariants}

The symbols $\kch$, $\kcat$, $\kBKM$, and $\kfib$ refer to distinct
constructions. For a compact Calabi-Yau $X$,
\[
  \kch^{\mathrm{comp}}(X)=\sum_q(-1)^q h^{0,q}(X),
  \qquad
  \kcat(X)=\chi(\cO_X).
\]
The BKM invariant is the Borcherds weight
$\kBKM=c_N(0)/2$ in the primitive CHL single-cusp normalisation. The
fibre invariant records the K3 Mukai rank.

\begin{center}
\begin{tabular}{c|c|c|l}
Object & Invariant & Value & Source of the number \\
\hline
$K3$ & $\kch^{\mathrm{comp}}$ & $2$ & $h^{0,0}+h^{0,2}=2$ \\
$K3$ & $\kcat$ & $2$ & $\chi(\cO_{K3})$ \\
$K3$ & $\kfib$ & $24$ & $\operatorname{rk}H^*(K3,\Z)$ \\
$\Delta_5$ & $\kBKM$ & $5$ & $c_1(0)/2=10/2$ \\
$K3\times E$ & $\kch^{\mathrm{comp}}$ & $0$ & $(1-1)(1+1)=0$ \\
$K3\times E$ & $\kcat$ & $0$ & $\chi(\cO_{K3})\chi(\cO_E)=2\cdot0$ \\
$K3\times E$ & $\kch^{\mathrm{Heis}}$ & $3$ & Heisenberg-Mukai specialisation \\
$\mathfrak g_{\Delta_5}$ & $\kBKM$ & $5$ & $\Delta_5$ Borcherds weight \\
$K3\times E$ & $\kfib$ & $24$ & K3 fibre Mukai rank
\end{tabular}
\end{center}

The $d=2$ object $\Db\Coh(K3)$ has an $E_2$-chiral output in the
CY-to-chiral framework. The product $K3\times E$ is a compact
Calabi-Yau threefold; its chiral output belongs to the $d\geq 3$
$E_1$ scope, while $E_2$-structures occur on derived centres or on
specialised two-dimensional factors.

\section{CY-to-chiral interpretation}
\label{sec:cy-to-chiral}

Let
\[
  A_S=\Phimap_2(\Db\Coh(S))
\]
denote the $E_2$-chiral object assigned to a K3 surface by a
CY-to-chiral construction. The published theorem is
Theorem~\ref{thm:eot-mock-modularity} together with
Theorem~\ref{thm:mathieu-module}. A stronger assertion identifying
the mock modular form $H$ with a bar-complex reconstruction of
$A_S$ consists of the following mathematical inputs.

\begin{enumerate}[label=\textup{(\roman*)}]
\item an explicit $E_2$-chiral model for $A_S$ whose current sector
recovers the small $\mathcal N=4$ K3 character theory;
\item an identification of the relevant bar Euler character with the
small $\mathcal N=4$ massive-character generating function;
\item a proof that the non-holomorphic completion induced by the bar
complex has $\xi_{1/2}$-shadow $24\eta^3$;
\item an $M_{24}$-equivariant lift matching the traces
$\operatorname{tr}(g\mid K_n^\natural)$ with the twined bar characters.
\end{enumerate}

These four inputs upgrade the object-level interpretation to a
functorial CY-to-chiral theorem. The present CY-to-chiral paragraph is
a compatibility statement: the $d=2$ target has the
correct dimension, the K3 elliptic genus supplies the canonical
weak Jacobi form, the EOT/GHV/Gannon theorem supplies the mock
modular and Mathieu structures, and the Borcherds lift of the
primitive Jacobi generator supplies the $\Delta_5$ BKM normalisation.
