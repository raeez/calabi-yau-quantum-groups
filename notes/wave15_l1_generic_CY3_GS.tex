\section{Green--Schwarz for 6D hCS on generic compact CY$_3$
(Wave-15 L1/20)}
\label{wn:sec:wave15-l1-generic-cy3-GS}

\subsection{Setting}

The anomaly density of Wave-14 I1 is
\begin{equation}
\label{eq:omega-anom}
\omega_{\mathrm{anom}}
\;=\; \ch_3(F_\cA)\,\Omega_X
\;=\; \tfrac{1}{6}\Tr(F_\cA^3)\,\Omega_X,
\qquad
F_\cA = \bar\partial\cA+\tfrac{1}{2}[\cA,\cA],
\end{equation}
integrated against $\Omega_X$ to land in $H^{3,3}(X) = \CC$. Its
class is
\begin{equation}
\label{eq:kappa-anom-compact}
\kanom(X,\gfrak)
\;=\;
\frac{\hdual}{(4\pi)^3}\int_X \ch_3(T X)\,\Omega_X
\;=\;
\frac{\hdual}{2(4\pi)^3}\int_X c_3(T X)\,\Omega_X
\;=\;
\frac{\hdual\cdot\chitop(X)}{2(4\pi)^3}\,\|\Omega_X\|^2_X,
\end{equation}
using $\ch_3 = \tfrac{1}{6}(c_1^3-3c_1c_2+3c_3) = \tfrac{1}{2}c_3$
on a CY$_3$ ($c_1 = 0$) and $\int_X c_3(T X) = \chitop(X)$
(Gauss--Bonnet). On $\CC^3$ and $K3\times E$ the class
\eqref{eq:kappa-anom-compact} vanishes (Wave-14 I1 Cycle~5). On the
quintic $X_5\subset\mathbb{P}^4$, $\chitop(X_5) = -200$, so $\kanom
\ne 0$. The question is whether a Green--Schwarz-type counterterm
trivialises \eqref{eq:kappa-anom-compact}.

\subsection{Cycle 1 --- What the 6D GS counterterm must be}

\ClaimStatusDefinition\ A Green--Schwarz trivialisation of
$\omega_{\mathrm{anom}}$ is a pair $(B,H)$ with $B \in
\Omega^{2,2}(X)/\mathrm{im}(\bar\partial)$, $H = dB \in
\Omega^{\bullet}(X)$, and a modified Bianchi identity
\begin{equation}
\label{eq:bianchi-6d}
dH \;=\; \ch_3(F_\cA) - \ch_3(T X),
\end{equation}
together with the counterterm
\begin{equation}
\label{eq:SGS}
S_{\mathrm{GS}}
\;=\;
\hbar\int_X B\wedge\Bigl(\ch_2(F_\cA)-\ch_2(T X)\Bigr)\,\Omega_X.
\end{equation}
The Costello--Gwilliam local cohomology class
$[\omega_{\mathrm{anom}}]\in H^1(\mathrm{Loc},d_{\mathrm{BV}})$
admits a primitive exactly when the obstruction $[\ch_3(F_\cA)] -
[\ch_3(T X)]$ is $d$-exact (Witten 1984, \emph{Comm.\ Math.\ Phys.}\
92, 455; Alvarez-Gaum\'e--Witten 1983, \emph{Nucl.\ Phys.}\ B234,
269, for the abelian case). The 6D \emph{descent datum} is the
sequence
\begin{equation}
I_8^{\mathrm{6D}}(\cA) = \ch_3(F_\cA)\cdot\Omega_X \xrightarrow{d}
I_7^{(1)}(\cA,v)
\xrightarrow{d} I_6^{(2)}(\cA,v_1,v_2),
\end{equation}
the holomorphic-twist analogue of the 10D Bardeen--Zumino descent
(Stora 1984).

The critical structural fact is that $dH = \ch_3(F_\cA) -
\ch_3(T X)$ demands $[\ch_3(F_\cA)] = [\ch_3(T X)]$ \emph{in
$H^6(X,\QQ)$}. This is an integrality / cohomological constraint,
not a local field-theoretic one.

\subsection{Cycle 2 --- Comparison with 10D type-I Green--Schwarz}

\ClaimStatusHeuristic\ The 10D type-I / heterotic anomaly polynomial
factorises (Witten 1984; Green--Schwarz 1984) as
\begin{equation}
\label{eq:I12}
I_{12} \;=\; \bigl(\mathrm{tr}R^2 - \tfrac{1}{30}\Trfund F^2\bigr)
\wedge X_8(\cA,R),
\end{equation}
where the factorisation \emph{forces} the gauge group to be
$E_8\times E_8$ or $\mathrm{SO}(32)$ (dual Coxeter numbers
$30$ each), and the coefficient $\tfrac{1}{30}$ is $1/\hdual$ for
$E_8$. The GS counterterm is
$S_{\mathrm{GS}}^{\mathrm{10D}} = \int B_2\wedge X_8$ with $dH_3 =
\tfrac{1}{30}\Trfund F^2 - \mathrm{tr}R^2$.

For 6D $\hCS$ on CY$_3$ the analogue of \eqref{eq:I12} is the
bilinear decomposition of $\omega_{\mathrm{anom}}$ on a Hodge-wedge
basis:
\begin{equation}
\label{eq:I8-6d}
\ch_3(F_\cA)
\;=\; \tfrac{1}{2!}\ch_2(F_\cA)\cdot \tfrac{1}{1!}\ch_1(F_\cA)
\;+\; \text{(cubic adjoint trace residue)},
\end{equation}
which on a simple $\gfrak$ with $\ch_1(F_\cA) = 0$ reduces to the
second term: a pure cubic Casimir $d^{abc}F_aF_bF_c$. For
$\gfrak = \mathrm{SU}(N)$, $N\ge 3$, this cubic is nonzero; for
$\gfrak = \mathrm{SU}(2), \mathrm{SO}(N), E_6, E_7, E_8, F_4, G_2$
(all $d^{abc} = 0$ simple Lie algebras) the cubic residue
vanishes identically and \eqref{eq:kappa-anom-compact} is
\emph{zero for every CY$_3$}. This is the 6D $\hCS$ analogue of
the Witten 1984 group-theoretic selection rule.

\subsection{Cycle 3 --- $\int_X c_3(T X)\cdot\ch_3(\gfrak)$: which
$\gfrak$ is admissible on which $X$?}

\ClaimStatusTheorem\ \emph{Group-theoretic selection.} Let $X$ be
compact CY$_3$ with $\chitop(X) \ne 0$, and let $\gfrak$ be simple.
Then $\kanom(X,\gfrak) = 0$ (with no GS counterterm needed) iff
$d^{abc}(\gfrak) = 0$, i.e.\ $\gfrak \in
\{\mathrm{SU}(2), \mathrm{SO}(N), E_6, E_7, E_8, F_4, G_2\}$. For
$\gfrak = \mathrm{SU}(N)$, $N\ge 3$, $\kanom(X,\mathrm{SU}(N))\propto
A(\mathrm{SU}(N))\cdot\chitop(X)$ with $A(\mathrm{SU}(N)) = N$ the
cubic-Casimir anomaly coefficient (Frampton--Kephart 1983;
Okubo 1982; Georgi 1999 \S10.6). On the quintic $\chitop = -200$,
so $\kanom(X_5,\mathrm{SU}(N)) = -200\,N\,\hdual/(2(4\pi)^3) \ne 0$
for every $N\ge 3$.

\ClaimStatusConjectured\ \emph{CY-$\gfrak$ admissibility pairing.}
The admissibility of $\gfrak$ on CY$_3$ $X$ at the 6D $\hCS$ level is
captured by the pairing
\begin{equation}
\label{eq:admissibility-pairing}
\langle X,\gfrak\rangle_{\mathrm{6D}}
\;=\;
A(\gfrak)\cdot\chitop(X)/24,
\end{equation}
which is an integer when $\chitop(X)\equiv 0\pmod{24/\gcd(24,A)}$;
the $24$ is the universal K3 periodicity (Noether--Lefschetz;
Dolgachev 1996). On $K3\times E$, $\chitop = 0$ and every
$\gfrak$ is admissible; on the quintic the admissibility pairing
lands in $\QQ\setminus\ZZ$ for $\gfrak = \mathrm{SU}(N)$, $N\ge 3$,
and no GS counterterm is available for those $\gfrak$.

\subsection{Cycle 4 --- Heterotic 10D on $X$: $\ch_2(E) =
\ch_2(T X)$, and its 6D hCS analogue}

\ClaimStatusTheorem\ \emph{Heterotic constraint (Candelas--Horowitz--
Strominger--Witten 1985).} For heterotic 10D compactified on a CY$_3$
$X$ with a holomorphic vector bundle $E\to X$, anomaly cancellation
demands
\begin{equation}
\label{eq:CHSW}
\ch_2(E) \;=\; \ch_2(T X) \;\in\; H^4(X,\ZZ).
\end{equation}
The standard embedding $E = T X$ saturates \eqref{eq:CHSW}
tautologically and gives an $E_8\times E_6$ low-energy
gauge group (the $E_8$-factor is broken to $E_6$ by the $T X$
holonomy $\mathrm{SU}(3)\subset E_8$).

\ClaimStatusConjectured\ \emph{6D $\hCS$ analogue.} The 6D $\hCS$
theory on CY$_3$ $X$ with structure group $\gfrak$ admits a quantum
factorisation algebra (i.e.\ $\Phi(A_X,\gfrak)$ exists) iff the
$\hCS$ analogue of \eqref{eq:CHSW} holds at the level of
$\ch_3$:
\begin{equation}
\label{eq:hCS-CHSW}
\ch_3(F_\cA) \;=\; \ch_3(T X) \;\in\; H^6(X,\QQ),
\qquad
\text{together with } \ch_2(F_\cA) = \ch_2(T X).
\end{equation}
This is the \emph{6D holomorphic analogue of the CHSW standard
embedding}. The standard solution is $F_\cA = R$ with
$\gfrak = \mathrm{su}(3)$ embedded via the $T X$ holonomy ---
i.e., the $\hCS$ theory on $X$ is forced into the tangent-bundle
gauge sector. For the quintic, this selects $\gfrak\subset
\mathrm{su}(3)$ as the admissible structure group; for $K3\times E$,
any $\gfrak$ is admissible since both sides of \eqref{eq:hCS-CHSW}
vanish.

The $\ch_2$-side of \eqref{eq:hCS-CHSW} is the 6D shadow of the
Costello--Li BCOV anomaly-cancellation condition on CY$_3$
(Costello--Li 2015, \texttt{arXiv:1201.4522}, \S2--3): BCOV
holomorphic anomaly closes quantum-mechanically on CY$_3$ exactly
when $\ch_2(E) - \ch_2(T X) = 0$, which is the chain-level origin
of the cache entry $\kch(\mathrm{quintic}) = \chitop/24 = -25/3$ via
the BCOV constant-map formula.

\subsection{Cycle 5 --- Consequence for $\Phi$ on generic CY$_3$}

\ClaimStatusTheorem\ \emph{(Obstruction dichotomy.)} Let $X$ be
compact CY$_3$. The functor $\Phi_3: \mathrm{CY}_3\text{-cat}\to
\mathrm{ChirAlg}^{E_1}$ on $A_X$ with structure Lie-algebra
$\gfrak$:
\begin{enumerate}
\item is \emph{unobstructed} at one loop iff
$\kanom(X,\gfrak) = 0$, i.e.\ either $\chitop(X) = 0$ (e.g.\
$K3\times E$, abelian $T^6$, half-K3 products) or $A(\gfrak) = 0$
(e.g.\ $\gfrak \in \{E_8, E_7, E_6, F_4, G_2, \mathrm{so}(N),
\mathrm{su}(2)\}$);
\item admits a Green--Schwarz trivialisation iff
\eqref{eq:hCS-CHSW} holds, i.e.\ $F_\cA$ is cohomologous to the
tangent-bundle curvature $R$ in $H^{\le 6}(X,\QQ)$.
\end{enumerate}

\ClaimStatusConjectured\ \emph{(Scope consequence.)} For the compact
landscape:
\begin{center}
\begin{tabular}{lcccc}
$X$ & $\chitop(X)$ & admissible $\gfrak$ & GS needed? &
$\kBKM$ \\\hline
$K3\times E$ & $0$ & all simple & no & $5$\\
$T^6$ & $0$ & all simple & no & ---\\
abelian / bielliptic & $0$ & all simple & no & ---\\
half-K3 $\times$ half-K3 & $0$ & all simple & no & ---\\
quintic $X_5$ & $-200$ & $A(\gfrak) = 0$ only & yes,
CHSW-embedded & N/A \\
generic CICY & $\ne 0$ & $A(\gfrak) = 0$ only & yes,
CHSW-embedded & N/A \\
\end{tabular}
\end{center}

The programme's BKM sector (K3 $\times$ E, abelian surfaces,
half-K3) is precisely the $\chitop = 0$ locus where $\Phi_3$ is
unobstructed for \emph{every} simple $\gfrak$. The Gritsenko
$\Delta_5$ paramodular anomaly cancellation (Vol III
\texttt{k3e\_cy3\_programme.tex} L3669) lives here: on $K3\times E$
the 6D BV obstruction vanishes identically by \eqref{eq:kappa-anom-compact}
with $c_3 = 0$, and the $\kBKM = 5$ Gritsenko weight is
\emph{not} obstructed by Green--Schwarz.

For the generic CY$_3$ (quintic, CICYs, Schoen, Tian--Yau) the
programme restricts either to the $A(\gfrak) = 0$ structure-algebra
family (the anomaly-free simple Lie algebras) \emph{or} to the
CHSW-embedded gauge bundles $F_\cA\sim R$, where the 6D $\hCS$
theory reduces at the Lagrangian level to the holomorphic twist of
heterotic on $X$. Either route leaves $\Phi_3$ defined on the
quintic, but at a higher cost: the BCOV holomorphic-anomaly tower
is nontrivial (Vol III Theorem~\ref{thm:hae-mc-structural}), and
the $\kch = \chitop/24 = -25/3$ value is the $\Eone$-shadow
fingerprint of the Costello--Li CY$_3$ BV datum, not a
$\kBKM$-witness.

\subsection{Summary and handoff}

\ClaimStatusTheorem\ (Wave-15 L1 theorem.) The one-loop BV anomaly
$\kanom(X,\gfrak)$ of 6D $\hCS$ on compact CY$_3$ $X$ with simple
structure algebra $\gfrak$ factors as
\[
\kanom(X,\gfrak)
\;=\;
\frac{\hdual\cdot A(\gfrak)\cdot\chitop(X)}{2(4\pi)^3}
\cdot\|\Omega_X\|^2_X,
\]
with $A(\gfrak)$ the cubic-Casimir coefficient. The anomaly
vanishes iff $A(\gfrak) = 0$ or $\chitop(X) = 0$. A
Green--Schwarz trivialisation exists iff the CHSW-type condition
$[\ch_3(F_\cA)] = [\ch_3(T X)]$ holds, which selects the
tangent-bundle-embedded gauge sector (standard 6D holomorphic
analogue of Candelas--Horowitz--Strominger--Witten). The CY-to-chiral
functor $\Phi_3$ is unobstructed on the $\chitop = 0$ locus
($K3\times E$, abelian, half-K3 products), which is exactly the
BKM / Gritsenko $\Delta_5$ sector of the programme; on the
$\chitop \ne 0$ locus (quintic, CICYs) $\Phi_3$ is defined only
through the CHSW-embedded GS trivialisation or by restricting
$\gfrak$ to the cubic-Casimir-free simple algebras.
