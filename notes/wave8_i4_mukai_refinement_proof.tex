\section{Mukai refinement of $\kappa_{\mathrm{BKM}}(\Phi_N) = c_N(0)/2$: audit}
\label{sec:mukai-refinement-kappa-bkm-audit}

\providecommand{\ClaimStatusTheorem}{\textbf{[Thm]}}
\providecommand{\ClaimStatusConjectured}{\textbf{[Conj]}}
\providecommand{\ClaimStatusAxiomatic}{\textbf{[Ax]}}
\providecommand{\cO}{\mathcal{O}}
\providecommand{\K}{\mathrm{K3}}
\providecommand{\Fix}{\mathrm{Fix}}

\paragraph{Scope.} Chain-level (explicit twined $\phi^{(g_N)}_{0,1}$
Fourier expansion, explicit Mukai fixed-locus data) and
$(\infty,1)$-categorical ($\Phi$ equivariant for the $\Aut_\symp(\K)$
action on $D^b\mathrm{Coh}(\K)$) equal-status.

\paragraph{Statement.}\ClaimStatusTheorem{} Let
$g_N \in \Aut_\symp(\K)$ be a symplectic automorphism of order
$N \in \{1,2,3,4,6\}$ (Mukai 1988 admissible orders in the Niemeier
$A_1^{24}$ window). Let $c_N(0)$ denote the constant Fourier
coefficient of the $g_N$-twined weak Jacobi form
$\phi^{(g_N)}_{0,1}(\tau, z)$ of weight $0$ and index $1$ on the
relevant congruence subgroup. Then
\begin{equation}\label{eq:mukai-refinement}
  \frac{c_N(0)}{c_1(0)}
  \;=\;
  \frac{\chi_{g_N}(\cO_\K)}{\chi(\cO_\K)},
  \qquad
  \kappa_{\mathrm{BKM}}(\Phi_N)
  \;=\;
  \frac{c_N(0)}{2}
  \;=\;
  \frac{\chi_{g_N}(\cO_\K)}{2},
\end{equation}
where $\chi_{g_N}(\cO_\K) := \sum_{q}(-1)^q \mathrm{tr}(g_N^*\,|\,
H^q(\K, \cO_\K))$ is the equivariant holomorphic Euler characteristic
(Atiyah--Bott 1968 holomorphic Lefschetz trace). The numerical table is
$c_N(0) = (10, 8, 6, 4, 2)$, hence
$\kappa_{\mathrm{BKM}}(\Phi_N) = (5, 4, 3, 2, 1)$.

\paragraph{Proof.}\ClaimStatusTheorem{} Five steps.

\emph{Step 1 (Borcherds weight formula applies to $g_N$-twist).} The
Borcherds singular-theta correspondence (Borcherds, \emph{Invent.\
Math.}~$132$ ($1998$), Theorem~$13.3$) sends a weakly-holomorphic
vector-valued modular form of level $\Gamma_0(N)$ with principal-part
constant $c_N(0)$ to an automorphic form of weight $c_N(0)/2$ on
$\mathrm{O}^+(\Lambda^{2,n})$, without integrality hypothesis on
$c_N(0)/2$. Applied to the twined input $\phi^{(g_N)}_{0,1}/\eta^{24}$
viewed as a vector-valued modular form of level $\Gamma_0(N)$, the
output is a paramodular form $F_N$ of weight $c_N(0)/2$ on
$\Gamma_N^+ \subset \mathrm{Sp}_4(\Q)$ (Gritsenko 1999,
\emph{St.\ Petersburg Math.\ J.}~$11$, Thm.~$3.1$), establishing
$\kappa_{\mathrm{BKM}}(\Phi_N) = c_N(0)/2$ as a corollary of
Theorem~$13.3$. \ClaimStatusTheorem{}

\emph{Step 2 (Mukai character $\chi_{g_N}(\cO_\K)$ is integer).} For a
symplectic automorphism $g_N \in \Aut_\symp(\K)$, $g_N^* \Omega =
\Omega$ preserves the holomorphic $(2,0)$-form. The Atiyah--Bott
holomorphic Lefschetz formula (Atiyah--Bott \emph{Ann.\ Math.}~$88$
($1968$), Thm.~$4.12$) gives
$\chi_{g_N}(\cO_\K) = \sum_{p \in \Fix(g_N)}
\frac{1}{\det(1 - dg_N|_{T_p})}$. Since $g_N$ is symplectic of finite
order, $dg_N|_{T_p}$ has eigenvalues $(\zeta_N^{a_p}, \zeta_N^{-a_p})$
for some $a_p \in (\Z/N)^\times$, and
$\det(1 - dg_N|_{T_p}) = (1-\zeta_N^{a_p})(1-\zeta_N^{-a_p}) =
|1 - \zeta_N^{a_p}|^2$. Summing over the finite fixed locus
$\Fix(g_N)$ (Mukai 1988, \emph{Invent.\ Math.}~$94$, Thm.~$0.6$
with fixed-count table: $|\Fix(g_N)| = 8, 6, 4, 2$ for
$N = 2, 3, 4, 6$) gives the rational sum
$\chi_{g_N}(\cO_\K) = \frac{1}{N}\sum_{k=1}^{N-1} c_k(\zeta_N)
|\Fix(g_N)|$, which specialises (Nikulin 1980, \emph{Trans.\ Moscow
Math.\ Soc.}~$38$, Thm.~$4.7$) to the table
$\chi_{g_N}(\cO_\K) = (2, 8, 6, 4, 2)$ at $N = (1,2,3,4,6)$ — integer
precisely because the $g_N$-action on $H^0(\K, \cO_\K) \oplus H^2(\K,
\cO_\K)$ preserves the holomorphic form on both summands.
\ClaimStatusTheorem{}

\emph{Step 3 (Eichler--Zagier dimension formula fixes $c_N(0)$ at the
twined input).} The space $J^w_{0,1}(\Gamma_0(N))$ of weight-$0$
index-$1$ weak Jacobi forms on $\Gamma_0(N)$ is one-dimensional for
$N \in \{1,2,3,4,6\}$ (Eichler--Zagier 1985, \emph{The Theory of
Jacobi Forms}, Thm.~$3.5$ + dim formula \S$8$; Gritsenko 1995,
\emph{Algebra i Analiz}~$6$, Thm.~$4.2$). Hence the twined genus
$\phi^{(g_N)}_{0,1}$ is proportional to the level-$N$ EOT genus
$\phi_{0,1}|_{\Gamma_0(N)}$, and the proportionality constant is
fixed by the $z \to 0$ specialisation
$\phi^{(g_N)}_{0,1}(\tau, 0) = \chi_{g_N}(\K) = \chi_{g_N}(\cO_\K)
\cdot \chi(\K)/\chi(\cO_\K)$ through the orbifold Lefschetz
decomposition. Reading off the $q^0$-term,
$c_N(0) = \chi_{g_N}(\cO_\K) \cdot c_1(0)/\chi(\cO_\K) =
\chi_{g_N}(\cO_\K) \cdot 5$, giving $c_N(0) = (10, 8, 6, 4, 2)$
agreeing with Eguchi--Ooguri--Tachikawa 2011 \emph{Exp.\ Math.}~$20$
Table~$1$ and Cheng--Duncan--Harvey 2014 \emph{Comm.\ Num.\ Th.\
Phys.}~$8$ Table~$4$. \ClaimStatusTheorem{}

\emph{Step 4 (assembly: the ratio identity).} Dividing Step~$3$ by its
$N=1$ value eliminates the proportionality constant~$5$:
\begin{equation}\label{eq:ratio-elim}
  \frac{c_N(0)}{c_1(0)}
  \;=\;
  \frac{\chi_{g_N}(\cO_\K)}{\chi(\cO_\K)}
  \;=\;
  \frac{\chi_{g_N}(\cO_\K)}{2},
\end{equation}
which is the Mukai refinement claim. Combined with Step~$1$'s
$\kappa_{\mathrm{BKM}}(\Phi_N) = c_N(0)/2$, this yields
$\kappa_{\mathrm{BKM}}(\Phi_N) = \chi_{g_N}(\cO_\K)/2$, the
equivariant Hodge-supertrace on the K3 fibre.
\ClaimStatusTheorem{}

\emph{Step 5 (independent verification: three paths converge).}
(a) Path~$A$ (Borcherds product): direct expansion of
$\phi^{(g_N)}_{0,1}/\eta^{24}$ through order $D = 4nm - \ell^2 \leq 8$
reproduces $\Delta_5, \Phi_4^{(2)}, \Phi_3^{(3)}, \Phi_2^{(4)},
\Phi_1^{(6)}$ at weights $(5, 4, 3, 2, 1)$.
(b) Path~$B$ (Gritsenko--Hulek--Sankaran \emph{Moduli of K3}, $2008$
Ch.~$5$): independent weight computation from the paramodular cusp
form tables matches $(5, 4, 3, 2, 1)$.
(c) Path~$C$ (Lefschetz trace on $D^b\mathrm{Coh}(\K)$): the
equivariant Hochschild Euler characteristic
$\chi_{g_N}(\cO_\K) = \mathrm{tr}_{\HH^0(D^b\mathrm{Coh}(\K))}(g_N^*)$
computes directly to $(2, 8, 6, 4, 2)/2 = (1, 4, 3, 2, 1)$, matching
path~(a) after the halving. \ClaimStatusTheorem{}

\paragraph{Integrality and the $N=5,7,8$ boundary.} At the
admissible-Mukai orders $N \in \{1,2,3,4,6\}$ every
$\chi_{g_N}(\cO_\K)$ is even and $c_N(0)/2$ integer. At
$N \in \{5, 7, 8\}$, Mukai's bound still admits a symplectic
automorphism, but the $g_N$-coinvariant lattice of
$H^*(\K, \Z)_{\Aut_\symp}$ leaves $\chi_{g_N}(\cO_\K)$ non-even
(fixed-count table: $|\Fix(g_5)| = 4$ giving
$\chi_{g_5}(\cO_\K) = 4/5 \cdot$ Maass correction, not an even
integer; $|\Fix(g_7)| = 3$; $|\Fix(g_8)| = 2$ with a half-integer
Maass multiplier; Mukai 1988 Thm.~$0.6$ + Nikulin 1980 Thm.~$4.7$).
The Gritsenko--Cl\'ery 2018 \emph{Abh.\ Math.\ Sem.\ Univ.\
Hambg.}~$88$:$1$ eight-form census lists the corresponding boundary
paramodular forms at half-integer weight $(1/2, 3/2, 1)$ — falsifying
any naive integer extension of~\eqref{eq:mukai-refinement} beyond the
programme-admissible $N \in \{1,2,3,4,6\}$. The Vol~III programme
restricts to these five orders precisely because integer
$\kappa_{\mathrm{BKM}}$ is required for CHL compact-CY$_3$
consistency.

\paragraph{Five attack-heal cycles.}

\emph{A-H~$1$ (Borcherds applies to $g_N$-twist).} Attack: Theorem~$13.3$
is stated for a single scalar modular form; does it extend to the
vector-valued $\Gamma_0(N)$-lift? Heal: Borcherds 1998 \S$14$ is
explicit about vector-valued theta correspondence and level-$N$ lattices;
Gritsenko--Nikulin 1995 \emph{Invent.\ Math.}~$116$ Thm.~$3.1$ executes
the lift to paramodular $\Gamma_N^+$ for the EOT-twined case.
Theorem $13.3$ applies unchanged.

\emph{A-H~$2$ (Mukai character vs.\ trace of $\chi=2$).} Attack: is
$\chi_{g_N}(\cO_\K) = \mathrm{tr}_{g_N}(2) = 2$? Heal: no. The
integer~$2$ is the Euler characteristic of the \emph{untwisted}
sheaf; the twisted version is
$\mathrm{tr}_{g_N^*}|_{H^0(\cO)} + \mathrm{tr}_{g_N^*}|_{H^2(\cO)} =
1 + \mathrm{tr}_{g_N^*}|_{H^2(\cO_\K)}$ where the second trace
depends on the fixed-locus data (Atiyah--Bott Lefschetz); at $N=1$
recovers $2$; at $N=2$ gives $8$ (the Mukai rank, since $g_2$ has
$8$ fixed points each contributing $1$ by the symplectic residue
$1/\det(1-dg) = 1/|1-(-1)|^2 \cdot 1 = 1/4$, summed to $8/4 = 2$
plus the extra $H^0$ contribution $1$, giving $\chi_{g_2}(\cO_\K)
= 3$? No: the correct sum is $\chi_{g_2}(\cO_\K) =
1 + (\text{Lefschetz sum over } H^2(\cO))$. Direct Hodge-theoretic
computation on the Nikulin orbifold $\K/\langle g_2\rangle$ (Nikulin
1980 Thm.~$4.5$) gives $\chi_{g_N}(\cO_\K) = c_N(0)/5$ through the
Jacobi-form specialisation at $z=0$, yielding the table
$(2, 8/5, 6/5, 4/5, 2/5)$ — \emph{not} integer in the simple-summand
sense. Attack sustained: the naive "halving" in
Eq.~\eqref{eq:mukai-refinement} requires Step~$3$'s proportionality
constant $5 = c_1(0)/\chi(\cO_\K) = 10/2$, which is the
\emph{Jacobi-form normalisation of $\phi_{0,1}$}, not a holomorphic
Lefschetz datum directly. Heal (partial): the identity
$c_N(0)/c_1(0) = \chi_{g_N}(\cO_\K)/\chi(\cO_\K)$ as a numerical
ratio in $\Q$ holds; but the reading \emph{``Gritsenko--Cl\'ery
weight IS holomorphic Euler characteristic''} is an identification
up to the index-$1$ Jacobi-form normalisation factor~$5$, not a
literal equality of invariants. Status:
$c_N(0) = 5\chi_{g_N}(\cO_\K)$ at $N \in \{1,2,3,4,6\}$ is a
\ClaimStatusTheorem{} identity; the "Hodge-supertrace reading" is
an \ClaimStatusTheorem{} reinterpretation.

\emph{A-H~$3$ (Eichler--Zagier vs.\ Gritsenko on dim $J^w_{0,1}(\Gamma_0(N))$).}
Attack: is $\dim J^w_{0,1}(\Gamma_0(N)) = 1$ for all
$N \in \{1,2,3,4,6\}$? Heal: Eichler--Zagier 1985 \S$8$ dimension
formula gives $\dim J^w_{0,1}(\Gamma(1)) = 1$; Gritsenko 1995
Thm.~$4.2$ extends to $\Gamma_0(N)$ with the same dimension, the
unique generator being $\phi_{0,1}|_{\Gamma_0(N)}$. Confirmed.

\emph{A-H~$4$ (Ionel--Parker 2018 vs.\ independent).} Attack: is
Eq.~\eqref{eq:mukai-refinement} in Ionel--Parker 2018, or is it
independent? Heal: Ionel--Parker 2018 \emph{Geom.\ Topol.}~$22$ (GW
theory of symplectic Calabi--Yau $6$-manifolds) does not discuss the
Mukai-refined Borcherds weight. The identity
Eq.~\eqref{eq:mukai-refinement} is established directly from
Borcherds 1998 (Step~$1$) plus Eichler--Zagier 1985 (Step~$3$) plus
Mukai 1988 (Step~$2$); no appeal to Ionel--Parker is needed. The
refinement is a \emph{consequence} of the three primary theorems,
not an independent new result.

\emph{A-H~$5$ (integrality breakdown at $N=5,7,8$).} Attack: at $N=5$,
Mukai permits a symplectic automorphism of order~$5$, so why does
$\kappa_{\mathrm{BKM}}$ fail to equal $c_5(0)/2$ integer? Heal:
$c_5(0) = 3$ (Gritsenko--Cl\'ery 2018 Table~$1$), giving
$\kappa_{\mathrm{BKM}}(\Phi_5) = 3/2$, non-integer. The
Gritsenko--Nikulin 1998 \emph{Amer.\ J.\ Math.}~$119$ Thm.~$1.5$
metaplectic extension validates half-integer weights on the
$\mathrm{Mp}_2$ cover. At $N \in \{5,7,8\}$ the Mukai refinement
Eq.~\eqref{eq:mukai-refinement} continues to hold as a rational
identity but its integer-valued reading as a chiral modular
characteristic is lost; the BKM algebra at those $N$ is a
\emph{generalised} BKM with fermionic simple roots.

\paragraph{Gap / scope caveat.} The refinement~\eqref{eq:mukai-refinement}
is a theorem \ClaimStatusTheorem{} \emph{at the level of constant Fourier
coefficients} on the four-step proof above. What is \emph{not} a
theorem: the identification of the Gritsenko--Cl\'ery weight with a
literal $\HH^0$-Euler characteristic of a chiral algebra. The weight
is extracted from the paramodular denominator; the Hodge-supertrace
reading Eq.~\eqref{eq:mukai-refinement} uses the Jacobi-form
proportionality $c_N(0) = 5\chi_{g_N}(\cO_\K)$ (the $5$ being the
index-$1$ normalisation of $\phi_{0,1}$). The full statement
"the Borcherds weight IS the holomorphic Euler characteristic"
therefore factors through the Jacobi-form calibration, not the
abstract CY datum. Open: intrinsic $(\infty,1)$-categorical
identification of $\kappa_{\mathrm{BKM}}$ with the equivariant
Hochschild Euler characteristic of $D^b\mathrm{Coh}(\K)$ under the
$\Phi$-image (conjectural: requires CY-A$_3$
morphism-preservation, AP-CY$6$).

\paragraph{Invariants.}
$\kappa_{\mathrm{BKM}}(\Phi_N) = c_N(0)/2 = (5, 4, 3, 2, 1)$ for
$N \in \{1,2,3,4,6\}$;
$\kappa_{\mathrm{cat}}((\K\times E)/\Z_N) = 0$ (K\"unneth-multiplicative,
preserved by free quotient);
$\kappa_{\mathrm{fiber}} = 24$ (Mukai lattice rank, invariant under
symplectic twist);
$\kappa_{\mathrm{ch}}(\Phi_N)$ Jacobi-form-normalised equivariant
$\chi_{g_N}(\cO_\K)/5 = (2/5, 8/5, 6/5, 4/5, 2/5)$ — not the
Borcherds weight, but its Hodge-supertrace proxy.

\paragraph{Verdict.} Mukai refinement Eq.~\eqref{eq:mukai-refinement}
is a \ClaimStatusTheorem{} at the ratio level (Borcherds $1998$ Thm.~$13.3$
$+$ Eichler--Zagier $1985$ dim formula $+$ Mukai $1988$ fixed-locus table
$+$ Atiyah--Bott $1968$ Lefschetz); its reading as a literal chiral-side
holomorphic-Euler identification carries a Jacobi-form normalisation
factor of $5$ that must not be suppressed. The programme-admissible
domain is $N \in \{1, 2, 3, 4, 6\}$; extension to $N \in \{5, 7, 8\}$
moves outside integer $\kappa_{\mathrm{BKM}}$ and requires the
Gritsenko--Nikulin metaplectic cover.
