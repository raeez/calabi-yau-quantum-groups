\title{Exact Rational Correction for the Zamolodchikov Tetrahedron Equation}
\author{Raeez Lorgat}

\begin{abstract}
The Yang \(R\)-matrix
\[
 R(z)=\frac{z\Id+\hbarY P}{z+\hbarY}
\]
solves the Yang-Baxter equation, but the factored face operator
\[
 S^{\mathrm{fact}}_{ijk}=R_{ij}(u_i-u_j)R_{ik}(u_i-u_k)R_{jk}(u_j-u_k)
\]
does not solve the Zamolodchikov tetrahedron equation on
\(V^{\otimes 4}\).  For \(V=\C^2\), \(u=(0,1,3,7)\), and
\(\hbarY=1/5\), the charge-\(2\) obstruction is killed to first
linearized order by a per-face ternary correction.  The resulting
minimum-norm rational representative is a \(6\times 6\) matrix with
ten exact rational values, symmetric, persymmetric, full rank, and
zero anti-diagonal.  This is a finite rational computation.  It is not
a BKM, \(K3\times E\), Maulik-Okounkov, or all-order physical
identification.
\end{abstract}

\section{Finite Yang Model}
\label{sec:finite-yang-model}

Let \(V=\C|0\rangle\oplus\C|1\rangle\), and let
\(P(x\otimes y)=y\otimes x\).  For spectral parameters
\[
 u=(u_0,u_1,u_2,u_3)=(0,1,3,7),
 \qquad \hbarY=\frac{1}{5},
\]
set
\[
 R_{ij}=R(u_i-u_j),\qquad
 R(z)=\frac{z\Id+\hbarY P}{z+\hbarY}.
\]
The face operator on the ordered face \(i<j<k\) is
\[
 S^{\mathrm{fact}}_{ijk}=R_{ij}R_{ik}R_{jk}.
\]
The Zamolodchikov tetrahedron residual is
\[
 \cO =
 S^{\mathrm{fact}}_{012}S^{\mathrm{fact}}_{013}
 S^{\mathrm{fact}}_{023}S^{\mathrm{fact}}_{123}
 -
 S^{\mathrm{fact}}_{123}S^{\mathrm{fact}}_{023}
 S^{\mathrm{fact}}_{013}S^{\mathrm{fact}}_{012}
 \in \End(V^{\otimes4}).
\]
The charge-\(2\) basis is
\[
\mathcal B_2=
(|0011\rangle,|0101\rangle,|0110\rangle,
 |1001\rangle,|1010\rangle,|1100\rangle).
\]

\begin{proposition}[Charge-\(2\) obstruction]
\label{prop:charge-two-obstruction}
On \(V^{\otimes4}_{q=2}\), the residual
\(\cO_2=\cO|_{V^{\otimes4}_{q=2}}\) is nonzero, antisymmetric, and
persymmetric:
\[
 \cO_2^T=-\cO_2,\qquad J\cO_2J=\cO_2,
\]
where \(J\) is the anti-diagonal identity matrix in the basis
\(\mathcal B_2\).  Its rank is \(4\), its diagonal is zero, and
\[
 \|\cO_2\|_\infty=\frac{2100043585}{235846972032}.
\]
\end{proposition}

\begin{proof}
Substitution of the displayed \(R\)-matrix into the four face
operators gives sixteen by sixteen rational matrices.  Projection to
the six charge-\(2\) basis states gives \(\cO_2\).  Direct rational
matrix arithmetic gives a nonzero matrix, \(\cO_2^T=-\cO_2\),
\(J\cO_2J=\cO_2\), \(\rk(\cO_2)=4\), and the stated maximum entry.
\end{proof}

\section{Linearized Ternary Correction}
\label{sec:linearized-correction}

Let
\[
 C_{ijk}\in \End(V_i\otimes V_j\otimes V_k)^{\mathrm{cp}}
\]
be a charge-preserving ternary correction on a face.  Since
\[
\dim\End(V^{\otimes3})^{\mathrm{cp}}
=\binom30^2+\binom31^2+\binom32^2+\binom33^2
=20,
\]
the four faces give \(80\) rational unknowns.  The linearized
tetrahedron equation on \(V^{\otimes4}_{q=2}\) is
\[
 A x=-\operatorname{vec}(\cO_2),
 \qquad A\in \operatorname{Mat}_{36\times80}(\Q).
\]

\begin{proposition}[Exact linear algebra]
\label{prop:exact-linear-algebra}
The matrix \(A\) has rank \(35\).  The system
\[
 A x=-\operatorname{vec}(\cO_2)
\]
is consistent.  Hence the \(O(\hbarY^2)\) obstruction is exact in the
gauge-extended ternary complex.  The solution is not unique; the
rank defect is gauge freedom.  The representative below is the
minimum-norm rational solution obtained by exact row-space projection.
\end{proposition}

\begin{proof}
Each column of \(A\) is the charge-\(2\) flattening of
\[
 S_{012}\cdots C_f\cdots S_{123}
 -
 S_{123}\cdots C_f\cdots S_{012},
\]
with \(C_f\) inserted in the corresponding face position.  All entries
lie in \(\Q\).  Exact row reduction gives \(\rk(A)=35\), and direct
multiplication gives \(A x=-\operatorname{vec}(\cO_2)\) for the
displayed representative.
\end{proof}

\begin{theorem}[Exact rational \(T\)-matrix]
\label{thm:exact-t-matrix}
The matrix \(T_2\) has exactly ten distinct rational entries.  It is
symmetric, persymmetric, has zero anti-diagonal, and has full rank:
\[
 T_2^T=T_2,\qquad JT_2J=T_2,\qquad (T_2)_{i,7-i}=0,\qquad
 \rk(T_2)=6.
\]
The corrected face operators
\[
 S^{\mathrm{corr}}_{ijk}=S^{\mathrm{fact}}_{ijk}+C_{ijk}
\]
kill the linearized \(O(\hbarY^2)\) obstruction exactly.  The nonlinear
ZTE residual is nonzero and of higher order; in the canonical
normalisation its maximum entry is strictly smaller than
\(\|\cO_2\|_\infty\).
\end{theorem}

\begin{proof}
The displayed matrix has the symmetry \(N^T=N\) and \(JNJ=N\) by
inspection.  Its anti-diagonal is zero by inspection.  Exact rational
rank computation gives \(\rk(D^{-1}N)=6\).  The linearized equation is
Proposition~\ref{prop:exact-linear-algebra}; substituting
\(S^{\mathrm{fact}}_{ijk}+C_{ijk}\) into the nonlinear tetrahedron
product leaves a nonzero residual, with exact rational maximum smaller
than the maximum entry of \(\cO_2\).  Thus the displayed \(T_2\) is an
exact first-correction representative, not an all-order tetrahedron
solution.
\end{proof}

\section{Scope}
\label{sec:scope}

The theorem above is a rational finite-dimensional statement.  It
does not assert a configuration-space Feynman integral evaluation.
It does not identify \(T_2\) with a Maulik-Okounkov stable-envelope
discrepancy.  It does not produce a BKM denominator, a Borcherds lift,
or a \(K3\times E\) automorphic class.  It does not prove an all-order
solution of the Zamolodchikov tetrahedron equation on every charge
sector.

The physical reading is heuristic until a comparison theorem supplies
the map from \(6\)-dimensional holomorphic Chern-Simons observables to
the finite rational Yang model.  The BKM and \(K3\times E\) readings
require independent compactification and automorphic comparison data.

\begin{problem}[Comparison with \(6\)-dimensional hCS]
Construct a chain-level comparison from the \(6\)-dimensional
holomorphic Chern-Simons factorization observables on \(\C^3\) to the
finite Yang model above, preserving the face operators and the
linearized tetrahedron obstruction.
\end{problem}

\begin{problem}[Stable-envelope comparison]
Define a three-face stable-envelope discrepancy whose charge-\(2\)
matrix equals \(T_2\) after the basis and parameter normalization of
Section~\ref{sec:finite-yang-model}.  The equality is not a consequence
of the rational computation alone.
\end{problem}

\begin{problem}[All-order ZTE completion]
Find corrections beyond \(C_{ijk}\) which kill the nonlinear residual
on all charge sectors of \(V^{\otimes4}\), or prove that no such
finite per-face ternary completion exists in the chosen gauge class.
\end{problem}
