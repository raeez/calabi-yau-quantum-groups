\title{\texorpdfstring{The Pfaffian embedding
$\Lat^{3,2}\subset \II_{3,3}$ and the

and the Delta5 denominator}}
\author{Raeez Lorgat}
\date{}

\begin{abstract}
The exterior square of a rank-four lattice carries the Pfaffian form
of signature $(3,3)$.  A primitive vector
$q_0=e_1\wedge e_3+e_2\wedge e_4$ of square $-2$ has orthogonal
complement
\[
q_0^\perp \cong U_-\oplus U_-\oplus\langle 2\rangle,
\qquad
U_-=\begin{pmatrix}0&-1\\-1&0\end{pmatrix}.
\]
After multiplying the form by $-1$ this is the standard type-IV
lattice $L_{\Delta_5}\cong U\oplus U\oplus\langle -2\rangle$ on which
the Gritsenko-Nikulin denominator $\Delta_5$ lives.  The proved
extension from $\Lat^{3,2}$ to $\II_{3,3}$ supplies a primitive lattice
embedding and a stabilizer identity.  An $\OO(3,3)$ Borcherds product
or ambient generalized Kac-Moody algebra requires additional
denominator data.
\end{abstract}

\section{The Pfaffian lattice}

Let $V=\Z e_1\oplus\Z e_2\oplus\Z e_3\oplus\Z e_4$, choose
$\mathrm{vol}=e_1\wedge e_2\wedge e_3\wedge e_4$, and define a
symmetric bilinear form on $M=\bigwedge^2 V$ by
\[
(\omega_1,\omega_2)_{\mathrm{Pf}}\mathrm{vol}
=\omega_1\wedge\omega_2 .
\]

\begin{theorem}[Pfaffian model of $\II_{3,3}$]
\label{thm:pfaffian-II33}
The lattice $(M,(\ ,\ )_{\mathrm{Pf}})$ is even unimodular of
signature $(3,3)$, hence isometric to $\II_{3,3}=U^{\oplus 3}$.
\end{theorem}

\begin{proof}
Set
\[
a_1=e_1\wedge e_2,\quad b_1=e_3\wedge e_4,\quad
a_2=e_1\wedge e_3,\quad b_2=-e_2\wedge e_4,\quad
a_3=e_1\wedge e_4,\quad b_3=e_2\wedge e_3 .
\]
The only nonzero pairings among these basis vectors are
$(a_i,b_i)_{\mathrm{Pf}}=1$.  The Gram matrix is therefore
$U\oplus U\oplus U$.  This matrix is even, unimodular, and has
signature $(3,3)$.
\end{proof}

\section{The primitive codimension-one sublattice}

Define
\[
q_0=e_1\wedge e_3+e_2\wedge e_4.
\]
In the basis of Theorem~\ref{thm:pfaffian-II33}, $q_0=a_2-b_2$, so
$(q_0,q_0)_{\mathrm{Pf}}=-2$.

\begin{proposition}[Primitive Pfaffian complement]
\label{prop:primitive-complement}
The orthogonal complement $q_0^\perp\subset M$ is primitive and has
basis
\[
f_1=e_1\wedge e_2,\quad
f_2=e_2\wedge e_3,\quad
f_3=e_1\wedge e_3-e_2\wedge e_4,\quad
f_{-2}=e_4\wedge e_1,\quad
f_{-1}=e_4\wedge e_3 .
\]
Its Gram matrix is
\[
\begin{pmatrix}
0&0&0&0&-1\\
0&0&0&-1&0\\
0&0&2&0&0\\
0&-1&0&0&0\\
-1&0&0&0&0
\end{pmatrix}.
\]
Thus
\[
q_0^\perp\cong U_-\oplus U_-\oplus\langle 2\rangle,
\qquad
L_{\Delta_5}:=q_0^\perp(-1)\cong U\oplus U\oplus\langle -2\rangle .
\]
\end{proposition}

\begin{proof}
The displayed five vectors are orthogonal to $q_0$ by direct wedge
calculation.  The equation $(\omega,q_0)_{\mathrm{Pf}}=0$ imposes the
single integral relation that the coefficients of
$e_1\wedge e_3$ and $e_2\wedge e_4$ are negatives of one another, so
the displayed vectors form a basis of $q_0^\perp$.  The Gram matrix is
obtained by wedging each pair and reading the coefficient of
$\mathrm{vol}$.  Since $(e_1\wedge e_3,q_0)_{\mathrm{Pf}}=-1$, the map
$M\to\Z$, $\omega\mapsto(\omega,q_0)_{\mathrm{Pf}}$, is surjective;
therefore $q_0^\perp$ is primitive.
\end{proof}

The sign in $L_{\Delta_5}=q_0^\perp(-1)$ is structural.  The Pfaffian
convention makes $q_0^\perp$ a lattice of signature $(3,2)$.
The Borcherds type-IV convention uses the sign-reversed lattice of
signature $(2,3)$:
\[
\mathcal D(L_{\Delta_5})
=\left\{[Z]\in\mathbb P(L_{\Delta_5}\otimes\C):
(Z,Z)=0,\ (Z,\overline Z)>0\right\}^{+}.
\]

\section{\texorpdfstring{The stabilizer is $\Sp_4$}{The stabilizer is Sp4}}

The vector $q_0$ is the symplectic form hidden inside
$\bigwedge^2 V$.  Define
\[
-x\wedge y\wedge q_0
=B_{q_0}(x,y)\,\mathrm{vol},
\qquad x,y\in V.
\]
Then $B_{q_0}$ is a unimodular alternating form on $V$.

\begin{proposition}[Exterior-square stabilizer]
\label{prop:stabilizer-sp4}
The stabilizer of $q_0$ in $\SL(V)$ is $\Sp(V,B_{q_0})$.  The
exterior-square representation induces
\[
\Sp_4(\Z)/\{\pm I\}\longrightarrow \SO(q_0^\perp)_+
\]
with finite-index image, and over $\R$ it is the exceptional isogeny
\[
\PSp_4(\R)\simeq \SO(3,2)_+ .
\]
Under the type-IV realization this is the usual identification between
the genus-two Siegel upper half-space $\mathfrak H_2$ and
\(\mathcal D(L_{\Delta_5})\).
\end{proposition}

\begin{proof}
For $g\in\SL(V)$,
\[
(\bigwedge^2 g)(q_0)=q_0
\quad\Longleftrightarrow\quad
B_{q_0}(gx,gy)=B_{q_0}(x,y)\quad\text{for all }x,y\in V.
\]
This is precisely the defining condition for $\Sp(V,B_{q_0})$.  The
Pfaffian form is preserved by $\bigwedge^2\SL(V)$, hence the
stabilizer acts on $q_0^\perp$.  The kernel on $q_0^\perp$ is
$\{\pm I\}$.  The real Lie algebras have type $C_2=B_2$, giving
$\PSp_4(\R)\simeq\SO(3,2)_+$.  The integral statement follows by
restricting the exterior-square action to the lattice basis in
Proposition~\ref{prop:primitive-complement}.
\end{proof}

Humbert divisors in $\mathcal D(L_{\Delta_5})$ come from negative
vectors of $L_{\Delta_5}$.  The external vector $q_0$ produces the
type-IV lattice by orthogonal complement.  Humbert divisor classes live
inside the sign-reversed complement \(L_{\Delta_5}\).

\section{\texorpdfstring{The $\Delta_5$ denominator}{The Delta5 denominator}}

Let
\[
\phi_{0,1}^{\mathrm{EZ}}(\tau,z)
=y^{-1}+10+y+O(q)
=\sum f(n,\ell)q^ny^\ell
\]
be the primitive Eichler-Zagier weak Jacobi form in the Borcherds
normalization.  Gritsenko and Nikulin construct the Borcherds product
on $\mathcal D(L_{\Delta_5})$
\[
\Dfive(Z)=64^{-1}\Delta_5(Z)
=q^{1/2}r^{1/2}s^{1/2}
\prod_{(n,\ell,m)>0}
(1-q^nr^\ell s^m)^{f(nm,\ell)} .
\]
The coefficient $c_1(0)=10$ gives the Borcherds weight
\[
\kBKM(\mathfrak g_{\Delta_5})
=\kBKM(\Delta_5)=c_1(0)/2=5.
\]
For the primitive CHL ladder
$N\in\{1,2,3,4,6\}$,
\[
c_N(0)=(10,8,6,4,2),\qquad
\kBKM(\Phi_N)=c_N(0)/2=(5,4,3,2,1).
\]

\begin{proposition}[Denominator algebra on the type-IV lattice]
\label{prop:delta5-denominator}
The Gritsenko-Nikulin Borcherds-Kac-Moody superalgebra
$\mathfrak g_{\Delta_5}$ is attached to the type-IV lattice
$L_{\Delta_5}$.  The full Pfaffian lattice $\II_{3,3}$ supplies the
ambient exterior-square lattice.  On the active Borcherds cone its
signed root supermultiplicities are
\[
\sdim \mathfrak g_{\Delta_5,\alpha(n,\ell,m)}=f(nm,\ell),
\qquad
\alpha(n,\ell,m)=2nf_2-\ell f_3+2mf_{-2},
\]
and its denominator is
\[
\operatorname{den}(\mathfrak g_{\Delta_5})=\Dfive(2Z).
\]
\end{proposition}

\begin{proof}
The product formula above is the Weyl-Kac-Borcherds denominator
identity after the standard type-II chamber choice.  The exponent of
the factor indexed by $(n,\ell,m)$ is the signed dimension of the
corresponding root space.  The map
\(\alpha(n,\ell,m)=2nf_2-\ell f_3+2mf_{-2}\) is the integral
identification between the Jacobi exponent lattice and the Borcherds
root lattice inside $L_{\Delta_5}$.
\end{proof}

The Igusa square and the Oberdieck-Pandharipande scalar use different
normalizations:
\[
\PhiTen=\Delta_5^2,\qquad
\Phi_{10}^{\mathrm{OP}}=\Dfive^2=4096^{-1}\Delta_5^2,\qquad
Z_{\mathrm{OP/DT}}^{K3\times E}=-4096\,\Delta_5^{-2}.
\]
These scalar equalities belong to the automorphic and reduced-DT rows.
The compact Hall source requires its own construction.  The primitive
denominator value remains \(\kBKM(\Delta_5)=5\).

\section{\texorpdfstring{Ambient $\OO(3,3)$ denominator data}{Ambient O(3,3) denominator data}}

\begin{proposition}[The envelope obstruction]
\label{prop:envelope-obstruction}
The primitive embedding $L_{\Delta_5}\subset\II_{3,3}(-1)$ supplies
lattice and stabilizer data.  An $\OO(3,3)$ Borcherds product, an
$\OO(3,3)$ generalized Kac-Moody algebra, or a fixed-subalgebra
realization of $\mathfrak g_{\Delta_5}$ requires additional
automorphic, multiplicity, and restriction data.
\end{proposition}

\begin{proof}
Borcherds products used for $\Delta_5$ are holomorphic automorphic
forms on type-IV domains attached to lattices of signature $(2,n)$, up
to global sign.  The full Pfaffian lattice has signature $(3,3)$.  The
symmetric space $\OO(3,3)/(\OO(3)\times\OO(3))$ and the Grassmannian of
oriented positive three-planes are real symmetric spaces distinct from
the type-IV Hermitian domain of the Borcherds product theorem.  Choosing
$q_0$ and passing to $L_{\Delta_5}=q_0^\perp(-1)$ is exactly the step
which restores signature $(2,3)$ and supplies the domain carrying
$\Delta_5$.

The reflection in $q_0$ is an isometry of the ambient lattice.  An
automorphism of an ambient denominator algebra requires that algebra as
input.  Vectors with nonzero pairing against $q_0$ are ambient lattice
vectors; they become roots of $\mathfrak g_{\Delta_5}$ only after an
ambient denominator identity, root-multiplicity function, and
restriction theorem have been proved.
\end{proof}

An $\OO(3,3)$ envelope requires an automorphic or regularized
theta-lift datum on a specified
Hermitian replacement for the $\OO(3,3)$ space, a denominator identity
whose $q_0^\perp$-orthogonal root sector restricts to $\Dfive(2Z)$,
multiplicity preservation for all roots in $q_0^\perp$, and a proof
that no additional relations survive on restriction.
With these four inputs absent, the proved statement is
\[
\operatorname{Stab}_{\SL_4(\Z)}(q_0)/\{\pm I\}
\simeq \PSp_4(\Z)
\longrightarrow \SO(q_0^\perp)_+,
\qquad
\mathfrak g_{\Delta_5}\text{ lives on }L_{\Delta_5}.
\]

\section{Separated invariants}

The only invariant computed in this note is the Borcherds weight
\(\kBKM(\Delta_5)=5\).  For the compact space $K3\times E$, the
standard values remain distinct:
\[
\kappa_{\mathrm{cat}}(K3\times E)=0,\qquad
\kappa_{\mathrm{ch}}(K3\times E)=0,\qquad
\kappa_{\mathrm{ch}}^{\mathrm{Heis}}=3,\qquad
\kBKM(\Delta_5)=5,\qquad
\kappa_{\mathrm{fiber}}=24.
\]
The Borcherds weight is read from the principal-part constant.  The
primitive CHL row gives \((5,4,3,2,1)\), independently of the compact
Hodge supertrace.
