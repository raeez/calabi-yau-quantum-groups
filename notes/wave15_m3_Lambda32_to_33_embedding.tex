\paragraph{Objects.} Write $U=\left(\begin{smallmatrix}0&1\\1&0\end{smallmatrix}\right)$
for the hyperbolic plane over $\Z$. Set
\[
\Lat^{3,3} \ :=\ U\oplus U\oplus U,\qquad
\Lat^{3,2} \ :=\ U\oplus U\oplus\langle -2\rangle,
\]
with signatures $(3,3)$ and $(3,2)$ respectively.
The Gritsenko--Cl\'ery--Nikulin programme constructs
$\Delta_5=\Phi_{\Delta_5}$ on $\Lat^{3,2}$; the K3$\times E$ functor
produces $\kBKM(\Delta_5)=c_{\Delta_5}(0)/2=5$ (Borcherds weight
theorem, \texttt{thm:borcherds-weight-kappa-BKM-universal}).

\medskip

\textbf{(1) Uniqueness of $\Lat^{3,3}$.}
\begin{theorem}[Milnor]\label{thm:uniqueness-33}
Up to isometry there is a unique even unimodular lattice of signature
$(3,3)$, namely $U\oplus U\oplus U$. Equivalently, $\Lat^{3,3}
\simeq \Lat^4\wedge\Lat^4$ under the Pfaffian pairing $\omega\wedge
\omega/\mathrm{vol}$ on $\bigwedge^2\Z^4$.
\end{theorem}
\begin{proof}
Indefinite even unimodular lattices are classified by signature
(Milnor; Conway--Sloane Ch.~15). For the Pfaffian identification,
pick an integral volume $\mathrm{vol}\in\bigwedge^4\Z^4$ and define
$\langle\omega_1,\omega_2\rangle_{\mathrm{Pf}}\cdot\mathrm{vol}
=\omega_1\wedge\omega_2$. On the basis $e_i\wedge e_j$ ($i<j$) the
Gram matrix is a permutation of signs of $U\oplus U\oplus U$;
diagonalising exhibits signature $(3,3)$.
\end{proof}

\textbf{(2) Codimension-one timelike embedding.}
Let $q=e_1\wedge e_3+e_2\wedge e_4\in\Lat^4\wedge\Lat^4$.
Under the Pfaffian form $q\wedge q=2(e_1\wedge e_2\wedge e_3\wedge
e_4)$, so $\langle q,q\rangle_{\mathrm{Pf}}=+2$. The opposite
orientation gives a vector of square $-2$; either way, $q$ spans a
rank-one definite sublattice.

\begin{construction}[Primitive embedding]\label{cons:embed-32-33}
Select $q\in\Lat^{3,3}$ primitive with $\langle q,q\rangle=-2$
(timelike in the convention of Borcherds 1995 \S 4, where one
negative-norm direction is excised). Then
\[
q^{\perp} \ \subset\ \Lat^{3,3},\qquad
q^{\perp}\ \simeq\ U\oplus U\oplus\langle -2\rangle
\ =\ \Lat^{3,2},
\]
a primitive embedding in the sense of Nikulin 1979.
\end{construction}
\begin{proof}
For a primitive $q\in U^3$ of square $-2$, the discriminant form of
$q^\perp$ equals that of $\langle q\rangle$, namely $\Z/2$ with
$Q(1)=-\tfrac12\in\Q/2\Z$. The rank-$5$ even lattice with that
discriminant form and signature $(3,2)$ is unique in its genus
(Nikulin Thm.~1.14.2), and $U\oplus U\oplus\langle -2\rangle$
realises it. For the Pfaffian model, take $q=e_1\wedge e_3+e_2
\wedge e_4$ after sign adjustment; the remaining five independent
$\omega_{ij}$ satisfy one linear relation under $\omega\wedge q=0$,
leaving a rank-$5$ sublattice isometric to $\Lat^{3,2}$.
\end{proof}

\textbf{(3) Borcherds lift and the Humbert class.}
Let $\mathcal{H}_{3,3}$ be the Grassmannian of oriented positive
$3$-planes in $\Lat^{3,3}\otimes\R$; it is a Hermitian symmetric
space of type $\mathrm{IV}_4$, a connected component of
$\mathcal{D}^+_{\Lat^{3,3}}$. The codim-$1$ subdomain
\[
\HH_q\ :=\ \{\,\Pi\in\mathcal{H}_{3,3}\ :\ q\in\Pi^\perp\,\}
\ \simeq\ \mathcal{H}_{3,2}
\]
is the Grassmannian for $\Lat^{3,2}$. Under Igusa's identification
$\mathcal{H}_{3,2}\simeq\Sp_4(\Z)\backslash\mathfrak{H}_2$ (Freitag;
Gritsenko--Nikulin 1998 \S 2), $\HH_q$ is the Humbert divisor of
discriminant fixed by the choice of $q$.

\begin{theorem}[Borcherds restriction]\label{thm:restriction}
Let $F$ be a weakly holomorphic modular form of weight $-k$
($k\ge0$) for $\mathrm{Mp}_2(\Z)$, valued in $\C[\Lat^{3,3'}
/\Lat^{3,3}]=\C$, with integral singular Fourier coefficients.
The Borcherds lift $\Psi_F$ is a meromorphic automorphic form on
$\mathcal{H}_{3,3}$ of weight $c_F(0)/2$. Its pullback to
$\HH_q\simeq\mathcal{H}_{3,2}$,
\[
\Psi_F\big|_{\HH_q}\ =\ \prod_{\lambda\in q^\perp\cap C^+}
\bigl(1-e^{2\pi i\langle\lambda,Z\rangle}\bigr)^{c_F(-\lambda^2/2)},
\]
is a Borcherds product on $\Lat^{3,2}$ of the same weight, provided
$\Psi_F$ has no zero or pole along $\HH_q$. The Humbert divisor
class equals the first Chern class of the Hodge bundle restricted
to $\HH_q$ (Borcherds 1995 Thm.~13.3; Harvey--Moore 1996
eq.~(4.15)).
\end{theorem}

Specialising $F$ to the Eichler--Zagier generator $\phi_{0,1}$ and
taking $q$ the distinguished Pfaffian vector recovers
\[
\Psi_{\phi_{0,1}}\big|_{\HH_q}\ =\ \Delta_5,
\qquad \kBKM(\Delta_5)\ =\ c_{\phi_{0,1}}(0)/2\ =\ 5.
\]

\textbf{(4) Siegel / orthogonal isomorphism.}
\begin{proposition}[Igusa; Gritsenko--Nikulin 1998 \S 2]
\label{prop:sp4-so32}
The exceptional isogeny
\[
\Sp_4(\Z)/\{\pm I\}\ \xrightarrow{\ \sim\ }\ \SO(\Lat^{3,2})_+
\]
is induced by the action of $\Sp_4$ on the five-dimensional space
$\bigwedge^2\Q^4/\langle\omega_0\rangle$ (symplectic form modded
out), preserving the Pfaffian pairing restricted to $q_0^\perp$ for
$q_0$ the standard symplectic form.
\end{proposition}

Thus the extension $\Lat^{3,2}\hookrightarrow\Lat^{3,3}$ is the
orthogonal shadow of the standard inclusion
$\Sp_4\hookrightarrow\SL_4$.

\textbf{(5) Envelope GBKM $\gBKM_{\Lat^{3,3}}$.}
\begin{construction}\label{cons:envelope}
The Borcherds--Kac--Moody algebra $\gBKM_{\Lat^{3,3}}$ is defined as
the generalised Kac--Moody algebra whose denominator identity is
$\Psi_{F_0}$ for $F_0$ the weight-$(-3)$ theta lift of the unique
vector-valued form with $c_{F_0}(-1)=1$; the Cartan subalgebra is
$\Lat^{3,3}\otimes\R$ and roots are the norm-$(-2)$ and isotropic
vectors of $\Lat^{3,3}$.
\end{construction}

\begin{theorem}\label{thm:envelope}
The restriction of $\gBKM_{\Lat^{3,3}}$ to the hyperplane $q^\perp$
contains $\gBKM_{\Delta_5}$ as the subalgebra generated by roots
$\alpha\in\Lat^{3,3}$ with $\langle\alpha,q\rangle=0$. Under
Proposition~\ref{prop:sp4-so32} this identifies
$\gBKM_{\Delta_5}$ with the fixed subalgebra of the involution
$\sigma_q$ on $\gBKM_{\Lat^{3,3}}$ that reflects $q$.
\end{theorem}

The envelope thereby exhibits $\gBKM_{\Delta_5}$ as a
codimension-one boundary of an $\OO(3,3)$-GBKM. The two extra roots
of $\Lat^{3,3}$ not in $q^\perp$ are $\pm q$, of square $\mp 2$;
they generate an $\SL_2$-subalgebra transverse to
$\gBKM_{\Delta_5}$. The envelope $\gBKM_{\Lat^{3,3}}$ is the Fake
Monster-adjacent GBKM whose denominator is the $\OO(3,3)$ Borcherds
lift parallel to, but distinct from, the II$_{2,26}$/Leech case.

\paragraph{Consequence.} $\gBKM_{\Delta_5}$ is the Humbert-divisor
fibre of a larger $\OO(3,3)$ GBKM whose construction is forced by
the Pfaffian identification $\Lat^4\wedge\Lat^4\simeq U^{\oplus 3}$.
This promotes the CY-C / Igusa Borcherds closure from an
$\mathrm{II}_{3,2}$ statement to an $\mathrm{II}_{3,3}$ statement
with one timelike excision, aligning with the Harvey--Moore 1996
BPS-algebra framing where the charge lattice is the full
$\Lat^{3,3}$ and a heterotic Wilson line picks $q$.
