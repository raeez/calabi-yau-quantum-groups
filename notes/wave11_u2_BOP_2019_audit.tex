\section*{Wave-11 U2/20 audit: the claimed ``Bryan--Oberdieck--Pandharipande 2019''}

\subsection*{Primary-source finding (refutation of R3)}

\ClaimStatusTheorem\ (as an audit result).
The identifier \texttt{arXiv:1903.07729} does \emph{not} correspond to
a Bryan--Oberdieck--Pandharipande paper. It is
\emph{``Constitutive modelling of mechanically induced martensitic
transformations''} by D.~de~Bortoli, F.~Adziman, E.~A.~de~Souza~Neto
and F.~M.~Andrade~Pires, a materials-science paper (March~2019).
No three-author Bryan--Oberdieck--Pandharipande paper exists on
\texttt{arXiv} in any year; no Oberdieck--Pandharipande paper appears
in \emph{Duke Math.~J.} \textbf{168}:7 (the issue contains
Bergeron--Li, Soundararajan--Thorner, Xu--Zhang, Beuzart-Plessis--Wan).
The R3 citation as a ``decisive primary-source'' for programme-restricted
Conjecture~1$'$ at $N\in\{1,2,3,4,6\}$ is \textbf{not supported by the
literature}.

\subsection*{The actual BOP-adjacent literature (verified)}

\textbf{(A) Oberdieck--Pandharipande~2014, arXiv:1411.1514},
\emph{``Curve counting on K3$\times$E, the Igusa cusp form $\chi_{10}$,
and descendent integration''} (Springer LNM, \emph{K3 Surfaces and Their
Moduli}, 2016).
\emph{Conjectures} the identity
$Z^{\mathrm{red}}_{\mathrm{GW}}(K3\times E) = -1/\chi_{10}(\tau,z,\tilde\tau)$
at $N=1$; proves primitive-class descendent cases. \textbf{Conjectural at $N=1$}.

\textbf{(B) Bryan~2015, arXiv:1504.02920}, \emph{``The DT theory of
K3$\times$E via the topological vertex''}. Reduced-DT computation for
primitive classes of square $-2$ and $0$, via topological vertex.
\textbf{Partial verification} of (A) on the DT side.

\textbf{(C) Oberdieck--Pixton~2017, arXiv:1706.10100},
\emph{``Holomorphic anomaly equations and the Igusa cusp form
conjecture''} (\emph{Invent.~Math.}~\textbf{222} (2020), 667--759).
\ClaimStatusTheorem\ \emph{Main Theorem}: the reduced GW theory of
$S\times E$ is solved for all curve classes \textbf{primitive in the K3
factor}; the Igusa cusp-form conjecture (A) at $N=1$ follows as a
corollary. \textbf{Scope: $N=1$ only; no CHL orbifold; GW side only}.

\textbf{(D) Bryan--Oberdieck~2018, arXiv:1811.06102},
\emph{``CHL Calabi--Yau threefolds: curve counting, Mathieu moonshine
and Siegel modular forms''}. \textbf{Conjecture}: the primitive DT
partition function of an elliptic CHL model
$[K3\times E/\mathbb{Z}_N]$ is the Borcherds/Gritsenko lift of the
corresponding twisted-twined Mathieu elliptic genus, \emph{for each
$N\in\{2,3,4,6\}$} (symplectic $M_{24}$ conjugacy class pairing).
\emph{Theorem 0.1} proves \emph{base cases via the topological vertex}
at \textbf{$N=2$ only}, primitive square~$-2$, $0$.
\textbf{General $N$ remains conjectural}.

\textbf{(E) Bryan--Oberdieck--Pandharipande--Yin~2015, arXiv:1506.00841}.
Four authors; \emph{abelian} surfaces/threefolds, not CHL orbifolds of
K3$\times$E. Not a substitute for the claimed reference.

\subsection*{What R3 attempted to cite}

R3 asserted that ``Bryan--Oberdieck--Pandharipande 2019, Theorem~1''
extends Oberdieck--Pixton at $N=1$ to all $N\in\{1,2,3,4,6\}$,
establishing $Z^{\mathrm{red}}_{\mathrm{GW}} = 1/\Phi_N$ on the crepant
resolution of $[K3\times E/\mathbb{Z}_N]$. The audit of items (A)--(E)
shows: (i) no such paper exists; (ii) the Bryan--Oberdieck~2018 paper
(D) \emph{conjectures} the identity at $N\in\{2,3,4,6\}$ but
\emph{proves} only base cases at $N=2$; (iii) the Oberdieck--Pixton
theorem (C) is GW only at $N=1$; (iv) no MNOP-type GW/DT comparison
theorem converts (C) to reduced DT in general.

\subsection*{Verbatim status of the primary result on the DT side}

\ClaimStatusConjectured\ \textit{(Bryan--Oberdieck, arXiv:1811.06102,
Conj.~0.1 + 0.2)}: for each $N\in\{2,3,4,6\}$ with symplectic automorphism
data $(g,N)$ a Mathieu-moonshine pair, the primitive reduced DT
partition function of the CHL Calabi--Yau threefold
$[K3\times E/\mathbb{Z}_N]$ equals
\(
-1/\Phi_{g,N}(\tau,z,\tilde\tau),
\)
where $\Phi_{g,N}$ is the Gritsenko--Borcherds lift of the twisted-twined
elliptic genus $\phi_{g,N}$ of $M_{24}$-moonshine.
\emph{Proved at $N=2$ primitive, square $\in\{-2,0\}$}; open otherwise.

\subsection*{Implication for programme-restricted Conjecture~1$'$}

The cache-key identification $\kBKM(\Phi_N) = c_N(0)/2$ remains \emph{a
Borcherds-theorem consequence on the automorphic side}
(Borcherds~1995; Gritsenko~1999), independent of the geometric
DT-reduced interpretation. Programme-restricted Conjecture~1$'$,
\emph{insofar as it claims a primary-source GW$=$DT proof at
$N\in\{1,2,3,4,6\}$}, is presently supported only at $N=1$
(Oberdieck--Pixton GW + Bryan primitive DT base cases, plus the
unconditional MNOP for local K3) and remains conjectural at
$N\in\{2,3,4,6\}$ (Bryan--Oberdieck 2018 conjectures, $N=2$ base
cases only).
