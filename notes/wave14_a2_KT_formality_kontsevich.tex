\author{Raeez Lorgat}

\begin{document}

\begin{abstract}
A Koszul--Tate resolution is a commutative-algebraic resolution of a
constraint algebra, controlled by regularity of the defining equations.
A BV--BRST complex records equations of motion, gauge symmetries, ghosts,
and antifields. Costello quantisation adds gauge fixing, heat-kernel
renormalisation, completed local functionals, and the quantum master
equation. Kontsevich--Tamarkin formality supplies an operadic zigzag
between chains on little disks and the corresponding Poisson operad.
These four layers meet in holomorphic Chern--Simons theory, with separate
hypotheses at each layer. For a Calabi--Yau threefold, the
surface--curve specialisation of the CY-to-chiral construction has
$\Eone$ output; $\Etwo$ structures live on centres, doubles, or module
layers after their own dualisability, gluing, and completion data have
been supplied.
\end{abstract}

\section{Koszul--Tate Regularity}
\label{sec:koszul-tate-regularity}

Let $R$ be a commutative algebra over a characteristic-zero field and
let $f_1,\ldots,f_r\in R$. The Koszul--Tate algebra of the constraint
system $f_1=\cdots=f_r=0$ is
\[
K_R(f_1,\ldots,f_r)
=
\bigl(R[\epsilon_1,\ldots,\epsilon_r],d\bigr),
\qquad
|\epsilon_i|=-1,\qquad d\epsilon_i=f_i,
\]
with the graded-commutative product and the derivation extension of $d$.
It carries an augmentation
\[
K_R(f_1,\ldots,f_r)\longrightarrow R/(f_1,\ldots,f_r).
\]

\begin{proposition}[Koszul--Tate resolution criterion]
\label{prop:kt-regularity}
If $f_1,\ldots,f_r$ is an $R$-regular sequence, then
\[
K_R(f_1,\ldots,f_r)\xrightarrow{\simeq}R/(f_1,\ldots,f_r)
\]
is a quasi-isomorphism. For an arbitrary sequence, the same complex
models the derived zero locus
\[
R\otimes^{\mathbf L}_{\mathbb K[t_1,\ldots,t_r]}\mathbb K,
\qquad
t_i\mapsto f_i,\quad t_i\mapsto 0,
\]
and its negative cohomology is the Koszul homology measuring the failure
of regularity.
\end{proposition}

\begin{proof}
For $r=1$, the complex is $R\epsilon_1\to R$, with differential given by
multiplication by $f_1$. Exactness in degree $-1$ is precisely the
non-zero-divisor condition. The general case follows by induction:
after quotienting by $(f_1,\ldots,f_{i-1})$, multiplication by $f_i$ is
injective exactly when the next two-term Koszul step is exact in
negative degree. Tensoring the elementary two-term complexes over
$\mathbb K[t_1,\ldots,t_r]$ gives the displayed derived tensor product.
\end{proof}

\begin{definition}[Higher Tate data]
\label{def:higher-tate-data}
A higher Tate resolution of $R/(f_1,\ldots,f_r)$ is a semi-free
commutative dg algebra
\[
T=
K_R(f_1,\ldots,f_r)[\eta_j\mid j\in J],
\qquad |\eta_j|\leq -2,
\]
whose differential sends each new generator to a cycle representing a
chosen syzygy class in the preceding stage. It is a resolution when the
augmentation $T\to R/(f_1,\ldots,f_r)$ induces
\[
H^0(T)\cong R/(f_1,\ldots,f_r),
\qquad
H^i(T)=0\quad (i<0).
\]
\end{definition}

\begin{proposition}[Acyclicity source]
\label{prop:kt-acyclicity-source}
Koszul--Tate acyclicity has two standard sources. A regular sequence,
or sheafwise a local complete-intersection presentation, gives the
length-one complex of Proposition~\ref{prop:kt-regularity}. For a
non-regular constraint system, acyclicity follows from explicit higher
Tate data of Definition~\ref{def:higher-tate-data} plus a filtration
argument whose associated graded complex kills the syzygy homology stage
by stage. In the absence of either source, the first Koszul complex
computes the derived zero locus and carries the remaining negative
homology as obstruction data.
\end{proposition}

\begin{proof}
The regular case is Proposition~\ref{prop:kt-regularity}. In the higher
Tate case, filter $T$ by the number of added generators. The associated
graded differential separates the old differential from the new
syzygy-killing terms. The spectral sequence has first page equal to the
previous Koszul homology tensored with the free algebra on the new
generators, and the prescribed differentials send the chosen homology
generators to boundaries. Iterating over the filtered tower leaves
$R/(f_1,\ldots,f_r)$ in degree zero and kills negative degrees.
\end{proof}

\begin{proposition}[Koszul--Tate and BV--BRST are separate layers]
\label{prop:kt-bv-brst}
Koszul--Tate regularity belongs to the commutative algebra of
constraints and is detected by negative Koszul homology or by a higher
Tate tower. A BV--BRST complex has separate typed entries: fields,
equations of motion, infinitesimal gauge symmetries, ghosts, antifields,
an odd symplectic pairing, a classical master equation, and an
observables complex. A quantum BV theory adds a BV Laplacian, a
renormalisation scheme, counterterms, and a quantum master equation.
Gauge-theoretic acyclicity supplies information about the quotient by
symmetries; KT acyclicity supplies information about the constraint
algebra.
\end{proposition}

\section{Operadic Formality}
\label{sec:operadic-formality}

Let $\cD_n$ be the little $n$-disks operad and let
\[
e_n := H_\bullet(\cD_n;\QQ) \simeq \Pois_n .
\]
In cohomological conventions, an $e_n$-algebra has a commutative product
of degree $0$ and a Lie bracket of degree $1-n$. Thus an $e_3$-algebra
has a degree $-2$ bracket, and restriction along the shifted Lie
suboperad gives an $\Linf$-structure on the shifted object $V[2]$.

\begin{theorem}[Little three-disks formality]
\label{thm:D3-formality}
The chain operad $C_\bullet(\cD_3;\mathbb K)$ is formal over a field of
characteristic zero after the choice of a formality model:
\[
C_\bullet(\cD_3;\mathbb K)\simeq e_3 .
\]
Over $\RR$ this is Kontsevich's configuration-space formality theorem,
with the Fulton--MacPherson model developed by Lambrechts--Volic. Over
$\QQ$, the case $n=3$ is covered by the rational intrinsic formality
theorem of Fresse--Willwacher for $E_n$-operads with $n\geq 3$.
\end{theorem}

\begin{proof}
Kontsevich constructs a graph cooperad model and a configuration-space
integral map to forms on the Fulton--MacPherson operad $\FM_n$; for
$n=3$ this gives a real formality zigzag
$C_\bullet(\cD_3;\RR)\simeq H_\bullet(\cD_3;\RR)$. The rational
statement follows from Fresse--Willwacher: for $n\geq 3$, the rational
homotopy type of an $E_n$-operad is forced by the $n$-Poisson cohomology
cooperad. The reflection condition that appears in their theorem imposes
no extra condition when $4\nmid n$, hence in particular when $n=3$.
\end{proof}

\begin{proposition}[Shifted Lie data and the full $e_3$ datum]
\label{prop:lie-part-e3}
Let $V$ be an $\Ethree$-algebra and choose a formality transfer to an
$e_3$-model. The induced $\Linf$-structure on $V[2]$ is the image of the
Lie generator of $e_3$. Reconstruction of the $e_3$-structure requires
the commutative product, the degree $-2$ bracket, the Poisson
compatibility, and the chosen formality zigzag.
\end{proposition}

\begin{proof}
The homology operad $e_3$ is generated by a commutative multiplication
and a bracket of degree $-2$, modulo Jacobi and Poisson relations. The
restriction to shifted Lie algebras retains the bracket generator. The
product and the Leibniz relation are independent operadic generators and
relations in the reconstruction problem.
\end{proof}

\begin{remark}[Associator parameters]
\label{rem:associator-parameters}
Drinfeld associators and the prounipotent Grothendieck--Teichmuller
action parameterise Tamarkin formality for $E_2$ and deformation
quantisation. Severa--Willwacher compare Kontsevich graph formality with
Tamarkin formality for a special associator. An analogous
parameter-space statement for $E_3$ is an additional graph-complex
theorem.
\end{remark}

\section{Hochschild Objects}
\label{sec:hochschild-objects}

For a dg category or stable $\infty$-category $\cC$,
$C^\bullet(\cC,\cC)$ denotes categorical Hochschild cochains, the
derived endomorphisms of the identity functor. Its cohomology is
$\HH^\bullet(\cC)$.

\begin{theorem}[Deligne and higher Deligne]
\label{thm:deligne-higher}
For an associative dg algebra, or for a dg category regarded as an
$\Eone$ object, Hochschild cochains carry a natural $\Etwo$ structure.
More generally, Hochschild cochains of an $\En$-algebra carry an
$E_{n+1}$ structure under the hypotheses of the higher Deligne theorem
of Francis and Lurie. These are chain-level Hochschild constructions
with hypotheses separate from operadic formality of little disks.
\end{theorem}

\begin{proposition}[Five distinct objects]
\label{prop:five-objects}
For an $\Eone$ chiral output $A$, the following objects carry distinct
definitions and comparison maps:
\[
A,\qquad B(A),\qquad A^i,\qquad A^!,\qquad
Z^{\mathrm{der}}_{\mathrm{ch}}(A).
\]
Here $B(A)$ is the bar coalgebra, $A^i$ is the Koszul-dual coalgebra when
the conilpotent hypotheses hold, $A^!$ is the Verdier or linear dual
algebraic object when dualisability is present, and
$Z^{\mathrm{der}}_{\mathrm{ch}}(A)$ is the derived chiral centre. The
bar--cobar counit $\Omega B(A)\to A$, Koszul duality
$A\leftrightarrow A^i$, Verdier duality $A\mapsto A^!$, and the centre
map $A\to Z^{\mathrm{der}}_{\mathrm{ch}}(A)$ are different
constructions.
\end{proposition}

\begin{corollary}[Compact CY3 Hochschild closure]
\label{cor:cy3-global-closure-hypotheses}
Let $X$ be a compact Calabi--Yau threefold and let $\cC=\Perf(X)$. The
usual Hochschild cochains of $\cC$ carry the $\Etwo$ structure of
Deligne's theorem. A compact global $\Ethree$ statement on
$\HH^\bullet(\cC)$ requires a TCFT or framed higher-Deligne construction
compatible with the Calabi--Yau trace, a completed observables complex,
orientation and framing data, a solved anomaly problem, gauge fixing,
and an obstruction filtration controlling $\HH^{-2}(\cC)$.
\end{corollary}

\begin{proposition}[HKR and Todd normalisation]
\label{prop:hkr-todd}
For a smooth projective variety $X$, Hochschild--Kostant--Rosenberg
gives a graded-vector-space isomorphism
\[
\HH^m(\Perf(X))\cong
\bigoplus_{p+q=m}H^q(X,\Lambda^pT_X).
\]
Compatibility with products, pairings, and Chern characters requires the
usual Todd-class normalisation. The Calabi--Yau condition $c_1(X)=0$
leaves the $c_2(X)$ and higher terms in $\sqrt{\mathrm{td}_X}$.
\end{proposition}

\section{Holomorphic Chern--Simons BV Data}
\label{sec:hcs-bv-data}

Let $X$ be a complex threefold with holomorphic volume form $\Omega_X$,
let $\gfrak$ be a finite-dimensional Lie algebra with invariant
nondegenerate pairing $B_\gfrak$, and set
\[
\mathcal E_X=\Omega^{0,\bullet}(X,\gfrak)[1].
\]
The classical holomorphic Chern--Simons functional is
\begin{equation}
S_{\hCS}(\mathcal A)=
\int_X\Omega_X\wedge
B_\gfrak\!\left(
\frac12\mathcal A\bar\partial\mathcal A+
\frac16\mathcal A[\mathcal A,\mathcal A]
\right).
\label{eq:hcs-action}
\end{equation}

\begin{proposition}[Maurer--Cartan equation and BV presentation]
\label{prop:hcs-mc}
The Euler--Lagrange equation of \eqref{eq:hcs-action} is
\[
\bar\partial\mathcal A+\frac12[\mathcal A,\mathcal A]=0.
\]
Equivalently, classical hCS is governed by the dg Lie algebra
\[
L_{\hCS}(X,\gfrak)=
\bigl(\Omega^{0,\bullet}(X,\gfrak),\bar\partial,[-,-]\bigr),
\]
where the bracket combines the Lie bracket on $\gfrak$ with wedge product
of forms. The shift $[1]$ and the pairing
$\int_X\Omega_X\wedge B_\gfrak(-,-)$ give the BV field theory.
\end{proposition}

\begin{proof}
Varying \eqref{eq:hcs-action} and using invariance of $B_\gfrak$ gives
\[
\delta S_{\hCS}(\mathcal A)=
\int_X\Omega_X\wedge
B_\gfrak\!\left(
\delta\mathcal A,
\bar\partial\mathcal A+\frac12[\mathcal A,\mathcal A]
\right).
\]
Nondegeneracy of the pairing gives the displayed Maurer--Cartan
equation. The differential and bracket are the operations of
$L_{\hCS}(X,\gfrak)$.
\end{proof}

\begin{definition}[BV--BRST differential]
\label{def:bv-brst}
The hCS BV--BRST complex adjoins a ghost $c\in
\Omega^{0,\bullet}(X,\gfrak)$ for infinitesimal gauge transformations
and antifields dual to the Euler--Lagrange equations. The BRST part has
the local form
\[
Q_{\mathrm{BRST}}\mathcal A
=
\bar\partial c+[\mathcal A,c],
\qquad
Q_{\mathrm{BRST}}c=-\frac12[c,c],
\]
and the BV part pairs fields with antifields through the odd symplectic
form induced by $\Omega_X$ and $B_\gfrak$.
\end{definition}

\begin{definition}[Classical and quantum BV equations]
\label{def:cme-qme}
The classical BV datum consists of an odd symplectic field complex, an
action functional $S$, and the classical master equation
\[
Q_{\mathrm{lin}}S+\frac12\{S,S\}=0.
\]
Its BRST quotient is encoded by the Chevalley--Eilenberg complex of the
gauge dg Lie algebra acting on fields. A quantum BV datum at scale $L$
adds a second-order operator $\Delta_L$ and an effective interaction
$I[L]$ satisfying the scale-$L$ quantum master equation.
\end{definition}

\begin{definition}[Costello quantisation datum]
\label{def:costello-quantization}
A perturbative Costello quantisation of hCS consists of a gauge fixing,
a heat-kernel propagator $P_{\epsilon<L}$, completed local functionals,
effective interactions $I[L]\in\cO_{\mathrm{loc}}(\mathcal E_X)[[\hbar]]$,
renormalisation-group compatibility, a counterterm scheme, and the
scale-$L$ quantum master equation
\[
QI[L]+\frac12\{I[L],I[L]\}_L+\hbar\Delta_LI[L]=0.
\]
The kernel satisfies the homotopy identity
\[
[\bar\partial,P_{\epsilon<L}]=K_L-K_\epsilon,
\]
where $K_t$ is the Dolbeault heat kernel.
\end{definition}

\begin{proposition}[Observables from a solved QME]
\label{prop:propagator-observables}
When the Costello quantisation datum of
Definition~\ref{def:costello-quantization} is fixed and the QME is
solved, the completed quantum observables form a renormalised
pre-factorisation algebra $\Obs^q_{\hCS}$. The propagator belongs to the
renormalisation datum; an operadic formality morphism is a separate
quasi-isomorphism of operads or a transfer of algebra structure along
such a quasi-isomorphism.
\end{proposition}

\begin{proposition}[Compact hCS closure datum]
\label{prop:compact-hcs-closure}
A compact holomorphic Chern--Simons quantisation on a Calabi--Yau
threefold requires the following data before it supplies a global
observable theory:
\begin{enumerate}
\item orientation and determinant-line framing;
\item elliptic gauge fixing and heat-kernel regularisation;
\item completed local functionals and an RG-compatible counterterm
scheme;
\item a solution of the QME in the completed BV complex;
\item vanishing or explicit trivialisation of the local anomaly class;
\item descent or TCFT sewing for compact opens;
\item comparison maps to Hochschild, Hall, or CY-to-chiral objects
preserving filtrations and completions.
\end{enumerate}
\end{proposition}

\section{Quartic Anomaly and Two-Point Term}
\label{sec:quartic-anomaly-two-point}

For a representation $\rho:\gfrak\to\operatorname{End}(V)$, define the
symmetric quartic invariant
\[
I_4^\rho(x_1,x_2,x_3,x_4)
=
\frac1{4!}\sum_{\sigma\in S_4}
\operatorname{Tr}_V\!\left(
\rho(x_{\sigma(1)})\rho(x_{\sigma(2)})
\rho(x_{\sigma(3)})\rho(x_{\sigma(4)})
\right),
\]
so $I_4^\rho\in\operatorname{Sym}^4(\gfrak^\vee)^\gfrak$. If
$\{T_a\}$ is a basis of $\gfrak$ and $\{T^a\}$ is dual for the invariant
pairing, set
\[
C_2^\rho=\sum_a\rho(T_a)\rho(T^a).
\]

\begin{proposition}[Quartic QME obstruction]
\label{prop:quartic-qme}
In complex dimension three, the nonabelian one-loop gauge contribution
to the QME obstruction is represented in local CE cochains by
\[
\Theta^\rho_U(\alpha_0,\alpha_1,\alpha_2,\alpha_3)
=
\int_U
\left[
I_4^\rho\bigl(
\alpha_0,\partial\alpha_1,\partial\alpha_2,\partial\alpha_3
\bigr)
\right]_{(3,3)},
\qquad
\alpha_i\in\Omega_c^{0,\bullet}(U,\gfrak).
\]
Its cohomology class lies in
\[
H^1_{\mathrm{loc}}
\bigl(\Omega^{0,\bullet}(X,\gfrak)[1],\cO_{\mathrm{loc}}\bigr).
\]
The degree-zero element $C_2^\rho$ governs the two-point field-strength
counterterm and the kinetic-pairing renormalisation.
\end{proposition}

\begin{proof}
The one-loop obstruction graph in hCS has four external gauge insertions.
The colour factor is the symmetrised trace of four Lie algebra inputs,
hence $I_4^\rho$. The form factor has three holomorphic derivatives; its
top component is a local functional of cohomological degree one. Closing
two legs of the cubic vertex gives the contraction
$\sum_a\rho(T_a)\rho(T^a)=C_2^\rho$, a degree-zero two-point term with
the tensor type of the kinetic pairing.
\end{proof}

\section{Local Graph Formality}
\label{sec:local-graph-formality}

On a holomorphic polydisk $U\simeq\CC^3$ with flat metric and constant
volume form, translation-invariant hCS observables form a local
holomorphic factorisation algebra. Configuration-space integrals of
propagator forms define graph weights.

\begin{proposition}[Graph comparison datum]
\label{prop:graph-comparison}
A comparison between hCS graph weights and a Kontsevich formality
morphism requires a graph-operad map
\[
\Graphs_3\longrightarrow\Omega^\bullet(\FM_3),
\]
together with boundary-stratum cancellation, orientation signs, symmetry
factors, propagator normalisation, and compatibility with the chosen
formality zigzag of Theorem~\ref{thm:D3-formality}.
\end{proposition}

\section{Surface--Curve Specialisation}
\label{sec:surface-curve-specialisation}

The CY-to-chiral construction for a Calabi--Yau threefold factorises as
\[
\Phi_3=\Sp_{\Sigma_2,C}\circ\Phi^{\mathrm{FA}}_3.
\]
The factorisation-algebra term requires locality, formality, framing,
completion, and anomaly hypotheses. The functor $\Sp_{\Sigma_2,C}$
specialises along the surface-and-curve datum $(\Sigma_2,C)$.

\begin{proposition}[$\Eone$ output at $d=3$]
\label{prop:d3-output-level}
For $d=3$, the specialised object
\[
A_{X,\Sigma_2,C}:=
\Sp_{\Sigma_2,C}\bigl(\Phi^{\mathrm{FA}}_3(\Perf(X))\bigr)
\]
is an $\Eone$ chiral output. An $\Etwo$ structure belongs to
$Z^{\mathrm{der}}_{\mathrm{ch}}(A_{X,\Sigma_2,C})$, to a Drinfeld
double, or to a module-centre construction after the corresponding
dualisability, completion, and gluing hypotheses are supplied.
\end{proposition}

\begin{proof}
The CY-to-chiral dimension rule assigns $\Etwo$-chiral output for
$d\leq2$ and $\Eone$-chiral output for $d\geq3$ after
surface--curve specialisation. Centre, double, and module-layer
constructions carry their own existence hypotheses and their own
operadic level.
\end{proof}

\begin{proposition}[Invariant separation on \texorpdfstring{$K3\times E$}{K3 x E}]
\label{prop:k3e-invariant-separation}
For $X=K3\times E$, the compact and specialisation invariants are
\[
\kappa_{\mathrm{cat}}(X)=\chi(\cO_X)=0,
\qquad
\kappa_{\mathrm{ch}}(A_X)=\sum_q(-1)^qh^{0,q}(X)=0,
\]
\[
\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(X)=3,
\qquad
\kappa_{\mathrm{BKM}}(\mathfrak g_{\Delta_5})=5,
\qquad
\kappa_{\mathrm{fiber}}(K3)=24.
\]
The first two values are compact total-space invariants. The last three
values are the Heisenberg specialisation, the Borcherds automorphic
weight, and the K3 fibre-rank invariant.
\end{proposition}

\begin{proof}
K\"unneth gives
\[
\chi(\cO_{K3\times E})=\chi(\cO_{K3})\chi(\cO_E)=2\cdot0=0.
\]
The Hodge column of $K3\times E$ is
$(h^{0,0},h^{0,1},h^{0,2},h^{0,3})=(1,1,1,1)$, hence the compact
Hodge supertrace is $1-1+1-1=0$. The Heisenberg witness records the
Mukai-Heisenberg rank $2+1=3$ after the K3-fibre and elliptic-base
specialisation. Borcherds' weight formula gives
$\kappa_{\mathrm{BKM}}(\mathfrak g_{\Delta_5})=c_1(0)/2=10/2=5$.
The fibre value is $\dim_\CC H^\bullet(K3,\CC)=24$.
\end{proof}

\section{Conditional Threefold Statement}
\label{sec:conditional-threefold-statement}

\begin{theorem}[Koszul--Tate, formality, and hCS at $d=3$]
\label{thm:kt-formality-hcs-d3}
Let $X$ be a smooth Calabi--Yau threefold with holomorphic volume form
$\Omega_X$, let $\gfrak$ be a finite-dimensional Lie algebra with
invariant nondegenerate pairing, and let $\cC=\Perf(X)$.

\begin{enumerate}
\item A Koszul--Tate resolution of a constraint algebra attached to
$X$ is controlled by the regular or local complete-intersection
criterion of Proposition~\ref{prop:kt-regularity}, or by explicit
higher Tate data as in Proposition~\ref{prop:kt-acyclicity-source}.
BV--BRST acyclicity and QME solvability are separate hypotheses.

\item The operad $C_\bullet(\cD_3)$ is formal after choosing a
formality zigzag. The transferred $e_3$ structure consists of a product,
a degree $-2$ bracket, and Poisson compatibility; the shifted
$\Linf$ structure is its Lie part.

\item The hCS action \eqref{eq:hcs-action}, together with BV--BRST data
and a solved Costello QME, produces the completed local quantum
observables $\Obs^q_{\hCS}$ as a pre-factorisation algebra.

\item Compact global hCS statements require a compact BV package:
orientation and determinant-line framing, gauge fixing, heat-kernel
regularisation, counterterms, completed local functionals, factorisation
or TCFT sewing, QME solution, and vanishing or trivialisation of the quartic
anomaly class of Proposition~\ref{prop:quartic-qme}.

\item On a formal holomorphic three-disk, equality between hCS graph
weights and a named Kontsevich formality morphism is the graph-integral
comparison datum of Proposition~\ref{prop:graph-comparison}.

\item The surface--curve specialisation at $d=3$ produces the chiral
object $A_{X,\Sigma_2,C}$ with $\Eone$ structure. The centre, double, or
module-centre layer carries the $\Etwo$ structure under the relevant
hypotheses.

\item At $X=K3\times E$, the compact total-space and specialisation
invariants are the five typed values of
Proposition~\ref{prop:k3e-invariant-separation}.
\end{enumerate}
\end{theorem}

\begin{proof}
Part (1) is Propositions~\ref{prop:kt-regularity},
\ref{prop:kt-acyclicity-source}, and~\ref{prop:kt-bv-brst}. Part (2) is
Theorem~\ref{thm:D3-formality} and Proposition~\ref{prop:lie-part-e3}.
Part (3) is Propositions~\ref{prop:hcs-mc} and
\ref{prop:propagator-observables}. Part (4) is the QME and anomaly
package of Definitions~\ref{def:bv-brst} and
\ref{def:costello-quantization}, Corollary~\ref{cor:cy3-global-closure-hypotheses},
and Propositions~\ref{prop:compact-hcs-closure} and
\ref{prop:quartic-qme}. Part (5) is
Proposition~\ref{prop:graph-comparison}. Part (6) is
Proposition~\ref{prop:d3-output-level}. Part (7) is
Proposition~\ref{prop:k3e-invariant-separation}.
\end{proof}

\section{Construction Problems}
\label{sec:construction-problems}

\begin{problem}[Compact CY3 global closure]
\label{obl:compact-cy3-global}
A theorem upgrading compact CY3 Hochschild or hCS data to a global
$\Ethree$ object requires:
\begin{enumerate}
\item a TCFT, framed higher-Deligne, or equivalent factorisation
construction compatible with the Calabi--Yau trace;
\item a Costello quantisation satisfying the QME;
\item trivialisation of the quartic anomaly class in local BV cohomology;
\item orientation, determinant-line framing, gauge fixing, and regulator
data;
\item an RG-compatible counterterm scheme and compact descent or TCFT
sewing;
\item a completed observables complex and a comparison map to
categorical Hochschild cochains;
\item a filtration or vanishing statement controlling
$\HH^{-2}(\Perf(X))$;
\item HKR and Todd normalisation in the Hochschild comparison.
\end{enumerate}
\end{problem}

\begin{problem}[hCS graph weights and Kontsevich formality]
\label{obl:hcs-graphs}
A theorem identifying hCS graph weights with a specific
Kontsevich-formality morphism requires the graph complex, the
configuration-space compactification, the propagator differential forms,
boundary cancellation, orientation signs, symmetry factors, and the
volume-form normalisation.
\end{problem}

An $\Etwo$ structure attached to a $d=3$ output is placed on

dualisability, completion, and gluing hypotheses under which that centre
exists.
\end{problem}
