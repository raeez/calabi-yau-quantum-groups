The expression ``DT zeta root'' has a precise meaning in the
K3-fibred genus-two setting.  It is the reciprocal square-root
denominator \(F\) in a reduced Donaldson-Thomas scalar identity
\[
        Z^{X,\mathrm{red}}_{\DT/\PT}(Z)=C_{\mathrm{red}}F(Z)^{-2},
        \qquad C_{\mathrm{red}}\in\CC^\times .
\]
The divisor attached to \(F\) is the pole divisor of the scalar
series:
\[
        \operatorname{div}_{\infty}
        Z^{X,\mathrm{red}}_{\DT/\PT}
        =2\,\operatorname{div}(F).
\]
Zeros of \(Z^{X,\mathrm{red}}_{\DT/\PT}\) play no role in this
comparison.  The square-root sign belongs to an orientation or
automorphic character datum, while the squared scalar sees only
\(F^2\).

\subsection{Variables, Chamber, and Comparison Map}
\label{subsec:dt-square-normalisation}

Let
\[
        Z=\begin{pmatrix}\tau&z\\ z&\sigma\end{pmatrix}\in\HH_2,
        \qquad
        q=e^{2\pi i\tau},\quad r=e^{2\pi iz},\quad
        s=e^{2\pi i\sigma}.
\]
The product chamber used for the theta-normalized Igusa form is
\[
        0<|s|\ll |q|\ll |r|^{-1}\ll 1 .
\]
If a reduced DT series is written with variables \((q,t,p)\), the
comparison with Siegel variables is
\[
        (q,t,p)=(e^{2\pi i\tau},e^{2\pi iz},e^{2\pi i\sigma}).
\]
The group comparison is Igusa's exterior-square identification
\[
        \Sp_4(\ZZ)/\{\pm I\}\simeq \SO(\Lambda^{3,2})_+,
\]
which transports the genus-two Siegel domain to the orthogonal domain
of the Borcherds product.  All divisor statements below are made after
this comparison map and in the displayed chamber.

The theta-normalized cusp form \(\Delta_5\) has leading coefficient
\[
        [q^{1/2}r^{1/2}s^{1/2}]\Delta_5=64 .
\]
The monic Borcherds denominator is therefore
\[
        D_5=64^{-1}\Delta_5 .
\]
Igusa's square normalization and Oberdieck-Pandharipande's scalar
normalization are
\[
        \Phiten^{\mathrm{un}}=\Delta_5^2,\qquad
        \Phiten^{\mathrm{OP}}=D_5^2=4096^{-1}\Delta_5^2 .
\]
For \(X=S\times E\), \(S\) a K3 surface, and \(\beta\in H_2(S,\ZZ)\)
primitive, the reduced scalar branch is
\[
        Z^{S\times E,\mathrm{red}}_{\DT/\PT}
        =
        -(\Phiten^{\mathrm{OP}})^{-1}
        =
        -D_5^{-2}
        =
        -4096\,\Delta_5^{-2}.
\]
The sign is the reduced virtual class convention in the
Oberdieck-Pandharipande/Oberdieck-Pixton normalization.  A
Gromov-Witten potential enters through the reduced MNOP comparison as a
separate series with its own variables and wall convention.  A BKM
denominator identity is a further automorphic and Lie-theoretic
statement.

\subsection{The Level-One Theorem}
\label{subsec:level-one-theorem}

\begin{theorem}[Reduced \(K3\times E\) scalar and \(\Delta_5\)]
\label{thm:reduced-k3e-delta5-square}
Let \(X=S\times E\), with \(S\) a K3 surface and \(E\) an elliptic
curve.  For primitive curve class \(\beta\in H_2(S,\ZZ)\), the reduced
DT/PT scalar in the chamber of
Subsection~\ref{subsec:dt-square-normalisation} satisfies
\[
        Z^{X,\mathrm{red}}_{\DT/\PT}
        =
        -4096\,\Delta_5^{-2}.
\]
The reciprocal square-root denominator is the theta-normalized Igusa
cusp form \(\Delta_5\), while the monic denominator in the
Gritsenko-Nikulin BKM identity is \(D_5=64^{-1}\Delta_5\).  The divisor
of \(\Delta_5\) on the genus-two Siegel threefold is the diagonal
Humbert divisor \(H_1\) with multiplicity one, hence the scalar pole
divisor is \(2H_1\).
\end{theorem}

\begin{proof}
Oberdieck-Pandharipande and Oberdieck-Pixton identify the reduced
DT/PT scalar of \(K3\times E\) with
\(-(\Phiten^{\mathrm{OP}})^{-1}\) for primitive K3 class.  The
normalization
\(\Phiten^{\mathrm{OP}}=D_5^2\) gives \(-D_5^{-2}\).  The
theta-normalized and monic products are related by
\(D_5=64^{-1}\Delta_5\), so the scalar becomes
\(-4096\,\Delta_5^{-2}\).  Igusa gives
\(\Phiten^{\mathrm{un}}=\Delta_5^2\), and
Cl\'ery-Gritsenko, in the \(N=1\) row, gives
\(\operatorname{div}(\Delta_5)=H_1\) with multiplicity one.
\end{proof}

\begin{theorem}[Borcherds denominator at level one]
\label{thm:delta5-bkm-denominator}
Let
\[
        \phi_{0,1}(\tau,z)=\sum_{n,\ell} f(n,\ell)q^nr^\ell
        =r^{-1}+10+r+O(q)
\]
be the Eichler-Zagier weak Jacobi form.  In the chamber above,
\[
        D_5(Z)=q^{1/2}r^{1/2}s^{1/2}
        \prod_{(n,\ell,m)>0}
        (1-q^nr^\ell s^m)^{f(nm,\ell)} .
\]
The Gritsenko-Nikulin generalized Borcherds-Kac-Moody superalgebra
\(\fg_{\Delta_5}\) has denominator
\[
        \operatorname{den}(\fg_{\Delta_5})=D_5(2Z)
        =64^{-1}\Delta_5(2Z)
\]
and signed root supermultiplicities
\[
        \sdim(\fg_{\Delta_5,\alpha(n,\ell,m)})=f(nm,\ell).
\]
Its BKM weight invariant is
\[
        \kBKM(\fg_{\Delta_5})
        =\wt(\Delta_5)
        =\frac{c_1(0)}{2}
        =\frac{10}{2}
        =5 .
\]
\end{theorem}

\begin{proof}
Borcherds' singular theta lift gives weight \(c(0)/2\) for an
automorphic product from the constant term of the input.  The
Gritsenko-Nikulin product for \(\phi_{0,1}\) gives the displayed
monic product \(D_5\), with \(c_1(0)=f(0,0)=10\).  The same product is
the Weyl-Kac-Borcherds denominator for \(\fg_{\Delta_5}\) after the
argument \(Z\mapsto 2Z\), and its exponents are precisely the signed
superdimensions of the root spaces.
\end{proof}

\subsection{Primitive CHL Denominators}
\label{subsec:primitive-chl-denominators}

The primitive CHL denominator ladder has five compact \(K3\times E\)
BKM orders:
\[
        N\in\{1,2,3,4,6\}.
\]
With \(c_N(0)\) the constant term of the Borcherds input at the chosen
cusp, the constants and weights are
\[
\begin{array}{c|ccccc}
N & 1 & 2 & 3 & 4 & 6\\
\hline
c_N(0) & 10 & 8 & 6 & 4 & 2\\
\kBKM(\Phi_N)=c_N(0)/2 & 5 & 4 & 3 & 2 & 1
\end{array}
\]
This table is the primitive CHL denominator table.  It is indexed by
the cyclic order \(N\), by the cusp normalization of the CHL Borcherds
input, and by the corresponding denominator \(\Phi_N\).

\begin{proposition}[Primitive CHL weight formula]
\label{prop:primitive-chl-weight-formula}
For every \(N\in\{1,2,3,4,6\}\),
\[
        \kBKM(\Phi_N)=\wt(\Phi_N)=\frac{c_N(0)}{2}.
\]
The tuple of constants is
\[
        (c_1(0),c_2(0),c_3(0),c_4(0),c_6(0))
        =(10,8,6,4,2),
\]
and the BKM weights are
\[
        (\kBKM(\Phi_1),\kBKM(\Phi_2),\kBKM(\Phi_3),
        \kBKM(\Phi_4),\kBKM(\Phi_6))
        =(5,4,3,2,1).
\]
\end{proposition}

\begin{proof}
Borcherds' weight theorem gives \(c_N(0)/2\).  The CHL input constants
are the five values displayed above in the Gritsenko-Nikulin and
Scheithauer normalizations.  Substitution gives the stated weights.
\end{proof}

\subsection{The Eight Diagonal-Divisor Rows}
\label{subsec:eight-diagonal-divisor-rows}

The Gritsenko-Cl\'ery catalogue is a different object from the primitive
CHL ladder.  It classifies genus-two modular forms with character or
multiplier for Hecke-type congruence subgroups \(\Gamma_t(N)\) whose
divisor is the diagonal Humbert divisor on the corresponding
paramodular threefold with multiplicity one.  In the standard order:
\[
\begin{array}{c|c|c|c|c}
j & F_j & (t,N) & \wt(F_j) & 2\wt(F_j)\\
\hline
1 & \Delta_5 & (1,1) & 5 & 10\\
2 & \Delta_2 & (2,1) & 2 & 4\\
3 & \nabla_3 & (1,2) & 3 & 6\\
4 & \Delta_1 & (3,1) & 1 & 2\\
5 & \nabla_2 & (1,3) & 2 & 4\\
6 & \Delta_{1/2} & (4,1) & \tfrac12 & 1\\
7 & \nabla_{3/2} & (1,4) & \tfrac32 & 3\\
8 & Q_1 & (2,2) & 1 & 2
\end{array}
\]
Thus the catalogue row constants are
\[
        (10,4,6,2,4,1,3,2),
\]
whereas the primitive CHL constants are
\[
        (10,8,6,4,2).
\]
The two tables meet at \(\Delta_5\).  Every other comparison requires a
row map specifying the group, multiplier cover, branch, Borcherds
input, chamber, orientation character, and scalar host.

\begin{proposition}[Data required for a reduced DT host]
\label{prop:dt-host-data-required}
For a catalogue row \(F_j\), a reduced DT realization consists of the
following data:
\[
        (X_j,\ \beta_j,\ Z_j^{\mathrm{red}},\
        \Gamma_j,\ \nu_j,\ \mathcal C_j,\ \mathfrak o_j,\ \rho_j).
\]
Here \(X_j\) is a CY$_3$ or orbifold CY$_3$ with reduced obstruction
theory, \(\beta_j\) is the curve-class convention, \(Z_j^{\mathrm{red}}\)
is the DT/PT scalar in fixed variables, \((\Gamma_j,\nu_j)\) is the
modular group and character or multiplier of \(F_j\), \(\mathcal C_j\)
is the product chamber, \(\mathfrak o_j\) is the orientation or
Pfaffian square-root datum, and \(\rho_j\) is the comparison map from
curve-counting variables to the Siegel or orthogonal domain.  With
these data fixed, the target identity has the form
\[
        Z_j^{\mathrm{red}}(Z)=C_jF_j(Z)^{-2}.
\]
\end{proposition}

\begin{proof}
The scalar square, the automorphic denominator, and the BKM
denominator are different layers.  The equality above has a definite
meaning only after fixing the chamber and comparison map.  A square
root \(F_j\) additionally carries a character or multiplier that the
scalar \(F_j^2\) has forgotten.  The orientation datum records that
character on the curve-counting side.
\end{proof}

\subsection{Twined K3 Inputs and Root Multiplicities}
\label{subsec:twined-root-multiplicities}

Let \(g\) and \(h\) be commuting finite-order symplectic automorphisms
of a K3 surface, and let
\[
        Z_{g|h}(\tau,z)
\]
be the corresponding twisted and twined K3 elliptic genus in the
normalization fixed by the chosen sector.  When it is an index-one K3
elliptic genus, the Borcherds input used for root multiplicities is
\[
        \phi^{(g,h)}_{0,1}=\frac12 Z_{g|h}
        =\sum_{n,\ell}c^{(g,h)}(n,\ell)q^nr^\ell .
\]
For a row whose product datum is supplied, the signed
supermultiplicity formula is
\[
        \sdim(\fg_{F,\alpha(n,\ell,m)})
        =c^{(g,h)}(nm,\ell).
\]
At \(g=h=1\), this recovers
\[
        \phi_{0,1}=r^{-1}+10+r+O(q),\qquad
        c_1(0)=10,\qquad
        \kBKM(\Delta_5)=5 .
\]

For \(F_j\) beyond \(\Delta_5\), numerical root multiplicities must be
read from the row's actual Jacobi input, after the row map to
\(\Gamma_t(N)\), its multiplier cover, and its product chamber have
been specified.  Catalogue weights alone determine
\(\kBKM(F_j)=\wt(F_j)\); a reduced DT scalar host and a compact Hall
algebra require the additional data listed above.

\subsection{Four \(K3\times E\) Invariants}
\label{subsec:k3e-invariants-separated}

For \(Y=K3\times E\), the compact Hodge column is
\[
        h^{0,\bullet}(Y)=(1,1,1,1),
\]
hence
\[
        \kch(Y)=\sum_q(-1)^qh^{0,q}(Y)=0 .
\]
The categorical Euler invariant is
\[
        \kcat(Y)=\chi(\cO_{K3})\chi(\cO_E)=2\cdot0=0 .
\]
The Heisenberg-Mukai specialization, the BKM denominator, and the
K3 fibre rank are separate constructions:
\[
        \kHeis(Y)=3,\qquad
        \kBKM(\fg_{\Delta_5})=5,\qquad
        \kfib(Y)=24 .
\]
Thus the canonical \(K3\times E\) values are
\[
        \kcat(Y)=0,\quad
        \kch(Y)=0,\quad
        \kHeis(Y)=3,\quad
        \kBKM(\fg_{\Delta_5})=5,\quad
        \kfib(Y)=24.
\]
They arise from the total-space Euler characteristic, the compact
Hodge supertrace, the Heisenberg-Mukai specialization, the Borcherds
weight of \(\Delta_5\), and the Mukai-lattice rank
\[
        \operatorname{rk}\widetilde H(K3,\ZZ)=1+22+1=24 .
\]
These constructions are independent.  In particular,
\(\kBKM(\Phi_N)\) is always read as \(c_N(0)/2\) from the denominator
input.

\subsection{CY-to-Chiral Type and Positive Halves}
\label{subsec:cy-to-chiral-type}

For a CY$_3$ datum the native specialized output of the functor
\[
        \Phi_3=\mathrm{Sp}_{\Sigma_2,C}\circ\Phi^{\mathrm{FA}}_3
\]
is \(E_1\)-chiral.  Any \(E_2\)-structure belongs to a centre, double,
or module-category layer associated with that \(E_1\)-object.  Thus a
DT/Borcherds denominator can be compared with a chiral construction
only after the target layer has been named.

The local model follows the same convention:
\[
        \CoHA(\CC^3)=Y^+(\widehat{\mathfrak{gl}}_1).
\]
The full \(\mathcal W_{1+\infty}\) appears after passing to a double,
centre, completion, or evaluation layer.  A compact \(K3\times E\)
Hall-Drinfeld construction must specify which layer is being compared
with the BKM denominator.

\subsection{BKM Branch and Yangian Branch}
\label{subsec:bkm-yangian-branches}

The BKM/Hall-Drinfeld branch and the Mukai/Yangian branch have
different inputs.
\[
\begin{array}{c|c|c}
\text{branch} & \text{input} & \text{output}\\
\hline
\text{BKM/Hall-Drinfeld} &
\text{Borcherds product, Weyl chamber, reflection character} &
\fg_{\Delta_5}\text{ or }\fg_{F_j}\\
\text{Mukai/Yangian} &
\text{Mukai lattice, stable envelopes, root-of-unity evaluation} &
\text{Yangian or quantum toroidal specialization}
\end{array}
\]
Root-of-unity Yangian phenomena therefore require their own
specialization map, centre or double layer, and comparison with the
DT/PT scalar.  A CHL or Gritsenko-Cl\'ery denominator identity supplies
an automorphic BKM product; a root-of-unity Yangian theorem supplies a
different algebraic object.

\subsection{Clean DT-Side Statement}
\label{subsec:clean-dt-side-statement}

\begin{theorem}[The proved level-one DT-side identity]
\label{thm:clean-dt-side-n1}
For \(X=K3\times E\), primitive K3 curve class, the chamber
\[
        0<|s|\ll |q|\ll |r|^{-1}\ll1,
\]
and the theta-normalized Igusa form \(\Delta_5\), one has
\[
        Z^{X,\mathrm{red}}_{\DT/\PT}
        =
        -4096\,\Delta_5^{-2}.
\]
The associated BKM denominator is
\[
        \operatorname{den}(\fg_{\Delta_5})=64^{-1}\Delta_5(2Z),
\]
and
\[
        \kBKM(\fg_{\Delta_5})=5.
\]
\end{theorem}

\begin{proof}
This is Theorem~\ref{thm:reduced-k3e-delta5-square} together with
Theorem~\ref{thm:delta5-bkm-denominator}.  The first theorem is the
reduced DT/PT scalar identity in OP normalization.  The second theorem
is the Gritsenko-Nikulin BKM denominator identity in monic
normalization.  The conversion is exactly
\[
        D_5=64^{-1}\Delta_5,\qquad
        \Phiten^{\mathrm{OP}}=D_5^2,\qquad
        \Phiten^{\mathrm{un}}=\Delta_5^2 .
\]
\end{proof}

\begin{conjecture}[Catalogue-row reduced DT realization]
\label{conj:catalogue-row-dt-realization}
Let \(F_j\) be one of the seven Gritsenko-Cl\'ery rows different from
\(\Delta_5\).  A reduced DT realization of \(F_j\) consists of the data
listed in Proposition~\ref{prop:dt-host-data-required}, together with a
twisted K3 input \(\phi^{(g,h)}_{0,1}\) or its higher-index analogue.
The conjectural scalar identity is
\[
        Z_j^{\mathrm{red}}(Z)=C_jF_j(Z)^{-2},
        \qquad C_j\in\CC^\times,
\]
and the conjectural BKM root multiplicities are
\[
        \sdim(\fg_{F_j,\alpha(n,\ell,m)})
        =c^{(g,h)}(nm,\ell).
\]
The assertion is rowwise: each row fixes its own group, multiplier,
chamber, orientation, and comparison map.
\end{conjecture}

\subsection{Primary Anchors}
\label{subsec:primary-anchors}

\begin{enumerate}
\item Igusa \(1962,1964\): \(\Phiten^{\mathrm{un}}=\Delta_5^2\),
\(\wt(\Delta_5)=5\), and the Maass character \(\nu_{\Delta_5}\).
\item Gritsenko-Nikulin \(1995,1997\): the monic product
\(D_5=64^{-1}\Delta_5\), the denominator
\(\operatorname{den}(\fg_{\Delta_5})=D_5(2Z)\), and root
supermultiplicities from \(\phi_{0,1}\).
\item Borcherds \(1998\), Theorem \(13.3\): the automorphic product
weight is \(c(0)/2\).
\item Cl\'ery-Gritsenko \(2008\): the eight diagonal-divisor rows and
their multiplier systems.
\item Oberdieck-Pandharipande, Oberdieck-Pixton: the reduced
\(K3\times E\) DT/PT scalar in OP square normalization.
\item Gaberdiel-Hohenegger-Volpato and subsequent K3 twining tables:
the twined elliptic genera whose coefficients supply rowwise
Borcherds inputs after a row map is fixed.
\end{enumerate}
