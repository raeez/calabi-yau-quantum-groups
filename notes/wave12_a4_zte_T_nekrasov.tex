\title{A Gauge-Fixed Rational Matrix for the Tetrahedron Equation}
\author{Raeez Lorgat}

\begin{abstract}
The rational Yang matrix
\[
  R(z)=\frac{z\,\mathrm{Id}+\hbar_{\mathrm{Y}}P}{z+\hbar_{\mathrm{Y}}}
\]
on $\C^2\otimes \C^2$ satisfies the Yang--Baxter equation and spectral
unitarity.  The pairwise-factored face operator
$S^{\mathrm{fact}}_{ijk}=R_{ij}R_{ik}R_{jk}$ has a nonzero
Zamolodchikov tetrahedron residual of order $\hbar_{\mathrm{Y}}^2$.
On the charge-$2$ sector of $(\C^2)^{\otimes 4}$, at
$\hbar_{\mathrm{Y}}=1/5$ and $u=(0,1,3,7)$, the gauge-extended
linearised equation has rank $35$ and a $45$-dimensional affine space of
solutions.  The row-reduced gauge gives an exact rational face correction
whose charge-$2$ residual vanishes.  The same correction leaves rank-$2$
residuals on charges $1$ and $3$, so the resulting tetrahedral braiding
statement is sectoral and gauge-fixed.
\end{abstract}

\section{Yang Face Operators}

\begin{definition}[The Yang matrix and the tetrahedron residual]
Let $V=\C^2$ with ordered basis $\ket{0},\ket{1}$, let
$P(v\otimes w)=w\otimes v$, and let
\[
  R(z)=\frac{z\,\mathrm{Id}+\hbar_{\mathrm{Y}}P}{z+\hbar_{\mathrm{Y}}}
  \in \End(V\otimes V).
\]
The denominator is assumed nonzero whenever $R(z)$ is used.
For spectral parameters $u_0,u_1,u_2,u_3$, put
\[
  S^{\mathrm{fact}}_{ijk}
  =
  R_{ij}(u_i-u_j)R_{ik}(u_i-u_k)R_{jk}(u_j-u_k)
  \in \End(V^{\otimes 4}).
\]
The Zamolodchikov tetrahedron residual is
\[
  \mathcal O
  =
  S_{012}S_{013}S_{023}S_{123}
  -
  S_{123}S_{023}S_{013}S_{012}.
\]
The charge-$q$ sector of $V^{\otimes 4}$ is the span of words with
exactly $q$ entries equal to $1$.
\end{definition}

On the charge-$2$ sector the ordered basis is
\[
  \mathcal B_2=
  \bigl(
    \ket{0011},\ket{0101},\ket{0110},
    \ket{1001},\ket{1010},\ket{1100}
  \bigr).
\]
For the Yang matrix above, exact multiplication gives
$\mathcal O|_{q=2}\neq 0$, $\mathcal O^T=-\mathcal O$,
$\rk(\mathcal O|_{q=2})=4$, and
$\|\mathcal O|_{q=2}\|_\infty=O(\hbar_{\mathrm{Y}}^2)$.

\section{Parameter Normalisation}

The scalar $\hbar_{\mathrm{Y}}$ is the Yangian rational parameter in the
normalisation
\[
  R(z)=\frac{z\,\mathrm{Id}+\hbar_{\mathrm{Y}}P}{z+\hbar_{\mathrm{Y}}}.
\]
It is the variable called \texttt{kappa} in the exact arithmetic engines.
The variables used below are $\hbar_{\mathrm{Y}}$ and the four spectral
parameters $u_i$.  The $q,t$ parameters belong to the stable-envelope
comparison, root-of-unity specialisations belong to a different
integrable-system regime, and the $\kappa_\bullet$ invariants belong to
the Calabi--Yau stratification.  At the canonical point
\[
  \hbar_{\mathrm{Y}}=\frac15,\qquad u=(0,1,3,7),
\]
the six denominators $u_i-u_j+\hbar_{\mathrm{Y}}$ for $i<j$ are
\[
  -\frac45,\ -\frac{14}{5},\ -\frac{34}{5},\
  -\frac95,\ -\frac{29}{5},\ -\frac{19}{5},
\]
so every displayed rational matrix product is defined.

\section{Matrix Symmetries}

\begin{definition}[Transpose, persymmetry, reflection, bipersymmetry]
Let $J\in\End(\C^6)$ be the anti-diagonal identity,
$J_{ab}=1$ precisely when $a+b=5$, with indices
$0\leq a,b\leq 5$.
For $M\in\End(\C^6)$:
\[
\begin{array}{ll}
\text{transpose symmetry} & M^T=M,
  \quad M_{ab}=M_{ba},\\[2mm]
\text{anti-diagonal reflection invariance} & JMJ=M,
  \quad M_{ab}=M_{5-a,5-b},\\[2mm]
\text{classical persymmetry} & JMJ=M^T,
  \quad M_{ab}=M_{5-b,5-a},\\[2mm]
\text{bipersymmetry} & M^T=M \text{ and } JMJ=M.
\end{array}
\]
Bipersymmetry implies classical persymmetry.  Classical persymmetry is
strictly weaker.
\end{definition}

The involution $J$ is the particle--hole involution on
$\mathcal B_2$: it sends each two-particle word to its complement.
Anti-diagonal vanishing means
\[
  M_{a,5-a}=0,\qquad 0\leq a\leq 5.
\]
This vanishing is an additional equation on $M$; it is independent of
transpose symmetry and anti-diagonal reflection invariance.

\section{The Row-Reduced Charge-Two Matrix}

\begin{theorem}[Gauge-fixed rational correction]
\label{thm:rref-zte-correction}
Set $\hbar_{\mathrm{Y}}=1/5$ and $u=(0,1,3,7)$.  The linearised
charge-$2$ equation for charge-preserving per-face ternary corrections
$C_{ijk}\in\End(V_i\otimes V_j\otimes V_k)$ has matrix size
$36\times 80$, rank $35$, and nullity $45$.  In the row-reduced gauge,
with free variables set to zero, the cumulative charge-$2$ correction is
\[
T_{\operatorname{rref}}=
\begin{pmatrix}
-25/21 & 5/42 & 0 & 1/14 & 0 & 0\\
5/42 & -233/168 & 0 & 15/56 & 0 & 0\\
0 & 0 & -75/56 & 0 & 15/56 & 1/14\\
1/14 & 15/56 & 0 & -75/56 & 0 & 0\\
0 & 0 & 15/56 & 0 & -233/168 & 5/42\\
0 & 0 & 1/14 & 0 & 5/42 & -25/21
\end{pmatrix}.
\]
It satisfies
\[
  T_{\operatorname{rref}}^T=T_{\operatorname{rref}},
  \qquad
  JT_{\operatorname{rref}}J=T_{\operatorname{rref}},
  \qquad
  (T_{\operatorname{rref}})_{a,5-a}=0,
\]
and $\rk(T_{\operatorname{rref}})=6$.  Its entries take exactly seven
rational values:
\[
  \left\{
  -\frac{233}{168},-\frac{75}{56},-\frac{25}{21},
  0,\frac{1}{14},\frac{5}{42},\frac{15}{56}
  \right\}.
\]
For the corrected face operators
$S^{\mathrm{rref}}_{ijk}=S^{\mathrm{fact}}_{ijk}+C^{\mathrm{rref}}_{ijk}$,
the charge-$2$ tetrahedron residual is exactly zero.
\end{theorem}

\begin{proof}
The charge-preserving endomorphisms of $V^{\otimes 3}$ have dimensions
$1,9,9,1$ in charges $0,1,2,3$, hence dimension $20$ per face and
$80$ for the four faces.  Linearising the tetrahedron residual at
$S^{\mathrm{fact}}$ gives a $36\times 80$ rational system on
$\End(V^{\otimes 4}|_{q=2})$.  Exact row reduction at
$\hbar_{\mathrm{Y}}=1/5$ and $u=(0,1,3,7)$ gives rank $35$ and the
displayed representative; consistency gives a $45$-dimensional affine
solution space.  Direct rational multiplication of the four corrected
face operators gives zero in every entry of the $6\times 6$ charge-$2$
residual; this nonlinear vanishing is a separate exact check after the
linearised solve.  The equations for transpose symmetry, anti-diagonal
reflection, anti-diagonal vanishing, and full rank are entrywise checks
on the displayed matrix.
\end{proof}

\section{Minimum-Norm Representative}

\begin{proposition}[The ten-value representative]
\label{prop:minnorm-ten-values}
At the same parameters, the minimum-norm solution of the same
linearised equation gives a different rational matrix
$T_{\min}$ on the charge-$2$ sector.  It has
\[
  T_{\min}^T=T_{\min},\qquad JT_{\min}J=T_{\min},\qquad
  (T_{\min})_{a,5-a}=0,\qquad \rk(T_{\min})=6,
\]
with $30$ nonzero entries and exactly ten distinct rational values.
The nonlinear charge-$2$ residual after this first minimum-norm
correction is nonzero of order $\hbar_{\mathrm{Y}}^4$ and is strictly
smaller than the original order-$\hbar_{\mathrm{Y}}^2$ residual.
\end{proposition}

\begin{proof}
The minimum-norm representative is the unique solution in the row space
of the rank-$35$ system matrix.  Equivalently, it is the point of least
Euclidean norm in the affine solution space.  The entries are exact
rational numbers.  Direct entrywise comparison gives symmetry,
anti-diagonal reflection invariance, zero anti-diagonal, full rank, $30$
nonzero entries, and ten distinct values.  Recomputing the full product
$S_{012}S_{013}S_{023}S_{123}-S_{123}S_{023}S_{013}S_{012}$ after the
first correction gives a nonzero charge-$2$ residual whose infinity norm
is strictly smaller than the uncorrected residual and scales as
$O(\hbar_{\mathrm{Y}}^4)$.
\end{proof}

The row-reduced and minimum-norm gauges are distinct points in the same
$45$-dimensional affine solution space of the linearised equation.  A
second exact Newton step after the minimum-norm first step has full rank
$36$ on the charge-$2$ sector and kills the charge-$2$ residual exactly;
this second-step statement is restricted to charge $2$.

\section{Charge-Sector Residuals}

\begin{proposition}[Residual ranks after the row-reduced correction]
\label{prop:charge-residuals}
For $S^{\mathrm{rref}}$ at
$\hbar_{\mathrm{Y}}=1/5$ and $u=(0,1,3,7)$, the exact residual by
charge sector is
\[
\begin{array}{c|c|c|c}
q & \dim V^{\otimes 4}_{q} & \rk(\mathcal O^{\mathrm{rref}}|_q)
  & \|\mathcal O^{\mathrm{rref}}|_q\|_\infty\\
\hline
0 & 1 & 0 & 0\\
1 & 4 & 2 & 260557465/2105776536\\
2 & 6 & 0 & 0\\
3 & 4 & 2 & 260557465/2105776536\\
4 & 1 & 0 & 0
\end{array}
\]
The row-reduced correction therefore solves the tetrahedron equation on
the even charge sectors and leaves equal rank-$2$ residuals on the two
odd charge sectors.
\end{proposition}

\begin{proof}
The corrected face operators are multiplied as $16\times 16$ rational
matrices and then projected to the five charge sectors.  The charge-$0$
and charge-$4$ sectors are one-dimensional.  The charge-$2$ sector is
Theorem~\ref{thm:rref-zte-correction}.  The charge-$1$ and charge-$3$
blocks have rank $2$, and their maximum entries agree as the same
rational number by particle--hole conjugation.
\end{proof}

\section{Yang--Baxter, Unitarity, Braiding}

\begin{proposition}[Consequences of the calculation]
\label{prop:ybe-unitarity-braiding}
When the displayed denominators are nonzero, the Yang matrix satisfies
\[
  R_{12}(x)R_{13}(x+y)R_{23}(y)
  =
  R_{23}(y)R_{13}(x+y)R_{12}(x),
  \qquad
  R(z)R(-z)=\mathrm{Id}.
\]
At the same parameters the tetrahedron calculation gives the following
additional facts.
\begin{enumerate}[label=\textup{(\roman*)}]
\item The per-face ternary linearised system has rank $35$ in the
$36$ charge-$2$ constraints and gives the row-reduced solution of
Theorem~\ref{thm:rref-zte-correction}.
\item For the row-reduced representative, the finite-dimensional
pairwise projection used in the exact computation has Frobenius
norm-squared $12095/168$, and the residual three-body component has
Frobenius norm-squared $2209/72$.
\item The tetrahedral braid relation represented by the four face
operators holds exactly on charges $0,2,4$ at the displayed parameters.
The rank-$2$ residuals on charges $1,3$ obstruct a representation on
all of $V^{\otimes 4}$.
\item The proved unitarity identity in this note is the two-body identity
$R(z)R(-z)=\mathrm{Id}$ for the Yang matrix.
\end{enumerate}
\end{proposition}

\begin{proof}
The Yang--Baxter and unitarity identities are direct $4\times 4$ matrix
identities for $R(z)$, with all displayed denominators nonzero.  The
rank, residual, and norm statements are computed over $\Q$ from the
linearised ternary system and the row-reduced corrected product.  The
pairwise and three-body norm-squares use the same finite-dimensional
projection as the exact computation; stable-envelope residue
interpretations require an additional comparison map.
Proposition~\ref{prop:charge-residuals} gives the braid-relation scope.
\end{proof}

\section{Stable-Envelope Boundary}

In the fundamental two-particle Maulik--Okounkov setting, after the
scalar normalisation by $z+\hbar_{\mathrm{Y}}$, the displayed $R(z)$ is
the rational Yangian $R$-matrix attached to stable envelopes.  The
theorem proved here is the finite rational tetrahedron calculation on
$(\C^2)^{\otimes 4}$.  A comparison with an Aganagic--Okounkov elliptic
stable envelope requires an elliptic modulus, equivariant $K$-theory
parameters, a degeneration map to $\hbar_{\mathrm{Y}}$, and an
identification of the Taylor coefficient with the ternary correction
$C^{\mathrm{rref}}_{ijk}$.  The result here fixes the charge sector, the
spectral parameters, the gauge, and the residual ranks.
