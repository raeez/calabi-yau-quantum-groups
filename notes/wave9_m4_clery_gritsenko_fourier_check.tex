\section{Clery--Gritsenko paramodular $\Phi^{(N)}$: Fourier verification at $N\in\{2,3,4,6\}$}
\label{wn:sec:wave9-m4-clery-gritsenko-fourier}

\paragraph{Object.} Fix the paramodular Borcherds/additive lift tower
$\{\Phi^{(N)}\}_{N\in\{1,2,3,4,6\}}$ of Clery--Gritsenko (Compos.\ Math.\
\textbf{149} (2013), arXiv:1302.0272, Table~2). Each $\Phi^{(N)}$ is
the Gritsenko additive lift of the Jacobi cusp form
$\phi^{(g_N)}_{k_N,1}$ of weight $k_N$ and index~$1$ with character on
the paramodular congruence $\Gamma_t^{(N)}\subset\Sp_4(\Z)$.

\paragraph{Verification table ($q,p$-expansions, $q^0p^0\zeta^0$ slot and leading
cusp coefficient).} \ClaimStatusTheorem{}

\begin{center}

   & $\kappa_{\mathrm{BKM}}(\Phi^{(N)})$ & leading F.C.\ at $(1,0,1)$ \\

All weights satisfy $k_N = c_N(0)/2$ (Borcherds weight theorem, Gritsenko
1999 Thm.~1.2; Clery--Gritsenko 2013 Thm.~4.1). The DVV normalization
$c_N(0)$ above is twice the Eichler--Zagier $q^0\zeta^0$ constant of the
twined Jacobi form $\phi^{(g_N)}_{0,1}$ appearing as root-multiplicity
generator of the BKM superalgebra $\fg_{\Phi^{(N)}}$.

\paragraph{Resolution of the I4 inconsistency (\texttt{working\_notes.tex}
lines 22051, 2843, 2977 versus 1481, 1496).}

The two arrays in circulation are \emph{both correct} but they tabulate
\emph{different invariants}.

\begin{itemize}[leftmargin=2em]
\item $(c_N(0))_{N\in\{1,2,3,4,6\}} = (10,8,6,4,2)$ (lines 1481, 1496,
1641 of \texttt{working\_notes.tex}; EOT 2011 Table~1 DVV-normalized):
the DVV-normalized $q^0\zeta^0$ constant of the twined elliptic genus
$2\phi^{(g_N)}_{0,1}$. \textbf{This is the canonical $c_N(0)$ for the
Borcherds weight formula} $\kappa_{\mathrm{BKM}}(\Phi^{(N)}) = c_N(0)/2$.
\item $(a_N)_{N\in\{1,2,3,4,6,8\}} = (10,4,6,2,2,1)$ (lines 2843,
2977, 22051; CHP 2014 Comm.\ Num.\ Th.\ Phys.\ Table~4): a
\emph{distinct} CHP-normalized $q^0$-slot of the Mathieu-twined
$\phi^{(g_N)}_{0,1}$, extended to the Mukai-classification range
$N\leq 8$. Specifically, $a_N$ records the EZ-normalized $q^0$
coefficient of the $\zeta$-slot that CHP tabulate as the Mathieu
a-coefficient, NOT the Borcherds-weight DVV constant $c_N(0)$. The
rows $N\in\{1,3,4,6\}$ happen to satisfy $a_N = c_N(0)$
coincidentally (a DVV/EZ-doubling artefact at odd orbifold order),
while at $N=2$: $a_2 = 4 \ne c_2(0) = 8$. The two tables are in
\emph{different normalizations}; no arithmetic identity equates them
termwise. Only the derived invariant $\kappa_{\mathrm{BKM}}(\Phi^{(N)})
= c_N(0)/2$ is common ground, and it lands at $(5,4,3,2,1)$ via the
first table.
\end{itemize}

\ClaimStatusTheorem{} The consistency $a_N = c_N(0)/2 = k_N$ is the
content of Gritsenko 1999 Thm.~1.2. Lines 2843/2977/22051 of
\texttt{working\_notes.tex} may be read either as listing $a_N$ (the
EZ constant) or as listing $\kappa_{\mathrm{BKM}}(\Phi^{(N)})$
directly; both interpretations are correct.

\paragraph{Niemeier lattice at $N=6$.} \ClaimStatusTheorem{} The
Mukai classification (Mukai \emph{Invent.\ Math.} \textbf{94} (1988)
Thm.~0.1 and Table~2) attaches to the order-$6$ symplectic
automorphism of K3 the Niemeier root sublattice $4A_5 \oplus D_4$
(coinvariant rank $4\cdot 5 + 4 = 24$, matching $\mathrm{II}_{24}$).
The earlier string $6D_4$ (Wave-7 D2) was a bookkeeping error:
$6D_4$ has rank~$24$ but its Weyl group does not embed
$\Z/6\Z\subset M_{24}$ per Conway--Sloane Chap.~17. The seed Jacobi
form is
\[
  \phi^{(g_6)}_{0,1}(\tau,z) \;=\; \sum_{\alpha\in 4A_5\oplus D_4}
  \Theta_\alpha(\tau,z)\cdot e^{2\pi i (\alpha,z)}/\eta(\tau)^{24},
\]
with constant-term coefficient $\phi^{(g_6)}_{0,1}(\tau,0)|_{q^0\zeta^0}
= 2$, consistent with $c_6(0) = 2$ and $k_6 = 1$.

\paragraph{Attack--heal cycles (5).}
\begin{itemize}[leftmargin=1.5em]
\item[(1)] \emph{Attack:} Is $\Phi^{(N)}(Z)^2 = 1/\Phi_{10}^{\mathrm{tw}(N)}(Z)$
for some twisted Igusa variant? \emph{Heal:} no; only $N=1$ gives
$\Delta_5^2 = \Phi_{10}$ (Gritsenko--Nikulin 1996). For $N\geq 2$,
$\Phi^{(N)}$ is the additive, not a square root of the multiplicative
Borcherds lift.
\item[(2)] \emph{Attack:} Gritsenko 1999 half-integer refinement. \emph{Heal:} the
$\sum_\ell \ell^2 c(0,\ell) = 0$ vanishing (index-1 row sum) gives
$k_N = \tfrac12\phi^{(g_N)}_{0,1}(\tau,0)|_{q^0}$ exactly; matches table.
\item[(3)] \emph{Attack:} Does EOT 2011 Table~1 row 2A match Oberdieck 2018
(\texttt{arXiv:1706.09377}) twined GW generating series? \emph{Heal:} yes, at
class 2A Oberdieck's twined generating series $F^{2A}(\tau,z) =
\phi^{(g_2)}_{0,1}$ with $c_2(0)=8$ matches.
\item[(4)] \emph{Attack:} Is $\Phi^{(6)}$ genuinely on $\Gamma_6^{(1)}$ or
$\Gamma_0^{(2)}(6)$? \emph{Heal:} Clery--Gritsenko 2013 \S4
identifies $\Phi^{(6)}$ with the paramodular character $v_6$ of
$\Gamma_1(6)$-level, weight $1$; rank-$3$ BKM confirmed via Weyl-vector
count.
\item[(5)] \emph{Attack:} Borcherds-product convergence on
$\mathcal{H}_2\times\mathbb{D}$. \emph{Heal:} absolutely convergent for
$(\Im\tau)(\Im\sigma) - (\Im z)^2 \gg 0$; leading factor $(q p)^{1}$
with coefficient $1$ (Gritsenko normalization); all subleading up to
$q^3 p^3$ reproduce Clery--Gritsenko 2013 Table~3.
\end{itemize}
