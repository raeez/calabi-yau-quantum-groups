\begin{abstract}
The correct descent input for perturbative hCS on
$X=K3\times E$ is a Costello--Gwilliam factorisation-cosheaf statement
on a Dolbeault/Weiss/Ran basis. It is not a finite Cech sheaf statement
on an ordinary polydisk atlas. This note records the repaired core:
finite coordinate polydisk atlases exist and seed local coordinates;
Weiss descent is taken over finite disjoint relatively compact
polydisk families in the Ran direction; the maps are covariant
compact-support extension maps; the descent object is a homotopy
colimit; harmonic zero modes remain global; and the comparison between
the global $K3\times E$ propagator and local $\C^3$ propagators
requires RG homotopy, counterterms, and overlap compatibility. The
atlas/Weiss result is only an hCS-side input. It does not construct the
hCS-to-Hall comparison.
\end{abstract}

\section{The DWR/Ran Basis}
\label{sec:dwr-ran-basis}

\begin{definition}[DWR/Ran polydisk basis]
\label{def:dwr-ran-basis}
\ClaimStatusConstruction{}
Let $U\subset X$ be open. The DWR/Ran polydisk basis
$\cB_{\mathrm{DWR}}(U)$ has as objects finite pairwise disjoint
families
\[
 \sigma=(S,\{P_s\Subset U\}_{s\in S})
\]
of relatively compact holomorphic coordinate polydisks. Its associated
open is
\[
 |\sigma|=\bigsqcup_{s\in S}P_s.
\]
Morphisms are refinements by inclusions $P'_t\Subset P_s$ over maps of
finite sets $t\mapsto s$, with Cech face maps supplied by intersections
with a chosen ordinary cover and Ran multiplication supplied by
disjoint union.
\end{definition}

\begin{lemma}[The basis is Weiss]
\label{lem:dwr-basis-weiss}
\ClaimStatusProvedHere{}
The opens $|\sigma|$ with $\sigma\in\cB_{\mathrm{DWR}}(U)$ form a Weiss
cover of $U$.
\end{lemma}

\begin{proof}
For a finite subset $F\subset U$, choose pairwise disjoint coordinate
polydisks $P_x\Subset U$ around the distinct points $x\in F$. The
finite family $\{P_x\}_{x\in F}$ is an object of
$\cB_{\mathrm{DWR}}(U)$ and its associated open contains $F$.
\end{proof}

\begin{remark}[Why finite ordinary subcovers do not solve Weiss descent]
\label{rem:finite-subcovers-not-weiss}
A finite ordinary subcover of $U$ need not be Weiss: a finite subset of
$U$ need not lie in a single member of the subcover. Enlarging a finite
atlas by all finite unions either includes $U$ itself, making descent
tautological, or assigns values on unions by the descent statement one
is trying to prove. That construction is not a polydisk basis and is
not a proof of Weiss descent.
\end{remark}

\section{Cosheaf Variance}
\label{sec:cosheaf-variance}

\begin{definition}[hCS observables on DWR/Ran objects]
\label{def:hcs-observables-dwr}
\ClaimStatusConstruction{}
For a DWR/Ran object $\sigma=(S,\{P_s\})$, put
\[
 \Obs_{\hCS}^{q}(\sigma,\frakg)
 :=
 \widehat{\bigotimes}_{s\in S}
 \Obs_{\hCS}^{q}(P_s,\frakg),
\]
where each factor is the Costello--Gwilliam renormalised BV complex
\[
 \Obs_{\hCS}^{q}(P,\frakg)
 =
 \left(
 \mathcal O_{\mathrm{ren,loc/multiloc}}
 (\Omega_c^{0,\bullet}(P,\frakg)[1])[[\hbar]],
 Q_{\hCS}+\{I[L],-\}_{\mathrm{BV}}+\hbar\Delta_L
 \right).
\]
For a refinement $\rho:\sigma'\to\sigma$, the source-side map
\[
 \rho_*:\Obs_{\hCS}^{q}(\sigma',\frakg)
 \longrightarrow
 \Obs_{\hCS}^{q}(\sigma,\frakg)
\]
is induced by compact-support extension of fields/test insertions,
followed by the chosen RG transport between scales. For disjoint
families, the product is the completed tensor product followed by
extension of supports into the larger DWR/Ran open.
\end{definition}

\begin{proposition}[Correct Weiss-cosheaf descent shape]
\label{prop:correct-weiss-cosheaf-shape}
\ClaimStatusConditional{}
Assume an anomaly-cancelled Costello--Gwilliam/Costello--Li hCS quantum
package on $X$ whose compact-support extension maps, effective
interactions, BV Laplacians, and counterterms are compatible on all
DWR/Ran refinements. Then the factorisation descent statement on an
open $U\subset X$ has the form
\[
 \hocolim_{\sigma\in\cB_{\mathrm{DWR}}(U)}
 \Obs_{\hCS}^{q}(\sigma,\frakg)
 \xrightarrow{\;\simeq\;}
 \Obs_{\hCS}^{q}(U,\frakg).
\]
The indexing category is covariant, and the displayed object is a
homotopy colimit. It is not a homotopy limit over restriction maps.
\end{proposition}

\begin{proof}
Costello--Gwilliam factorisation algebras are Weiss cosheaves. In the
compact-support presentation, inclusions are implemented by extension
of supports; disjoint unions give the factorisation product. Lemma
\ref{lem:dwr-basis-weiss} supplies a Weiss cover of $U$ by DWR/Ran
objects. Applying the Weiss-cosheaf axiom to this cover gives the
displayed homotopy colimit. The analytic content is not the existence
of a finite atlas; it is the assumed compatible quantum hCS package on
all refinements.
\end{proof}

\section{Harmonic Zero Modes Are Global}
\label{sec:harmonic-global}

\begin{lemma}[No local compact-support harmonic functor]
\label{lem:no-local-harmonic-functor}
\ClaimStatusProvedHere{}
Let $P\Subset X$ be a proper coordinate polydisk. The global harmonic
projector
\[
 P_{\cH}^{X}:\Omega^{0,\bullet}(X,\frakg)\to
 \cH^{0,\bullet}(X,\frakg)
\]
does not define an endomorphism of
$\Omega_c^{0,\bullet}(P,\frakg)$ and therefore does not give a
functorial decomposition of local compact-support hCS fields into
harmonic and off-harmonic summands.
\end{lemma}

\begin{proof}
Extend $\alpha\in\Omega_c^{0,\bullet}(P,\frakg)$ by zero to $X$ and
apply $P_{\cH}^{X}$. The result is a global harmonic form. A harmonic
form on the connected compact complex manifold $X$ that is supported in
the proper open $P$ is zero by unique continuation. For a general
$\alpha$ the projection is not zero and is not compactly supported in
$P$. Thus $P_{\cH}^{X}$ does not preserve local compact support.
Restriction to smaller polydisks cannot commute with this projection.
\end{proof}

\begin{corollary}[Use of zero modes]
\label{cor:zero-modes-global-only}
\ClaimStatusVerified{}
The harmonic/off-harmonic split for $K3\times E$ is a global Hodge
decomposition used in $\Obs_{\hCS}^{q}(X,\frakg)$ and in the residual
finite-dimensional BV integral. It is not a local DWR/Ran cosheaf
splitting. Local descent must be carried by the full compact-support
renormalised BV complex and its extension maps.
\end{corollary}

\section{Propagator and Counterterm Compatibility}
\label{sec:propagator-compatibility}

\begin{proposition}[Local flat propagator comparison is extra data]
\label{prop:propagator-comparison-extra-data}
\ClaimStatusOpen{}
For a coordinate polydisk $P\subset X$, the restricted global
propagator $P_X|_{P\times P}$ and the flat Costello--Li propagator on
the coordinate model $\C^3$ have the same local diagonal singularity,
but equality of the resulting quantum hCS factorisation algebras
requires an RG homotopy, counterterm matching, and compatibility on
overlaps and DWR/Ran refinements.
\end{proposition}

\begin{proof}
The heat-kernel asymptotic on a K\"ahler threefold gives the same
order-$r^{-4}$ singularity near the diagonal as the flat model. The
difference between two choices of parametrix is smooth only after the
singular germ is matched. In perturbative BV quantisation, changing the
propagator by such a smooth kernel changes the effective interaction by
renormalisation-group flow and possible local counterterms. On a cover,
these changes must satisfy Cech/Ran compatibility on pairwise and
higher overlaps. The finite coordinate atlas supplies charts; it does
not by itself supply this compatible RG homotopy.
\end{proof}

\section{Calabi--Yau Coordinates}
\label{sec:cy-coordinates}

\begin{lemma}[Volume form does not force volume-preserving transitions]
\label{lem:volume-transitions}
\ClaimStatusVerified{}
If $z_\alpha$ and $z_\beta$ are arbitrary holomorphic coordinate
systems on overlapping polydisks and
\[
 \Omega_X=c_\alpha(z_\alpha)\,dz_{\alpha,1}dz_{\alpha,2}dz_{\alpha,3}
 =c_\beta(z_\beta)\,dz_{\beta,1}dz_{\beta,2}dz_{\beta,3},
\]
then
\[
 \det\!\left(\frac{\partial z_\beta}{\partial z_\alpha}\right)
 =\frac{c_\alpha}{c_\beta}.
\]
Thus the global Calabi--Yau form trivialises $K_X$ but does not make
arbitrary coordinate transitions volume-preserving.
\end{lemma}

\section{Scope: hCS Descent Is Not hCS-to-Hall}
\label{sec:scope}

\begin{proposition}[Source-side nature of this repair]
\label{prop:source-side-only}
\ClaimStatusVerified{}
The DWR/Ran Weiss descent statement above constructs, or assumes after
renormalisation, the hCS source as a factorisation cosheaf. It does not
construct an oriented hCS-to-Hall comparison. The latter is the separate
problem of producing a continuous natural transformation on the whole
DWR/Ran nerve
\[
 \Theta_{\hCS\to\Hall}^{\mathrm{or}}:
 \Obs_{\hCS}^{q}(-,\frakg)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(-)
\]
with orientation data, shifts, Tate twists, completions,
Thom--Sebastiani coherences, and vertexwise quasi-isomorphism
certificates.
\end{proposition}

\begin{proof}
The source descent theorem only identifies how the hCS observables glue
as a factorisation algebra. The Hall target requires critical CoHA
charts and orientation systems, and a comparison requires maps
compatible with all Cech/Ran faces and factorisation products. This is
exactly the typed datum isolated in
\texttt{chapters/theory/cy3\_chain\_level\_bridge.tex}; no atlas or
Weiss argument supplies it.
\end{proof}
