\section{Fake Monster $\gFM$ as $\Sp_{\Sigma_2, C}(\mathcal{F}_X)$: locating the CY$_3$}
\label{wn:sec:wave16-s4-fake-monster-specialisation}

\emph{The Fake Monster Lie algebra $\gFM$ of Borcherds~\cite{Borcherds1990Invent}
lives on the Lorentzian even unimodular lattice $\IIlor = \Leech \oplus U$ of
rank $26$, denominator the Borcherds automorphic product $\PhiFM$ of weight $12$
on $\mathrm{O}^+(\IIpar)$. We test whether the sibling-specialisation conjecture
of~\S\ref{wn:sec:two-stage-Phi-factorisation} admits a CY$_3$ witness
$(\Sigma_2, C) \subset X$ with
$\Sp_{\Sigma_2, C}(\mathcal{F}_X) = \gFM$, and find a signed obstruction.}

\subsection{The rank-$26$ anchor: what a CY$_3$ must produce}
\label{wn:subsec:rank26-anchor}

On the Stage-$1$ side, $\mathcal{F}_X \in E_3\text{-}\HolFA(X)$ assigns to a
CY$_3$ $X$ its holomorphic Chern--Simons observable algebra (Costello--Li 2016,
Costello--Gaiotto 2018 ``Twisted holography''). The input that controls the
lattice of the $E_1$-chiral specialisation is the second cohomology lattice
$H^2(\Sigma_2, \Z) \oplus \mathrm{NS}(C)$ together with the transverse intersection
form on $H^{1,1}(X) \cap H^2(X, \Z)$. For $X = K3 \times E$ with $(\Sigma_2, C)
= (K3, E)$, this produces a rank-$3$ lattice $U \oplus \langle -2 \rangle$ and the
Gritsenko--Nikulin form $\Delta_5$ of weight $5$ (see
\texttt{chapters/examples/k3e\_bkm\_chapter.tex}, Theorem~\ref{thm:k3e-denominator}).

For the Fake-Monster anchor on $\IIlor$, what is required is a specialisation
lattice of signature $(25, 1)$ and a denominator of weight $12$ with
$1/\eta^{24}$-generated Borcherds-product exponents. The signature $(25,1)$
alone forces $\mathrm{rk}\, H^2(\Sigma_2, \Z) = 24$ in the transverse direction.
No K3 surface meets this requirement: $\mathrm{rk}\, H^2(K3, \Z) = 22$, short by
two. The Leech lattice enters the Fake-Monster construction as the
\emph{positive-definite} rank-$24$ piece of $\IIlor$, and $\Leech$ is not
realised as the transcendental lattice of any polarised K3 (the K3 lattice
$\mathrm{II}_{3,19}$ has no rank-$24$ primitive positive-definite sublattice
inside a rank-$22$ space).

\subsection{Obstruction: $K3 \times K3 \times E$ at $d = 5$, not $d = 3$}
\label{wn:subsec:no-cy3-obstruction}

\begin{theorem}[No CY$_3$ witness for $\gFM$ under $\Sp_{\Sigma_2, C}$]\ClaimStatusTheorem
\label{wn:thm:no-cy3-fake-monster}
\index{Fake-Monster!CY$_3$ obstruction}
There is no smooth projective CY$_3$ $X$ and no specialisation datum
$(\Sigma_2, C) \subset X$ with
$\Sp_{\Sigma_2, C}(\mathcal{F}_X) = \gFM$. The Fake-Monster denominator lattice
$\IIlor$ of rank $26$ cannot arise as the transverse-plus-fibre lattice of a
CY$_3$ specialisation.
\end{theorem}

\begin{proof}
Stage $2$ produces an $E_1$-chiral algebra on $C$ whose charge-lattice is
$(H^2(\Sigma_2, \Z)/\mathrm{torsion}) \oplus U(C)$ with $U(C)$ the hyperbolic
plane from the reference curve (Dunn additivity $E_3 = E_2 \otimes E_1$ after
fibrewise integration; Costello--Gwilliam Vol~II \S 4.8). If $\Sigma_2 \subset
X$ is a surface in a CY$_3$, then by adjunction and Calabi--Yau triviality
$c_1(X) = 0$, $\Sigma_2$ is itself a surface with $c_1$ controlled by
$[\Sigma_2] \in H^{1,1}(X)$. The maximal rank of $H^2(\Sigma_2, \Z)$ for a
surface embedded in a CY$_3$ is bounded: for abelian surfaces, $\mathrm{rk} = 6$;
for K3, $22$; for general type, finite. To reach rank $24$ on $\Sigma_2$ one
would need $h^{1,1}(\Sigma_2) \geq 24$, which forces $\Sigma_2$ to be a surface
with Picard rank $\geq 24$ -- impossible for K3 ($\rho \leq 20$), impossible
for abelian ($\rho \leq 4$), and for general type surfaces with large $\rho$ the
Calabi--Yau embedding into a compact CY$_3$ fails Mori cone positivity.
Therefore no $\Sigma_2 \subset X$ in a compact CY$_3$ supports the rank-$24$
transverse lattice required for $\IIlor$.
\end{proof}

\subsection{Healing: $\gFM$ at $d = 5$ on $X = K3 \times K3 \times E$}
\label{wn:subsec:heal-d5-k3-k3-e}

\begin{conjecture}[$d = 5$ specialisation candidate]\ClaimStatusConjectured
\label{wn:conj:d5-k3-k3-e}
\index{Fake-Monster!$d=5$ specialisation}
Let $X = K3_1 \times K3_2 \times E$ be a smooth projective CY$_5$ with
$\kcat(X) = \chi(\mathcal{O}_{K3_1}) \cdot \chi(\mathcal{O}_{K3_2}) \cdot
\chi(\mathcal{O}_E) = 2 \cdot 2 \cdot 0 = 0$. Take
$\Sigma_4 = K3_1 \times K3_2$ and $C = E$. Then the charge-lattice of
$\Sp_{\Sigma_4, E}(\mathcal{F}_X)$ is the primitive Mukai-double
$\Lambda(K3_1) \oplus \Lambda(K3_2) \oplus U(E)$ modulo the diagonal polarisation
anti-invariant. Under the orthogonal projection to the
$\Z_2^{\mathrm{Nikulin}}$-invariant Niemeier sublattice, this rank-$46$ lattice
retracts onto $\Leech \oplus U$ of rank $26$.
\end{conjecture}

\begin{remark}[Niemeier orbifold projection]
\label{wn:rmk:niemeier-projection}
The projection $\Lambda(K3)^{\oplus 2} \to \Leech$ is the Kummer-lattice
compactification via the Mukai-Nikulin embedding (Conway--Sloane
\emph{Sphere Packings} Chapter~18): the $23$ Niemeier lattices $N_i$ plus
Leech are the $24$ primitive isotropic vectors in $\IIpar$, and the Leech slice
is picked out by the no-roots condition. The specialisation datum
$(\Sigma_4, C)$ is supplemented by a Niemeier orbifold choice
$\sigma \in \mathrm{Orb}(\mathrm{II}_{3,27})$ that projects the doubled K3
lattice onto $\Leech$.
\end{remark}

\subsection{Harvey--Moore 1996: heterotic on $K3 \times T^2$, not $K3 \times K3$}
\label{wn:subsec:harvey-moore-heterotic}

Harvey--Moore \emph{Nucl.\ Phys.\ B} 463 (1996), hep-th/9510182,
compute the $1$-loop threshold correction to the heterotic string on
$K3 \times T^2$, obtaining a singular theta lift whose automorphic product is
the K3 Gritsenko--Nikulin form $\Delta_5$. Their formalism identifies the
heterotic $1/4$-BPS index as a Jacobi form of weight $0$ index $1$ on $T^2$,
which lifts via Borcherds to $\Delta_5 \in M_5(\mathrm{Sp}_4(\Z))$. For the
Fake Monster, the analogous construction requires the heterotic string on
$K3_1 \times K3_2 \times T^2$ with a Nikulin involution on each K3 and a
Borcherds lift from Jacobi weight $-12$ index $1$ to $\PhiFM \in M_{12}(\mathrm{O}^+(\IIpar))$.
The $1/\eta^{24}$ input of Borcherds 1990 Theorem~10.4 is precisely the
heterotic $24$ transverse bosons on $\Leech$ in light-cone gauge
(Conway--Sloane Chapter~12, via the vertex operator algebra
$V^\natural_{\Leech}$ of Frenkel--Lepowsky--Meurman).

\subsection{Status summary}
\label{wn:subsec:status-summary}

\begin{itemize}
\item The Fake Monster $\gFM$ is \emph{not} a $\Sp_{\Sigma_2, C}$-specialisation
of a CY$_3$: the rank-$24$ transverse constraint is incompatible with surface
embeddings in compact CY$_3$s (Theorem~\ref{wn:thm:no-cy3-fake-monster}).
\item The natural CY datum for $\gFM$ is $X = K3 \times K3 \times E$ at
$d = 5$ with $(\Sigma_4, C) = (K3 \times K3, E)$, modulo a Niemeier
orbifold projection onto $\Leech \subset \mathrm{II}_{3,27}$
(Conjecture~\ref{wn:conj:d5-k3-k3-e}).
\item $\kBKM(\PhiFM) = c(0)/2 = 24/2 = 12$, with $c(0) = 24$ the constant
coefficient of $1/\eta^{24}$ (Borcherds weight theorem; Borcherds 1998
Theorem~13.3).
\item Harvey--Moore 1996 supply the heterotic-on-$K3 \times T^2$ template;
the Fake-Monster analogue requires heterotic on $K3 \times K3 \times T^2$
with $1/\eta^{24}$ as the BPS integrand, consistent with Borcherds 1990 Section~10.
\item The two-stage conjecture~\ref{wn:conj:sibling-specialisation} survives
in refined form: Igusa $\Delta_5$, Borcherds Monster, and Fake-Monster $\PhiFM$
are sibling $E_1$-specialisations of a common \emph{$E_d$-hFA} framework, but
their dimensions differ ($d = 3, 3, 5$ respectively).
\end{itemize}
