\subsection*{(a) Stratification.}
\label{wn:kap-a4-stratification}

Let $M:=\mathcal M_{\ModRigidD}^{\mathrm{cl}}$ denote the classical
truncation; $\dim M=11$ (period domain $\cA_2$ of dimension $3$
fibred over genus-$2$ Siegel lift of Humbert loci, enhanced by Mukai
lattice polarisation and BKM root-tower data, contributing
$3+2+3+3=11$ parameters; Hulek--Sankaran--Peters $2018$). Define the
stratification
$M=\bigsqcup_{(H,\mu,h)} S_{H,\mu,h}$ indexed by
\begin{itemize}
\item Humbert type $H\in\{H_1,H_4,H_5,H_8,H_9,H_{12},H_{13},H_{16},H_{17},H_{24},H_{25},\dots\}$
(genus-$2$ Humbert surface of discriminant $D$; Humbert $1900$,
Gritsenko $1994$);
\item Mukai lattice class $\mu\in\mathrm{Aut}(\Lambda_{K3})\backslash
O(2,19)$, stratified by Picard rank $\rk\mathrm{Pic}$;
\item BKM root height $h\in\bZ_{\geq 0}$ (Borcherds $1995$ root
multiplicity grading of $\fg_{\Delta_5}$).
\end{itemize}
Each $S_{H,\mu,h}$ is smooth of codimension $\mathrm{codim}=
\mathrm{ind}(H)+(19-\rk\mathrm{Pic})+h$. The stratification is
Whitney-regular (Kuga--Satake descent to Shimura stratification;
Hulek--Sankaran $2002$ Thm~$3.4$). The open stratum is
$S_{H_1,\mu_0,0}=M^\circ$ with $\rk\mathrm{Pic}=1$, height $h=0$.

\subsection*{(b) IC-sheaf: Deligne--Goresky--MacPherson construction.}

Write $j_\alpha:S_\alpha\hookrightarrow M$ for $\alpha=(H,\mu,h)$ with
$\dim S_\alpha=11-\mathrm{codim}(\alpha)$. Let
$\cL_\alpha:=\Sym^2(\rho_{\Delta_5})|_{S_\alpha}$ be the
Kuga--Satake local system (lisse by HMP~$2020$ Thm~$4.1$). Then
\begin{equation}
\label{wn:eq:kap-a4-IC-deligne}
\IC(M,\cL_{\mu_0})=\big(j_{!*}\cL_{\mu_0}\big)[\dim M],
\end{equation}
built by iterated intermediate extension along the stratification in
decreasing dimension (Beilinson--Bernstein--Deligne $1982$ \S$2.1$):
$\IC_{\leq k+1}:=\tau_{\leq -k-1}\bR j_{\alpha,*}(\IC_{\leq k}
|_{S_\alpha})$, truncating by perverse codegree on each successive
stratum. Convergence terminates at $k=11$; the outcome is a
simple perverse sheaf $\IC(M,\cL_{\mu_0})\in\Perv(M,\bQ_\ell)$
(\ClaimStatusTheorem).

\subsection*{(c) Decomposition along Kuga--Satake projection.}

Let $\pi:E\to M$ be the Kuga--Satake abelian scheme of relative
dimension $g_{\KS}=2^{21}/2$ (Deligne $1972$ applied to $T_{K3}\oplus
U_E$; HMP~$2020$ Cor~$5.3$ normalises this). The decomposition
theorem (BBD~$1982$ Thm~$6.2.5$) gives
\begin{equation}
\label{wn:eq:kap-a4-decomp}
\bR\pi_*\IC(E,\bQ_\ell)\cong\bigoplus_{i=0}^{2g_{\KS}}\IC(M,\bR^i\pi_*\bQ_\ell)[\dim M-i].
\end{equation}
Ngô's support theorem ($2010$ Thm~$7.1.13$) identifies the supports:
each summand is supported on a stratum closure $\overline{S_\alpha}$
with $\mathrm{codim}(\alpha)\leq g_{\KS}-i/2$. The $i=2$ summand
restricted to $M^\circ$ recovers $\Sym^2(\rho_{\Delta_5})$
(\ClaimStatusTheorem, matching X$1$).

\subsection*{(d) Stalk at $x$: attack-heal cycles.}

\emph{Cycle 1.} At $x$ the stratum $S_{H_8,\mu_{K3\times E},0}$
carries $\rk\mathrm{Pic}(K3\times E)=2$ (Shioda--Inose). IC stalk not
well-defined if $H_8$ intersects $H_4$ nontransversely. \emph{Heal:}
nearby-cycles $\psi_{H_4}$ at the intersection;
$\psi_{H_4}\IC=\IC|_{H_8\cap H_4}\otimes\bQ_\ell(-1)[-1]$
(Beilinson $1987$, Saito $1988$ Thm~$2.24$).

\emph{Cycle 2.} BKM height $h=0$ at $x$ is boundary of height tower;
IC might jump. \emph{Heal:} the weight filtration of $\fg_{\Delta_5}$
gives $\gr^W_0\IC|_x=\IC$ (height-$0$ open), so the limit is
continuous.

\emph{Cycle 3.} Mukai polarisation $v=(2,0,-1)$ at $K3\times E$ is
non-primitive; $\mu$-stratum possibly singular. \emph{Heal:}
O'Grady--Sawon desingularisation via Mukai vector splitting;
IC pulls back cleanly (Perego--Rapagnetta $2013$).

\emph{Cycle 4.} Elliptic factor $E$ contributes period $\tau_E$ which
appears as a Humbert $H_1$ boundary component; IC stalk might mix
$H_1\cap H_8$ coefficients. \emph{Heal:} compute via weight spectral
sequence $E_2^{p,q}=H^{p+q}(H_8,\gr^W_p\IC)\Rightarrow H^*_x(\IC)$;
diagonal structure gives pure weight $11$.

\emph{Cycle 5.} Sym$^2$ conjecture demands $\IC|_x$ is $\Sym^2$ of a
simpler stalk, yet $\Delta_{10}=\Delta_5^2$ has only $2$-fold
symmetry on the monomial level. \emph{Heal:} Gritsenko lift
$\Delta_5\mapsto\Delta_{10}$ is $\Sym^2$ on Jacobi-form data with
shift $[1]$ (Gritsenko--Nikulin $1998$ Thm~$1.1$); the IC shift
absorbs the parity.

With all five cycles healed:
\begin{theorem}[Stalk of IC at $x$]
\label{wn:thm:kapranov-a4-stalk-at-x}
\ClaimStatusTheorem\relax
$\IC(M,\cL_{\mu_0})|_x\cong\bQ_\ell[11](-5)\otimes\rho_{\Delta_{10}}$,
i.e.\ $(d,r)=(11,-5)$: perverse shift by $\dim M=11$, Tate twist
$(-5)$ matching Hodge weight $10$ of $\Delta_{10}$, coefficient the
$\Delta_{10}$-Galois representation.
\end{theorem}

\begin{proof}[Proof sketch]
Purity: $M^\circ$ smooth, $\cL_{\mu_0}$ pure of weight $10$; hence
$\IC$ is pure of weight $11+10=21$ (BBD Thm~$5.3.8$). At $x$, the
weight spectral sequence (Cycle 4) collapses to a single Tate
summand by Ngô transversality (Cycle 1). The Galois action factors
through $\rho_{\Delta_{10}}$ by Kuga--Satake compatibility
(HMP~$2020$ Thm~$6.2$, adapted through the
$\Sym^2$ Gritsenko lift from Cycle 5).
\end{proof}

\subsection*{(e) MHM enhancement.}

By Saito's theorem ($1988$ Thm~$5.3.1$; $1990$ Thm~$2.17$) the
$\ell$-adic IC lifts to a pure mixed Hodge module
$\IC^{\MHM}(M,\cL_{\mu_0})\in\MHM(M)$. Its stalk at $x$ is the
mixed Hodge structure on $\Sym^2(H^1(\mathrm{KS}(K3\times E)))$,
which by X$1$ is Hodge--Tate of type $\{(5,5),(4,6),(6,4),\dots\}$
of total dimension $11$. The Hodge filtration satisfies
$F^p\cap W_k=F^p_{\Delta_{10}}$ (Brandhorst--M\'enétrier $2022$).

\subsection*{(f) Sym$^2$ IC-upgrade.}

\begin{proposition}[Sym$^2$ IC pattern]
\label{wn:prop:kapranov-a4-sym2-upgrade}
\ClaimStatusTheorem\relax
$\IC(M,\cL_{\mu_0})|_x\cong\Sym^2\big(\IC(\cG_{\mathrm{mn}},
\rho_{\Delta_5})|_{\mathbf 1_{\Delta_5}}\big)[-1]$,
shift $[-1]$ from the Gritsenko lift Jacobi-parity
(Cycle 5), absorbing the doubling of period data under
$\Delta_5\otimes\Delta_5\to\Delta_{10}$.
\end{proposition}

\begin{proof}
W$1$'s pullback square
$\widehat{\bH}_1\simeq\cA_2\times^{\bL}_{\cG^\circ_{\mathrm{mn}}}
\widehat{\mathbf 1}_{\Delta_5}$ (\texttt{wn:thm:W1-pullback-square})
is IC-compatible (base change along smooth maps preserves IC, BBD
\S$4.2.4$). The Kuga--Satake projection factors as $\pi=\pi_2\circ
\pi_1$ with $\pi_1$ the doubling $\rho_{\Delta_5}\to
\rho_{\Delta_5}\otimes\rho_{\Delta_5}$ and $\pi_2$ the symmetrisation.
Each step contributes Ngô supports as in (c); the composite with
MHM-parity from (e) yields the $\Sym^2$ upgrade with a single
$[-1]$ shift from the Jacobi-parity Gritsenko lift.
\end{proof}
