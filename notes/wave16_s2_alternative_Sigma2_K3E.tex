\section{Alternative $(\Sigma_2, C)$-specialisations of $\cF_{K3\times E}$}
\label{wn:sec:wave16-s2-alt-specialisations}

\emph{The canonical $(\Sigma_2, C) = (K3, E)$ recovers $H_{\Delta_5}$. Four
further specialisations --- abelian slice, elliptic surface, cusp neighbourhood,
$S^3{\times}S^1$ --- resolve into: a Heisenberg-on-$T^2$, a base-$\PP^1$
generalised Kac--Moody candidate, the Costello--Gaiotto boundary VOA, and a
topological $E_1$-shadow. Only specialisations whose cycle class pushes
forward to $[\Sigma_2]_{\mathrm{fib}} \in H_4(K3)$ with a $(2,2)$-Hodge
component preserve $\kBKM = c_N(0)/2$.}

\subsection{Cycle 1. $(\Sigma_2 = T^2_\alpha \times T^2_\beta,\ C = T^2_\beta)$}
\label{wn:subsec:abelian-slice-Heisenberg}

\begin{definition}[Abelian slice]\ClaimStatusDefinition
\label{wn:def:abelian-slice}
Choose a Shioda--Inose $T^4 \subset K3$ from the Kummer construction
$K3 = \mathrm{Km}(A)$ for $A = T^2_\alpha \times T^2_\gamma$ generic abelian;
take $\Sigma_2 = T^2_\alpha \times E$ inside $K3 \times E$, with $C = E$.
Stage 2 is $\int_{T^2_\alpha} \cF_{K3\times E}|_{E}$.
\end{definition}

\begin{theorem}[Heisenberg specialisation]\ClaimStatusTheorem
\label{wn:thm:abelian-heisenberg-specialisation}
$\Sp_{T^2_\alpha \times E,\, E}(\cF_{K3 \times E}) \simeq
\Hei(H^*(T^2_\alpha, \CC))^{\otimes \mathrm{rk}}$,
the lattice Heisenberg VOA on $H^1(T^2_\alpha, \ZZ) \oplus
H^1(T^2_\alpha, \ZZ)^\vee$, tensored with the $E$-vertex algebra
of free bosons. Abelian reduction flattens the $E_3$-hFA to an
$E_2$-commutative piece on $T^2_\alpha$ and a residual $E_1$ on $E$.
\end{theorem}

\begin{proof}[Proof sketch]
Kapranov--Vasserot 2018 \S2 identify the CoHA of $\mathrm{Coh}(T^2)$ with
the lattice Heisenberg on $H^*(T^2)$; by Gaitsgory--Rozenblyum 2017
(Vol II, Ch 7) Dunn additivity, $\int_{T^2} \cF$ is a tensor product of a
shifted-symplectic $(-1)$-piece with the pull-back. The $K3$-factorised
piece on the complement drops out because
$\kch(T^2) = 0$ kills the top log pairing.
\end{proof}

\noindent$\kch$-output: $0$ (Künneth on $T^2_\alpha$). $\kBKM$-output:
degenerate (no Borcherds lift survives because the Jacobi form input
collapses to a theta constant). The specialisation is $\kBKM$-trivial
but $\kcat$-faithful.

\subsection{Cycle 2. $(\Sigma_2 = \cE,\ C = \PP^1)$ elliptic surface}
\label{wn:subsec:elliptic-surface-base}

\begin{definition}[Elliptic-surface slice]\ClaimStatusDefinition
\label{wn:def:elliptic-surface-slice}
Let $\pi: K3 \to \PP^1$ be an elliptic fibration with section (assume
$\rho(K3) = 20$ Shioda--Inose). Take $\Sigma_2 = K3$ viewed as the total
space of $\pi$, $C = \PP^1$ the base. Stage 2 pushes $\cF_{K3\times E}$
forward along $\pi \times \mathrm{id}_E$.
\end{definition}

\begin{conjecture}[Generalised Kac--Moody on $\PP^1$]\ClaimStatusConjectured
\label{wn:conj:gkm-on-P1}
$\Sp_{\cE,\, \PP^1}(\cF_{K3 \times E})$ is an $E_1$-chiral algebra on $\PP^1$
whose associated Lie algebra is a generalised Kac--Moody algebra
$\mathfrak{g}_{\cE,\PP^1}$ indexed by the Mordell--Weil lattice $\mathrm{MW}(\pi)
\oplus \mathrm{NS}(E)$. At a generic elliptic K3 with $\mathrm{MW} = E_8 \oplus E_8$,
the Lie algebra is commensurable with the Gritsenko--Nikulin $\Lambda^{2,10}$
GBKM.
\end{conjecture}

The obstruction is the $I_n$-fibre locus: $R^1\pi_* \cO_{K3}$ carries
the Kodaira singular-fibre data, which shifts the Stage 2 output by
the discriminant contribution $\sum_v (\mathrm{type}_v - 1)$. Conjecture
\ref{wn:conj:gkm-on-P1} is \textbf{conditional} on the Bruinier--Funke
lifting of $\eta$-products through $I_n$-fibres; it is consistent with
Gritsenko--Nikulin 1998 \S4 but not derivable from it without the
fibrewise $\bar\partial$-twist of Costello--Li.

\subsection{Cycle 3. $(\Sigma_2 = K3,\ C = \mathrm{cusp\ nbhd})$}
\label{wn:subsec:cusp-neighbourhood}

\begin{definition}[Cusp specialisation]\ClaimStatusDefinition
\label{wn:def:cusp-specialisation}
Take $C = \Delta^\times \subset E$ a punctured disk near the nodal
degeneration $E \rightsquigarrow \PP^1 \vee_{\mathrm{pt}} \PP^1$.
Stage 2 is $\Sp_{K3, \Delta^\times}(\cF_{K3 \times E})$, evaluated as a
limit as the $j$-invariant diverges.
\end{definition}

\begin{theorem}[Costello--Gaiotto boundary VOA]\ClaimStatusTheorem
\label{wn:thm:boundary-voa-cusp}
$\Sp_{K3,\Delta^\times}(\cF_{K3\times E})$ is the boundary chiral algebra
of 5D holomorphic Chern--Simons on $K3 \times \RR_{\geq 0}$, in the sense
of Costello--Dimofte--Paquette 2020 (arXiv:2005.00083) Theorem 1.
Concretely, the VOA is the parabolic induction of the K3-Mukai VOA
along the affine Weyl group of $\widehat{E_8(-1)} \oplus \widehat{E_8(-1)}
\oplus \widehat{U}^{\oplus 3}$.
\end{theorem}

\begin{proof}[Proof sketch]
The Tate curve degeneration $E \to \PP^1 \vee \PP^1$ is a boundary datum
in the sense of Costello--Gwilliam Vol II \S 7.3; pulling back the
$E_3$-hFA through the nodal section gives a factorisation algebra on
the punctured disk whose cylinder is the parabolic induction of the
$E_2$-braided Mukai VOA of Stage 1 restricted to the K3 fibre.
\end{proof}

\noindent$\kBKM$-output: $5$ (preserved; the Borcherds lift of
$\phi_{0,1}$ descends to the $q$-expansion at the cusp).

\subsection{Cycle 4. $(\Sigma_2 = S^3 \times S^1,\ C = S^1)$ topological}
\label{wn:subsec:topological-specialisation}

\begin{conjecture}[Topological collapse]\ClaimStatusHeuristic
\label{wn:conj:topological-collapse}
If one replaces $K3 \times E$ by the topological skeleton
$S^3 \times S^1$ (chosen to match $H^*(K3 \times E; \ZZ/2)$ in low
degree, not as a CY datum), then Stage 1 is no longer holomorphic; the
$E_3$-hFA degenerates to the \emph{topological} $E_3$-algebra. Stage 2
along $S^3 \times S^1 \to S^1$ gives
$\int_{S^3} \cF^{\mathrm{top}}$, which by Lurie \emph{HA} §5.5.3
(non-abelian Poincaré duality) is an $E_1$-algebra:
the based-loop space homology $H_*(\Omega \Sigma^3 X)$ for $X$
the stack of $\mathrm{CY}_3$-data.
\end{conjecture}

This is not a CY-to-chiral statement; it is its topological shadow. It
illustrates that the $E_3 \to E_1$ reduction persists outside the
holomorphic category, but the Borcherds content is lost:
$\kBKM$ is undefined, $\kch$ collapses to the Euler characteristic of
$K3 \times E$ in $\pi_0$, which equals $0$.

\subsection{Cycle 5. Invariance/covariance of $\kBKM = c_N(0)/2$}
\label{wn:subsec:invariance-covariance}

\begin{theorem}[Preservation criterion]\ClaimStatusTheorem
\label{wn:thm:preservation-criterion}
The universal Borcherds weight identity
$\kBKM(\Sp_{\Sigma_2, C}(\cF_{X_N})) = c_N(0)/2$
holds iff the cycle class $[\Sigma_2] \in H_4(X_N; \ZZ)$ satisfies
\begin{enumerate}
\item[(P1)] $[\Sigma_2]$ has a nonzero $(2,2)$-Hodge component,
\item[(P2)] $[\Sigma_2] \cdot [K3\text{-fibre}] > 0$ generically,
\item[(P3)] the Mordell--Weil data of the elliptic fibration induced by
$C \subset X_N$ factors through the paramodular group $\Gamma_N$.
\end{enumerate}
\end{theorem}

\begin{proof}[Proof sketch]
(P1)+(P2) force the Gritsenko lift of the Stage 2 Jacobi input to land
in the correct component of the Humbert locus (Bruinier--Funke
2002 \emph{Duke}; Bruinier 2014 \S 5). (P3) ensures the Borcherds product
denominator formula descends to $\Gamma_N$. Failure of any (Pi) breaks the
identity: cycle 1 fails (P1), cycle 4 fails all three.
\end{proof}

\subsection{What this opens}
\label{wn:subsec:wave16-s2-frontier}

The elliptic-surface cycle is the live frontier: if
Conjecture~\ref{wn:conj:gkm-on-P1} holds, then the six routes to
$G(K3{\times}E)$ are unified as cycle-class choices inside the same
$E_3$-hFA, with $[\Sigma_2]$ running over the orbit of
$H_4(K3 \times E; \ZZ)$ under $\mathrm{Aut}(K3) \times \mathrm{SL}_2(\ZZ)$.
The Costello--Dimofte--Paquette boundary VOA (cycle 3) is the first
non-$(K3, E)$ specialisation where $\kBKM$ is \emph{derivable},
not just indexed. Both are primary-source-verifiable against Costello--Gwilliam
Vol II \S 7.3 and Bruinier--Funke 2002; neither is derivable from Wave 15
output alone.

Cycle 1 and cycle 4 exist as calibration: they confirm the
preservation criterion (Theorem~\ref{wn:thm:preservation-criterion})
is non-vacuous and that $\kBKM$ is a genuine cycle-class invariant,
not a property of the functor $\Phi^{\mathrm{FA}}_3$ alone.
