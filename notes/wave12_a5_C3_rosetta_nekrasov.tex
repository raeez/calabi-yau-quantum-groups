\subsection*{A3. The Drinfeld centre and the $\Etwo$ enhancement

\textsc{Attack.} Theorem~\ref{wn:thm:c3-drinfeld-center} asserts
\[
\mathcal Z\bigl(\mathrm{Rep}^{\Eone}(Y^+(\gghone))\bigr)\;\simeq\;
\mathrm{Rep}^{\Etwo}(Y(\gghone))\;\simeq\;
\mathrm{Rep}^{\Etwo}(\Wilpha).
\]
This is precise --- but the prose around it occasionally slides
``$\Etwo$ on $\Wilpha$'' (meaning: on the algebra object itself).
Top-15 \#3 and F8: at $d\ge 3$, $A$ is $E_1$; $\Etwo$ lives on
$\mathcal Z(\mathrm{Rep}(A))$, not on $A$ itself.

\noindent
\textbf{Ghost theorem.} Müger's theorem: for $A$ a finite tensor category

\noindent
\textbf{Precise error.} ``$\Wilpha$ is $\Etwo$'' vs.\ ``$\mathrm{Rep}$
of $\Wilpha$ has an $\Etwo$-enhancement via the Drinfeld centre''. The
chiral algebra itself is $\Eone$ on a reference curve; only the
(braided) category of modules is $\Etwo$.

\noindent
\textbf{Correct relationship (character).}
The character identity
\[
\mathrm{ch}\,\mathcal Z(\mathrm{Rep}(Y^+))
  \;=\;M(q)^2\cdot P(q)
  \;=\;\prod_{n\ge 1}(1-q^n)^{-(2n+1)}

($M(q)\cdot M(q)$) times neutral partition data ($P(q)$). Two

lift. The $\Etwo$ braiding --- the Yang $R$-matrix on the
charge-$2$ sector --- is a computation \emph{on the centre}; it is
meaningless on $\Wilpha$ itself.

\textsc{Heal.} In
Theorem~\ref{wn:thm:c3-drinfeld-center} replace any sentence of the
shape \emph{``$\Wilpha$ is $\Etwo$''} with the unambiguous
\emph{``$\mathcal Z(\mathrm{Rep}^{\Eone}(Y^+(\gghone)))$ is $\Etwo$; the
underlying $Y^+(\gghone)$ and $\Wilpha$ are $\Eone$''}. Reference F8 /
top-15 \#3 as the structural discipline.

\emph{$\Etwo$-braiding witness.} On the charge-$2$ sector of

\subsection*{Scope guard: $\Omega$-background is toric-specific}

\emph{Universal propagation warning.}
Nekrasov's $\Omega$-background is defined on \emph{toric} geometries
admitting a torus action with fixed-point localisation. Of the four
CY$_3$ in this wave's scope:
\begin{itemize}
\item $\mathbb C^3$: $T^3$-toric. $\Omega$-background applies.
\item Conifold: $T^3$-toric. $\Omega$-background applies
    perturbatively; non-perturbatively, wall-crossing is extra data.
\item Local $\mathbb P^2$: $T^3$-toric. $\Omega$-background applies.
\item Local $\mathbb P^1\times\mathbb P^1$: $T^3$-toric.
    $\Omega$-background applies.
\end{itemize}
$K3\times E$, the quintic, and compact CY$_3$ in general are \emph{not}
toric; there is no $T^3$ action. Theorems proved via
$\Omega$-background, 3d-partition localisation, Nekrasov partition
function, or MacMahon weight sum \emph{do not propagate} to the
compact non-toric case. This is cache entry \#1 in the CLAUDE.md
``Watch'' list (toric $\to$ all CY) and is enforced at every step above.

\subsection*{Healed statement of the $\mathbb C^3$ Rosetta Stone}

\begin{theorem}[$\mathbb C^3$ Rosetta Stone, Nekrasov-audited]
\label{wn:thm:c3-rosetta-nekrasov}
\ClaimStatusTheorem\ Let $X=\mathbb C^3$ with $T^3$-action of weights
$(\epsilon_1,\epsilon_2,\epsilon_3)$ satisfying $\epsilon_1+\epsilon_2
+\epsilon_3=0$ (CY$_3$ identity). Then:
\begin{enumerate}[label=\textup{(\alph*)}]
\item (\emph{SV 2013}) $\CoHA(X)\simeq Y^+(\gghone)$ as associative
    algebras over $\mathbb F(\!(z)\!)$ --- positive half of the affine
    Yangian, NOT the full $\Wilpha$.
\item (\emph{Tsymbaliuk 2017}) The Hall--Drinfeld double
    $\mathcal D_\hbar(Y^+)=Y(\gghone)$ is isomorphic to $\Wilpha
    [\lambda]$ at central charge $c=1$ and
    $\lambda=\epsilon_1/\epsilon_3+\epsilon_2/\epsilon_3$.
\item (\emph{CY-A$_3$}) $\Phi_3(\mathrm{Perf}(X))$ is an $E_1$-chiral
    algebra on a reference curve, isomorphic to $\Wilpha|_{c=1}=H_1$
    (Heisenberg VOA).
\item (\emph{Drinfeld centre / Müger}) $\mathcal Z(\mathrm{Rep}^{\Eone}
    (Y^+(\gghone)))\simeq\mathrm{Rep}^{\Etwo}(Y(\gghone))$, with
    character $M(q)^2 P(q)$ and $\Etwo$-braiding given by the Yang
    $R$-matrix $R(u)=1+\hbar P/u+\ldots$
\item (\emph{Five-path $\kappa_{\mathrm{ch}}$}) $\kappa_{\mathrm{ch}}(X)
    =\kappa_{\mathrm{ch}}(H_1)=1$, witnessed by: (i) OPE $J(z)J(w)\sim
    1/(z-w)^2$; (ii) genus-1 free energy $F_1=1/24$ via MacMahon; (iii)
    DT degree-0 $F_1=1/24$; (iv) affine Yangian structure function at
    the self-dual point; (v) CY categorical trace.
\item (\emph{Shadow tower, independent check}) $m_5=-80/9$ via
    $G_5^{\mathrm{conn}}=775/5184$ is the unique value consistent with
    graph-enumeration (Kontsevich 1994) + single-valued MZV projection
    (Brown 2013) + shadow recursion (Costello--Gwilliam 2017).
\end{enumerate}
\end{theorem}
