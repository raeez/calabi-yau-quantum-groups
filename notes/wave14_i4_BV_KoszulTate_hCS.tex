\title{BV and Koszul--Tate model for holomorphic Chern--Simons theory on
\texorpdfstring{$\C^3$}{C3}}
\date{2026-04-22}

\begin{document}
\maketitle

Let \(X=\C^3\), let
\(\Omega=dz_1\wedge dz_2\wedge dz_3\), and let \(\g\) be a finite
dimensional complex Lie algebra with non-degenerate invariant symmetric
pairing \(\tr(-,-)\). Write
\[
  \cE_{\mathrm{cl}}=\Omega^{0,1}(X,\g)
\]
for the classical hCS fields in the fixed \(C^\infty\)-trivialisation.
The bracket on \(\Omega^{0,\bullet}(X,\g)\) is the wedge product of
forms combined with the Lie bracket of \(\g\).

\section{Jets and local functionals}

The infinite jet bundle \(\Jinf\cE_{\mathrm{cl}}\to X\) has fibre over
\(x\in X\) equal to the formal Taylor expansions at \(x\) of
\((0,1)\)-forms with values in \(\g\). Its function sheaf
\(\cO(\Jinf\cE_{\mathrm{cl}})\) is a left \(\cD_X\)-module: the total
derivatives \(\partial_{z_i}\) and \(\partial_{\bar z_i}\) shift the jet
multi-index. Local functionals are coinvariants for total derivatives:
\[
  \cO_{\mathrm{loc}}(\cE_{\mathrm{cl}})
  =
  \Dens_X\otimes_{\cD_X}\cO(\Jinf\cE_{\mathrm{cl}})_{\mathrm{red}}
  \simeq
  \Omega_X^{3,3}\otimes_{\cD_X}\cO(\Jinf\cE_{\mathrm{cl}})/\C .
\]
This quotient is the algebraic form of integration by parts. A local
action is an element of \(\cO_{\mathrm{loc}}\), and its Euler--Lagrange
section is obtained by moving all total derivatives off the variation.

\section{Euler--Lagrange ideal}

For \(A\in\Omega^{0,1}(X,\g)\), the holomorphic Chern--Simons action is
\[
  S_{\hCS}(A)=
  \int_X\Omega\wedge
  \tr\!\left(\frac12 A\wedge\dbar A+\frac16 A\wedge[A,A]\right).
\]
The variation is
\[
  \delta S_{\hCS}(A)=
  \int_X\Omega\wedge
  \tr\!\left(\delta A,\,
    F_A^{0,2}\right),
  \qquad
  F_A^{0,2}=\dbar A+\frac12[A,A]\in\Omega^{0,2}(X,\g).
\]
Thus the equation of motion is the Maurer--Cartan equation
\[
  F_A^{0,2}=0 .
\]
Let \(I_{\EOM}\subset\cO(\Jinf\cE_{\mathrm{cl}})\) be the differential
ideal generated by the components of \(F_A^{0,2}\) and by all their
total derivatives. In coordinates,
\[
  F^a_{\bar\imath\bar\jmath}
  =
  \partial_{\bar\imath}A^a_{\bar\jmath}
  -\partial_{\bar\jmath}A^a_{\bar\imath}
  +\frac12 f^a{}_{bc}
    \bigl(A^b_{\bar\imath}A^c_{\bar\jmath}
          -A^b_{\bar\jmath}A^c_{\bar\imath}\bigr),
  \qquad 1\leq i<j\leq3 .
\]
The underived critical locus in the chosen trivialisation is the jet
scheme cut out by \(I_{\EOM}\). It is the formal moduli of integrable
\(\dbar\)-operators before division by gauge transformations.

\begin{lemma}[Bianchi syzygy]
\label{lem:bianchi}
The Euler--Lagrange section satisfies
\[
  \dbar_A F_A^{0,2}=0,\qquad
  \dbar_A=\dbar+[A,-].
\]
\end{lemma}

\begin{proof}
The operator \(\dbar_A=\dbar+[A,-]\) has curvature
\((\dbar_A)^2=[F_A^{0,2},-]\). Applying the graded Jacobi identity to
\(\dbar_A^2\) gives
\(\dbar_A F_A^{0,2}=0\). Equivalently, expanding
\(\dbar(\dbar A+\frac12[A,A])+[A,\dbar A+\frac12[A,A]]\) leaves only
the Jacobi sum in \(\g\), which is zero.
\end{proof}

\section{Koszul--Tate resolution}

Let
\[
  P=\Omega^{0,2}(X,\g),\qquad R=\Omega^{0,3}(X,\g).
\]
The section \(F_A^{0,2}\) takes values in \(P\), and
Lemma~\ref{lem:bianchi} gives a first syzygy in \(R\). Denote by
\[
  A^\dagger_{0,2}\in P^\vee[1],\qquad
  c^\dagger_{0,3}\in R^\vee[2]
\]
the corresponding antifield generators, of ghost numbers \(-1\) and
\(-2\). Via the invariant pairing and the Calabi--Yau form, these duals
are represented by \((0,2)\)- and \((0,3)\)-forms with values in
\(\g\).

\begin{construction}[Koszul--Tate algebra]
\label{cons:KT}
The Koszul--Tate algebra of the hCS Euler--Lagrange locus is the free
graded-commutative \(\cD_X\)-algebra
\[
  \KT_{\hCS}
  =
  \cO(\Jinf\cE_{\mathrm{cl}})
  \otimes
  \operatorname{Sym}\bigl(P^\vee[1]\oplus R^\vee[2]\bigr).
\]
Its differential is the \(\cD_X\)-linear derivation determined by
\[
  d_{\KT}A=0,\qquad
  d_{\KT}A^\dagger_{0,2}=F_A^{0,2},\qquad
  d_{\KT}c^\dagger_{0,3}=\dbar_A A^\dagger_{0,2}.
\]
\end{construction}

\begin{lemma}
\label{lem:KT-square}
\(d_{\KT}^2=0\).
\end{lemma}

\begin{proof}
The only generator with a square to check is \(c^\dagger_{0,3}\):
\[
  d_{\KT}^2c^\dagger_{0,3}
  =d_{\KT}(\dbar_AA^\dagger_{0,2})
  =\dbar_A F_A^{0,2}=0
\]
by Lemma~\ref{lem:bianchi}. The connection coefficient \(A\) in
\(\dbar_A\) is \(d_{\KT}\)-closed.
\end{proof}

\begin{proposition}[Resolution on complete-intersection charts]
\label{prop:KT-resolution}
On a jet chart where \(F_A^{0,2}\) is a regular section and the Bianchi
identity generates the first syzygy sheaf of the Euler--Lagrange ideal,
\[
  H^0(\KT_{\hCS},d_{\KT})
  \cong
  \cO(\Jinf\cE_{\mathrm{cl}})/I_{\EOM},
  \qquad
  H^{<0}(\KT_{\hCS},d_{\KT})=0 .
\]
\end{proposition}

\begin{proof}
For a regular section, the ordinary Koszul complex on
\(F_A^{0,2}\) resolves the quotient by \(I_{\EOM}\). The displayed
Bianchi identity is a relation among the Koszul generators. Adding
\(c^\dagger_{0,3}\) is the Tate step killing precisely that relation.
Completeness of the listed first syzygy sheaf on the chart makes the
resulting Tate complex acyclic in negative degrees.
\end{proof}

This is a local statement on the jet scheme. Reducible connections and
non-complete-intersection charts require the full Tate tower of higher
syzygies for a global derived critical-locus theorem.

\section{Classical BV theory}

Set
\[
  \cE_{\BV}=\Omega^{0,\bullet}(X,\g)[1].
\]
A homogeneous element of \(\Omega^{0,q}(X,\g)\) has ghost number
\(\gh=1-q\), so the BV superfield decomposes as
\[
  \mathcal A
  =
  c+A_{0,1}+A^\dagger_{0,2}+c^\dagger_{0,3},
  \qquad
  \gh(c,A,A^\dagger,c^\dagger)=(1,0,-1,-2).
\]
Serre duality on \(X\) gives the degree \(-1\) pairing
\[
  \omega_{\BV}(\alpha,\beta)
  =
  \int_X\Omega\wedge\tr(\alpha\wedge\beta),
  \qquad
  \alpha\in\Omega^{0,p},\ \beta\in\Omega^{0,3-p}.
\]
Indeed \(\gh(\alpha)+\gh(\beta)=2-(p+3-p)=-1\).

\begin{theorem}[Classical master equation]
\label{thm:CME}
The local functional
\[
  S_{\BV}(\mathcal A)=
  \int_X\Omega\wedge
  \tr\!\left(
    \frac12\mathcal A\wedge\dbar\mathcal A
    +\frac16\mathcal A\wedge[\mathcal A,\mathcal A]\right)
\]
satisfies
\[
  \{S_{\BV},S_{\BV}\}_{\BV}=0 .
\]
Its Hamiltonian vector field is
\[
  Q_{\BV}\mathcal A=\dbar\mathcal A+\frac12[\mathcal A,\mathcal A],
  \qquad
  Q_{\BV}^2=0 .
\]
\end{theorem}

\begin{proof}
The variation of \(S_{\BV}\) is
\[
  \delta S_{\BV}
  =
  \int_X\Omega\wedge
  \tr\!\left(\delta\mathcal A,\,
  \dbar\mathcal A+\frac12[\mathcal A,\mathcal A]\right),
\]
using invariance of \(\tr\) and integration by parts in
\(\cO_{\mathrm{loc}}\). Hence the Hamiltonian vector field is the
displayed Maurer--Cartan operator \(Q_{\BV}\). The square of this
operator is
\[
  Q_{\BV}^2\mathcal A
  =
  \dbar\!\left(\dbar\mathcal A+\frac12[\mathcal A,\mathcal A]\right)
  +
  [\mathcal A,\dbar\mathcal A+\frac12[\mathcal A,\mathcal A]],
\]
which vanishes by \(\dbar^2=0\), the derivation rule for \(\dbar\), and
the graded Jacobi identity. Since \(Q_{\BV}\) is Hamiltonian for
\(S_{\BV}\), the identity \(Q_{\BV}^2=0\) is equivalent to the classical
master equation.
\end{proof}

The ghost-independent, antifield-number lowering part of \(Q_{\BV}\) is
exactly the differential of Construction~\ref{cons:KT}. The remaining
terms are the Chevalley--Eilenberg differential for the gauge action:
\[
  Q_{\BV}A=\dbar_Ac,\qquad
  Q_{\BV}c=\frac12[c,c],
\]
and the corresponding coadjoint action on \(A^\dagger\) and
\(c^\dagger\). Thus the BV complex is the BRST quotient of the
Koszul--Tate resolution. Proposition~\ref{prop:KT-resolution} still
requires its regularity hypotheses.

\section{Quantisation and anomaly}

A Costello quantisation of the classical hCS theory consists of
effective interactions
\(\{I[L]\}_{L>0}\), a propagator and gauge fixing, local counterterms,
and a completion of the space of fields such that the homotopy
renormalisation-group equation holds and the scale-\(L\) quantum master
equation
\[
  QI[L]+\frac12\{I[L],I[L]\}_L+\hbar\Delta_L I[L]=0
\]
is satisfied. At loop order \(n\), failure of the equation is a local
cohomology class
\[
  [\Theta_n]\in H^1\bigl(\Def_{\mathrm{loc}}(\cE_{\BV})\bigr).
\]
Solving the quantum theory requires \([\Theta_n]=0\) and a chosen local
counterterm primitive at every order.

For holomorphic Chern--Simons theory in complex dimension \(d\), the
one-loop gauge anomaly is governed by the invariant polynomial of degree
\(d+1\). In dimension \(3\) this is the quartic invariant
\[
  \theta_{\g}(x_1,x_2,x_3,x_4)
  =
  \operatorname{Tr}_{\mathrm{ad}}
  (x_1x_2x_3x_4)_{\mathrm{sym}}
  \in (\operatorname{Sym}^4\g^\vee)^\g ,
\]
and its descent class in the local deformation complex. The QME
obstruction is this quartic class. Cubic Casimir terms form a separate
counterterm family. Contractibility of \(X=\C^3\) removes global
Dolbeault cohomology, while quantisation still requires the vanishing or
local trivialisation of the quartic anomaly for the chosen gauge algebra
and renormalisation data. On a compact Calabi--Yau threefold, a
quantisation theorem must exhibit the QME solution, anomaly
cancellation, orientation or framing datum, factorisation or TCFT
descent, local counterterms, and the analytic or formal completion in
which the effective interactions live.

\section{Operadic and Hall-algebra scope}

The construction above proves a classical BV model and, under the
quantisation hypotheses just stated, supplies the local observable
factorisation algebra of hCS on \(\C^3\). For the CY-to-chiral
dimension convention, the \(d=3\) specialization produces an
\(E_1\)-chiral output. Braided \(E_2\) structures belong to derived
centres, Drinfeld doubles, or module-category layers under their own
hypotheses. The shifted symplectic form in the BV theory supplies the
classical bracket, while the \(E_2\) upgrade requires those additional
centre or double constructions.

The Hall-algebra comparison is a separate theorem. In the toric
\(\C^3\) case the accepted identification is
\[
  \CoHA(\C^3)=Y^+(\widehat{\mathfrak{gl}}_1),
\]
the positive half. The full \(\mathcal W_{1+\infty}\) object appears
only after passing to the appropriate double, centre, or representation
category. Constructing the CoHA product, the Drinfeld double, or a
Borcherds--Kac--Moody denominator identity requires the corresponding
Hall-algebra and completion data in addition to the BV and
Koszul--Tate complexes written here.
