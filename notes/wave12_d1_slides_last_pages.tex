\paragraph{2. Explicit CY3 quotient construction (slides 857--866).}
Let $(S, g_N, h_M)$ be a K3 with two symplectic automorphisms of
finite order~$N, M \leq 8$. Let $(E, e_0)$ be a non-singular elliptic
curve with $N$-torsion point~$e_0$. The action
\[
  \mathbb{Z}/N\mathbb{Z} \;\curvearrowright\; S \times E,
  \qquad (s, e) \mapsto (gs,\, e + e_0)
\]
is free and preserves the CY form; the quotient
$X = (S \times E)/(\mathbb{Z}/N\mathbb{Z})$ is a smooth projective
CY3. For $N=1$, $X = S \times E$ (untwisted K3 fibration).

\paragraph{3. Elliptic K3 fibration datum (slides 867--882).} Take
$\pi : S \to \mathbb{P}^1$ elliptically fibered with sections
$s_1, s_2 : \mathbb{P}^1 \to S$, with $s_2$ of order~$N$ relative to
the zero-section~$s_1$. The iterated translates
$s_k = s + s_2(\pi(s_{k-1}))$ for $1 \leq k \leq N$ define $N$
sections. Let $F \in \mathrm{Pic}(S)$ be a fiber class; set
\[
  \beta_h \;=\; \tfrac{1}{N}\,(s_1 + s_2 + \cdots + s_N + hF),
  \qquad h \geq 0.
\]

\paragraph{4. DT series $\Zpartition^{X}(q, t, p)$ (slides 883--909).}
With $\langle \alpha, \beta \rangle = \int_S \alpha \cup \beta$ the
K3 intersection pairing,
\[
  \Zpartition^{X}(q, t, p)
  \;=\; \sum_{h=0}^{\infty} \sum_{d=0}^{\infty} \sum_{n \in \mathbb{Z}}
        \mathrm{DT}_{n, (\beta_h, d)} \; q^{d-1} \, t^{\tfrac{1}{2}\langle \beta_h, \beta_h \rangle} \, (-p)^n.
\]
The DT invariant is defined via Behrend weighting of the Hilbert
scheme $\mathrm{Hilb}^n(X, \beta_h)$ modulo the $E$-action:
\[
  \mathrm{DT}_{n, (\beta_h, d)}
  \;=\; \int_{\mathrm{Hilb}^n(X, \beta_h)/E} \nu_{\mathrm{de}}
  \;=\; \sum_{k \in \mathbb{Z}} k \cdot e(\nu^{-1}(k)),
\]
with $\nu : \mathrm{Hilb}^n(X, \beta_h) \to \mathbb{Z}$ the Behrend
function. The prefactor $q^{d-1}$ is load-bearing: it shifts the
curve-class exponent by the elliptic-fiber degree, matching the
Borcherds lift.

\paragraph{5. Theorem $N = 1$ (slide 912).} For $X = S \times E$ a
smooth projective CY3 from elliptically fibered K3,
\[
  \Zpartition^{S \times E} \;=\; -\frac{C}{(\Delta_5)^2}.
\]
The K3 elliptic genus is $2\phi_{0,1}$. This is the direct DT-side
witness of $\kBKM(\Phi_{10}) = 10$ and hence
$\kBKM(\Delta_5) = 5 = c_1(0)/2$ (Borcherds weight theorem applied
to $\Phi_{10} = \Delta_5^2$).

\paragraph{6. Conjecture: all $d_d$ Siegel forms arise via twisting
(slides 920--936).} For $X$ from $(S, g_N^r, g_N^s)$ and $(E, e_0)$
with $S$ carrying a lattice polarization~$L$, define the
$L$-twisted $(h_M = g_N^s)$-series $\Zpartition^{X}_{L, h_M}(q, t, p)$.

\emph{Conjecture.} All $d_d$ Siegel modular forms arise, up to
constant, as
\[
  -\frac{1}{\Zpartition^{X}_{L, h_M}(q, t, p)},
\]
and the associated GBKM is a \emph{correction}
$\gBKM_{(g_N, h_M)} \subset \gBKM_L$ determined by the
$(g_N, h_M)$-twisted elliptic genus of K3,
$F_L^{(g_N, h_M)}$, a Jacobi form of weight~0, index~1.

\paragraph{7. Case decomposition (slides 938--952).} With the

\[
  A \;=\; \tfrac{1}{4}\,\phi_{0,1}, \qquad
  B \;=\; \phi_{-2,1} \;=\; \frac{\nu_1^2}{\eta^6},

the twisted elliptic genus breaks into four regimes:
\begin{itemize}
  \item \textbf{Case $N \in \{1, 2, 3, 5, 7\}$}: single uniform
    ansatz for $1 \leq s, r \leq N-1$, $0 \leq k \leq N-1$

  \item \textbf{Case $N = 4$}: four-parameter family indexed by

  \item \textbf{Case $N = 6$}: four-parameter family indexed by

\paragraph{8. Vol III programme connections.}
\begin{itemize}
  \item The quotient $X = (S \times E)/(\mathbb{Z}/N)$ enlarges the
    Vol III CY3 census beyond $S \times E$: it gives a
    one-parameter family of CY3's whose $\kBKM$ spectrum is
    governed by Gritsenko's $\Delta_{k(N)}$ for $N \in \{2, 3, 4, 6\}$,
    matching the Igusa/Gritsenko weights
    $\{5, 2, 1, 1/2\}$ (cf.\ working\_notes.tex Table).
  \item The $N \in \{1, 2, 3, 5, 7\}$ uniform ansatz is precisely
    the \emph{prime} case; $N = 4, 6, 8$ break into blocks by the
    $\mathbb{Z}/N$-character decomposition of $H^*(S)^{g_N}$.
  \item The conjecture states a $\Phi$-inverse principle:
    $-1/\Zpartition^{X}_{L, h_M}$ is the Borcherds denominator
    of~$\gBKM_{(g_N, h_M)}$, so DT-invariants on~$X$ reconstruct
    BKM root multiplicities.
\end{itemize}
