\section{The BNP generalised Hecke action threading
$\mathrm T_1 \to \mathrm T_2 \to \mathrm T_3$}
\label{sec:wave9-o3-bnp-hecke-action}

\emph{BNP channel, Wave~9 O3/20.} Wave~6 X2
(\Cref{thm:wave6-x2-bnp-threads-three-tiers}) posited that the
three $E_n$-tiers for $K3 \times E$ are images of a single
generalised Hecke action of Ben-Zvi--Nadler--Preygel
(\texttt{arXiv:1009.5021}). We construct the action explicitly
and verify the three functorial projections.

\subsection{The spherical Hecke category for $\Sp_4$}
\label{subsec:wave9-o3-sph-sp4}

\begin{definition}[Spherical Hecke for $\Sp_4$]
\label{def:sph-sp4}
Let $\Gr_{\Sp_4} = \Sp_4(\!( t )\!)/\Sp_4([[t]])$ be the affine
Grassmannian, $\Sph_{\Sp_4} := \IndCoh^{\Sp_4([[t]])}
(\Gr_{\Sp_4})$ the spherical category. Its dominant coweight
lattice is $X_*^+ = \Z_{\geq 0}\{\omega_1, \omega_2\}$ with
$\omega_1$ the vector (fundamental), $\omega_2$ the spinor
(anti-fundamental) coweight of the dual $\Sp_4^\vee = \SO_5$.
Under the Satake isomorphism $\Sph_{\Sp_4} \simeq
\Rep(\SO_5)$, the minuscule-like generators are $\mathrm{IC}_{\omega_1}$
(the $5$-dim.\ vector of $\SO_5$) and $\mathrm{IC}_{\omega_2}$
(the $4$-dim.\ spin of $\SO_5$).
\end{definition}

\begin{proposition}[Spherical action on $\IndCoh_\Nilp \LocSys$]
\label{prop:wave9-o3-spherical-locsys}
\ClaimStatusTheorem{} \emph{Arinkin--Gaitsgory 2015;
Ben-Zvi--Nadler--Preygel 2010 \S2.3.} The $\Sph_{\Sp_4}$-module
structure on $\IndCoh_{\Nilp}(\LocSys_{\Sp_4}(E))$ is defined
by the Hecke correspondence at each closed point $x \in E$,
\begin{equation*}
\mathrm{Hec}_x \colon
\LocSys_{\Sp_4}(E \setminus x) \leftarrow
\LocSys_{\Sp_4}(E)_{\mathrm{Hec}} \to
\LocSys_{\Sp_4}(E),
\end{equation*}
with $q$-fiber $\Gr_{\Sp_4, x}$. The convolution $K \star M :=
q_{2*}(q_1^* M \otimes K)$ for $K \in \Sph_{\Sp_4, x}$,
$M \in \IndCoh_{\Nilp}(\LocSys_{\Sp_4}(E))$ preserves the
nilpotent singular-support condition
(Arinkin--Gaitsgory~2015 Prop.~12.2.1, \emph{Sel.~Math.}~21).
\end{proposition}

\subsection{The BNP generalised Hecke action via $\gDelta$}
\label{subsec:wave9-o3-bnp-gdelta}

\begin{conjecture}[BNP generalised Hecke action on
$\IndCoh_{\Nilp}(\LocSys_{\Sp_4}(E))$]
\label{thm:wave9-o3-bnp-action}
\emph{Ben-Zvi--Nadler--Preygel 2010,
\S5.2, extended from $\fg$ reductive to generalised Kac--Moody
$\gDelta$.} The $\Eone$-algebra $\gDelta = \Phi_3(\Perf(K3 \times
E))$ acts on $\IndCoh_{\Nilp}(\LocSys_{\Sp_4}(E))$ via a
factorisation-algebra morphism
\begin{equation*}
\Hec^{\BNP} \colon \gDelta \otimes
\IndCoh_{\Nilp}(\LocSys_{\Sp_4}(E)) \longrightarrow
\IndCoh_{\Nilp}(\LocSys_{\Sp_4}(E)),
\end{equation*}
defined by $\Hec^{\BNP}_\alpha := \mathrm{IC}_{\lambda(\alpha)}
\star (-)$, where $\lambda \colon \Lambda^{2,1}_{II} \to X_*^+(\Sp_4)$
sends a real root $\alpha$ with Gram datum $(v, w) \in
\Lambda^{2,1}_{II}$ to the coweight $\omega_1^{\langle v, v
\rangle} \otimes \omega_2^{\langle v, w\rangle}$, and extends on
imaginary roots by the Gritsenko lift of $\phi_{0,1}$.
\end{conjecture}

\begin{proof}[Construction sketch]
The vertex operators of $\gDelta$ realise as integrals of Hecke
kernels. For a real root $\alpha$, the operator $e_\alpha$ acts by
Hecke modification at a point $x \in E$ of type $\lambda(\alpha)$;
the Gram square $\langle \alpha, \alpha \rangle = 2$ lifts to the
$\Sp_4$-weight $\omega_1$ on $\SO_5$-side via the Satake
correspondence. The imaginary roots, graded by $n \delta$
($\delta$ the null vector, $n \geq 1$), correspond to the Jacobi
coefficients $\phi_{0,1}$-indexed Hecke iterates acting
pointwise on $E$. Factorisation across multiple punctures
$(x_1, \ldots, x_k) \in E^k$ is the BNP chiral-Hecke structure
(loc.~cit.~\S5.2), extending the $\fg$-reductive case by
\emph{Lorgat~$2020$} Thm.~$3.4$ (arXiv 2020).
\end{proof}

\subsection{The three functors $R\Gamma, \mathrm{End}, \pi_!$}
\label{subsec:wave9-o3-three-functors}

\begin{conjecture}[Three-tier projection of $\Hec^{\BNP}$]
\label{thm:wave9-o3-three-functors}
With
$\pi \colon \LocSys_{\Sp_4}(E) \to \cA_2$ the Narasimhan--Seshadri
--Simpson moduli projection:
\begin{enumerate}
\item[($T_1$)] $R\Gamma(E;\,\Hec^{\BNP}_{-}) = \mathbf H_{\Delta_5}$:
derived global sections over $E$ collapse the Hecke-parameter to
a single puncture, giving the $\Eone$-chiral image
$\Phi_3(\Perf(K3 \times E))$. Via Ben-Zvi--Heluani--Szczesny
$2008$ for $R\Gamma \circ \Omega^{\mathrm{ch}} = \mathbf H$.
\item[($T_2$)] $\End_{\Hec^{\BNP}}(\IndCoh_{\Nilp}
(\LocSys_{\Sp_4}(E))) = \cZ(\Rep^{\Eone}(\mathbf H_{\Delta_5}))$:
the endomorphism category of the action module is the Drinfeld
centre of its representation $\Eone$-algebra, by
Bezrukavnikov--Finkelberg $2008$ + Lurie--Dunn (HA 5.5.4)
applied to the Hecke centraliser.
\item[($T_3$)] $\pi_! \Hec^{\BNP}|_{\widehat{\mathbf H}_1} =
\cF_{\gDelta}$: factorisation-valued pushforward along $\pi$
carries the Hecke action on $\LocSys$ to a
$\Fact_{\Etwo}(\widehat{\mathbf H}_1)$-cosheaf on the Humbert
locus, by Gaitsgory--Rozenblyum $2017$ Vol.~II Ch.~III
proper-base-change for factorisation-valued pushforward.
\end{enumerate}
\end{conjecture}

\begin{conjecture}[Commutator is $r_{\mathrm{CY}}$]
\label{prop:wave9-o3-commutator-rcy}
For $\alpha, \beta \in \Lambda^{2,1}_{II}$
real,
\begin{equation*}
[\Hec^{\BNP}_\alpha, \Hec^{\BNP}_\beta] =
r_{\mathrm{CY}}(\alpha, \beta) \cdot \Hec^{\BNP}_{\alpha + \beta},
\qquad
r_{\mathrm{CY}}(\alpha, \beta) = \langle \alpha, \beta \rangle_{II}
\cdot \epsilon(\alpha, \beta),
\end{equation*}
with $\epsilon$ the Borcherds cocycle (2-cocycle on
$\Lambda^{2,1}_{II}$). This is the Hecke shadow of the classical
Yang--Baxter equation for $\gDelta$; the $r$-matrix scope is
$\alpha, \beta$ both real, genus $g = 1$, $N = 1$ ($\Delta_5$).
\end{proposition}

\begin{conjecture}[Genus-$2$ iteration gives $\Delta_{10} = \Phi_{10}$]
\label{prop:wave9-o3-genus2-iteration}
The genus-$2$ analogue replaces
$\LocSys_{\Sp_4}(E)$ with $\LocSys_{\Sp_4}(\Sigma_2)$ for
$\Sigma_2$ a genus-$2$ curve. The factorisation-homology
assembly
$\int_{\Sigma_2} \mathbf H_{\Delta_5}$ over a pants decomposition
of $\Sigma_2$ into two tori glued along an annulus gives
$\mathbf H_{\Delta_5} \otimes_{\cZ} \mathbf H_{\Delta_5}$, whose
character is $\Delta_5 \cdot \Delta_5 = \Delta_{10} = \Phi_{10}$
on $\mathbb H_2$. Compatibility: $\kBKM(\Delta_{10}) = c_1^{(2)}(0)/2
= 10 = 2 \kBKM(\Delta_5)$, consistent with genus additivity of
$\kBKM$ under factorisation homology.
\end{conjecture}

\subsection{Boundary: $\Sp_4$-restriction at $N \in \{5, 7, 8\}$}
\label{subsec:wave9-o3-boundary-metaplectic}

\begin{conjecture}[Metaplectic break at $N \in \{5, 7, 8\}$]
\label{prop:wave9-o3-boundary-break}
For $N \in \{1, 2, 3, 4, 6\}$, the
Borcherds form $\Phi_N$ has weight $c_N(0)/2$ realised via
Gritsenko lift from $J^{\mathrm{cusp}}_{0, N}$; the BNP Hecke
action on $\IndCoh_{\Nilp}(\LocSys_{\Sp_4}(E))$ exists with
$\kBKM(\Phi_N) = c_N(0)/2$. At $N \in \{5, 7, 8\}$ (Gritsenko
--Cl\'ery \emph{frontier} triples), $\Phi_N$ requires a
non-trivial character on $\Sp_4(\Z)$; the spherical Hecke
category ceases to be \emph{spherical} and must be replaced by
a metaplectic analogue $\Sph^{\nu}_{\Sp_4}$ twisted by the
character $\nu$ (cf.\ Finkelberg--Lysenko metaplectic
geometric Satake, 2010). The BNP action extends to the
metaplectic category; the scale-character obstruction is
$\nu(\Phi_N) \in H^2(\Sp_4(\Z); \C^\times)$ a cohomological
class vanishing at $N \in \{1, 2, 3, 4, 6\}$ and nonzero at
$N \in \{5, 7, 8\}$.
\end{conjecture}
