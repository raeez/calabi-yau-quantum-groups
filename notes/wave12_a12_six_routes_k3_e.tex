\begin{abstract}
Six routes to the conjectural quantum vertex chiral group
$G(K3 \times E)$ are distinct construction machines. In the
normalisation used here, only the Heisenberg route applies the
CY-to-chiral functor $\Phi_3$ directly to
$\mathrm{Perf}(K3\times E)$. The organising lattice is the
Narain orthogonal group $O(\Lambda_{\mathrm{Nar}})$,
$\Lambda_{\mathrm{Nar}} = \mathrm{II}_{4,20} \oplus \mathrm{II}_{1,1}(-1)$,
acting on primitive Mukai vectors through the invariants
$D = 4mn-\ell^2$ and $\gcd(m,n,\ell)$. Four of the six orbit
types produce the canonical spectrum
$\{\kappa_{\mathrm{cat}}, \kappa_{\mathrm{ch}}^{\mathrm{Heis}},
\kappa_{\mathrm{BKM}}, \kappa_{\mathrm{fiber}}\}
= \{0, 3, 5, 24\}$;
the K3 fibre $\kappa_{\mathrm{cat}}(K3) = 2$ is the fibrewise value,
while the total-space value is
$\kappa_{\mathrm{cat}}(K3 \times E) = 0$. A symmetric-monoidal
$1$-functor $\Phi \colon \mathrm{CY\text{-}cat}_3 \to \mathrm{Alg}$
cannot have one scalar shadow reproducing all four $\kappa_\bullet$
simultaneously, since the four invariants carry four different
multiplicativity types. A coherent $6$-simplex
$\sigma \in N_6(\mathcal G_{\mathrm{mn}})$ exists only after the
comparison morphisms and higher homotopies in the moonshine groupoid
are constructed. The 6d holomorphic Chern--Simons theory on
$K3 \times E$ supplies the boundary models for the same six
strata, again under the corresponding hCS-to-Hall comparison
hypotheses.
\end{abstract}

\section{The construction problem}
\label{sec:intro}

The symbol $G(K3 \times E)$ denotes the conjectural $(E_1\text{-chiral})$
quantum vertex chiral group associated with the compact Calabi--Yau
threefold $K3 \times E$. The six construction machines separated
below are:
\[
\begin{array}{ll}
\mathrm{R1} & \text{categorical Euler real-root construction},\\
\mathrm{R2} & \text{Heisenberg--Mukai }E_1\text{-chiral construction},\\
\mathrm{R3} & \text{Igusa squaring and light-like boundary construction},\\
\mathrm{R4} & \text{primitive Gritsenko--Nikulin BKM construction},\\
\mathrm{R5} & \text{CHL }N\text{-twisted paramodular construction},\\
\mathrm{R6} & \text{K3 fibre-rank construction}.
\end{array}
\]
They are six constructions, with six inputs and six comparison
problems. The functor $\Phi_3$ supplies the R2 Heisenberg
specialisation of the single native object
$\Phi^{\mathrm{FA}}_3(\mathrm{Perf}(K3\times E))$. The Borcherds
weight, the Igusa square, the CHL twists, and the Mukai-rank fibre
invariant come from their own automorphic, Hall, or lattice data.
Schiffmann--Vasserot CoHA, Maulik--Okounkov stable envelopes,
BFN Coulomb branches, Kummer free fields, and the BLLPR Schur
sector occur as comparison avatars inside this diagram; they stay
inside the six Narain orbit types.

\section{The four invariants}
\label{sec:four-invariants}

Before enumerating constructions, four scalar invariants are
attached to $K3 \times E$, each a different chiral shadow of a
different algebraic structure.

\begin{definition}[Categorical Euler characteristic]
\label{def:kappa-cat}
For a compact Calabi--Yau $d$-fold $X$ with structure sheaf $\mathcal O_X$,
$\kappa_{\mathrm{cat}}(X) := \chi(\mathcal O_X)
= \sum_q (-1)^q h^{0,q}(X)$.
On a product $\kappa_{\mathrm{cat}}$ is K\"unneth-multiplicative:
$\kappa_{\mathrm{cat}}(X \times Y)
= \kappa_{\mathrm{cat}}(X) \cdot \kappa_{\mathrm{cat}}(Y)$.
On $K3$: $\chi(\mathcal O_{K3}) = 1 - 0 + 1 = 2$. On $E$:
$\chi(\mathcal O_E) = 1 - 1 = 0$. On $K3 \times E$:
$\kappa_{\mathrm{cat}}(K3 \times E) = 2 \cdot 0 = 0$.
\end{definition}

\begin{definition}[Heisenberg chiral characteristic]
\label{def:kappa-ch-heis}
$\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(K3 \times E) := 3$
is the Heisenberg--Mukai Cartan rank of the Stage-$2$ specialisation
\[
\mathrm{Sp}^{\mathrm{ch}}_{K3,E}\!
\bigl(\Phi^{\mathrm{FA}}_3(\mathrm{Perf}(K3 \times E))\bigr)
\]
on the elliptic base. It reads the rank-$3$ signature-$(2,1)$
Cartan line of the $\Delta_5$ comparison target. The compact
Hall--Borcherds recognition gate is separate: the scalar rank
$3$ is only a rank witness, while the Hall product, primitive
Borcherds bracket, and PBW comparison require finite recognition data.
\end{definition}

\begin{definition}[Borcherds--Kac--Moody weight]
\label{def:kappa-BKM}
Let $\Phi_N$ be a Borcherds product with weak-Jacobi or
vector-valued theta input
\[
\phi_N(\tau,z)=\sum_{n,\ell} c_N(4n-\ell^2)\,q^n\zeta^\ell .
\]
The BKM modular characteristic is the Borcherds product weight
\[
\kappa_{\mathrm{BKM}}(\Phi_N) := \mathrm{wt}(\Phi_N)=c_N(0)/2,
\]
where $c_N(0)$ is the discriminant-zero input coefficient; the
cuspidal form $\Phi_N$ itself has no constant term. For the
Gritsenko--Nikulin $\Delta_5$, the $q^0$ input is
$\zeta^{-1}+10+\zeta$, hence $c_1(0) = 10$, so
$\kappa_{\mathrm{BKM}}(\Delta_5) = 5$. The universal formula
$\kappa_{\mathrm{BKM}}(\Phi_N) = c_N(0)/2$ holds unconditionally
for the CHL denominator scope $N \in \{1, 2, 3, 4, 6\}$, with
\[
 (c_1(0),c_2(0),c_3(0),c_4(0),c_6(0))=(10,8,6,4,2),
\]
so $\kappa_{\mathrm{BKM}}(\Phi_N)=(5,4,3,2,1)$. It is the weight of
the Borcherds automorphic product attached to the lattice
$\mathrm{II}_{2,2}(N)$.
\end{definition}

\begin{definition}[Fibre-rank characteristic]
\label{def:kappa-fiber}
$\kappa_{\mathrm{fiber}}(K3 \times E) :=
\chi_{\mathrm{top}}(K3) = 24$. This is the rank of the Mukai
lattice $\Lambda_{\mathrm{Muk}}(K3) = H^\ast(K3, \mathbb Z) \cong
\mathrm{II}_{4,20}$, equivalently the second Chern number
$c_2(K3)$ integrated against the fundamental class, equivalently
the topological Euler characteristic of a single $K3$ fibre.
In the relative construction the K3 fibre supplies
$24$ free chiral bosons (lattice VOA $V_{\Lambda_{\mathrm{Muk}}}$).
\end{definition}

\begin{remark}[The four multiplicativity types]
\label{rem:four-mult}
$\kappa_{\mathrm{cat}}$ is K\"unneth-multiplicative;
$\kappa_{\mathrm{fiber}}$ is restriction to a fibre; it is determined
by the fibration structure, and the total space alone leaves it
undetermined;
$\kappa_{\mathrm{BKM}}$ is additive under Borcherds tensor product
of the underlying automorphic inputs;
$\kappa_{\mathrm{ch}}^{\mathrm{Heis}}$ is a Cartan-rank count in the
Heisenberg--Mukai specialisation.
Four multiplicativity types, four invariants: no single binary
operation on scalars reproduces all four consistently
(Proposition~\ref{prop:no-single-functor}).
\end{remark}

\section{The six routes: construction-by-construction}
\label{sec:six-routes}

Each subsection names the construction, the chiral-side avatar as an
algebraic object $\mathbf A^{(i)}$, the Narain orbit in which its
primitive Mukai vectors live, the $\kappa_\bullet$ it computes, and

\subsection{Route R$1$: categorical Euler real-root orbit}
\label{subsec:R1-mukai}

\emph{Construction.} The categorical Euler construction takes the
Hodge supertrace of the total CY$_3$ category
$\mathrm{Perf}(K3\times E)$. Its real-root lattice witness is the
K3 cohomological lattice $H^2(K3,\mathbb Z)\cong \mathrm{II}_{3,19}$,
viewed inside the Mukai lattice only after the two null directions
$H^0\oplus H^4$ have been separated.
\emph{Scalar shadow.} On the total space
\[
\kappa_{\mathrm{cat}}(K3\times E)
=\chi(\mathcal O_{K3})\,\chi(\mathcal O_E)
=2\cdot 0=0.
\]
The K3 fibre value $\kappa_{\mathrm{cat}}(K3)=2$ is a per-fibre
Hodge supertrace used as a witness for the calculation.
\emph{Orbit.} Real-root orbit, discriminant $D < 0$, gcd $=1$:
primitive vectors $v$ with $\langle v, v \rangle = -2$ in the
$\mathrm{II}_{3,19}$ sublattice.
\emph{Chiral-side avatar.} The Hodge-supertrace truncation of the
K3 fibre, transported through K\"unneth to the total-space
categorical scalar $0$.

\subsection{Route R$2$: Heisenberg $E_1$-chiral orbit}
\label{subsec:R2-heis}

\emph{Construction.} The $E_1$-chiral algebra
$\mathbf H_{\Delta_5}$ produced by the Stage-$2$ specialisation
$\mathrm{Sp}^{\mathrm{ch}}_{K3,E}\circ \Phi^{\mathrm{FA}}_3$ on
$\mathrm{Perf}(K3 \times E)$. This is the unique route in this
normalisation involving the CY-to-chiral functor $\Phi_3$ directly.
The dimension $d = 3$ gives an $E_1$-chiral algebra on the elliptic
base; the braided $E_2$ structure belongs to the derived chiral centre
of its representation category.
\emph{Scalar shadow.}
$\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(K3 \times E) = 3$
(Definition~\ref{def:kappa-ch-heis}).
\emph{Orbit.} Heisenberg orbit, $D < 0$, gcd $>1$: imprimitive
Mukai vectors along the $\mathrm{II}_{2,2}$ sublattice generated
by the Cartan of $\mathfrak g_{\Delta_5}$ plus the elliptic
coupling $z$.
\emph{Chiral-side avatar.}
$\mathbf H_{\Delta_5}$ as the Stage-$2$ Hall--Heisenberg avatar.
With the monic Borcherds product $D_5=64^{-1}\Delta_5$, the
Oberdieck--Pandharipande scalar normalisation is
$\Phi_{10}^{\mathrm{OP}}=D_5^2$ and
$Z_{K3\times E}^{\mathrm{DT}}=-(\Phi_{10}^{\mathrm{OP}})^{-1}$.
This character identity tests the compact Hall source; the Hall
bracket requires the compact critical correspondence data.

\subsection{Route R$3$: Light-like Igusa squaring orbit
($\Phi_{10}^{\mathrm{un}} = \Delta_5^2$)}
\label{subsec:R3-igusa}

\emph{Construction.} The Igusa cusp form
$\Phi_{10}^{\mathrm{un}} \in S_{10}(\mathrm{Sp}_4(\mathbb Z))$ is the unique
weight-$10$ Siegel cusp form with trivial character; it factors
as $\Phi_{10}^{\mathrm{un}} = \Delta_5^2$ with $\Delta_5 \in
S_5(\mathrm{Sp}_4(\mathbb Z), \nu_{\Delta_5})$ the
Gritsenko--Nikulin paramodular form of weight $5$ with
nebentypus. The OP scalar convention is
$\Phi_{10}^{\mathrm{OP}}=D_5^2=4096^{-1}\Delta_5^2$.
The construction of $\mathfrak g_{\Delta_5}$
by Borcherds lift of the weak Jacobi form $\phi_{0,1}$
(the K3 elliptic genus / $2$) is the additive-lift side;
the multiplicative-lift side produces $\Delta_5$ itself
as the denominator identity.
\emph{Scalar shadow.}
The Igusa square has weight $10$. It is the squared scalar
denominator used by DT theory and is separate from the primitive
BKM invariant $\kappa_{\mathrm{BKM}}(\Delta_5)=5$. On the
light-like ray, the shadow is the $\eta^9$ multiplicity
$\tau(a) = 9$.
\emph{Orbit.} Light-like orbit, $D = 0$: primitive null vectors
in $\mathrm{II}_{4,20} \oplus \mathrm{II}_{1,1}(-1)$.
\emph{Chiral-side avatar.} The character-level Borcherds tensor
square $V_{\Phi_{10}^{\mathrm{un}}}
= V_{\Delta_5} \otimes V_{\Delta_5}$, weight $10$.

\subsection{Route R$4$: BKM paramodular orbit ($\Phi_N$, $N \in \{1,2,3,4,6\}$)}
\label{subsec:R4-BKM}

\emph{Construction.} The Borcherds--Kac--Moody superalgebra
$\mathfrak g_{\Delta_5}$ is the $N = 1$ (untwisted) member of
the paramodular family $\{\Phi_N\}_{N \in \{1, 2, 3, 4, 6\}}$
of Gritsenko--Nikulin. Each $\Phi_N$ is a Borcherds
automorphic product on the paramodular group
$K(N) = \mathrm{II}_{2,2}(N) \cdot \mathrm{Sp}_4$; its Fourier
coefficients compute signed BKM root multiplicities
$\mathrm{mult}(\alpha) = c_N(4nm - \ell^2)$.
\emph{Scalar shadow.}
$\kappa_{\mathrm{BKM}}(\Phi_N) = c_N(0)/2$ for $N \in \{1, 2, 3, 4, 6\}$
(Borcherds $1998$; Gritsenko $1999$), with
$(c_1(0),c_2(0),c_3(0),c_4(0),c_6(0))=(10,8,6,4,2)$ and weights
$(5,4,3,2,1)$; for $N = 1$,
$c_1(0) = 10$ and $\kappa_{\mathrm{BKM}}(\Delta_5) = 5$.
\emph{Orbit.} Imaginary orbit, $D > 0$, gcd $= 1$: positive-norm
primitive vectors.
\emph{Chiral-side avatar.} The denominator-formula chiral algebra
$V_0(\mathfrak g_{\Delta_5})$ reading the Borcherds product
$\Delta_5(Z) = q_1^{1/2} q_2^{-1/2} q_3^{1/2}
\prod_{(m,n,\ell) > 0} (1 - q_1^m q_2^\ell q_3^n)^{c_1(4mn - \ell^2)}$.

\subsection{Route R$5$: CHL $N$-twisted paramodular orbit ($\gcd = N$)}
\label{subsec:R5-CHL}

\emph{Construction.} The CHL-twisted paramodular construction
starts from symplectic K3 automorphisms of orders
$N\in\{2,3,4,6\}$ compatible with the elliptic factor and forms
the Borcherds products $\Phi_N$ from the twined weak Jacobi input
$\phi_{0,1}^{(N)}$. This is the $N$-twisted sibling of R$4$,
separate from the Monster or Fake-Monster branch on
$\mathrm{II}_{25,1}$.
\emph{Scalar shadow.}
$\kappa_{\mathrm{BKM}}(\Phi_N)=c_N(0)/2$ with
$(c_2(0),c_3(0),c_4(0),c_6(0))=(8,6,4,2)$ and weights
$(4,3,2,1)$.
\emph{Orbit.} $N$-twisted orbit, gcd $= N$: imprimitive BPS
charges of fixed denominator in the Narain lattice.
\emph{Chiral-side avatar.} The CHL Hall--Borcherds target
$\mathbf H_{\Delta_5}^{(N)}$, conditional on the equivariant
hCS-to-Hall comparison and the $g_N$-equivariant specialisation
datum.

\subsection{Route R$6$: K3 fibre-rank orbit}
\label{subsec:R6-fiber}

\emph{Construction.} The relative chiral algebra construction
$\pi \colon K3 \times E \to E$ produces the fibration
$K3$ over the elliptic base and sews the fibres along $E$ via the
DMVV elliptic-genus formula. Each $K3$ fibre contributes a
boundary algebra with $\chi_{\mathrm{top}}(K3) = 24$ zero modes;
the $24$ reads the Mukai lattice rank and also the anomaly
coefficient of the $K3$ sigma model.
\emph{Scalar shadow.}
$\kappa_{\mathrm{fiber}}(K3 \times E) = 24$
(Definition~\ref{def:kappa-fiber}).
\emph{Orbit.} Fibre-rank orbit: the anti-diagonal embedding
$\mathrm{II}_{3,19} \hookrightarrow \mathrm{II}_{4,20}
\oplus \mathrm{II}_{1,1}(-1)$ selecting the K3-fibre summand.
\emph{Chiral-side avatar.} The relative vertex algebra
$\pi_\ast V_{\Lambda_{\mathrm{Muk}}}$ on $E$, equivalently the
$24$-boson lattice VOA $V_{\Lambda_{\mathrm{Muk}}}$ evaluated on
the $K3$-fibre, with Euler contribution $24$ sewn along the
elliptic fibre.

\subsection{BLLPR Schur-sector avatar of R$2$}
\label{subsec:BLLPR-sixth}

The Beem--Lemos--Liendo--Peelaers--Rastelli
Schur sector starts from the $6$d $(2,0)$ theory, compactifies to a
$4$d $\mathcal N = 2$ theory, and applies the BLLPR theorem to
obtain a $2$d vertex algebra from the Schur operators.
\emph{Feature.} The BLLPR construction is $\Omega$-independent:
it uses no $\Omega$-background localisation.
\emph{Reading.} This is a Schur-sector comparison with the
Heisenberg/Hall branch. It supplies a vertex-algebraic shadow of the
protected sector and a useful central-charge check, remaining inside
the Heisenberg orbit.

\section{The Narain orthogonal orbit structure}
\label{sec:orbits}

\begin{definition}[Narain orbit decomposition]
\label{def:narain-orbits}
Let $\Lambda_{\mathrm{Nar}} = \mathrm{II}_{4,20} \oplus
\mathrm{II}_{1,1}(-1)$ of signature $(5, 21)$, and let
$G_{\mathrm{Nar}} = O(\Lambda_{\mathrm{Nar}})$ act on
primitive lattice vectors of fixed Mukai norm
$\mathrm{Muk}(v) := \langle v, v \rangle_{\Lambda_{\mathrm{Nar}}}
\in \mathbb Z$.
For each orbit type $i \in \{\mathrm I, \mathrm{II}, \mathrm{III},
\mathrm{IV}, \mathrm V, \mathrm{VI}\}$ the characteristic data
$(\text{discriminant } D, \gcd, \text{sublattice})$ are:
\[
\begin{array}{lcccl}
\text{Orbit} & D & \gcd & \text{Sublattice} & \kappa_\bullet \\
\hline
\mathrm I \ (\text{categorical Euler}) & <0 & 1 & \mathrm{II}_{3,19} & \kappa_{\mathrm{cat}}(K3\times E) = 0 \\
\mathrm{II} \ (\text{Heisenberg}) & <0 & >1 & \mathrm{II}_{2,2} & \kappa_{\mathrm{ch}}^{\mathrm{Heis}} = 3 \\
\mathrm{III} \ (\text{light-like}) & 0 & 1 & \mathbb Z \cdot \delta & \eta^9, \tau(a) = 9 \\
\mathrm{IV} \ (\text{imaginary/BKM}) & >0 & 1 & \mathrm{II}_{2,3} & \kappa_{\mathrm{BKM}} = 5 \\
\mathrm V \ (\text{CHL }N\text{-twisted}) & >0 & N & \Lambda_N & \kappa_{\mathrm{BKM}}(\Phi_N) \\
\mathrm{VI} \ (\text{fibre-rank}) & \text{mixed} & 1 & \mathrm{II}_{4,20} & \kappa_{\mathrm{fiber}} = 24.
\end{array}
\]
\end{definition}

\begin{theorem}[Conditional orbit--route correspondence]
\label{thm:orbit-route}
\noindent\emph{Hypotheses.} Assume the $U$-duality stratification of the
type-IIA charge lattice identifies the relevant BPS charges by
$(D,\gcd)$, and assume each route's comparison map to the candidate
Hall--Drinfeld/Borcherds target is constructed on its stated locus.
The six routes R$1, \ldots,$ R$6$ of \S\ref{sec:six-routes} are in
correspondence with the six Narain orbit types
$(\mathrm I), \ldots, (\mathrm{VI})$ of
Definition~\ref{def:narain-orbits}:
R$i$ is the chiral-side avatar attached to the sublattice of orbit
$(i)$ via the corresponding Hodge, Heisenberg, Igusa, primitive
Borcherds, CHL-twisted, or fibre-rank construction.
\end{theorem}

\begin{proof}[Proof sketch]
Pass through the Narain moduli stratification of
type-IIA $K3 \times E$: primitive BPS states are Mukai vectors
in $\Lambda_{\mathrm{Nar}}$, and the BPS-spectrum generating function
$Z_{\mathrm{BPS}}(K3 \times E)$ is controlled, at the scalar level, by
the Igusa denominator (Oberdieck--Pandharipande) and has
Fourier coefficients stratified by orbit type. Each stratum $i$
has its own automorphic-lift construction producing a distinct
chiral or scalar avatar $\mathbf A^{(i)}$: $(\mathrm I)$ via
categorical Hodge supertrace, $(\mathrm{II})$ via Heisenberg
specialisation, $(\mathrm{III})$ via
Borcherds tensor squaring, $(\mathrm{IV})$ via Gritsenko paramodular
lift at $N=1$, $(\mathrm V)$ via CHL-twisted paramodular lifts,
$(\mathrm{VI})$ via relative fibre DMVV sewing. The bijection is
the organising statement; its upgrade to a common target is the
finite-height Hall--Drinfeld comparison problem.
\end{proof}

\section{Route separations}
\label{sec:route-separations}

\subsection{Categorical Euler vs fibre-rank (R$1$ vs R$6$)}
\label{subsec:route-mukai-fiber}

\emph{Question.} R$1$ uses the K3 real-root lattice witness
$\mathrm{II}_{3,19}$, while R$6$ uses the full Mukai rank
$\mathrm{rk}\,\widetilde H(K3,\mathbb Z)=24$. The common K3 source
can obscure the scalar distinction.

\emph{Resolution.} R$1$ computes the total-space Hodge supertrace
\[
\kappa_{\mathrm{cat}}(K3\times E)=
\chi(\mathcal O_{K3})\chi(\mathcal O_E)=2\cdot 0=0.
\]
The fibre value $\kappa_{\mathrm{cat}}(K3)=2$ is a witness in that
calculation. R$6$ computes the topological fibre-rank invariant
\[
\kappa_{\mathrm{fiber}}(K3\times E)=
\mathrm{rk}\,\widetilde H(K3,\mathbb Z)=1+22+1=24.
\]
The inclusion $\mathrm{II}_{3,19}\subset\mathrm{II}_{4,20}$ has
quotient $U$, and the two null directions are precisely what R$6$
retains and R$1$ excludes.

\subsection{Igusa squaring vs BKM (R$3$ vs R$4$)}
\label{subsec:route-igusa-bkm}

\emph{Question.} $\Phi_{10}^{\mathrm{un}} = \Delta_5^2$ by the Gritsenko--Nikulin
squaring; so R$3$ (Igusa $\Phi_{10}^{\mathrm{un}}$) is R$4$ (BKM $\Delta_5$)
raised to the second power. Are they the same construction?

\emph{Resolution.} The scalar shadow differs,
$\mathrm{wt}(\Phi_{10}^{\mathrm{un}})=10$ while
$\kappa_{\mathrm{BKM}}(\Delta_5) = c_1(0)/2 = 5$; and the
orbit differs, R$3$ lives on the light-like ray $D = 0$
with $\eta^9$-multiplicity datum $\tau(a) = 9$, R$4$ on the
imaginary orbit $D > 0$ with BKM root multiplicity
$c_1(4mn - \ell^2)$. The light-like $\eta^9$ datum is
$\tau(a) = 9 = c_1(0) - 1 = 2\kappa_{\mathrm{BKM}} - 1$, a
shifted shadow of R$4$ by one unit. The squaring map is a
character-level relation between automorphic forms; the primitive
BKM branch remains the $\Delta_5$ denominator.

\subsection{CHL-twisted BKM vs primitive BKM (R$5$ vs R$4$)}
\label{subsec:route-chl-bkm}

\emph{Question.} The R$4$ denominator and the R$5$ CHL siblings are all
Borcherds products with the same formal weight formula
$c_N(0)/2$.

\emph{Resolution.} R$4$ is the primitive $N=1$ denominator
$\Delta_5=\Phi_1$, with $c_1(0)=10$ and
$\kappa_{\mathrm{BKM}}(\Delta_5)=5$. R$5$ changes the Jacobi input by
a symplectic K3 automorphism and gives
$c_N(0)=(8,6,4,2)$ for $N=(2,3,4,6)$, hence weights
$(4,3,2,1)$. The Monster and Fake-Monster constructions on
$\mathrm{II}_{25,1}$ are dimension-stratified moonshine siblings,
with Fake-Monster weight $12$; they belong to a different
dimension-stratified Borcherds branch from the $K3\times E$
Hall--Drinfeld/BKM construction.

\subsection{Hall--Drinfeld/BKM vs K3 self-mirror Yangian}
\label{subsec:hall-drinfeld-vs-yangian}

\emph{Question.} The local K3 and ADE/Kummer charts carry Yangian
language, and the compact $K3\times E$ BKM branch is often compared
with a Hall positive half.

\emph{Resolution.} The compact BKM-side object is the
Hall--Drinfeld double of the compact critical Hall positive half,
with primitive Lie superalgebra $\mathfrak g_{\Delta_5}$ after the
finite Hall--Borcherds recognition data are supplied. The
self-mirror K3 Yangian is a separate Mukai-lattice current branch
valid on K3/ADE/Kummer charts. The equality
$\mathrm{CoHA}(\mathbb C^3)=Y^+$ is a toric positive-half statement;
the full affine $\mathcal W_{1+\infty}$ or doubled Yangian appears
only after Drinfeld-double, centre, or evaluation operations.

\subsection{CoHA vs MO vs BFN (are R$2$'s equivalents
multiple?)}
\label{subsec:route-coha-mo-bfn}

\emph{Question.} The Schiffmann--Vasserot CoHA, the Maulik--Okounkov
stable-envelope construction, the Braverman--Finkelberg--Nakajima
Coulomb branch, and the Kummer-orbifold free-field VOA all produce
algebras of relevance to $G(K3 \times E)$ and all are labelled
``R$2$-like''. Are they $4$ separate constructions, making the
true route count $9$?

\emph{Resolution.} They are $1$-morphisms between R$2$ and its avatars
within the Heisenberg/Hall orbit $(\mathrm{II})$. Schiffmann--Vasserot
CoHA supplies a Hall positive half. Maulik--Okounkov stable envelopes
produce the associated $R$-matrix via stable bases on
$\mathrm{Hilb}^n(K3)$. BFN Coulomb branches give a deformation
quantisation of the relevant Coulomb branch. Kummer free fields give
an orbifold-point model. These comparison avatars share the
Heisenberg scalar $\kappa_{\mathrm{ch}}^{\mathrm{Heis}}=3$ on the
witnessed locus. Their full compact comparison requires the same
Hall--Drinfeld, orientation, PBW, and completion data as the BKM
branch.

\subsection{Single-functor unification}
\label{subsec:route-single-functor}

\emph{Question.} Six constructions compute four invariants;
the surplus suggests that a unifying functor exists that produces
a single algebraic output carrying all four scalar shadows.

\emph{Resolution.} The obstruction is already visible at the
$1$-categorical level.

\begin{proposition}[Single-target $1$-functor obstruction]
\label{prop:no-single-functor}
A symmetric-monoidal functor
$\Phi^{\mathrm{gen}} \colon \mathrm{CY\text{-}cat}_3 \to
\mathrm{Alg}$ into a single $1$-category of algebras cannot have a
scalar shadow simultaneously computing all four of
$\kappa_{\mathrm{cat}}$, $\kappa_{\mathrm{ch}}^{\mathrm{Heis}}$,
$\kappa_{\mathrm{BKM}}$, $\kappa_{\mathrm{fiber}}$ on $K3 \times E$.
\end{proposition}

\begin{proof}
The four scalars are functions on four different kinds of data.
$\kappa_{\mathrm{cat}}$ is defined on the total CY category and is
K\"unneth-multiplicative:
$\kappa_{\mathrm{cat}}(K3\times E)=2\cdot 0=0$.
$\kappa_{\mathrm{ch}}^{\mathrm{Heis}}$ is defined after choosing the
Stage-$2$ Heisenberg--Mukai specialisation and reads a Cartan rank.
$\kappa_{\mathrm{BKM}}$ is defined after choosing a Borcherds input
and reads $c_N(0)/2$. $\kappa_{\mathrm{fiber}}$ is defined only
after choosing the K3 fibration and reads the Mukai rank of the
fibre. A scalar shadow of a single symmetric-monoidal $1$-functor has
one source type and one multiplicativity law. These four assignments

\end{proof}

\texorpdfstring{$\sigma \in N_6(\mathcal G_{\mathrm{mn}})$}{sigma in N6}]

$\mathcal G_{\mathrm{mn}}$, and assume the U-duality lattice embeddings

\section{6d holomorphic Chern--Simons on $K3 \times E$: six boundary
reductions}
\label{sec:6dhCS}

\begin{definition}[6d hCS on a CY$_3$]
\label{def:6dhCS}
For a Calabi--Yau $3$-fold $X$ with holomorphic volume form $\Omega$,
the Costello $6$d holomorphic Chern--Simons theory at gauge algebra
$\mathfrak g$ has action
\[
S_{\mathrm{hCS}}[A]
= \int_{X} \Omega \wedge \bigl(\tfrac{1}{2} A \,\bar\partial A
+ \tfrac{1}{3} A \, [A, A]\bigr),
\]
where $A \in \Omega^{0, \ast}(X, \mathfrak g)$ is a $\mathfrak g$-valued
Dolbeault form. The theory is topological in the antiholomorphic
direction and holomorphic in the holomorphic direction;
factorisation algebras of observables on $X$ are $E_3$-algebras in
the antiholomorphic directions but carry additional
$\bar\partial$-structure in the holomorphic directions. The boundary
algebra is extracted by specialising to a codimension-$2$ cycle
$\Sigma \subset X$.
\end{definition}

\begin{theorem}[Conditional boundary reductions of 6d hCS on $K3 \times E$]
\label{thm:six-reductions}
\noindent\emph{Hypotheses.} Assume the oriented hCS-to-Hall comparison,
the six boundary specialisation functors, and the comparison maps to
the Hall--Drinfeld/Borcherds target are constructed on the stated
loci.
Let $X = K3 \times E$. Corresponding to the six Narain orbits
$(\mathrm I), \ldots, (\mathrm{VI})$ of
Definition~\ref{def:narain-orbits} there are six boundary-reduction
procedures of the 6d hCS theory on $X$, each producing a distinct
chiral-side avatar:
\begin{enumerate}[label=(B\arabic*)]
\item (B$1$) Reduction to $\mathrm{Pic}(K3)$-cycles: traces primitive
$(-2)$-classes of the K3 fibre and records
$\kappa_{\mathrm{cat}}(K3\times E)=0$, with
$\kappa_{\mathrm{cat}}(K3)=2$ as the fibre witness.
\item (B$2$) Reduction along the $E$-cycle: traces the Heisenberg
$\hbar$-deformation along the elliptic base; boundary is the
$E_1$-chiral algebra $\mathbf H_{\Delta_5}$ with
$\kappa_{\mathrm{ch}}^{\mathrm{Heis}} = 3$. This is the boundary
dual to $\Phi$ on $\mathrm{Perf}(K3 \times E)$.
\item (B$3$) Squared doubling: takes two copies of B$2$ and contracts
along the diagonal Borcherds tensor square; boundary is
$V_{\Phi_{10}^{\mathrm{un}}}$ with Igusa weight $10$.
\item (B$4$) Reduction via paramodular period integration:
integrates $\Omega \wedge A^3$ against the paramodular form $\Phi_N$;
boundary is $V_0(\mathfrak g_{\Delta_5})$ with
$\kappa_{\mathrm{BKM}} = 5$ at $N = 1$.
\item (B$5$) CHL-twisted reduction: inserts the order-$N$
symplectic K3 automorphism into the boundary Hall datum; boundary is
the conditional CHL sibling $\mathbf H_{\Delta_5}^{(N)}$ with
$\kappa_{\mathrm{BKM}}(\Phi_N)=(4,3,2,1)$ for
$N=(2,3,4,6)$.
\item (B$6$) Relative reduction over $E$: fibre-by-fibre pushforward
$\pi_\ast$ of 6d hCS on $K3 \times E \to E$; boundary is the
$24$-boson lattice VOA $V_{\Lambda_{\mathrm{Muk}}}$ on $E$ with
$\kappa_{\mathrm{fiber}} = 24$.
\end{enumerate}
The six boundary-reduction procedures are related by the homotopy
coherence of the $6$-simplex $\sigma \in N_6(\mathcal G_{\mathrm{mn}})$
of Theorem~\ref{thm:six-simplex}.
\end{theorem}

\begin{proof}[Proof sketch]
The six procedures are the six boundary conditions consistent with
the $\mathrm{CY}_3$ structure on $X$ and the action of the Narain
U-duality on the $\mathrm{II}_{4,20} \oplus \mathrm{II}_{1,1}(-1)$
charge lattice. For each orbit $i$, the corresponding boundary
cycle is the primitive lattice vector representative modulo
$O(\Lambda_{\mathrm{Nar}})$-stabiliser; the boundary algebra is
extracted by evaluating the 6d hCS partition function along this
cycle and reading the factorisation-algebra restriction to the
boundary. That the six boundary algebras assemble into a coherent
homotopy-commutative diagram is exactly the conditional content of
Theorem~\ref{thm:six-simplex}.
\end{proof}

\begin{remark}[$E_n$-structure on the boundary]

\end{remark}
