\begin{abstract}
Donaldson--Thomas, Pandharipande--Thomas, and Gromov--Witten
theories attach three curve-counting series to a smooth projective
Calabi--Yau threefold.  A trace formulation packages those series as
images of three proper dualisable modules over a common ambient
\(\Ethree\)-factorisation algebra \(\cF_X\), with values in an
\(\Etwo\)-centre, under supplied orientation, cyclic trace, quantum
master equation, descent, completion, and comparison data.  The theorem
loci used here are nonsingular toric Calabi--Yau threefolds and the
reduced primitive \(K3\times E\) branch.  For a general compact
Calabi--Yau threefold the Maulik--Nekrasov--Okounkov--Pandharipande
comparison remains conjectural.  The variable change \(-q=e^{iu}\),
the Behrend weighting, the DT/PT wall crossing, and the reduced
K3-fibred comparison are part of the theorem data.

For \(X=K3\times E\), the reduced Oberdieck--Pandharipande scalar
belongs to the scalar partition function.  With
\(\Dfive=64^{-1}\DeltaFive\) and
\(\PhiTen^{\mathrm{OP}}=\Dfive^2\), this scalar is
\[
  Z_{\mathrm{OP/DT}}
  =-(\PhiTen^{\mathrm{OP}})^{-1}
  =-\Dfive^{-2}
  =-4096\,\DeltaFive^{-2}.
\]
This partition-function scalar is separate from
\(\kcat(K3\times E)=0\), \(\kchcomp(K3\times E)=0\),
\(\kchHeis=3\), \(\kBKM(\DeltaFive)=5\), and \(\kfib=24\).
\end{abstract}

\section{Three Curve-Counting Theories}

Let \(X\) be a smooth projective Calabi--Yau threefold and let
\(\beta\in H_2(X,\ZZ)\) be an effective curve class.

\begin{definition}[Donaldson--Thomas series]
The ideal-sheaf moduli space
\[
  I_n(X,\beta)=\Hilb(X,\beta,n)
\]
parametrises subschemes \(Z\subset X\) with
\([Z]=\beta\) and \(\chi(\cO_Z)=n\).  Its symmetric perfect
obstruction theory carries Behrend's constructible function
\(\nu_{I_n(X,\beta)}\).  The numerical DT invariant is
\[
  \DT_{n,\beta}(X)
  =
  \int_{[I_n(X,\beta)]^{\vir}}1
  =
  \chi\bigl(I_n(X,\beta),\nu_{I_n(X,\beta)}\bigr),
\]
with the second equality in Behrend's convention.  The partition
function is
\[
  Z_X^{\DT}(q,Q)
  =
  \sum_{\beta,n}\DT_{n,\beta}(X)\,q^n Q^\beta .
\]
The reduced DT series is
\[
  Z_X^{\DT\prime}(q,Q)
  =
  \frac{Z_X^{\DT}(q,Q)}
       {Z_{X,0}^{\DT}(q)},
  \qquad
  Z_{X,0}^{\DT}(q)=M(-q)^{\chi_{\mathrm{top}}(X)},
\]
where \(M(q)=\prod_{m\geq 1}(1-q^m)^{-m}\) is MacMahon's function.
\end{definition}

\begin{definition}[Pandharipande--Thomas series]
The stable-pair moduli space \(P_n(X,\beta)\) parametrises pairs
\[
  \cO_X \longrightarrow F
\]
with \(F\) pure of dimension one, \([F]=\beta\), and
\(\chi(F)=n\).  Its virtual class defines
\[
  \PT_{n,\beta}(X)
  =
  \int_{[P_n(X,\beta)]^{\vir}}1,
  \qquad
  Z_X^{\PT}(q,Q)
  =
  \sum_{\beta,n}\PT_{n,\beta}(X)\,q^n Q^\beta .
\]
The DT/PT comparison is the wall-crossing identity
\[
  Z_X^{\DT\prime}(q,Q)=Z_X^{\PT}(q,Q)
\]
in the numerical Calabi--Yau threefold settings covered by the
Joyce--Song, Toda, and Bridgeland wall-crossing theorems.
\end{definition}

\begin{definition}[Gromov--Witten series]
The stable-map moduli stack \(\overline{\cM}_g(X,\beta)\) has virtual
dimension zero.  The primary GW invariant is
\[
  \GW_{g,\beta}(X)
  =
  \int_{[\overline{\cM}_g(X,\beta)]^{\vir}}1 .
\]
The disconnected reduced partition function is written
\[
  Z_X^{\GW}(u,Q)
  =
  \exp\!\left(
    \sum_{\beta\neq 0}\sum_{g\geq 0}
      \GW_{g,\beta}(X)\,u^{2g-2}Q^\beta
  \right)
\]
with the usual constant-map convention separated from the nonzero
curve-class series.
\end{definition}

\begin{theorem}[MNOP comparison on theorem loci]
For every smooth projective Calabi--Yau threefold \(X\) in a proved
MNOP locus, including the nonsingular toric locus and the reduced
primitive \(K3\times E\) locus after the Oberdieck--Pandharipande
comparison, the equality
\[
  Z_X^{\GW}(u,Q)
  =
  Z_X^{\DT\prime}(q,Q)\big|_{-q=e^{iu}}
\]
holds after the MNOP change of variables and after the degree-zero DT
factor has been divided out.  The same comparison can be read through
PT theory as
\[
  Z_X^{\GW}(u,Q)
  =
  Z_X^{\PT}(q,Q)\big|_{-q=e^{iu}}.
\]
For a general smooth projective Calabi--Yau threefold the displayed
equality is the MNOP conjecture.
\end{theorem}

\begin{remark}[The substitution]
The equality \(-q=e^{iu}\) relates the DT Euler-characteristic
variable \(q\) to the GW loop variable \(u\).  The sign is tied to the
Behrend-weighted symmetric obstruction theory, while the exponential
form records the passage from connected GW invariants to the
disconnected partition function.  The substitution is part of the
comparison theorem.
\end{remark}

\section{Trace Maps}

Let \(\cF_X\) denote an ambient \(\Ethree\)-factorisation algebra
attached to the Calabi--Yau threefold \(X\).  On constructed
chain-level loci this is the holomorphic-Chern--Simons or
Costello--Li observable algebra.  In the categorical lane it is the
\(\Ethree\)-monoidal structure carried by the Calabi--Yau category
\(\Perf(X)\), subject to the stated finiteness and orientation
hypotheses.

\begin{definition}[Trace comparison datum]
A trace comparison datum for \(X\) consists of:
\[
  \bigl(\cF_X;\,
  M_{\DT}^X,M_{\PT}^X,M_{\GW}^X;\,
  \theta,\tau_{\mathrm{cyc}},S_{\QME},\mathfrak c,\widehat{(-)},\rho\bigr).
\]
Here \(M_{\DT}^X,M_{\PT}^X,M_{\GW}^X\) are proper dualisable
\(\cF_X\)-modules; \(\theta\) is an orientation or square-root datum for
the symmetric obstruction theories; \(\tau_{\mathrm{cyc}}\) is the
cyclic trace pairing; \(S_{\QME}\) is the BV action with quantum master
equation and anomaly cancellation; \(\mathfrak c\) is the chosen
counterterm system; \(\widehat{(-)}\) is the Novikov, \(q\)-adic, or
automorphic completion in which the traces converge; and \(\rho\) is
the family of sewing and comparison maps relating the DT, PT, and GW
characters.
\end{definition}

\begin{definition}[\(\Ethree\)-trace target]
Let \(A\) be an \(\bE_3\)-algebra in a symmetric monoidal
\((\infty,1)\)-category and let \(M\) be a dualisable \(A\)-module.
Dunn--Lurie additivity gives a trace
\[
  \Tr_A^{\bE_3}(M,\id_M)
  \in
  \End_A^{\bE_2}(\mathbf 1_A).
\]
The target \(\End_A^{\bE_2}(\mathbf 1_A)\) is the \(\Etwo\)-centre
in which the numerical partition function appears after applying the
character or Euler-characteristic map.
\end{definition}

\begin{conjecture}[Trace formulation of MNOP]
Assume \(X\) carries a trace comparison datum and that the DT, PT, and
GW moduli theories determine the dualisable \(\cF_X\)-modules
\[
  M_{\DT}^X,\qquad M_{\PT}^X,\qquad M_{\GW}^X
\]
whose trace characters reproduce the Behrend-weighted DT series, the
stable-pair series, and the GW series.  Then MNOP is the assertion
that the diagram
\[
\begin{tikzcd}[column sep=large]
  \Tr_{\cF_X}^{\bE_3}(M_{\DT}^X)
    \arrow[r, "{/\,M(-q)^{\chi_{\mathrm{top}}(X)}}"]
    \arrow[dr, "\sigma_{\MNOP}"']
  &
  \Tr_{\cF_X}^{\bE_3}(M_{\PT}^X)
    \arrow[d, "\sigma_{\MNOP}"]
  \\
  &
  \Tr_{\cF_X}^{\bE_3}(M_{\GW}^X)
\end{tikzcd}
\]
commutes after applying the character map to
\(\End_{\cF_X}^{\bE_2}(\mathbf 1)\), where
\(\sigma_{\MNOP}\) implements \(-q=e^{iu}\).
\end{conjecture}

\begin{remark}[Hypotheses carried by the trace statement]
The common ambient algebra \(\cF_X\) alone gives a common trace target.
Equality of trace images requires the comparison theorem and the full
trace comparison datum.  The dualisability of \(\Coh(I_n(X,\beta))\),
\(\Coh(P_n(X,\beta))\), and
\(\Coh(\overline{\cM}_g(X,\beta))\) as \(\cF_X\)-modules is a separate
geometric input.  Smooth and proper categorical dualisability results
cover the curve-counting stacks only after the orientation, cyclic
trace, descent, sewing, and completion hypotheses are checked.
\end{remark}

\begin{proposition}[Local and compact hCS inputs]
On a formal polydisc of a Calabi--Yau threefold, the holomorphic
Chern--Simons BV complex
\[
  \Obs_{\hCS}^q(U)
\]
has an \(\Ethree\)-factorisation structure after the Costello
renormalisation data solve the quantum master equation.  For compact
\(X\), the global trace comparison additionally requires a compact
quantisation package: compatible local counterterms, anomaly
cancellation for the chosen volume form, nuclear or Novikov completion
of observables, descent over the factorisation cover, and sewing
compatibility for boundary strata of the curve-counting moduli spaces.
\end{proposition}

\begin{proof}
Locality supplies the \(\Ethree\)-factorisation products on disjoint
polydiscs.  The BV bracket and BV Laplacian define the obstruction to
quantisation; the quantum master equation cancels this obstruction
after the counterterms are chosen.  Compactness introduces infinite
sums over curve classes and boundary strata, so the trace must be
taken in a completed target.  Descent and sewing ensure that the local
traces glue to a global class in
\(\End_{\cF_X}^{\bE_2}(\mathbf 1)\).
\end{proof}

\section{K3 \(\times\) E}

Let \(X=K3\times E\).  The reduced curve-counting theory uses the
holomorphic two-form on the K3 factor and the elliptic factor \(E\).
The scalar partition function and the five CY-to-chiral numbers
belong to different constructions.

\begin{proposition}[The reduced OP scalar]
With
\[
  [q^{1/2}r^{1/2}s^{1/2}]\DeltaFive=64,
  \qquad
  \Dfive=64^{-1}\DeltaFive,
\]
the monic Oberdieck--Pandharipande scalar square is
\[
  \PhiTen^{\mathrm{OP}}=\Dfive^2=4096^{-1}\DeltaFive^2.
\]
The reduced DT/PT scalar is
\[
  Z_{\mathrm{OP/DT}}
  =
  -(\PhiTen^{\mathrm{OP}})^{-1}
  =
  -\Dfive^{-2}
  =
  -4096\,\DeltaFive^{-2},
\]
with the sign fixed by the Behrend-weighted orientation convention.
\end{proposition}

\begin{proof}
The coefficient condition makes \(\Dfive\) the monic
Oberdieck--Pandharipande product.  Squaring gives
\(\PhiTen^{\mathrm{OP}}\), so
\[
  (\PhiTen^{\mathrm{OP}})^{-1}
  =
  \Dfive^{-2}
  =
  4096\,\DeltaFive^{-2}.
\]
The reduced DT/PT theorem on \(K3\times E\) inserts the Behrend sign,
giving the displayed scalar.
\end{proof}

\begin{proposition}[The five distinct K3 \(\times\) E numbers]
The following numbers have separate origins:
\[
\begin{array}{c|c|c}
\text{symbol} & \text{value} & \text{construction} \\
\hline
\kcat(K3\times E) & 0
  & \chi(\cO_{K3})\,\chi(\cO_E)=2\cdot 0 \\
\kchcomp(K3\times E) & 0
  & \sum_q(-1)^q h^{0,q}(K3\times E)=1-1+1-1 \\
\kchHeis & 3
  & \text{Mukai--Heisenberg Stage-2 specialization} \\
\kBKM(\DeltaFive) & 5
  & c_1(0)/2=10/2 \\
\kfib & 24
  & \mathrm{rank}\,H^*(K3,\ZZ)
\end{array}
\]
The scalar \(Z_{\mathrm{OP/DT}}\) lies in partition-function data and
leaves the five table entries in separate construction slots.
\end{proposition}

\begin{proof}
The \(\kcat\) value is Kunneth multiplicativity for the holomorphic
Euler characteristic.  The compact \(\kchcomp\) value is the Hodge
supertrace.  The Heisenberg value counts the Stage-2 Mukai--Heisenberg
specialization.  The Borcherds value is the weight of the primitive
Gritsenko form \(\DeltaFive\), read as \(c_1(0)/2\).  The fibre value
is the rank of the K3 Mukai lattice.
\end{proof}

\section{Borcherds Constants and the OP Square}

Let \(N\in\{1,2,3,4,6\}\).  Let \(c_N(0)\) be the
\(\zeta^0q^0\)-coefficient of the CHL Borcherds input weak Jacobi
form in Eichler--Zagier normalization.

\begin{proposition}[Primitive CHL constants]
The primitive constants are
\[
  (c_1(0),c_2(0),c_3(0),c_4(0),c_6(0))
  =
  (10,8,6,4,2),
\]
and hence
\[
  \bigl(\kBKM(\Phi_1),\kBKM(\Phi_2),\kBKM(\Phi_3),
        \kBKM(\Phi_4),\kBKM(\Phi_6)\bigr)
  =
  (5,4,3,2,1).
\]
At \(N=1\), \(\Phi_1=\DeltaFive\),
\[
  c_1(0)=10,\qquad \kBKM(\DeltaFive)=5,\qquad
  \Dfive=64^{-1}\DeltaFive .
\]
These are Borcherds-weight constants.  The OP scalar uses only the
\(N=1\) monic square
\[
  \PhiTen^{\mathrm{OP}}
  =
  (64^{-1}\DeltaFive)^2.
\]
\end{proposition}

\begin{proof}
The \(N=1\) coefficient is read from
\[
  \phi_{0,1}(\tau,z)|_{q^0}=\zeta^{-1}+10+\zeta .
\]
Borcherds' weight formula gives
\(\mathrm{wt}(\Phi_N)=c_N(0)/2\) in the Borcherds-denominator
normalization.  The CHL ladder gives the remaining coefficients
\(8,6,4,2\), so the primitive weights are \((5,4,3,2,1)\).  The
Gritsenko--Clery eight-form catalogue uses its own
triple-indexed \((t,N;k)\) normalization; the OP square uses the
monic \(N=1\) normalization above.
\end{proof}

\section{Generic Compact Calabi--Yau Threefolds}

For a compact Calabi--Yau threefold \(X\), four numerical surfaces
often sit near one another:
\[
  \kchcomp(X),\qquad
  \kcat(X),\qquad
  \kBCOV(X)=\frac{\chi_{\mathrm{top}}(X)}{24},\qquad
  Z_X^{\DT\prime},Z_X^{\PT},Z_X^{\GW}.
\]
They answer different questions.

\begin{proposition}[Compact CY3 separation]
For every compact Calabi--Yau threefold \(X\),
\[
  \kchcomp(X)=\sum_q(-1)^q h^{0,q}(X)=0,
  \qquad
  \kcat(X)=\chi(\cO_X)=0.
\]
The BCOV one-loop scalar is
\[
  \kBCOV(X)=\chi_{\mathrm{top}}(X)/24,
\]
which can be nonzero.  The MNOP series are partition functions in
curve variables.  A BKM denominator requires a separate Borcherds
input, root lattice, and denominator theorem.
\end{proposition}

\begin{proof}
Serre duality gives \(h^{0,q}(X)=h^{0,3-q}(X)\), so the alternating
Hodge supertrace vanishes.  The holomorphic Euler characteristic is
the same alternating sum.  The BCOV constant uses the topological
Euler characteristic.  DT, PT, and GW count curves through virtual
classes and Behrend-weighted Euler characteristics.  Borcherds
denominators come from automorphic products with prescribed Fourier
coefficients.
\end{proof}

\begin{remark}[Hall and centre layers]
For \(d\geq 3\), the specialized CY-to-chiral output is native
\(\Eone\).  The \(\Etwo\)-structure relevant to trace comparisons lives
on a centre, double, or module category such as
\(\cZ(\Rep^{\Eone}(A))\).  In the local \(\CC^3\) model, the CoHA gives
the positive half \(Y^+\); the \(\mathcal W_{1+\infty}\)-side belongs
to the double, centre, completion, and evaluation layer.  Thus the
chain
\[
  \CoHA(\CC^3)=Y^+
  \longrightarrow D(Y^+)
  \longrightarrow \cZ(\Rep^{\Eone}(Y^+))
  \longrightarrow \mathcal W_{1+\infty}
\]
records four separate operations: positive Hall integration,
Drinfeld doubling, centre formation, and Fock or vacuum evaluation.
MNOP numerics supply curve-counting partition functions. A Hall double,
BKM algebra, or chiral quantum group requires the additional Hall,
pairing, completion, and comparison data.
\end{remark}

\section{Resulting Formulation}

The trace formulation has the following precise content.

\begin{enumerate}
\item DT, PT, and GW are three theories with three moduli problems:
ideal sheaves, stable pairs, and stable maps.
\item DT uses Behrend weighting; the degree-zero DT factor is
\(M(-q)^{\chi_{\mathrm{top}}(X)}\).
\item PT removes the degree-zero DT factor through wall crossing.
\item GW is compared with DT or PT only after the MNOP substitution
\(-q=e^{iu}\).
\item A trace statement requires three dualisable modules over a common
\(\Ethree\)-factorisation algebra, cyclic trace data, QME/anomaly
cancellation, descent and sewing, counterterms, completion, comparison
maps, and a common \(\Etwo\)-centre target.
\item For \(K3\times E\), the reduced OP scalar is
\(-(\PhiTen^{\mathrm{OP}})^{-1}=-\Dfive^{-2}\), with
\(\Dfive=64^{-1}\DeltaFive\).  This scalar is separate from
\(\kcat=0\), \(\kchcomp=0\), \(\kchHeis=3\), \(\kBKM=5\), and
\(\kfib=24\).
\item For a generic compact Calabi--Yau threefold, MNOP supplies the
DT/PT/GW comparison after the variable substitution. A BKM denominator,
Hall double, or constructed CY-to-chiral quantum group requires separate
Hall and Borcherds data.
\item For \(d\geq 3\), the native \(\Phi\)-output is \(\Eone\);
\(\Etwo\)-structures arise at centre, double, module, or enhancement
layers.
\end{enumerate}
