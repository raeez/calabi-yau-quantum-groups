\subsection{Holomorphic Chern--Simons and categorical Hochschild presentations}
\label{wn:subsec:hcs-vs-cat-synthesis}

Let $X$ be a Calabi--Yau threefold with holomorphic volume form
$\Omega_X$, and let $\cC$ be a smooth proper Calabi--Yau category of
dimension $3$. Put
\[
  \cF_{\cC}:=\PhiFA_3(\cC)\in E_3\text{-}\mathrm{HolFA}(X).
\]
For a specialisation datum $(\Sigma_2,C)$, the chiral output is
\[
  A_{\cC,\Sigma_2,C}
  :=
  \Sp_{\Sigma_2,C}(\cF_{\cC})
  \in E_1\text{-}\mathrm{HolFA}(C).
\]
The braided $E_2$ layer belongs to
$\mathcal Z(\Rep^{E_1}(A_{\cC,\Sigma_2,C}))$, to a Drinfeld double, or
to a module enhancement after the relevant construction has been made.

\begin{definition}[Typed layers and allowed arrows]
\label{def:hcs-cat-typed-layers}
The comparison uses six typed layers:
\[
\begin{array}{ccl}
\mathsf{BV}^{\mathrm{loc}}_{\hCS}(U,\mathfrak g)
  &=& \Obs_{\hCS}^{q}(U,\mathfrak g),\\[0.4ex]
\mathsf{BV}^{\mathrm{glob}}_{\hCS}(X,\mathfrak g;\mathcal H_X)
  &=& \text{global quantum hCS built from }\mathcal H_X,\\[0.4ex]
\mathsf{CY}_3(\cC,\eta)
  &=& \text{smooth proper CY category with trace }\eta,\\[0.4ex]
Y^+_\sigma(X)
  &=& \text{critical Hall positive half on a stratum }\sigma,\\[0.4ex]
\mathsf{Cent}(A)
  &=& \mathcal Z(\Rep^{E_1}(A))\ \text{or}\ \cD(Y^+_\sigma(X)),\\[0.4ex]
\mathsf{Chir}(A)
  &=& A_{\cC,\Sigma_2,C}.
\end{array}
\]
The allowed comparison arrows are
\[
  \Theta_{\mathrm{oc}}\colon
  \mathsf{BV}^{\mathrm{loc}}_{\hCS}(U,\gl(V))
  \longrightarrow
  C^\bullet_{\mathrm{Hoch},E_3}(\Perf(U);\operatorname{End}(V)),
\]
\[
  \Theta_{\mathrm{glob}}\colon
  \mathsf{BV}^{\mathrm{glob}}_{\hCS}(X,\gl(V);\mathcal H_X)
  \longrightarrow
  \PhiFA_3(\Perf(X)),
\]
\[
  \Theta_{\mathrm{Hall}}\colon
  C^\bullet_{\mathrm{Hoch},E_3}(\Perf(X))
  \longrightarrow
  Y^+_\sigma(X),
  \qquad
  \Sp_{\Sigma_2,C}\colon
  \PhiFA_3(\cC)\longrightarrow A_{\cC,\Sigma_2,C}.
\]
Every equivalence below is produced by one of these arrows together
with its displayed hypotheses.
\end{definition}

The hCS--Hochschild comparison is local on Stein polydiscs. On a
compact threefold it requires a Costello--Li all-scale BV quantisation
package satisfying the quantum master equation, trivialisation of the
local anomaly class, orientation, framing, descent, sewing,
counterterms, analytic completion, and RG-compatible gauge fixing. The
absence of this package is the compact proof obligation.

\subsubsection{Input data}
\label{wn:subsubsec:hcs-cat-inputs}

\begin{definition}[Holomorphic Chern--Simons input]
\label{def:hcs-input}
An hCS input is a triple $(X,\mathfrak g,\Omega_X)$ in which $X$ is a
complex threefold with $\Omega_X\in H^0(X,K_X)$ nowhere vanishing,
$\mathfrak g$ is a finite-dimensional complex Lie algebra with invariant
symmetric pairing, and the BV field is
\[
  A\in\Omega^{0,\bullet}(X,\mathfrak g)[1].
\]
The classical action is
\[
  S_{\hCS}[A]
  =
  \int_X \Omega_X\wedge
  \operatorname{tr}
  \left(\frac12 A\dbar A+\frac13 A[A,A]\right),
\]
with Maurer--Cartan equation
$\dbar A+\frac12[A,A]=0$.
\end{definition}

\begin{definition}[Calabi--Yau category input]
\label{def:cy3-category}
A Calabi--Yau category of dimension $3$ is a $\mathbb C$-linear smooth
proper stable category $\cC$, or a smooth proper pre-triangulated
dg-category on a chain model, equipped with a non-degenerate cyclic trace
\[
  \eta\colon \HH_\bullet(\cC)\longrightarrow \mathbb C[-3]
\]
identifying the Serre functor with the shift $[3]$. For
$\cC=\Perf(X)$, $\eta$ is the Serre-duality residue pairing determined
by $\Omega_X$.
\end{definition}

\begin{definition}[Hall input]
\label{def:hall-input}
A Hall input on a stability or equivariant stratum $\sigma$ is a
critical cohomological Hall algebra
\[
  Y^+_\sigma(X)
  =
  H^\bullet_{\mathrm{crit},\sigma}
  (\mathcal M_{\mathrm{eff}}(X),\phi_W)
\]
with orientation data, vanishing-cycle sheaf, and associative Hall
product. For the toric chart $X=\mathbb C^3$ this input is the
Jordan triple-loop critical CoHA. Its algebra is the positive half
of the affine Yangian.
\end{definition}

\begin{definition}[Specialised chiral output]
\label{def:specialised-chiral-output}
For $d=3$ and $(\Sigma_2,C)$, the object
$A_{\cC,\Sigma_2,C}=\Sp_{\Sigma_2,C}(\PhiFA_3(\cC))$ is the native
$E_1$-chiral algebra on $C$. The $E_2$ structure used for braiding is
the centre, double, or module-enhancement layer attached to
$A_{\cC,\Sigma_2,C}$.
\end{definition}

\begin{remark}[Five separated Hochschild objects]
\label{rmk:five-objects-separated}
For a strict $E_1$ Hochschild model $A$ attached to $\cC$, the objects
\[
  A,\qquad B(A),\qquad A^i,\qquad A^!,\qquad
  Z^{\mathrm{der}}_{\mathrm{ch}}(A)
\]
play different roles. The identity $\Omega B(A)\simeq A$ is bar--cobar
inversion. The object $A^!$ is the Verdier or Koszul dual. The bulk
object is accessed through Hochschild theory and the derived chiral
centre. The comparison below keeps these objects separated.
\end{remark}

\subsubsection{Local open--closed comparison}
\label{wn:subsubsec:local-open-closed-comparison}

\begin{proposition}[Local hCS observables and Hochschild cochains]
\label{prop:local-hcs-hochschild-comparison}
Let $U\subset X$ be a Stein polydisc with coordinates
$z_1,z_2,z_3$ and volume form
$\Omega=dz_1\wedge dz_2\wedge dz_3$. Fix a vector space $V$ and gauge
algebra $\gl(V)$. Assume a formality point for the $E_3$ operad, a
Costello--Li anomaly-free hCS quantisation package on $U$, and completed
local observables. Then the open--closed map gives an $E_3$
quasi-isomorphism of local factorisation observables
\[
  \Obs_{\hCS}^{q}(U,\gl(V))
  \;\simeq\;
  C^\bullet_{\mathrm{Hoch},E_3}(\Perf(U);\operatorname{End}(V)).
\]
At the level of symbols, Hochschild--Kostant--Rosenberg identifies
\[
  \HH^\bullet(\Perf(U))
  \simeq
  \PV^\bullet(U)
  =
  \bigoplus_{p,q}\Omega^{0,q}(U,\wedge^p T^{1,0}U),
\]
and contraction by $\Omega$ sends $\PV^{p,q}(U)$ to
$\Omega^{3-p,q}(U)$. The hCS field complex
$\Omega^{0,\bullet}(U,\gl(V))[1]$ is the matrix-framed open-string
sector. The full polyvector complex is the closed-string Hochschild
sector acting on that open sector.
\end{proposition}

\begin{proof}
HKR is the antisymmetriser quasi-isomorphism from polyvector fields to
Hochschild cochains. The Calabi--Yau volume form converts polyvectors
to Dolbeault forms with complementary holomorphic degree. Matrix
framing tensors the open sector with $\operatorname{End}(V)$ and gives
the hCS dg Lie algebra
$\Omega^{0,\bullet}(U,\gl(V))[1]$ with differential $\dbar$ and
wedge--Lie bracket. Costello--Gwilliam locality turns an anomaly-free
BV package into a factorisation algebra of quantum observables. The
Costello open--closed map identifies closed Hochschild cochains with
local operations on the hCS open sector. The $V\to\infty$ stabilisation
is the usual trace-stable matrix limit.
\end{proof}

\begin{remark}[Raw complexes and observable algebras]
\label{rmk:raw-complexes-observables}
The equality in Proposition~\ref{prop:local-hcs-hochschild-comparison}
is an equality of completed $E_3$ observable algebras. The raw complexes
$\PV^\bullet(U)$ and $\Omega^{0,\bullet}(U,\gl(V))[1]$ have different
degrees and different meanings: the first is closed Hochschild bulk
data, the second is the open hCS field complex.
\end{remark}

\subsubsection{Formality and the associator}
\label{wn:subsubsec:formality-associator}

\begin{theorem}[Formality datum for the Stage-\texorpdfstring{$1$}{1} comparison]
\label{thm:stage-one-formality-datum}
After a point of the Kontsevich--Tamarkin formality torsor is fixed, the
local Hochschild cochain complex of a Calabi--Yau threefold carries the
$E_3$ structure used by $\Phi^{\mathrm{FA}}_3$. The ambiguity of the
chain-level comparison is governed by the prounipotent group
$\Gtone(\mathbb Q)$. A Costello--Li heat-kernel regulator on the flat
chart selects the Kontsevich configuration-integral point of this
torsor when its graph weights satisfy the all-scale QME and
boundary-stratum compatibility conditions.
\end{theorem}

\begin{proof}
Kontsevich formality identifies polyvectors and Hochschild cochains on
the formal disc. Tamarkin's refinement and the operadic formality of
little disks promote the comparison to the $E_3$ level once a Drinfeld
associator is chosen. Willwacher identifies the degree-zero graph-complex
symmetries with $\mathfrak{grt}_1$, giving the $\Gtone$ action. In the
Costello--Li hCS regulator, the small-scale propagator on $\mathbb C^3$
has Bochner--Martinelli leading term. Pulling graph weights to
Fulton--MacPherson compactifications gives the same configuration-space
integrals as the chosen Kontsevich formality point, provided the QME and
boundary terms close.
\end{proof}

\subsubsection{Compact Calabi--Yau threefolds and the anomaly slot}
\label{wn:subsubsec:compact-cy3-anomaly-slot}

\begin{definition}[Compact comparison package]
\label{def:compact-comparison-package}
For compact $X$, an hCS--Hochschild comparison package consists of:
\[
  \mathcal H_X =
  \left(
  I[L],\Delta_L,\QME,\mathcal A_{\QME},
  \mathcal K,\mathrm{or},\mathrm{fr},\mathrm{desc},
  \mathrm{TCFT},\widehat{\mathrm{comp}},\mathrm{gf},
  \Theta_{\mathrm{oc}}
  \right).
\]
Here $I[L]$ are scale-dependent effective interactions, $\Delta_L$ is
the BV Laplacian, $\mathcal A_{\QME}$ is the local QME anomaly class,
$\mathcal K$ is the counterterm space in the same local functional
complex, $\mathrm{or}$ and $\mathrm{fr}$ are orientation and framing
data, $\mathrm{desc}$ is descent data, $\mathrm{TCFT}$ denotes the
factorisation or TCFT sewing datum, $\widehat{\mathrm{comp}}$ is
analytic completion, $\mathrm{gf}$ is regulator-compatible gauge
fixing, and $\Theta_{\mathrm{oc}}$ is the open--closed Hochschild
comparison.
\end{definition}

\begin{proposition}[Quartic QME anomaly and degree-two two-point term]
\label{prop:quartic-qme-quadratic-counterterm}
For six-dimensional non-abelian hCS on a complex threefold, the
one-loop QME anomaly is represented by a quartic invariant
\[
  I_4^\rho\in\operatorname{Sym}^4(\mathfrak g^\vee)^{\mathfrak g}
\]
attached to the chosen representation $\rho$. On a coordinate chart
$U$, its local cocycle has the form
\[
  \Theta^\rho_U(\alpha_0,\alpha_1,\alpha_2,\alpha_3)
  =
  \int_U
  I_4^\rho
  \bigl(\alpha_0,\partial\alpha_1,
  \partial\alpha_2,\partial\alpha_3\bigr),
  \qquad
  \alpha_i\in\Omega^{0,\bullet}_c(U,\mathfrak g).
\]
The degree-two invariant $C_2^\rho(\mathfrak g)$ appears in the
two-point field-strength counterterm
\[
  S_{\mathrm{ct}}^{(2)}
  =
  -\,\hbar\,\frac{C_2^\rho(\mathfrak g)}{(4\pi)^3}
  \log(L/\varepsilon)
  \int_X\Omega_X\wedge\operatorname{Tr}_\rho(A\dbar A).
\]
This term renormalises the kinetic pairing. The QME obstruction is the
degree-one class $[\hbar\Theta^\rho]$ in the Costello local functional
complex.
\end{proposition}

\begin{proof}
The hCS cubic vertex has one field and three holomorphic derivatives in
the one-loop anomaly descent on a complex threefold, so its invariant
polynomial has arity four. The two-point graph has arity two and
contracts through the invariant bilinear form, producing the
$C_2^\rho(\mathfrak g)$ kinetic counterterm. The quartic descent class
and the two-point counterterm live in different cohomological slots of
the BV local functional complex.
\end{proof}

\begin{proposition}[Compact proof obligations]
\label{prop:compact-proof-obligations}
For compact $X$, a global hCS presentation of
$\PhiFA_3(\Perf(X))$ requires:
\begin{enumerate}
\item the local QME for the chosen quartic tensor $I_4^\rho$;
\item a primitive for $\mathcal A_{\QME}$ in the same local functional
complex, or a proof of its vanishing;
\item orientation and framing data for the BV pairing;
\item determinant-line, BCOV, or TCFT background data;
\item descent and sewing for the factorisation algebra;
\item counterterms compatible with the regulator and RG flow;
\item analytic OPE completion.
\end{enumerate}
A vanishing Euler coefficient verifies the scalar BCOV entry alone. The
representation-dependent quartic class remains part of the QME problem.
\end{proposition}

\begin{corollary}[Compact presentation theorem under the package]
\label{cor:compact-presentation-package}
Let $X$ be compact and let $\mathcal H_X$ be a compact comparison
package in the sense of Definition~\ref{def:compact-comparison-package}.
If $\mathcal A_{\QME}=0$, or if a counterterm primitive kills
$\mathcal A_{\QME}$ in the same local functional complex, then hCS
observables and categorical Hochschild cochains give equivalent
presentations of $\Phi^{\mathrm{FA}}_3(\Perf(X))$ in the completed
holomorphic $E_3$ homotopy category.
\end{corollary}

\begin{proof}
The QME makes the renormalised BV differential square-zero at every
scale. RG compatibility identifies the factorisation algebras obtained
from different scales. The orientation, framing, TCFT sewing,
completion, and gauge-fixing data allow the local open--closed maps of
Proposition~\ref{prop:local-hcs-hochschild-comparison} to glue over
$X$. The resulting glued object is the Stage-$1$ factorisation algebra
attached to $\Perf(X)$.
\end{proof}

\begin{remark}[The product \texorpdfstring{$K3\times E$}{K3 x E}]
\label{rmk:k3xe-anomaly}
For $X=K3\times E$, the tangent characteristic contribution
$c_3(T_X)$ vanishes by Kunneth: $T_E\simeq\cO_E$ and $K3$ has complex
dimension $2$. This removes the geometric $c_3$ contribution to the
compact anomaly calculation. The gauge-theoretic quartic slot
$I_4^\rho$ remains part of the QME package and must vanish or be
trivialised for the chosen representation. The scalar lanes attached
to the $K3\times E$ comparison are:
\[
  \kcat(K3\times E)=0,\qquad
  \kch(K3\times E)=0,\qquad
  \kch^{\mathrm{Heis}}=3,\qquad
  \kBKM(\mathfrak g_{\Delta_5})=\frac{c_1(0)}2=5,\qquad
  \kfib=24.
\]
The first two coincide numerically here and remain separate functionals:
$\kcat$ is the categorical Euler characteristic, while compact
$\kch$ is the Hodge supertrace
$\sum_q(-1)^q h^{0,q}(K3\times E)=
\chi(\cO_{K3})\chi(\cO_E)=2\cdot0=0$.
\end{remark}

\subsubsection{Traces and enumerative theories}
\label{wn:subsubsec:trace-identities}

\begin{proposition}[DT, GW, and PT traces]
\label{prop:dt-gw-pt-traces}
Assume the compact comparison package of
Definition~\ref{def:compact-comparison-package}. Then Donaldson--Thomas,
Gromov--Witten, and Pandharipande--Thomas theories define trace
functionals on the completed Stage-$1$ object
$\Phi^{\mathrm{FA}}_3(\Perf(X))$:
\[
  Z_{\DT}(X;q)=
  \operatorname{Tr}_{\DT}^{E_3}
  \left(\int_X \Phi^{\mathrm{FA}}_3(\Perf(X))\right),
\]
\[
  Z_{\GW}(X;u)=
  \operatorname{Tr}_{\GW}^{E_3}
  \left(\int_X \Phi^{\mathrm{FA}}_3(\Perf(X))\right),
\]
and
\[
  Z_{\PT}(X;q)=
  \operatorname{Tr}_{\PT}^{E_3}
  \left(\int_X \Phi^{\mathrm{FA}}_3(\Perf(X))\right).
\]
The DT/PT correspondence identifies the DT and PT traces under its
usual hypotheses. MNOP identifies the reduced DT trace with the GW
trace after $q=-e^{iu}$ on the proven toric locus and remains
conjectural for general compact threefolds.
\end{proposition}

\begin{proof}
The PTVV shifted symplectic form on the moduli of objects in
$\Perf(X)$ gives the virtual orientation data behind DT and PT counts.
The stable-map moduli give the GW trace. The compact package supplies
the same completed Stage-$1$ factorisation algebra to each trace. The
known DT/PT and toric MNOP theorems identify the resulting trace
functionals after their standard changes of variables and reductions.
\end{proof}

\begin{remark}[Borcherds weight from the trace on \texorpdfstring{$K3\times E$}{K3 x E}]
\label{rmk:k3xe-trace-weight}
On primitive classes of $K3\times E$, the reduced DT trace is governed
by the Oberdieck--Pixton and Gritsenko--Nikulin--Igusa product. The
Borcherds weight invariant is
\[
  \kBKM(\Phi_N)=\frac{c_N(0)}2,
\]
so the $N=1$ row has $c_1(0)=10$ and
$\kBKM(\Phi_1)=5$. This invariant is separate from
$\kcat$, $\kch$, and $\kfib$.
\end{remark}

\subsubsection{Local \texorpdfstring{$\mathbb C^3$}{C3} CoHA and the chiral output}
\label{wn:subsubsec:c3-coha-output}

\begin{proposition}[\texorpdfstring{$\CoHA(\mathbb C^3)$}{CoHA(C3)} and the positive half]
\label{prop:c3-coha-positive-half}
For the Jordan triple-loop quiver with cubic potential,
Schiffmann--Vasserot identify the critical cohomological Hall algebra
with the positive half of the affine Yangian:
\[
  \CoHA(\mathbb C^3)\cong Y^+(\widehat{\mathfrak{gl}}_1).
\]
The full $\mathcal W_{1+\infty}$ algebra is reached through the typed
chain
\[
  Y^+(\widehat{\mathfrak{gl}}_1)
  \longrightarrow
  \mathcal D\bigl(Y^+(\widehat{\mathfrak{gl}}_1)\bigr)
  \longrightarrow
  \mathcal W_{1+\infty},
\]
where $\mathcal D$ denotes the Drinfeld double and the final arrow
uses a completed Fock or evaluation representation.
\end{proposition}

\begin{proof}
The critical CoHA multiplication is the shuffle multiplication on
symmetric functions with the three equivariant parameters
$\epsilon_1,\epsilon_2,\epsilon_3$. This is the positive half
$Y^+(\widehat{\mathfrak{gl}}_1)$. Negative modes and the Cartan current
enter after forming the double and choosing the representation that
realises the vertex algebra.
\end{proof}

\begin{corollary}[Geometric and categorical presentations]
\label{cor:hcs-geometric-resolution}
Under the hypotheses stated above, hCS is the geometric BV presentation
of the same Stage-$1$ object that categorical Hochschild theory presents
through $\HH^\bullet(\Perf(X))$. The typed chain is
\[
  \Obs_{\hCS}^{q}
  \xrightarrow{\Theta_{\mathrm{oc}}}
  C^\bullet_{\mathrm{Hoch},E_3}(\Perf(X))
  \xrightarrow{\Theta_{\mathrm{glob}}}
  \PhiFA_3(\Perf(X))
  \xrightarrow{\Sp_{\Sigma_2,C}}
  A_{\Perf(X),\Sigma_2,C}.
\]
The hCS presentation supplies propagators, RG flow, QME, counterterms,
and Feynman weights. The categorical presentation supplies the
Hochschild $E_3$ structure, Serre-duality trace, centre, and
open--closed action. The last object is the $E_1$ chiral output for
$d=3$.
\end{corollary}

\begin{proof}
The local statement is Proposition~\ref{prop:local-hcs-hochschild-comparison}.
The associator and formality dependence are recorded in
Theorem~\ref{thm:stage-one-formality-datum}. Compact targets require
Definition~\ref{def:compact-comparison-package} and
Corollary~\ref{cor:compact-presentation-package}. The passage from
$\CoHA(\mathbb C^3)$ to the chiral vertex algebra is typed by
Proposition~\ref{prop:c3-coha-positive-half}. The operadic level after
specialisation is the $E_1$ level prescribed by the two-stage
factorisation theorem for $d\geq3$.
\end{proof}
