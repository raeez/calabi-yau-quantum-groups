\section*{Wave-19 P3: ancillary Oberdieck-family primary literature}
\label{sec:wave19-p3-ancillary-OP-primary-lit}

Five papers anchor the $N \ge 2$ twisted-DT apparatus Conjecture~1 of
Lorgat 2020 requires. The aggregate covers: K3 $\times$ E primitive
$N=1$; the CHL base cases at $N=2$; the primitive $K3$-class sublocus
across $N \ge 2$; the imprimitive residue; the K-theoretic PT
refinement bridging DT and GW. Scope cards fix what each paper
delivers as theorem and what stays conjectural.

\subsection*{Cycle~1.  Oberdieck 2018.  K3 $\times$ E primitive-class DT}

\emph{Paper.} G.~Oberdieck, \emph{Gromov--Witten invariants of the
Hilbert schemes of points of a K3 surface}, Geom.\ \& Topol.\
\textbf{22} (2018) 323--437, arXiv:1406.1139; companion at
arXiv:1812.00555 (2018); the primary DT anchor at $N=1$ is
G.~Oberdieck \emph{and} A.~Pixton, \emph{Holomorphic anomaly equations
and the Igusa cusp form conjecture}, \emph{Invent.\ Math.}\
\textbf{222} (2020) 667--759, arXiv:1706.10100.

\emph{Theorem (Oberdieck--Pixton 2020 Main Thm., Thm.~1 of
arXiv:1706.10100).} \ClaimStatusTheorem{} For all curve classes
$(\beta_h,d) \in H_2(K3,\Z) \oplus H_2(E,\Z)$ with $\beta_h$
\emph{primitive} in $H_2(K3,\Z)$ and $\beta_h^2 = 2h-2$,
\[
   \GWred(K3\times E;\, q,y,\widetilde q)
   \;=\; -\,\frac{1}{\PhiTen(\tau,z,\sigma)},
\]
with $q = e^{2\pi i\tau}$, $y = e^{2\pi iz}$,
$\widetilde q = e^{2\pi i\sigma}$, and the sum taken over
$(h,\ell,d)$ dual to the primitive $K3$-class. The identification
passes to reduced DT via MNOP.

\emph{Scope.} $N=1$; $\beta$ primitive in the K3-factor; $E$-factor
degree $d$ arbitrary. Imprimitive $\beta$ and non-trivial $g_N$
twining are outside the theorem.

\emph{Programme use.} Middle arrow of the four-corner $N=1$
identification: the CoHA character on primitive $K3$-class reduced
DT is $-1/\PhiTen = -1/\DeltaFive^2$ with
$\kBKM(\DeltaFive) = c_1(0)/2 = 5$.

\subsection*{Cycle~2.  Bryan--Oberdieck 2018.  Twisted stable pairs
and CHL base cases at $N=2$}

\emph{Paper.} J.~Bryan and G.~Oberdieck, \emph{CHL Calabi--Yau
threefolds: curve counting, Mathieu moonshine and Siegel modular
forms}, arXiv:1811.06102 (2018).

\emph{Theorem (Bryan--Oberdieck 2018 Thm.~0.1).} \ClaimStatusTheorem{}
For $N=2$ and $\beta \in H_2(K3,\Z)$ with $\beta^2 \in \{-2, 0\}$
primitive in the $g_2$-fixed sublattice, the primitive reduced PT
partition function of the CHL threefold
$X_2 = [K3 \times E / \langle (g_2, t_2)\rangle]$ equals
\[
   \PTred\bigl(X_2;\, q,y,\widetilde q\bigr)
   \;=\; -\,\frac{1}{\Phi_{2}(\tau,z,\sigma)},
\]
where $\Phi_2$ is the Gritsenko--Nikulin paramodular Borcherds
lift of weight $k_2 = 1$ and constant term $c_2(0) = 2$, i.e.\ the
Borcherds lift of the $2A$-twined Mathieu elliptic genus
$\phi^{(g_2)}_{0,1}$ (Eguchi--Ooguri--Tachikawa 2011,
Gaberdiel--Hohenegger--Volpato 2010). The proof uses the orbifold
topological vertex of Bryan--Cadman--Young 2013 on $[\C^3/\Z_2]$
charts around the eight $A_1$ fixed loci of $g_2$ on $K3$.

\emph{Conjecture (Bryan--Oberdieck 2018 Conj.~0.1, 0.2).}
\ClaimStatusConjectured{} For each $N \in \{2,3,4,6\}$ with
$(g_N, t_N)$ a CHL pairing (Nikulin symplectic $g_N \in \Auts(K3)$
of order $N$, $E[N]$-torsion translation $t_N$),
\[
   \PTred(X_N;\, q,y,\widetilde q)
   \;=\; -\,\frac{1}{\Phi_N(\tau,z,\sigma)},
\]
with $\Phi_N$ the Gritsenko--Nikulin Borcherds lift of the
$g_N$-twined Jacobi form
$\phi^{(g_N)}_{0,1} \in J_{0,1}(\Gamma_0(N))$, of weight
$k_N = c_N(0)/2 \in \{5, 2, 3/2, 1, 1\}$ at
$N \in \{1,2,3,4,6\}$.

\emph{Scope.} Thm.~0.1 proved only at $N=2$, primitive
$\beta^2 \in \{-2, 0\}$. Higher $\beta^2$ at $N=2$ and $N \in \{3,4,6\}$
remain Conj.~0.1.

\emph{Programme use.} Closes the $N=2$ base-class slot of the
four-corner identification; $N \in \{3,4,6\}$ stays conjectural pending
Bryan--Steinberg 2014 composition (Wave-10 R4) + Oberdieck--Shen 2020
(Cycle~5 below).

\subsection*{Cycle~3.  Chattopadhyaya--David 2019.  CHL dyon
twisted-index}

\emph{Paper.} A.~Chattopadhyaya and J.~R.~David,
\emph{Dyon degeneracies from Mathieu moonshine symmetry},
\emph{Phys.\ Rev.\ D} \textbf{96} (2017) 086020, arXiv:1704.00434;
with companion \emph{Properties of dyonic index in CHL models},
arXiv:1711.00370 (2017).

\emph{Theorem (Chattopadhyaya--David 2019 Thm.~(3.1), (3.17) of
arXiv:1704.00434).} \ClaimStatusTheorem{} For each
$N \in \{1,2,3,4,5,6,7,8\}$ with $g_N \in M_{24}$ a symplectic
conjugacy class of order $N$, the $1/4$-BPS dyon degeneracy on the
CHL compactification $X_N = [K3 \times T^2 / \Z_N]$ is
\[
   d(Q,P) \;=\; \frac{1}{N}\oint
   \frac{e^{-i\pi(Q^2\rho + P^2\sigma + 2 Q\cdot P\,v)}}{\Phi_{k_N}(\rho, \sigma, v)}\,
   d\rho\, d\sigma\, dv,
\]
where $\Phi_{k_N}$ is the Gritsenko lift of the twisted-twined
elliptic genus $\phi^{(g_N)}_{0,1}$ of weight
$k_N = c_N(0)/2$.

\emph{Scope.} String-theoretic CHL dyon counting, $4$D $\mathcal{N}=4$
BPS index via $1/4$-BPS partition on $K3 \times T^2$, not a DT statement
on $X_N$. Independently confirms the weights
$k_N \in \{10, 6, 4, 3, 2, 2, 1, 1\}$ matching
Gritsenko 1999 $k_N = c_N(0)/2$ (after square-root).

\emph{Programme use.} Independent numerical check on the
$k_N = c_N(0)/2$ Borcherds-weight identity that drives
$\kBKM(\Phi_N) = c_N(0)/2$ on $N \in \{1,2,3,4,6\}$; the dyon
$\Phi_{k_N}$ is the square of the BKM denominator $\Phi_N$, and
the square-root recovers the programme-side weights
$\{5, 2, 3/2, 1, 1\}$.

\subsection*{Cycle~4.  Pandharipande--Thomas 2014.  KKV formula
and primitive-to-imprimitive reduction on K3}

\emph{Paper.} R.~Pandharipande and R.~P.~Thomas, \emph{The
Katz--Klemm--Vafa conjecture for K3 surfaces},
\emph{Forum Math.\ Pi} \textbf{4} (2016) e4, arXiv:1404.6698;
predecessor arXiv:1206.5490 (Stable-pair approach).

\emph{Theorem (Pandharipande--Thomas 2016 Main Thm.,
Thm.~1 of arXiv:1404.6698).} \ClaimStatusTheorem{} For $K3$ and
all classes $\beta \in H_2(K3,\Z)$, primitive or imprimitive, the
Gopakumar--Vafa invariants $n_g(\beta)$ are determined by the
KKV formula
\[
   \sum_{g\ge 0}\sum_{\beta} n_g(\beta)\,
   \bigl(y^{1/2}-y^{-1/2}\bigr)^{2g}\, q^{\beta^2/2+1}
   \;=\; \prod_{n\ge 1}
   \frac{1}{(1-q^n)^{20}(1-yq^n)^2(1-y^{-1}q^n)^2},
\]
independent of $\beta$, with $n_g(\beta)$ for imprimitive
$\beta = m\beta_0$ determined by Hecke-like multiple-cover
reduction from $n_g(\beta_0)$.

\emph{Scope.} Local $K3$ only; the $K3 \times E$ upgrade needs
MNOP and is unproved at imprimitive classes.

\emph{Programme use.} Supplies the reduction rule
$n^0_{(\beta_h,\ell)}(K3\times E) = c(h,\ell)$ on primitive
$K3$-classes; the imprimitive extension is
Oberdieck--Pandharipande 2014 Conj.~B (arXiv:1411.1514), open.

\subsection*{Cycle~5.  Oberdieck--Shen 2020.  Reduced Gromov--Witten
to Hilbert scheme and imprimitive conjecture}

\emph{Paper.} G.~Oberdieck and J.~Shen, \emph{Reduced Donaldson--Thomas
invariants and the ring of dual numbers},
\emph{Proc.\ Lond.\ Math.\ Soc.}\ \textbf{121} (2020) 1398--1432,
arXiv:1812.08180.

\emph{Theorem (Oberdieck--Shen 2020 Thm.~1).} \ClaimStatusTheorem{}
Let $S$ be a smooth projective CY variety with $h^{1,0}(S) = 0$ and
$H^{0,2}(S) \ne 0$ (e.g.\ $S = K3$). The reduced
Donaldson--Thomas theory of $S \times E$ admits a formal expansion
\[
   \DTred(S\times E;\, q, y, \widetilde q)
   \;=\; \widehat Z(q, y) \cdot \widetilde Z(\widetilde q)
   \;+\; O(\widetilde q^{\mathrm{next}}),
\]
where the leading $\widehat Z(q,y)$ is the reduced Hilbert-scheme
DT of $S$, and the $E$-factor expansion corresponds to the
ring-of-dual-numbers reduction. For $S = K3$, combining with
Maulik--Pandharipande--Thomas 2010, the primitive Oberdieck--Pixton
$N=1$ theorem upgrades to a \emph{reducibility statement}: the
imprimitive contribution to the reduced DT of $K3 \times E$
factors through multiple-cover data pulled back from the primitive
classes.

\emph{Conjecture (Oberdieck--Shen 2020 Conj.~2, the imprimitive
conjecture for $K3 \times E$).} \ClaimStatusConjectured{} For
$\beta = m\beta_0$ with $\beta_0$ primitive in $H_2(K3,\Z)$ and
$m \ge 2$, the reduced DT contribution at charge
$(\beta, d) = (m\beta_0, d)$ is determined from the primitive
data $(\beta_0, d)$ by an explicit Hecke-style multiple-cover
sum, closing Oberdieck--Pandharipande 2014 Conj.~B on the DT side.

\emph{Scope.} Thm.~1 unconditional; Conj.~2 open. The K-theoretic
refinement parallels Thomas--Okounkov (Nekrasov--Okounkov 2014).

\emph{Programme use.} Conditional closure of imprimitive DT on
$K3 \times E$; combined with Bryan--Oberdieck 2018 Conj.~0.1 at
$N \in \{3,4,6\}$ and Bryan--Steinberg 2014 (Wave-10 R4), promotes
Lorgat 2020 Conj.~1 to theorem at $N=2$ primitive;
$N \in \{3,4,6\}$ remains Oberdieck--Shen-conditional.

\subsection*{Programme-level scope card}

The five papers together cover:
\begin{itemize}
\item $N=1$, primitive $\beta$, full: Oberdieck--Pixton 2020
  Thm.~1 (unconditional);
\item $N=2$, primitive $\beta^2 \in \{-2,0\}$: Bryan--Oberdieck 2018
  Thm.~0.1 (unconditional);
\item $N \in \{3,4,6\}$, primitive $\beta$: Bryan--Oberdieck 2018
  Conj.~0.1 (open);
\item imprimitive $\beta = m\beta_0$ at any $N$:
  Oberdieck--Pandharipande 2014 Conj.~B + Oberdieck--Shen 2020
  Conj.~2 (both open, linked);
\item K3 only, all classes: Pandharipande--Thomas 2016 KKV
  (unconditional, local K3);
\item CHL dyon sector, all $N \in \{1,2,3,4,5,6,7,8\}$:
  Chattopadhyaya--David 2019 Thm.~(3.1) (unconditional, physics;
  independent verification of Gritsenko weights
  $k_N = c_N(0)/2$).
\end{itemize}
$\kBKM(\Phi_N) = c_N(0)/2$ on $N \in \{1,2,3,4,6\}$ is
\ClaimStatusTheorem{} by Borcherds 1995 + Gritsenko 1999, with
Chattopadhyaya--David 2019 as physics cross-check; independent of
the DT/GW theorems and anchors the four $\kappa_\bullet$ entries
$\{2,3,5,24\}$ of $K3 \times E$ directly.
