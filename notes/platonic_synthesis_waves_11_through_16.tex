\subsection{The canonical object: two-stage factorisation of $\Phi$}
\label{wn:subsec:plat-two-stage}

\begin{theorem}[Two-stage factorisation]\ClaimStatusTheorem
\label{wn:thm:plat-two-stage}
The CY-to-chiral functor decomposes canonically:
\[
\Phi_d \;=\; \Sp_{\Sigma_{d-1},C} \circ \Phi^{\mathrm{FA}}_d : \CYcat_d
\xrightarrow{\;\Phi^{\mathrm{FA}}_d\;} E_d\text{-}\HolFA(X)
\xrightarrow{\;\Sp_{\Sigma_{d-1},C}\;} E_1\text{-}\ChirAlg(C).
\]
Stage~1 is canonical up to contractible choice (Kontsevich--Tamarkin
$E_d$-formality; Costello--Gwilliam--Li locality). Stage~2 is a
specialisation: factorisation homology over a $(d{-}1)$-cycle
$\Sigma_{d-1} \subset X$, restricted to a reference curve $C \subset X$.
Different $(\Sigma_{d-1}, C)$ produce different $E_1$-chiral shadows.
\end{theorem}

\begin{corollary}[``Many BKMs from one CY$_3$'' is theorem-grade]
\label{wn:cor:plat-many-bkms}
Borcherds Monster and Igusa $\fg_{\Delta_5}$ are sibling $E_1$-specialisations
of a common $E_3$-hFA framework, differentiated by $(\Sigma_2, C)$ choice.
The Fake Monster is a $d=5$ cousin, obstructed from $d=3$ by a rank
count: the Leech lattice $\Lambda_{\mathrm{Leech}}$ has rank $24$, while
$h^{1,1}(K3) = 20$; no transverse surface $\Sigma_2$ in a compact CY$_3$
supplies the Niemeier data.
\end{corollary}
