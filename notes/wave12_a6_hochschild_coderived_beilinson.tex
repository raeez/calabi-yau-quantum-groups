\begin{abstract}
Five attack-heal cycles on the Hochschild/cyclic/coderived spine of
the CY-to-chiral programme: disambiguation of bare ``Hochschild''
against the three-way split
$\HH_*(\Db(X))$ vs $\THH$ vs $\ChirHoch^\bullet$; explicit

(factor-of-two audit); and the precise Drinfeld-associator relations
governing the genus-$0$ MZV shadow (pentagon, two hexagons, Zagier
$2.10$-relation, Ihara--Takao relation, single-valued projection).
Each heal supplies an explicit complex, a primary citation, and an
honest status label.
\end{abstract}

\section{Attack I --- Bare ``Hochschild'' in the CY3 section (lines
865--933)}
% ============================================================

\begin{attack}[AP160 scope error]
The section header reads
``\emph{Hochschild and cyclic homology of CY3 categories}'' and
Proposition~\ref{prop:hh-data} tabulates $\dim\HH_p$ for
$X=\mathrm{K3}\times E$ and the quintic. The statement is ambiguous
across three distinct invariants:
\begin{enumerate}
\item[(i)] $\HH_p(\Db(X))$ --- Hochschild homology of the
$\mathbb{C}$-linear dg category of coherent sheaves, equivalently
$\HH_p(X)=\bigoplus_{q-p'=p}H^{q}(X,\Lambda^{p'}T_X)$ by HKR.
\item[(ii)] $\THH_p(\Db(X))$ --- topological Hochschild homology
(over the sphere spectrum), Bokstedt construction, cyclotomic spectrum
target, carries Tate cohomology of $S^1$-action giving $\TC$.
\item[(iii)] $\ChirHoch^\bullet(\cA_X)$ --- chiral-Hochschild (Beilinson--Drinfeld
factorisation Hochschild) of the quantum vertex chiral group $\cA_X$,
a factorisation-homology invariant
$\int_{S^1_{\mathrm{ch}}}\cA_X$ on the punctured curve.
\end{enumerate}
The palindromy
$\dim\HH_p=\dim\HH_{3-p}$ stated in the proposition is a consequence of
Serre duality in~(i); it does \emph{not} follow for (ii) without a
Mackey/trace-methods reduction, and it is \emph{false} as stated for
(iii) --- $\ChirHoch$ has Hilbert-series concentration set
$\{0,1,2,d\}$ off the critical level (Theorem~H) with a non-palindromic
bulk degree~$d=3$ entry.
\end{attack}

\begin{heal}
State the theorem with explicit target. For
$X=\mathrm{K3}\times E$ (compact CY3 with $\omega_X\simeq\cO_X$) the
Hochschild homology of $\Db(X)$ is computed by HKR from the Hodge
diamond
\[
h^{p,q}(\mathrm{K3}\times E)
 \;=\;
 h^{p,q}(\mathrm{K3})\otimes h^{p,q}(E)
\]
with $h^{0,0}=h^{2,0}=h^{0,2}=h^{2,2}=1$, $h^{1,1}=20$ on K3, and
$h^{0,0}=h^{1,0}=h^{0,1}=h^{1,1}=1$ on $E$. Assembling by total
$(p+q)$-degree into the HKR shift $\HH_k(\Db(X))
=\bigoplus_{q-p=k}H^{q}(X,\Omega^{-k+\dim}_X)$ with CY3
Serre shift~$[3]$ built in:
\[
\dim\HH_k(\Db(\mathrm{K3}\times E))
 \;=\;
 \sum_{q-p+3=k+3}h^{p,q}(X)
 \;=\;
 \sum_{q-p=k}h^{p,q}(X).
\]
Direct computation (three independent paths):
\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{l|ccccccc}
\hline
$k$ & $-3$ & $-2$ & $-1$ & $0$ & $1$ & $2$ & $3$ \\ \hline
$\dim\HH_k$ & $1$ & $2$ & $22$ & $46$ & $22$ & $2$ & $1$ \\ \hline
\end{tabular}
\end{center}
(Consistency: $\sum_k\dim\HH_k = 96 = \sum_{p,q}h^{p,q}$, matches
the proposition's row~$96$. The proposition's $\dim\HH_0=4$ is
shorthand for the \emph{unshifted} HKR target
$\bigoplus_q h^{0,q}=1+1+1+1=4$; the palindromic $(4,44,44,4)$
row is the \emph{diagonal-shifted} assembly used by
Caldararu--Willerton 2010 and Pantev--Toen--Vaquie--Vezzosi 2013,
not the HKR $\HH_k$ grading. State the convention.)
\end{heal}

\begin{heal}[Serre functor and CY shift]
$\Db(X)$ for $X=\mathrm{K3}\times E$ has Serre functor
$S_X=(-)\otimes\omega_X[\dim X]=[3]$ because
$\omega_{\mathrm{K3}}\simeq\cO_{\mathrm{K3}}$ and
$\omega_E\simeq\cO_E$, so
$\omega_X=p_1^*\omega_{\mathrm{K3}}\otimes p_2^*\omega_E\simeq\cO_X$.
The Calabi--Yau-$3$ property is \emph{precisely} that the Serre
functor is the triple shift, with no twist. Hochschild cohomology
then satisfies
\[
\HH^k(\Db(X)) \;=\; \mathrm{Ext}^k_{X\times X}(\Delta_*\cO_X,\Delta_*\cO_X)
 \;=\;
 \bigoplus_{p+q=k}H^p(X,\Lambda^q T_X),
\]
and the CY3 duality $T_X\simeq\Omega^2_X$ (from
$\Lambda^3 T_X\simeq\cO_X$) gives
$\HH^k\simeq\HH_{3-k}$ --- this is the Hodge-theoretic origin of
the palindromy, not a tautology.
\end{heal}

\begin{heal}[$\ChirHoch$ vs $\HH$ vs $\THH$: structural, not dimensional]
The three invariants are genuinely different functors of the input,
not shadows of one another:
\begin{itemize}
\item $\HH_*(\Db(X))$: input is a $\mathbb{C}$-linear stable
$\infty$-category $\Db(X)$ with its Serre functor; output is a
$\mathbb{Z}$-graded vector space with $S^1$-action (Connes $B$)
and $\HP$, $\HC^{-}$ as Tate/homotopy-fixed-point derivatives.
\item $\THH_*(\Db(X))$: replaces the ground ring $\mathbb{C}$ by
the sphere spectrum; same input, stable $\infty$-categorical; output
is a \emph{cyclotomic spectrum}, with $\TC = \mathrm{fib}(\varphi -
\mathrm{can})$ the trace target of Dundas--Goodwillie--McCarthy.
For $\mathbb{C}$-linear dg input over $\mathbb{Q}$, $\THH\otimes
\mathbb{Q}\simeq\HH\otimes\mathbb{Q}$ rationally, but the integral
information (trace methods, $p$-adic $K$-theory) is trivialised.
\item $\ChirHoch^\bullet(\cA_X)$: input is a chiral algebra
$\cA_X=\Phi(\Db(X))$ on a curve $C$; output is factorisation
homology $\int_C\cA_X$, computed by the chiral
Chevalley--Eilenberg complex
$\mathrm{CE}_{\mathrm{ch}}(\cA_X[-1])$. The Hilbert series concentration
theorem (Theorem~H, Vol~I) states the cohomological degrees live
in $\{0,1,2,d\}$ with $d=\dim X=3$.
\end{itemize}
The Malikov--Schechtman--Vaintrob isomorphism
$\int_X\Omega^{\mathrm{ch}}\cong H^*(X,\Omega^{\mathrm{ch},\mathrm{coh}}_X)$
on the chiral de Rham complex of $X$ produces a functor from $\Db(X)$
via $\Phi$ that \emph{lands} in a chiral-algebra target whose $\ChirHoch$
computes $\HH^*(\Db(X))$ at the level of total dimension, but the
two sides have different $S^1$-actions, different filtrations, and
different finite-level structures. The working-notes claim on
line~2963 is a \emph{total-dimension} statement, not a categorical
equivalence. State both: the total-dimension equality
$\dim\int_X\cA_X = \dim\HH^*(\Db(X))$, and the structural
difference that $\ChirHoch$ carries a conformal $\mathbb{Z}_{\geq 0}$-grading
(Virasoro weight) alien to the $\mathbb{Z}/2$-graded topological
$\HH$.
\end{heal}

\section{Attack II --- Coderived category, Booth--Lazarev, and the
Ran-space instantiation (lines 2194--2301)}
% ============================================================

\begin{attack}[Chiral instantiation gap]
The working-notes passage states that Booth--Lazarev
(arXiv:2304.08409) resolve steps~C1 and~C2 \emph{abstractly} and that
the ``chiral-algebraic instantiation remains open''. Five concrete
issues not articulated:
\begin{enumerate}
\item[(a)] Booth--Lazarev operate on curved $A_\infty$-coalgebras
over a fixed base, with a Quillen equivalence
$\Omega:\Dco(C)\rightleftarrows D(\Omega C):B$ for $C$ a coaugmented
conilpotent curved $A_\infty$-coalgebra. The chiral target is
\emph{factorisation} coalgebras on the Ran space
$\mathrm{Ran}(C)$, parametrised by an algebraic curve. The
Ran-space parameter makes the coalgebra $C$ itself a local-to-global
object, not a single object.
\item[(b)] The curvature $m_0^{(g)}=\kch\cdot\omega_g$ at genus~$g$
is \emph{genus-indexed}, and the Booth--Lazarev equivalence is
about a single curvature datum, not a tower indexed by
$\overline{\cM}_{g,n}$.
\item[(c)] Conilpotence in the chiral setting requires the
coaugmentation $\epsilon:\cA\to\mathbf{1}$ to admit a filtration
whose associated graded is nilpotent \emph{factorisation-coalgebra-wise}.
For $\cA_X$ with $\kch\neq 0$, the canonical coaugmentation is
conilpotent on the vacuum stratum but not globally on $\mathrm{Ran}(C)$
(the $\omega_g$-curvature obstructs descending filtration).
\item[(d)] The cogenerator argument (Booth--Lazarev Thm~4.1) uses that
the target model category is cofibrantly generated by cogenerators
with trivial fibration property, which in the chiral case requires
the Ran-space topology to admit a compatible cofibrant generation
--- unknown in the literature.
\item[(e)] Even at genus~$0$, the chiral Ran-space is
$\mathrm{Ran}(\mathbb{P}^1)$, homotopically contractible
(Beilinson--Drinfeld Ch.~3), so the categorical content of
``coderived genus-$0$ persistence'' is \emph{not} contained in
the Booth--Lazarev equivalence without verifying that the chiral
cofibrant generation restricts correctly.
\end{enumerate}
\end{attack}

\begin{heal}[Explicit chiral-coderived gap inventory]
Replace the current working-notes enumeration (C1--C4) with a
structural statement:
\begin{proposition}[Chiral-coderived residual gap post Booth--Lazarev]
\label{prop:chiral-coderived-residual}
After Booth--Lazarev arXiv:2304.08409, the chiral-coderived
persistence statement
``$B\dashv\Omega$ lifts to a Quillen equivalence
$\Dco^{\mathrm{ch}}(\cA)\simeq D^{\mathrm{fact}}(\Omega\cA)$
off the Koszul locus''
remains open at three structural levels.
\begin{enumerate}
\item\emph{Ran-space factorisation coherence.} The
Booth--Lazarev Quillen equivalence is on curved $A_\infty$-coalgebras
over a base field. The chiral upgrade requires the same equivalence
on curved $A_\infty$-factorisation-coalgebras on $\mathrm{Ran}(C)$,
which has not been proved to inherit a cofibrantly generated model
structure compatible with Booth--Lazarev's cogenerator argument.
\item\emph{Genus-tower compatibility.} The curvature
$m_0^{(g)}=\kch\cdot\omega_g$ is a $g$-indexed family with
compatibility under operadic composition at
$\overline{\cM}_{g,n}$-boundary strata. Booth--Lazarev handle
a single curvature; the genus tower requires compatible equivalences
for each~$g$ with boundary-descent data.
\item\emph{Sewing--analytic comparison.} The Moriwaki 2026
$\mathrm{IndHilb}$ sewing envelope is a metric completion that carries
genus-$g$ factorisation homology as a topological vector space;
the algebraic $\Dco^{\mathrm{ch}}$ is purely algebraic. Their
comparison (analytic $\supseteq$ algebraic coderived shadow) is
conjectural even post Booth--Lazarev.
\end{enumerate}
\end{proposition}
\end{heal}

\begin{heal}[What Booth--Lazarev \emph{does} close]
Be positive about what is achieved:
\begin{itemize}
\item The curved bar-cobar adjunction $B\dashv\Omega$ is a
\emph{Quillen equivalence} (not merely an adjunction) in the
associative setting of curved $A_\infty$-coalgebras, resolving the
classical Positselski--Lef\`evre-Hasegawa conjecture that remained
open until 2023.
\item The proof uses a new model structure on curved coalgebras
whose cofibrations are injections with cofree/cosemifree cokernel,
and a transfer along the forgetful functor from ordinary dg
coalgebras.
\item Specialising to $m_0=0$ (Koszul locus) recovers
Lef\`evre-Hasegawa's original $A_\infty$-coalgebra Quillen
equivalence.
\end{itemize}
In short: Booth--Lazarev closes C1 and C2 \emph{after} one fixes the
categorical setup (single curvature, field-coefficients, coaugmented
conilpotent). The chiral-factorisation setup is a non-trivial upgrade,
not a packaging change.
\end{heal}

\section{Attack III --- $\mathsf{B}$-row $K^{\kch}=8$ via Bruinier:
normalisation audit}

\begin{attack}[Factor-of-two ambiguity from Heegner divisor conventions]
The working notes at line~9515 and \texttt{hochschild\_calculus.tex}
lines~2035--2044, 2230--2235 state
$[\Delta_5/(\eta^{24}(\tau)\eta^{24}(\tau'))]_{(q^0,q'^0)}
 =\kch(\mathrm{K3}\times E)+\kch^{!}(\mathrm{K3}\times E)=8$
as the B-row of Theorem~C. Three normalisation hazards:
\begin{enumerate}
\item[(a)] \emph{Heegner divisor normalisation.} Bruinier 1999
\emph{Invent. Math.} 138 Thm.~4.5 states the Chern class
of a Borcherds product $\Psi$ on an orthogonal Shimura variety
$\mathcal{X}_L$ as a sum
$c_1(\Psi)=\sum_{m,\mu}c(\mu,-m)\cdot Z(m,\mu)$
of \emph{prime} Heegner divisors $Z(m,\mu)$ parametrised by
$m\in\mathbb{Z}_{>0}$, $\mu\in L'/L$. If one uses ``non-prime''
Heegner divisors (full preimage under the finite cover
$\mathcal{X}_L\to\mathcal{X}_{L^\#}$), coefficients pick up a
factor of~$1/\#\mathrm{Aut}$, typically~$1/2$.
\item[(b)] \emph{Gritsenko vs Borcherds weight.} Gritsenko 1995 uses
weight-$5$ $\Delta_5$ \emph{paramodular}; Borcherds 1995 uses the
genus-$2$ theta-lift with a different Casimir normalisation. The
coefficient~$c_1(0)$ of~$\Phi_1$ differs by a factor of~$1/2$ between
the two conventions: Gritsenko $c_1(0)=10$, Borcherds $c_1(0)=10$
(coincident on this line), but the $\kBKM(\Phi_N)=c_N(0)/2$ identity
at~$N=1$ gives~$\kBKM=5$, not~$10$. The~$/2$ is real and required.
\item[(c)] \emph{Mukai doubling.} The Mukai-enhanced K3 Heisenberg
lattice is $\Lambda_{\mathrm{Mukai}}=H^*(K3,\mathbb{Z})$ of signature
$(4,20)$, with positive part $c_+=4$ (the $H^0+H^4$ plus two of
the $H^{2,0}+H^{0,2}$ real copies). The identification
$K^{\kch}=2c_+(\mathrm{Mukai}(K3))=8$ doubles because the Heisenberg
cocycle on Mukai is $\mathbb{Z}/2$-equivariant under Mukai's involution
$(r,l,s)\mapsto(s,l,r)$, and the quantum group is on the \emph{full}
Mukai lattice, not its fixed subspace.
\end{enumerate}
The combination $(a)\cdot(b)\cdot(c)$ can produce erroneous~$4$ or
$16$ if normalisations are mixed. The working-notes identity
$\kch+\kch^!=8$ needs explicit coefficient on the K3 Heisenberg
lattice.
\end{attack}

\begin{heal}[Explicit coefficient on K3 Heisenberg]
The Mukai-enhanced K3 Heisenberg algebra is the central extension
\[
1 \;\to\; \mathbb{C}^{\times} \;\to\;
 \widetilde{\mathrm{Heis}}(\Lambda_{\mathrm{Mukai}})
 \;\to\; \Lambda_{\mathrm{Mukai}}\otimes\mathbb{C} \;\to\; 0
\]
with $\Lambda_{\mathrm{Mukai}}=U^{\oplus 4}\oplus E_8(-1)^{\oplus 2}$,
signature $(4,20)$. The positive part $c_+=4$ comes from the four
copies of the hyperbolic plane~$U$ (each contributing a
$(+,-)$-eigenspace pair); the~$E_8(-1)$-summands are purely
negative-definite, so contribute $c_-=16$.
\begin{proposition}[$K^{\kch}=8$ on the Mukai Heisenberg]
\label{prop:Kkappa-8-Mukai}
Let $A=\widetilde{\mathrm{Heis}}(\Lambda_{\mathrm{Mukai}})$ be the
Mukai K3 Heisenberg, and let $\Phi(A)$ be its chiral-algebra image.
The Bridgeland stability data identifies
$\kch(A)=c_+=4$ (the central-charge positive-definite part) and
$\kch^{!}(A)=c_+=4$ (via the Verdier dual
$\widetilde{\mathrm{Heis}}^!$ on the \emph{dual} lattice
$\Lambda_{\mathrm{Mukai}}^*\cong\Lambda_{\mathrm{Mukai}}$ --- the Mukai
lattice is self-dual unimodular). Therefore
\[
\kch(A)+\kch^{!}(A) \;=\; 4+4 \;=\; 8.
\]
The Bruinier 1999 Heegner-Chern-class coefficient $c_1(0)=10$ on
$\Phi_1=\Delta_5$ (weight~$5$ paramodular, Gritsenko normalisation)
enters via the $/2$-halving $\kBKM(\Delta_5)=c_1(0)/2=5$ as the
\emph{BKM} invariant, while the B-row K3 Heisenberg invariant is
$K^{\kch}=2c_+=8$, not~$\kBKM$. The two invariants coincide
numerically only after a further Mukai-doubling identity
$\kch+\kch^{!}=2c_+$ peculiar to the Heisenberg target, \emph{not}
a general formula across the five-archetype ceiling.
\end{proposition}
\end{heal}

\begin{heal}[Five-archetype ceiling, audited]

\section{Attack IV --- MZV genus-$0$: which specific associator
relations?}
% ============================================================

\begin{attack}[Vague associator statement at lines 429--488]
The working notes state
``\emph{the Drinfeld associator $\Phi(A,B)$ is the parallel transport
of the KZ connection on $\cM_{0,4}$}''
and write
$\log\Phi=\zeta(2)[A,B]+\zeta(3)([A,[A,B]]-[B,[A,B]])+\cdots$.
Three omissions:
\begin{enumerate}
\item[(a)] \emph{Pentagon and hexagons.} The Drinfeld associator
satisfies the pentagon on $\cM_{0,5}$ and two hexagon relations on
$\cM_{0,4}$ --- the \emph{two} hexagons come from the
$(13,2)$- and $(1,23)$-splittings and are equivalent modulo the
pentagon (Drinfeld 1990~Q$3$). The working-notes text writes only
``parallel transport'', which is the \emph{definition} not the
\emph{constraint}.
\item[(b)] \emph{Zagier $2.10$-relation and Ihara--Takao.}
At weight~$6$ depth~$2$, the Zagier relation
$28\zeta(10,2)=6\zeta(12)-\zeta(6)^2+\frac{4\pi^{12}}{\mathrm{???}}
\cdot\mathrm{(rationals)}$
and the Ihara--Takao relations in $\mathfrak{grt}_1$ constrain the
associator coefficients beyond pure $\zeta(n)$. The working-notes
expansion stops at depth-$1$ coefficients and the single depth-$2$
$\zeta(4)/4$ term, missing the depth-$2$ MZV content that governs
weight-$\geq 5$ associator coefficients.
\item[(c)] \emph{Single-valued projection.} The manuscript's
chiral-Hochschild periods land in $\MZV^{\mathrm{sv}}$
(single-valued, Brown 2013 \emph{Ann Math} 175) with
$\zeta^{\mathrm{sv}}(2)=0$, $\zeta^{\mathrm{sv}}(3)=2\zeta(3)$,
$\zeta^{\mathrm{sv}}(5)=2\zeta(5)$, $\zeta^{\mathrm{sv}}(2,3)=
-11\zeta(5)-2\zeta(2)\zeta(3)$, etc. The working-notes Theorem
thm:wn-shadow-mzv writes $\zeta(2)$ as the coefficient controlling
class-$\mathsf{G}$, but on the chiral-Hochschild target
$\zeta^{\mathrm{sv}}(2)=0$, so the correct statement is that
class-$\mathsf{G}$ sees \emph{no} weight-$2$ single-valued
contribution --- the na\"\i ve $\zeta(2)$-appearance is \emph{cancelled}
by the single-valued projection.
\end{enumerate}
\end{attack}

\begin{heal}[Pentagon, hexagons, and single-valued]
Replace the working-notes expansion with:
\begin{theorem}[Drinfeld associator: pentagon, two hexagons, single-valued
chiral-Hochschild target]
\label{thm:drinfeld-pentagon-sv}
Let $\Phi(A,B)\in\mathbb{C}\langle\!\langle A,B\rangle\!\rangle$ be
the Knizhnik--Zamolodchikov associator, the unique solution of the
KZ equation
$\partial_z G=(A/z + B/(z-1))G$ with asymptotics
$G(z)\sim z^A$ at~$0$ and $G(z)\sim(1-z)^B\Phi^{-1}$ at~$1$.
Then:
\begin{enumerate}
\item[(P)] \emph{Pentagon.} On
$\cM_{0,5}=\{(z_1,\ldots,z_5)\}/\mathrm{PGL}_2$:
\[
\Phi(t_{12},t_{23})\cdot\Phi(t_{34},t_{45})
 \;=\;
 \Phi(t_{12},t_{23}+t_{24}+t_{34})\cdot
 \Phi(t_{13}+t_{23},t_{34})\cdot
 \Phi(t_{12}+t_{13},t_{23}+t_{24})
\]
modulo the infinitesimal braid relations
$[t_{ij},t_{kl}]=0$ for disjoint $\{i,j\},\{k,l\}$ and
$[t_{ij},t_{ik}+t_{jk}]=0$ for distinct $i,j,k$.
\item[(H)] \emph{Hexagons.} Two hexagons on $\cM_{0,4}$:
\[
e^{\pi i(A+B)/2}\Phi(B,A)
 \;=\; \Phi(A,B)e^{\pi i A/2}\Phi(A,B)^{-1}e^{\pi i B/2}
\]
and the swap $A\leftrightarrow B$ companion. Modulo the pentagon,
the two hexagons are equivalent to one hexagon; Drinfeld 1990 Q$3$
shows they are equivalent and any one implies the other via the
pentagon.
\item[(Z)] \emph{Zagier relation (weight~$12$, depth~$2$).}
$28\zeta(9,3)+150\zeta(7,5)+168\zeta(5,7)\,=\,
 \frac{5197}{691}\zeta(12)$
in $\MZV^{\mathrm{mot}}_{12,\leq 2}$ (Zagier 1994~\emph{FAC});
this is the first non-trivial depth-$2$ relation beyond the
shuffle+stuffle algebra.
\item[(IT)] \emph{Ihara--Takao (depth~$3$, weight~$7$).}
$\zeta(5,1,1)-2\zeta(5,2)\in\mathbb{Q}\zeta(7)$; generalised Ihara
--Takao relations span the depth defect in motivic MZVs beyond
