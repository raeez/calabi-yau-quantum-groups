\begin{abstract}
At Calabi-Yau dimension three the CY-to-chiral construction is a
framed object-level assignment.  A smooth proper CY$_3$ category
$\mathcal C$, together with a CY$_3$ target $X$ and explicit
formality, Costello-Li, TCFT, completion, and $S^3$-framing witnesses,
produces an $E_3$-holomorphic factorisation algebra
$\Phi^{\mathrm{FA}}_3(\mathcal C)$ on $X$.  A witnessed surface-curve
datum $(\Sigma_2,C)$ specialises it to
\[
  \Phi^{(\Sigma_2,C)}_3(\mathcal C)
  =
  \mathrm{SpCh}_{\Sigma_2,C}\bigl(\Phi^{\mathrm{FA}}_3(\mathcal C)\bigr)
  \in E_1\text{-}\mathrm{ChirAlg}(C).
\]
Functoriality on arbitrary CY$_3$ morphisms, inverse reconstruction,
compact non-formal strictification, and the global Drinfeld
centre/double are additional constructions.  The native output at
$d=3$ is ordered: its braided layer lives on the centre, double, or
module category attached to the specialised $E_1$ algebra.
\end{abstract}

\section{Source, Target, and Operadic Level}
\label{sec:source-target}

Let $k=\mathbb C$.  A CY$_3$ category in this note is a smooth proper
dg category $\mathcal C$ with connected unit, a negative-cyclic
Calabi-Yau class
\[
  [\sigma_{\mathcal C}]\in HC^-_3(\mathcal C),
\]
and a Hochschild shadow giving a non-degenerate degree $-3$ Serre
pairing.  The associated Euler form is antisymmetric, and the
specialised chiral output at $d=3$ is ordered.

\begin{definition}[$S^3$-framing and $A_\infty$ twisting data]
\label{def:s3-ainfty-datum}
A witnessed chain-level $S^3$-framing datum for $\mathcal C$ consists
of the following data in a chosen cyclic $A_\infty$ or dg model:
\begin{enumerate}[label=\textup{(\roman*)}]
\item a cyclic model
$(M,\{m_r\}_{r\geq 1},\langle-,-\rangle,\sigma_{\mathcal C})$
representing the negative-cyclic CY$_3$ class;
\item a chain homotopy $H_{\mathrm{fr}}$ trivialising the framed
obstruction class in the cyclic bar complex and compatible with the BV
operator determined by $\sigma_{\mathcal C}$;
\item a Costello TCFT correction operator
$B^{(2)}_{\mathrm{TCFT}}$ satisfying the total corrected identity
\[
  \{b,B^{(2)}_{\mathrm{TCFT}}\}=0;
\]
\item a strictified $E_1$ Hochschild model $A$ and a comparison map from
the relative $E_3/E_2$ obstruction complex to
$C^\bullet_{E_1}(A,A)[-2]$;
\item a complete, exhaustive, separated, strongly convergent filtration
on $C^\bullet_{E_1}(A,A)$ whose total degree $-2$ line is empty;
\item analytic completion of fields, observables, OPE kernels, and
factorisation products.
\end{enumerate}
The homotopy $H_{\mathrm{fr}}$ is chain-level data.  The topological
vanishing of $\pi_3(B\mathrm{Sp})$ supplies obstruction-theoretic room
for such a datum; the chain homotopy remains additional input.
\end{definition}

\begin{definition}[Native and specialised outputs]
\label{def:native-specialised}
Fix the data of Definition~\ref{def:s3-ainfty-datum}, a CY$_3$ target
$(X,\Omega_X)$, and a formality point for the little $3$-disks operad.
The native output is an $E_3$-holomorphic factorisation algebra
\[
  \Phi^{\mathrm{FA}}_3(\mathcal C)\in E_3\text{-}\mathrm{HolFA}(X).
\]
A witnessed admissible specialisation datum is a compact or properly
supported holomorphic surface $\Sigma_2\subset X$, a smooth reference
curve $C\subset X$, an orientation and normal-support convention, an
incidence kernel on $X\times C$, Beck-Chevalley and Fubini cells for
pushforward and restriction, and the completion data needed for
factorisation products.  The specialised output is
\[
  A_{\mathcal C}^{(\Sigma_2,C)}
  :=
  \Phi^{(\Sigma_2,C)}_3(\mathcal C)
  =
  \mathrm{SpCh}_{\Sigma_2,C}\bigl(\Phi^{\mathrm{FA}}_3(\mathcal C)\bigr).
\]
\end{definition}

\begin{proposition}[Operadic level at $d=3$]
\label{prop:e1-level}
For every constructed specialised output
$A_{\mathcal C}^{(\Sigma_2,C)}$, the native operadic level on the curve
$C$ is $E_1$:
\[
  A_{\mathcal C}^{(\Sigma_2,C)}\in E_1\text{-}\mathrm{ChirAlg}(C).
\]
When the derived centre and representation category have been
constructed, the braided layer is the centre/double layer
\[
  \mathcal Z\bigl(\mathrm{Rep}^{E_1}(A_{\mathcal C}^{(\Sigma_2,C)})\bigr)
  \simeq
  \mathrm{Rep}^{E_2}\bigl(Z^{\mathrm{der}}_{\mathrm{ch}}
  (A_{\mathcal C}^{(\Sigma_2,C)})\bigr).
\]
\end{proposition}

\begin{proof}
The two-stage construction first forms an $E_3$-holomorphic
factorisation algebra on $X$.  Factorisation homology over a
two-dimensional specialisation cycle and restriction to the curve $C$
leave one ordered complex direction.  Dunn additivity gives the
residual $E_{3-2}=E_1$ structure.  The ordered two-point configuration
space in $\mathbb R^3$ is simply connected:
\[
  \pi_1(\mathrm{Conf}^{\mathrm{ord}}_2(\mathbb R^3))=0.
\]
Unordered exchange gives the symmetric transposition.  The braid
parameter of an $E_2$ chiral algebra is absent at the native level, so
quantum-group braiding is centre data for $\mathrm{Rep}^{E_1}(A)$.
\end{proof}

\section{The Two-Stage Formula}
\label{sec:two-stage}

The formula
\[
  \Phi^{(\Sigma_{d-1},C)}_d
  =
  \mathrm{SpCh}_{\Sigma_{d-1},C}\circ \Phi^{\mathrm{FA}}_d
\]
becomes a theorem once the source, the target, and the proof inputs are
fixed.  At $d=3$ those inputs are part of the mathematical datum.

\begin{theorem}[Framed object-level CY-A$_3$ construction]
\label{thm:framed-cya3}
Let $\mathcal C$ be a smooth proper CY$_3$ category over $\mathbb C$.
Assume the following data are supplied.
\begin{enumerate}[label=\textup{(H\arabic*)}]
\item A negative-cyclic CY$_3$ class whose Hochschild shadow is a
non-degenerate degree $-3$ Serre pairing.
\item A chain-level $S^3$-framing and $A_\infty$ twisting datum as in
Definition~\ref{def:s3-ainfty-datum}.
\item A TCFT correction datum giving the total corrected relation
$\{b,B^{(2)}_{\mathrm{TCFT}}\}=0$ in the chosen cyclic model.
\item A strictified $E_1$ Hochschild model with a comparison map from
the $E_3/E_2$ obstruction complex and a complete, exhaustive,
separated, strongly convergent filtration whose total degree $-2$ line
vanishes.
\item A Costello-Li holomorphic factorisation package on the target
$X$, including completion of fields and observables.
\item A witnessed admissible specialisation datum $(\Sigma_2,C)$.
\end{enumerate}
Then the composite
\[
  \Phi^{(\Sigma_2,C)}_3(\mathcal C)
  =
  \mathrm{SpCh}_{\Sigma_2,C}\bigl(\Phi^{\mathrm{FA}}_3(\mathcal C)\bigr)
\]
is a well-defined object of $E_1\text{-}\mathrm{ChirAlg}(C)$.  Its
ordered bar coalgebra satisfies, on the same witnessed locus,
\[
  B^{E_1}\bigl(A_{\mathcal C}^{(\Sigma_2,C)}\bigr)
  \simeq
  \mathrm{CC}^{E_1}_{\bullet}(\mathcal C)
\]
as an $E_1$ factorisation coalgebra.
\end{theorem}

\begin{proof}
Kontsevich-Tamarkin formality upgrades the degree $-2$ Gerstenhaber
bracket on Hochschild cochains to an $E_3$ algebra after the formality
point is chosen.  Costello-Gwilliam locality factorises this algebra on
the underlying topological space of $X$.  The Costello-Li holomorphic
twist and the CY form refine the topological factorisation algebra to
$\Phi^{\mathrm{FA}}_3(\mathcal C)$ exactly on the locus where the
framing, TCFT correction, anomaly, and completion data are present.

The filtered Hochschild hypothesis identifies the primary relative
framing obstruction with
\[
  \mathrm{HH}^{-2}_{E_1}(A,A)
\]
for the chosen strictified model $A$ and kills that group in the model.
This degree $-2$ vanishing proves primary obstruction vanishing.  Full
lift-space contractibility requires vanishing of the higher Goodwillie
layers as an additional hypothesis.
The witnessed specialisation datum supplies the factorisation-homology
pushforward over $\Sigma_2$ and the restriction to $C$.  Proposition
\ref{prop:e1-level} gives the residual $E_1$ operadic level, and the
bar comparison is the $E_1$ ordered bar comparison transported through
the same native and specialisation data.
\end{proof}

\begin{remark}[Scope]
\label{rem:scope}
Theorem~\ref{thm:framed-cya3} is an object-level construction.  A
functor on arbitrary CY$_3$ morphisms requires cyclic kernel data,
Costello-Li naturality, Beck-Chevalley compatibility, compact-support
control, and completion compatibility.  Image classification,
mapping-space reconstruction, recovery of $\mathcal C$ from
$A_{\mathcal C}^{(\Sigma_2,C)}$, and construction of a global quantum
vertex group $G(\mathcal C)$ are separate problems.
\end{remark}

\section{Rigidity Criterion}
\label{sec:rigidity}

\begin{proposition}[$E_1$ Hochschild criterion]
\label{prop:hh-minus-two}
Let $A$ be a supplied native model for a CY$_3$ input, equipped with
the comparison map from the relative $E_3/E_2$ obstruction complex to
$E_1$ Hochschild cochains.  If the selected filtration proves
\[
  \mathrm{HH}^{-2}_{E_1}(A,A)=0,
\]
then the primary obstruction to the compatible $E_1$ enhancement
vanishes.  If the higher Goodwillie-Andre-Quillen obstruction groups in
degrees below $-2$ vanish in the same model, the resulting enhancement
is unique up to a contractible choice space.
\end{proposition}

\begin{proof}
Dunn additivity identifies the relative tangent complex of the
restriction problem with
\[
  C^\bullet_{E_1}(A,A)[-2].
\]
The first obstruction lies in
$\pi_0 C^\bullet_{E_1}(A,A)[-2]=\mathrm{HH}^{-2}_{E_1}(A,A)$.  The
remaining obstruction layers are the higher Goodwillie-Andre-Quillen
groups.  The conclusion follows from the displayed vanishings, which
are extra hypotheses beyond bare smooth proper CY$_3$ structure.
\end{proof}

\section{Holomorphic Chern-Simons Input}
\label{sec:hcs}

Holomorphic Chern-Simons theory supplies an explicit model for
the native factorisation algebra on affine and toric CY$_3$ loci.
Compact targets require the package below.

\begin{definition}[Compact hCS quantisation package]
\label{def:compact-hcs-package}
A compact CY$_3$ hCS construction consists of:
\begin{enumerate}[label=\textup{(\roman*)}]
\item a compact Calabi-Yau threefold $(X,\Omega_X)$ with orientation
and determinant-line framing compatible with the BV pairing;
\item a quadratic gauge dg Lie algebra
$(\mathfrak g,\langle-,-\rangle)$ and, when trace expressions are used,
a representation $\rho\colon\mathfrak g\to\mathrm{End}(W)$;
\item an elliptic Costello gauge fixing, heat-kernel regularisation,
propagators, and renormalisation counterterms;
\item completed fields, local functionals, and observables on which the
BV Laplacian, bracket, and RG flow are defined;
\item a scale-dependent solution of the quantum master equation;
\item vanishing or explicit trivialisation of the anomaly class, with
open-closed counterterms when required;
\item factorisation descent over $X$, plus TCFT sewing data whenever
global traces or partition functions are asserted.
\end{enumerate}
\end{definition}

\begin{proposition}[hCS observables on the compact locus]
\label{prop:compact-hcs}
Given the package of Definition~\ref{def:compact-hcs-package}, the
completed quantum hCS observables form a quantum factorisation algebra
on $X$.  A surviving anomaly class in the Costello local BV obstruction
complex blocks the corresponding compact quantum observable algebra at
that order.
\end{proposition}

\begin{proof}
Costello's renormalisation constructs observables by heat-kernel flow
between scales.  The quantum master equation is the compatibility
condition for the BV differential, bracket, and renormalised effective
interaction.  Anomaly cancellation or trivialisation is the obstruction
vanishing needed for the scale-$L$ theory to define a factorisation
algebra.  Completion and descent make the local construction global.
\end{proof}

\begin{construction}[Affine hCS model]
\label{cons:affine-hcs}
For $X=\mathbb C^3$ with holomorphic volume form $\Omega$ and
$A\in\Omega^{0,*}(X,\mathfrak g)[1]$, the classical hCS action is
\[
  S_{\mathrm{hCS}}(A)
  =
  \int_X \Omega\wedge
  \operatorname{Tr}\left(
  \frac12 A\bar\partial A+\frac13 A\wedge A\wedge A
  \right).
\]
Costello-Li quantisation gives the local holomorphic factorisation
algebra of observables on the affine toric locus after gauge fixing,
anomaly cancellation, and completion.  In the formal normal-bundle model
of a smooth curve $C$, reduction along the normal directions gives the
chiral Chevalley-Eilenberg complex
\[
  \mathrm{CE}^{\mathrm{ch}}_{\bullet}
  \bigl(\mathfrak g\otimes\Omega^{0,*}_{\nu(C)}[-1]\bigr)
\]
and its chiral envelope
$U^{\mathrm{ch}}(\mathfrak g\otimes\Omega^{0,*}_{\nu(C)}[-1])$ as the
$E_1$ factorisation algebra on $C$.
\end{construction}

\begin{remark}[Hall comparison on $\mathbb C^3$]
\label{rem:c3-hall}
Schiffmann-Vasserot identify the equivariant critical cohomological
Hall algebra of the Jordan three-loop quiver with potential
$\operatorname{Tr}(X[Y,Z])$ with the positive half
\[
  \mathrm{CoHA}(\mathbb C^3)\simeq
  Y^+(\widehat{\mathfrak{gl}}_1).
\]
The Drinfeld double or centre adds the Cartan and negative halves.  A
Prochazka-Rapcak Fock or evaluation representation then produces the
$\mathcal W_{1+\infty}$ vertex-algebra realisation.  The Hall output
itself is the positive half.
\end{remark}

\section{Five Separated Objects}
\label{sec:five-objects}

Let $A=A_{\mathcal C}^{(\Sigma_2,C)}$ be a constructed $E_1$ chiral
algebra.  The following five objects have different meanings:
\[
  A,\qquad
  B(A),\qquad
  A^i,\qquad
  A^!,\qquad
  Z^{\mathrm{der}}_{\mathrm{ch}}(A).
\]
Here $B(A)$ is the ordered bar coalgebra, $A^i$ is the Koszul dual
coalgebra, $A^!$ is the Verdier or Koszul dual algebra according to the
duality context, and $Z^{\mathrm{der}}_{\mathrm{ch}}(A)$ is the derived
chiral centre carrying the braided layer on the verified centre locus.
The identity $\Omega B(A)\simeq A$ is bar-cobar inversion.  Verdier
duality and the centre construction are separate operations with
separate input data.

\section{The $K3\times E$ Constants}
\label{sec:k3e}

\begin{proposition}[Construction-specific invariants]
\label{prop:k3e-constants}
For the product CY$_3$ threefold $X=K3\times E$, the relevant readings
are
\[
\begin{array}{c|c|c}
\text{reading} & \text{value} & \text{source of the value}\\
\hline
\kappa_{\mathrm{cat}}(X) & 0 &
\chi(\mathcal O_{K3})\chi(\mathcal O_E)=2\cdot 0\\
\kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(A_X) & 0 &
\sum_q(-1)^q h^{0,q}(K3\times E)=1-1+1-1\\
\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(X) & 3 &
\text{Heisenberg-Cartan specialisation}\\
\kappa_{\mathrm{BKM}}(\Delta_5) & 5 &
c_1(0)/2=10/2\\
\kappa_{\mathrm{fiber}}(K3) & 24 &
\mathrm{rank}\,H^*(K3,\mathbb Z)
\end{array}
\]
The four-value spectrum is $\{0,3,5,24\}$; the two zero rows are
different constructions with the same numerical value.  A relation
among rows is extra mathematical structure.
\end{proposition}

\begin{proof}
The categorical value is K\"unneth-multiplicative:
$\chi(\mathcal O_{K3\times E})=\chi(\mathcal O_{K3})\chi(\mathcal O_E)$.
The compact chiral value is the corresponding compact Hodge supertrace.
The Heisenberg value counts the chosen Cartan-Heisenberg
specialisation.
The BKM value is the Borcherds weight formula
$\kappa_{\mathrm{BKM}}=c_1(0)/2$ for the $\Delta_5$ denominator
normalisation.  The fibre value is the Mukai lattice rank of K3.
These constructions have independent inputs.
\end{proof}

\section{Consequences and Proof Obligations}
\label{sec:obligations}

\begin{corollary}[Dimension-three scope]
\label{cor:d3-scope}
On a witnessed CY$_3$ locus, CY-A$_3$ supplies the specialised
$E_1$-chiral object
$A_{\mathcal C}^{(\Sigma_2,C)}$.  Statements using this object, its
ordered bar coalgebra, and the supplied specialisation factorisation
data are in the scope of Theorem~\ref{thm:framed-cya3}.
Statements requiring Koszul reconstruction, a global Drinfeld double,
PBW/no-extra-relations, arbitrary morphism functoriality, or compact
hCS quantisation require additional proof.  Compact hCS quantisation
uses the package of Definition~\ref{def:compact-hcs-package}.
\end{corollary}

\begin{question}[Quintic strictification]
\label{q:compact-strictification}
Let $Q^5=X_5\subset\mathbb P^4$ be a smooth quintic threefold.  A
strict framed witness for $\mathrm{Perf}(Q^5)$ must absorb the
large-volume Kodaira-Spencer cubic
\[
  m^{\mathrm{KS}}_3\colon
  \mathrm{Sym}^3 H^1(T_{Q^5})\longrightarrow
  H^3(\mathcal O_{Q^5})\simeq\mathbb C,
  \qquad
  \int_{Q^5}H^3=5.
\]
The scalar values
\[
  \chi(\mathcal O_{Q^5})=1-0+0-1=0,\qquad
  \chi_{\mathrm{top}}(Q^5)/24=-200/24=-25/3
\]
are diagnostics.  The witness is the chain homotopy or curvature data
absorbing $m^{\mathrm{KS}}_3$.
Construct a cyclic curved $A_\infty$ or $L_\infty$ model whose curvature
absorbs this class while preserving the negative-cyclic CY$_3$ trace,
then construct the chain-level $S^3$-framing, TCFT correction, filtered
Hochschild comparison, QME solution, anomaly trivialisation, regulator,
completion, and witnessed specialisation datum.
\end{question}

\begin{question}[Morphism functoriality]
\label{q:morphism-functoriality}
Find the exact class of cyclic Fourier-Mukai or $A_\infty$ kernels for
which $\Phi^{(\Sigma_2,C)}_3$ is functorial on morphisms and compatible
with specialisation factorisation homology.
\end{question}

\begin{question}[Centre and double]
\label{q:center-double}
Construct the Drinfeld centre or double for the specialised
$E_1$-chiral algebra of a compact CY$_3$ and compare its braided
representation category with the expected quantum group.
\end{question}
