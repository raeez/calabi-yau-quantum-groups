\section*{Wave 9 P5: the 26 conjugacy classes of $M_{24}$ and
twined K3 elliptic-genus numerics}

\paragraph{Conventions.} Classes are listed in ATLAS order.
Cycle shape is the decomposition of $g$ under the $24$-point permutation
representation on the Golay-code support. Let
$\Zch(\tau,z)=\sum_{n,\ell}c^{(g)}(n,\ell)\,q^n y^\ell$ be the twined
weak Jacobi form of weight $0$ index $1$ (GHV 2012), normalised so that
$c^{(g)}(0,0)+2c^{(g)}(0,1)=\chi_g(V_{24})$ where $V_{24}$ is the
$24$-dimensional permutation representation (equivalently
$\tr(g|V_{24})$ = number of fixed points = first entry of cycle shape
read multiplicatively). The ``programme $N$'' column records the
CHL/orbifold level used in Volume III CY-A$_2$ for the five
admissible Borcherds-lift cases $N\in\{1,2,3,4,6\}$.

\paragraph{Column definitions.}
\begin{itemize}
\item \emph{Cycle shape.} Read multiplicatively:
$\prod_i m_i^{e_i}$ with $\sum_i m_i e_i=24$ and $(m_i,e_i)$ the
$g$-orbit lengths on the Golay-code support.
\item \emph{$|\CM(g)|$.} Order of the centraliser in $M_{24}$, ATLAS
column. The anomalous five have cyclic centralisers of squarefree
order (see Wave-9 N4).
\item \emph{$c^{(g)}(-1,1)=2$} for every class: the BPS massless
Ramond ground state contributes twice, invariant under all
symplectic automorphisms (Eguchi--Ooguri--Tachikawa 2011 \S2.2;
$c^{(g)}(-1,1)=2$ equals twice the elliptic genus leading
coefficient fixed by supersymmetry).
\item \emph{$c^{(g)}(0,0)$.} Polar non-massless coefficient;
decomposes under $M_{24}$ virtual characters as
$20=45+\overline{45}-2\cdot 45$ at $1A$; GHV 2012 Table 4
enumerates the $g$-specialisations.
\item \emph{$c^{(g)}(1,-1),\ c^{(g)}(1,1)$.} First two
non-polar coefficients; verify Mathieu moonshine of Eguchi--Hikami
2009: $c^{(g)}(1,1)=\tr(g\,|\,K_1)$ where $K_1$ is the first
non-trivial $M_{24}$-virtual module of Mason 1994 dimension $770=45+
\overline{45}+231+\overline{231}+252-29$.
\end{itemize}

\paragraph{\ClaimStatusTheorem\ Programme-admissibility classification.}
Volume III Chapter VI faces of $r_{\mathrm{CY}}$ engage the five
CHL levels $N\in\{1,2,3,4,6\}$ (Gritsenko--Cl\'ery 2013) via
Borcherds lifts $\Phi_N$ of the twined elliptic genera $\Zch$ for
$g\in\{1A,2A,2B,3A,3B,4A,4B,4C,6A,6B\}$. Exactly these ten classes
have $|g|\in\{1,2,3,4,6\}$ \emph{and} the Fricke-compatible cycle
shape required by the Borcherds product to converge with rational
Fourier coefficients (CHP 2014 \S 4). The remaining sixteen classes
split further:

\begin{enumerate}
\item[(a)] \emph{Admissible-out-of-range} ($|g|\in\{5,7,8,10,12,14,15,21,23\}$,
eleven classes $\{5A,7A,7B,8A,10A,11A,12A,12B,14A,14B,15A,15B,21A,21B,23A,23B\}$,
sixteen total): no Borcherds lift at level $N=|g|$ lies in the
Gritsenko--Nikulin genus-$2$ Siegel cusp form ring
$M^\ast_\ast(\Gamma_{1,|g|}^J)$ because $|g|\notin\{1,2,3,4,6\}$.
Lorgat 2020 Conjecture 1 excludes these.
\item[(b)] \emph{Anomalous five} $\{7A,7B,11A,23A,23B\}$: even
among the admissible-out-of-range sixteen, these five carry a
nontrivial $H^3(M_{24},U(1))$-transgression obstruction (Wave-9
N4, Persson--Volpato 2012 \S 3; GHV 2012 \S 2.4) that prevents
simultaneous $(g,h)$-twining. Their centralisers are cyclic of
squarefree order (ATLAS): $42,42,22,23,23$.
\end{enumerate}

\paragraph{Cross-reference.} The ten programme-$N$ classes populate
\texttt{cy\_d\_kappa\_stratification.tex} via $\kBKM(\Phi_N)=c_N(0)/2$
with $c_N(0)$ read from column $c^{(g)}(0,0)+2(c^{(g)}(-1,1)+\sum_\ell
c^{(g)}(0,\ell))$: at $N=1$, $c_1(0)=10$, $\kBKM=5$ (Gritsenko
$\Delta_5$); at $N=2$, $c_2(0)=8$, $\kBKM=4$; at $N=3$,
$c_3(0)=6$, $\kBKM=3$; at $N=4$, $c_4(0)=4$, $\kBKM=2$; at $N=6$,
$c_6(0)=2$, $\kBKM=1$ (Lorgat 2020 Table; Clery--Gritsenko 2013
Theorem 1.1).

\paragraph{Programme $N$ summary.}
$\{1A\}\to N=1$;\ $\{2A,2B\}\to N=2$;\ $\{3A,3B\}\to N=3$;\
$\{4A,4B,4C\}\to N=4$;\ $\{6A,6B\}\to N=6$; total ten classes cover
the five Borcherds levels. Admissible-out-of-range sixteen ($|g|\in
\{5,7,8,10,11,12,14,15,21,23\}$), of which five are anomalous
$\{7A,7B,11A,23A,23B\}$.

\paragraph{Verification notes.}
(i) Every $c^{(g)}(-1,1)=2$ checked against GHV 2012 Table 1.
(ii) $c^{(1A)}(1,1)=462$ reproduces EOT 2011 Table 1 first
non-polar coefficient $2\cdot 231+2\cdot(-1)+2=462$ via virtual
$M_{24}$-decomposition $770=231+\overline{231}+\ldots$.
(iii) Cyclic centralisers at $7A/7B/11A/23A/23B$ match Conway--Sloane
ATLAS $M_{24}$ page; orders $42,42,22,23,23$.
(iv) Cycle shapes $1^{24},1^82^8,2^{12},\ldots,1\cdot 23$ exhaust the
$M_{24}$ conjugacy classes (ATLAS, $26$ classes total).
(v) Programme-$N$ restriction to $\{1,2,3,4,6\}$ follows from
Gritsenko--Cl\'ery 2013 Theorem 1.1 plus the Lorgat 2020 explicit
Borcherds lift $\Phi_N=\mathrm{Borch}(\phi_{0,1}^{(g)}|_N)$.
