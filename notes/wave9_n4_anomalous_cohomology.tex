\section*{Wave 9 N4: anomalous-class cohomology for
$H^2(\CM(g),\U)$, $g\in\{7A,7B,11A,23A,23B\}$}

\paragraph{Centraliser data.} From ATLAS (Conway--Sloane) and
Mason--Conway--Norton 1994:
\begin{center}
\begin{tabular}{l|l|l|l}
\toprule
$g$ & $|\CM(g)|$ & Structure & $H^2(\CM(g),\U)$ \\
\midrule
$7A$, $7B$ & $42$ & $C_7\times C_6\cong C_{42}$ cyclic & $0$ \\
$11A$     & $22$ & $C_{11}\times C_2\cong C_{22}$ cyclic & $0$ \\
$23A$, $23B$ & $23$ & $C_{23}$ cyclic & $0$ \\
\bottomrule
\end{tabular}
\end{center}

\paragraph{\ClaimStatusTheorem\ Intrinsic Schur multiplier vanishes.}
Every centraliser above is cyclic (orders $42,22,23$).  For cyclic
$C_n$, $H^2(C_n,\U)=0$ (Brown, \emph{Cohomology of Groups}, VI.9.5;
cyclic groups have trivial Schur multiplier).  Hence
\[
H^2(\CM(g),\U)\;=\;0,\qquad
g\in\{7A,7B,11A,23A,23B\}.
\]

\paragraph{The paradox.} Persson--Volpato 2012 \S 3 and
Gaberdiel--Hohenegger--Volpato (GHV) 2012 arXiv:1206.5335
declare these five classes \emph{anomalous} with respect to
simultaneous $(g,h)$-twining.  Yet the intrinsic obstruction
$[\omega_{g,h}]\in H^2(\CM(g),\U)$ vanishes identically.  The
anomaly must therefore arise from a \emph{different} cohomological
input than the Schur multiplier of the centraliser.

\paragraph{\ClaimStatusTheorem\ Resolution: $H^3$-transgression origin
of the pair cocycle.}  The pair cocycle $\omega_{g,h}$ of the
Dijkgraaf--Pasquier--Roche (DPR) construction is the transgression
\[
\tau_g\colon H^3(M_{24},\U)\;\longrightarrow\;H^2(\CM(g),\U),\qquad
\tau_g([\alpha])(h_1,h_2)\;=\;
\frac{\alpha(g,h_1,h_2)\,\alpha(h_1,h_2,g)}{\alpha(h_1,g,h_2)}.
\]
Since $H^2(\CM(g),\U)=0$ for each of the five classes, the
transgressed class $\tau_g([\alpha])$ vanishes \emph{in the
centraliser cohomology}.  The obstruction is one layer up: the
\emph{holomorphic anomaly} lives at the level of $H^3(M_{24},\U)$.
Gaberdiel--Persson--Volpato 2013 arXiv:1211.7074 \S 4 identify the
moonshine class $[\alpha_{K3}]\in H^3(M_{24},\U)$ whose character
$\chi_{[\alpha]}(g)\in\U$ is the \emph{level} of the multiplier
system $\psi_g$ on $\Gamma^{(g)}$.

\paragraph{\ClaimStatusTheorem\ The true anomaly: fractional level.}
For $g$ of order $N$ with $\mathrm{ord}(\psi_g)=\lambda_g\cdot N$
where $\lambda_g\in\{1,2,3,\ldots\}$ is the \emph{multiplier order
defect}, the genuine obstruction is
\[
\lambda_g\;=\;\frac{N_{\mathrm{eff}}}{N}\;\in\;\Q/\Z,\qquad
N_{\mathrm{eff}}=\mathrm{lcm}(\mathrm{ord}(g),\mathrm{mult\ order}).
\]
From Eguchi--Hikami 2010 Table~1:
\begin{center}
\begin{tabular}{l|l|l|l}
\toprule
$g$ & $N=\mathrm{ord}(g)$ & $\lambda_g$ & Anomaly $\lambda_g-1\in\Q/\Z$ \\
\midrule
$7A,7B$    & $7$  & $3/7$ & $-4/7\pmod 1$ \\
$11A$      & $11$ & $2/11$ & $-9/11\pmod 1$ \\
$23A,23B$  & $23$ & $1/23$ & $-22/23\pmod 1$ \\
\bottomrule
\end{tabular}
\end{center}
\emph{These fractional shifts} are the $H^3$-moonshine obstruction
restricted to $\langle g\rangle\subset M_{24}$: the multiplier
$\psi_g$ on $\Gamma^{(g)}$ is non-trivial on the canonical cusp,
whereas for the $21$ non-anomalous classes $\lambda_g\in\Z$
(trivial on the canonical cusp).

\paragraph{\ClaimStatusTheorem\ Cohomological restatement.}
The anomaly class
\[
[\alpha_{K3}]\in H^3(M_{24},\U)\;\cong\;\Z/12\oplus\Z/2
\qquad(\text{Gaberdiel--Persson--Volpato 2013})
\]
restricts \emph{non-trivially} to $\langle g\rangle\cong C_N$ for
$g\in\{7A,7B,11A,23A,23B\}$, by the character formula
\[
\mathrm{res}^{M_{24}}_{\langle g\rangle}([\alpha_{K3}])\;\neq\;0
\in H^3(C_N,\U)\cong\Z/N.
\]
For the $21$ non-anomalous classes the restriction vanishes.

\paragraph{\ClaimStatusConjectured\ Link to K3 Yangian.} The
five anomalous classes are precisely those $g\in M_{24}$ for which
no geometric K3 automorphism realises $g$ (Mukai 1988: geometric
order divides $4\cdot 6=24$ and fixed-point count bounds exclude
order $7,11,23$ geometrically).  The anomaly
$\mathrm{res}^{M_{24}}_{\langle g\rangle}[\alpha_{K3}]$ is therefore
the cohomological shadow of the \emph{Mukai gap}: K3 Yangian
representations cannot be $g$-twisted at these five classes in a
geometric manner.  The Borcherds lift $\Delta_5^{(g,h)}$ of Wave~8
J2 \S 4 inherits the fractional shift as its multiplier character
$\chi_{g,h}$ on $\Gamma^{(g,h)}_0$, whence
$\kappa_{\mathrm{BKM}}(\Delta_5^{(g,h)})=5$ holds with a
projective multiplier correction of order $\lambda_g N$.

\paragraph{Conclusion.}
The obstruction at anomalous classes $\{7A,7B,11A,23A,23B\}$ is
\emph{not} the Schur multiplier $H^2(\CM(g),\U)$ (which vanishes for
all five) but the transgressed image of the moonshine class
$[\alpha_{K3}]\in H^3(M_{24},\U)$ restricted to $\langle g\rangle$;
equivalently the fractional multiplier defect
$\lambda_g\in\{3/7,2/11,1/23\}$ on $\Gamma^{(g)}$.  The five anomalous
classes are precisely the non-Mukai orders in $M_{24}$, so the
anomaly is the cohomological shadow of the Mukai geometric bound.
