\section{The sheaf $\cY^+_X$}

Fix a compact Calabi--Yau threefold $X$ with holomorphic volume form
$\Omega$ and a cover $\cU=\{U_\alpha\}$ by toric/quiver--affine charts:
$U_\alpha=\Spec\cA_\alpha$, where $(\cA_\alpha,W_\alpha)$ is a
Ginzburg--Calabi--Yau algebra of a quiver with superpotential $W_\alpha$
whose Jacobi algebra $J(\cA_\alpha,W_\alpha)$ presents $D^b\operatorname{Coh}(U_\alpha)$
in the Bocklandt sense (Bocklandt 2013, Thm.~3.6; Van den Bergh 2004,
\S\S 3--4). On each chart the positive--half cohomological Hall algebra
is the Kontsevich--Soibelman object
\[
\cY^+_{U_\alpha}\;=\;\CoHA(\cA_\alpha,W_\alpha)
\;=\;\bigoplus_{d\in\ZZ^{Q_0}}
H^\bullet_c\bigl(\cM_d(\cA_\alpha),\varphi_{W_\alpha}\bigr),
\]
with product the $\star$-convolution on moduli of representations
(Kontsevich--Soibelman 2008, Ch.~6, \S 6.1--6.2). We upgrade the
collection $\{\cY^+_{U_\alpha}\}$ to a sheaf on $X$.

\begin{definition}[positive--half factorisation sheaf]\label{def:Yplus-sheaf}
$\cY^+_X$ is the sheaf on $X$ assigning to an affine toric open
$U\subset X$ the chiral factorisation algebra
$\cY^+_X(U)=\CoHA(U,W_U)$, with $W_U$ the restriction of the chart
superpotential; to a general open $V$, the
Beilinson--Drinfeld unital factorisation algebra obtained by
left Kan extension along the inclusion of the toric--affine site
(BD 2004, Ch.~3, \S 3.4).
\end{definition}

The unit and the product are the factorisation $j_!j^*\to\Delta_!$
structure maps of BD \S 3.4.11; $E_1$--associativity follows from the
critical CoHA $\star$--product on each chart, $E_3$--compatibility from
the Wave~13--16 6D holomorphic Chern--Simons locality of observables
(Costello 2013, \S\S 8--10; Costello--Gaiotto 2018).

\section{The \v{C}ech nerve and Francis--Gaitsgory descent}

Let $N_\bullet(\cU)\to X$ be the \v{C}ech nerve of the cover, i.e.\ the
simplicial scheme with $N_p=\coprod_{\alpha_0\dots\alpha_p}
U_{\alpha_0\cdots\alpha_p}$, $U_{\alpha_0\cdots\alpha_p}
=U_{\alpha_0}\cap\dots\cap U_{\alpha_p}$.

\begin{proposition}[\v{C}ech presentation]\label{prop:cech-presentation}
Under the \'etale--to--fppf refinement on the toric--affine site, the
factorisation sheaf $\cY^+_X$ of Definition~\ref{def:Yplus-sheaf}
satisfies
\[
\cY^+_X\;\cong\;
\hocolim\Bigl(
\coprod_{\alpha\beta}\cY^+_{U_{\alpha\beta}}\rightrightarrows
\coprod_\alpha\cY^+_{U_\alpha}\Bigr)
\quad\text{in}\quad\Fact(\Ran X),
\]
where the arrows are restriction to the double--overlap charts.
\end{proposition}

\begin{proof}[Proof sketch]
Francis--Gaitsgory chiral--Koszul duality (Francis--Gaitsgory 2011,
Thm.~6.3.2) identifies factorisation algebras on $X$ with
$\Fact$--coalgebras on $\Ran X$; both categories are
$\infty$--presentable and satisfy \'etale descent. For the toric--affine
cover $\cU$, BD 2004 Ch.~3, Prop.~3.4.6, gives that the assignment
$U\mapsto\CoHA(U,W_U)$ is a co--sheaf on the toric site; passing to
chiral duals yields the claimed $\hocolim$. The comparison with the
Kontsevich--Soibelman critical CoHA on each chart is the content of
KS~2008, Thm.~6.1, specialised to the local potential $W_U$.
\end{proof}

\section{Overlap cocycle: the glued potential}

\begin{construction}[overlap cocycle]\label{cstr:overlap}
Let $U_\alpha,U_\beta$ be two charts with
$U_{\alpha\beta}=U_\alpha\cap U_\beta$. On $U_{\alpha\beta}$ both
$(\cA_\alpha,W_\alpha)|_{U_{\alpha\beta}}$ and
$(\cA_\beta,W_\beta)|_{U_{\alpha\beta}}$ present the same
CY$_3$--affine derived category $D^b\operatorname{Coh}(U_{\alpha\beta})$,
hence are Morita--equivalent as Ginzburg dg--algebras (Van den Bergh
2004, Thm.~A). Write
$M_{\alpha\beta}\colon\cA_\alpha|_{U_{\alpha\beta}}\xrightarrow{\sim}
\cA_\beta|_{U_{\alpha\beta}}$ for the mutation/tilting--complex
equivalence. The cocycle is the induced isomorphism of critical loci
\[
g_{\alpha\beta}\colon
\Crit(W_\alpha)|_{U_{\alpha\beta}}
\xrightarrow{\,M_{\alpha\beta}\,}
\Crit(W_\beta)|_{U_{\alpha\beta}},
\]
together with the comparison of vanishing cycles
$\varphi_{W_\alpha}\simeq g_{\alpha\beta}^{*}\varphi_{W_\beta}$
(KS 2008, \S 4.3). The collection $\{g_{\alpha\beta}\}$ is a \v{C}ech
1--cocycle in $\mathrm{Aut}(\Perv(X))$ by construction, and its
cohomology class is the obstruction to a global
$(\cA,W)$--presentation.
\end{construction}

\begin{lemma}[cocycle condition]\label{lem:cocycle}
On triple overlaps $U_{\alpha\beta\gamma}$,
\[
g_{\beta\gamma}\circ g_{\alpha\beta}\;=\;g_{\alpha\gamma}
\quad\text{in}\quad\mathrm{Aut}(\cY^+_{U_{\alpha\beta\gamma}}),
\]
up to a $2$--cell recorded by the Seidel--Thomas spherical--twist class
in $HH^2$ of the local derived category.
\end{lemma}

\begin{proof}
Mutation equivalences compose to the identity on triple overlaps up to
a spherical twist (Van den Bergh 2004, \S 5; Bocklandt 2013, \S 4).
Passing to vanishing cycles converts the twist to a $2$--cell in the
$\infty$--groupoid $\mathrm{Aut}(\Perv)$; the critical CoHA transports
this to an $E_1$--equivalence of $\cY^+_{U_{\alpha\beta\gamma}}$.
\end{proof}

\section{Derived global sections}

\begin{theorem}[$Y^+(X)$ as derived global sections]\label{thm:global-Yplus}
With $\cY^+_X$ and $\{g_{\alpha\beta}\}$ as above,
\[
Y^+(X)\;\coloneqq\;R\Gamma\bigl(X,\cY^+_X\bigr)
\;=\;\holim\Bigl(
\prod_\alpha\cY^+_{U_\alpha}\rightrightarrows
\prod_{\alpha\beta}\cY^+_{U_{\alpha\beta}}
\rightthreetimes\dots\Bigr),
\]
the totalisation of the \v{C}ech cosimplicial object of
Prop.~\ref{prop:cech-presentation}. The result carries an
$E_1$--bialgebra structure, and an $E_3$--hFA refinement whenever the
Wave 13--16 6D holomorphic Chern--Simons locality holds.
\end{theorem}

\begin{proof}[Proof sketch]
Take derived global sections term--by--term in the \v{C}ech cosimplicial
resolution. Homotopy limits commute with $R\Gamma$ on presentable
stable $\infty$--categories (BD 2004 \S 3.4; Lurie HA, 1.2.4), yielding
the totalisation. The $E_1$--bialgebra product is the chart--local
$\star$--product; gluing preserves it because each $g_{\alpha\beta}$ is
an $E_1$--algebra equivalence (Lemma~\ref{lem:cocycle}). The
$E_3$--hFA refinement is the Costello--Gaiotto holomorphic twist
locality on 6D hCS, which passes to factorisation homology on $X$
(Costello 2013, \S 9).
\end{proof}

\section{Homotopy descent: survival of the bialgebra structure}

\begin{proposition}[bialgebra descent]\label{prop:bialg-descent}
The CoHA bialgebra structure (coproduct, antipode up to homotopy)
survives the \v{C}ech totalisation: $Y^+(X)$ is an $E_1$--bialgebra in
the symmetric--monoidal $\infty$--category of factorisation complexes
on $X$.
\end{proposition}

\begin{proof}[Proof sketch]
On each chart the Davison--Meinhardt coproduct on $\CoHA(\cA_\alpha,
W_\alpha)$ is natural under mutation (Davison--Meinhardt 2020,
Thm.~A). Naturality implies the chart coproducts assemble into a map
of \v{C}ech cosimplicial objects, and homotopy limits of
$E_1$--bialgebras in presentable stable $\infty$--categories are
$E_1$--bialgebras (Lurie HA, 7.3.4.7 applied to the monad for $E_1$).
Antipode descends up to coherent homotopy via the $2$--cells of
Lemma~\ref{lem:cocycle}.
\end{proof}

\begin{remark}[scope]
On toric Calabi--Yau threefolds the cover by mutation--equivalent
quiver charts is global (KS 2008, Ch.~8); on compact CY$_3$'s such as
the quintic the cover is \'etale--local and the descent is genuinely
\v{C}ech. $\kch(X)$ is chart--independent:
$\kch=\sum_q(-1)^q h^{0,q}(X)$, hence coincides with $\kcat=\chi(\cO_X)$
(HZ--3 supertrace, CY$_d$ identification).
\end{remark}
