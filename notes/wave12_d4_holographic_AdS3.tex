\section{AdS$_3$ holographic realization of $\fg_{\Delta_5}$}
\label{wn:sec:wave12-d4-holographic-AdS3}

\paragraph{Setup.} Type IIB on $K3\times S^1$ with $Q_1$ D1-branes on
$S^1$ and $Q_5$ D5-branes on $K3\times S^1$, decoupling limit
$AdS_3\times S^3\times K3$ (Maldacena 1998 hep-th/9711200;
Strominger--Vafa 1996 hep-th/9601029). Boundary CFT$_2$: sigma model
on $\Sym^N(K3)$ with $N=Q_1 Q_5$, central charge
$\cAdS=6N = 3\ell_3/(2 G_3)$, $\mathcal{N}{=}(4,4)$ superconformal.
Twist by $T^2$-reduction to $AdS_3\times S^3\times K3\times S^1$
realizes the Vol~III $K3\times E$ threefold as the internal
compactification; the BKM $\fg_{\Delta_5}$ with
$\kBKM(\Phi_1)=c_1(0)/2=5$ is the conjectural BPS symmetry
algebra of this bulk (Dijkgraaf--Verlinde--Verlinde 1997
hep-th/9607026).

\paragraph{Cycle 1: DMVV and $\Phi_{10}^{-1}$ as $\Sym^\infty(K3)$ genus.}
\ClaimStatusTheorem\ (DMVV 1997 hep-th/9608096.) The K3 elliptic
genus has weak Jacobi form expansion
$\chi(K3;q,y)=\sum_{n\geq 0,\ell} c(4n-\ell^2)\,q^n y^\ell$ with
$c(-1)=2$, $c(0)=20$, $c(3)=0$, total $\chi_y(K3)=2y+20+2y^{-1}$.
DMVV's generating identity
\[
  Z_{\mathrm{ell}}\bigl(\SymInf K3;p,q,y\bigr)
  \;=\; \prod_{n>0,m\geq 0,\ell}
   \frac{1}{(1-p^n q^m y^\ell)^{c(4nm-\ell^2)}}
\]
equals $p/\Phi_{10}(p,q,y)$ up to the Hodge anomaly prefactor,
matching the Igusa cusp form denominator of Wave-11 W2. Thus
$\Sym^N(K3)$ elliptic genera are Fourier coefficients of
$\Phi_{10}^{-1}$, and $\fg_{\Delta_5}=\fg_{\Phi_{10}^{1/2}}$ is
the denominator-formula BKM reading of this same automorphic object:
\emph{the BKM algebra whose Weyl--Kac--Borcherds denominator
reproduces the $\Sym^N(K3)$ orbifold partition function.}
At Vol~III subscript discipline,
$\kBKM=5$, $\kcat(K3\times E)=0$, $\kch(K3\times E)=0$ via
Hodge supertrace; the three are distinct.

\paragraph{Cycle 2: Sen quantum entropy and $\Phi_{10}^{-1}$.}
\ClaimStatusTheorem\ (Sen 2008 arXiv:0809.3304 quantum entropy
function; Jatkar--Sen 2006 hep-th/0510147; Shih--Strominger--Yin
2005 hep-th/0506151 for $N=1$.) The 1/4-BPS degeneracy on $K3\times
T^2$ is the Siegel modular contour
\[
  d_1(Q,P)=(-1)^{Q\cdot P+1}\oint_{\mathcal{C}}
  \frac{e^{-i\pi(\rho Q^2+\sigma P^2+2v\,Q\!\cdot\!P)}}{\Phi_{10}(\rho,\sigma,v)}\,
  d\rho\,d\sigma\,dv,
\]
equal to $\langle \exp(-S_{\mathrm{bdy}})\rangle_{\mathrm{AdS}_2}$
under Sen's AdS$_2/$CFT$_1$ attractor. Saddle: $\rho=P^2/\tau$,
$\sigma=Q^2/\tau$, $v=Q\!\cdot\!P/\tau$ gives
$\log d_1=\pi\sqrt{Q^2 P^2-(Q\!\cdot\!P)^2}+\cdots$ (Cardy leading).
At CHL $N\in\{2,3,4,6\}$ the same identity holds with $\Phi_{10}\to
\Phi_{k(N)}$, $k(N)\in\{10,6,4,2,4/7\}$ (Jatkar--Sen 2006;
David--Sen 2006 hep-th/0609074), with level-$N$ contour
$\mathcal{C}_N$ and subleading Gauss-Bonnet $4\log(Q^2 P^2-\cdots)$
corrections pinned by Wave-9 P4.

\paragraph{Cycle 3: BLLPRvR 4D $\mathcal{N}{=}2$ chiral algebra
and $\fg_{\Delta_5}$.}
\ClaimStatusTheorem\ (Beem--Lemos--Liendo--Peelaers--Rastelli--van~Rees
2015 \cite{BLLPR2015}, arXiv:1312.5344.) The class-$\mathcal{S}$
theory $T[A_1,\Sigma_{0,24}]$ has
$c_{\mathrm{4d}}=(2n_v+n_h)/12=107/6$ via Chacaltana--Distler 2010
\S5.14 pants decomposition with trinion $(n_v,n_h)=(63,88)$;
BLLPRvR gives boundary VOA central charge
\[
  c_{\mathrm{2d}}(T[A_1,\Sigma_{0,24}])\;=\;-12\,c_{\mathrm{4d}}\;=\;-214\;=\;-2\cdot 107
\]
(107 prime). \ClaimStatusConjectured\ (Wave-11 W2 + BLLPRvR
composition.) The BKM super-algebra $\fg_{\Delta_5}$ embeds as the
Higgs-branch chiral symmetry of $T[A_1,\Sigma_{0,24}]$: its imaginary
simple roots are the Coulomb-branch Schur operators of the 24-punctured
sphere, its real simple roots the 24 affine $\widehat{A_1}$ level-1
currents at the punctures. The embedding
$V(\fg_{\Delta_5})\hookrightarrow \chi[T[A_1,\Sigma_{0,24}]]$ at
$c=-214$ is compatible with: (a) the $\kBKM=5$ level matching
$(5n-13)/6|_{n=24}=107/6$ via $2\cdot 107/(-214/107)$; (b) Wave-9
P4 Monster rank~2 embedding as centralizer of the 24 fermion zero
modes.

\paragraph{Cycle 4: Higher-spin $\mathcal{W}_\infty[\lambda]$
and CHL lift.}
\ClaimStatusConjectured\ (Gaberdiel--Gopakumar 2011 arXiv:1011.2986;
Gaberdiel--Hartman 2011 arXiv:1101.2910 higher-spin AdS$_3/$CFT$_2$;
Castro--Gopakumar--Gutperle--Raeymaekers 2011 arXiv:1106.4788.)
Vasiliev hs$[\lambda]$ gravity in AdS$_3$ has boundary
$\Wlambda$ algebra with 't Hooft parameter $\lambda=N/(N+k)$.
The Wave-11 X5 M5 Chern--Simons identification
$k^{\mathrm{CS}}_N=\kBKM(\Phi_N)$ suggests the CHL lift
$k=\kBKM(\Phi_N)$, $N=\mathrm{rank}\,\fg^{(N)}$ (finite at
$N\in\{2,3,4,6\}$), giving
\[
  \lambda_N \;=\; \frac{\mathrm{rank}\,\fg^{(N)}}{\mathrm{rank}\,\fg^{(N)}+\kBKM(\Phi_N)}
  \;\in\;\bigl\{\tfrac{r_1}{r_1+5},\tfrac{r_2}{r_2+4},\tfrac{r_3}{r_3+3},
  \tfrac{r_4}{r_4+2},\tfrac{r_6}{r_6+1}\bigr\}.
\]
\ClaimStatusConjectured\ The $N\to\infty$ collapse $\lambda\to 1$
matches the semiclassical Vasiliev limit; the $N=1$ Monster
embedding places $\fg_{\Delta_5}$ inside a $\Wlambda$ with
$\lambda=r_1/(r_1+5)$ at Monster rank~2, pinning the higher-spin
asymptotic symmetry as a BKM enhancement.

\paragraph{Cycle 5: Triangulated holographic identity.}
Collecting Cycles 1--4: $\fg_{\Delta_5}$ is simultaneously
(i) the BPS symmetry of $\Sym^\infty K3$ whose characters are
$\Phi_{10}^{-1}$ Fourier coefficients (Cycle 1);
(ii) the generator of Sen's AdS$_2$ quantum-entropy contour
integral that equals $d_1(Q,P)$ (Cycle 2);
(iii) the Higgs-branch VOA of $T[A_1,\Sigma_{0,24}]$ at
$c_{\mathrm{2d}}=-214$ (Cycle 3);
(iv) the boundary $\Wlambda[\lambda_1]$ enhancement of
Vasiliev hs$[\lambda]$ gravity (Cycle 4).
The four arrows close into the \emph{holographic quadrangle}
\[
\begin{array}{ccc}
 \SymInf(K3)_{\mathrm{DMVV}} & \longleftrightarrow &
   \chi[T[A_1,\Sigma_{0,24}]]_{\mathrm{BLLPRvR}} \\
 \updownarrow\mathrm{Sen} & & \updownarrow\mathrm{Vasiliev} \\
 Z_{\mathrm{AdS}_2}^{\mathrm{Sen}}=1/\Phi_{10} & \longleftrightarrow &
   \Wlambda[\lambda_1]_{\mathrm{Monster}}
\end{array}
\]
with $\fg_{\Delta_5}$ at each vertex the same BKM super-algebra
viewed through the four lenses. At CHL $N\geq 2$ the quadrangle
persists with $\Phi_{10}\to\Phi_{k(N)}$,
$c_{\mathrm{2d}}\to -12 c_{\mathrm{4d}}^{(N)}\in\{-422/5,-196/5,
-226/5,-286/5\}$ (Wave-11 X5 Q1), $\lambda_1\to\lambda_N$.

\paragraph{Healed observation.} The BLLPRvR identity
$c_{\mathrm{2d}}=-12c_{\mathrm{4d}}$ composed with Wave-11 X5
M5 Chern-Simons $k^{\mathrm{CS}}_N=\kBKM(\Phi_N)$ and Sen's
quantum-entropy $1/\Phi_{10}$ produces the \emph{triangulated
holographic identity}
\[
  c_{\mathrm{2d}}^{(N)}\;+\;12\,c_{\mathrm{4d}}^{(N)}\;=\;0,\qquad
  k^{\mathrm{CS}}_N\;=\;\kBKM(\Phi_N)\;=\;c_N(0)/2,\qquad
  Z_{\mathrm{AdS}_2}^{(N)}=1/\Phi_{k(N)}.
\]
The three identities are algebraically independent but share
$\fg^{(N)}$ as their common representation-theoretic content.
The $N=1$ case is self-consistent at $c_{\mathrm{2d}}=-214$,
$k^{\mathrm{CS}}_1=5$, $Z=1/\Phi_{10}$; the $N\in\{2,3,4,6\}$
extensions are conjectural pending independent verification of
the CHL class-$\mathcal{S}$ trinion counts and CHL modular contour.
