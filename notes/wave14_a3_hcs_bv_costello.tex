\title{Bochner--Martinelli Propagators and the Quartic BV Obstruction\\
for Holomorphic Chern--Simons Theory on Calabi--Yau Threefolds}
\author{}
\date{}

\begin{document}

\begin{abstract}
Holomorphic Chern--Simons theory on a Calabi--Yau threefold is a
Costello BV theory only after choosing a holomorphic volume form, a
quadratic gauge Lie algebra, an elliptic gauge fixing, a heat-kernel
regulator, and a completion of the factorisation algebra of observables.
On flat $\C^3$, the Costello heat-kernel propagator has the
Bochner--Martinelli kernel as its off-diagonal limit. This kernel is the
canonical representative selected by the Euclidean metric in the
homotopy class of $\dbar$-gauge fixings. For a nonabelian gauge algebra
in complex dimension three, the degree-one obstruction to the quantum
master equation is governed by the quartic invariant
$I_4 \in \Sym^4(\gfrak^\vee)^{\gfrak}$. The quadratic Casimir governs
the two-point field-strength counterterm and the kinetic pairing. A
compact statement therefore requires an actual BV anomaly
trivialisation together with orientation, framing, regulator,
gauge-fixing, analytic completion, and factorisation or TCFT sewing
hypotheses.
\end{abstract}

\section{BV Data}
\label{sec:bv-data}

\begin{construction}[Holomorphic Chern--Simons datum]
\label{cons:hcs-datum}
A six-real-dimensional holomorphic Chern--Simons datum consists of
\[
  (X,\Omega_X,\gfrak,\langle -,-\rangle,h,Q^{\mathrm{GF}}),
\]
where $X$ is a complex threefold with a nowhere-vanishing holomorphic
volume form $\Omega_X$, $\gfrak$ is a finite-dimensional Lie algebra
with a non-degenerate invariant symmetric pairing, $h$ is a Hermitian
metric, and $Q^{\mathrm{GF}}$ is an elliptic Costello gauge-fixing
operator for $Q=\dbar$.

The BV fields are
\[
  \cE_X=\Omega^{0,\bullet}(X,\gfrak)[1].
\]
The classical action is
\[
S_{\mathrm{cl}}(\cA)=
  \int_X \Omega_X\wedge
  \left\langle
  \frac12\,\cA\wedge\dbar\cA+
  \frac16\,\cA\wedge[\cA,\cA],
  1
  \right\rangle ,
  \qquad
  \cA\in\Omega^{0,\bullet}(X,\gfrak).
\]
Equivalently, the integrand is
$\Omega_X\wedge\langle \frac12 \cA\dbar\cA+\frac16\cA[\cA,\cA]\rangle$.
The classical master equation is the Jacobi identity for the Dolbeault
dg Lie algebra
$(\Omega^{0,\bullet}(X,\gfrak),\dbar,[-,-])$
together with invariance of $\langle -,-\rangle$.
\end{construction}

\begin{definition}[Costello renormalisation datum]
\label{def:costello-rg}
A perturbative quantisation is a family of effective interactions
$I[L]\in\cO_{\mathrm{loc}}(\cE_X)[[\hbar]]$ satisfying the renormalisation
group equation and the scale-$L$ quantum master equation
\[
  QI[L]+\frac12\{I[L],I[L]\}_L+\hbar\Delta_L I[L]=0.
\]
Here $\{-,-\}_L$ and $\Delta_L$ are defined by the regularised BV kernel
at scale $L$. The obstruction at order $\hbar^r$ lies in the degree-one
cohomology of the local BV complex.
\end{definition}

\begin{remark}[Gauge fixing]
\label{rem:gauge-fixing}
For a fixed Hermitian metric the standard gauge fixing is
$Q^{\mathrm{GF}}=\dbarstar$. A homotopy of Costello gauge fixings changes
the propagator by a $Q$-exact kernel. When the QME obstruction vanishes,
Costello's independence-of-gauge-fixing theorem identifies the resulting
quantum observables up to equivalence of factorisation algebras.
\end{remark}

\section{Flat Propagator}
\label{sec:flat-propagator}

\begin{proposition}[Bochner--Martinelli kernel from the flat heat kernel]
\label{prop:bm-kernel}
On $(\C^3,h_{\mathrm{eucl}})$, let
$K_t(z,w)=(4\pi t)^{-3}\exp(-\|z-w\|^2/(4t))$ be the scalar heat kernel.
For $0<\varepsilon<L<\infty$ define
\[
  P_{\varepsilon,L}(z,w)=
  \int_{\varepsilon}^{L}(\dbarstar_{\mathrm{eucl}}\otimes 1)K_t(z,w)\,dt .
\]
Off the diagonal,
\[
  \lim_{\varepsilon\to0,\;L\to\infty}P_{\varepsilon,L}(z,w)
  =P_{\mathrm{BM}}(z,w),
\]
where
\[
P_{\mathrm{BM}}(z,w)=
\frac{2!}{(2\pi i)^3}
\sum_{k=1}^{3}(-1)^{k-1}
\frac{\overline{z_k-w_k}}{\|z-w\|^6}\,
\widehat{d\bar z_k}\wedge dw_1\wedge dw_2\wedge dw_3 .
\]
The distributional identity
$\dbar_wP_{\mathrm{BM}}=\delta_{\Delta}$ holds on
$\C^3\times\C^3$.
\end{proposition}

\begin{proof}
The Green operator for the flat $\dbar$-Laplacian is
$G=\int_0^\infty K_t\,dt$ off the diagonal. Applying
$\dbarstar_{\mathrm{eucl}}$ in one variable gives
$(\dbarstar_{\mathrm{eucl}}\otimes1)G$, whose standard coordinate form
is the Bochner--Martinelli kernel. The finite interval
$[\varepsilon,L]$ smooths the diagonal singularity and cuts the infrared
tail. Removing both cutoffs recovers the Green kernel away from the
diagonal, hence the displayed formula.
\end{proof}

\begin{corollary}[Flat representative]
\label{cor:flat-representative}
$P_{\mathrm{BM}}$ is the canonical representative determined by the
Euclidean metric and the gauge fixing $\dbarstar_{\mathrm{eucl}}$.
Changing the Hermitian metric or adding a $Q$-exact gauge-fixing
deformation changes the representative inside the same propagator
homotopy class.
\end{corollary}

\section{Quartic Gauge Obstruction}
\label{sec:quartic-obstruction}

\begin{definition}[Symmetric quartic invariant]
\label{def:quartic-invariant}
For a representation $\rho:\gfrak\to\End(V)$, set
\[
  I_4^\rho(x_1,x_2,x_3,x_4)
  =
  \frac1{4!}\sum_{\sigma\in S_4}
  \Tr_V\!\left(
    \rho(x_{\sigma(1)})\rho(x_{\sigma(2)})
    \rho(x_{\sigma(3)})\rho(x_{\sigma(4)})
  \right).
\]
Then $I_4^\rho\in\Sym^4(\gfrak^\vee)^{\gfrak}$. The adjoint choice is
denoted $I_4^{\ad}$.
\end{definition}

\begin{proposition}[One-loop QME obstruction]
\label{prop:qme-obstruction}
In complex dimension three, the nonabelian gauge contribution to the
one-loop QME obstruction is represented in the local CE complex by
\[
  \Theta^\rho_U(\alpha_0,\alpha_1,\alpha_2,\alpha_3)
  =
  \int_U
  \left[
  I_4^\rho\bigl(
  \alpha_0,\partial\alpha_1,\partial\alpha_2,\partial\alpha_3
  \bigr)
  \right]_{(3,3)},
  \qquad
  \alpha_i\in\Omega_c^{0,\bullet}(U,\gfrak),
\]
with the usual CE Koszul signs. Its class
\[
  \kanom(X,\gfrak,\rho)
  =
  [\,\hbar\,\Theta^\rho\,]
  \in H^1_{\mathrm{loc}}
  \bigl(\Omega^{0,\bullet}(X,\gfrak)[1],\cO_{\mathrm{loc}}\bigr)
\]
is the first obstruction to solving the QME for the open hCS theory.
For $\rho=\ad$ this is the Costello--Li expression
$\int_X \Tr_{\ad}(c\,F_A^3)$ in ghost-field components.
\end{proposition}

\begin{proof}
The open-string one-loop anomaly is the wheel with four external gauge
insertions. The Lie factor is the symmetrised trace of four Lie algebra
inputs, hence an element of $\Sym^4(\gfrak^\vee)^{\gfrak}$. The form
factor contains three holomorphic derivatives, giving a $(3,\bullet)$
density on a complex threefold. Taking the $(3,3)$ component gives a
local functional of cohomological degree one. Costello--Li identify the
same class as the descent of $\Tr_{\ad}(F_A^4)$, equivalently
$\Tr_{\ad}(cF_A^3)$ after inserting the gauge ghost.
\end{proof}

\begin{remark}[Green--Schwarz factorisation in complex dimension three]
\label{rem:gs-factorisation}
For the type-I open--closed B-model, the closed sector cancels the
one-loop open anomaly when the adjoint quartic trace factors through the
closed-sector quadratic expression. Costello--Li write the condition in
dimension three as
\[
  \Tr_{\ad}(X^4)\propto \bigl(\Tr_{\mathrm{fun}}(X^2)\bigr)^2 .
\]
For $\gfrak=\mathfrak{so}_n$,
\[
  \Tr_{\ad}(X^4)
  =(n-8)\Tr_{\mathrm{fun}}(X^4)
  +3\bigl(\Tr_{\mathrm{fun}}(X^2)\bigr)^2,
\]
so the type-I cancellation selects $\mathfrak{so}_8$. The stable
superalgebra version is $\mathfrak{osp}(n+8\mid n)$.
\end{remark}

\section{Two-Point Renormalisation}
\label{sec:two-point}

\begin{definition}[Quadratic Casimir coefficient]
\label{def:quadratic-casimir}
Let $\{T_a\}$ be a basis of $\gfrak$ and let $\{T^a\}$ be dual for the
chosen invariant pairing. The representation-theoretic quadratic
Casimir is the scalar or endomorphism
\[
  C_2^\rho=\sum_a\rho(T_a)\rho(T^a).
\]
For the adjoint representation of a simple Lie algebra in the usual
normalisation, $C_2^{\ad}=2h^\vee$.
\end{definition}

\begin{proposition}[Field-strength counterterm]
\label{prop:field-strength}
With the flat heat-kernel normalisation, the two-point graph contributes
a degree-zero counterterm of the form
\[
  S_{\mathrm{ct}}^{(2)}
  =
  -\hbar\,\frac{C_2^\rho}{(4\pi)^3}
  \log(L/\varepsilon)
  \int_X\Omega_X\wedge\Tr_\rho(\cA\,\dbar\cA),
\]
up to the convention for the invariant pairing and the heat-kernel
normalisation. This term renormalises the kinetic pairing and the field
$\cA$. It is a two-point field-strength term, while
Proposition~\ref{prop:qme-obstruction} is the degree-one QME
obstruction.
\end{proposition}

\begin{proof}
Closing two legs of the cubic vertex yields the colour contraction
$\sum_a\rho(T_a)\rho(T^a)=C_2^\rho$. The remaining two external fields
have the same tensor type as the kinetic term
$\int\Omega_X\wedge\Tr_\rho(\cA\dbar\cA)$. The logarithmic divergence is
the usual short-distance term in six real dimensions for the heat-kernel
regularisation. Since the resulting local functional has cohomological
degree zero, it changes the kinetic representative. The degree-one
obstruction class is the quartic anomaly class.
\end{proof}

\section{Compact Threefolds}
\label{sec:compact-threefolds}

\begin{hypothesis}[Compact BV package]
\label{hyp:compact-package}
A compact CY3 hCS quantisation theorem requires the following data.
\begin{enumerate}
\item A compact Calabi--Yau threefold $(X,\Omega_X)$ with orientation and
determinant-line framing compatible with the BV pairing.
\item A quadratic gauge Lie algebra or dg Lie algebra
$(\gfrak,\langle-,-\rangle)$ and, when traces are used, a representation
$\rho$ fixing $I_4^\rho$ and $C_2^\rho$.
\item An elliptic Costello gauge fixing $Q^{\mathrm{GF}}$, heat-kernel
regulator $P_{\varepsilon,L}$, and RG-compatible counterterm scheme.
\item A completed space of local functionals and observables on which
$\Delta_L$, $\{-,-\}_L$, and the RG flow are defined.
\item A factorisation or TCFT sewing structure with descent from
polydisks to $X$.
\item A trivialisation of the anomaly class
$\kanom(X,\gfrak,\rho)$, either by vanishing in local BV cohomology or
by an explicit local counterterm or closed-sector Green--Schwarz term.
\end{enumerate}
\end{hypothesis}

\begin{theorem}[Conditional compact quantisation]
\label{thm:conditional-compact}
Assume Hypothesis~\ref{hyp:compact-package}. If
$\kanom(X,\gfrak,\rho)=0$ in the Costello local BV obstruction complex
after the chosen open or open--closed counterterms are included, then
the effective interactions $I[L]$ can be chosen to satisfy the QME
through one loop. Under the corresponding higher-loop vanishing or
counterterm hypotheses, the completed observables form a quantum
factorisation algebra on $X$. If the class remains non-zero, the same
compact package has no QME solution with the specified regulator and
field content.
\end{theorem}

\begin{proof}
Costello's obstruction theory solves the QME inductively in powers of
$\hbar$. At the first quantum step, the obstruction is the degree-one
local class of Proposition~\ref{prop:qme-obstruction}, corrected by any
closed-sector or local counterterm contribution. Vanishing of this class
allows the first quantum correction. The same obstruction-theoretic
argument applies at each later loop order, provided the completed local
functional complex and sewing operations in
Hypothesis~\ref{hyp:compact-package} are fixed. A surviving obstruction
class is precisely the failure of the QME in that complex.
\end{proof}

\begin{remark}[Euler characteristics and gauge traces]
\label{rem:euler-gauge}
The topological Euler number
$\chi_{\mathrm{top}}(X)=\int_Xc_3(TX)$ is a characteristic number of the
base. The gauge obstruction keeps the independent quartic datum
$I_4^\rho$. Compact index or BCOV terms may contain
$\chi_{\mathrm{top}}(X)$, while the nonabelian open gauge anomaly is
still measured by the class
$\kanom(X,\gfrak,\rho)$.
\end{remark}

\begin{corollary}[Basic compact examples]
\label{cor:compact-examples}
For $X=K3\times E$,
\[
  \chi_{\mathrm{top}}(K3\times E)=24\cdot0=0,
  \qquad
  \kcat(K3\times E)=\chi(\cO_{K3})\chi(\cO_E)=2\cdot0=0.
\]
The Euler and categorical factors vanish by different Kunneth
computations. A nonabelian compact hCS quantisation on $K3\times E$
still requires the quartic anomaly class
$\kanom(K3\times E,\gfrak,\rho)$ to vanish or be trivialised in the
chosen BV complex.

For the quintic threefold $X_5$,
\[
  \chi_{\mathrm{top}}(X_5)=\int_{X_5}c_3(TX_5)=-200.
\]
This number fixes the base characteristic contribution. The open gauge
obstruction is determined by $I_4^\rho$, the trace normalisation, the
regulator, and the descent representative.
\end{corollary}

\section{Boundary Specialisation}
\label{sec:boundary-specialisation}

\begin{proposition}[$\Eone$ output in complex dimension three]
\label{prop:e1-output}
Let $C\subset X$ be a holomorphic curve and suppose the compact BV
package of Hypothesis~\ref{hyp:compact-package} has been solved on a
neighbourhood of $C$. Integrating the completed quantum observables in
the normal directions gives
\[
  \mathcal A_C=
  \int_{N_{C/X}/C}\Obs^q_{\hCS}
  \in \mathrm{Alg}^{\mathrm{ch}}_{\Eone}(C).
\]
For $d=3$ this specialised chiral output is $\Eone$. The $\Etwo$
structure appears on the centre, double, or module layer, for example
\[
  Z\bigl(\Rep^{\Eone}(\mathcal A_C)\bigr)
  \simeq
  \Rep^{\Etwo}\!\left(
    C^\bullet_{\CE,\mathrm{ch}}(\mathcal A_C,\mathcal A_C)
  \right)
\]
under the chiral BZFN hypotheses.
\end{proposition}

\begin{remark}[$\C^3$ CoHA normalisation]
\label{rem:c3-coha}
On the toric CoHA side,
\[
  \mathrm{CoHA}(\C^3)=Y^+
\]
is the positive half. The full $\mathcal W_{1+\infty}$ object arises
after applying the appropriate double, centre, completion, and
evaluation operations.
\end{remark}

\section{Numerical Invariants}
\label{sec:numerical-invariants}

The BV obstruction $\kanom(X,\gfrak,\rho)$ is separate from the four
Calabi--Yau numerical invariants used elsewhere in the manuscript:
\[
  \kch,\qquad
  \kcat=\chi(\cO_X),\qquad
  \kBKM=c_N(0)/2,\qquad
  \kfib.
\]
The hCS anomaly calculation above uses $\kanom$ and the Lie-theoretic
classes $I_4^\rho$ and $C_2^\rho$. Comparisons with
$\kch,\kcat,\kBKM,\kfib$ require an additional CY-to-chiral
identification and do not follow from the BV calculation alone.
