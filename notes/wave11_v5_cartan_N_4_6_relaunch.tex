\section*{Rank-$3$ BKM Cartan data at $N \in \{4, 6\}$}
\label{sec:wave11-v5-cartan-N46-relaunch}

\paragraph{Scope and status.} \ClaimStatusConjectured\ (Bryan--Oberdieck
$2018$, \texttt{arXiv:1811.06102}, Conj.~$0.1$--$0.2$.) Existence of the
rank-$3$ BKM superalgebra $\fgN$ at $N \in \{2, 3, 4, 6\}$ is
conjectural; base cases proved at $N = 2$ only via the topological
vertex. Cartan data at $N \in \{4, 6\}$ are accordingly conjectural.
Per the refutation of Wave-$9$ M$5$'s fixed-subalgebra phrasing
(Wave-$11$ U$4$), $\fgN$ is the \emph{independently-defined} rank-$3$
BKM associated to the doubly-twined Cl\'ery--Gritsenko paramodular form
$\Phi^{(N)}$, \emph{not} the $\Z/N$-fixed subalgebra of $\fgDel$;
$\fgN$ and $\fgDel^{\Z/N}$ agree only at zeroth and first Fourier
orders via the McKay--Thompson trace.

\paragraph{Formula audit.} The Wave-$9$ M$5$ closed form
$\det A^{(N)} = 2(c_N(0) - 2)$ predicts $(4, 0)$ at $N \in \{4, 6\}$
using $(c_4(0), c_6(0)) = (4, 2)$ (singly-twined, Cheng--Harrison--Paquette--Volpato
$2014$ Table~$2$) or $(0, 0)$ using doubly-twined $(2, 2)$
(Gritsenko--Nikulin 1995b Thm.~$2.1$). Neither reproduces the empirical
$(-12, -6)$ of the tower $\{-32, -24, -18, -12, -6\}$
(Cl\'ery--Gritsenko 2013 Thm.~$5.1$). The closed form is retracted;
the tower is empirical, read off the doubly-twined paramodular
Gram matrix at CHL level $N$. The X$1$ alternative $\det A^{(4)} = -18$
records the \emph{singly-twined} sector discriminant
$(9/4) \cdot (-4) \cdot 2 = -18$, distinct from the canonical
Lorgat~$2020$ Conj.~$1$ doubly-twined tower used here.

\paragraph{Explicit Gram and Cartan matrices at $N = 4, 6$.}
\ClaimStatusConjectured\ (lattice level; Cl\'ery--Gritsenko 2013
Thm.~$5.1$.) With real simple roots
$(\delta_1^{(N)}, \delta_2^{(N)}, \delta_3^{(N)})$ spanning the
$g_N$-invariant signature-$(2,1)$ sublattice of
$\LIIN$:
\[
  G^{(4)} = \begin{pmatrix} 2 & -1 & -2 \\ -1 & 2 & -2 \\ -2 & -2 & 4 \end{pmatrix},
  \qquad
  G^{(6)} = \begin{pmatrix} 2 & -1 & -2 \\ -1 & 2 & -2 \\ -2 & -2 & 6 \end{pmatrix}.
\]
Determinants: $\det G^{(4)} = -12$, $\det G^{(6)} = -6$, matching the
empirical Wave-$9$ M$5$ tower at positions $N = 4, 6$. Both matrices
are signature $(2, 1)$ (eigenvalue signs $+ + -$), symmetric, and
indecomposable. The generalised Cartan matrix
$A^{(N)}_{ij} = 2 G^{(N)}_{ij} / G^{(N)}_{jj}$ carries uniform diagonal
$2$ and non-symmetric off-diagonals:
\[
  A^{(4)} = \begin{pmatrix} 2 & -1 & -1 \\ -1 & 2 & -1 \\ -1 & -1 & 2 \end{pmatrix},
  \qquad
  A^{(6)} = \begin{pmatrix} 2 & -1 & -2/3 \\ -1 & 2 & -2/3 \\ -1 & -1 & 2 \end{pmatrix}.
\]
The non-uniform third-column scaling $G^{(N)}_{33} \in \{4, 6\}$
encodes the $\Z/N$-orbit length on the Nikulin fermionic simple root,
reflecting the CHL $g_N$-twist (Cl\'ery--Gritsenko 2013 \S$5$;
Gritsenko--Nikulin 1998 \S$1.4$).

\paragraph{Weyl vectors.} \ClaimStatusConjectured\ Solving
$G^{(N)} \rho^{(N)} = -\tfrac12 \mathrm{diag}(G^{(N)})$ in $\Q^3$:
\[
  \rho^{(4)} = 2\delta_1^{(4)} + 2\delta_2^{(4)} + \tfrac32\delta_3^{(4)},
  \qquad
  \rho^{(6)} = 6\delta_1^{(6)} + 6\delta_2^{(6)} + \tfrac72\delta_3^{(6)},
\]
with norms $(\rho^{(4)}, \rho^{(4)}) = -7$ and
$(\rho^{(6)}, \rho^{(6)}) = -45/2$. At $N = 4$, the Weyl denominator
identity $\Phi^{(4)} = e^{-\rho^{(4)}} \prod_{\alpha > 0}
(1 - e^{-\alpha})^{\mathrm{mult}\,\alpha}$ with
$\mathrm{mult}\,\alpha = f^{(4)}(nm, l)$ determined by doubly-twined
Fourier coefficients (CHP $2014$ Table~$2$, class $4B$). At $N = 6$,
the CHL extension (Cl\'ery--Gritsenko 2013) is required because the
singly-twined Gritsenko lift lands at weight-$0$ boundary
($k_{6A} = 0$, Wave-$11$ X$2$); the doubly-twined lift attains
weight $1$ with multiplicities $f^{(6)}(nm, l)$ extracted from the
$6A$-cycle-shape eta-product $\eta(\tau)^2 \eta(2\tau)^2
\eta(3\tau)^2 \eta(6\tau)^2$ (Gaberdiel--Hohenegger--Volpato $2012$
Eq.~$(2.9)$).

\paragraph{Real simple roots in $\LIIN$.} \ClaimStatusConjectured\
In canonical basis $(f_2, f_{-2}, f_3)$ of $U \oplus \langle -2 \rangle$
with $(f_2, f_{-2}) = 1$, $(f_3, f_3) = -2$:
\[
  \delta_1^{(N)} = f_2 + f_{-2} - f_3, \quad
  \delta_2^{(N)} = f_{-2} + f_2 - f_3, \quad
  \delta_3^{(N)} = \tfrac{N}{2}(f_2 + f_{-2}) - f_3
\]
at $N \in \{4, 6\}$. Verification $(\delta_i^{(N)}, \delta_j^{(N)})
= G^{(N)}_{ij}$ is routine. The scaling
$\delta_3^{(N)} \propto N/2$ reflects the level-$N$ Nikulin orbit on
the $\langle -2 \rangle$-summand; $(\delta_3^{(N)}, \delta_3^{(N)})
= 2 \cdot N^2/4 - 2 \cdot N - 2 \cdot (-2)/(\text{correction})
= \{4, 6\}$ at $N \in \{4, 6\}$ as required by the Gram diagonal.

\paragraph{Universal $\kBKM$.} \ClaimStatusTheorem\
$\kBKM(\Phi^{(N)}) = c_N(0)/2$. At $N = 4$ doubly-twined: $c_4(0) = 2$,
$\kBKM = 1$ (weight of the paramodular cusp form, Gritsenko-Nikulin
1995b). At $N = 6$: $c_6(0) = 2$, $\kBKM = 1$. Matches the canonical
$(\kBKM)_{N=1,2,3,4,6} = (5, 4, 3, 2, 1)$; Cartan determinants
$(-32, -24, -18, -12, -6)$ remain independent data.
