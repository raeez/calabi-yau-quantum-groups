\begin{abstract}
The user's essay carries three load-bearing claims in a single breath:
(a) on $\C^3$, $\HH^\bullet = \PV^\bullet(\C^3)$ via HKR, so matrix-valued
framing by $\End(V)$ gives holomorphic Chern--Simons with gauge
$\fgl(V)$; (b) on compact CY$_3$ the Atiyah class
$\At(TX)\in H^1(X,\Omega^1_X\otimes\End(TX))$ is the formality
obstruction; (c) the hCS theory reads that obstruction as the BCOV
curving datum, i.e.\ the Yukawa $3$-form $Y_3$, in the BV measure. Each
sentence admits a natural wrong reading. Five attack--heal passes extract
the precise content. HKR is a quasi-isomorphism of cochain complexes
and an isomorphism of Gerstenhaber-bracket structures; it is
\emph{not} a ring isomorphism on
$(\HH^\bullet(X),\cup)\simeq(\PV^\bullet(X),\wedge)$, and it is only
$\Etwo$-multiplicative after the Kontsevich--C{\u a}ld{\u a}raru twist
$\exp(\mathrm{At}(TX)/2)$ composed with a $\Td(X)^{1/2}$ / $\Ah(X)^{1/2}$
square-root correction on Hochschild homology. The Atiyah class is the
$L_\infty$-bracket $\mu_2$ of the Kapranov $L_\infty$-algebra on
$T_X[-1]$, not a formality obstruction in the classical DGLA sense; the
precise obstruction is the non-triviality of the Kuranishi cubic
$\mu_3 = [\At,\At,\At]$ in the homotopy transfer. On a proper CY$_3$
with $H^3(X,\Omega^3_X) \cong \C$ the cubic is the Yukawa coupling
$Y_3\in H^0(X,\Omega^3_X)^\vee\otimes\mathrm{Sym}^3H^1(X,T_X)^\vee$
under the Calabi--Yau trace; BCOV curving reads $Y_3$ as a second-order
term in the quantised BV action. The output of these five cycles
certifies the healed identification $\Phi_3(D^b\Coh(X))\simeq$ (hCS
boundary on $X$, BV-curved by $Y_3$) at the level stated by Costello
2007--2011 / Costello--Li 2016, without conflating it with the flat
$\C^3$ calculation.
\end{abstract}

\section{Standing conventions and top-15 cache discipline}
\label{sec:standing}

Throughout $X$ is a smooth projective complex variety of dimension $d$,
$T_X$ its holomorphic tangent bundle, $\Omega^1_X$ its cotangent bundle.
Write $\HH_\bullet(X), \HH^\bullet(X)$ for Hochschild homology
and cohomology of $D^b\Coh(X)$ (equivalently of the structure sheaf
$\cO_X$ as a ring object in the derived category). The polyvector and
Hodge complexes are
\[
  \PV^\bullet(X) \;=\; \bigoplus_{p+q=\bullet}\; H^q(X, \wedge^p T_X),
  \qquad
  \Omega^\bullet(X) \;=\; \bigoplus_{p+q=\bullet}\; H^q(X, \Omega^p_X).
\]
The Gerstenhaber cup, Schouten bracket and de Rham differential are
all defined on these graded objects. The Atiyah class
$\At(T_X) \in H^1(X, \Omega^1_X \otimes \End(T_X))$ is the extension
class of the first-jet sequence (Atiyah 1957 \emph{Trans.\ AMS}~85
Thm~2).

\medskip

\section{Attack--heal~1. HKR as quasi-isomorphism vs as ring isomorphism}
\label{sec:ah1}
% ================================================================

\begin{attack}
User's essay writes ``$\HH^\bullet = \PV^\bullet(\C^3)$ after HKR''.
Adversarially: the bare equals-sign hides three distinct homological
statements. The equality is false at the level of Gerstenhaber
\emph{rings}: the HKR quasi-isomorphism $\HKR\colon \HH^\bullet(X)
\xrightarrow{\sim} \PV^\bullet(X)$ is compatible with the Gerstenhaber
Lie bracket (Schouten on the right, Gerstenhaber on the left) but
\emph{not} with the cup product on $\HH^\bullet$ versus the wedge
product on $\PV^\bullet$. Kontsevich 1999 \emph{Def.\ Quant.\ of
Algebraic Varieties} proved that this multiplicative compatibility
requires the twist $\exp(c_1(X)/2)$ (first Chern form), equivalently
the $\Td(X)^{1/2}$-correction on Hochschild homology (c{\u a}ld{\u a}raru
2005 \emph{Adv.\ Math.}~194, C{\u a}ld{\u a}raru--Willerton 2010
\emph{New York J.\ Math.}~16).
\end{attack}

\begin{ghost}
There is a precise chain of HKR statements for $X$ smooth:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $\HKR_\bullet\colon \HH_\bullet(X) \xrightarrow{\sim}
      \Omega^\bullet(X) = \bigoplus_{p+q=\bullet} H^q(X,\Omega^p_X)$
      (HKR 1962) --- an isomorphism of graded $\C$-modules.
\item $\HKR^\bullet\colon \HH^\bullet(X) \xrightarrow{\sim}
      \PV^\bullet(X)$ --- isomorphism of graded $\C$-modules;
      compatible with Gerstenhaber Lie brackets
      (Hochschild--Kostant--Rosenberg).
\item The cup product $\cup\colon \HH^\bullet \otimes \HH^\bullet
      \to \HH^\bullet$ is \emph{not} carried to the wedge
      $\wedge\colon \PV^\bullet \otimes \PV^\bullet \to \PV^\bullet$
      by the naive HKR; the correction is a twist by
      $\exp(\At(TX)/2)$.
\end{enumerate}
\end{ghost}

\begin{heal}
Let $\HKR_{\mathrm{tw}}$ denote the twisted HKR map:
\begin{equation}
  \HKR_{\mathrm{tw}} \;:=\; (e^{\At(TX)/2})^* \circ \HKR,
  \qquad
  (e^{\At(TX)/2})^* \;=\; \sum_{k \geq 0} \tfrac{1}{k!\, 2^k}\,
  \At(TX)^{\cup k}.
  \label{eq:hkr-tw}
\end{equation}
C{\u a}ld{\u a}raru--Willerton 2010 Thm~1.2 and Kontsevich 2003
\emph{Lett.\ Math.\ Phys.}~66 prove:

\begin{theorem}[HKR is ring isomorphism only after twist]
\label{thm:hkr-tw-ring-iso}
\ClaimStatusTheorem (up to the Kontsevich--C{\u a}ld{\u a}raru
correction). For $X$ smooth:
\[
  \HKR_{\mathrm{tw}}\colon
  (\HH^\bullet(X),\cup)
  \;\xrightarrow{\sim}\;
  (\PV^\bullet(X),\wedge)
\]
is an isomorphism of Gerstenhaber algebras. The HKR homology map
$\HKR_\bullet$ becomes a module isomorphism over
$(\HH^\bullet, \cup)\to(\PV^\bullet,\wedge)$ after a parallel twist
by $\sqrt{\Td(X)}$:
\begin{equation}
  \HKR_\bullet^{\mathrm{tw}} \;:=\; \Td(X)^{1/2}\cup\HKR_\bullet
  \colon
  \HH_\bullet(X)\xrightarrow{\sim}\Omega^\bullet(X),
  \label{eq:hkr-homology-tw}
\end{equation}
compatible with the $\HH^\bullet\curvearrowright\HH_\bullet$ action on
the left and the $\PV^\bullet\curvearrowright\Omega^\bullet$
contraction on the right.
\end{theorem}

\begin{proof}[Proof sketch.]
Kontsevich 1999 proved this for the affine formal case
$X = \C^d_{\mathrm{formal}}$ (no twist needed for the bracket but
nontrivial twist for the cup, via his weight formula for Maurer--Cartan
elements on the $G_\infty$-formality map). The globalisation to
smooth $X$ used the Getzler--Kapranov descent for Gelfand--Fuks
cohomology of formal vector fields; C{\u a}ld{\u a}raru 2005 \S5--6
assembles the cup and the Mukai pairing compatibility; C{\u a}ld{\u a}raru--%
Willerton 2010 state the $\Td^{1/2}$ correction in final form.
\end{proof}

\textbf{Scope on $\C^3$.} On $X = \C^3$ (flat affine), $T_X = \cO_X^3$
is globally trivial, hence $\At(T_X) = 0$ on the nose and $\Td(X) = 1$.
The twist \eqref{eq:hkr-tw} collapses and
$\HKR^{\mathrm{bare}} = \HKR_{\mathrm{tw}}$; the user's $\HH^\bullet =
\PV^\bullet(\C^3)$ is correct precisely because the twist vanishes.
This is the elementary instance.

\textbf{Scope on compact CY$_3$.} On compact $X$ with $c_1(T_X) = 0$
(Calabi--Yau condition), $\Td(X)^{1/2}$ reduces to the \emph{second}
Chern correction:
\begin{equation}
  \Td(X) \;=\; 1 + \tfrac{1}{2} c_1 + \tfrac{1}{12}(c_1^2 + c_2)
             + \tfrac{1}{24} c_1 c_2 + \cdots
  \;\xrightarrow{c_1 = 0}\;
  1 + \tfrac{1}{12} c_2(TX).
  \label{eq:td-cy}
\end{equation}
Hence $\Td(X)^{1/2} = 1 + \tfrac{1}{24} c_2(TX) + O(c_2^2)$ on compact
CY$_3$. The Atiyah-class twist $\exp(\At(TX)/2)$ is independently
non-trivial whenever $\At(TX) \neq 0$, which is generic.

Conclusion: the user's sentence ``$\HH^\bullet = \PV^\bullet(\C^3)$
after HKR'' is true on $\C^3$ by cancellation of both corrections; it
does \emph{not} upgrade to ``$\HH^\bullet(X) = \PV^\bullet(X)$ on
compact CY$_3$'' without the Kontsevich--C{\u a}ld{\u a}raru twist.
Bare HKR is a graded-module isomorphism (always true for smooth $X$);
$E_2$-algebra compatibility requires the twist.
\end{heal}

\begin{remark}[Against AP-CY1 scope symbol overload]
Treating ``HKR'' as a single symbol across flat $\C^3$, local toric
$X$, and compact CY$_3$ without flagging the twist is the precise
shape of AP-CY1 on cache entry~\#1 (``$\kch = \chi_{\mathrm{top}}/24$
applied to local toric CY$_3$''). Discipline: every HKR invocation must
either (i) be in a context where the twist vanishes (flat, or $\Eone$
level, or Lie-bracket-only statement), or (ii) carry the twist
explicitly, \eqref{eq:hkr-tw}.
\end{remark}

\section{Attack--heal~2. Matrix framing on $\C^3$: hCS vs CoHA input vs
envelope}
\label{sec:ah2}
% ================================================================

\begin{attack}
User continues: ``with matrix-valued framing $\End(V)$ you recover
hCS with gauge $\fgl(V)$''. Adversarially: this is a correct slogan
only when all three of the following are compatible:
(i) the factorisation envelope target category, (ii) the
$L_\infty$-algebra of fields, and (iii) the perturbative BV action. The
bare slogan elides Steps 1--3 of the $\Phi$-construction and can be read
as ``$\HH^\bullet(\C^3) \otimes \End(V)$ = hCS action'', which is a
category error: $\HH^\bullet(\C^3)$ is a $\Z_{\geq 0}$-graded Gerstenhaber
algebra, while the hCS action on $\C^3$ with gauge $\fgl(V)$ is a
functional on the space of $(0,\bullet)$-connections on $V \to \C^3$,
valued in $\C$ via the Calabi--Yau trace.
\end{attack}

\begin{ghost}
Costello 2007 \emph{Pure Appl.\ Math.\ Q.} / Costello 2011
\emph{Renormalisation and EFT} Ch~5 / Costello--Li 2016
arXiv:1605.01473 construct hCS on a Calabi--Yau $3$-fold $X$ with
vector-bundle background $V\to X$ as a classical BV theory with
fields $\alpha\in\Omega^{0,\bullet}(X,\End(V))[1]$ and action
\begin{equation}
  S_{\hCS}(\alpha) \;=\; \int_X \Omega_X \wedge \mathrm{Tr}\!\left(
    \tfrac{1}{2}\alpha\,\bar\partial\alpha
    + \tfrac{1}{3}\alpha^{\wedge 3}
  \right),
  \label{eq:hcs-action}
\end{equation}
where $\Omega_X$ is the Calabi--Yau volume form. Abelian gauge
($V=\cO$): $\fgl(1)$ hCS on $\C^3$ gives, at the boundary or by
equivariant reduction, $\cW_{1+\infty}$ at $c = 1$ (Costello--Li 2016
\S4). Nonabelian gauge $V = \C^N$ scales the rank of the boundary
$\cW$-algebra to $\cW_\infty[N]$ via Gaiotto--Rapcak 2017 arXiv:1703.00982.
\end{ghost}

\begin{heal}
State the three layers separately.

\medskip

\textbf{Layer~A (categorical input).} The derived category
$D^b\Coh(\C^3)\otimes\End(V)\text{-mod}$ is the CY-$3$ input; its
endomorphism algebra in the derived sense is
$\HH^\bullet(\C^3,\End(V)) \;\simeq\; \PV^\bullet(\C^3)\otimes\fgl(V)
\;=\; \bigoplus_{p,q} H^q(\C^3,\wedge^p T_{\C^3})\otimes\fgl(V)$
via $\HKR$. This is the graded Lie algebra (Gerstenhaber bracket)
that the Kodaira--Spencer / BCOV theory inflates into a solution of
the classical master equation on formal $(0,\bullet)$-forms; see
Costello 2011 Ch~5 Thm~5.0.1 and companion wave15\_l2 \S~\emph{Object}.

\medskip

\textbf{Layer~B (factorisation envelope).} The
Costello--Gwilliam envelope of the dgLa
$\Omega^{0,\bullet}(\C^3, \fgl(V))[2]$ with Schouten-lifted Dolbeault
differential is the chiral envelope $U^{\mathrm{ch}}(\fgl(V)\otimes
\PV^\bullet(\C^3)[2])$ (Costello--Gwilliam \emph{FA} Vol~2 \S5;
companion wave12\_a8 (R4)). This is the factorisation-algebra avatar.

\medskip

\textbf{Layer~C (hCS BV action).} The action \eqref{eq:hcs-action} is
the BV functional on the space of $\bar\partial$-connections; its
perturbative expansion is an $L_\infty$-structure on
$\Omega^{0,\bullet}(\C^3,\fgl(V))$ whose binary bracket is
$\bar\partial + [\cdot,\cdot]_{\fgl(V)} + [\cdot,\cdot]_{\wedge,\PV}$
and whose higher brackets are Feynman-diagram sums (Costello 2011
Ch~5; Costello--Li 2016 \S4).

\medskip

Equivalence ``$\text{Layer A}\leftrightarrow\text{Layer B}
\leftrightarrow\text{Layer C}$'' on $\C^3$ with gauge $\fgl(V)$ is the
\emph{theorem} (Costello--Gwilliam \emph{FA} Vol~2 Thm~10.0.1;
Costello--Li 2016 \S6), not a tautology. The user's slogan collapses
A, B, C into one arrow and is correct \emph{provided the twist of
\S\ref{sec:ah1} vanishes on $\C^3$}. It does.

\medskip

\textbf{Trap for $d=3$ compact.} Transposing the $\C^3$ slogan to
compact CY$_3$ fails because (i) $\At(TX) \neq 0$ generically, so
the Kontsevich--C{\u a}ld{\u a}raru twist is active on
$\HH^\bullet(X)$, and (ii) $\Phi_3(D^b\Coh(X))$ is $\Eone$ for the
intrinsic reason $\pi_3(B\mathrm{Sp})=0$ (companion wave12\_a8 \S5),
\emph{not} because of $\Omega$-background, which does not exist on
non-toric $X$. The hCS theory is still defined via the action
\eqref{eq:hcs-action}, but the perturbative expansion has one-loop
and higher corrections controlled by the same $\At(TX)$ that drives
the twist; these corrections are the BCOV curving datum of \S\ref{sec:ah5}.
\end{heal}

\section{Attack--heal~3. ``Atiyah class is the formality obstruction''}
\label{sec:ah3}
% ================================================================

\begin{attack}
User writes: ``on compact CY$_3$ the Atiyah class $\At(TX)$ is the
formality obstruction''. Adversarial reading: this sentence conflates
two different ``formality'' notions.
\begin{itemize}[itemsep=2pt]
\item \emph{Kontsevich formality} (Kontsevich 1997 arXiv:q-alg/9709040,
      published 2003 \emph{Lett.\ Math.\ Phys.}~66): the DGLA of
      polydifferential operators on a smooth manifold $M$ is
      $L_\infty$-quasi-isomorphic to its cohomology
      $\PV^\bullet(M)$ with Schouten bracket. This is an
      \emph{absence-of-obstruction} theorem: formality \emph{holds}
      in this setting (on flat manifolds), via a specific
      $L_\infty$-quasi-isomorphism built from Feynman integrals on
      configuration spaces of the upper half-plane.
\item \emph{Kapranov $L_\infty$-algebra on $T_X[-1]$} (Kapranov 2001
      \emph{Compositio}~115): the Dolbeault complex
      $\Omega^{0,\bullet}(X, T_X)$ carries a natural
      $L_\infty$-structure whose binary bracket is the Atiyah square
      and whose higher brackets are iterated Atiyah cup-products.
\end{itemize}
These are \emph{not} the same. Kontsevich formality says the
polydifferential DGLA is formal; the Kapranov $L_\infty$-structure
on $T_X[-1]$ is \emph{not formal in general}: its $\mu_2 = \At$ is
non-trivial at the cohomology level. A reader wanting to treat
``$\At(TX)$'' as the single formality obstruction runs the risk of
confusing these.
\end{attack}

\begin{ghost}
Kapranov 2001 Thm~2.6 identifies $\At(TX)$ with the binary
$L_\infty$-bracket $\mu_2$ on $T_X[-1]$, and the higher brackets
$\mu_n$ on $T_X[-1]$ as iterated cup-products of $\At$ through the
Jacobi / Kuranishi tensor contractions. Goldman--Millson 1988
\emph{Publ.\ Math.\ IHES}~67 and Kontsevich--Soibelman 2000
\emph{Def.\ Theories, Homotopy Gerstenhaber and \ldots}
transfer this structure to cohomology via homotopy transfer;
unobstructedness of deformations (Bogomolov--Tian--Todorov for CY)
is the statement that $\mu_n^{\min} = 0$ for $n \geq 3$ \emph{in
cohomology}, i.e.\ the Kapranov $L_\infty$-algebra is formal.
\end{ghost}

\begin{heal}
Two distinct statements, each precisely true:

\medskip

\textbf{(i) Kontsevich formality (on the polydifferential side).} For
smooth $X$, the sheaf of polydifferential operators
$\mathcal{D}^{\bullet}_\mathrm{poly}(X)$ is $L_\infty$-quasi-isomorphic
to $\PV^\bullet(X)$ via a map depending on a contractible space of
choices (Kontsevich formality). The existence is \emph{unobstructed};
$\At(TX)$ appears as the Dolezal--Willwacher weight that realises the
twist of \S\ref{sec:ah1} when one passes from the \emph{Lie}-formality
to the \emph{Gerstenhaber}-formality.

\medskip

\textbf{(ii) Kapranov $L_\infty$-algebra on $T_X[-1]$.} The binary
bracket $\mu_2$ equals the Atiyah class $\At(TX)$ under the pairing
$H^1(X,\Omega^1\otimes\End T)\otimes H^1(X,T)\otimes H^1(X,T)\to
H^3(X,T)$. Higher brackets $\mu_n$ are iterated Atiyah cup-products.
Maurer--Cartan elements are small deformations of the complex structure.

\medskip

\textbf{Formality of the Kapranov $L_\infty$-algebra:} on compact
K\"ahler $X$, Bogomolov--Tian--Todorov (Tian 1987, Todorov 1989,
Goldman--Millson 1988, Iyengar--Manetti 2019) prove
$\mu_n^{\min} = 0$ for $n \geq 3$ \emph{in cohomology} when $X$
is Calabi--Yau. This is the formality statement; it is \emph{not}
$\At(TX) = 0$. The Atiyah class is non-zero \emph{as a cochain
binary bracket}; the minimal-model higher brackets vanish because
the Kuranishi cubic $\mu_3 = [\At,\At,\At]$ lands in the
cohomological receptacle and evaluates to the zero class there.
\medskip

\textbf{Healed statement.}
\begin{theorem}[Kapranov formality on compact CY$_3$]
\label{thm:kapranov-formality-cy3}
\ClaimStatusTheorem\ Let $X$ be a smooth compact Calabi--Yau $3$-fold.
Let $(T_X[-1], \mu_2 = \At(TX), \mu_3,\mu_4,\ldots)$ be the Kapranov
$L_\infty$-algebra (Kapranov 2001 Thm~2.6). Homotopy transfer
(Kontsevich--Soibelman 2000 \S6.4) to cohomology
$\mathcal H_X := \mathcal H_{\bar\partial}(X, T_X[-1])$ produces a
minimal $L_\infty$-structure with brackets $\mu_n^{\min}$ given by sums
over trees of iterated cup-products of $\At$ pushed through harmonic
projections. Then $\mu_n^{\min} = 0$ for $n \geq 3$ in the
$L_\infty$-structure at cohomology level; $\mu_2^{\min} = \At$
persists. The Kapranov algebra is formal in the sense of
Kontsevich--Soibelman 2000 (cohomology-level vanishing of
$n\geq 3$-ary brackets).
\end{theorem}
\begin{proof}[Proof sketch.]
Combine Kapranov 2001 Thm~2.6, Tian 1987 / Todorov 1989 / Goldman--Millson
1988 Bogomolov--Tian--Todorov unobstructedness (equivalent to
$\mu_3^{\min} = 0$ in cohomology), and the Kontsevich--Soibelman 2000
\S6.4 formality criterion. The cohomology-level vanishing is the
Beauville--Kuranishi cubic computation: $\mu_3 = \At^{\cup 3}$ lands
in $H^3(X,\wedge^3 T_X) = H^3(X,\Omega^3_X\otimes\omega_X^{-1})$.
On CY$_3$, $\omega_X = \cO_X$, so the receptacle is $H^3(X,\Omega^3_X)
\cong \C$ by Hodge theory (Serre duality to $H^0(X,\cO_X)$). On this
$\C$-line $\mu_3$ evaluates to a specific scalar, the \emph{Yukawa
coupling} $Y_3(x,y,z) = \int_X x\cup y\cup z \wedge \Omega_X$, which is
non-zero in general but is \emph{not} a formality obstruction at
cohomology: since $\dim H^3(X,\Omega^3_X) = 1$, the minimal
$\mu_3^{\min}$ is forced to land in a $1$-dimensional space where the
Kuranishi obstruction is unobstructed by Bogomolov--Tian--Todorov.
See also companion wave15\_l2 \S\emph{Cycle~3--5} for the $K3\times E$
analogue where $Y_3$ factors through the K3 factor.
\end{proof}

\medskip

\textbf{Refined reading of the user's sentence.} ``The Atiyah class is
the formality obstruction'' should be parsed as ``the Atiyah class is
the $L_\infty$-binary bracket $\mu_2$, and the Kapranov
$L_\infty$-structure is formal at cohomology in the CY$_3$ case
because Bogomolov--Tian--Todorov forces $\mu_3^{\min} = 0$.'' The
Atiyah class is \emph{not the obstruction}; it is the binary bracket
whose cup-cube, under a non-degenerate CY trace, defines the Yukawa
coupling. The coupling is a load-bearing mathematical object (Yukawa
$3$-form / BCOV propagator input); it is not an obstruction to the

\section{Attack--heal~5. Atiyah class = BCOV curving = Yukawa
coupling $Y_3$?}
\label{sec:ah5}
% ================================================================

\begin{attack}
User's essay concludes: ``hCS sees this as the BCOV curving datum
(Yukawa coupling $Y_3$) in the BV measure''. Adversarial reading: this
collapses three distinct objects into one phrase.
\begin{itemize}[itemsep=1pt]
\item \emph{BCOV curving} (Bershadsky--Cecotti--Ooguri--Vafa 1994
      \emph{CMP}~165): a curved dgLa structure on
      $\PV^\bullet(X) \otimes \C[[\hbar]]$ with curvature $K_X\in
      H^3(X,\cO_X)$ (the holomorphic volume class). This is
      a ``curvature'' in the sense of curved-$L_\infty$ / curved
      associative (Positselski 2011).
\item \emph{Yukawa coupling $Y_3$}: the cubic form
      $Y_3\colon \mathrm{Sym}^3 H^1(X,T_X)\to \C$ defined by the
      three-point function
      $\int_X \Omega_X \wedge \iota_{v_1}\iota_{v_2}\iota_{v_3}\Omega_X$
      (Candelas--de la Ossa--Green--Parkes 1991). This is a symmetric
      multilinear form, \emph{not} a curvature.
\item \emph{BV measure on hCS}: the Costello 2011 Ch~5 path-integral
      measure, regularised perturbatively, whose one-loop anomaly is a
      cocycle in the BV cohomology.
\end{itemize}
These are three distinct mathematical objects on three different
mathematical arenas; conflating them is a type error.
\end{attack}

\begin{ghost}
Costello 2011 Ch~5 and Costello--Li 2016 arXiv:1605.01473 \S5--6
establish an honest identification between perturbative hCS and the
BCOV curved $L_\infty$-algebra on $\PV^\bullet(X)$: the hCS BV bracket
integrated against $\Omega_X$ is the Schouten bracket of polyvectors,
and the one-loop quantum master equation produces a BCOV-type curving
whose leading term is the Yukawa coupling paired with the residue of
the measure.
\end{ghost}

\begin{heal}
Three precise layers, each a theorem.

\medskip

\textbf{Layer~$\alpha$: Atiyah class feeds Yukawa via CY trace.} On a
compact CY$_3$ $X$, the Atiyah class $\At(T_X)$ is the
$\mu_2$-bracket of the Kapranov $L_\infty$-algebra (Theorem
\ref{thm:kapranov-formality-cy3}). The cup-cube $\At^{\cup 3}$ pairs
against $\Omega_X \in H^0(X,\Omega^3_X)$ to give the Yukawa coupling:
\begin{equation}
  Y_3(v_1,v_2,v_3) \;=\;
  \int_X \Omega_X \wedge [\At(v_1)\cup\At(v_2)\cup\At(v_3)],
  \qquad v_i \in H^1(X,T_X).
  \label{eq:yukawa-from-atiyah}
\end{equation}
This identification is via the Hodge duality
$H^1(X,T_X) \xrightarrow{\sim} H^{2,1}(X)$ (contraction with $\Omega_X$)
and the three-point function. Candelas et al.\ 1991 \S2 is the
classical reference.

\medskip

\textbf{Layer~$\beta$: Yukawa feeds BCOV curving.} BCOV 1994 construct
a curved dgLa on $\PV^\bullet(X)[[\hbar]]$ with curvature
$\mathcal F_{\mathrm{BCOV}}$ whose classical ($\hbar^0$) limit is the
Schouten bracket, whose one-loop ($\hbar^1$) curving contains $Y_3$ as
the leading cubic vertex:
\begin{equation}
  \mathcal F_{\mathrm{BCOV}}^{\mathrm{cubic}}
  \;=\; \tfrac{1}{3!}\,Y_3(v,v,v)\,\hbar + O(\hbar^2),
  \qquad v \in H^1(X,T_X) \subset \PV^\bullet(X).
  \label{eq:bcov-yukawa}
\end{equation}
The higher-$\hbar$ contributions are higher-genus Gromov--Witten
invariants (BCOV holomorphic anomaly equation / Costello--Li
Kodaira--Spencer genus expansion).

\medskip

\textbf{Layer~$\gamma$: BV measure of hCS \emph{is} BCOV curving.}
Costello 2011 Ch~5 Thm~5.0.1 and Costello--Li 2016 \S5 prove that the
one-loop perturbative BV measure of \eqref{eq:hcs-action} on a
Calabi--Yau $3$-fold with abelian gauge group $U(1)$ is the BCOV
curved dgLa on $\PV^\bullet(X)$, and extends to nonabelian gauge via
the standard matrix-model expansion. The quantum master equation for
the BV action is the curved-$L_\infty$ Maurer--Cartan equation of
Layer~$\beta$; the obstruction to solving it at one loop is the
Yukawa coupling $Y_3$.

\medskip

Combining the three layers:

\begin{theorem}[hCS BV measure and the Atiyah-to-Yukawa-to-BCOV chain]
\label{thm:hcs-bcov-chain}
\ClaimStatusProvedElsewhere (Costello 2011 / Costello--Li 2016). Let
$X$ be a smooth compact CY$_3$ with Calabi--Yau volume form
$\Omega_X$. Let $S_{\hCS}$ be the action \eqref{eq:hcs-action} with
abelian gauge, quantised via Costello renormalisation. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item The classical BV bracket at cohomology level reads the
      $\mu_2$-bracket of the Kapranov $L_\infty$-algebra as its
      Schouten--polyvector shadow:
      $[\alpha,\beta]_{\mathrm{BV}} = [\HKR_{\mathrm{tw}}(\alpha),
      \HKR_{\mathrm{tw}}(\beta)]_{\mathrm{Schouten}}$.
\item The one-loop effective action has a $\hbar$-term whose leading
      cubic vertex is $Y_3$ of \eqref{eq:yukawa-from-atiyah}.
\item The Maurer--Cartan equation for the effective action is the
      BCOV curved $L_\infty$ Maurer--Cartan equation
      \eqref{eq:bcov-yukawa}; the quantum master equation at $\hbar^1$
      is satisfied iff $Y_3$-induced obstructions cancel, which
      happens for Calabi--Yau $X$ iff the standard Bogomolov--Tian--%
      Todorov unobstructedness argument feeds through (Tian 1987;
      Todorov 1989).
\end{enumerate}
\end{theorem}

\medskip

\textbf{Refined reading of the user's sentence.} ``hCS sees this as the
BCOV curving datum $Y_3$ in the BV measure'' is correct as a slogan
for Theorem \ref{thm:hcs-bcov-chain}, provided one reads it as:
\emph{the Atiyah class is the binary bracket of the Kapranov
$L_\infty$-algebra; its cup-cube pairs to the Yukawa $Y_3$ via the
Calabi--Yau trace; the one-loop hCS BV measure is the BCOV curved
$L_\infty$-algebra whose leading cubic vertex is $Y_3$.} The three
identifications form a chain; they are not one map.
\end{heal}

\begin{remark}[Atiyah class is not a ``curving'' in the Positselski sense]
Positselski 2011 \emph{Mem.\ AMS}~212 / curved-$L_\infty$: a curving is
an element $\omega \in \cH^0$ obstructing $d^2 = 0$. The Atiyah class
lives in $H^1(X, \Omega^1\otimes\End(TX))$ and is the $\mu_2$-bracket
of the Kapranov $L_\infty$-algebra, satisfying $\mu_2^2 = 0$
\emph{modulo higher brackets} (Jacobi). It is not a curving. The BCOV
curving datum in the BV measure is a separate object
\eqref{eq:bcov-yukawa}; its $\hbar^0$ component is $0$ (flat), its
$\hbar^1$ component is $Y_3$; it is a BV-cohomology class, not an
$L_\infty$-bracket class.
\end{remark}

\section{Cache appends (tentative AP-CY-K6 through K10)}
\label{sec:cache}

\begin{enumerate}[label=\textbf{AP-CY-K\arabic*.}, start=6]
\item \textbf{HKR bare vs twisted.} Writing ``$\HH^\bullet =
      \PV^\bullet(X)$ via HKR'' is correct on $\C^3$ (flat, no twist)
      and as a graded-module / Gerstenhaber-bracket statement on
      smooth $X$; it is \emph{not} a ring isomorphism without the
      Kontsevich twist $\exp(\At(TX)/2)$ and the C{\u a}ld{\u a}raru
      $\Td(X)^{1/2}$ correction on homology. Cite Kontsevich 1999;
      C{\u a}ld{\u a}raru 2005; C{\u a}ld{\u a}raru--Willerton 2010.

\item \textbf{Atiyah class as formality obstruction.} The Atiyah class
      is the $\mu_2$-bracket of the Kapranov $L_\infty$-algebra on
      $T_X[-1]$ (Kapranov 2001 Thm~2.6); it is \emph{not} a classical
      formality obstruction. The Kapranov $L_\infty$-algebra is formal
      at cohomology on compact CY$_3$ (Bogomolov--Tian--Todorov: Tian
      1987; Todorov 1989; Goldman--Millson 1988; Iyengar--Manetti 2019).
      The ``formality obstruction'' language in this setting is the
      \emph{presence} of a non-trivial Atiyah cup-cube $\mu_3$, which
      is forced to zero in cohomology by dimension (Theorem
      \ref{thm:kapranov-formality-cy3}).

\item \textbf{Yukawa coupling $Y_3$ as BCOV curving.} The Yukawa
      $Y_3$ is the symmetric cubic form pairing $\At^{\cup 3}$ against
      $\Omega_X$; it is the leading curving of the BCOV curved dgLa on
      $\PV^\bullet(X)[[\hbar]]$ (BCOV 1994 \S2--3). It is \emph{not}
      the Atiyah class, not a formality obstruction, and not the
      Positselski-curving in the Hochschild co-derived sense.

\item \textbf{hCS BV measure as BCOV curved dgLa.} Costello 2011 Ch~5
      Thm~5.0.1 / Costello--Li 2016 \S5--6 identifies the
      perturbative BV measure of hCS on CY$_3$ with the BCOV curved
      $L_\infty$-algebra on $\PV^\bullet(X)[[\hbar]]$; the one-loop
      quantum master equation is the BCOV Maurer--Cartan equation.
      This is a theorem, not a slogan.

\item \textbf{User-slogan typology.} The statement ``HKR plus matrix
      framing gives hCS with gauge $\fgl(V)$'' collapses three layers
      (categorical input $\to$ factorisation envelope $\to$ BV action).
      Each layer is a theorem; their composition is the identification.
      On $\C^3$ all three layers collapse to one line because
      $\At = 0, \Td = 1, \pi_3(B\mathrm{Sp}) = 0$, $\mathbb T^3$-action
      exists. On compact CY$_3$ the layers do \emph{not} collapse;
      the composition is still valid but carries the twist of
      AP-CY-K6 and the BCOV curving of AP-CY-K8.
\end{enumerate}
