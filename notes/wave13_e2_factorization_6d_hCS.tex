Fix $X=\C^3$ with holomorphic volume form
\[
 \Omega=dz_1\wedge dz_2\wedge dz_3,
\]
and fix a finite-dimensional quadratic Lie algebra
$(\g,\langle-,-\rangle_\g)$. For an open set $U\subset X$ set
\[
 L_U=\Omega_c^{0,\bullet}(U,\g)[1].
\]
The local differential is $\dbar$, the bracket is the pointwise bracket
on $\g$ multiplied by wedge product, and the holomorphic Chern-Simons
functional is
\[
 S_U(A)=\int_U \Omega\wedge
 \left(\frac{1}{2}\langle A,\dbar A\rangle_\g
 +\frac{1}{6}\langle A,[A,A]\rangle_\g\right).
\]
The shifted symplectic pairing
\[
 \omega_U(\alpha,\beta)=\int_U\Omega\wedge
 \langle\alpha,\beta\rangle_\g
\]
pairs $\Omega^{0,p}_c(U,\g)$ with $\Omega^{0,3-p}(U,\g)$ and has BV
degree $-1$ after the shift by $[1]$. In unshifted Dolbeault degree,
the $(0,0)$ component is the ghost, the $(0,1)$ component is the
physical hCS field, and the $(0,2)$ and $(0,3)$ components are the BV
antifields.

\section{Classical Observables}

\begin{definition}[Completed local observables]\label{def:e2-local-observables}
The classical local observables are the completed Chevalley-Eilenberg
cochains
\[
 \Obs^{\mathrm{cl}}(U)
 =
 \CE^\bullet_{\mathrm{cont}}\bigl(L_U\bigr)
 =
 \widehat{\Sym}\bigl(L_U^\vee[-1]\bigr)
\]
in the differentiable-vector-space completion of Costello-Gwilliam.
The differential is the sum of the dual Dolbeault differential and the
dual bracket differential. The completed $n$th symmetric power is
described by distributional kernels on $U^n$; the local-functional
subspace is cut out by support on the diagonals.
\end{definition}

\begin{remark}[Structural completion]\label{rem:e2-completion}
The Bochner-Martinelli propagator is a distributional kernel on
$U\times U$ singular along the diagonal. Polynomial observables alone
fail closure under the BV Laplacian. The distributional completion is
part of the observable algebra.
\end{remark}

\section{Factorisation Structure}

\begin{proposition}[Classical factorisation algebra]\label{prop:e2-classical-factorisation}
The assignment $U\mapsto\Obs^{\mathrm{cl}}(U)$ is a factorisation algebra
on $\C^3$.
\end{proposition}

\begin{proof}
For mutually disjoint opens $U_1,\ldots,U_m\subset V$, extension by zero
sends distributional kernels on $U_i^{n_i}$ to kernels on $V^{n_i}$.
Multiplication in the completed symmetric algebra gives the structure
map
\[
 \bigotimes_i \CE^\bullet_{\mathrm{cont}}(L_{U_i})
 \longrightarrow
 \CE^\bullet_{\mathrm{cont}}(L_V).
\]
Associativity is associativity of kernel extension and of the symmetric
product. Weiss descent follows from Costello-Gwilliam descent for
elliptic local moduli problems: compactly supported Dolbeault fields
form a cosheaf, and completed symmetric powers preserve the required
colimits in the differentiable-vector-space category.
\end{proof}

\section{Quantum Deformation}

\begin{definition}[Costello renormalisation datum]\label{def:e2-renormalisation-datum}
A perturbative quantum hCS datum on $U$ consists of an elliptic gauge
fixing $Q^{\mathrm{GF}}$, heat kernels $K_t$, regularised BV kernels
$P_{\varepsilon,L}$, effective interactions
$I[L]\in\Loc(L_U)[[\hb]]$, and a completed local-functional complex on
which
\[
 QI[L]+\frac12\{I[L],I[L]\}_L+\hb\Delta_L I[L]=0
\]
is meaningful. The bracket $\{-,-\}_L$ and the BV Laplacian $\Delta_L$
are defined by the scale-$L$ regularised kernel. The obstruction at
each order of $\hb$ is a degree-one class in the local BV complex.
\end{definition}

\begin{definition}[Renormalised quantum observables]\label{def:e2-renormalised-bv}
For a datum of Definition~\ref{def:e2-renormalisation-datum} satisfying
the quantum master equation, set
\[
 \Obs^{\mathrm{q}}_L(U)
 =
 \left(
  \widehat{\Sym}\bigl(L_U^\vee[-1]\bigr)[[\hb]],
  d_{\Obs}^{\mathrm{cl}}+\{I[L],-\}_L+\hb\Delta_L
 \right).
\]
\end{definition}

\begin{proposition}[Quantum factorisation under the QME]\label{prop:e2-quantum-factorisation}
If the anomaly class vanishes in the chosen local BV complex or is
trivialised by explicit local or closed-sector counterterms, then
$U\mapsto\Obs^{\mathrm{q}}_L(U)$ is an $\hb$-complete factorisation
algebra deforming
$\Obs^{\mathrm{cl}}$.
\end{proposition}

\begin{proof}
The Costello renormalisation group equation transports the effective
interactions between scales. The quantum master equation gives
\((d_{\Obs}^{\mathrm{cl}}+\{I[L],-\}_L+\hb\Delta_L)^2=0\). The BV
kernel decomposes over disjoint unions, so the classical factorisation
maps commute with the renormalised differential. A surviving
degree-one local class is exactly the obstruction to this differential
squaring to zero.
\end{proof}

\section{Bochner-Martinelli Kernel}

\begin{proposition}[Flat heat-kernel propagator]\label{prop:e2-bm-kernel}
Let \(Q^{\mathrm{GF}}=\dbarstar\) for the Euclidean Hermitian metric and
let \(K_t(z,w)=(4\pi t)^{-3}\exp(-\|z-w\|^2/(4t))\). For
\(0<\varepsilon<L<\infty\), put
\[
 P_{\varepsilon,L}(z,w)
 =
 \int_\varepsilon^L(\dbarstar\otimes1)K_t(z,w)\,dt .
\]
Then \(\dbar P_{\varepsilon,L}=K_L-K_\varepsilon\). Off the diagonal,
as \(\varepsilon\to0\) and \(L\to\infty\),
\[
 P_{\mathrm{BM}}(z,w)
 =
 \frac{2!}{(2\pi i)^3}\sum_{k=1}^3
 (-1)^{k-1}\frac{\overline{z_k-w_k}}{\|z-w\|^6}\,
 d\bar z_1\wedge\cdots\wedge\widehat{d\bar z_k}
 \wedge\cdots\wedge d\bar z_3 .
\]
The distributional identity
\[
 \dbar_w P_{\mathrm{BM}}=\delta_\Delta
\]
holds after the standard residue normalisation around the diagonal.
\end{proposition}

\begin{proof}
The identity \(\dbar P_{\varepsilon,L}=K_L-K_\varepsilon\) is the
heat-kernel telescoping relation for
\(\Delta_{\dbar}=\dbar\dbarstar+\dbarstar\dbar\). Removing the cutoffs
away from the diagonal gives the Green kernel of the flat Dolbeault
Laplacian. Its standard coordinate representative is the displayed
Bochner-Martinelli kernel, and the residue computation gives the
diagonal current.
\end{proof}

\section{One-Loop Obstruction}

\begin{definition}[Quartic invariant]\label{def:e2-quartic-invariant}
For a representation \(\rho:\g\to\End(V)\), define
\[
 I_4^\rho(x_1,x_2,x_3,x_4)
 =
 \frac1{4!}\sum_{\sigma\in S_4}
 \Tr_V\!\left(
  \rho(x_{\sigma(1)})\rho(x_{\sigma(2)})
  \rho(x_{\sigma(3)})\rho(x_{\sigma(4)})
 \right).
\]
Then \(I_4^\rho\in\Sym^4(\g^\vee)^\g\).
\end{definition}

\begin{proposition}[Quartic QME anomaly]\label{prop:e2-qme-obstruction}
In complex dimension three, the nonabelian one-loop obstruction to the
QME is represented in the local Chevalley complex by
\[
 \Theta^\rho_U(\alpha_0,\alpha_1,\alpha_2,\alpha_3)
 =
 \int_U
 \left[
 I_4^\rho\bigl(
  \alpha_0,\partial\alpha_1,\partial\alpha_2,\partial\alpha_3
 \bigr)
 \right]_{(3,3)},
 \qquad
 \alpha_i\in\Omega_c^{0,\bullet}(U,\g).
\]
Its class
\[
 \kanom(X,\g,\rho)
 =
 [\,\hb\,\Theta^\rho\,]
 \in
 H^1_{\mathrm{loc}}
 \bigl(\Omega^{0,\bullet}(X,\g)[1],\cO_{\mathrm{loc}}\bigr)
\]
is the first gauge obstruction to solving the scale-\(L\) QME.
\end{proposition}

\begin{proof}
The relevant one-loop graph is the wheel with four external gauge
insertions. Its Lie factor is the symmetrised trace of four Lie algebra
inputs, hence \(I_4^\rho\). The form factor contributes three
holomorphic derivatives; in complex dimension three this gives a
holomorphic top density. Taking the \((3,3)\)-component produces a
degree-one local functional, the degree in which Costello's obstruction
theory places the first QME obstruction.
\end{proof}

\begin{definition}[Quadratic Casimir]\label{def:e2-casimir}
Let \(\{T_a\}\) be a basis of \(\g\), and let \(\{T^a\}\) be the dual
basis for \(\langle-,-\rangle_\g\). The representation-theoretic
quadratic Casimir is
\[
 C_2^\rho=\sum_a\rho(T_a)\rho(T^a).
\]
For a simple Lie algebra in the usual adjoint normalisation,
\(C_2^{\mathrm{ad}}=2h^\vee\).
\end{definition}

\begin{proposition}[Two-point counterterm]\label{prop:e2-c2-counterterm}
With the flat heat-kernel normalisation, the two-point graph contributes
a degree-zero counterterm of the form
\[
 S_{\mathrm{ct}}^{(2)}
 =
 -\hb\,\frac{C_2^\rho}{(4\pi)^3}
 \log(L/\varepsilon)
 \int_X\Omega\wedge\Tr_\rho(A\,\dbar A),
\]
up to the invariant-pairing and heat-kernel normalisations. This term
renormalises the kinetic pairing and the field \(A\). The QME anomaly
is the quartic class of Proposition~\ref{prop:e2-qme-obstruction}.
\end{proposition}

\begin{proof}
Closing two legs in the cubic vertex gives the colour contraction
\(\sum_a\rho(T_a)\rho(T^a)=C_2^\rho\). The remaining two external
fields have the tensor type of the kinetic functional
\(\int\Omega\wedge\Tr_\rho(A\dbar A)\). The short-distance heat-kernel
integral in six real dimensions supplies the logarithmic factor. The
cohomological degree is zero, so the term changes the kinetic
representative. Proposition~\ref{prop:e2-qme-obstruction} contains the
degree-one obstruction class.
\end{proof}

\section{CE Identification}

\begin{theorem}[Dolbeault CE model]\label{thm:e2-dolbeault-ce}
For every open $U\subset\C^3$ there is an equality of completed graded
objects
\[
 \Obs^{\mathrm{cl}}(U)
 =
 \CE^\bullet_{\mathrm{cont}}\bigl(\Omega_c^{0,\bullet}(U,\g)[1]\bigr).
\]
After a quantum solution of the master equation, the same completed
graded object carries the CE differential of the renormalised effective
$L_\infty$ algebra on $L_U$.
\end{theorem}

\begin{proof}
Definition~\ref{def:e2-local-observables} is the CE cochain definition:
linear observables are $L_U^\vee[-1]$, and the completed symmetric
algebra on them is the CE cochain algebra. The CE differential is dual
to $\dbar$ and to the bracket on $L_U$. Quantisation changes the
differential by the BV Laplacian and counterterm operators. These
operators are local and satisfy the $L_\infty$ identities exactly when
the quantum master equation holds.
\end{proof}

\begin{remark}[Fields, antifields, and categorical polyvectors]\label{rem:e2-separation}
The hCS field complex \(L_U=\Omega_c^{0,\bullet}(U,\g)[1]\) is a
gauge-theoretic Dolbeault dg Lie algebra. Its BV antifields are the
complementary Dolbeault components paired by \(\omega_U\). The
Hochschild polyvectors of a CY category are a separate categorical
object; for example
\[
 \HH^\bullet(\operatorname{Perf}(\C^3))
 \simeq
 \C[z_1,z_2,z_3]\otimes
 \bigwedge\nolimits^\bullet
 \langle\partial_{z_1},\partial_{z_2},\partial_{z_3}\rangle .
\]
These polyvectors control categorical deformations and centre
constructions. A finite-dimensional hCS gauge algebra enters only after
a framing object or external gauge coefficient has been chosen.
\end{remark}

\section{Operadic Scope}

\begin{proposition}[\(\PhiFA_3\) and the specialised output]\label{prop:e2-operadic-scope}
The local hCS observable factorisation algebra on \(\C^3\) is an
\(E_3^{\mathrm{hol}}\) factorisation object at the \(\PhiFA_3\) level
after the framing, formality, QME, and completion witnesses are fixed.
For a specialisation datum \((\Sigma_2,C)\), the specialised
CY-to-chiral output
\[
 \Phi_3^{(\Sigma_2,C)}(\mathcal C)
 =
 \SpCh_{\Sigma_2,C}\bigl(\PhiFA_3(\mathcal C)\bigr)
\]
is \(E_1\)-chiral. An \(E_2\) structure is carried by a centre, double,
or representation-category construction with its own hypotheses.
\end{proposition}

\begin{proof}
The first factor \(\PhiFA_3\) records holomorphic collisions in three
complex directions. The specialisation functor integrates over the
complex surface \(\Sigma_2\) and restricts to the curve \(C\). The
residual level is
\[
 E_{3-2}=E_1.
\]
The Drinfeld centre construction is Morita-theoretic and starts from
the \(E_1\) object. It supplies the \(E_2\) layer after the module
category, pairing, and completion data have been fixed.
\end{proof}

\begin{remark}[$\C^3$ Hall normalisation]\label{rem:e2-c3-coha}
The toric Hall algebra attached to \(\C^3\) is
\[
 \CoHA(\C^3)\simeq \Yp(\widehat{\mathfrak{gl}}_1),
\]
the positive half in the Schiffmann-Vasserot normalisation. The full
\(\Woneinf\) object appears after the appropriate Drinfeld double or
centre, completion, and vacuum-evaluation representation have been
constructed.
\end{remark}

\section{Compact Targets}

\begin{hypothesis}[Compact quantum hCS package]\label{hyp:e2-compact-package}
A compact CY\(_3\) hCS quantisation with Hall comparison requires the
following data.
\begin{enumerate}
\item a compact Calabi-Yau threefold \((Y,\Omega_Y)\) with orientation
and determinant-line framing compatible with the BV pairing;
\item a quadratic gauge Lie algebra or dg Lie algebra
\((\g,\langle-,-\rangle_\g)\) and, when traces are used, a
representation \(\rho\) fixing \(I_4^\rho\) and \(C_2^\rho\);
\item an elliptic Costello gauge fixing, a heat-kernel regulator
\(P_{\varepsilon,L}\), and an RG-compatible counterterm scheme;
\item a nuclear or otherwise completed space of local functionals and
observables on which \(\Delta_L\), \(\{-,-\}_L\), and the RG flow are
defined;
\item factorisation or TCFT sewing data giving descent from local
polydiscs to \(Y\);
\item a trivialisation of \(\kanom(Y,\g,\rho)\) by vanishing in local BV
cohomology, an explicit local counterterm, or a closed-sector
Green-Schwarz term;
\item a completed hCS-to-Hall comparison preserving the positive half,
pairing, coproduct, and orientation data.
\end{enumerate}
\end{hypothesis}

\begin{theorem}[Conditional compact factorisation]\label{thm:e2-compact-conditional}
Assume Hypothesis~\ref{hyp:e2-compact-package}. If
\(\kanom(Y,\g,\rho)=0\) after the chosen open or open-closed
counterterms are included, then the effective interactions can be
chosen to satisfy the QME through one loop. Under the corresponding
higher-loop vanishing or counterterm hypotheses, the completed
observables form a quantum factorisation algebra on \(Y\).
\end{theorem}

\begin{proof}
Costello's obstruction theory solves the QME inductively in powers of
\(\hb\). At the first quantum step the obstruction is the degree-one
class of Proposition~\ref{prop:e2-qme-obstruction}, modified by the
specified local or closed-sector counterterms. Vanishing gives the
first quantum correction. The same induction applies at higher loop
order inside the completed local functional complex, and the sewing
data make the local solutions compatible on overlaps.
\end{proof}

\begin{remark}[The product $K3\times E$]\label{rem:e2-k3e-values}
For $Y=K3\times E$,
\[
 \kcat(Y)=\chi(\cO_{K3})\chi(\cO_E)=2\cdot0=0,
 \qquad
 \kch^{\mathrm{compact}}(Y)=
 \sum_q(-1)^qh^{0,q}(Y)=1-1+1-1=0,
\]
while the adjacent constructions give
\[
 \kch^{\mathrm{Heis}}(Y)=3,\qquad
 \kBKM(\mathfrak g_{\Delta_5})=\frac{c(0)}2=\frac{10}{2}=5,
 \qquad
 \kfib(Y)=24.
\]
These values are construction-separated: total-space Euler, compact
Hodge supertrace, Heisenberg-Mukai specialisation, Gritsenko's
\(\Delta_5\) Borcherds denominator, and the K3 Mukai lattice rank.
The actual Borcherds weight is \(5\), while the proposed total-space
plus fibre sum gives \(24\).  The table is governed by the individual
constructions.
\end{remark}
