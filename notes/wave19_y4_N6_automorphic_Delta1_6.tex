\section*{$N = 6$ automorphic Borcherds lift and the BKM denominator for $\DeltaOneSix$}
\label{wn:sec:wave19-y4-N6-automorphic}

\paragraph{Datum.} Fix $g_6 \in \Auts(K3)$ of order~$6$: Mukai 1988 Thm.~0.1
plus Kondo 1998 Thm.~1 classify the coinvariant lattice
$\Lambda_{g_6} \hookrightarrow N_6 = D_4^{\oplus 6}$
(Niemeier item \#19, Coxeter number $h = 6$; umbral group
$G^{(6D_4)} = 3.\mathrm{Sym}_6$, order $2160$; CDH 2014 Table~2).
Mukai rank $|\Lambda_{g_6}| = 8$; invariant rank $r_6 = 14$; Hashimoto
2012 gives the K3 fixed-locus stratification $K3^{g_6} = 2A_5 + 2D_4$,
orbifold Euler $\chi_{g_6}(K3) = 4$ (the twined K3 Euler character at
class $6A \subset M_{24}$; CHP 2014 Table~4 column $6A$).

\paragraph{Twined Jacobi seed.} \ClaimStatusTheorem\ The CDH 2014 twined
weak Jacobi form of weight $0$ and index $1$ at class $6A$ expands
(arXiv:1204.2779 Table~2; M4/DVV convention):
\[
  \phi^{(g_6)}_{0,1}(\tau, z)
  \;=\; \bigl(\zeta + 2 + \zeta^{-1}\bigr) + q(\cdots) + O(q^2),
  \qquad \zeta = e^{2\pi i z},
\]
with Borcherds-weight constant $c_6(0) = 2$ (the $q^0 r^0$ coefficient
in the paramodular convention of Gritsenko--Cl\'ery 2013). Character
$v_6 = \chi_{g_6}$ on $\GammaSix$ of order $6$, determined by the
$M_{24}$-class permutation representation on $H^*(K3, \Z)$
(Cl\'ery--Gritsenko 2013 \S 4.2).

\paragraph{Borcherds lift to paramodular weight~$1$.}
\ClaimStatusTheorem\ (Gritsenko--Cl\'ery 2013 Thm.~4.1; Gritsenko 1999
Thm.~1.2.) The singular theta lift of Borcherds 1998 Thm.~13.3 sends
$\phi^{(g_6)}_{0,1}$ to
\[
  \DeltaOneSix \;=\; \Bor\bigl(\phi^{(g_6)}_{0,1}\bigr)
  \;\in\; S_{1}\bigl(\GammaSix, v_6\bigr),
\]
of weight $k_6 = c_6(0)/2 = 1$ on the paramodular group
$\GammaSix \subset \Sp_4(\Q)$ of polarisation level~$6$. This is the
level-$6$ entry of the Gritsenko--Cl\'ery diagonal-divisor octet
(arXiv:0812.3962). The Borcherds product expansion:
\[
  \DeltaOneSix(Z)
  \;=\; e^{-2\pi i(\rho_6, Z)}
  \!\prod_{\substack{(n, \ell, m) \\ n, m \geq 0, \, 4nm > \ell^2 \\ n = m = 0 \Rightarrow \ell > 0}}
  \!\bigl(1 - e^{-2\pi i(n\tau + \ell z + m\sigma)}\bigr)^{c^{(g_6)}(nm, \ell)},
\]
exponents $c^{(g_6)}(nm, \ell)$ from $\phi^{(g_6)}_{0,1} = \sum
c^{(g_6)}(n, \ell) q^n \zeta^\ell$. Leading factor $(qp)^{k_6/6}$
matches Cl\'ery--Gritsenko 2013 Table~3, $N = 6$.

\paragraph{Weyl vector and hyperbolic Cartan.} \ClaimStatusTheorem\ The
BKM superalgebra $\gSix := \fg_{\DeltaOneSix}$ carries a rank-$3$
hyperbolic Cartan of signature $(2, 1)$ in the paramodular lattice
$\Lambda^{2,1}_{\mathrm{II}}(6) = U \oplus U(6) \oplus \langle -12
\rangle$ (Gritsenko--Nikulin 1998 \S 3; Lorgat 2020 \S 5, level-$6$).
The Weyl vector $\rho_6 = (\tfrac16, -\tfrac1{12}, \tfrac16)$ on the
hyperbolic basis $(f_2, f_3, f_{-2})$ is the $1/N$-scaled analogue of
the $N = 1$ Weyl vector $\rho = f_2 - \tfrac12 f_3 + f_{-2}$. Three
real simple roots $\{\delta_1^{(6)}, \delta_2^{(6)}, \delta_3^{(6)}\}$
of squared length $2/6$ generate $W^{(2)}(\Lambda^{2,1}_{\mathrm{II}}(6))$,
of index $6$ in the full orthogonal group on the positive cone.

\paragraph{Denominator identity.} \ClaimStatusTheorem\ The
Weyl--Kac--Borcherds denominator identity for $\gSix$ reads
\[
  \DeltaOneSix(Z)
  \;=\; e^{-2\pi i(\rho_6, Z)}
  \!\!\prod_{\alpha \in \Delta_{+}(\gSix)}\!\!
  \bigl(1 - e^{-2\pi i(\alpha, Z)}\bigr)^{\mathrm{mult}_{\gSix}(\alpha)}
  \;=\; \sum_{w \in W^{(2)}}\!\det(w)\,
  \Bigl(e^{-2\pi i(w\rho_6, Z)}
  - \!\sum_{a} m_6(a)\,e^{-2\pi i(w(\rho_6 + a), Z)}\Bigr),
\]
with even root multiplicities
$\mathrm{mult}_{\gSix}(\alpha) = c^{(g_6)}(nm, \ell)$ for
$\alpha = (n, \ell, m)$ in positive Weyl chamber, and imaginary simple
root data $m_6(a)$ read from the $(a, a) < 0$ Fourier coefficients of
$\phi^{(g_6)}_{0,1}$ (sign-twist by $v_6$ on odd-indexed roots).
Odd imaginary simple roots encode the super structure: $\gSix$ is a
Lie superalgebra, not a Lie algebra (Lorgat 2020 \S 3;
Gritsenko--Nikulin 1998 Lemma~2.3 square-map obstruction at $N \geq 2$).

\paragraph{Subscripted invariants.} \ClaimStatusTheorem\
$\kBKM(\DeltaOneSix) = c_6(0)/2 = 1$. On the CHL threefold
$X_6 = (K3 \times E)/\langle(g_6, t_6)\rangle$:
$\kcat(X_6) = \chi(\mathcal{O}_{K3})^{g_6}\,\chi(\mathcal{O}_E) = 0$
(K\"unneth, free diagonal quotient);
$\kch(X_6) = \sum_q (-1)^q h^{0, q}(X_6)^{(g_6, t_6)} = 0$
(Hodge supertrace at $d = 3$);
$\kfib(X_6) = \chi_{g_6}(K3) = 4$ (CHP 2014 Table~4, class $6A$;
distinguished from $\kfib(K3) = 24$ on the untwisted fibre).
The four-tuple $\{0, 0, 1, 4\}$ is the level-$6$ row of the
$\{2, 3, 5, 24\}$-spectrum programme from four distinct constructions
(Mukai lattice, $\DeltaOneSix$ via $\phi^{(g_6)}_{0,1}$, Borcherds
weight, twined fibre Euler).

\paragraph{Match to Y3 BPS characters.} \ClaimStatusConjectured\ (Lorgat
2020 Conj.~1, level-$6$ branch.) Let $Y^{+}(X_6) =
H^{*}_{\mathrm{eq}}(\mathcal{M}^{+}_{\mathrm{eff}}(X_6), \phi_{W})$ be
the positive half of the $(g_6, t_6)$-equivariant CoHA on the effective
BPS moduli (Davison--Meinhardt 2020; Bryan--Oberdieck 2018). Then
\[
  \mathrm{ch}\bigl(Y^{+}(X_6)\bigr)(\tau, \sigma, z)
  \;=\; -\,\frac{1}{\DeltaOneSix(Z)},
\]
termwise matched to the Borcherds product under $(n, \ell, m)
\leftrightarrow (\beta_{nm}^{(g_6)}, \ell)$. The KS wall-crossing factor
$(1 - x^{\alpha})^{-\Omega^{(g_6, t_6)}(\alpha)}$ matches the Borcherds
factor with exponent $c^{(g_6)}(nm, \ell)$ in the Oberdieck chamber;
Joyce--Song integrality plus Davison--Meinhardt integration pin
$\Omega^{(g_6, t_6)}(\alpha) = c^{(g_6)}(nm, \ell) = \dim \gSix_\alpha$
at primitive $K3$-classes. The bracket-level isomorphism
$\fg^{(6)}_{\mathrm{BPS}} \cong \gSix$ is conjectural (AP-CY34 open at
all $N$); graded dimensions are theorem at primitive scope via
Oberdieck 2018 Thm.~$1_{g_6}$ + Pandharipande--Thomas 2014 twined by
$g_6$.

\paragraph{Five attack--heal cycles.}

\emph{(1) Niemeier anchor.} \emph{Attack.} $4A_5 \oplus D_4$ (\#18) or
$6D_4$ (\#19)? \emph{Heal.} $6D_4$, per Wave-11 U1. Both lattices share
$h = 6$, but umbral moonshine selects \#19 via the $H^{(6)}$ shadow
matching the weight-$1/2$ unary theta series of the $6D_4$ glue code
and the umbral character $\chi_{6D_4}\colon 3.\mathrm{Sym}_6 \to
\C^\times$. Assignment to \#18 forces umbral order $144$, incompatible
with the $M_{24}$-twined weight of $\phi^{(g_6)}_{0,1}$.

\emph{(2) Character order.} \emph{Attack.} $\mathrm{ord}(v_6)$?
\emph{Heal.} Cycle shape $1^2 2^2 3^2 6^2$ of class $6A$ forces
$\mathrm{ord}(v_6) \mid 6$; Gritsenko--Cl\'ery 2013 \S 4.2 pins
$\mathrm{ord}(v_6) = 6$ via the $\pm I$ involution and Atkin--Lehner
$w_6$.

\emph{(3) Weight.} \emph{Attack.} $k_6 = 9/2$ per the A-series formula
$k_N = (N + 3)/2$? \emph{Heal.} That formula covers the
$A_{N-1}$-Niemeier substitution for $N \in \{2, 3, 4, 5\}$ under
$(N - 1) \mid 24$; at $N = 6$ it breaks (W19 ledger item~19), the
Niemeier substitute is $6D_4$, and $k_6 = c_6(0)/2 = 1$. The two
weights refer to different forms on different congruence subgroups.

\emph{(4) Paramodular level.} \emph{Attack.} $\GammaSix \subset
\Sp_4(\Q)$ or $\Gamma_0^{(2)}(6) \subset \Sp_4(\Z)$? \emph{Heal.}
$\GammaSix$ (Cl\'ery--Gritsenko 2013 \S 4): the seed has paramodular
(not Siegel) transformation under level-$6$ Humbert divisors.

\emph{(5) Convergence.} The Borcherds product converges absolutely on
$(\Im\tau)(\Im\sigma) - (\Im z)^2 \gg 0$ (Borcherds 1998 \S 14); the
twined exponents grow sub-exponentially via Hardy--Ramanujan applied
to the Mathieu-twined genus. Leading factor $(qp)^1$ with coefficient
$1$ matches Cl\'ery--Gritsenko 2013 Table~3 row $N = 6$.
