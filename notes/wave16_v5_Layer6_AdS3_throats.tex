\section*{Layer 6: eight AdS$_3$ throats of the Gritsenko--Cl\'ery
  family (Wave-16 V5/20)}
\label{sec:wave16-v5-layer6-ads3-throats}

\paragraph{Thesis.}
The sixth holographic layer of Wave-15 R5 --- the lattice-polarised
$\gL$ family over polarisation moduli $\cM_L$ --- is a fibration of
AdS$_3$ throats over the eight Gritsenko--Cl\'ery cusp points. Each
throat has central charge $c_N=24 k_N$ with paramodular weight
$k_N\in\{10,6,4,3,2\}$ for $N\in\{1,2,3,4,6\}$ (theorem) and
conjectural half-integer weights for $N\in\{5,7,8\}$. The dyon
partition function $1/\Phi_N^2$ is the Weyl--Kac--Borcherds
denominator of $\gL$, and Sen's quantum entropy
$S_{\BH}=2\pi\sqrt{\Delta/N}$ extends through the eight throats as
the CHL-reduced 1/(4N)-BPS count.

\subsection*{Attack-heal 1 --- Throat metric at weight $k_N$}

\ClaimStatusTheorem{} \emph{($N=1$ throat.)} Type IIB on $K3\times
S^1\times\widetilde S^1$ with Strominger--Vafa charges $(Q_1,Q_5,n,J)$
has near-horizon geometry $\AdS_3\times S^3\times K3$ with radius
$\ell_3^2=4G_3 c_{2d}/3$; at $N=1$,
\[
  c_{2d}\;=\;6 Q_1 Q_5\;=\;24 k_1\bigl|_{k_1=10,\,Q_1Q_5=40}
\]
(Maldacena 1998 \texttt{hep-th/9711200}; Strominger--Vafa 1996
\texttt{hep-th/9601029}). The normalisation $c_N=24 k_N$ reads
$k_N$ as the \emph{chiral half-rank} of the $\gL$-denominator
Cartan subalgebra at the $N$-th throat.

\ClaimStatusConjectured{} \emph{(Eight-throat extension.)} For the
eight Gritsenko--Cl\'ery forms $\{\Phi_N\}_{N\in\{1,\dots,8\}}$ with
paramodular weights
\[
  k_N\;=\;\bigl(10,\,6,\,4,\,3,\,\tfrac{5}{2},\,2,\,\tfrac{12}{7},
    \,\tfrac{5}{3}\bigr)_{N\in\{1,2,3,4,5,6,7,8\}},
\]
the throat metric $\AdS_3\times S^3/\Z_N\times K3^{g_N}$ carries
central charge $c_N=24 k_N$; the integral $N\in\{1,2,3,4,6\}$
branches are CHL orbifolds with theorem-status Sen entropy, the
half-integral $N\in\{5,7,8\}$ branches are conjectural fractional-orbifold
completions (Lorgat 2020 Conjecture~1; Gritsenko--Nikulin 1998
\emph{Int.\ J.\ Math.} 9; Cl\'ery--Gritsenko 2011
\texttt{arXiv:0812.3962}).

\subsection*{Attack-heal 2 --- Dyon count $d_N(Q)=\oint e^{2\pi i
    Q\cdot\Omega}/\Phi_N^2$}

\ClaimStatusTheorem{} (DVV 1997 \texttt{hep-th/9607026};
Jatkar--Sen 2006 \texttt{hep-th/0510147}.) The 1/4-BPS degeneracy
at CHL level $N$ is
\[
  d_N(Q,P)\;=\;(-1)^{Q\cdot P+1}\oint_{\cC_N}
  \frac{e^{-i\pi(\rho Q^{2}+\sigma P^{2}+2v\,Q\!\cdot\!P)}}
    {\Phi_{k_N}(\rho,\sigma,v)^{2}}\,
  d\rho\,d\sigma\,dv,
\]
with contour $\cC_N$ threading the Humbert divisor
$2v\in\Z,\ \rho,\sigma\in\mathbb{H}_+$ and $\Phi_{k_N}$ the Siegel
modular form of weight $k_N$ on the paramodular group
$K(N)\subset\mathrm{Sp}_4(\Q)$. The square $\Phi_N^2$ arises because
the Borcherds product of the weight-$k_N$ Jacobi form
$\phi^{(N)}_{0,1}$ is $\Phi_N$, and the physical
denominator on the $\AdS_3$ symmetric orbifold
$\mathrm{Sym}^\infty(K3^{g_N})$ is the square
(DVV 1997 Eq.~(1.7); Shih--Strominger--Yin 2005
\texttt{hep-th/0506151}).

\ClaimStatusTheorem{} \emph{(Borcherds reading.)} The logarithm
$-\log\Phi_{k_N}^2$ is twice the $\gL$-denominator
\[
  -\tfrac12\log\Phi_{k_N}^2\;=\;\sum_{\alpha\in\Delta^+_L}
    \mathrm{mult}(\alpha)\,\log(1-e^{\alpha}),
\]
identifying $d_N(Q)$ with the character of $\gL^+$ evaluated at the
dyon charge vector $Q$ (Borcherds 1995 \texttt{alg-geom/9508032}
Thm.~10.1; Lorgat 2020 Lemma~1 for $N=1$).

\subsection*{Attack-heal 3 --- $N=1$ Sen entropy
  $2\pi\sqrt{\Delta/N}$}

\ClaimStatusTheorem{} (Sen 2008 \texttt{arXiv:0708.1270}
Eq.~(6.9); Shih--Strominger--Yin 2005.) At $N=1$ with
$\Delta(Q,P)=Q^2 P^2 - (Q\cdot P)^2$ the T-duality invariant,
the saddle of $\oint 1/\Phi_{10}^2$ at $\rho,\sigma,v$ extremising
the Siegel contour gives
\[
  \log d_1(Q,P)\;=\;2\pi\sqrt{\Delta(Q,P)}\;+\;
    \bigl(k_1+2\bigr)\log\Delta(Q,P)\;+\;\cdots,
\]
with the leading term the Bekenstein--Hawking entropy
$S_{\BH}=2\pi\sqrt{\Delta}$ of the four-dimensional BMPV black hole,
the logarithmic correction fixed by the Gauss--Bonnet density and
matching $k_1=10$ weight, and $\cdots$ the Sen quantum-entropy
sub-leading AdS$_2$ boundary corrections.

\subsection*{Attack-heal 4 --- CHL 1/(4N)-BPS reduction}

\ClaimStatusTheorem{} (Jatkar--Sen 2006; Chattopadhyaya--David 2019
\texttt{arXiv:1711.00370}.) The CHL $\Z_N$-orbifold of Type II on
$K3\times T^2$ by a Nikulin involution $g_N$ of order
$N\in\{2,3,4,6\}$ reduces the 1/4-BPS spectrum to 1/(4N)-BPS:
four supercharges out of sixteen are broken, and the projected
dyons carry level-$N$ Wilson lines on the CHL-cycle. The counting
function becomes $1/\Phi_{k_N}^2$ with
\[
  k_N\;=\;\frac{24}{N+1}-2\;\in\;\{10,6,4,3,2\}\qquad
  (N\in\{1,2,3,4,6\}),
\]
where the Nikulin identity $k_N(N+1)+2N=24$ is the CHL analogue of
the K3 Euler characteristic (Chaudhuri--Hockney--Lykken 1995;
Cheng 2010 \texttt{arXiv:1005.5415}). The 1/(4N)-BPS Bekenstein--Hawking
entropy is
\[
  S_{\BH}^{(N)}\;=\;2\pi\sqrt{\Delta(Q,P)/N}
\]
by the rescaling $\Delta\to\Delta/N$ of the CHL T-duality invariant
(Sen 2005 \texttt{hep-th/0504005}; David--Jatkar--Sen 2006
\texttt{hep-th/0609074}), exactly matching the weight shift
$k_1=10\to k_N=10/N+\cdots$ at leading order.

\ClaimStatusConjectured{} \emph{(Half-integer completion.)} For
$N\in\{5,7,8\}$ the Conjecture~1 weights $k_N\in\{5/2,12/7,5/3\}$
correspond to non-Nikulin orbifolds --- in the $N=5$ case to a
$\Z_5$ symplectic automorphism of an FK3-fibration (Garbagnati 2012
\texttt{arXiv:1106.1337}). The entropy would read
$S_{\BH}^{(N)}=2\pi\sqrt{\Delta/N}$ at half-integral weight, with
the square-root structure preserved by the Borcherds-lift identity
$\kBKM(\Phi_N)=c_N(0)/2$ evaluated on the rational-weight
Jacobi form.

\subsection*{Attack-heal 5 --- Bulk-side avatar of $\cM_L$}

\ClaimStatusConjectured{} \emph{(Holographic moduli duality.)} The
sixth-layer $\gL$-family over $\cM_L$ is the bulk-side avatar of the
boundary moduli space of the symmetric-orbifold chiral algebra
$\mathrm{Sym}^\infty(K3^{g_N})$:
\[
  \cM_L\;=\;\Gamma_L\backslash\Omega_{T}\;\simeq\;
    \cM_{\AdS_3}^{\mathrm{throat}}
\]
where $\Omega_T=\{z\in T\otimes\C:\langle z,z\rangle=0,\,
\langle z,\bar z\rangle>0\}/\C^{*}$ is the type-IV period domain of
the transcendental lattice $T=L^\perp\subset\Lambda_{K3}$ and
$\cM_{\AdS_3}^{\mathrm{throat}}$ is the moduli space of
$\AdS_3$-throat geometries with fixed $(k_N,g_N)$ (Nakajima 2001;
Harvey--Moore 1996 \texttt{hep-th/9510182}). Three independent
channels support the identification:
\begin{enumerate}
\item \emph{Period map.} The complex structure on
$T\otimes\C$ is dual to the attractor flow of the $\AdS_3$ throat
moduli: Dolgachev mirror $(L,\check L)$ pairs match T-duality
orbits on the throat (Dolgachev 1996 \emph{J.\ Math.\ Sci.} 81;
Aspinwall--Morrison 1997 \texttt{hep-th/9404151}).
\item \emph{BKM/root system.} The real simple roots of $\gL$
index the 24 punctures of the class-$\cS$ sphere $\Sigma_{0,24}$
whose boundary VOA is $\chi[T[A_1,\Sigma_{0,24}]]$ at
$c_{2d}=-214$ (Wave-12 D4 Cycle 3); Nikulin's classification of
$L$-polarisations is equivalent to Chacaltana--Distler's
classification of admissible puncture decorations of
$\Sigma_{0,24/N}$ at CHL level $N$ (Chacaltana--Distler 2010
\texttt{arXiv:1008.5203}).
\item \emph{Automorphic lift.} The Borcherds lift
$\phi^{(N)}_{0,1}\mapsto\Phi_N$ is the bulk $\AdS_3$ partition
function on the $N$-th throat (DMVV 1997; Lorgat 2020
\S3). The lift extends to Conjecture~1's eight Gritsenko--Cl\'ery
forms, and the eight throats are the DT zeta roots
$d_N\in\{24,12,8,6,5,4,3,2\}$ of the $K3\times E$
Donaldson--Thomas zeta function (Oberdieck--Pixton 2017
\texttt{arXiv:1706.10100}).
\end{enumerate}

\paragraph{Unification.}
The six holographic layers of Wave-15 R5 --- class-$\cS$ 4D,
BLLPR chiral 2D, $\hCS$ bulk $E_3$, Miki $S_3$ triality,
$\mathrm{HH}^2_{E_2}$ rigidity, $\gL$-family --- are six projections
of one CY$_3$ datum $(K3\times E,\Omega,\gL)$. The sixth layer
resolves into \emph{eight AdS$_3$ throats} indexed by the
Gritsenko--Cl\'ery family, with central charges $c_N=24 k_N$,
dyon counts $d_N(Q)=\oint 1/\Phi_N^2$, Bekenstein--Hawking
entropies $S_{\BH}^{(N)}=2\pi\sqrt{\Delta/N}$ (theorem at
$N\in\{1,2,3,4,6\}$, conjectural at $N\in\{5,7,8\}$), and bulk
moduli $\cM_L$ dual to the boundary attractor flow. Four
invariants $\kch=0$, $\kcat=0$, $\kBKM(\Phi_N)=c_N(0)/2$, $\kfib=24$
of the four distinct constructions sit at the four
corners of each throat, and the eight-throat fibration is the
Lorgat 2020 Conjecture~1 in holographic form.
