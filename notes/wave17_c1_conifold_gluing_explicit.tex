\section*{Wave-17 C1/20: the resolved conifold as two $\C^{3}$ charts, and
          the gluing of their positive halves}
\label{wn:sec:conifold-gluing-explicit}

The resolved conifold $\widetilde{Y} = \cO(-1)\oplus\cO(-1)\to\bP^{1}$ is
the smallest non-trivial toric CY$_{3}$: one compact curve, two affine
charts, one overlap.  What is to be shown is that $\Yplus(\widetilde{Y})$
is not two copies of $\Yplus(\C^{3}) = \Yplus(\ghat)$ glued naively ---
the overlap imposes a cocycle that converts the pair into the
two-vertex Klebanov--Witten CoHA of Szendr\H{o}i.

\subsection*{Cycle 1 --- the toric fan of $\widetilde{Y}$}

\ClaimStatusDerivation\ The defining lattice is $N = \Z^{3}$ with rays
\[
  v_{1}=(1,0,0),\quad v_{2}=(0,1,0),\quad v_{3}=(1,0,1),\quad v_{4}=(0,1,1).
\]
These are coplanar (the CY condition: all rays lie on $z=0$ or $z=1$ in
dual coordinates, equivalently $\langle v_{i}, (0,0,1)\rangle \in \{0,1\}$,
which places them on an affine hyperplane and gives $\sum (-1)^{\cdot}$
divisor = canonical = trivial).  The fan $\Sigma_{\widetilde{Y}}$ has
two maximal cones
\[
  \sigma_{+} = \mathrm{cone}(v_{1},v_{2},v_{3}), \qquad
  \sigma_{-} = \mathrm{cone}(v_{1},v_{2},v_{4}),
\]
sharing the wall $\tau = \mathrm{cone}(v_{1},v_{2})$.  Each
$\sigma_{\pm}$ is generated by a $\Z$-basis of $N$, hence each affine
toric variety $U_{\pm} = \mathrm{Spec}\,\C[\sigma_{\pm}^{\vee}\cap M]$
is $\C^{3}$.  The wall $\tau$ is two-dimensional, so
$U_{+}\cap U_{-} = \mathrm{Spec}\,\C[\tau^{\vee}\cap M] = \C^{*}\times\C^{2}$
--- a single $\C^{*}$ direction (the $\bP^{1}$ coordinate away from its
two torus fixed points) and two fibre coordinates.

The distinguished flop $\widetilde{Y}^{\vee}$ is obtained by replacing
$\sigma_{\pm}$ with the other triangulation
$\sigma'_{\pm}=\mathrm{cone}(v_{3},v_{4},v_{1})\cup\mathrm{cone}(v_{3},v_{4},v_{2})$;
Bridgeland's theorem
(arXiv:\texttt{math/0502050}) gives $D^{b}(\widetilde{Y})\cong
D^{b}(\widetilde{Y}^{\vee})$ as a derived equivalence.  This is the
flop that the Klebanov--Witten quiver will absorb into its Seiberg
duality.

\subsection*{Cycle 2 --- classical gluing at the ring level}

\ClaimStatusDerivation\ Pick homogeneous $\bP^{1}$-coordinates $[t_{0}:t_{1}]$
and fibre coordinates $u,v$ with weights $(-1,-1)$.  On
$U_{+}=\{t_{0}\neq 0\}$ set $t = t_{1}/t_{0}$ and keep $u,v$ as
$\cO(-1)^{\oplus 2}$ trivialisations; on
$U_{-}=\{t_{1}\neq 0\}$ set $\tilde t = t_{0}/t_{1} = t^{-1}$ and
$\tilde u = t\,u$, $\tilde v = t\,v$ (the $\cO(-1)$ twist absorbed into
the fibre coordinate).  Both charts are $\C^{3}$; the overlap ring is
\[
  \C[t,t^{-1},u,v] \;=\; \C[\tilde t,\tilde t^{-1}, \tilde u,\tilde v]
\]
with cocycle $\tilde t = t^{-1}$, $\tilde u = t u$, $\tilde v = t v$.
The global holomorphic $3$-form $\Omega_{\widetilde{Y}}$ restricts as
\[
  \Omega_{\widetilde{Y}}|_{U_{+}} \;=\; dt\wedge du\wedge dv, \qquad
  \Omega_{\widetilde{Y}}|_{U_{-}} \;=\; -\,t^{-1}\,d\tilde t\wedge d\tilde u\wedge d\tilde v
\]
(the Jacobian $|\partial(\tilde t,\tilde u,\tilde v)/\partial(t,u,v)| =
-t^{-1}\cdot t\cdot t = -t$ and the differentials cancel all but a sign
--- the minus is the $\sigma_{+}\to\sigma_{-}$ orientation flip, as
expected of a flop).  Non-vanishing of $\Omega_{\widetilde Y}$ on the
overlap is the Calabi--Yau witness.

\subsection*{Cycle 3 --- chart-level CoHA}

\ClaimStatusTheorem\ For each chart $U_{\pm}\simeq\C^{3}$, the positive
half of the associated cohomological Hall algebra is
\[
  \CoHA(U_{\pm}) \;=\; \Yplus(\ghat)
  \;=\; \bigoplus_{n\ge 0} H^{*}_{T}\bigl(\Hilb^{n}(\C^{2})\bigr),
\]
with the action of $\Hilb^{n}$-cohomology on its tripled tangent
(Schiffmann--Vasserot 2013, arXiv:\texttt{1202.2756};
Maulik--Okounkov 2012, arXiv:\texttt{1211.1287}).  The charge lattice is
$\Gamma(U_{\pm}) = \Z$, with $n\in\N$ labelling the rank.  The
supertrace is $\kch(\Yplus(\ghat)) = 1$; at the categorical level
$\kcat(\C^{3}) = \chi(\cO_{\C^{3}}) = 1$ (affine, no higher $\cO$-cohomology).

\subsection*{Cycle 4 --- overlap and the Szendr\H{o}i cocycle}

\ClaimStatusDerivation\ On $U_{+}\cap U_{-}=\C^{*}\times\C^{2}$ the
positive half is
\[
  \CoHA(\C^{*}\times\C^{2}) \;=\; H^{*}_{T}\bigl(\bigsqcup_{n\ge 0}
  \mathrm{Quot}^{n}(\C^{*}\times\C^{2})\bigr)
\]
and carries \emph{two} such structures --- one inherited from $U_{+}$,
one from $U_{-}$ --- related by the transition that the cocycle
$(t\to t^{-1},\,u\to tu,\,v\to tv)$ induces on torus-equivariant Quot
schemes.  The cocycle acts on the charge lattice
$\Gamma(\C^{*}\times\C^{2})=\Z^{2}$ (one $\Z$ per rank, one $\Z$ per
winding around $\C^{*}$) as
\[
  \begin{pmatrix} n_{+} \\ w \end{pmatrix}
  \;\longmapsto\;
  \begin{pmatrix} n_{-} \\ -w \end{pmatrix}, \qquad n_{-} = n_{+},
\]
because the rank is unchanged under the trivialisation switch but the
winding reverses.  The \emph{two-vertex quiver} $Q_{\mathrm{KW}}$ of
Szendr\H{o}i (\texttt{arXiv:0705.3419}) absorbs this:
$Q_{0}=\{0,1\}$ (one vertex per chart), arrows
$a_{1},a_{2}\colon 0\to 1$ and $b_{1},b_{2}\colon 1\to 0$ (two per
fibre direction), potential
\[
  W_{\mathrm{KW}} \;=\; \tr(a_{1}b_{1}a_{2}b_{2}-a_{1}b_{2}a_{2}b_{1}).
\]
The dimension vector $(n_{0},n_{1})\in\N^{2}$ doubles the chart-level
$\Z$-lattice; the shift from $(n,0)\leftrightarrow(0,n)$ under Seiberg
duality on $Q_{\mathrm{KW}}$ \emph{is} the transition cocycle above
(Morrison--Mozgovoy--Nagao--Szendr\H{o}i, \texttt{arXiv:1107.5017},
Proposition 2.4).

\subsection*{Cycle 5 --- global positive half and verification}

\ClaimStatusTheorem\ (Davison--Meinhardt, \texttt{arXiv:1601.02479},
specialised to $(Q_{\mathrm{KW}},W_{\mathrm{KW}})$; MMNS
\texttt{1107.5017}, Theorem~1.)  The global positive half is
\[
  \Yplus(\widetilde{Y}) \;=\; \CoHA(Q_{\mathrm{KW}},W_{\mathrm{KW}})
  \;=\; U\bigl(\mathfrak{n}_{+}^{\widehat{\mathfrak{sl}}_{2}}\bigr)
         \otimes \Lambda_{\mathrm{Cartan}},
\]
matching the affine-$A_{1}^{(1)}$ root system:
$\Delta_{+}^{\mathrm{re}} = \{(n+1,n),(n,n+1):n\ge 0\}$,
$\Delta_{+}^{\mathrm{im}}=\{(n,n):n\ge 1\}$.  The real roots come
from individual chart generators pushed to the global object; the
imaginary roots $(n,n)$ come from the compact $\bP^{1}\subset\widetilde Y$
and carry dimension $2$ per level (the $\bP^{1}$-cohomology
$H^{*}(\bP^{1}) = \C\langle[\mathrm{pt}],[\bP^{1}]\rangle$).
The $\DT$-invariants are $(1,1,\dots; -2,-2,\dots)$.

\emph{Overlap verification.}  Restrict the global
$\Yplus(\widetilde{Y})$ to $U_{+}$: the imaginary roots disappear
(no compact curve in $U_{+}\simeq\C^{3}$), the real roots collapse
to a single tower $e_{n}$ indexed by $n\ge 0$, and one recovers
$\Yplus(\ghat)$ --- the chart-level positive half.  Restrict to
$U_{-}$: same result, with $e_{n}\leftrightarrow f_{n}$ the Seiberg
duality exchange.  The two restrictions agree on $U_{+}\cap U_{-}$
modulo the cocycle of Cycle~4.  The supertrace invariants:
\[
  \kch(\Yplus(\widetilde{Y})) \;=\; 1,
\]
computed three ways: (i) additivity under toric glue,
$\kch(U_{+})+\kch(U_{-}) - \kch(U_{+}\cap U_{-}) = 1+1-1 = 1$; (ii)
$\chi(\cO_{\widetilde{Y}}) = 1$ by Riemann--Roch on the non-compact
CY$_{3}$ (the compactly-supported Euler characteristic of the
structure sheaf is $1$); (iii) direct: the vacuum character of
$\Yplus(\widetilde{Y})$ at $\mathbf{y}=(1,1)$ reduces to
$\prod_{n\ge 1}(1-q^{n})^{-1}$ with $\kch$ read as the central-charge
normalisation (MacMahon-like formula restricted to the trivial
charge).  All three agree.

The gluing is complete: \emph{two} $\Yplus(\ghat)$'s are not glued
naively; they are assembled by doubling the vertex set (reflecting the
two affine charts), introducing four arrows (two per fibre direction),
and imposing the Klebanov--Witten potential $W_{\mathrm{KW}}$ whose
Jacobi algebra is the NCCR of $\widetilde{Y}$ (Van den Bergh
\texttt{arXiv:math/0207170}).  The overlap cocycle is realised as
Seiberg duality on $(Q_{\mathrm{KW}},W_{\mathrm{KW}})$; the flop
$\widetilde{Y}\leftrightarrow\widetilde{Y}^{\vee}$ is the associated
derived equivalence (Bridgeland \texttt{math/0502050}).  This closes
the conifold instance of the general toric CY$_{3}$ gluing programme
and is consistent with the $\kch$-spectrum entry for the resolved
conifold ($\kch=1$) tabulated in
\texttt{chapters/examples/cy\_d\_kappa\_stratification.tex}.\qed
