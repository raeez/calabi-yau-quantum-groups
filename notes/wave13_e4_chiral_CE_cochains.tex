\section{Dolbeault--chiral CE complex}
Let $X$ be a complex threefold and let $\cL$ be a holomorphic dg Lie
algebroid with Dolbeault resolution. The complex
\[
  \CE^*_{\dbar,\chir}(\cL,\cM)
\]
computes chiral Chevalley--Eilenberg cochains of $\cL$ with coefficients
in a chiral module $\cM$, with the Dolbeault differential included in
the total differential. For
\[
  \cE=\Omega^{0,\bullet}(\C^3,\g)
\]
with pointwise Lie bracket, this complex is the classical observable
factorisation algebra of affine holomorphic Chern--Simons theory.

\section{Classical Chevalley--Eilenberg cochains}
For a Lie algebra $\g$ over $k$ with module $M$, classical CE cochains are
\[
 \CE^n(\g,M)=\Hom_k(\Lambda^n\g,M),
\]
with differential
\[
\begin{aligned}
 (d_{\CE}f)(x_0,\ldots,x_n)
 &=
 \sum_i(-1)^i\,x_i\cdot
 f(x_0,\ldots,\widehat{x_i},\ldots,x_n) \\
 &\quad+
 \sum_{i<j}(-1)^{i+j}\,
 f([x_i,x_j],x_0,\ldots,\widehat{x_i},\ldots,\widehat{x_j},\ldots,x_n),
\end{aligned}
\]
with the standard Koszul signs in the graded case. The chiral
replacement of $\Lambda^n\g$ is the Beilinson--Drinfeld chiral
symmetric power $\Sym^n_{\chir}\cL$ on the Ran space
(\cite[\S3.4]{BD04}).

\section{Chiral CE cochains}
On a smooth curve, a chiral Lie algebra is a right $\cD_X$-module
$\cL$ with bracket
\[
  \mu:j_*j^*(\cL\boxtimes\cL)\longrightarrow\Delta_*\cL
\]
satisfying skew-symmetry and the chiral Jacobi identity. Higher
dimensional complex manifolds are treated here through holomorphic
factorisation algebras on the Ran space, with the curve case as the
Beilinson--Drinfeld model.

\begin{definition}[Chiral CE cochains]
\label{def:chiral-ce-cochains}
For a chiral $\cL$-module $\cM$, set
\[
 \CE^n_{\chir}(\cL,\cM)
 :=
 \Hom_{\cD_{X^n}}\!\bigl(\Sym^n_{\chir}\cL,\Delta^{(n)}_*\cM\bigr),
\]
where $\Delta^{(n)}:X\hookrightarrow X^n$ is the diagonal. The
differential is
\[
 (d_{\chir} f)(l_0,\ldots,l_n)
 =
 \sum_{i<j}(-1)^{i+j}\,
 f(\mu_{ij}(l_i,l_j),\ldots,\widehat{l_i},\ldots,\widehat{l_j},\ldots)
 +
 \sum_i(-1)^{i+1}\rho_i\cdot f(\ldots,\widehat{l_i},\ldots),
\]
with $\mu_{ij}$ the bracket on positions $(i,j)$ and $\rho_i$ the
chiral action on $\cM$. The chiral Jacobi identity on
$\Ran^{\leq3}$ gives $d_{\chir}^2=0$.
\end{definition}

The $\cD$-module Hom records chiral operations because
$\Sym^n_{\chir}\cL$ is the universal input for $n$-ary chiral
operations in Beilinson--Drinfeld's construction. Dualising the chiral
bar construction gives the familiar comparison with
$B(U^{\chir}(\cL))$ on the Koszul locus.

\section{Dolbeault refinement}
For a complex threefold $X$, use $\cD_X$-modules in Dolbeault degrees.
The current example is
\[
  \cL=\Omega^{0,\bullet}(X,\g),
\]
with $\dbar$ and pointwise bracket on $\g$-coefficients. The chiral
bracket is the diagonal distribution
\[
  \mu(\alpha\otimes\beta)=[\alpha,\beta]_\g\cdot\delta_\Delta .
\]

\begin{definition}[Dolbeault--chiral CE]
\label{def:dolbeault-chiral-ce}
\[
 \CE^*_{\dbar,\chir}(\cL,\cM)
 :=
 \left(
   \bigoplus_{n\geq0}
   \Omega^{0,\bullet}\!\bigl(
     X^n,\Hom(\Sym^n_{\chir}\cL,\cM)
   \bigr)^{\Sigma_n},
   \dbar+d_{\chir}
 \right).
\]
The total differential is $D=\dbar+d_{\chir}$.
\end{definition}

\begin{lemma}[Differential]
\label{lem:dolbeault-chiral-differential}
For $\cL=\Omega^{0,\bullet}(X,\g)$ with pointwise coefficient bracket,
$D^2=0$.
\end{lemma}

\begin{proof}
$\dbar^2=0$ and $d_{\chir}^2=0$. The commutator
$[\dbar,d_{\chir}]$ vanishes because $\dbar$ acts on Dolbeault form
degree while the bracket acts on the $\g$-coefficient and the diagonal
distribution. Equivalently,
\[
  \dbar[\alpha,\beta]_\g
  =
  [\dbar\alpha,\beta]_\g+(-1)^{|\alpha|}[\alpha,\dbar\beta]_\g .
\]
\end{proof}

\section{Holomorphic Chern--Simons observables}
For hCS on $\C^3$ with structure Lie algebra $\g$, the Costello BV
fields are
\[
  \cA\in\Omega^{0,\bullet}(\C^3,\g)[1]
\]
and the classical action is
\[
  S=
  \int_{\C^3}\Omega\wedge
  \left\langle \cA,\dbar\cA+\tfrac13[\cA,\cA]\right\rangle .
\]

\begin{proposition}[hCS observables as Dolbeault--chiral CE]
\label{prop:hcs-obs-chiral-ce}
Let $\cE:=\Omega^{0,\bullet}(\C^3,\g)$ with $\dbar$ and pointwise
bracket. Then
\[
 \Obs_{\mathrm{hCS}}^{\cl}(\C^3)
 \simeq
 \CE^*_{\dbar,\chir}\!\bigl(\cE,\cO_{\C^3}\bigr)
\]
as completed classical factorisation algebras on $\C^3$.
\end{proposition}

\begin{proof}
The completed algebra of classical observables is
\[
  \Sym(\cE_c^\vee[-1])
\]
with differential $\dbar^\vee+\{S,-\}$. The quadratic part of $S$
contributes $\dbar^\vee$. The cubic hCS vertex contributes the CE
differential induced by the pointwise Lie bracket. Locality of $S$
identifies the multiplication over disjoint opens with the factorisation
product, so the CE model and the BV observable model define the same
factorisation algebra.
\end{proof}

\begin{remark}[Quantum deformation]
Quantum observables are the Costello--Gwilliam perturbative
factorisation algebra
\[
  \bigl(\cO(\cE(U))[[\hbar]],
  Q+\{I[L]_U,-\}_{\BV}+\hbar\Delta^{P_L}_{\BV}\bigr)
\]
after gauge fixing, renormalisation, counterterms, anomaly cancellation,
and analytic completion. They are a filtered deformation of the
classical CE complex, not the result of adding a single trace term to
$d_{\chir}$.
\end{remark}

\section{Native $E_3$ structure and specialised $E_1$ output}
The affine hCS observable factorisation algebra on $\C^3$ carries the
native holomorphic $E_3$ structure of a complex threefold. This is the
Stage-$1$ object $\PhiFA_3$ in the CY-to-chiral construction. After a
witnessed specialisation datum $(\Sigma_2,C)$, the curve output is
\[
  A_X^{(\Sigma_2,C)}
  =
  \SpCh_{\Sigma_2,C}\bigl(\PhiFA_3(X)\bigr),
\]
and it is $E_1$-chiral for $d\geq3$. The $E_2$ object is the Drinfeld
centre $\cZ(\Rep^{E_1}(A_X^{(\Sigma_2,C)}))$, or the corresponding
completed double when a Hall pairing and completion have been fixed.

\section{Koszul duality}
Francis \cite[\S3]{Francis13} defines $E_n$-factorisation homology
$\int_M A$ for an $E_n$-algebra $A$ and framed $n$-manifold $M$.

\begin{proposition}[Completed $E_3$ Koszul duality]
\label{prop:koszul-factorisation}
Let $\cE=\Omega^{0,\bullet}(\C^3,\g)$ and let
$U^{\chir}(\cE)$ be the augmented completed $E_3$ chiral envelope.
Under the usual augmentation, connectivity, and completion hypotheses,
\[
  U^{\chir}(\cE)
  \simeq
  \Omega_{E_3}\!\bigl(\CE^*_{\dbar,\chir}(\cE,\C)\bigr).
\]
After compact-support and continuous-dual choices,
\[
  \int_{\C^3}U^{\chir}(\cE)
  \simeq
  \CE^*_{\dbar,\chir}(\cE,\C)^\vee .
\]
\end{proposition}

\begin{proof}
This is Francis' $E_n$ bar--cobar duality for enveloping algebras and
CE cochains, read in the completed holomorphic factorisation category.
The coefficient $\C$ is the augmentation costalk. The
$\cO_{\C^3}$-coefficient observable complex is the module-coefficient
variant and should not be identified with the augmented Koszul dual.
\end{proof}

\section{Hodge supertrace}
For compact CY$_3$ input where the TCFT correction, chain-level
$S^3$-framing, Costello--Li anomaly cancellation, OPE completion, and
witnessed Stage-$2$ specialisation are supplied, the abelian
Hodge-supertrace formula is
\[
  \kappa_{\mathrm{ch}}(A_X)=\sum_q(-1)^q h^{0,q}(X).
\]
For $X=\C^3$ this compact formula is replaced by the affine local
generator computation. Non-abelian CE cohomology contributes only after a
finite model and completion have been fixed.
