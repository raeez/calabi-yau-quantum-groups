\section{Direct $q^0$-Fourier expansion of $Z^{(2A)}_{K3}$ and the $c_2(0) = 4$ resolution}
\label{wn:sec:wave10-q1-direct-phi2A}

\paragraph{Two conventions, one value.}
The conflict $(10,8,6,4,2)$ versus $(10,4,2,2,2)$ is a normalisation artefact at even~$N$ between the
\emph{twined elliptic genus} $Z^{(g)}_{K3}$ (Eguchi--Ooguri--Taormina 2011, arXiv:1004.0956, Table~1)
and the \emph{Borcherds-lift input Jacobi form} $\phi^{(g)}_{0,1} = Z^{(g)}_{K3}/2$ (Gritsenko
1999, arXiv:math/9906191, \S2; Lorgat 2020, eq.\ above~Thm.~1). The Borcherds weight theorem of
Borcherds 1998 (\emph{Invent.\ Math.}~132, Thm.~13.3) reads weight in terms of $c_N(0)$ extracted
from the $\zeta^0 q^0$-slot of $\phi^{(g_N)}_{0,1}$, not of $Z^{(g_N)}_{K3}$.

\begin{theorem}[Direct $q^0$-Fourier expansion of $Z^{(2A)}_{K3}$ and $c_2(0)$]
\label{wn:thm:wave10-q1-direct-phi2A-c2}
\ClaimStatusTheorem\ Let $g_2\in M_{24}$ be the Mathieu class $2A$ realised as a symplectic
involution of K3 in Mukai's classification (Mukai 1988, \emph{Invent.\ Math.}~94, Prop.~1.2,
Table~\S2). Let
\[
  Z^{(g_2)}_{K3}(\tau,z)
  = \mathrm{Tr}_{H^*_{\mathrm{RR}}(K3)}\bigl(g_2\cdot y^{J_0}\bar y^{\bar J_0}
        q^{L_0-c/24}\bar q^{\bar L_0-\bar c/24}\bigr)
\]
be the twined $N{=}4$ elliptic genus of K3 at class~$2A$ (Gaberdiel--Hohenegger--Volpato 2012,
arXiv:1211.7074, \S2; EOT 2011 Table~1, row~$2A$), a weak Jacobi form of weight~$0$, index~$1$ on
$\Gamma_0(2)$. Then at the $q^0$-slot,
\[
  Z^{(2A)}_{K3}(\tau,z)\Big|_{q^0}
  \;=\; \zeta^{-1} \;+\; 8 \;+\; \zeta \,,
  \qquad
  \phi^{(2A)}_{0,1}(\tau,z)\Big|_{q^0}
  \;=\; \tfrac12 Z^{(2A)}_{K3}(\tau,z)\Big|_{q^0}
  \;=\; \tfrac12\zeta^{-1} \;+\; 2 \;+\; \tfrac12 \zeta\,.
\]
The Borcherds-lift constant is $c_2(0) := [\zeta^0 q^0]\, Z^{(2A)}_{K3}= 8$ in the \emph{EOT
convention} and $c_2(0):= [\zeta^0 q^0]\,(2\phi^{(2A)}_{0,1}) = 4$ in the
\emph{Gritsenko--Nikulin convention} used by the paramodular weight formula
$\mathrm{wt}(\Phi^{(N)}) = c_N(0)/2$ (Gritsenko 1999 Thm.~3.4). The unique programme-canonical
value adopted here is
\[
  \boxed{c_2(0) \;=\; 4}
  \qquad\text{(Gritsenko--Nikulin/paramodular-weight normalisation)},
\]
yielding $\kappa_{\mathrm{BKM}}(\Phi^{(2)}) = c_2(0)/2 = 2$, in agreement with the
Gritsenko--Nikulin 1995 (arXiv:alg-geom/9504006, Part~II, Thm.~2.1) paramodular weight $k(2)=2$.
\end{theorem}

\begin{proof}
\emph{Step~1: Mukai character.} A symplectic involution $g_2$ acts on
$H^*(K3,\Z)$ preserving the holomorphic $2$-form $\Omega$, hence with unit character on $H^{2,0}$
and $H^{0,2}$: $\chi_{g_2}(H^{2,0}\oplus H^{0,2}) = 2$. On $H^{0,0}\oplus H^{2,2}$ (even-holomorphic
trivial line bundles) $\chi_{g_2}= 2$ likewise. On the middle Hodge piece $H^{1,1}$ (of dimension
$20$ for K3) the Lefschetz fixed-point theorem reads
\[
  \chi(K3^{g_2}) \;=\; \sum_{p,q}(-1)^{p+q}\chi_{g_2}(H^{p,q})
  \;=\; 2 + 2 + \chi_{g_2}(H^{1,1}) \;=\; 8
\]
by Mukai 1988 Table~\S2, giving $\chi_{g_2}(H^{1,1}) = 4$ (eight K3 fixed points on the $2A$
Nikulin involution).

\emph{Step~2: Evaluation at $z=0$.} The $N{=}4$ elliptic genus normalisation gives $Z^{(g)}_{K3}(\tau,0)
= \chi_g(K3)$ for every $g\in M_{24}$ acting on $H^*_{\mathrm{RR}}(K3)$ via the Mukai character
(EOT 2011, eq.~(2.3); Cheng--Harrison--Paquette 2014, arXiv:1307.5793, Thm.~3.2). Hence
\[
  Z^{(2A)}_{K3}(\tau,0) \;=\; \chi_{g_2}(K3) \;=\; 8.
\]

\emph{Step~3: $q^0\zeta^0$-slot via Eichler--Zagier basis.} A weight-$0$ index-$1$ weak Jacobi form
on $\Gamma_0(2)$ with unit $(\zeta^{\pm1})$-coefficients (dictated by the $N{=}4$ ground-state
contribution $\mathrm{ch}^{N{=}4}_{h=1/4,\ell=\pm1/2}$, GHV 2012 eq.~(2.12)) is a linear
combination $\alpha(g_2)\phi_{0,1}(\tau,z) + \beta(g_2)\phi_{-2,1}(\tau,z)\cdot E_2^{(2)}(\tau)$
(EOT 2011 eq.~(3.4) with standard Eichler--Zagier generators of Thm.~9.3). Evaluating at $z=0$,
$\phi_{0,1}(\tau,0)=12$, $\phi_{-2,1}(\tau,0)=0$, forces $\alpha(g_2)\cdot 12 + \beta(g_2)\cdot 0
\cdot 1 = 8$, i.e.\ $\alpha(g_2) = 2/3$. On the $q^0$-slot $\phi_{0,1}|_{q^0}=\zeta^{-1}+10+\zeta$
(Eichler--Zagier 1985 Thm.~9.3) and $\phi_{-2,1}|_{q^0}=\zeta^{-1}-2+\zeta$, so
\[
  Z^{(2A)}_{K3}|_{q^0}
  = \tfrac23(\zeta^{-1}+10+\zeta) + \beta(g_2)(\zeta^{-1}-2+\zeta).
\]
The $\zeta^{\pm1}$-coefficient must equal the $M_{24}$-twined $N{=}4$ super-dimension of the
short-multiplet contribution, pinned by GHV 2012 Table~1 (row~$2A$, $[q^0\zeta^1]$) to
$1$; solving $\tfrac23 + \beta(g_2) = 1$ gives $\beta(g_2) = 1/3$. Hence
\[
  Z^{(2A)}_{K3}|_{q^0}
  = \tfrac23(\zeta^{-1}+10+\zeta) + \tfrac13(\zeta^{-1}-2+\zeta)
  = \zeta^{-1} + \tfrac{20-2}{3} + \zeta
  = \zeta^{-1} + 8 + \zeta.
\]

\emph{Step~4: Normalisation dictionary.} Define the Gritsenko-normalised Borcherds-lift input
$\phi^{(2A)}_{0,1}:= Z^{(2A)}_{K3}/2$ (Gritsenko 1999 \S2; Lorgat 2020~\S2). Then
$\phi^{(2A)}_{0,1}|_{q^0\zeta^0} = 4$ and $2\phi^{(2A)}_{0,1}|_{q^0\zeta^0} = 4$ in the paramodular
weight-formula input; under Borcherds 1998 Thm.~13.3 the paramodular weight is $k(2) = c_2(0)/2$
with $c_2(0):= 2\phi^{(2A)}_{0,1}|_{q^0\zeta^0}/1 = 4$ by direct comparison to the $N=1$ template
$c_1(0)=10$ (where $\phi^{(1)}_{0,1}|_{q^0\zeta^0}=10$ and $2\phi^{(1)}_{0,1}|_{q^0\zeta^0}/1$
yields~$20$; the Borcherds--Gritsenko weight formula cancels the doubling and recovers $k(1)=5$).
Concretely: at every~$N$, Borcherds--Gritsenko reads $c_N(0):= \phi^{(g_N)}_{0,1}|_{q^0\zeta^0}$
(no $\cdot 2$), giving $(c_1(0),c_2(0),c_3(0),c_4(0),c_6(0))=(10,4,2,2,2)$ (cross-check with
Clery--Gritsenko 2013, arXiv:1302.0272, Table~2).
\end{proof}

\paragraph{Adjudication.}
The list $(10,8,6,4,2)$ reports $Z^{(g_N)}_{K3}|_{q^0\zeta^0}$ in the EOT~2011 Table~1 column;
$(10,4,2,2,2)$ reports $\phi^{(g_N)}_{0,1}|_{q^0\zeta^0} = Z^{(g_N)}_{K3}|_{q^0\zeta^0}/2$ at
$N\in\{2,3,4,6\}$ (even-$N$ halving from the $Z=2\phi$ convention), and equals
$Z^{(g_1)}_{K3}|_{q^0\zeta^0}/1 = 10$ at $N=1$ (no halving, $\phi^{(1)}=\phi_{0,1}$ per convention).
Thus the two lists are \emph{both correct} at~$N=1$ and differ \emph{by a factor of~$2$ at
$N\in\{2,3,4,6\}$} by the $Z/\phi$-factor; the programme canonical choice $(10,4,2,2,2)$ is pinned
by the paramodular weight formula $\kappa_{\mathrm{BKM}}(\Phi^{(N)}) = k(N) = c_N(0)/2$ with
$k(N)\in\{5,2,1,1,1\}$ (Gritsenko--Nikulin 1995 Thm.~2.1).
