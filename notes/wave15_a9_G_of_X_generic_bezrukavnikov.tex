\title{The boundary of the quantum vertex chiral group \(G(X)\) for a
generic Calabi--Yau threefold}
\author{}
\date{2026-04-30}

\begin{document}
\maketitle

\begin{abstract}
For a smooth proper Calabi--Yau threefold \(X\) outside the toric,
Maulik--Okounkov accessible, \(K3\), and \(K3\times E\) loci, the
global quantum vertex chiral group \(G(X)\) is a construction problem.
The representability theorem in \texttt{quantum\_groups\_foundations.tex}
applies after a precise equivariant input: a torus
\(T\subseteq \Aut(X)\) of rank at least \(2\), isolated fixed points on
the Hilbert schemes, dualisable completed equivariant \(K\)-theory, and
an accessible limit-preserving CoHA-compatibility functor.  Compact
holomorphic Chern--Simons input further requires a solved \(\QME\),
anomaly cancellation, orientation and framing data, descent and sewing,
counterterms, and a compatible completion.  The statements below
separate the constructed equivariant object \(G^T(X)\), the framed
\(\Phi_3\)-output, the Hall positive half, the completed Drinfeld
double, the Borcherds denominator, the Yangian branch, and the
class-\(\mathcal S\) or VOA branch.
\end{abstract}

\section{Constructed Locus}

\begin{definition}[MO-accessible equivariant locus]
Let \(X\) be a smooth Calabi--Yau threefold.  The pair \((X,T)\) is
\emph{MO-accessible} when a torus \(T\subseteq \Aut(X)\) of rank
\(\geq 2\) acts with isolated fixed points on each \(\Hilb^n(X)\), the
completed \(K_T\)-theory of these fixed-point spaces is dualisable over
\(K_T(\mathrm{pt})\), and the CoHA-compatibility functor \(F_X\) of
\texttt{quantum\_groups\_foundations.tex}, Definition
\(\ref{def:qgf-F-X}\), is accessible and limit-preserving.
\end{definition}

\begin{theorem}[Conditional equivariant representability]
\label{thm:generic-GX-equivariant-representability}
On the MO-accessible equivariant locus, Theorem
\(\ref{thm:qgf-G-X-representability}\) of
\texttt{quantum\_groups\_foundations.tex} corepresents \(F_X\) by an
object \(G^T(X)\in\QChirGrp_\kk\).  This object is an abstract Hopf
object in the completed equivariant quantum-chiral category.  A
presentation of \(G^T(X)\) as a Yangian, a completed Drinfeld double, a
Borcherds--Kac--Moody algebra, or a \(\mathcal W\)-algebra requires
extra structure: a positive half, a Cartan part, a Serre-dual negative
half, a non-degenerate Hopf pairing, a topological completion, and the
corresponding \(R\)-matrix or denominator identity.
\end{theorem}

\begin{proof}
The cited theorem is Brown--Lurie representability in the presentable
\((\infty,1)\)-category \(\QChirGrp_\kk\).  Presentability and
limit-preservation give a corepresenting object for \(F_X\).  The proof
constructs a universal \(F_X\)-compatible module datum; generators,
relations, triangular decomposition, Hopf pairing, \(R\)-matrix, and
Borcherds denominator are presentation data beyond representability.
\end{proof}

\begin{proposition}[Generic compact threefolds]
\label{prop:generic-GX-unconstructed}
Let \(X\) be a generic smooth proper Calabi--Yau threefold, for
example a generic quintic hypersurface \(X_5\subset\bP^4\).  Then \(X\)
lies outside the MO-accessible locus: its algebraic automorphism group
is finite, so the required positive-rank torus action with isolated
Hilbert-scheme fixed points is absent.  Theorem
\ref{thm:generic-GX-equivariant-representability} therefore leaves
global \(G(X)\) unconstructed for \(X\).  In this stratum \(G(X)\) is a
construction problem.  A theorem for this stratum must supply the
compact Hall object, the framed compact \(\Phi_3\)-output, and the
center or double comparison as part of its hypotheses.
\end{proposition}

\begin{proof}
For the generic quintic, finiteness of \(\Aut(X_5)\) excludes a
\(\bC^\times\)-action.  The Maulik--Okounkov stable-envelope chamber
and the completed equivariant \(K\)-theory functor required by
Definition~\ref{def:qgf-F-X} lack input.  Critical CoHA objects and
vanishing-cycle sheaves on critical moduli, where separately
constructed, are enumerative positive-half data.  A full quantum group
additionally needs a pairing, Cartan, negative half, completion, and
comparison with the framed \(\Phi_3\)-output.
\end{proof}

\section{Operadic Level}

\begin{proposition}[Native \(d\geq 3\) output]
\label{prop:generic-GX-native-E1}
For a framed threefold specialisation
\[
  A_X^{(\Sigma_2,C)}
  =
  \SpCh_{\Sigma_2,C}\bigl(\Phi^{\mathrm{FA}}_3(\Perf(X))\bigr),
\]
the specialised chiral output is natively \(\Eone\).  Braided
\(\Etwo\)-data at \(d=3\) live on centre, double, or module layers,
for instance
\[
  \mathcal Z\bigl(\Rep^{\Eone}(A_X^{(\Sigma_2,C)})\bigr)
  \qquad\text{or}\qquad
  D(Y^+(X)),
\]
after the relevant half-braidings, Hopf pairing, Cartan part, negative
half, and completion have been constructed.
\end{proposition}

\begin{proof}
This is Proposition~\(\ref{prop:native-en-level}\) of
\texttt{cy\_to\_chiral.tex}: the native operadic level is
\(\infty\) at \(d=1\), \(2\) at \(d=2\), and \(1\) at \(d\geq3\).
Remark~\(\ref{rem:e2-on-drinfeld-centre}\) identifies the braided
layer with the Drinfeld centre of the \(\Eone\)-monoidal
representation category.  Dunn additivity and the centre theorem move
from ordered \(\Eone\)-data to braided \(\Etwo\)-representation data;
they operate on a new layer separate from the original threefold output.
\end{proof}

\begin{proposition}[Positive half and full quantum group]
\label{prop:positive-half-scope-GX}
An isomorphism
\[
  \CoHA(X)\simeq Y^+(X)
\]
identifies positive halves as associative \(\Eone\)-algebras.  A full
Yangian, completed Drinfeld double, or \(\mathcal W\)-object additionally
requires
\[
  Y^-(X),\qquad Y^0(X),\qquad
  \langle-,-\rangle_{\mathrm{Hopf}},\qquad
  \widehat{\otimes},\qquad
  \mathcal R.
\]
The equality of positive halves alone implies neither
\(G(X)=Y(X)\) nor a vertex-algebra identification.
\end{proposition}

\begin{proof}
For \(X=\bC^3\), Schiffmann--Vasserot identify the CoHA with
\(Y^+(\widehat{\mathfrak{gl}}_1)\).  The full affine Yangian is obtained
after adjoining the Cartan and negative halves and completing the
Drinfeld double.  The \(\mathcal W_{1+\infty}\) object is reached
through the double and an evaluation representation on the vertex side.
\end{proof}

\section{Compact Holomorphic Chern--Simons}

\begin{hypothesis}[Compact hCS quantisation package]
\label{hyp:compact-hcs-package}
For a compact smooth Calabi--Yau threefold \(X\), a global holomorphic
Chern--Simons input for \(G(X)\) consists of:
\begin{enumerate}
\item a \(\BV\) field theory on \(X\) whose effective action
  \(S_{\mathrm{eff}}\) satisfies the quantum master equation
  \[
    Q S_{\mathrm{eff}}
    + \frac12\{S_{\mathrm{eff}},S_{\mathrm{eff}}\}_{\BV}
    + \hbar\,\Delta_{\BV}S_{\mathrm{eff}}=0;
  \]
\item cancellation of the local Costello--Li anomaly class, including
  every one-loop and higher local term used in the compact descent;
\item orientation data, determinant-line square roots, and the framed
  \(\bS^3\)-compatible TCFT correction datum;
\item \v{C}ech--Ran descent and sewing over a finite analytic cover,
  with overlap and triple-overlap coherences;
\item local counterterms whose restrictions agree on overlaps and whose
  finite-cutoff limits form a Mittag--Leffler compatible system;
\item a topological completion compatible with the Hall product, the
  chiral OPE, and the center or Drinfeld double comparison.
\end{enumerate}
\end{hypothesis}

\begin{proposition}[Compact hCS boundary for generic \(X\)]
\label{prop:compact-hcs-boundary-generic}
For a generic compact Calabi--Yau threefold \(X\), holomorphic
Chern--Simons theory supplies \(G(X)\) only after the data of
Hypothesis~\ref{hyp:compact-hcs-package} and the Hall package of
Hypothesis~\ref{hyp:generic-GX-input-package} have been constructed.
Local quantisation on charts, a BCOV anomaly formula, or a critical
moduli stack supplies input for this package without producing the
global quantum vertex chiral group.
\end{proposition}

\begin{proof}
The \(\QME\) gives a differential on compact observables.  Anomaly
cancellation makes that differential square to zero after quantisation.
Orientation and determinant-line data define signs in vanishing-cycle
and Hall products.  Descent and sewing compare chartwise BV actions on
overlaps and triple overlaps.  Counterterms and completion make the
cutoff limit a continuous object.  The Hall pairing, negative half,
Cartan, and center or double comparison then turn the positive-half data
into a candidate quantum group.
\end{proof}

\section{Conditional Generic Object}

\begin{hypothesis}[Generic threefold input package]
\label{hyp:generic-GX-input-package}
Let \(X\) be a smooth proper Calabi--Yau threefold.  A construction of
\(G(X)\) beyond the MO-accessible and toric loci requires the following
data.
\begin{enumerate}
\item A framed Stage-2 specialisation
  \(\SpCh_{\Sigma_2,C}\circ \Phi^{\mathrm{FA}}_3\) with explicit TCFT
  background, chain-level framing, orientation line, compact hCS
  \(\QME\), anomaly cancellation, descent, sewing, counterterms, and
  completed state space.
\item A critical CoHA or Hall object \(Y^+_{\mathrm{crit}}(X)\) as an
  associative \(\Eone\)-algebra, with a PBW theorem and a specified
  stability chamber.
\item A Cartan factor, Serre-dual negative half, non-degenerate Hopf
  pairing, and topological completion producing
  \(D(Y^+_{\mathrm{crit}}(X))\).
\item A comparison between the doubled Hall object and the framed
  \(\Phi_3\)-output, including the centre or half-braiding whenever a
  braided \(\Etwo\)-category is asserted.
\item For a Borcherds or automorphic presentation, a denominator product
  with a specified Jacobi input and Fourier coefficient \(c(0)\).  Its
  BKM modular characteristic is \(\kBKM=c(0)/2\).
\end{enumerate}
\end{hypothesis}

\begin{definition}[Candidate generic object]
\label{def:candidate-generic-GX}
Under Hypothesis~\ref{hyp:generic-GX-input-package}, define
\[
  G^{(\Sigma_2,C)}_{\mathrm{cand}}(X)
  :=
  D\bigl(Y^+_{\mathrm{crit}}(X)\bigr).
\]
When any item of Hypothesis~\ref{hyp:generic-GX-input-package} is
absent, the symbol \(G(X)\) names the corresponding construction
problem.
\end{definition}

\begin{conjecture}[Generic CY-C comparison]
\label{conj:generic-GX-CYC}
For a smooth proper Calabi--Yau threefold satisfying
Hypothesis~\ref{hyp:generic-GX-input-package}, the candidate
\(G^{(\Sigma_2,C)}_{\mathrm{cand}}(X)\) is the quantum-group object
predicted by Conjecture CY-C.  The comparison is an isomorphism after
the framed \(\Eone\) specialisation and, where a braided category is
claimed, after passage to the Drinfeld centre or completed double.
\end{conjecture}

\label{thm:generic-GX-numerical-anchors}
The constructed loci carry the following invariants.

\hline
\end{tabular}
\end{center}
The primitive CHL denominator constants, when a Borcherds presentation
is part of the constructed data, are
\[
  c_N(0)=(10,8,6,4,2),\qquad
  \kBKM(\Phi_N)=c_N(0)/2=(5,4,3,2,1),
  \qquad
  N\in\{1,2,3,4,6\}.
\]
At \(N=1\),
\[
  D_5=64^{-1}\Delta_5,\qquad
  \Phi_{10}^{\mathrm{un}}=\Delta_5^2,\qquad
  \Phi_{10}^{\mathrm{OP}}=D_5^2.
\]
\end{theorem}

\begin{proof}
The \(\bC^3\) row is Schiffmann--Vasserot:
\(\CoHA(\bC^3)=Y^+(\widehat{\mathfrak{gl}}_1)\) as an associative
\(\Eone\)-Hall algebra.  The vertex algebra
\(\mathcal W_{1+\infty}\) appears after the Drinfeld double and an
evaluation representation.

The conifold and local-surface values are the toric rows of the Vol III
CY-d stratification chapter: direct McKay gives
\(\kch(\mathrm{conifold})=1\), and the local-surface shadow gives
\(\kch(\mathrm{local}\,\bP^2)=\frac12\chi_{\mathrm{top}}(\bP^2)=3/2\).

For \(Y=K3\times E\), K\"unneth gives
\(\kcat(Y)=\chi(\cO_{K3})\chi(\cO_E)=2\cdot0=0\), while the compact
Hodge supertrace is
\[
  \kch(Y)=\sum_{q=0}^3(-1)^q h^{0,q}(Y)=1-1+1-1=0.
\]
The Heisenberg--Mukai specialisation has
\(\kappa_{\mathrm{ch}}^{\mathrm{Heis}}=3\).  The Borcherds denominator
of \(\fg_{\Delta_5}\) has
\(\kBKM(\fg_{\Delta_5})=c_1(0)/2=10/2=5\).  The fibre invariant is the
Mukai-lattice rank \(\kfib=\mathrm{rk}\,\widetilde H(K3,\mathbb Z)=24\).
These are four constructions and five named numerical entries once the
compact \(\kch\)-supertrace is displayed; additive formulas are excluded
by the distinct construction sources.

The CHL constants are the Gritsenko--Nikulin and Borcherds primitive
denominator weight constants recorded in the Vol III CY-d
stratification chapter and its closed-form \(c_N(0)\) lemma.  The
normalisations
\(D_5=64^{-1}\Delta_5\),
\(\Phi_{10}^{\mathrm{un}}=\Delta_5^2\), and
\(\Phi_{10}^{\mathrm{OP}}=D_5^2\) are those of
\texttt{k3e\_bkm\_chapter.tex}, Proposition
\(\ref{prop:k3e-delta5-phi10-phi12-separation}\).
MNOP and Oberdieck--Pandharipande supply reduced GW/DT and stable-pairs
scalar identities in the OP lane; the primitive BKM denominator remains
the \(\Delta_5\) lane with weight \(5\).
\end{proof}

\section{Proof Obligations}

For a generic compact Calabi--Yau threefold, a proof of
Conjecture~\ref{conj:generic-GX-CYC} must construct the following
objects.
\begin{enumerate}
\item The framed compact threefold \(\Phi_3\)-specialisation with TCFT,
  \(\QME\), anomaly cancellation, orientation, framing, descent, sewing,
  counterterms, and completion data.
\item The critical positive-half CoHA as an \(\Eone\)-algebra, with
  stability dependence controlled by wall crossing.
\item The Cartan and negative halves and a non-degenerate Hopf pairing.
\item The completed Drinfeld double and its centre-level
  \(\Etwo\)-braiding.
\item The comparison isomorphism between this doubled object and the
  framed \(\Phi_3\)-output.
\item If a BKM or automorphic form is claimed, the denominator product
  and the constant term \(c(0)\) giving \(\kBKM=c(0)/2\).
\end{enumerate}
