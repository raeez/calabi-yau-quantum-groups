\title{Gauge 2-Groupoids, Framings, and Quantisation in Holomorphic Chern-Simons Theory}
\author{}
\date{}

\begin{document}
\maketitle

\begin{abstract}
Holomorphic Chern-Simons theory takes a finite-dimensional gauge
coefficient as external field-theoretic input. A Calabi-Yau dg category
supplies Hochschild polyvectors, autoequivalence mapping spaces, and
framed endomorphism algebras. The Deligne gauge 2-groupoid, derived
quotient stack, BRST complex, BV field-antifield theory, hCS
quantisation, Drinfeld centre, and chiral output form a typed ladder
with explicit hypotheses at every arrow. A framing object and geometric
realisation are required before categorical endomorphisms yield a
finite colour algebra. Compact nonabelian hCS requires a Costello QME
solution, quartic anomaly trivialisation, orientation and framing,
analytic completion, gauge fixing with regulator, counterterms, and
descent or sewing data. In complex dimension three the QME obstruction
is the quartic class \(I_4^\rho\). The element \(C_2^\rho\) supplies a
two-point kinetic counterterm. The specialised CY-to-chiral output at
\(d\geq3\) is native \(\Eone\); \(\Etwo\) structure belongs to centre,
double, or module layers.
\end{abstract}

\section{Typed Inputs}
\label{sec:typed-inputs}

\begin{definition}[Geometric hCS coefficient]
\label{def:hcs-coefficient}
Let \(X\) be a complex threefold with holomorphic volume form
\(\Omega_X\). A geometric holomorphic Chern-Simons coefficient consists
of a complex reductive Lie algebra \(\fg\), an invariant pairing or trace
\(\Tr_\rho\) coming from a representation
\(\rho:\fg\to\End(W)\), and a principal \(G\)-bundle \(P\to X\). The
classical field dg Lie algebra is
\[
  \cL_{\hCS}(X,P,\fg)
  =
  \Omega^{0,\bullet}(X,\operatorname{ad}P)[1],
  \qquad d=\dbar_P .
\]
In a trivialisation \(P|_U\simeq U\times G\),
\[
  \cL_{\hCS}(U,\fg)
  =
  \Omega^{0,\bullet}(U)\otimes\fg[1].
\]
The classical action on \(a\in\Omega^{0,\bullet}(X,\operatorname{ad}P)\)
is
\[
  S_{\hCS}(a)
  =
  \int_X \Omega_X\wedge
  \Tr_\rho\!\left(
    \frac12 a\,\dbar_Pa+\frac16 a[a,a]
  \right).
\]
\end{definition}

\begin{definition}[Gauge 2-groupoid and derived quotient]
\label{def:gauge-2groupoid-derived-stack}
Let
\[
  L_P=\Omega^{0,\bullet}(X,\operatorname{ad}P),
  \qquad d=\dbar_P .
\]
For a nilpotent Artinian cdga ideal \(\mathfrak m\), the formal
Deligne gauge 2-groupoid is the two-truncation of the Maurer-Cartan
simplicial set
\[
  \Del_P(\mathfrak m)
  =
  \tau_{\leq2}
  \MC\!\left(
    L_P\otimes\mathfrak m\otimes\Omega^\bullet(\Delta^\bullet)
  \right).
\]
Its objects are elements \(a\in(L_P\otimes\mathfrak m)^1\) satisfying
\[
  \dbar_Pa+\frac12[a,a]=0.
\]
The degree-zero part gives gauge 1-morphisms, and negative degrees in
the test cdga give gauge 2-morphisms. The associated formal derived
quotient stack at \(P\) is
\[
  \cM^{0,1}_G(X)_P
  =
  [\,\MC(L_P)/\cG_P\,],
  \qquad
  \cG_P=\Gamma(X,\operatorname{Ad}P),
\]
with tangent complex \(R\Gamma(X,\operatorname{ad}P)[1]\).
\end{definition}

\begin{definition}[BRST and BV packages]
\label{def:brst-bv-packages}
The BRST complex of the formal quotient in
Definition~\ref{def:gauge-2groupoid-derived-stack} is the
Chevalley-Eilenberg model for functions on the derived quotient,
\[
  C^\bullet_{\mathrm{BRST}}(X,P)
  =
  \CE^\bullet\!\left(
    L_P,
    \cO(\MC(L_P))
  \right),
\]
with differential given by \(\dbar_P\), the gauge action, and
contraction with the Maurer-Cartan equation. The BV field theory is the
shifted cotangent theory
\[
  T^\ast[-1]\cM^{0,1}_G(X)_P,
\]
represented by
\[
  L_P[1]\oplus L_P^\vee[-2]
  \simeq
  \Omega^{0,\bullet}(X,\operatorname{ad}P)[1]
  \oplus
  \Omega^{0,3-\bullet}_c(X,\operatorname{ad}P)^\vee[-2],
\]
where the last identification uses the CY$_3$ volume form and the trace
pairing. A quantum hCS theory is a Costello quantisation of this BV
theory. A chiral centre is a further construction applied to a chiral
algebra or its representation category.
\end{definition}

\begin{definition}[Oriented and framed CY category]
\label{def:framed-cy-category}
An oriented CY$_3$ dg category is a smooth proper \(\C\)-linear
pretriangulated dg category \(\cC\) with a negative cyclic Calabi-Yau
class \(\eta\in HC^-_3(\cC)\) whose Hochschild image identifies the
inverse dualising bimodule with \(\cC[-3]\). A brane framing of
\((\cC,\eta)\) is a perfect object \(\cE\in\cC\). It determines the
\(\Eone\)-algebra
\[
  A_{\cE}
  =
  \REnd_{\cC}(\cE)
  =
  \RHom_{\cC}(\cE,\cE)
\]
and its commutator dg Lie algebra \(A_{\cE}^{\operatorname{Lie}}\).
A Morita presentation framing is a compact generator \(V\in\cC\);
then \(A_V=\REnd_{\cC}(V)\) and \(\Perf(A_V)\simeq\cC\).
\end{definition}

\begin{proposition}[Typed layers]
\label{prop:typed-layers}
For a geometric framed input \((X,\Omega_X,P,\rho,\cE)\), the following
objects have different definitions and functorial behaviour:
\[
\begin{array}{c|c|c}
\text{layer} & \text{object} & \text{role}\\
\hline
\text{external hCS coefficient} & (\fg,\rho,\Tr_\rho) & \text{finite colour input}\\
\text{Deligne data} & \Del_P & \text{formal gauge 2-groupoid}\\
\text{gauge stack} & \cM^{0,1}_G(X)_P & \text{derived quotient}\\
\text{BRST functions} & C^\bullet_{\mathrm{BRST}}(X,P) & \text{functions on the quotient}\\
\text{BV fields} & T^\ast[-1]\cM^{0,1}_G(X)_P & \text{field-antifield theory}\\
\text{framed endomorphisms} & \REnd_{\cC}(\cE) & \text{brane endomorphism algebra}\\
\text{Hochschild polyvectors} & \HH^\bullet(\cC) & \text{intrinsic CY deformation complex}\\
\end{array}
\]
For \(\cC=\Perf(X)\) and \(\cE=E\) a vector bundle, the Dolbeault model is
\[
  \REnd_{\Perf(X)}(E)
  \simeq
  \Omega^{0,\bullet}(X,\End E),
  \qquad d=\dbar_E .
\]
If \(E=\cO_X^{\oplus r}\) on a trivialising chart, this becomes
\[
  \Omega^{0,\bullet}(X)\otimes\operatorname{Mat}_r(\C),
\]
whose fibre coefficient is \(\fgl_r\). The recovery of \(\fgl_r\)
therefore uses the framed geometric object \(E\) and the trivialisation.
The unframed oriented category \((\cC,\eta)\) supplies
\(\HH^\bullet(\cC)\), the derived centre, and categorical symmetries.
\end{proposition}

\begin{proof}
The first row is the finite colour datum in the hCS action. The next
four rows are Definitions~\ref{def:gauge-2groupoid-derived-stack}
and~\ref{def:brst-bv-packages}: the Deligne object remembers formal
gauge homotopies, the stack is the derived quotient, the BRST complex
models functions on that quotient, and the BV theory is the shifted
cotangent field theory. The framed endomorphism row is the algebra of
the selected brane object. The Hochschild row is the Morita-invariant
object
\[
  \HH^\bullet(\cC)
  =
  \REnd_{\cC\otimes\cC^{\mathrm{op}}}(\cC),
\]
which for \(\Perf(X)\) is identified by HKR with polyvectors, with the
usual Todd correction in global comparisons:
\[
  \HH^\bullet(\Perf(X))
  \simeq
  \bigoplus_{p,q}H^q(X,\wedge^pT_X).
\]
The CY volume form and trace pairing identify the cotangent term in the
BV row with the displayed antifield complex. These constructions use
different universal properties.
\end{proof}

\section{Symmetries, Framings, and Morita Presentations}
\label{sec:symmetries-framings-morita}

\begin{definition}[Aut-equivalence space]
\label{def:aut-infty}
The categorical symmetry of a dg category \(\cC\) is the mapping space
\[
  \Aut_\infty(\cC)
  =
  \Map_{\mathrm{dgCat}_\C}(\cC,\cC)^\simeq .
\]
Its \(\pi_0\) is the ordinary group of exact dg autoequivalences. Its
loop space at the identity is the unit space of the derived centre.
\end{definition}

\begin{definition}[Framed object groupoid and Morita presentation groupoid]
\label{def:two-framing-groupoids}
Let \(\Fr(\cC)\) be the groupoid of perfect objects of \(\cC\), with
isomorphisms in the maximal subgroupoid of \(\cC\). Let
\(\Mor_{\mathrm{gen}}(\cC)\) be the bicategory whose objects are compact
generators \(V\), whose 1-morphisms from \(V\) to \(W\) are perfect
\(A_W\)-\(A_V\) bimodules implementing Morita equivalences, and whose
2-morphisms are bimodule equivalences.
\end{definition}

\begin{proposition}[Exact functors between categorical inputs]
\label{prop:exact-functors}
There are canonical functors
\[
  \Aut_\infty(\cC)\times\Fr(\cC)
  \longrightarrow
  \Fr(\cC),
  \qquad
  (F,\cE)\longmapsto F(\cE),
\]
\[
  \Fr(\cC)
  \longrightarrow
  \Eone\operatorname{-Alg},
  \qquad
  \cE\longmapsto\REnd_{\cC}(\cE),
\]
and
\[
  \Eone\operatorname{-Alg}
  \longrightarrow
  \operatorname{dgLie},
  \qquad
  A\longmapsto A^{\operatorname{Lie}} .
\]
The Morita presentation groupoid carries a separate functor
\[
  \Mor_{\mathrm{gen}}(\cC)
  \longrightarrow
  \operatorname{Cat}^{\mathrm{Mor}},
  \qquad
  V\longmapsto \Perf(\REnd_{\cC}(V)).
\]
The finite hCS coefficient \(\fg\) is obtained from a geometric
trivialisation of \(A_{\cE}^{\operatorname{Lie}}\) or supplied as an
external coefficient in Definition~\ref{def:hcs-coefficient}. It is a
finite-dimensional Lie algebra, while \(\Aut_\infty(\cC)\) is an
\(\infty\)-group and \(\Mor_{\mathrm{gen}}(\cC)\) is a Morita
bicategory of presentations.
\end{proposition}

\begin{proof}
The first functor is the action of self-equivalences on objects. The
second is functorial because an isomorphism \(\cE\simeq\cF\) conjugates
endomorphism algebras. The third is the commutator construction on
\(\Eone\)-algebras. The fourth is Bondal-Van den Bergh and Keller
derived Morita theory: a compact generator \(V\) presents \(\cC\) as
\(\Perf(A_V)\), and an invertible bimodule gives the Morita equivalence
between two such presentations. These functors have distinct codomains,
which separates gauge coefficients, framed objects, Morita
presentations, and autoequivalences.
\end{proof}

\begin{theorem}[Rank framings on \(\C^3\)]
\label{thm:c3-rank-framings}
Let \(X=\C^3\), \(R=\C[x,y,z]\), \(\cC=\Perf(R)\), and
\(V_r=\cO_X^{\oplus r}\) for \(r\geq1\). Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \(V_r\) is a compact generator of \(\cC\).
\item \(\REnd_{\cC}(V_r)=\operatorname{Mat}_r(R)\), concentrated in
cohomological degree zero.
\item The associated current Lie algebra is
\[
  \operatorname{Mat}_r(R)^{\operatorname{Lie}}
  =
  \fgl_r\otimes R .
\]
\item The fibre coefficient of the trivial hCS bundle is \(\fgl_r\),
and the local hCS field complex is
\[
  \Omega^{0,\bullet}(\C^3)\otimes\fgl_r[1].
\]
\item \(V_r\) and \(V_s\) lie in the same \(\Aut_\infty(\cC)\)-orbit
exactly when \(r=s\).
\item The algebras \(\operatorname{Mat}_r(R)\) and
\(\operatorname{Mat}_s(R)\) are Morita equivalent for all \(r,s\geq1\).
\end{enumerate}
\end{theorem}

\begin{proof}
The object \(\cO_X\) generates \(\Perf(R)\) because \(R\) is affine and
regular; finite direct sums of a compact generator remain compact
generators. The endomorphism computation is the usual matrix
calculation over \(R\). The derived Picard description for affine
space gives
\[
  \operatorname{DPic}(R)
  \simeq
  \Aut(R)\ltimes\Z,
\]
so an autoequivalence sends \(R^{\oplus r}\) to
\({}_{\varphi}R^{\oplus r}[n]\). If this object is isomorphic to
\(R^{\oplus s}\), then \(n=0\) and comparison after tensoring with
\(R/(x,y,z)\) gives \(r=s\). Equivalently, an autoequivalence preserves
endomorphism dg algebras up to quasi-isomorphism:
\[
  \REnd_{\cC}(F(V_r))\simeq \REnd_{\cC}(V_r).
\]
The fibre at the origin of \(\operatorname{Mat}_r(R)\) has dimension
\(r^2\), so it is isomorphic to the fibre of
\(\operatorname{Mat}_s(R)\) exactly when \(r=s\). Morita equivalence of
matrix algebras over a fixed ring is implemented by the standard
rectangular bimodules.
\end{proof}

\section{Quantum hCS and the Anomaly Slot}
\label{sec:quantum-hcs-anomaly-slot}

\begin{definition}[Quartic invariant]
\label{def:quartic-invariant}
For a representation \(\rho:\fg\to\End(W)\), define
\[
  I_4^\rho(x_1,x_2,x_3,x_4)
  =
  \frac1{4!}\sum_{\sigma\in S_4}
  \Tr_W\!\left(
    \rho(x_{\sigma(1)})\rho(x_{\sigma(2)})
    \rho(x_{\sigma(3)})\rho(x_{\sigma(4)})
  \right).
\]
Then \(I_4^\rho\in\Sym^4(\fg^\vee)^\fg\).
\end{definition}

\begin{proposition}[One-loop QME obstruction]
\label{prop:qme-obstruction}
In complex dimension three, the nonabelian one-loop hCS obstruction is
represented locally by
\[
  \Theta^\rho_U(\alpha_0,\alpha_1,\alpha_2,\alpha_3)
  =
  \int_U
  \left[
  I_4^\rho\bigl(
    \alpha_0,\partial\alpha_1,\partial\alpha_2,\partial\alpha_3
  \bigr)
  \right]_{(3,3)},
  \qquad
  \alpha_i\in\Omega^{0,\bullet}_c(U,\fg).
\]
Its obstruction class is
\[
  \kanom(X,\fg,\rho)
  =
  [\,\hbar\,\Theta^\rho\,]
  \in
  H^1_{\mathrm{loc}}
  \bigl(\Omega^{0,\bullet}(X,\fg)[1],\cO_{\mathrm{loc}}\bigr).
\]
For \(\rho=\operatorname{ad}\), the corresponding descent expression is
\[
  \int_X \Tr_{\operatorname{ad}}(c\,F_A^3).
\]
\end{proposition}

\begin{proof}
The open one-loop anomaly graph is the wheel with four external gauge
insertions. Its colour factor is the symmetrised trace of four
representation matrices, which is \(I_4^\rho\). The holomorphic form
factor contributes three \(\partial\)-derivatives; in complex dimension
three this produces a holomorphic top density. Projection to bidegree
\((3,3)\) gives a degree-one local functional. Costello's obstruction
theory identifies this class as the first obstruction to solving the
scale-\(L\) QME.
\end{proof}

\begin{definition}[Degree-two colour contraction]
\label{def:degree-two-colour}
Let \(\{T_a\}\) be a basis of \(\fg\) and \(\{T^a\}\) the dual basis
for the chosen invariant pairing. The degree-two colour contraction in
the representation \(\rho\) is
\[
  C_2^\rho=\sum_a \rho(T_a)\rho(T^a).
\]
For a simple Lie algebra in the adjoint normalisation,
\[
  C_2^{\operatorname{ad}}=2h^\vee .
\]
\end{definition}

\begin{proposition}[Two-point counterterm]
\label{prop:c2-counterterm}
The two-point graph contributes a degree-zero kinetic counterterm of
the following trace-channel form:
\[
  S_{\mathrm{ct}}^{(2)}
  =
  -\hbar\,\lambda_\rho(L,\varepsilon)
  \int_X\Omega_X\wedge\Tr_\rho(A\,\dbar A),
\]
where \(\lambda_\rho(L,\varepsilon)\) is the heat-kernel scalar obtained
from the contraction \(C_2^\rho\) after the chosen pairing and regulator
are fixed. Thus \(C_2^\rho\) belongs to the field-strength
renormalisation slot, while
\(\kanom(X,\fg,\rho)\) is governed by \(I_4^\rho\).
\end{proposition}

\begin{proof}
Closing two legs in the cubic hCS vertex contracts the colour indices
as \(\sum_a\rho(T_a)\rho(T^a)\). The remaining two fields have the
tensor type of the kinetic functional. The cohomological degree is zero.
The QME obstruction of Proposition~\ref{prop:qme-obstruction} has
degree one and four external gauge insertions.
\end{proof}

\begin{hypothesis}[Compact nonabelian hCS package]
\label{hyp:compact-hcs-package}
A compact nonabelian hCS quantisation theorem on a CY$_3$ target
requires:
\begin{enumerate}[label=\textup{(\roman*)}]
\item a compact Calabi-Yau threefold \((X,\Omega_X)\) with orientation
and determinant-line framing compatible with the BV pairing;
\item a quadratic gauge Lie algebra or dg Lie algebra
\((\fg,\langle-,-\rangle)\) and a trace representation \(\rho\) fixing
\(I_4^\rho\) and \(C_2^\rho\);
\item a Costello gauge fixing, heat-kernel regulator, propagators, and
RG-compatible counterterms;
\item completed local functionals and observables on which
\(\Delta_L\), \(\{-,-\}_L\), and RG flow are defined;
\item a solution of the scale-\(L\) QME;
\item a vanishing or explicit trivialisation of
\(\kanom(X,\fg,\rho)\) in the local BV obstruction complex;
\item factorisation descent over \(X\), sewing data for global traces
or partition functions, and compatibility of all counterterms with the
chosen completion.
\end{enumerate}
\end{hypothesis}

\begin{theorem}[Conditional compact hCS observables]
\label{thm:conditional-compact-hcs}
Assume Hypothesis~\ref{hyp:compact-hcs-package}. If the obstruction
class \(\kanom(X,\fg,\rho)\) is trivial after the specified local and
closed-sector counterterms are included, then the effective
interactions can be chosen to solve the QME through one loop. Under the
analogous higher-loop obstruction trivialisation hypotheses, the
completed observables form a quantum factorisation algebra on \(X\).
A surviving obstruction class is exactly the failure of the QME for the
specified field content and regulator.
\end{theorem}

\begin{proof}
Costello's renormalisation theorem solves the QME inductively in powers
of \(\hbar\). At the first quantum step the obstruction is the
degree-one local class represented by \(\Theta^\rho\). Triviality gives
the first quantum correction. The same induction handles higher loops
inside the completed local functional complex. A surviving class means
the inductive equation has no solution at that order.
\end{proof}

\begin{remark}[Characteristic shadows]
\label{rem:characteristic-shadows}
The expression \(A(\fg)\chi_{\mathrm{top}}(X)\) is a numerical shadow
defined after a separate index calculation, trace normalisation, and
descent representative have been fixed. The local QME obstruction
class is \([\,\hbar\Theta^\rho\,]\), controlled by the quartic invariant
\(I_4^\rho\). For \(K3\times E\),
\[
  \chi_{\mathrm{top}}(K3\times E)=24\cdot0=0,
\]
and this topological vanishing leaves the quartic gauge class dependent
on \((\fg,\rho,I_4^\rho)\), the regulator, and the chosen
trivialisation.
\end{remark}

\section{CY-to-Chiral Output}
\label{sec:cy-to-chiral-output}

\begin{proposition}[Native output at \(d\geq3\)]
\label{prop:native-output-dgeq3}
Let \(\cC\) be a CY$_d$ category on a verified locus of the two-stage
construction, and let
\[
  A_\cC^{(\Sigma_{d-1},C)}
  =
  \SpCh_{\Sigma_{d-1},C}\bigl(\PhiFA_d(\cC)\bigr)
\]
be the specialised chiral algebra on the reference curve \(C\). Its
native operadic level is
\[
  n_{\mathrm{native}}(d)
  =
  \begin{cases}
  \infty, & d=1,\\
  2, & d=2,\\
  1, & d\geq3.
  \end{cases}
\]
Thus for \(d\geq3\), \(A_\cC^{(\Sigma_{d-1},C)}\) is an \(\Eone\)-chiral
algebra. Braided \(\Etwo\) structure belongs to the representation
centre, Drinfeld double, or module layer, for example
\[
  \cZ\!\left(\Rep^{\Eone}(A_\cC^{(\Sigma_{d-1},C)})\right),
\]
when the centre construction and its comparison data have been supplied.
\end{proposition}

\begin{proof}
The Gerstenhaber bracket on \(\HH^\bullet(\cC)\) has degree \(1-d\).
Kontsevich formality gives the \(E_d\)-structure on the witnessed
locus. Factorisation homology over the
\((d-1)\)-dimensional holomorphic specialisation submanifold
\(\Sigma_{d-1}\) drops the residual operadic level to \(E_1\) for
\(d\geq3\). The nonsymmetric quantum-group braiding is recovered by
adjoining half-braidings in the representation centre or by constructing
the corresponding Drinfeld double.
\end{proof}

\begin{proposition}[Toric \(\C^3\) Hall normalisation]
\label{prop:c3-hall-normalisation}
On the toric \(\C^3\) chart, the critical cohomological Hall algebra is
\[
  \CoHA(\C^3)\simeq \Yp(\widehat{\fgl}_1),
\]
the positive half of the affine Yangian. The full
\(\Winf\) vertex algebra appears after the additional chain
\[
  \Yp(\widehat{\fgl}_1)
  \longrightarrow
  D_\hbar\!\left(\Yp(\widehat{\fgl}_1)\right)
  \longrightarrow
  \text{completed centre or evaluation representation}
  \longrightarrow
  \Winf .
\]
\end{proposition}

\begin{proof}
The first identification is the Kontsevich-Soibelman and
Schiffmann-Vasserot toric CoHA calculation in shuffle normalisation.
The displayed chain adds the Hopf pairing, negative half, Cartan
completion, centre data, and evaluation representation. These data are
external to the positive Hall half.
\end{proof}

\begin{proposition}[\(K3\times E\) scalar slots]
\label{prop:k3xe-scalar-slots}
For \(X=K3\times E\), the compact categorical and chiral entries are
\[
  \kappa_{\mathrm{cat}}(K3\times E)
  =
  \chi(\cO_{K3})\chi(\cO_E)
  =
  2\cdot0
  =
  0,
  \qquad
  \kappa_{\mathrm{ch}}(A_{K3\times E})=0.
\]
The Heisenberg-Mukai specialisation, Borcherds denominator, and fibre
rank give the separate values
\[
  \kappa_{\mathrm{ch}}^{\mathrm{Heis}}(K3\times E)=3,
  \qquad
  \kappa_{\mathrm{BKM}}(\mathfrak g_{\Delta_5})=c_1(0)/2=5,
  \qquad
  \kappa_{\mathrm{fiber}}(K3)=24.
\]
The Hall-Drinfeld/Borcherds branch attached to \(K3\times E\) uses the
completed double
\[
  \mathcal D_\hbar\!\left(
    \CoHA^{\mathrm{or}}_{\mathrm{crit}}(K3\times E)
  \right)^{\wedge},
\]
together with orientation, compact-support, Hopf-pairing, completion,
and Borcherds-product comparison data. The Mukai self-mirror K3
Yangian branch uses the K3 Mukai lattice and stable-envelope Yangian
presentation. The branches share Mukai input and occupy separate
construction slots.
\end{proposition}

\begin{proof}
The first equality is Kunneth multiplicativity for
\(\chi(\cO_X)\), since \(\chi(\cO_{K3})=2\) and
\(\chi(\cO_E)=0\). The compact chiral value is the Hodge supertrace of
the native \(\Eone\) output on the total CY$_3\). The value \(3\) comes
from the Heisenberg-Mukai specialisation on the elliptic direction, the

\end{proof}

\section{Final Formulation}

\begin{theorem}[Gauge data as framed input]
\label{thm:gauge-data-framed-input}
Let \(X\) be a smooth complex CY$_3$ target and
\(\cC=\Perf(X)\) with its oriented CY$_3$ class. The hCS colour datum
enters by two procedures with a comparison on their intersection:
\begin{enumerate}[label=\textup{(\roman*)}]
\item choose an external principal \(G\)-bundle \(P\), a finite
Lie algebra \(\fg\), and a trace representation \(\rho\), giving
\(\Omega^{0,\bullet}(X,\operatorname{ad}P)[1]\);
\item choose a geometric brane framing \(E\in\Perf(X)\), identify
\(\REnd(E)\) with a Dolbeault algebra
\(\Omega^{0,\bullet}(X,\End E)\), and read the fibre coefficient from
a local trivialisation.
\end{enumerate}
On the common locus, \(P\) is the frame bundle or a reductive reduction
of \(E\), \(G\) acts through \(\rho\subset\End(E)\), and
\(\operatorname{ad}P\) maps to \(\End(E)\) compatibly with the trace.
The structures \(\Aut_\infty(\cC)\), \(\Fr(\cC)\),
\(\Mor_{\mathrm{gen}}(\cC)\), the Deligne gauge 2-groupoid
\(\Del_P\), the derived quotient \(\cM^{0,1}_G(X)_P\), the BRST
complex, the BV field theory, and the finite-dimensional gauge group
\(G\) have separate functorial homes. The categorical symmetries act on
framed objects, framings parametrise branes, Morita bimodules compare
compact-generator presentations, the derived quotient records gauge
orbits, BRST gives functions on that quotient, BV supplies the shifted
field-antifield theory, and \(G\) is field-theoretic colour input.
Compact nonabelian quantisation adds the package of
Hypothesis~\ref{hyp:compact-hcs-package}. At \(d=3\), the specialised
CY-to-chiral output is native \(\Eone\); \(\Etwo\) belongs to the
centre, double, or module layer.
\end{theorem}

\begin{proof}
Definitions~\ref{def:hcs-coefficient} and~\ref{def:framed-cy-category}
give the two inputs, and
Definition~\ref{def:gauge-2groupoid-derived-stack} supplies the
derived gauge stack attached to the first input.
Definition~\ref{def:brst-bv-packages} separates functions on the gauge
quotient from the shifted cotangent BV theory.
Proposition~\ref{prop:exact-functors} separates the functorial homes of
autoequivalences, framed objects, endomorphism algebras, Morita
presentations, and finite gauge coefficients. The \(\C^3\) calculation
of Theorem~\ref{thm:c3-rank-framings}
shows that rank framings may be Morita equivalent as presentations
while remaining separated as framed objects and as finite hCS
coefficients. Propositions~\ref{prop:qme-obstruction} and
\ref{prop:c2-counterterm} separate the quartic anomaly from the
degree-two kinetic counterterm. The compact package, the \(d=3\)

\end{proof}
