\author{Raeez Lorgat}

\begin{abstract}
Humbert surfaces live in the Siegel threefold \(\cA_2\).  The period
domain of \(S^{[2]}=\Hilb^2(S)\), for a K3 surface \(S\), is governed by
the Beauville-Bogomolov lattice
\[
  \Lambda_{K3^{[2]}}
  =
  U^{\oplus 3}\oplus E_8(-1)^{\oplus 2}\oplus\langle -2\rangle,
  \qquad
  \operatorname{sign}\Lambda_{K3^{[2]}}=(3,20).
\]
After a primitive polarization is fixed, the relevant orthogonal
complement has signature \((2,20)\), while the Siegel genus-two domain is
the accidental \(\Sp_4\simeq\operatorname{Spin}(3,2)\) type-IV domain of
signature \((2,3)\).  A Humbert divisor therefore reaches
\(\Hilb^2(K3)\) only through specified Kummer, Shioda-Inose, and
Noether-Lefschetz data.  The \(K3\times E\) BKM denominator, the
class-\(\cS\) Gaiotto curve, the Siegel form \(\Dfive\), and the
Mukai-Yangian branch are separate objects connected by named comparison
maps.  The primitive normalizations are
\[
  D_5=64^{-1}\Dfive,\qquad
  c_1(0)=10,\qquad
  \kBKM(\Dfive)=5,\qquad
  \PhiTen^{\mathrm{un}}=\Dfive^2.
\]
\end{abstract}

\section{Humbert Divisors in \texorpdfstring{\(\cA_2\)}{A2}}

Let
\[
  Z=\begin{pmatrix}\tau&z\\ z&\sigma\end{pmatrix}\in\HH_2,
  \qquad
  \cA_2=\Sp_4(\Z)\backslash\HH_2.
\]
For a positive integer \(D\equiv 0,1\pmod 4\), the Humbert surface
\(H_D\subset\cA_2\) is the image of the union of rational quadratic
divisors
\[
  a\tau+bz+c\sigma+d(z^2-\tau\sigma)+e=0,
  \qquad
  b^2-4ac-4de=D,
\]
where \((a,b,c,d,e)\in\Z^5\) is primitive.  The discriminant \(D\)
belongs to the Humbert surface.  It is distinct from the paramodular
index \(N\) in the BKM/CHL ladder unless a row map explicitly identifies
them.

\begin{proposition}[The diagonal Humbert divisor]
\label{prop:humbert-one}
The divisor \(H_1\) is the \(\Sp_4(\Z)\)-orbit of the diagonal locus
\[
  \cH_{\mathrm{diag}}
  =
  \left\{
    \begin{pmatrix}a&0\\0&b\end{pmatrix}
    : a,b\in\HH_1
  \right\}
  \subset\HH_2.
\]
It is the decomposable principally polarized abelian surface locus
\((E_1\times E_2,\lambda_1\boxplus\lambda_2)\).  The Humbert surface
\(H_4\) is the surface of discriminant \(4\), defined by the same
equations with \(b^2-4ac-4de=4\).
\end{proposition}

\begin{proof}
Taking \((a,b,c,d,e)=(0,1,0,0,0)\) gives the equation \(z=0\) and
discriminant \(1\).  Its \(\Sp_4(\Z)\)-translates are exactly the
decomposable locus in \(\cA_2\).  The formula above is Humbert's
definition for every admissible discriminant.
\end{proof}

\begin{proposition}[Igusa square-root normalization]
\label{prop:delta5-normalization}
In the theta normalization,
\[
  \Dfive\in M_5(\Sp_4(\Z),\nu_{\Dfive}),
  \qquad
  [q^{1/2}r^{1/2}s^{1/2}]\Dfive=64.
\]
The monic Borcherds product is
\[
  D_5=64^{-1}\Dfive,
  \qquad
  \operatorname{den}(\fg_{\Dfive})=D_5(2Z).
\]
The divisor is
\[
  \operatorname{supp}\Div(\Dfive)=H_1,
  \qquad
  \operatorname{mult}_{H_1}(\Dfive)=1.
\]
The scalar square normalizations are
\[
  \PhiTen^{\mathrm{un}}=\Dfive^2,
  \qquad
  \PhiTen^{\mathrm{OP}}=D_5^2=4096^{-1}\Dfive^2.
\]
\end{proposition}

\begin{proof}
Gritsenko and Nikulin identify the Borcherds product of the
Eichler-Zagier generator \(\phi_{0,1}=y^{-1}+10+y+O(q)\) with the
weight-five Igusa square root.  Borcherds' weight formula gives
\[
  \wt(\Borch(\phi_{0,1}))=\frac{[q^0y^0]\phi_{0,1}}{2}
  =\frac{10}{2}=5.
\]
The product theorem fixes the leading coefficient \(64\), hence the
monic product \(D_5=64^{-1}\Dfive\).  Clery and Gritsenko identify the
divisor of \(\Dfive\) with the diagonal Humbert divisor with
multiplicity one.  Squaring gives the scalar Igusa form of weight ten,
and the Oberdieck-Pandharipande convention squares the monic product.
\end{proof}

\section{\texorpdfstring{\(\Hilb^2(K3)\)}{Hilb2(K3)} Periods}

Let \(S\) be a projective K3 surface.  Beauville's form gives
\[
  H^2(S^{[2]},\Z)
  \cong
  H^2(S,\Z)\oplus\Z\delta,
  \qquad
  (\delta,\delta)=-2.
\]
Thus
\[
  \Lambda_{K3^{[2]}}
  \cong
  U^{\oplus 3}\oplus E_8(-1)^{\oplus 2}\oplus\langle -2\rangle,
  \qquad
  \operatorname{sign}(\Lambda_{K3^{[2]}})=(3,20).
\]
The marked period domain is
\[
  \Omega_{\Lambda}
  =
  \bigl\{[\omega]\in\PP(\Lambda_{K3^{[2]}}\otimes\C):
  (\omega,\omega)=0,\;(\omega,\bar\omega)>0\bigr\}.
\]
It has complex dimension \(21\).  For a primitive polarization
\(h\in\Lambda_{K3^{[2]}}\) with \(h^2>0\), the polarized period domain
lies in \(h^\perp\), whose signature is \((2,20)\).

\begin{proposition}[Typed pullback criterion from \(\cA_2\)]
\label{prop:typed-humbert-pullback}
A divisor \(H_D\subset\cA_2\) determines a divisor on a
\(K3^{[2]}\)-type moduli space only after the following data have been
fixed:
\[
  L_{\mathrm{Siegel}}\hookrightarrow\Lambda_{K3^{[2]}},
  \qquad
  \operatorname{sign}(L_{\mathrm{Siegel}})=(2,3),
\]
a compatible arithmetic subgroup, and a Noether-Lefschetz condition on
the orthogonal complement.  The accidental isomorphism
\(\Sp_4\simeq\operatorname{Spin}(3,2)\) controls this rank-five
Siegel lattice; its action is confined to the rank-five summand, with
the full rank-twenty-three Beauville-Bogomolov lattice outside its
domain.
\end{proposition}

\begin{proof}
The Siegel space \(\HH_2\) is the symmetric domain for a rank-five
lattice of signature \((2,3)\).  The full \(K3^{[2]}\) lattice has
signature \((3,20)\), and a polarized \(K3^{[2]}\) period domain uses a
rank-twenty-two lattice of signature \((2,20)\).  A morphism of
orthogonal locally symmetric varieties requires an isometric embedding
of lattices together with an arithmetic subgroup compatibility.  A
Humbert equation records the discriminant divisor; the embedding and
arithmetic subgroup are additional inputs.
\end{proof}

\section{Kummer and Shioda-Inose Comparison}

Let \(A\) be a principally polarized abelian surface.  Its Kummer K3
surface is the minimal resolution
\[
  \Km(A)\longrightarrow A/\{\pm1\}.
\]
The sixteen exceptional curves form the Kummer lattice.  The
transcendental lattice satisfies
\[
  T_{\Km(A)}\cong T_A(2).
\]
A Shioda-Inose K3 surface \(S_A\) attached to \(A\) carries a Nikulin
involution \(\iota\), a rational degree-two map
\[
  S_A\dashrightarrow\Km(A),
\]
and
\[
  T_{S_A}\cong T_A.
\]

\begin{definition}[Humbert-Kummer locus]
\label{def:humbert-kummer-locus}
For \(D\equiv0,1\pmod4\), define \(\cK_D\) to be the locus of Kummer
and Shioda-Inose K3 surfaces obtained from principally polarized
abelian surfaces \(A\in H_D\).  Equivalently, \(\cK_D\) is the
Noether-Lefschetz locus cut out by the extra algebraic class induced by
the discriminant-\(D\) endomorphism of \(A\), transported through the
Kummer lattice and the Shioda-Inose correspondence.
\end{definition}

\begin{proposition}[The correct Humbert route to K3 geometry]
\label{prop:humbert-route}
The comparison from a Humbert surface to K3 geometry factors as
\[
  H_D\subset\cA_2
  \xrightarrow{\ A\mapsto\Km(A)\ }
  \cM_{K3}^{\mathrm{Kum}}
  \xleftarrow{\ S_A\dashrightarrow\Km(A)\ }
  \cM_{K3}^{\mathrm{SI}}.
\]
Passing from \(S_A\) to \(S_A^{[2]}\) then uses the Beauville
construction \(S_A\mapsto\Hilb^2(S_A)\).  A cycle
\(\Sigma_D\subset S_A^{[2]}\) requires an algebraic correspondence or a
tautological Nakajima class in addition to the Humbert discriminant.
\end{proposition}

\begin{proof}
The Kummer construction and the Shioda-Inose correspondence are
classical.  They transport periods and lattice polarizations from
principally polarized abelian surfaces to K3 surfaces.  The Hilbert
scheme \(S_A^{[2]}\) is functorial in the surface after \(S_A\) has been
chosen.  The Humbert discriminant produces a Noether-Lefschetz
condition; selection of a canonical codimension-two cycle in the Chow
group of a fixed \(S^{[2]}\) requires an additional correspondence.
\end{proof}

\begin{problem}[Humbert-lifted Hilbert cycle]
\label{prob:humbert-hilbert-cycle}
Construct, for each \(D\) in a specified Humbert set, a cycle class
\[
  [\Sigma_D]\in\CH^2(S_A^{[2]})
\]
from the Kummer lattice, the Shioda-Inose correspondence, and the
Noether-Lefschetz class attached to \(H_D\), with cohomology class
lying in the image of the primitive embedding
\[
  L_{\mathrm{Siegel},D}\hookrightarrow
  H^2(S_A^{[2]},\Z)
\]
and with discriminant \(D\) computed in the Beauville-Bogomolov form.
The construction must specify the surface \(S_A\), the arithmetic group,
the product chamber, and the tautological operator producing the class.
\end{problem}

\section{\texorpdfstring{\(K3\times E\)}{K3xE} Comparison}

Let \(Y=S\times E\) with \(S\) a K3 surface and \(E\) an elliptic curve.
The Hilbert scheme \(S^{[2]}\) is a holomorphic symplectic fourfold.
The product \(Y\) is a Calabi-Yau threefold.  Their comparison runs
through the Mukai lattice and through Fock-space or Hall-algebra
functoriality.  A set-theoretic intersection is type-incoherent.

\begin{proposition}[Separated \(K3\times E\) invariants]
\label{prop:k3e-invariants}
For \(Y=K3\times E\),
\[
  h^{0,\bullet}(Y)=(1,1,1,1),
  \qquad
  \kch(Y)=1-1+1-1=0.
\]
Kunneth multiplicativity gives
\[
  \kcat(Y)=\chi(\cO_{K3})\chi(\cO_E)=2\cdot0=0.
\]
The Heisenberg-Mukai specialization and the BKM denominator give
\[
  \kHeis(Y)=3,
  \qquad
  \kBKM(\Dfive)=c_1(0)/2=5.
\]
The fibre invariant is
\[
  \kfib(Y)=\rk\widetilde H(K3,\Z)=1+22+1=24.
\]
These values arise from separate constructions:
\[
  \{\kcat(Y),\kch(Y),\kHeis(Y),
  \kBKM(\Dfive),\kfib(Y)\}
  =
  \{0,0,3,5,24\}.
\]
\end{proposition}

\begin{proof}
The compact Hodge column of \(K3\times E\) gives the alternating sum
displayed above.  The categorical Euler invariant is the Euler
characteristic of \(\cO_Y\) and is multiplicative on the product.  The
Heisenberg value \(3\) belongs to the specialisation
\(\SpCh_{K3,E}\circ\PhiFA_3\).  Proposition
\ref{prop:delta5-normalization} gives the BKM value \(5\).  The final
value is the Mukai-lattice rank of the K3 fibre.  Each equality uses a
different input object.
\end{proof}

\begin{proposition}[Primitive BKM/CHL denominator constants]
\label{prop:primitive-chl-constants}
On the primitive BKM/CHL denominator ladder
\[
  N\in\{1,2,3,4,6\},
\]
the Borcherds input constants and weights are
\[
\begin{array}{c|ccccc}
N&1&2&3&4&6\\
\hline
c_N(0)&10&8&6&4&2\\
\kBKM(\Phi_N)=c_N(0)/2&5&4&3&2&1
\end{array}
\]
The level-one entry is \(\Phi_1=\Dfive\).  The Gritsenko-Clery
diagonal-divisor catalogue is a different eight-row catalogue with
weights
\[
  (5,2,3,1,2,\tfrac12,\tfrac32,1)
\]
and row constants
\[
  (10,4,6,2,4,1,3,2).
\]
A comparison between the two lists requires a named row map and a
normalization.
\end{proposition}

\begin{proof}
Borcherds' singular-theta lift gives
\(\wt(\Phi_N)=c_N(0)/2\).  The primitive CHL inputs use the
\(g_N\)-twined K3 weak Jacobi forms in the normalization
\(\phi_{0,1}^{(g_N)}=\frac12 Z_{K3}^{(g_N)}\), whose zero Fourier
coefficients are \((10,8,6,4,2)\).  Gritsenko-Nikulin give the \(N=1\)
case, Eguchi-Hikami give the \(2A\) twining, and
Govindarajan-Krishna give the \(3A\), \(4A\), and \(6A\) cases.  The
Gritsenko-Clery list is indexed by \((t,N)\) and by multiplier systems,
so its row constants belong to a separate classification problem.
\end{proof}

\section{Class-\texorpdfstring{\(\cS\)}{S} and VOA Layer}

The four-dimensional class-\(\cS\) parent relevant to the \(K3\times E\)
Igusa corner is
\[
  \cT_{24}:=\cT[A_1,\Sigma_{0,24}],
\]
where \(\Sigma_{0,24}\) is a sphere with twenty-four maximal regular
\(\fsu(2)\)-punctures.  Its pants decomposition has \(22\) trinions and
\(21\) gauged \(SU(2)\) tubes, hence
\[
  (n_v,n_h)=22(0,4)+21(3,0)=(63,88).
\]
The Shapere-Tachikawa and Beem-Lemos-Liendo-Peelaers-Rastelli-van Rees
normalizations give
\[
  c_{4d}=\frac{2n_v+n_h}{12}=\frac{107}{6},
  \qquad
  c_{2d}=-12c_{4d}=-214.
\]
For the closed genus-two curve \(\Sigma_{2,0}\), the same pants count
has two trinions and three gauged tubes:
\[
  (n_v,n_h)=(9,8),
  \qquad
  c_{4d}=\frac{13}{6},
  \qquad
  c_{2d}=-26.
\]
The \(107/6\) parent is therefore the punctured sphere
\(\Sigma_{0,24}\).

\begin{proposition}[Typed Schur-Jacobi-Borcherds composite]
\label{prop:schur-jacobi-borcherds}
The comparison from the Gaiotto curve to the Igusa denominator has the
typed form
\[
  \cT[A_1,\Sigma_{0,24}]
  \xrightarrow{\ \chi\ }
  \chi(\cT_{24})
  \xrightarrow{\ \Pi_{M_{24}}\ }
  \phi_{0,1}^{K3}
  \xrightarrow{\ \Borch\ }
  \Dfive
  \rightsquigarrow
  \fg_{\Dfive}.
\]
Here \(\Pi_{M_{24}}\) is the Mathieu average on puncture-labelled Schur
data, \(\phi_{0,1}^{K3}=y^{-1}+10+y+O(q)\) is the
Gritsenko-Nikulin normalized Jacobi form, and
\(\fg_{\Dfive}\) is the generalized BKM superalgebra whose denominator
is \(D_5(2Z)\).
\end{proposition}

\section{Branch Separation}

For \(d\geq3\), the native CY-to-chiral output is \(E_1\)-chiral.  In
the \(K3\times E\) case,
\[
  A_Y=
  \SpCh_{K3,E}\bigl(\PhiFA_3(\Perf(K3\times E))\bigr)
  \in E_1\text{-}\ChirAlg(E).
\]
Higher braided structure belongs to named centre, double, or module
layers such as
\[
  Z^{\mathrm{der}}_{\mathrm{ch}}(A_Y),
  \qquad
  D(\CoHA(Y)),
  \qquad
  Z(\mathrm{Rep}(A_Y)),
  \qquad
  \chi(\cT_{24}).
\]

\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{lll}
\hline
Layer & Object & Controlling datum \\
\hline
Siegel divisor & \(H_D\subset\cA_2\) & discriminant \(D\) \\
Igusa denominator & \(D_5(2Z)\) & \(\Dfive=\Borch(\phi_{0,1})\) \\
BKM branch & \(\fg_{\Dfive}\) & Borcherds product and root multiplicities \\
Hall-Drinfeld branch & \(D(\CoHA(K3\times E))\) & compact Hall orientation data \\
Mukai-Yangian branch & local ADE/Kummer Yangian data & Maulik-Okounkov \(R\)-matrix \\
Class-\(\cS\) branch & \(\chi(\cT[A_1,\Sigma_{0,24}])\) & Schur VOA and \(M_{24}\)-average \\
Native CY output & \(A_Y\) & \(E_1\)-chiral specialisation \\
\hline
\end{tabular}
\end{center}

The table records distinct layers.  A theorem comparing two entries must
name the functor, the lattice, the chamber, the orientation character,
and the source of the product coefficients.

\section{Conclusions}

\begin{enumerate}[leftmargin=1.7em]
  \item Humbert surfaces \(H_D\) are divisors in \(\cA_2\) defined by
  discriminant \(D=b^2-4ac-4de\).  \(H_1\) is the diagonal decomposable
  locus; \(H_4\) is the discriminant-four Humbert surface.
  \item The divisor of \(\Dfive\) is \(H_1\) with multiplicity one.
  The monic denominator is \(D_5=64^{-1}\Dfive\), and the scalar square
  is \(\PhiTen^{\mathrm{un}}=\Dfive^2\).
  \item The period lattice of \(S^{[2]}\) has signature \((3,20)\);
  a polarized period domain uses signature \((2,20)\).  The Siegel
  Humbert domain uses signature \((2,3)\).
  \item Kummer and Shioda-Inose constructions provide the route from
  \(H_D\) to K3 Noether-Lefschetz geometry.  A canonical
  \(\Sigma_D\subset S^{[2]}\) is an additional cycle-construction
  problem.
  \item For \(K3\times E\),
  \[
    \kcat=0,\qquad
    \kch=0,\qquad
    \kHeis=3,\qquad
    \kBKM(\Dfive)=5,\qquad
    \kfib=24.
  \]
  These numbers obey separation by construction; a summation law is
  absent.
  \item The primitive BKM/CHL constants are
  \(c_N(0)=(10,8,6,4,2)\) and
  \(\kBKM(\Phi_N)=(5,4,3,2,1)\) for
  \(N\in\{1,2,3,4,6\}\).
  \item The class-\(\cS\) parent is
  \(T[A_1,\Sigma_{0,24}]\) with
  \((n_v,n_h)=(63,88)\), \(c_{4d}=107/6\), and \(c_{2d}=-214\).
  The closed genus-two curve \(\Sigma_{2,0}\) has \(c_{4d}=13/6\).
  \item At \(d\geq3\), the native \(\Phi\)-output is \(E_1\)-chiral.
  \(E_2\)-structures occur only on named centre, double, module, or VOA
  layers.
\end{enumerate}
