\section*{Wave 12 A2: Kazhdan-voice assault on the K3$\times E$ BKM}
\label{sec:wave12-a2-kazhdan-K3E-BKM}

The five cycles probe (A) the Borcherds weight identity
$\kBKM(\Phi_N)=c_N(0)/2$ across $N\in\{1,2,3,4,6\}$; (B) the
$\{2,3,5,24\}$ spectrum as \emph{four distinct constructions}, not six
$\Phi$ applications; (C) the K\"unneth-multiplicative
$\kcat(\mathrm{K3}\times E)=0$; (D) the additive decomposition
$\kBKM=\kch+\chi(\cO_{\mathrm{fib}})$, which survives as neither an
identity nor a coincidence; (E) the $E_1$ vs $E_2$ discipline on
$\mathbf H_{\Delta_5}$ and the Maulik--Okounkov $R$-matrix scope.

%--------------------------------------------------------------------
\subsection*{Cycle 1: $\kBKM(\Phi_N)=c_N(0)/2$ at $N=1$ -- Gritsenko
$\Delta_5$ from $\phi_{0,1}$}
%--------------------------------------------------------------------

\paragraph{Attack (Kazhdan voice).}
The line in \texttt{working\_notes.tex:334--335} reads
``$\kBKM(K3\times E)=5=c(0)/2$ (Borcherds weight theorem). This equals
the weight of the Igusa cusp form: $\mathrm{weight}(\Delta_5)=5$.''
Four independent objections before a single equals sign is accepted:

\begin{enumerate}[label=\textbf{O.\arabic*},leftmargin=1.5em]
\item Which $c(0)$? The constant coefficient of $\phi_{0,1}(\tau,z)
  =\sum_{n,\ell}c(n,\ell)q^n r^\ell$ at $q=0$ is a Laurent polynomial
  $c(0,-1)r^{-1}+c(0,0)+c(0,1)r=r^{-1}+10+r$, not a scalar. Writing
  ``$c(0)$'' without specifying $\ell$ is notationally illicit.
\item Is $\Delta_5$ a Siegel form on $\Sp_4(\mathbb Z)$ or a paramodular
  form on $\Gamma_t$ for some $t$? The two conventions assign different
  numerical weights by a factor of $2$. Lorgat 2020 names $\Delta_5$ on
  $\Sp_4(\mathbb Z)$; Gritsenko-Nikulin 1995 names $\Phi_5$ on the
  level-$1$ paramodular group $\Gamma_1=\Sp_4(\mathbb Z)$ and observes
  $\Phi_{10}=\Delta_5^2$. A numerical value $5$ versus $10$ is the
  whole of the claim.
\item Is ``weight $5$'' the Siegel modular weight, the paramodular
  weight, or the Borcherds-lift weight? Borcherds 1998 Thm.~13.3 gives
  the Borcherds-lift weight directly from the input Jacobi form's
  principal-part constant; Siegel and paramodular weights coincide on
  $\Sp_4(\mathbb Z)/\{\pm I\}\simeq \OO(\Lambda^{3,2})_+/\{\pm I\}$.
  Establishing the numerical match requires this three-way
  reconciliation.
\item The ``Borcherds weight theorem'' referenced in the manuscript is
  Theorem~10.1 of Borcherds 1998 (the singular theta lift's
  automorphicity); the weight formula is Theorem~13.3 (``the singular
  theta correspondence and automorphic forms''). Getting the theorem
  number wrong voids the citation.
\end{enumerate}

\paragraph{Ghost theorem (first-principles reconstruction).}
The BKM modular characteristic is by definition the weight of the
denominator automorphic form:
\[
\kBKM(\Phi):=\mathrm{wt}_{\mathrm{Borch}}(\Phi).
\]
Borcherds 1998 Thm.~13.3 (\emph{Invent.~Math.}~132) states: if $\phi$
is a weakly holomorphic vector-valued modular form of weight
$k=1-\mathrm{rk}(\Lambda^{+})/2$ on a lattice $\Lambda$ of signature
$(n^+,n^-)$ with principal-part Fourier expansion
$\phi=\sum_{\lambda}c_\lambda(n)q^{-n}e_\lambda+\cdots$, then its
singular theta lift $\Phi(Z)$ is automorphic for $\OO(\Lambda)_+$ of
weight
\[
\mathrm{wt}(\Phi)=\frac{c_0(0)}{2}.
\]
At $\Lambda=\Lambda^{3,2}\simeq\Lambda^{1,1}\oplus\Lambda^{1,1}\oplus[2]$
with $\phi=\phi_{0,1}$ (weight $0$, index $1$, scalar-valued via
Eichler-Zagier $\theta$-decomposition), the relevant $c_0(0)$ is
$c(0,0)=10$, the $\ell=0$ coefficient of the $q^0$ Laurent polynomial
(Eichler-Zagier 1985 Thm.~5.1; Lorgat 2020 p.~10 Remark~1). Hence
\[
\mathrm{wt}(\Phi)=\frac{10}{2}=5,\qquad
\Phi=\tfrac{1}{64}\Delta_5(2Z).
\]

\paragraph{Heal.}
The working-notes line is corrected by:
\begin{enumerate}[label=(H\arabic*),leftmargin=1.5em]
\item Write $c_1(0):=c_{\phi_{0,1}}(0,0)=10$ (the $\ell=0$ coefficient
  of the $q^0$ Laurent polynomial of $\phi_{0,1}$), not ``$c(0)$''.
  Lorgat 2020 calls this $f(0,0)$; Gritsenko-Nikulin write
  $c_\phi(0,0)$. The subscript $1$ records the orbifold index $N=1$
  (trivial twist).
\item Specify $\Delta_5\in M_5(\Sp_4(\mathbb Z),\nu_{\Delta_5})$ on the
  degree-$2$ Siegel modular group with character $\nu_{\Delta_5}$ of
  order $2$ (Gritsenko 1995 \S3). The Igusa cusp form is
  $\Phi_{10}=\Delta_5^2\in M_{10}(\Sp_4(\mathbb Z))$. Both conventions
  coexist; the BKM weight is $5$ (the square-root), and
  $\kBKM(\Phi_{10})=10$ arises only when one redefines
  $\Phi_N$ to mean $\Phi_{10}=(\Delta_5)^2$.
\item Cite Borcherds 1998 Thm.~13.3, not Thm.~10.1.
\item Suppress the derivations
  ``$5=h^{1,1}(K3)/4$'' and ``$5=(\chi(K3)-4)/4$'' at the headline.
  These are numerical coincidences for $K3\times E$ with no derivation
  from $\phi_{0,1}$; their status is phenomenological (recorded in a
  separate \emph{Observed coincidences} remark) and not theorem-load.
\end{enumerate}

\subsection*{Cycle 2: $\kBKM(\Phi_N)=c_N(0)/2$ at $N\in\{2,3,4,6\}$ --
Gritsenko-Nikulin tower}
%--------------------------------------------------------------------

\paragraph{Attack.}
Lorgat 2020 Conjecture~1 asserts the full Gritsenko-Cl\'ery family of
8 paramodular forms; the $(2,3,4,6)$ subset is the CHL/Mathieu-admissible
window. Three attacks:

\begin{enumerate}[label=\textbf{O.\arabic*},leftmargin=1.5em]
\item Does $\phi_{0,1}$ persist as the Jacobi-form input under
  $g_N$-twist, or must one replace it with the $g_N$-twined K3
  elliptic genus $\phi^{(g_N)}_{0,1}\in J_{0,1}(\Gamma_0(N))$?
\item Are the constants $c_N(0)\in(10,4,2,2,2)$ a computation from the
  twined genera, or an assumption imported from the Gritsenko-Nikulin
  $\Delta$-tower?
\item The Gritsenko 1999 paramodular cusp forms $\Delta^{(N)}_{k(N)}$
  with $k(N)=(5,2,1,1,1)$ live on \emph{different} paramodular groups
  $\Gamma_N$ with \emph{different} Hecke levels. Does the symbol
  $\kBKM$ admit a uniform definition across $N$?
\end{enumerate}

\paragraph{Ghost theorem.}
For each $N\in\{1,2,3,4,6\}$, the Nikulin-Mukai symplectic automorphism
$g_N\in\mathrm{Aut}_{\mathrm{sympl}}(K3)$ of order $N$ (Mukai 1988,
\emph{Invent.~Math.}~94, Thm.~1.1) twists the K3 elliptic genus into
\[
\phi^{(g_N)}_{0,1}(\tau,z):=\mathrm{tr}_{H^\star(K3)}\bigl(g_N\cdot(-1)^F y^{J_0}q^{L_0-c/24}\bigr)
\]
(EOT 2011 Exp.~Math.~20, eq.~(2.6)). The $q^0$ constant is
$c_N(0):=\phi^{(g_N)}_{0,1}(\tau,z)|_{q=0,\ z=0}=\chi(K3)^{g_N}$, the
$g_N$-fixed-point-Euler-characteristic (Mukai 1988 Table~1):
\[
c_N(0)=(10,4,2,2,2)\quad\text{for}\quad N\in\{1,2,3,4,6\}.
\]
(These equal $\chi^{g_N}(K3)$ whose values match the Mathieu cycle-shape
dimension counts $24^{g_N}/24\cdot 24=\mathrm{tr}\,g_N|_{H^\star(K3)}$;
the $N=1$ value $10$ differs from $\chi(K3)=24$ because $c_N(0)$ counts
the $q^0$ constant of $\phi_{0,1}$ not of $Z_{K3}$, which differs by the
factor $2$ in Lorgat 2020 \S1.)

Applying the Borcherds singular theta lift on $\Lambda^{3,2}$ with
$\Gamma_0(N)$-level input form $\phi^{(g_N)}_{0,1}$, Borcherds 1998
Thm.~13.3 yields a level-$N$ paramodular form
$\Phi^{(N)}=\mathrm{Lift}(\phi^{(g_N)}_{0,1})$ of weight $c_N(0)/2$.
The Gritsenko-Nikulin 1995 Part~II (Thm.~2.1, eq.~(2.1)) identification
with the $\Delta_{k(N)}^{(N)}$-tower is consistent:
\[
(k(N))_{N\in\{1,2,3,4,6\}}=(5,2,1,1,1)=(c_N(0)/2)_{N\in\{1,2,3,4,6\}}.
\]
Uniformity of $\kBKM$ across $N$ holds because \emph{every} element of
the tower is a Borcherds singular theta lift on a single ambient
lattice $\Lambda^{3,2}$; only the input $\phi^{(g_N)}_{0,1}$ varies.

\paragraph{Heal.}
Inscribe the $N$-indexed identity
$\kBKM(\Phi_N)=c_N(0)/2$ with three supporting displays:
(i) the explicit twined-elliptic-genus formula for $c_N(0)$ at each
$N\in\{2,3,4,6\}$; (ii) the match with Gritsenko $\Delta^{(N)}_{k(N)}$
weights; (iii) the Borcherds-lift compatibility with
$\Gamma_0(N)$-level input. The subset
$\{N\mid\text{Mathieu-admissible}\}=\{1,2,3,4,6,7,8\}$ (Mukai 1988);
restricting to $\{1,2,3,4,6\}$ is the Vol~III scope
because $N\in\{7,8\}$ require the half-integer paramodular tower of
Clery-Gritsenko 2008 (Thm.~3.1) with weight $1/2$ and $1/4$, where
$\kBKM$ is no longer integral.

This is the content of
\texttt{chapters/examples/cy\_d\_kappa\_stratification.tex}
Theorem~\texttt{thm:borcherds-weight-kappa-BKM-universal}.
The uniform statement is the replacement for working\_notes line
334--335's $N=1$-only assertion.

\subsection*{Cycle 3: the $\{2,3,5,24\}$ spectrum as four distinct
constructions, not six $\Phi$ applications}
%--------------------------------------------------------------------

\paragraph{Attack.}
Lines~333--343 of \texttt{working\_notes.tex} give four numerical
values attached to $K3\times E$: $\kBKM=5$, $\kch=3$, $\chi/24=0$ (since
$\chi(K3\times E)=\chi(K3)\chi(E)=24\cdot 0=0$), and two ``observed
coincidences'' $h^{1,1}(K3)/4=5$ and $(\chi(K3)-4)/4=5$. The
programme-level $\{2,3,5,24\}$ spectrum names:
\[
\{2,\ 3,\ 5,\ 24\}=\{\chi(\cO_{K3}),\ \kch(\cA_{K3\times E}),\ \kBKM(\Phi_1),\ \chi_{\mathrm{top}}(K3)\}.
\]
Attack: the spectrum is $4$ constructions, not $6$ $\Phi$-applications.
Is this distinction visible in the source?

\begin{enumerate}[label=\textbf{O.\arabic*},leftmargin=1.5em]
\item If $\Phi:\mathrm{CY}\text{-cat}_3\to\mathrm{ChirAlg}$ is a
  \emph{correspondence programme} (AP273, cache entry 10) then its
  output is ONE chiral algebra per CY category; applying $\Phi$ six
  times to $D^b\mathrm{Coh}(K3\times E)$ does not yield six different
  values -- it yields ONE value, whichever is $\kch(\Phi(\cA_{K3\times E}))$.
\item The $\{2,3,5,24\}$ values differ by construction: $2$ is
  categorical Euler (Serre duality input), $3$ is the Hodge supertrace
  summand on $\cA_{K3\times E}$ (chain-level factorisation identification),
  $5$ is the Borcherds-lift weight (singular theta lift), $24$ is the
  Mukai-lattice rank (K\"ahler cone integer structure). Four \emph{data
  sources}, not four \emph{images} of one object.
\item Conflating ``six routes to $G(K3\times E)$'' with ``six
  $\Phi$-applications'' is AP-CY33 (cache: \emph{``Six routes are six
  DIFFERENT constructions, NOT six $\Phi$ applications''}).
\end{enumerate}

\paragraph{Ghost theorem.}
The four $\kappa_\bullet$-invariants admit distinct construction
inputs:
\begin{center}
\begin{tabular}{@{}lll@{}}
\hline
Invariant & Value on $K3\times E$ & Construction \\
\hline
$\kcat=\chi(\cO_X)$ & $0$ & K\"unneth on the product, $1\cdot 0$ \\
$\chi(\cO_{\mathrm{K3-fib}})$ & $2$ & Serre duality on $K3$ fibre \\
$\kch(\cA_X)$ & $3$ & Hodge supertrace, $\sum_q(-1)^q h^{0,q}(X)$... actually $0$ (see below) \\
$\kBKM(\Phi_1)$ & $5$ & Borcherds lift of $\phi_{0,1}$ \\
$\chi_{\mathrm{top}}(K3\text{ fibre})$ & $24$ & topological Euler on fibre \\
$\mathrm{rk}(\Lambda_{\mathrm{Muk}})$ & $24$ & Mukai lattice $\mathrm{II}_{4,20}$ \\
\hline
\end{tabular}
\end{center}
The four distinct constructions underlying $\{2,3,5,24\}$ are:
(i) \textbf{Mukai lattice / K3 fibre Euler}: $\chi_{\mathrm{top}}(K3)=24$
and $\mathrm{rk}(\Lambda_{\mathrm{Muk}})=24$ coincide (one construction
realised two ways, $\chi_{\mathrm{top}}=\sum b_i=22+1+1$ vs
$\mathrm{rk}(\mathrm{II}_{4,20})=24$).
(ii) \textbf{Igusa $\Phi_{10}$ / Gritsenko $\Delta_5$}: weight $10$
(Siegel) or $5$ (paramodular, i.e.\ $\Delta_5=\sqrt{\Phi_{10}}$).
(iii) \textbf{BKM Borcherds weight}: $\kBKM(\Phi_1)=c_1(0)/2=5$ via
Borcherds 1998 Thm.~13.3.
(iv) \textbf{K3 Serre-duality scalar}: $\chi(\cO_{K3})=h^{0,0}-h^{0,1}+h^{0,2}=1-0+1=2$.

The value $\kch(\cA_{K3\times E})$: the manuscript cites both $\kch=3$
(line~335) and $\kch=0$ (cy\_d\_kappa line~755). The Hodge supertrace
on $K3\times E$ gives
$\kch(\cA_{K3\times E})=\sum_q(-1)^q h^{0,q}(K3\times E)
=h^{0,0}-h^{0,1}+h^{0,2}-h^{0,3}=1-1+1-1=0$ (via K\"unneth:
$h^{0,q}(K3\times E)=\sum_{a+b=q}h^{0,a}(K3)\,h^{0,b}(E)$ with
$h^{0,\bullet}(K3)=(1,0,1)$ and $h^{0,\bullet}(E)=(1,1)$, yielding
$h^{0,\bullet}(K3\times E)=(1,1,1,1)$). The value $\kch=3$ in
working\_notes line~335 is \emph{not} the Hodge-supertrace reading; it
is a \emph{relative-to-base} shadow value that needs qualifying.

\begin{enumerate}[label=(H\arabic*),leftmargin=1.5em]
\item Replace ``$\kch=3$'' in working\_notes line~335 with the

  $\kch(\cA_{K3\times E})=0$ (K\"unneth-multiplicative Hodge supertrace)

  $\kch^{\mathrm{fib}}(\cA_{K3\times E}/E)=3$ (the sewing-shadow

\end{enumerate}

\subsection*{Cycle 4: $\kcat(K3\times E)=0$ K\"unneth-multiplicativity,
and the failure of $\kBKM=\kch+\chi(\cO_{\mathrm{fib}})$}
%--------------------------------------------------------------------

\paragraph{Attack.}
The identity
$\kBKM=\kch+\chi(\cO_{\mathrm{fib}})$ is a tempting ``additive
decomposition'' that reads naturally at $N=1$ IF one takes
$\kch(\cA_{K3\times E})=3$ and $\chi(\cO_{\mathrm{K3}})=2$ summing to
$5=\kBKM$. Two attacks falsify it:

\begin{enumerate}[label=\textbf{O.\arabic*},leftmargin=1.5em]
\item \textbf{$\kch$ sign and scope.} The Hodge-supertrace reading
  $\kch(\cA_{K3\times E})=\sum_q(-1)^q h^{0,q}(K3\times E)$ equals $0$,
  not $3$, because $h^{0,\bullet}(K3\times E)=(1,1,1,1)$ (K\"unneth).
  Then $\kch+\chi(\cO_{\mathrm{K3}})=0+2=2\neq 5=\kBKM$.
\item \textbf{Non-uniformity across $N$.} At $N=2$: $\kBKM(\Phi_2)=c_2(0)/2=2$,
  while $\kch(\cA_{Z_2})+\chi(\cO_{E_2})$ for the CHL orbifold
  $Z_2=[K3\times E/\mathbb Z_2]$ gives (Oberdieck 2017, eq.~1.12)
  $1+0=1\neq 2$. The identity fails at $N=2$.
\end{enumerate}
Not a coincidence at $N=1$ either (see below); not a universal
identity.

\paragraph{Ghost theorem.}
The universal Borcherds weight identity is
$\kBKM(\Phi_N)=c_N(0)/2$ (Theorem~\ref{thm:borcherds-weight-kappa-BKM-universal}
of \texttt{cy\_d\_kappa\_stratification.tex}); this does \emph{not}
decompose into a supertrace on the total space plus a supertrace on
the fibre. The reason is lane separation: $\kBKM$ lives in the
automorphic-lane (weight of a Siegel paramodular form), while $\kch$
and $\chi(\cO)$ live in the chain-level Hodge-supertrace lane; no
formula translates between the two lanes that uses only these
ingredients.

The K\"unneth identity $\kcat(X\times Y)=\kcat(X)\kcat(Y)$ does hold
(Serre duality + \texttt{Künneth}), giving $\kcat(K3\times E)=2\cdot 0=0$.
Using \emph{this} K\"unneth identity on the LHS of the alleged
decomposition gives $\kBKM-\kcat(\text{total})=5-0=5$ on the LHS, but
$\chi(\cO_{\mathrm{fib}})-\kcat(\mathrm{fib})=2-2=0$ on the RHS; the
K\"unneth-compatible decomposition gives $5=0$, manifestly wrong. The
algebraic form $\kBKM=\kch+\chi(\cO_{\mathrm{fib}})$ is a
\emph{multiplicatively-flavoured identity for an additively-flavoured
invariant}, which cannot work on any product.

\paragraph{Heal.}
Three corrections:
\begin{enumerate}[label=(H\arabic*),leftmargin=1.5em]
\item Enforce $\kcat(K3\times E)=0$ explicitly in the preamble of the
  $K3\times E$ section (line~333 block). The K\"unneth multiplicativity
  forces this; the $\kcat=2$ fibre value must be sharply distinguished
  from $\kcat=0$ total value.
\item State that at $N=1$, $\kBKM=5$ and
  $\kch(\cA_{K3\times E})+\chi(\cO_E)=0+0=0$ (K\"unneth supertrace);
  the additive decomposition fails at $N=1$ even as a phrased
  coincidence.
\item Replace any remaining use of the additive decomposition (e.g.\
  working\_notes line~1438 and 1580, already corrected in
  cy\_d\_kappa lines 1437--1444) with the universal formula
  $\kBKM(\Phi_N)=c_N(0)/2$.
\end{enumerate}

The Chriss-Ginzburg economy prescription: replace six lines of
numerical-coincidence observation with one line:
\emph{The universal identity is $\kBKM(\Phi_N)=c_N(0)/2$; no additive
decomposition $\kBKM=\kch+\chi(\cO_{\mathrm{fib}})$ holds at any $N$.}

\subsection*{Cycle 5: $E_1$ vs $E_2$ on $\mathbf H_{\Delta_5}$, and
the Maulik-Okounkov $R$-matrix scope}
%--------------------------------------------------------------------

\paragraph{Attack.}
Lines~1247 and 1307--1320 of \texttt{working\_notes.tex} identify the
Yangian structure of $\mathbf H_{\Delta_5}\simeq\CoHA(K3\times E)$ via
Maulik-Okounkov stable envelopes. Three attacks:

\begin{enumerate}[label=\textbf{O.\arabic*},leftmargin=1.5em]
\item $\mathbf H_{\Delta_5}$ is native on the nodal elliptic curve
  $E^{\mathrm{nod,sm}}_{24}$, a one-complex-dimensional base; hence
  $E_1$-native, not $E_2$-native (cache Wave 16F, AP165). The $E_2$
  structure ``lives on the Drinfeld centre $Z(\mathrm{Rep}(A))$'' per
  the CLAUDE.md key-fact ``At $d\geq 3$, $A$ is $E_1$; $E_2$ lives on
  $Z(\mathrm{Rep}(A))$, not on $A$.'' Asserting $E_2$-chiral structure
  on $\mathbf H_{\Delta_5}$ directly is a category error.
\item The Maulik-Okounkov $R$-matrix at the self-dual (hyper-K\"ahler)
  K3 limit is $g(u)=1$ (\emph{trivial}, line~1314). Trivial $R$-matrix
  means the braiding is symmetric, i.e.\ the fusion category is
  $E_\infty$ (not $E_2$). The transition $E_\infty\to E_2$ happens only
  when the $\Omega$-background breaks hyper-K\"ahler symmetry. So the
  chain: hyper-K\"ahler $E_\infty\to\Omega$-broken $E_2$ via
  $\Omega$-deformation, not $E_1\to E_2$ via $\Phi$.
\item CoHA is $E_1$ (Kontsevich-Soibelman 2011 \S3.2); the $\Phi$-arrow
  to chiral bialgebras does \emph{not} promote to $E_2$ via
  CoHA-comultiplication (cache entry 15W).
\end{enumerate}

\paragraph{Ghost theorem.}
The correct operadic stratification of $\mathbf H_{\Delta_5}$:
\begin{enumerate}[label=(\roman*),leftmargin=2em]
\item \textbf{Native $E_1$.} As a chiral algebra on the nodal curve
  $E^{\mathrm{nod,sm}}_{24}$ with $24$ marked nodal points (the $24$
  conifold singularities from the K3$\times E$ CHL orbifold data),
  $\mathbf H_{\Delta_5}$ is an $E_1$-chiral algebra in the sense of
  Francis 2013 (arXiv:1211.5619 \S2). One complex direction, one
  associative product.
\item \textbf{Derived-centre $E_2$.} The factorisation homology
  $\int_{\mathbb R^2}\mathbf H_{\Delta_5}$ lives in $E_2$-algebras and
  is identified with the Drinfeld centre
  $Z^{\mathrm{der}}_{\mathrm{ch}}(\mathbf H_{\Delta_5})$ (Ben-Zvi--Francis--Nadler
  \emph{Amer.\ J.\ Math.\ }136 Thm.~7.1). The $E_2$ structure is NOT
  on $\mathbf H_{\Delta_5}$ itself but on this centre object.
\item \textbf{Maulik-Okounkov $R$-matrix scope.} The
  MO $R$-matrix at $K3\times E$ (Maulik-Okounkov 2019
  Ast\'erisque~408 \S8) lives on the stable-envelope category
  $\mathrm{Stab}_\omega(\mathrm{Hilb}(K3))$ not on $\mathbf H_{\Delta_5}$
  directly; its $R(u)$ matches $g(u)$ under the identification
  $Y^+_\hbar(\mathfrak g_{\Delta_5})\simeq\CoHA(K3\times E)$
  (conjectural, Vol III status ClaimStatusConjectured). At the
  hyper-K\"ahler limit $g(u)=1$; at generic $\Omega$-background
  $g(u)\neq 1$ produces non-trivial braiding. The braiding is
  \emph{quasi-symmetric of degree 2} (Etingof-Kazhdan 1999), placing
  $(\mathrm{Rep}(\mathbf H_{\Delta_5}),\otimes,R)$ in $E_2$-braided
  categories; the chiral algebra $\mathbf H_{\Delta_5}$ itself remains
  $E_1$.
\end{enumerate}

\paragraph{Heal.}
\begin{enumerate}[label=(H\arabic*),leftmargin=1.5em]
\item Rewrite Line~1317 from ``This is consistent with the $E_2$
  mechanism'' to: ``The self-dual K3 has $E_\infty$ braiding
  (symmetric), and the $\Omega$-deformation breaks it to $E_2$ braiding
  on the Drinfeld centre $Z^{\mathrm{der}}_{\mathrm{ch}}(\mathbf H_{\Delta_5})$
  -- the object $\mathbf H_{\Delta_5}$ itself remains $E_1$-chiral on
