\section{Lorgat 2020 Conjecture 1 at $(N,M) = (1,\mathrm{id})$:
rigorous proof chain and its obstructions}
\label{wn:sec:gelfand-manin-wave8-i1-OP-audit-N1-theorem}

\providecommand{\ClaimStatusTheorem}{\textbf{[Thm]}}
\providecommand{\ClaimStatusConjectured}{\textbf{[Conj]}}
\providecommand{\ClaimStatusAxiomatic}{\textbf{[Ax]}}
\providecommand{\phiZero}{\phi_{0,1}}
\providecommand{\gDelta}{\mathfrak{g}_{\Delta_5}}
\providecommand{\ZDTred}{Z^{\mathrm{red}}_{\mathrm{DT}}}

\paragraph{Target statement.}
Fix $X = K3 \times E$, $L$ a primitive polarisation, $h_M = \mathrm{id}$.
The $(N,M) = (1,\mathrm{id})$ specialisation of Lorgat 2020 Conjecture~1
asserts four equalities of a single integer $\mathrm{mult}$:
\[
  \mathrm{mult}_{\gDelta}(\alpha)
  \;=\; c_{K3}(h,\ell)
  \;=\; n^{0}_{(\beta_h,\ell)}(X)
  \;=\; N^{\mathrm{DT}}_{(\beta_h,\ell)}(X),
  \qquad \alpha = (n,\ell,m),\; h = nm,
\]
witnessed by the automorphic identity
$\ZDTred(q,\tilde q, y) = -\Phi_{10}^{-1} = -\Delta_5^{-2}$.

\paragraph{Audit (five attack--heal cycles).}

\emph{Cycle~I (reduced virtual class at $K3 \times E$).}
\textsc{Attack.} Oberdieck--Pixton \textup{(}arXiv:1411.1514\textup{)}
state the Igusa cusp conjecture; the proof appears in Oberdieck 2018
\textup{(}Duke 167:3045\textup{)}. Does this proof depend on open
MNOP conjectures? \textsc{Heal.} The MNOP $\mathrm{GW}/\mathrm{DT}$
correspondence in \emph{reduced} form on $K3 \times E$ is the
Maulik--Pandharipande--Thomas 2010 reduced extension, unconditional
for the primitive $K3$ class by Pandharipande--Thomas 2014 stable-pairs
proof of KKV. Oberdieck 2018 assembles the absolute reduced theory
from the relative theory of $K3 \times \mathbb{A}^1$ through the
degeneration $E \rightsquigarrow \mathbb{P}^1_{\mathrm{nodal}}$; the
rubber contribution integrates to $\eta^{24}$. \textbf{The reduced
virtual-class identification is \ClaimStatusTheorem}, not contingent
on the un-reduced MNOP conjecture (which remains open in general).

\emph{Cycle~II (Gopakumar--Vafa integrality).}
\textsc{Attack.} Is $\mathrm{mult}_{\gDelta}(\alpha) = n^{0}_{\mathrm{GV}}(\alpha)$
a theorem at $(1,\mathrm{id})$, or does it hang on Ionel--Parker~2018?
\textsc{Heal.} On $K3 \times E$ integrality bypasses Ionel--Parker: the
KKV formula \textup{(}Pandharipande--Thomas 2014, proving Katz--Klemm--Vafa
1999 in all classes via stable pairs\textup{)} expresses
$n^{0}_{(\beta_h,\ell)}$ as $c_{K3}(h,\ell)$, the Fourier coefficients
of $Z_{\mathrm{ell}}(K3) = 2\phiZero$ \textup{(}Eguchi--Ooguri--Tachikawa
2011\textup{)}, which are integers by construction of a weak Jacobi
form of weight $0$ index $1$. Ionel--Parker 2018 is sufficient but not
necessary here; the Oberdieck--Pandharipande multiple-cover refinement
completes the $\ell$-dependence. \textbf{$n^{0}_{\mathrm{GV}} = c_{K3}$ is
\ClaimStatusTheorem{} at $(1,\mathrm{id})$.}

\emph{Cycle~III (BKM denominator $\Delta_5 = \mathrm{Bor}(\phiZero)$).}
\textsc{Attack.} The Gritsenko--Nikulin 1998 theorem
$\Phi_{10} = \mathrm{Bor}(2\phiZero)$ rests on Lemmas~2--3 of GN
(anti-invariance of $\Delta_5$ under Type~II reflections of
$W^{(2)}(\Lambda^{2,1}_{II})$, invariance under Type~I). Are these
lemmas rigorously established? \textsc{Heal.} Lemma~2 (Type~I
reflection-invariance) is a direct check on the Gram matrix
$2(2I - J_3)$ with spectrum $\{-2,4,4\}$; Lemma~3 (Type~II reflection
anti-invariance) is the sign
$\det(w) = -1$ on reflections through $\delta^2 = 2$, $(\delta, \Lambda^{2,1}) = 2\mathbb{Z}$,
consistent with Borcherds 1995 singular-theta output. Both admit direct
Fourier-coefficient verification at the $f(1,1,1) = 64$ vertex (Poor--Yuen
2015 Appendix). Lorgat 2020 \S3--4 gives an independent re-derivation.
\textbf{$\Delta_5^2 = \mathrm{Bor}(2\phiZero)$ is
\ClaimStatusTheorem{} \textup{(}Borcherds 1995 + Gritsenko--Nikulin 1998\textup{)}.}

\emph{Cycle~IV (the four-way equality).}
\textsc{Attack.} Is $\mathrm{mult}_{\gDelta} = c_{K3} = n^{0}_{\mathrm{GV}} = N^{\mathrm{DT}}$
four theorems or three-plus-one conjecture? \textsc{Heal.} At
$(N,M) = (1,\mathrm{id})$: \textup{(}a\textup{)}
$\mathrm{mult}_{\gDelta}(\alpha) = c_{K3}(nm,\ell)$ is the Borcherds
product expansion of $\Delta_5^2$ with exponents drawn from $2\phiZero$
\textup{(}Gritsenko--Nikulin 1998, Thm.~1.1\textup{)};
\textup{(}b\textup{)} $c_{K3} = n^{0}_{\mathrm{GV}}$ is KKV \textup{(}PT14\textup{)};
\textup{(}c\textup{)} $n^{0}_{\mathrm{GV}} = N^{\mathrm{DT}}$ is reduced MNOP at
genus zero \textup{(}MPT 2010, Oberdieck 2018\textup{)}. Each link is a
theorem under reduced primitive hypothesis. \textbf{All four-way
equality is \ClaimStatusTheorem{} at $(1,\mathrm{id})$}, contingent only
on the $-1$ orientation prefactor tracking the Behrend sign on the
reduced component \textup{(}Maulik--Toda 2018\textup{)}.

\emph{Cycle~V (KS wall-crossing product formula).}
\textsc{Attack.} Is
$\prod_{\gamma}(1-x^\gamma)^{(-1)^{|\gamma|}\Omega(\gamma)} = \Delta_5^{-1}$
derived from Joyce--Song 2012 or restated conjecturally from
Kontsevich--Soibelman 2008? \textsc{Heal.} The motivic wall-crossing
formula of KS 2008 \S6.2 is the object-level definition of
$\Omega(\gamma,\sigma)$ via the motivic Hall algebra; Joyce--Song 2012
\textup{(}Memoirs AMS 217\textup{)} \S5 upgrades $\Omega$ to the
Behrend-weighted integer BPS count under chamber stability. The
identification with $\Delta_5^{-1}$ then follows by evaluating
$\Omega(\gamma,\sigma_0) = \mathrm{mult}_{\mathrm{BKM}}(\alpha_\gamma)$
at the $\Delta_5$-chamber, using Cycle~IV. \textbf{The KS product formula
is \ClaimStatusTheorem{} at $(1,\mathrm{id})$}, with Joyce--Song 2012
providing the integer-lift, and KS 2008 providing the motivic scaffolding.
The Bridgeland--Smith monodromy
$\rho_{\mathrm{BS}} : \Sp_4(\Z)/\{\pm I\} \hookrightarrow \mathrm{Auteq}(D^{\mathrm{b}}(X))/\sim$
supplies chamber-independence.

\paragraph{Rigorous proof chain at $(N,M) = (1,\mathrm{id})$.}
Eight steps, each scope-tagged:
\begin{enumerate}
\item $Z_{\mathrm{ell}}(K3) = 2\phiZero$, weak Jacobi of weight $0$ index $1$
  \textup{(}Eguchi--Ooguri--Tachikawa 2011\textup{)}. \ClaimStatusTheorem.
\item $\mathrm{Bor}(2\phiZero) = \Phi_{10}$, Borcherds singular-theta lift
  \textup{(}Borcherds 1995\textup{)}. \ClaimStatusTheorem.
\item $\Phi_{10} = \Delta_5^2$, Gritsenko additive-lift square
  \textup{(}Gritsenko--Nikulin 1998; Lorgat 2020 \S3\textup{)}.
  \ClaimStatusTheorem.
\item $\gDelta$ exists as a GKM superalgebra with $\Delta_5 = \mathrm{den}(\gDelta)$
  \textup{(}Gritsenko--Nikulin 1998 Thm.~1.1; Lorgat 2020 Thm.~3\textup{)}.
  \ClaimStatusTheorem.
\item KKV on $K3$: $\sum n^{0,K3}_{h} q^{h-1} = \prod(1-q^n)^{-24}$,
  integrality via $\mathrm{Hilb}^h(K3)$ $\chi_y$-genus
  \textup{(}Pandharipande--Thomas 2014\textup{)}. \ClaimStatusTheorem.
\item Reduced MNOP on $K3 \times E$: $\ZDTred = \ZDTred_{\mathrm{GW}}$
  \textup{(}Maulik--Pandharipande--Thomas 2010\textup{)}. \ClaimStatusTheorem.
\item Degeneration $E \rightsquigarrow \mathbb{P}^1_{\mathrm{nodal}}$:
  $\ZDTred(K3\times E) = -\Phi_{10}^{-1}$ \textup{(}Oberdieck 2018
  Thm.~1, completing Oberdieck--Pixton arXiv:1411.1514\textup{)}.
  \ClaimStatusTheorem.
\item Four-way mult identification
  $\mathrm{mult}_{\gDelta}(\alpha) = c_{K3}(h,\ell) = n^{0}_{\beta}(X) = N^{\mathrm{DT}}_{\beta}(X)$
  \textup{(}Step~3 + Step~5 + Step~6 + Step~7\textup{)}.
  \ClaimStatusTheorem.
\end{enumerate}

\paragraph{Verdict.} At $(N,M) = (1,\mathrm{id})$ Lorgat 2020
Conjecture~1 is \textbf{a theorem}, assembled from eight published
inputs. No step is \ClaimStatusAxiomatic; no step is \ClaimStatusConjectured;
Ionel--Parker 2018 is a sufficient but not necessary input
\textup{(}PT14 + OP suffice\textup{)}. The open portion of the
conjecture lies at $N \geq 2$, $h_M \neq \mathrm{id}$, or
$N \in \{5,7,8\}$ boundary cases.

\paragraph{Invariants.} $\kappa_{\mathrm{BKM}}(\Delta_5) = c_1(0)/2 = 5$
\textup{(}Borcherds weight theorem\textup{)};
$\kappa_{\mathrm{ch}}(K3 \times E) = 3$;
$\kappa_{\mathrm{cat}}(K3 \times E) = \chi(\mathcal{O}_{K3})\cdot\chi(\mathcal{O}_E) = 2 \cdot 0 = 0$
\textup{(}K\"unneth multiplicative, not the $K3$ fibre value $2$\textup{)};
$\kappa_{\mathrm{fiber}} = 24$. No conflation.
