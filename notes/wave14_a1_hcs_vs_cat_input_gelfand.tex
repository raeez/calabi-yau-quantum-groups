\title{The Inputs of $\PhiFA_3$: Holomorphic Chern--Simons and
Calabi--Yau Hochschild Theory}
\author{Raeez Lorgat}
\date{2026-04-30}

\begin{abstract}
Holomorphic Chern--Simons theory starts from geometric data
$(X,\fg,\Omega_X)$ and from a BV quantisation package.  Calabi--Yau
Hochschild theory starts from a Morita-invariant categorical datum
$(\cC,\eta)$.  Hall theory starts from a stack of objects, an orientation
or potential, a stability condition, and vanishing-cycle Borel--Moore
chains.  On the geometric locus $\cC=\Perf(X)$, after a framing object, a
formality model, completions, and anomaly control have been chosen, hCS
and Hochschild theory compare through HKR, open--closed maps, and
Costello--Gwilliam locality.  The comparison identifies completed
observable theories on the common locus.  It leaves the input data, the
Morita quotient, Hall products, associator choices, Atiyah--Kapranov--BCOV
cocycles, and MNOP trace functionals as separate mathematical structures.
\end{abstract}

\section{Typed Input Data}

\begin{definition}[Geometric hCS input]
\label{def:hcs-input}
A holomorphic Chern--Simons input in complex dimension three consists of
a smooth complex threefold $X$, a nowhere-vanishing holomorphic volume
form $\Omega_X\in H^0(X,K_X)$, and a finite type quadratic dg Lie algebra
or $L_\infty$ algebra $(\fg,\langle-,-\rangle)$.  The classical field is
\[
  A\in \Omega^{0,\bullet}(X,\fg)[1],
\]
and the classical action is
\begin{equation}
\label{eq:hcs-action}
  S_{\hCS}(A)
  =
  \int_X \Omega_X\wedge
  \left(
    \frac12\langle A,\dbar A\rangle
    +\frac16\langle A,[A,A]\rangle
  \right).
\end{equation}
For non-compact $X$ the integral is read locally or with compact support.
For compact $X$ it is part of a BV theory only after the global package of
Hypothesis~\ref{hyp:compact-hcs-package} has been supplied.
\end{definition}

\begin{definition}[Categorical CY3 input]
\label{def:cat-input}
A categorical Calabi--Yau input of dimension three is a smooth proper
$\Cx$-linear pretriangulated dg category, equivalently a smooth proper
stable $\infty$-category on a chain model, equipped with a negative cyclic
Calabi--Yau structure whose Hochschild trace is a non-degenerate pairing
\[
  \eta\colon \HH_\bullet(\cC)\longrightarrow \Cx[-3].
\]
Equivalently, $\eta$ identifies the Serre functor of $\cC$ with the shift
$[3]$.  For $\cC=\Perf(X)$, the trace is the Serre residue pairing
determined by $\Omega_X$.
\end{definition}

\begin{proposition}[Different inputs with a common comparison locus]
\label{prop:typed-inputs}
The datum $(X,\fg,\Omega_X)$ and the datum $(\cC,\eta)$ have different
types.  A comparison between them requires all of the following choices:
\begin{enumerate}[label=\textup{(\roman*)}]
\item a geometric realisation $\cC\simeq \Perf(X)$ or a specified Morita
equivalent dg model;
\item a framing object $E\in\cC$, giving the endomorphism algebra
$R\End_\cC(E)$ and the gauge algebra $\fg_E=H^0(R\End_\cC(E))$ in the
strict classical case;
\item a formality transfer from Hochschild cochains to polyvectors;
\item completions of fields and observables;
\item a BV quantisation satisfying the quantum master equation.
\end{enumerate}
On this locus the comparison relates the completed hCS observable algebra
to the Hochschild operations of the framed Calabi--Yau category.  A
smooth proper CY3 category can be non-geometric, and an hCS triple can
remember a selected gauge algebra that the Morita class of $\cC$ has
quotiented.
\end{proposition}

\begin{proposition}[Five typed layers]
\label{prop:five-typed-layers}
The comparison problem has five layers:
\[
\begin{array}{c|c|c}
\text{layer} & \text{primitive datum} & \text{structure carried}\\
\hline
\text{local hCS BV} &
(U,\fg,\Omega_U,S_{\hCS}) &
\text{elliptic local BV fields and brackets}\\
\text{compact hCS quantisation} &
(X,\fg,\Omega_X,Q^{\mathrm{GF}},I[L]) &
\QME,\ \text{anomaly trivialisation, descent, sewing}\\
\text{categorical CY3} &
(\cC,\eta,E) &
\text{Morita class, CY trace, framed endomorphisms}\\
\text{Hall/CoHA} &
(\mathcal M_\cC,\phi_W,\Gamma,\sigma,\mathrm{or}) &
\text{oriented vanishing-cycle correspondences}\\
\text{chiral output} &
A_\cC=\Sp_{\Sigma_{d-1},C}(\PhiFA_d(\cC)) &
\Eone\text{ output at }d\geq3
\end{array}
\]
An assertion crossing two rows requires the named comparison functor and
its hypotheses.  A local hCS chart on $U$ supplies the first row.  A
compact theorem supplies the second row only after the QME, anomaly
trivialisation, orientation, framing, counterterms, completion, descent,
and sewing data have been constructed.  The categorical row supplies a
Morita-invariant CY source and a framing object.  The Hall row supplies
the product by critical correspondences.  The chiral row is the
specialised output of $\Phi$.
\end{proposition}

\begin{corollary}[Positive half on affine space]
\label{cor:c3-positive-half}
For the smooth toric affine CY3,
\[
  \CoHA(\Cx^3)\simeq Y^+(\widehat{\fgl}_1).
\]
This is a Hall-side positive half.  The full vertex-algebraic
$\mathcal W_{1+\infty}$ object requires the Drinfeld double or centre,
the relevant completion, and an evaluation or vacuum representation.
\end{corollary}

\section{Local Comparison}

\begin{theorem}[Affine hCS and Hochschild operations]
\label{thm:local-comparison}
Let $U\subset \Cx^3$ be a Stein polydisc with
\[
  \Omega=dz_1\wedge dz_2\wedge dz_3,
\]
and let $V$ be a finite-dimensional complex vector space.  Assume a
chosen chain formality model, the Costello--Li anomaly-free local BV
quantisation of hCS on $U$ with gauge algebra $\fgl(V)$, and
completed local observables.  Then there is an $\Ethree$ quasi-isomorphism
between local quantum hCS observables and the matrix-framed Hochschild
operations of $\Perf(U)$:
\[
  \Obs^q_{\hCS}(U,\fgl(V))
  \simeq
  C^\bullet_{\mathrm{Hoch},\Ethree}
    \bigl(\Perf(U);\End(V)\bigr).
\]
At the symbol level, HKR gives
\[
  \HH^\bullet(\Perf(U))
  \simeq
  \PV^\bullet(U)
  =
  \bigoplus_{p,q}\Omega^{0,q}(U,\wedge^pT^{1,0}U),
\]
and contraction with $\Omega$ sends $\PV^{p,q}(U)$ to
$\Omega^{3-p,q}(U)$.
\end{theorem}

\begin{proof}
The HKR map identifies Hochschild cochains with polyvector fields on the
affine formal model.  The Calabi--Yau form converts polyvectors to
Dolbeault forms of complementary holomorphic degree.  The object $V$
supplies the open matrix sector $\End(V)$, giving the hCS dg Lie algebra
$\Omega^{0,\bullet}(U,\fgl(V))[1]$.  Costello--Gwilliam locality turns
the anomaly-free BV package into a factorisation algebra of observables.
The open--closed map identifies closed Hochschild operations with local
operations acting on the hCS open sector.  The result is local: compact
QME, Hall products, and chiral specialisation enter through the later
layers of Proposition~\ref{prop:five-typed-layers}.
\end{proof}

\section{Compact hCS Quantisation}

\begin{hypothesis}[Compact hCS package]
\label{hyp:compact-hcs-package}
A compact CY3 hCS quantisation result requires:
\begin{enumerate}[label=\textup{(\roman*)}]
\item a compact Calabi--Yau threefold $(X,\Omega_X)$ with orientation and
determinant-line framing compatible with the BV pairing;
\item a quadratic gauge dg Lie algebra $(\fg,\langle-,-\rangle)$ and,
when colour traces are used, a representation
$\rho\colon\fg\to\End(W)$;
\item an elliptic Costello gauge fixing $Q^{\mathrm{GF}}$, heat-kernel
regularisation, propagators, and renormalisation counterterms;
\item completed local functionals and observables on which the BV
Laplacian, bracket, and RG flow are defined;
\item a solution of the scale-$L$ quantum master equation;
\item vanishing or explicit trivialisation of the anomaly class
$\kanom(X,\fg,\rho)$, with open--closed counterterms included when the
chosen regulator requires them;
\item factorisation descent over $X$, and TCFT sewing data when global
traces or partition functions are asserted.
\end{enumerate}
\end{hypothesis}

\begin{proposition}[Conditional compact observable algebra]
\label{prop:compact-conditional}
Under Hypothesis~\ref{hyp:compact-hcs-package}, the completed quantum
hCS observables form a quantum factorisation algebra on $X$.  If the
anomaly class survives in the Costello local BV obstruction complex for
the chosen fields, regulator, and counterterms, the corresponding compact
quantum observable algebra is undefined at that order.
A polydisc computation supplies a local model for counterterms and
propagators; compact quantisation additionally requires the determinant
line, global orientation, all-scale QME solution, and descent/sewing
data in Hypothesis~\ref{hyp:compact-hcs-package}.
\end{proposition}

\section{Quartic Anomaly and Quadratic Casimir}

\begin{definition}[Quartic anomaly invariant]
\label{def:quartic-invariant}
For a representation $\rho\colon\fg\to\End(W)$ define
\[
  I_4^\rho(x_1,x_2,x_3,x_4)
  =
  \frac1{4!}\sum_{\sigma\in S_4}
  \Tr_W\!\left(
    \rho(x_{\sigma(1)})\rho(x_{\sigma(2)})
    \rho(x_{\sigma(3)})\rho(x_{\sigma(4)})
  \right).
\]
Then $I_4^\rho\in\Sym^4(\fg^\vee)^\fg$.
\end{definition}

\begin{proposition}[The hCS anomaly slot in complex dimension three]
\label{prop:quartic-anomaly}
The nonabelian one-loop gauge contribution to the hCS quantum master
equation in complex dimension three is represented locally by
\[
  \Theta^\rho_U(\alpha_0,\alpha_1,\alpha_2,\alpha_3)
  =
  \int_U
  \left[
  I_4^\rho
    \bigl(\alpha_0,\partial\alpha_1,\partial\alpha_2,\partial\alpha_3\bigr)
  \right]_{(3,3)},
  \qquad
  \alpha_i\in\Omega_c^{0,\bullet}(U,\fg).
\]
Its class
\[
  \kanom(X,\fg,\rho)
  =
  [\,\hbar\,\Theta^\rho\,]
  \in
  H^1_{\mathrm{loc}}
    \bigl(\Omega^{0,\bullet}(X,\fg)[1],\cO_{\mathrm{loc}}\bigr)
\]
is the first open-string obstruction to the QME.
\end{proposition}

\begin{definition}[Quadratic Casimir coefficient]
\label{def:quadratic-casimir}
Let $\{T_a\}$ be a basis of $\fg$ and let $\{T^a\}$ be dual for the chosen
invariant pairing.  The quadratic Casimir in the representation $\rho$ is
\[
  C_2^\rho=\sum_a \rho(T_a)\rho(T^a).
\]
For a simple Lie algebra in the adjoint representation, Killing-form
normalisation gives $C_2^{\mathrm{ad}}=2h^\vee$.
\end{definition}

\begin{proposition}[Quartic and two-point slots]
\label{prop:anomaly-casimir-slots}
The quartic invariant $I_4^\rho$ controls the degree-one QME obstruction.
The quadratic Casimir $C_2^\rho$ controls two-point field-strength
renormalisation through kinetic counterterms of the form
\[
  -\hbar\,\frac{C_2^\rho}{(4\pi)^3}\log(L/\varepsilon)
  \int_X \Omega_X\wedge\Tr_\rho(A\,\dbar A)
\]
up to pairing and heat-kernel conventions.  These two invariants occupy
different cohomological degrees and different Feynman graph arities.
\end{proposition}

\section{Morita Invariance and Framing}

\begin{proposition}[Morita-invariant categorical input]
\label{prop:morita}
Hochschild chains, Hochschild cochains, and negative cyclic
Calabi--Yau structures are Morita invariant.  If
$\cC\simeq_{\mathrm{Morita}}\cD$, then $\HH_\bullet(\cC)\simeq
\HH_\bullet(\cD)$ and the transported trace gives the same CY3 pairing.
Thus $(\cC,\eta)$ records the Morita class of the category with its
Calabi--Yau orientation.
\end{proposition}

\begin{proposition}[Framed gauge fibre]
\label{prop:framed-fibre}
For an object $E\in\cC$, the derived endomorphism algebra
$R\End_\cC(E)$ is an $\Eone$ algebra.  The Calabi--Yau trace $\eta$
induces a cyclic pairing on $R\End_\cC(E)$ when $E$ is perfect and
dualizable.  A strict classical gauge algebra is obtained from a chosen
Lie model of this endomorphism algebra, for instance
\[
  \fg_E=H^0(R\End_\cC(E))
\]
when higher operations are absent from the classical truncation.
The hCS triple fixes such a fibre.  The categorical datum keeps the
object family and its autoequivalences visible before a fibre is chosen.
\end{proposition}

\section{Formality Choices}

\begin{definition}[Formality datum]
\label{def:formality-datum}
A formality datum for the comparison is a chain-level zigzag transferring
the relevant little-disks or Gerstenhaber operations between Hochschild
cochains and polyvector fields.  In the $\Etwo$ or Gerstenhaber layer,
Drinfeld associators and the prounipotent Grothendieck--Teichmuller
action parameterise the familiar Tamarkin choices.  In the $\Ethree$
hCS comparison one still records the selected chain formality zigzag,
its gauge class, and its compatibility with the BV propagator and
descent.
\end{definition}

\begin{proposition}[Equality requires the chosen transfer]
\label{prop:formality-choice}
The local comparison of Theorem~\ref{thm:local-comparison} is an
equivalence of completed $\Ethree$ chain models after a formality datum
has been fixed.  A different associator or graph-formality transfer gives
a different chain representative in the same equivalence class.  Hence
the symbol-level HKR identification, the operadic transfer, and the
analytic hCS propagator are three separate inputs to the comparison.
\end{proposition}

\section{Atiyah, Kapranov, and BCOV Cocycles}

\begin{definition}[Atiyah class]
\label{def:atiyah}
For a smooth complex threefold $X$, the Atiyah class of the tangent
bundle is
\[
  \At(T_X)\in H^1(X,\Omega^1_X\otimes\End T_X).
\]
It is the extension class of the first jet sequence.  Kapranov identifies
it with the binary bracket in the $L_\infty$ algebra on $T_X[-1]$ in
the derived category.
\end{definition}

\begin{definition}[Yukawa cubic]
\label{def:yukawa}
For a compact CY3 $(X,\Omega_X)$ and
$v_i\in H^1(X,T_X)$, the Yukawa cubic is
\[
  Y_X(v_1,v_2,v_3)
  =
  \int_X
  \Omega_X\wedge
  \bigl((v_1\wedge v_2\wedge v_3)\,\lrcorner\,\Omega_X\bigr).
\]
In the Kapranov transfer it is the arity-three minimal bracket paired
with the Calabi--Yau trace.
\end{definition}

\begin{definition}[BCOV one-loop datum]
\label{def:bcov}
The BCOV one-loop datum is the genus-one determinant-line and
holomorphic-anomaly contribution of Kodaira--Spencer theory.  Its scalar
coefficient contains $\chi_{\mathrm{top}}(X)/24$ in the standard CY3
normalisation.  It is a closed-string BV datum.  The hCS open gauge
anomaly of Proposition~\ref{prop:quartic-anomaly} remains governed by
$I_4^\rho$.
\end{definition}

\begin{proposition}[Three related cocycle slots]
\label{prop:three-cocycles}
The classes in Definitions~\ref{def:atiyah}, \ref{def:yukawa}, and
\ref{def:bcov} occupy different slots:
\[
\begin{array}{c|c|c}
\text{datum} & \text{home} & \text{role}\\
\hline
\At(T_X) & H^1(X,\Omega^1_X\otimes\End T_X)
  & \text{Kapranov binary bracket}\\
Y_X & \Sym^3 H^1(X,T_X)^\vee
  & \text{tree-level CY3 cubic}\\
\BCOV_1 & \text{determinant-line anomaly datum}
  & \text{closed one-loop contribution}
\end{array}
\]
The Atiyah class feeds the Kapranov brackets; the Calabi--Yau trace pairs
the arity-three transfer to the Yukawa cubic; the BCOV datum belongs to
the closed one-loop theory.  The hCS gauge QME obstruction is the quartic
open-string class $\kanom(X,\fg,\rho)$.
\end{proposition}

\section{MNOP as a Trace Identity}

\begin{definition}[Reduced DT partition function]
\label{def:reduced-dt}
Let $M(q)$ be the MacMahon function and let $\chi_{\mathrm{top}}(X)$ be
the topological Euler characteristic of a compact CY3.  The degree-zero
DT factor is
\[
  Z^{\mathrm{DT}}_0(X;q)=M(-q)^{\chi_{\mathrm{top}}(X)}.
\]
The reduced DT series is
\[
  Z^{\mathrm{DT}\prime}(X;q,v)
  =
  Z^{\mathrm{DT}}(X;q,v)/M(-q)^{\chi_{\mathrm{top}}(X)}.
\]
\end{definition}

\begin{proposition}[MNOP comparison]
\label{prop:mnop}
The MNOP correspondence compares numerical trace functionals:
\[
  Z^{\mathrm{GW}\prime}(X;u,v)
  =
  Z^{\mathrm{DT}\prime}(X;q,v)\big|_{q=-e^{iu}},
\]
after the MNOP removal of unstable degree-zero factors and the variable
change $q=-e^{iu}$.  It is a theorem on the toric CY3 locus and a
conjectural identity for general compact CY3 targets.  In a
factorisation-algebra reading, DT, GW, and PT theories are trace
functionals on the completed $\Ethree$ object under the compact package;
MNOP identifies two such trace values.  The hCS input triple and the
categorical CY3 input remain separate data.
\end{proposition}

\section{Specialised Chiral Output}

\begin{definition}[Specialised output]
\label{def:specialised-output}
For a CY$_d$ category $\cC$, write
\[
  A_\cC
  =
  \Sp_{\Sigma_{d-1},C}\bigl(\PhiFA_d(\cC)\bigr)
\]
for the chiral algebra on the reference curve $C$ obtained by
holomorphic factorisation homology over the transverse cycle
$\Sigma_{d-1}$ and restriction to $C$.
\end{definition}

\begin{proposition}[Native operadic level]
\label{prop:native-level}
The specialised output has native operadic level
\[
  n_{\mathrm{native}}(d)=
  \begin{cases}
  \infty, & d=1,\\
  2, & d=2,\\
  1, & d\geq 3.
  \end{cases}
\]
Thus for $d\geq 3$ the specialised algebra $A_\cC$ is an $\Eone$ chiral
algebra.  The braided $\Etwo$ structure belongs to the centre, double, or
module layer, for example
\[
  \mathcal Z\bigl(\Rep^{\Eone}(A_\cC)\bigr)
  \simeq
  \Rep^{\Etwo}\bigl(\Zchder(A_\cC)\bigr)
\]
on the verified Drinfeld-centre locus.
\end{proposition}

\begin{remark}[Five separated objects]
\label{rem:five-objects}
For an $\Eone$ chiral algebra $A$, the five objects
\[
  A,\qquad B(A),\qquad A^i,\qquad A^!,\qquad \Zchder(A)
\]
are distinct.  The identity $\Omega B(A)\simeq A$ is bar--cobar
inversion.  The object $A^i$ is the Koszul dual coalgebra.  The object
$A^!$ is the Verdier or Koszul dual algebra, according to the chosen
duality context.  The derived chiral centre $\Zchder(A)$ is the centre
or bulk object carrying the braided layer in the $d\geq 3$ specialisation.
\end{remark}

\section{The $K3\times E$ Constants}

\begin{proposition}[Four construction-specific invariants]
\label{prop:k3e-values}
For $X=K3\times E$,
\[
  \kcat(X)=\chi(\cO_{K3})\chi(\cO_E)=2\cdot0=0,
\]
and the compact Hodge supertrace gives
\[
  \kch^{\mathrm{compact}}(X)
  =
  \sum_{q=0}^3(-1)^q h^{0,q}(X)
  =
  1-1+1-1
  =
  0.
\]
The Heisenberg--Mukai specialisation has
\[
  \kch^{\mathrm{Heis}}(X)=3,
\]
the Gritsenko $\Delta_5$ Borcherds construction has
\[
  \kBKM(\mathfrak g_{\Delta_5})=5,
\]
and the K3 fibre Mukai rank gives
\[
  \kfib(X)=24.
\]
These are four construction-specific invariants:
\[
  \{\kcat,\kch^{\mathrm{Heis}},\kBKM,\kfib\}(K3\times E)
  =
  \{0,3,5,24\},
\]
with the compact chiral supertrace
$\kch^{\mathrm{compact}}(K3\times E)=0$ recorded separately.
\end{proposition}
