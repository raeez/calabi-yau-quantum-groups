\begin{abstract}
For a class-$\mathcal{M}$ chiral algebra whose reduced one-handle external
$E_3$ bar page is $\Lambda^\bullet(\kk^3)$, the quartic shadow differential
has one-handle cohomology $[0,3,3,0]$. For $g$ independent handles the
post-quartic page has Poincar\'e polynomial $(3t+3t^2)^g$ and total dimension
$6^g$. Degree reasons identify this page with final cohomology exactly for
$1\leq g\leq 3$. For $g\geq 4$ the same number is the post-quartic page
dimension; final cohomology still depends on higher differentials. At $g=4$
the only remaining differential is
the quintic transfer
\[
 D_5(4,c):(\Lambda^2\kk^3)^{\otimes 4}\longrightarrow
          (\Lambda^1\kk^3)^{\otimes 4},
\]
an $81$ by $81$ matrix. If $r_5(c)=\operatorname{rk}D_5(4,c)$, then
\[
 \dim H^\bullet(\Bext(\Vir_c^{\oplus 4});\Sigma_4)=1296-2r_5(c).
\]
The computation of $r_5(c)$ is the genus-four obstruction.
\end{abstract}

\section{Layers}
\label{sec:layers}

The symbol $\mathcal{M}$ names a shadow class of chiral algebras. It means
that the transferred $A_\infty$ shadow tower has nonzero operations in
unbounded arity. The Virasoro algebra at generic central charge is the basic
example. In the normalization used here,
\[
 \kch(\Vir_c)=\frac{c}{2},\qquad
 S_4(c)=\frac{10}{c(5c+22)},\qquad
 \Delta_4(c)=8\kch(\Vir_c)S_4(c)=\frac{40}{5c+22}.
\]
Thus $\Delta_4(c)$ is nonzero for $c\neq -22/5$.

The genus parameter $g$ counts independent handle factors in the external
$E_3$ bar model. It records handle factorization data for this bar
construction. The class-$\mathcal{M}$ logarithmic centre, the external
$E_3$ bar construction, the genus handle factor, and the CY-to-chiral output
are separate objects.

For a CY$_d$ input with $d\geq 3$, the native CY-to-chiral output is
\[
 A_\cC=\Phi_d(\cC)=
 \SpCh_{\Sigma_{d-1},C}\bigl(\PhiFA_d(\cC)\bigr)\in E_1\text{-ChirAlg}.
\]
The braided layer is
\[
 \Rep^{E_2}(\Zder(A_\cC))\simeq \cZ(\Rep^{E_1}(A_\cC)).
\]
Further $E_3$ structures arise from centre, double, module, or ambient
factorisation constructions. They belong to layers beyond the native
$\Phi_d$ output at $d\geq 3$.

A compact or global hCS trace theorem carries the following data: a QME
solution, anomaly cancellation, an orientation or framing, descent and sewing
maps, local counterterms, a completed state space, and a comparison map to the
external bar page. The arguments below use the reduced external bar spectral
sequence as their input.

\section{The Reduced External Bar Model}
\label{sec:reduced-model}

\begin{definition}[One-handle external $E_3$ page]
\label{def:one-handle-page}
Let $W\simeq\kk^3$ be the span of the three external $E_3$ directions. The
reduced one-handle page of the class-$\mathcal{M}$ external bar model is
\[
 C_1^\bullet=\Lambda^\bullet W=[1,3,3,1],
\]
graded by exterior degree. For $g$ handles,
\[
 C_g^\bullet=(C_1^\bullet)^{\otimes g}
 \qquad\text{and}\qquad
 P_{C_g}(t)=(1+t)^{3g}.
\]
\end{definition}

The full chiral bar complex of $\Vir_c$ contains all derivative modes of the
Virasoro field and has unbounded mode degree. The finite model in
Definition~\ref{def:one-handle-page} is the reduced external page on which
the shadow differentials are computed. All dimension statements below concern
cohomology of this reduced spectral sequence.

\begin{definition}[Quartic shadow differential]
\label{def:quartic-differential}
For $c\neq -22/5$, let
\[
 d_4^{(1)}:\Lambda^3 W\longrightarrow \Lambda^0 W
\]
be multiplication by $\Delta_4(c)=40/(5c+22)$, and let $d_4^{(1)}$ vanish
on $\Lambda^1W$ and $\Lambda^2W$. On $g$ independent handles set
\[
 d_4=\sum_{i=1}^g
 1^{\otimes(i-1)}\otimes d_4^{(1)}\otimes 1^{\otimes(g-i)}.
\]
\end{definition}

\begin{lemma}[One-handle quartic cohomology]
\label{lem:one-handle}
For $c\neq -22/5$,
\[
 H^\bullet(C_1,d_4^{(1)})=\Lambda^1W\oplus\Lambda^2W,
 \qquad
 P_{H(C_1,d_4)}(t)=3t+3t^2.
\]
\end{lemma}

\begin{proof}
The scalar $\Delta_4(c)$ is nonzero, hence
$d_4^{(1)}:\Lambda^3W\to\Lambda^0W$ is an isomorphism between
one-dimensional spaces. Its image kills $\Lambda^0W$ and its kernel on
$\Lambda^3W$ is zero. The middle exterior degrees are outside the source
and target of $d_4^{(1)}$.
\end{proof}

\section{The Verified Range}
\label{sec:verified-range}

\begin{proposition}[Post-quartic page in genus $g$]
\label{prop:post-quartic}
For $g\geq 1$ and $c\neq -22/5$,
\[
 F_g^\bullet:=H^\bullet(C_g,d_4)
 =(\Lambda^1W\oplus\Lambda^2W)^{\otimes g}.
\]
Its Poincar\'e polynomial and degree dimensions are
\[
 P_{F_g}(t)=(3t+3t^2)^g=3^g t^g(1+t)^g,
\]
\[
 \dim F_g^n=
 \begin{cases}
  3^g\binom{g}{n-g}, & g\leq n\leq 2g,\\
  0, & \text{otherwise}.
 \end{cases}
\]
In particular $\dim F_g=6^g$.
\end{proposition}

\begin{proof}
The differential $d_4$ is the sum of independent one-handle contractions.
Over a field, the K\"unneth theorem gives
\[
 H^\bullet(C_g,d_4)=
 \bigotimes_{i=1}^g H^\bullet(C_1,d_4^{(1)}).
\]
Lemma~\ref{lem:one-handle} gives the one-handle factor
$\Lambda^1W\oplus\Lambda^2W$, with dimensions $3$ in degrees $1$ and $2$.
The displayed polynomial and binomial coefficient follow by choosing
$n-g$ handles in degree $2$ and the remaining handles in degree $1$.
\end{proof}

\begin{theorem}[Class-$\mathcal{M}$ $E_3$ bar cohomology through genus three]
\label{thm:g-leq-three}
For $1\leq g\leq 3$ and $c\neq -22/5$,
\[
 H^\bullet(\Bext(\Vir_c^{\oplus g});\Sigma_g)\cong F_g^\bullet,
 \qquad
 \dim H^\bullet(\Bext(\Vir_c^{\oplus g});\Sigma_g)=6^g.
\]
The values are
\[
 g=1:\ 6,\qquad g=2:\ 36,\qquad g=3:\ 216.
\]
\end{theorem}

\begin{proof}
After the quartic differential, $F_g$ is supported in total degrees
$[g,2g]$. A higher differential $D_r$ lowers total degree by $r-1$.
For such a differential to act on $F_g$, some source degree $n$ and target
degree $n-(r-1)$ must both lie in $[g,2g]$, so $r-1\leq g$. The first
higher shadow differential beyond the quartic one has $r=5$, hence shift
$4$. For $g\leq 3$ this shift exceeds the width of the support. All
higher differentials vanish by degree, and the post-quartic page is final.
\end{proof}

\section{The Genus-Four Differential}
\label{sec:genus-four}

At $g=4$, the support of $F_4$ is $[4,8]$. The shift of $D_5$ is $4$.
Thus the only possible higher differential is
\[
 D_5(4,c):F_4^8\longrightarrow F_4^4.
\]
The source and target are explicit:
\[
 F_4^8=(\Lambda^2W)^{\otimes 4},
 \qquad
 F_4^4=(\Lambda^1W)^{\otimes 4},
\]
and both have dimension $3^4=81$.

\begin{proposition}[Genus-four rank formula]
\label{prop:g4-rank-formula}
Let
\[
 r_5(c)=\operatorname{rk}\bigl(D_5(4,c):F_4^8\to F_4^4\bigr).
\]
Then
\[
 \dim H^\bullet(\Bext(\Vir_c^{\oplus 4});\Sigma_4)
 =6^4-2r_5(c)=1296-2r_5(c).
\]
Moreover $0\leq r_5(c)\leq 81$.
\end{proposition}

\begin{proof}
The page $F_4$ has dimensions
\[
 (81,324,486,324,81)
\]
in degrees $4,5,6,7,8$. A degree-lowering map of shift $4$ can only connect
degree $8$ to degree $4$. The cohomology of this single map has dimensions
$81-r_5(c)$ in degree $4$, $81-r_5(c)$ in degree $8$, and unchanged
dimensions in degrees $5,6,7$. Thus the total dimension is
\[
 (81-r_5(c))+324+486+324+(81-r_5(c))=1296-2r_5(c).
\]
A later differential would have shift at least $5$, leaving no source-target
pair inside the support interval $[4,8]$.
\end{proof}

\begin{problem}[The genus-four matrix]
\label{prob:g4-matrix}
Compute the induced transfer map
\[
 D_5(4,c):(\Lambda^2W)^{\otimes 4}\to(\Lambda^1W)^{\otimes 4}
\]
in the monomial bases of the two $81$-dimensional spaces, with the Koszul
signs inherited from the bar tensor product and with the quotient by the
quartic image imposed before the transfer. Its rank is the first invariant
needed beyond the theorem at $g\leq 3$.
\end{problem}

\section{Kontsevich Graph Scalar}
\label{sec:graph-scalar}

The one-channel Virasoro constants at $c=1$ are fixed independently of the
genus-four rank problem:
\[
 \kch(\Vir_1)=\frac12,\qquad S_4(1)=\frac{10}{27},\qquad
 \Delta_4(1)=\frac{40}{27},
\]
\[
 m_5(T,T,T,T,T)=-\frac{80}{9}T,\qquad S_5(1)=\frac{m_5}{5}=-\frac{16}{9}.
\]
The connected five-point Wick enumeration gives
\[
 G_5^{\mathrm{conn}}(1,2,3,4,5)=\frac{775}{5184}.
\]
With the integer-point Wick normalization the scalar product is
\[
 S_5(1)G_5^{\mathrm{conn}}(1,2,3,4,5)
 =-\frac{16}{9}\cdot\frac{775}{5184}
 =-\frac{775}{2916}.
\]
This scalar is a one-channel normalization datum. The rank of
$D_5(4,1)$ depends on its full $81$ by $81$ matrix after projection to
$F_4$. A nonzero one-channel scalar leaves that rank undecided.

\section{Higher Genus}
\label{sec:higher-genus}

\begin{proposition}[Finite obstruction list in genus $g$]
\label{prop:higher-genus-obstructions}
For $g\geq 4$, the post-quartic page $F_g$ has total dimension $6^g$ and
support $[g,2g]$. The available higher differentials on this page have
indices
\[
 5\leq r\leq g+1
\]
and degrees
\[
 D_r:F_g^n\longrightarrow F_g^{n-r+1},
 \qquad
 g\leq n-r+1<n\leq 2g.
\]
Consequently the final cohomology dimension is bounded above by $6^g$.
Equality holds precisely when every induced higher differential on the
post-quartic page vanishes.
\end{proposition}

\begin{proof}
The support calculation is Proposition~\ref{prop:post-quartic}. The degree
shift of $D_r$ is $r-1$. Such a shift fits inside $[g,2g]$ exactly when
$r-1\leq g$, which gives $r\leq g+1$. Since the quartic differential has
already been taken, the remaining shadow tower starts at $r=5$. Passing to
cohomology of a nonzero differential can only decrease total dimension.
\end{proof}

\section{Invariant Separations}
\label{sec:invariant-separations}

\begin{proposition}[Separate invariants on $K3\times E$]
\label{prop:k3e-separations}
For $X=K3\times E$, the four numerical invariants occupy four distinct
constructions:
\[
 \kcat(X)=\chi(\mathcal{O}_X)=
 \chi(\mathcal{O}_{K3})\chi(\mathcal{O}_E)=2\cdot 0=0,
\]
\[
 \kch(X)=\sum_q(-1)^qh^{0,q}(X)=1-1+1-1=0,
\]
\[
 \kch^{\mathrm{Heis}}(X)=3,\qquad
 \kBKM(\Delta_5)=\frac{c_1(0)}{2}=\frac{10}{2}=5,\qquad
 \kfib(K3)=24.
\]
The identity governing the Borcherds side is $\kBKM(\Phi_N)=c_N(0)/2$.
It is independent of the class-$\mathcal{M}$ external $E_3$ bar dimension.
\end{proposition}

\begin{proposition}[Hall, Yangian, and vertex layers]
\label{prop:hall-yangian-vertex-layers}
For the toric threefold $\C^3$,
\[
 \CoHA(\C^3)\cong Y^+(\widehat{\fgl}_1),
\]
the positive half of the affine Yangian. The full $\Winf$ object requires
the Yangian double or Drinfeld-centre passage, completion, and vacuum
evaluation. For $K3\times E$, the BKM-side Hall object is the
Hall-Drinfeld double attached to the K3-fibred Borcherds denominator. The
Mukai self-mirror Yangian is a separate K3 branch.
\end{proposition}

\begin{proof}
The first assertion is the Schiffmann--Vasserot toric Hall identification in
the positive half. The full vertex algebra acts after adding the negative
half and the central/evaluation data. The second assertion separates the
K3-fibred Borcherds denominator construction from the $d=2$ Mukai Yangian
construction; their representation-theoretic targets and hypotheses differ.
\end{proof}
