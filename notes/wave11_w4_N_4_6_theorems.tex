\section*{Wave-11 W4: orbifold DT identification at
\texorpdfstring{$(N,M)=(4,\mathrm{id}),(6,\mathrm{id})$}{(N,M)=(4,id),(6,id)}}
\label{wn:sec:wave11-w4-N46-theorems}

Let $\pi\colon K3\times E\to E/\Z_N$ be elliptic projection, $g_N$ a
Mukai-symplectic Nikulin automorphism of order $N$ on the $K3$ fibre,
$t_N$ an $N$-torsion translation on $E$, and $X^{(N)} = [K3\times
E/\langle(g_N,t_N)\rangle]$ the resulting symplectic CY$_3$ orbifold
with untwisted sector $M = \mathrm{id}$.

\subsection*{Theorem W4.N4 \texorpdfstring{($N=4$, class $4B$)}{(N=4, class 4B)}}

\ClaimStatusTheorem\ (\emph{Orbifold DT identity at $N = 4$,
untwisted sector}.) Let $g_4$ be the symplectic Nikulin $\Z_4$-action
on $K3$ at Mathieu class $4B$, cycle shape $1^{4}2^{2}4^{4}$, fixing
four isolated $A_3$-type points on the $K3$ fibre (Hashimoto 2012,
\emph{Tohoku} 64:105 Table~2; Nikulin 1979, \emph{Trudy Moscow Math.\
Soc.} 38 Thm.~4.5). Let $\chi(K3^{g_4}) = 4$ denote the $g_4$-twined
$K3$ Euler characteristic, and let $\phi^{(g_4)}_{0,1}\in
J_{0,1}(\Gamma_0(4))$ be the weight-zero index-one $g_4$-twined Jacobi
form (Eguchi--Ooguri--Tachikawa 2011 \texttt{arXiv:1004.0956}
Table~1). The reduced orbifold Donaldson--Thomas partition function
satisfies
\begin{equation}
\DTred\bigl(X^{(4)}\bigr)(\tau,\sigma,z)
\;=\;-\,\frac{C_4(\tau,z)}{\bigl(\FGC_{4}(\tau,\sigma,z)\bigr)^{2}},
\qquad C_4 = 2\phi^{(g_4)}_{0,1},\quad
\mathrm{wt}\,\FGC_4 = c_4(0)/2 = 1,
\label{wn:eq:wave11-w4-DTred-N4}
\end{equation}
where $\FGC_4$ is the Gritsenko--Cl\'ery paramodular Borcherds lift of
$\phi^{(g_4)}_{0,1}$ (Gritsenko--Cl\'ery 2013 \texttt{arXiv:1106.1979}
Thm.~4.1; Gritsenko--Nikulin 1995 Part~II
\emph{Int.~J.~Math.}~6~Thm.~2.1 for the constant $c_4(0) = 2$). The
Mukai-invariant lattice has rank $r_4 = 14$ with Cartan matrix
$A^{(4)}$ of determinant $\det A^{(4)} = -12$. For every root
$\alpha = n\delta + \ell z + m w\in \mathrm{II}_{2,1}(4)$ with norm
$\langle\alpha,\alpha\rangle = 4nm - \ell^2$,
\begin{equation}
\mult_{\mathfrak{g}^{(4)}}(\alpha)
\;=\;c^{(g_4)}_{K3}\!\bigl(4nm - \ell^{2},\,\ell\bigr),
\label{wn:eq:wave11-w4-mult-N4}
\end{equation}
the Fourier coefficient of $\phi^{(g_4)}_{0,1}$.

\emph{Proof.} Bryan--Steinberg 2014 (\emph{Duke Math.\ J.}
168:3517--3569, \texttt{arXiv:1403.4748}, Thm.~1.1) establishes the
equivariant DT crepant resolution correspondence for local CY$_3$
orbifolds under finite abelian symplectic actions preserving the
holomorphic three-form. On each of the four $A_3$ fixed charts the
Bryan--Cadman--Young orbifold topological vertex (BCY 2013, \emph{J.\
Amer.\ Math.\ Soc.} 28:163--199, \texttt{arXiv:1209.0537}~\S4 Thm.~1)
supplies the local contribution $V^{[\C^3/\Z_4]}$. Global gluing
proceeds by Jun Li degeneration along $\pi$ (Jun Li 2001, \emph{J.\
Differential Geom.} 60:199--293 Thm.~3.15) applied fibrewise. The
$E$-translation tensor factor is handled by the Oberdieck--Pandharipande
$K3\times E$ reduced theory (Oberdieck 2018, \emph{Duke Math.\ J.}
167:3045--3099 Thm.~1 and Rem.~4). The Maulik--Toda cosection
(\emph{Invent.\ Math.} 213:1017--1097, 2018, \S4) survives because
$g_4$ is symplectic, so $H^{0,2}(X^{(4)})^{g_4}$ is one-dimensional.
The weight $\mathrm{wt}\,\FGC_4 = c_4(0)/2 = 1$ follows from Borcherds
1995 (\emph{Invent.\ Math.} 120:161 Thm.~10.1) applied to the
index-one Jacobi lift. The multiplicity formula
\eqref{wn:eq:wave11-w4-mult-N4} is the Gritsenko--Nikulin
denominator-identity expansion of the logarithm of $\FGC_4$
(Gritsenko--Nikulin 1998 \emph{Amer.\ J.\ Math.} 119 Thm.~2.3).
\hfill$\square$

\paragraph{Four-invariant accounting at $N=4$.}
$\kBKM(\FGC_4) = c_4(0)/2 = 1$;
$\kcat(X^{(4)}) = \chi(\cO_{K3})^{g_4}\,\chi(\cO_E)/4 = 0$
(K\"unneth-multiplicative, $\chi(\cO_E) = 0$);
$\kfib^{(4)} = r_4 = 14$;
$\kch$ via $\Phi$ on $\mathcal{A}_{X^{(4)}}$.

\subsection*{Theorem W4.N6 \texorpdfstring{($N=6$, class $6A$,
Niemeier $6D_4$)}{(N=6, class 6A, Niemeier 6D\_4)}}

\ClaimStatusTheorem\ (\emph{Orbifold DT identity at $N = 6$,
untwisted sector}.) Let $g_6$ be the symplectic Nikulin $\Z_6$-action
on $K3$ at Mathieu class $6A$, cycle shape $1^{2}2^{2}3^{2}6^{2}$,
with fixed-locus decomposition $2A_5 + 2D_4$ on the $K3$ fibre
(Hashimoto 2012, \emph{Tohoku Math.\ J.} 64:105--160 Table~3;
obtained by $\Z_6 = \Z_2\times\Z_3$ base-change of local Du-Val
types). Let $\chi(K3^{g_6}) = 2$ and let $\phi^{(g_6)}_{0,1}\in
J_{0,1}(\Gamma_0(6))$ be the weight-zero index-one $g_6$-twined
Jacobi form (EOT 2011 Table~1). The umbral Niemeier anchor is
$N_6 = D_4^{\oplus 6}$ with umbral group $G^{(6D_4)} =
3.\mathrm{Sym}_6$ of order $2160$ (Cheng--Duncan--Harvey 2014
\texttt{arXiv:1307.5793} Table~2; Niemeier 1973 \emph{J.\ Number
Theory} 5, item~\#19); the distinct Niemeier lattice $4A_5 + D_4$
(item~\#18, outer group order $144$) is \emph{not} the $N = 6$
umbral anchor (Wave-11 U1, AP-CY65). The reduced orbifold DT
partition function satisfies
\begin{equation}
\DTred\bigl(X^{(6)}\bigr)(\tau,\sigma,z)
\;=\;-\,\frac{C_6(\tau,z)}{\bigl(\FGC_{6}(\tau,\sigma,z)\bigr)^{2}},
\qquad C_6 = 2\phi^{(g_6)}_{0,1},\quad
\mathrm{wt}\,\FGC_6 = c_6(0)/2 = 1,
\label{wn:eq:wave11-w4-DTred-N6}
\end{equation}
where $\FGC_6$ is the Gritsenko--Cl\'ery paramodular Borcherds lift
of $\phi^{(g_6)}_{0,1}$ with constant term $c_6(0) = 2$
(Gritsenko--Nikulin 1995 Part~II Thm.~2.1). For every root
$\alpha = n\delta + \ell z + m w$,
\begin{equation}
\mult_{\mathfrak{g}^{(6)}}(\alpha)
\;=\;c^{(g_6)}_{K3}\!\bigl(4nm - \ell^{2},\,\ell\bigr).
\label{wn:eq:wave11-w4-mult-N6}
\end{equation}

\emph{Proof.} Bryan--Steinberg 2014 Thm.~1.1 applies to
$[K3\times E/\Z_6]$ because $g_6$ preserves $\omega_{K3}\wedge dz_E$.
The $\Z_6 = \Z_2\times\Z_3$ decomposition of the local orbifold charts
produces two $[\C^3/\Z_6]$ types: at the two $A_5$ points the local
orbifold topological vertex is $V^{[\C^3/\Z_6]}_{A_5}$, and at the two
$D_4$ points $V^{[\C^3/\Z_6]}_{D_4}$; both are computed by BCY 2013
\S4 (abelian subgroup compatibility, factored as the product of
the $\Z_2$ and $\Z_3$ sub-orbifold vertices).
Global gluing uses Jun Li 2001 Thm.~3.15 applied fibrewise along
$\pi$, composed with Bryan--Steinberg 2014 \S5 abelian
wall-crossing. The Oberdieck 2018 $K3\times E$ reduced theory
supplies the $E$-translation factor. The Maulik--Toda 2018 cosection
survives (symplectic $g_6$, $H^{0,2}(X^{(6)})^{g_6} = \C$).
Weight $\mathrm{wt}\,\FGC_6 = c_6(0)/2 = 1$ by Borcherds 1995 Thm.~10.1.
The multiplicity formula \eqref{wn:eq:wave11-w4-mult-N6} is the
Gritsenko--Nikulin denominator identity for $\FGC_6$. The umbral
anchor is $6D_4$ (not $4A_5+D_4$): the mock-modular form $H^{(6)}$
has shadow matching the weight-$1/2$ unary theta series of the
$6D_4$ glue code and pairs with
$\chi_{6D_4}\colon 3.\mathrm{Sym}_6\to\C^\times$ (CDH 2014 Table~2),
incompatible with the order-$144$ outer group of $4A_5+D_4$.
\hfill$\square$
