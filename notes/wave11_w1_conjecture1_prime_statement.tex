\section{Conjecture 1 Prime}
\label{sec:conjecture-one-prime}

Let
\[
  \calNCHL=\{1,2,3,4,6\}.
\]
For each $N\in\calNCHL$, choose a symplectic K3 automorphism
$g_N$ of order $N$ in the Mathieu class used by the CHL construction
and a free order-$N$ translation $\tau_N$ of the elliptic curve $E$.
Set
\[
  X_N=(K3\times E)/\langle g_N\times \tau_N\rangle .
\]
Let $\phi_N=\phi_{g_N,\id}$ be the untwined weak Jacobi form in the
CHL normalisation, with constant term $c_N(0)$, and let
\[
  \Phi_N=\operatorname{Bor}(\phi_N)
\]
be the associated genus-two Borcherds product.  The symbol
$\mathfrak g_N$ denotes a Borcherds--Kac--Moody superalgebra only after
a lattice root datum $\cR_N$, a Weyl chamber $\cW_N$, parity data, and a
geometric or physical host $\cH_N$ have been supplied.

\begin{theorem}[CHL Borcherds weights]
\label{thm:conj-one-prime-chl-weights}
In the CHL row the Borcherds-weight invariant is
\[
  \kBKM(\Phi_N)=\frac{c_N(0)}{2}.
\]
The constants are
\[
\begin{array}{c|ccccc}
N & 1 & 2 & 3 & 4 & 6\\
\hline
c_N(0) & 10 & 8 & 6 & 4 & 2\\
\kBKM(\Phi_N) & 5 & 4 & 3 & 2 & 1
\end{array}
\]
For $N=1$, $\Phi_1=\Delta_5$ and the Gritsenko--Nikulin--Borcherds
construction gives the rank-three denominator superalgebra
$\mathfrak g_{\Delta_5}$.  For $N=2,3,4,6$, the displayed weights are
automorphic constants.  A construction of $\mathfrak g_N$ also requires
the root datum, Weyl chamber, parity assignment, multiplicities, and
denominator identity.
\end{theorem}

\begin{conjecture}[Conjecture 1 prime, CHL scope]
\label{conj:lorgat-one-prime-chl-scope}
For $N\in\calNCHL$, suppose the following data have been fixed:
\begin{enumerate}
\item the CHL Calabi--Yau threefold $X_N$ and the reduced
Donaldson--Thomas chamber in which curve classes have nonzero elliptic
degree;
\item the Borcherds product $\Phi_N=\operatorname{Bor}(\phi_N)$ with
the normalisation of Theorem~\ref{thm:conj-one-prime-chl-weights};
\item for $N\geq 2$, a root datum $\cR_N$, Weyl chamber $\cW_N$, parity
assignment, and host $\cH_N$ whose denominator identity is $\Phi_N$.
\end{enumerate}
Then the level-$N$ comparison asserts the scalar equality
\[
  \DTred(X_N)=C_N\Phi_N^{-2}
\]
for the normalising constant $C_N$ of the reduced theory, and, when the
Borcherds--Kac--Moody datum exists,
\[
  \operatorname{den}(\mathfrak g_N)=\Phi_N,\qquad
  \sdim \mathfrak g_{N,\alpha}
  = \text{the Jacobi coefficient assigned to }\alpha\text{ by }\cR_N .
\]
At $N=1$ this is the constructed $\Delta_5$ denominator algebra
$\mathfrak g_{\Delta_5}$.  At $N=2,3,4,6$ it is a conditional
root-data and host-realisation statement whose proof includes the
weight row as one input.
\end{conjecture}

\begin{remark}[Twined commuting pairs]
\label{rem:conj-one-prime-twined-pairs}
A commuting pair $(g,h)$ with $h\in C_{\Mtf}(g)$ enters the same
comparison only after its weak Jacobi form $\phi_{g,h}$, multiplier,
constant term, cocycle, and coefficient-to-root map have been fixed.
The untwined CHL constants above cannot be copied to all twined sectors.
\end{remark}

\begin{remark}[Orders outside the CHL row]
\label{rem:conj-one-prime-non-chl-orders}
The Mathieu orders $N\in\{5,7,8\}$ are not part of
$\calNCHL$.  Their automorphic and Nikulin weights are boundary
arithmetic.  They support a Borcherds--Kac--Moody or
Donaldson--Thomas comparison only after a root datum, Weyl chamber,
parity, coefficient map, and host are supplied.
\end{remark}

\begin{proposition}[Class-$\mathcal S$ parent]
\label{prop:conj-one-prime-class-s-parent}
The four-dimensional class-$\mathcal S$ parent used in the physical
comparison is
\[
  \cT[A_1,\Sigma_{0,24}],
\]
the $A_1$ theory on the $24$-punctured sphere with maximal regular
punctures.  Its trinion-tube count gives
\[
  (n_v,n_h)=(63,88),\qquad
  c_{4d}=\frac{2n_v+n_h}{12}=\frac{107}{6},\qquad
  c_{2d}=-12c_{4d}=-214 .
\]
\end{proposition}

\paragraph{Source anchors.}
Borcherds 1998 gives the automorphic weight formula
$\kBKM=c_N(0)/2$.  Gritsenko--Nikulin construct the $N=1$ denominator
algebra attached to $\Delta_5$.  The levels $N=2,3,4,6$ require
separate root-data and host constructions before the word
``Borcherds--Kac--Moody'' names an algebra.  The automorphic weight is
the scalar datum recorded above.  The class-$\mathcal S$ parent is fixed by Gaiotto 2009,
Shapere--Tachikawa 2008, and the $A_1$ genus-zero anomaly formula.
