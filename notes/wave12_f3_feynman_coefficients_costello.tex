\title{Feynman Coefficients and BV Anomalies for Six Dimensional
Holomorphic Chern-Simons Theory}
\author{Raeez Lorgat}
\date{22 April 2026}

\begin{document}
\maketitle

\begin{abstract}
Holomorphic Chern-Simons theory on a Calabi-Yau threefold becomes a
Costello BV theory after fixing a holomorphic volume form, a quadratic
gauge coefficient, an elliptic gauge fixing, heat-kernel regularisation,
a completed complex of local functionals, and factorisation descent.
On flat $\C^3$, the Costello propagator has the Bochner-Martinelli
kernel as its off-diagonal limit.  The one-loop two-point graph carries
the quadratic loop Casimir $\Ctwo$ and renormalises the kinetic pairing.
The one-loop quantum-master-equation obstruction in complex dimension
three is a different local cohomology class, governed by the quartic
invariant $I_4^\rho\in\Sym^4(\fg^\vee)^\fg$.  Compact CY3 statements
therefore require anomaly trivialisation, orientation and determinant
framing, regulator and gauge-fixing data, analytic completion, and
factorisation or TCFT sewing hypotheses.  The $d=3$ CY-to-chiral output
is $\Eone$ after specialisation; the $\Etwo$ structure lives on centre,
double, or module layers.  For the toric chart, $\CoHA(\C^3)$ is the
positive half $\Yp(\widehat{\fgl}_1)$; the full $\Winf$ vertex algebra
enters only after double, centre, completion, and evaluation data.
\end{abstract}

\section{BV Datum}
\label{sec:bv-datum}

\begin{construction}[Holomorphic Chern-Simons fields]
\label{cons:hcs-fields}
Let $X$ be a complex threefold with a nowhere-vanishing holomorphic
volume form $\Omega_X$.  Let $\fg$ be a finite-dimensional complex Lie
algebra with nondegenerate invariant symmetric pairing
$\langle-,-\rangle$.  On an open set $U\subset X$, the local BV field
space is the shifted Dolbeault dg Lie algebra
\[
  \cE_U=\Omega^{0,\bullet}(U,\fg)[1],
  \qquad
  Q=\dbar,
  \qquad
  \ell_2(\alpha,\beta)=[\alpha\wedge\beta]_{\fg}.
\]
The unshifted $(0,1)$ component is the holomorphic connection field,
the $(0,0)$ component is the ghost, and the $(0,2)$ and $(0,3)$
components are BV antifields paired by the holomorphic volume form.
\end{construction}

\begin{definition}[BV pairing and classical action]
\label{def:bv-action}
The BV pairing is
\[
  \omega_U(\alpha,\beta)
  =
  \int_U\Omega_X\wedge\langle\alpha,\beta\rangle ,
\]
pairing $\Omega^{0,p}(U,\fg)$ with compactly supported
$\Omega^{0,3-p}(U,\fg)$.  The classical interaction is
\[
  S_{\mathrm{cl}}(a)=
  \int_U\Omega_X\wedge
  \left(
    \frac12\,\langle a,\dbar a\rangle
    +\frac16\,\langle a,[a,a]\rangle
  \right).
\]
For a degree-one connection $A\in\Omega^{0,1}(U,\fg)$ this is the usual
functional
\[
  S_{\mathrm{cl}}(A)=
  \int_U\Omega_X\wedge
  \Tr_\rho\!\left(
    \frac12\,A\,\dbar A+\frac13\,A^3
  \right)
\]
after choosing a trace representation $\rho$.
\end{definition}

\begin{proposition}[Classical master equation]
\label{prop:cme}
The functional $S_{\mathrm{cl}}$ satisfies
\[
  \{S_{\mathrm{cl}},S_{\mathrm{cl}}\}=0 .
\]
\end{proposition}

\begin{proof}
The Hamiltonian vector field of $S_{\mathrm{cl}}$ is the Chevalley
differential of
$(\Omega^{0,\bullet}(U,\fg),\dbar,[-,-])$.  Its square vanishes because
$\dbar^2=0$, $\dbar$ acts as a derivation of the wedge Lie bracket,
the bracket satisfies Jacobi, and $\langle-,-\rangle$ is invariant.
\end{proof}

\section{Costello Renormalisation}
\label{sec:costello-renormalisation}

\begin{definition}[Renormalisation datum]
\label{def:renormalisation-datum}
A perturbative quantisation consists of a Costello gauge-fixing
operator $Q^{\mathrm{GF}}$, heat kernels $K_t$, regularised BV kernels
$P_{\varepsilon,L}$, and effective interactions
$I[L]\in\cO_{\mathrm{loc}}(\cE_X)[[\hbar]]$ satisfying the scale-$L$
quantum master equation
\[
  QI[L]+\frac12\{I[L],I[L]\}_{L}
  +\hbar\Delta_{L}I[L]=0 .
\]
The brackets and Laplacian are defined by the scale-$L$ BV kernel.  The
obstruction at each order in $\hbar$ lies in degree one of the completed
local BV complex.
\end{definition}

\begin{remark}[What the propagator controls]
\label{rem:propagator-controls}
The propagator is a homotopy for the free Dolbeault complex.  It fixes
representatives for Feynman integrals and counterterms.  The QME is a
separate cohomological condition in the Costello local functional
complex.
\end{remark}

\section{Flat Bochner-Martinelli Kernel}
\label{sec:bm-kernel}

\begin{proposition}[Heat-kernel propagator on $\C^3$]
\label{prop:bm-propagator}
Let $X=\C^3$ with Euclidean Hermitian metric, holomorphic volume
$\Omega=dz_1\wedge dz_2\wedge dz_3$, and
$Q^{\mathrm{GF}}=\dbarstar$.  For $0<\varepsilon<L<\infty$, set
\[
  P_{\varepsilon,L}(z,w)
  =
  \int_{\varepsilon}^{L}
  (\dbarstar\otimes1)K_t(z,w)\,dt .
\]
Then
\[
  \dbar P_{\varepsilon,L}=K_L-K_\varepsilon .
\]
Off the diagonal, as $\varepsilon\to0$ and $L\to\infty$,
$P_{\varepsilon,L}$ converges to the Bochner-Martinelli kernel
\[
  P_{\mathrm{BM}}(z,w)
  =
  \frac{2!}{(2\pi i)^3}
  \sum_{k=1}^{3}(-1)^{k-1}
  \frac{\overline{z_k-w_k}}{\|z-w\|^6}\,
  d\bar z_1\wedge\cdots\widehat{d\bar z_k}\cdots\wedge d\bar z_3 .
\]
Its singularity has homogeneity $r^{-5}$ in the real normal variable
$r=\|z-w\|$.
\end{proposition}

\begin{proof}
The scalar flat heat kernel is
$(4\pi t)^{-3}\exp(-\|z-w\|^2/(4t))$.  Applying $\dbarstar$ differentiates
the Gaussian in the antiholomorphic variable and supplies the numerator
$\overline{z_k-w_k}$.  Integration over $t\in(\varepsilon,L)$ gives a
smooth homotopy kernel.  The identity
$\dbar P_{\varepsilon,L}=K_L-K_\varepsilon$ is the heat-kernel
telescoping identity for
$\Delta_{\dbar}=\dbar\dbarstar+\dbarstar\dbar$.  Removing the cutoffs
away from the diagonal gives the displayed Bochner-Martinelli
representative.
\end{proof}

\begin{corollary}[Metric representative]
\label{cor:metric-representative}
For the Euclidean metric and the gauge fixing $\dbarstar$, the
Bochner-Martinelli kernel is the canonical off-diagonal representative
of the propagator homotopy class.  A homotopy of Costello gauge fixings
changes this representative by a $Q$-exact kernel and changes the
renormalised observables by the corresponding Costello equivalence once
the QME obstruction has been killed.
\end{corollary}

\section{Local Feynman Coefficients}
\label{sec:local-feynman-coefficients}

\begin{definition}[Graph weights]
\label{def:graph-weights}
Let $\{T_a\}$ be a basis for $\fg$, let $\{T^a\}$ be the dual basis for
the invariant pairing, and raise or lower indices with this pairing.
Set
\[
  f_{abc}=\langle T_a,[T_b,T_c]\rangle .
\]
A trivalent Feynman graph has vertex tensor $f_{abc}$, internal edge
kernel $P_{\varepsilon,L}$ contracted with the inverse pairing on
$\fg$, integration over internal vertices against $\Omega_X$, and the
standard factor $|\operatorname{Aut}(\Gamma)|^{-1}$.
\end{definition}

\begin{proposition}[Tree coefficient]
\label{prop:tree-coefficient}
For compactly supported sources $\alpha_i$ with disjoint support, the
connected tree-level three-point coefficient on $\C^3$ is
\[
\begin{aligned}
  \bigl\langle
  \cA^{a_1}(\alpha_1)
  \cA^{a_2}(\alpha_2)
  \cA^{a_3}(\alpha_3)
  \bigr\rangle_{\mathrm{tree,conn}}
  &=
  \hbar^2 f^{a_1a_2a_3}
  \int_{\C^3}\Omega\wedge
  (P_{\mathrm{BM}}\alpha_1)\wedge
  (P_{\mathrm{BM}}\alpha_2)\wedge
  (P_{\mathrm{BM}}\alpha_3).
\end{aligned}
\]
For an abelian gauge algebra, $f_{abc}=0$ and the interaction vanishes.
\end{proposition}

\begin{proof}
Each external contraction contributes one propagator and hence one
factor of $\hbar$; the cubic vertex appears from
$S_{\mathrm{cl}}/\hbar$ and contributes $\hbar^{-1}$.  The colour tensor
is the invariant cubic tensor $f^{a_1a_2a_3}$.  Disjoint support keeps
the Bochner-Martinelli kernels away from total collision; coincident
support is handled by Costello's heat-kernel extension.
\end{proof}

\begin{definition}[Quadratic loop Casimir]
\label{def:quadratic-casimir}
The invariant pairing fixes the adjoint loop scalar $\Ctwo$ by
\[
  \sum_{b,c}f^{a_1bc}f^{a_2}{}_{bc}
  =
  \Ctwo(\fg,\langle-,-\rangle)\,
  \langle T^{a_1},T^{a_2}\rangle .
\]
For a simple Lie algebra in the usual adjoint normalisation,
$\Ctwo=2h^\vee$.  If the pairing is induced from a trace
representation $\rho$, the relation between $\Ctwo$ and the endomorphism
$\sum_a\rho(T_a)\rho(T^a)$ depends on the Dynkin index of $\rho$.
\end{definition}

\begin{proposition}[Two-point local term]
\label{prop:two-point-local-term}
The one-loop two-point graph has colour contraction
\[
  \sum_{b,c}f^{a_1bc}f^{a_2}{}_{bc}
  =
  \Ctwo\,\langle T^{a_1},T^{a_2}\rangle .
\]
The raw product of two Bochner-Martinelli kernels has singular order
\[
  r^{-10},
  \qquad
  \int_\varepsilon^R r^5r^{-10}\,dr
  =
  \frac14(\varepsilon^{-4}-R^{-4}).
\]
Thus its scaling degree is $10$ in real dimension $6$, and an extension
across the diagonal is unique only after specifying local distributions
of derivative order at most $10-6=4$.  Costello heat-kernel
regularisation chooses such an extension.  After subtraction of the
power divergences, the scale-dependent kinetic projection in a flat
heat-kernel scheme has the form
\[
  S_{\mathrm{ct}}^{(2)}
  =
  -\hbar\,c_{\mathrm{kin}}^{\mathrm{hk}}\,
  \Ctwo\,
  \log(L/\varepsilon)
  \int_X\Omega_X\wedge\langle\cA,\dbar\cA\rangle ,
\]
where $c_{\mathrm{kin}}^{\mathrm{hk}}$ is fixed by the heat-kernel
normalisation and the projection to the kinetic local functional.
\end{proposition}

\begin{proof}
Closing two legs in the cubic vertex gives the adjoint contraction
defining $\Ctwo$.  The two remaining external fields have the tensor
type of the kinetic functional.  The displayed radial integral is the
diagonal scaling computation for the product of two
Bochner-Martinelli kernels in the six real normal directions; it fixes
the extension degree, separate from the quartic obstruction class.  The local
counterterms are precisely the allowed extensions of that distribution
in Costello's completed local functional complex, and the kinetic
projection gives the displayed degree-zero term.
\end{proof}

\begin{remark}[Coefficient versus obstruction]
\label{rem:c2-versus-i4}
The tensor $\Ctwo$ governs two-point field-strength renormalisation.
It has BV degree zero.  The QME anomaly has BV degree one and is the
quartic class of Section~\ref{sec:quartic-obstruction}.
\end{remark}

\section{Quartic One-Loop Obstruction}
\label{sec:quartic-obstruction}

\begin{definition}[Symmetric quartic invariant]
\label{def:quartic-invariant}
For a representation $\rho:\fg\to\End(V)$, define
\[
  I_4^\rho(x_1,x_2,x_3,x_4)
  =
  \frac1{4!}\sum_{\sigma\in S_4}
  \Tr_V\!\left(
    \rho(x_{\sigma(1)})\rho(x_{\sigma(2)})
    \rho(x_{\sigma(3)})\rho(x_{\sigma(4)})
  \right).
\]
Then $I_4^\rho\in\Sym^4(\fg^\vee)^\fg$.
\end{definition}

\begin{proposition}[One-loop QME obstruction in complex dimension three]
\label{prop:qme-obstruction}
The nonabelian one-loop gauge obstruction for holomorphic
Chern-Simons theory in complex dimension three is represented in the
local Chevalley complex by
\[
  \Theta^\rho_U(\alpha_0,\alpha_1,\alpha_2,\alpha_3)
  =
  \int_U
  \left[
  I_4^\rho\bigl(
    \alpha_0,\partial\alpha_1,\partial\alpha_2,\partial\alpha_3
  \bigr)
  \right]_{(3,3)},
  \qquad
  \alpha_i\in\Omega^{0,\bullet}_c(U,\fg).
\]
Its obstruction class is
\[
  \kanom(X,\fg,\rho)
  =
  [\,\hbar\,\Theta^\rho\,]
  \in
  H^1_{\mathrm{loc}}
  \bigl(\Omega^{0,\bullet}(X,\fg)[1],\cO_{\mathrm{loc}}\bigr).
\]
For $\rho=\ad$, the same Costello-Li descent class is written in
ghost-field components as
\[
  \int_X \Tr_{\ad}(c\,F_A^3).
\]
\end{proposition}

\begin{proof}
The open-string one-loop anomaly graph is the wheel with four external
gauge insertions.  Its Lie factor is the symmetrised trace of four Lie
algebra inputs, hence an element of $\Sym^4(\fg^\vee)^\fg$.  The form
factor carries three holomorphic derivatives, giving a holomorphic top
density on a complex threefold.  Taking the $(3,3)$ component gives a
degree-one local functional.  Costello obstruction theory identifies
this class as the first obstruction to solving the scale-$L$ QME.
\end{proof}

\begin{corollary}[Abelian and quartic-trivial cases]
\label{cor:abelian-quartic-trivial}
If $\fg$ is abelian, the cubic interaction vanishes and the theory is
free.  If $I_4^\rho=0$, or if $\Theta^\rho$ is exact after the specified
counterterm package is included, the one-loop gauge obstruction
vanishes for that datum.
\end{corollary}

\begin{remark}[Flat local scope]
\label{rem:flat-scope}
On $X=\C^3$, translation invariance and the Stein contraction give the
standard local perturbative model.  The anomaly slot remains the
quartic class of Proposition~\ref{prop:qme-obstruction}.  A quantum
closure statement for a chosen nonabelian open theory requires
vanishing or trivialisation of
$\kanom(\C^3,\fg,\rho)$ in the chosen Costello local functional complex.
\end{remark}

\section{Compact Calabi-Yau Threefolds}
\label{sec:compact-cy3}

\begin{hypothesis}[Compact hCS package]
\label{hyp:compact-package}
A compact nonabelian holomorphic Chern-Simons quantisation theorem on a
Calabi-Yau threefold requires the following data.
\begin{enumerate}[label=\textup{(\roman*)}]
\item A compact Calabi-Yau threefold $(X,\Omega_X)$ with orientation and
determinant-line framing compatible with the BV pairing.
\item A quadratic gauge Lie algebra or dg Lie algebra
$(\fg,\langle-,-\rangle)$ and, when traces are used, a representation
$\rho$ fixing $I_4^\rho$ and the loop scalar $\Ctwo$.
\item An elliptic Costello gauge fixing, heat-kernel regulator
$P_{\varepsilon,L}$, and RG-compatible counterterm scheme.
\item A completed space of local functionals and observables on which
$\Delta_L$, $\{-,-\}_L$, and RG flow are defined.
\item A factorisation or TCFT sewing structure with descent from
polydisks to $X$.
\item A trivialisation of $\kanom(X,\fg,\rho)$ by vanishing in local BV
cohomology, an explicit local counterterm, or a closed-sector
Green-Schwarz term.
\end{enumerate}
\end{hypothesis}

\begin{theorem}[Conditional compact quantisation]
\label{thm:conditional-compact-quantisation}
Assume Hypothesis~\ref{hyp:compact-package}.  If
$\kanom(X,\fg,\rho)=0$ in the Costello local BV obstruction complex
after the chosen counterterms and closed-sector contributions are
included, then the effective interactions can be chosen to satisfy the
QME through one loop.  Under the analogous higher-loop obstruction
vanishing or counterterm hypotheses, the completed observables form a
quantum factorisation algebra on $X$.  If a required obstruction class
survives, the specified regulator and field content fail to produce a
QME solution in that complex.
\end{theorem}

\begin{proof}
Costello's obstruction theory solves the QME inductively in powers of
$\hbar$.  At the first quantum step, the obstruction is the degree-one
local class of Proposition~\ref{prop:qme-obstruction}, modified by the
specified local or closed-sector counterterms.  Vanishing gives the
first quantum correction.  The same induction applies at higher loop
order inside the completed local functional complex.  A surviving class
is exactly the failure of the QME in that complex.
\end{proof}

\begin{remark}[Characteristic numbers and gauge traces]
\label{rem:characteristic-numbers}
The topological Euler number and the open gauge anomaly are separate
inputs.  For $K3\times E$,
\[
  \chi_{\mathrm{top}}(K3\times E)=24\cdot0=0,
  \qquad
  \kcat(K3\times E)=\chi(\cO_{K3})\chi(\cO_E)=2\cdot0=0.
\]
These characteristic-number vanishings leave the quartic gauge class
dependent on $(\fg,\rho,I_4^\rho)$, the regulator, and the descent
representative.  For the quintic threefold $X_5$,
\[
  \int_{X_5}c_3(TX_5)=\chi_{\mathrm{top}}(X_5)=-200.
\]
This base characteristic contribution is insufficient to determine the
open gauge obstruction without the trace normalisation and BV descent
datum.
\end{remark}

\section{Observables and Separations}
\label{sec:observables-separations}

\begin{proposition}[Classical and quantum observables]
\label{prop:observables}
For $U\subset X$,
\[
  \Obs^{\mathrm{cl}}_{\hCS}(U)
  =
  C^\bullet_{\CE,\mathrm{cont}}
  \bigl(\Omega^{0,\bullet}_c(U,\fg)[1]\bigr)
  =
  \widehat{\Sym}
  \bigl(\Omega^{0,\bullet}_c(U,\fg)[1]^\vee[-1]\bigr),
\]
with differential induced by $\dbar$ and the bracket on $\fg$.  A
Costello renormalisation datum gives
\[
  \Obs^q_{\hCS,L}(U)
  =
  \left(
    \Obs^{\mathrm{cl}}_{\hCS}(U)[[\hbar]],
    d_{\Obs}^{\mathrm{cl}}+\{I[L],-\}_L+\hbar\Delta_L
  \right)
\]
when the QME is satisfied.
\end{proposition}

\begin{proof}
The first formula is the Chevalley cochain algebra of the Dolbeault
local dg Lie algebra.  The second is Costello's BV deformation of this
cochain algebra by the renormalised interaction, regularised BV bracket,
and BV Laplacian.
\end{proof}

\begin{remark}[Fields, antifields, and Hochschild polyvectors]
\label{rem:field-polyvector-separation}
The hCS fields are
$\Omega^{0,\bullet}(X,\operatorname{ad}P)[1]$.  Their BV antifields are
the complementary Dolbeault components paired by
$\int_X\Omega_X\wedge\Tr_\rho(-\, -)$.  Hochschild polyvectors of a
Calabi-Yau category are a different object:
\[
  \HH^\bullet(\Perf(\C^3))
  \simeq
  \C[x,y,z]\otimes
  \bigwedge\nolimits^\bullet
  \langle\partial_x,\partial_y,\partial_z\rangle .
\]
They control intrinsic categorical deformations and the derived centre.
A finite-dimensional hCS gauge algebra enters only after a framing
object or external gauge coefficient is chosen.
\end{remark}

\section{Boundary Specialisation}
\label{sec:boundary-specialisation}

\begin{proposition}[$d=3$ output level]
\label{prop:d3-output-level}
Let $C\subset X$ be a holomorphic curve and suppose the hCS quantum
package has been solved on a neighbourhood of $C$.  Integration of
completed quantum observables in the normal directions gives
\[
  \mathcal A_C
  =
  \int_{N_{C/X}/C}\Obs^q_{\hCS}
  \in
  \operatorname{Alg}^{\mathrm{ch}}_{\Eone}(C).
\]
Thus the specialised output in complex dimension three is $\Eone$.
The $\Etwo$ structure belongs to a centre, double, or module layer, for
example
\[
  \cZ\!\left(\Rep^{\Eone}(\mathcal A_C)\right),
  \qquad
  D_\hbar(\mathcal A_C),
\]
under the relevant chiral centre or Hall-Drinfeld double hypotheses.
\end{proposition}

\begin{proof}
Under the stated quantum package, stage one of the CY-to-chiral
construction produces an $\Ethree$ holomorphic factorisation object.
Stage two integrates along the complex surface normal to $C$ and then
restricts to the reference curve.  Factorisation homology lowers the
operadic level from $\Ethree$ to $\Eone$ on the curve.  Braided $\Etwo$
data require the representation centre, a Hall-Drinfeld double, or an
equivalent module-layer construction; the native structure on
$\mathcal A_C$.
\end{proof}

\begin{proposition}[Toric CoHA normalisation]
\label{prop:c3-coha-normalisation}
For the toric chart $\C^3$, the equivariant critical CoHA of the
one-vertex three-loop quiver with potential
$W=\Tr(x[y,z])$ satisfies
\[
  \CoHA(\C^3)\simeq \Yp(\widehat{\fgl}_1),
\]
the positive half of the affine Yangian in the Schiffmann-Vasserot
normalisation.  The full $\Winf$ vertex algebra appears after the chain
\[
  \Yp(\widehat{\fgl}_1)
  \longrightarrow
  D^\wedge_\hbar\!\left(\Yp(\widehat{\fgl}_1)\right)
  \longrightarrow
  \text{completed Cartan and evaluation representation}
  \longrightarrow
  \Winf .
\]
\end{proposition}

\begin{proof}
The first is the Kontsevich-Soibelman and Schiffmann-Vasserot
identification of the toric critical CoHA with the shuffle positive
half of the affine Yangian.  Drinfeld double, Cartan completion, and
evaluation data are additional structure; they are the operations
producing the vertex-algebraic $\Winf$ layer.
\end{proof}

The local hCS calculation above supplies the following tensors and
constants:

  f_{abc}=\langle T_a,[T_b,T_c]\rangle,\qquad
  \sum_{b,c}f^{a_1bc}f^{a_2}{}_{bc}

\[
  I_4^\rho(x_1,x_2,x_3,x_4)

  \rho(x_{\sigma(1)})\rho(x_{\sigma(2)})

  \bigr),
\]
and the diagonal extension bound $10-6=4$ for the raw two-propagator

  \kanom(X,\fg,\rho)=[\,\hbar\,\Theta^\rho\,].
\]
These data are insufficient to determine the numerical shadow
coefficients of a chosen $\Winf$ normalisation by themselves.
\end{proposition}

\begin{proof}
The tensors are read directly from the cubic hCS vertex, the two-point
graph, and the four-wheel anomaly graph.  The $\Winf$ coefficients
depend on the one-dimensional chiral OPE normalisation, the double or
centre passage from $\Yp(\widehat{\fgl}_1)$, the completion, and the
evaluation representation.  Those choices are outside the local
six-dimensional BV coefficient computation.
\end{proof}

\end{enumerate}
\end{hypothesis}
