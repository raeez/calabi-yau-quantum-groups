\section*{Kontsevich--Tamarkin $E_3$-formality over $\Q$ and the $\mathrm{GRT}_1(\Q)$-torsor on Stage-$1$ of $\PhiFA_3$}
\label{sec:kt-e3-formality-grt1-torsor-cfg2026}

Fix the following categorical data. Let $\cC$ be a CY$_3$ category in
$\CY_3\text{-}\Cat^{\mathrm{dg}}$ (Definition~\ref{def:cy-d-cat-dg}),
so $\cC$ is smooth and proper over $\Q$ (rational coefficients are
forced: every construction below depends on a Drinfeld associator and
every rational associator is defined over $\Q$; the $\Q$-linearity is
not a convention, it is a theorem of Drinfeld $1991$). Let $X$ be a
smooth complex CY$_3$ with a chosen volume form $\Omega_X \in H^0(X,
K_X)$. The Stage-$1$ assembly of Proposition~\ref{prop:phifa-infty1-kernel}
factors as
\[
 \cC
 \xrightarrow{\;\mathrm{(a)}\;}
 E_3\text{-}\mathrm{Alg}\bigl(\HH^\bullet(\cC)\bigr)
 \xrightarrow{\;\mathrm{(b)}\;}
 \mathrm{FactAlg}^{\mathrm{top}}_{E_3}(X)
 \xrightarrow{\;\mathrm{(c)}\;}
 \EdHolFA(X),
\]
with the three morphisms named formality, locality, holomorphic twist.
This section constructs the first morphism from the configuration-space
ground up, identifies the space of such constructions as a torsor under
a pro-unipotent group $\mathrm{GRT}_1(\Q)$, and places the CFG $2026$
identification of observables of topological Chern--Simons as an $E_3$
algebra inside the same torsor. A CY$_3$ self-duality
$\omega_X \cong \cO_X$ does not rigidify the torsor: the torsor is a
universal invariant of the pair (little-$3$-disks operad, Gerstenhaber
operad of shift $-2$), independent of $X$.

\subsection*{The chain-level little $3$-disks operad and the Gerstenhaber operad of shift $-2$}

\begin{definition}[Little $3$-disks operad $E_3 = D_3$]
\label{def:little-3-disks-cfg}
\ClaimStatusDefinitional
For $k \geq 0$ let $D_3(k)$ be the topological space of $k$-tuples
$(f_1, \ldots, f_k)$ of affine embeddings $f_i \colon D^3 \hookrightarrow D^3$
of the open unit $3$-disk into itself, with pairwise disjoint images.
The operadic composition $D_3(k) \times D_3(m_1) \times \cdots \times
D_3(m_k) \to D_3(m_1 + \cdots + m_k)$ is substitution. The Fulton--MacPherson
compactification $\overline{F}(k, \R^3)^{\mathrm{FM}}$ is homotopy-equivalent
to $D_3(k)$ (Fulton--MacPherson $1994$, \textit{Ann. Math.} $139$): it
replaces open disks by formal collisions. The singular chain operad
$C_\bullet(D_3; \Q)$ is a dg-operad in $\mathrm{Ch}(\Q)$; its cohomology
ring is the Gerstenhaber operad of shift $-2$,
\[
 H^\bullet(D_3(k); \Q) \;=\; \mathrm{Ger}_3(k),
\]
with $\mathrm{Ger}_3(k)$ generated by a commutative product in degree $0$ and a
graded Lie bracket $\{-, -\}$ of degree $-2$ (F. Cohen $1976$,
\textit{LNM} $533$: the cohomology of configuration spaces in $\R^n$
is $\mathrm{Ger}_n$, the Gerstenhaber operad of shift $1 - n$).
\end{definition}

\begin{remark}[Side-by-side with CFG $2026$]
\label{rem:cfg-2026-d3-little-disks}
CFG $2026$ Section $1.2$ fixes precisely the same operad
$E_3 \simeq D_3$ as the source of their $E_3$-structure on local
observables of topological Chern--Simons. Their Theorem $A$ (E_3$
$-algebra structure) and the present Proposition~\ref{prop:phifa-infty1-kernel}
evaluate the \emph{same} dg-operad $C_\bullet(D_3; \Q)$ on two
different targets: CFG $2026$ targets $\Obs^{\mathrm{CS}}_{\mathrm{top}}
(M^3, \frakg)$ on a topological $3$-manifold with compact gauge;
the CY$_3$ Stage-$1$ assembly targets $\HH^\bullet(\cC)$ on a complex
CY$_3$ variety. The operadic input is identical; the difference is
where the $E_3$-algebra lives.
\end{remark}

\begin{definition}[Gerstenhaber operad of shift $-2$]
\label{def:gerstenhaber-shift-minus-2}
\ClaimStatusDefinitional
$\mathrm{Ger}_3 \in \mathrm{Op}(\mathrm{Ch}(\Q))$ is the quadratic operad
generated by a commutative associative product
$\mu \colon \mathrm{Ger}_3(2) \to \mathrm{Ger}_3(1)$ of degree $0$ and a graded Lie
bracket $\{-, -\} \colon \mathrm{Ger}_3(2) \to \mathrm{Ger}_3(1)$ of degree $-2$,
subject to the Leibniz relation
\[
 \{a, bc\} \;=\; \{a, b\}\, c \;+\; (-1)^{(|a| - 2)|b|}\, b\, \{a, c\}.
\]
The shift convention is Cohen's: the bracket on $\HH^\bullet(\cC)$
arising in the CY$_3$ setting has degree $-2$ after the Serre shift by
$d = 3$, so that the Gerstenhaber bracket and the Schouten--Nijenhuis
bracket on polyvector fields agree on the HKR-decomposed side. In the
topological CS setting of CFG the same $\mathrm{Ger}_3$ controls the $E_3$-algebra
of local observables.
\end{definition}

\begin{proposition}[Formality question: $C_\bullet(D_3; \Q) \simeq \mathrm{Ger}_3$ in $\mathrm{Op}(\mathrm{Ch}(\Q))$?]
\label{prop:e3-formality-question}
\ClaimStatusTheorem
The little $3$-disks operad is formal over $\Q$: there exists a
zig-zag of quasi-isomorphisms
\[
 C_\bullet(D_3; \Q) \;\xleftarrow{\;\sim\;}\; \mathrm{Graphs}_3 \;\xrightarrow{\;\sim\;}\; \mathrm{Ger}_3
\]
in $\mathrm{Op}(\mathrm{Ch}(\Q))$. Here $\mathrm{Graphs}_3$ is the
Kontsevich graph operad (Kontsevich $1999$
\textit{Operads and motives in deformation quantisation}; Tamarkin
$2003$ \textit{Another proof of Kontsevich's formality theorem}; extended
to $n \geq 3$ by Fresse--Willwacher $2020$, \textit{Acta Math.} $224$
Theorem $A$: $E_n$-operads are rationally formal for every $n \geq 2$).
\end{proposition}

\begin{proof}[Proof outline]
The left quasi-isomorphism $\mathrm{Graphs}_3 \xrightarrow{\sim}
C_\bullet(D_3; \Q)$ is the Kontsevich integral: for each labelled graph
$\Gamma$ with $k$ external vertices one integrates the product of
$\omega_{ij} = \mathrm{Vol}(S^2) / |x_i - x_j|^3$ over the
configuration space of internal vertices, giving a chain on $D_3(k)$.
The integral converges by Kontsevich's degree-counting on compactified
configuration spaces. The right quasi-isomorphism $\mathrm{Graphs}_3
\xrightarrow{\sim} \mathrm{Ger}_3$ identifies $\mathrm{Graphs}_3$'s
cohomology as the Gerstenhaber operad by explicit generator-counting:
in cohomological degree $0$ the only nontrivial graph is a single edge
(product $\mu$); in cohomological degree $-2$ the only nontrivial graph
is a single edge with an orientation flip (bracket $\{-, -\}$);
higher-valent graphs give cohomologically trivial classes by the Cohen
computation of $H^\bullet(F(k, \R^3); \Q)$.
\end{proof}

\subsection*{The Drinfeld associator over $\Q$ and the space of formality quasi-isomorphisms}

\begin{definition}[Associator set $\mathrm{Assoc}(\Q)$]
\label{def:drinfeld-associator-q}
\ClaimStatusDefinitional
$\mathrm{Assoc}(\Q)$ is the set of group-like elements
$\Phi \in \hat{U}(\mathfrak{t}_3)_{\geq 2}$ in the completed universal
enveloping algebra of the infinitesimal-pure-braid Lie algebra on three
strands over $\Q$ satisfying the pentagon and the two hexagon relations
(Drinfeld $1989$ \emph{Quasi-Hopf algebras}, \textit{Leningrad Math. J.}
$1$, $1419$--$1457$; Drinfeld $1991$ \emph{On quasitriangular
quasi-Hopf algebras and a group closely connected with
$\mathrm{Gal}(\overline{\Q}/\Q)$}, \textit{Leningrad Math. J.} $2$,
$829$--$860$; $\mathfrak{t}_3$ generated by $t_{12}, t_{13}, t_{23}$
with $[t_{ij}, t_{ik} + t_{jk}] = 0$ and $[t_{ij}, t_{kl}] = 0$ for
disjoint pairs). The Knizhnik--Zamolodchikov associator
$\Phi_{\mathrm{KZ}}$ is Drinfeld's transcendental element over $\C$
with coefficients multiple zeta values $\zeta(s_1, \ldots, s_r)$; an
associator in $\mathrm{Assoc}(\Q)$ exists by Drinfeld $1991$ Theorem $A'$
(a $\Q$-associator is constructed iteratively from degree data).
\end{definition}

\begin{theorem}[Torsor structure on $\mathrm{Assoc}(\Q)$: Drinfeld $1991$]
\label{thm:drinfeld-assoc-torsor}
\ClaimStatusTheorem
The set $\mathrm{Assoc}(\Q)$ is a torsor under the pro-unipotent
Grothendieck--Teichmüller group $\mathrm{GRT}_1(\Q)$: if
$\Phi, \Phi' \in \mathrm{Assoc}(\Q)$, there is a unique
$g \in \mathrm{GRT}_1(\Q)$ with $\Phi' = g \cdot \Phi$, where
$\mathrm{GRT}_1(\Q)$ is the pro-unipotent $\Q$-algebraic group
$\mathrm{Spec}\bigl(H^0(\mathrm{GC}_2; \Q)^\vee\bigr)$ of the
Kontsevich graph complex in arity $2$
(Drinfeld $1991$ §$5$; Willwacher $2015$ \textit{Invent. Math.} $200$
Theorem $1.1$: $\mathrm{GRT}_1(\Q) \cong \exp(H^0(\mathrm{GC}_2))$ as
pro-unipotent groups, with $\mathrm{GC}_2$ the Kontsevich graph complex
in $2$ directions).
\end{theorem}

\begin{remark}[Why $\mathrm{GRT}_1$ and not $\mathrm{GRT}_1^{\mathrm{super}}$ at $d = 3$]
\label{rem:grt1-vs-super-grt1-d3}
At $d = 2$ (K3, abelian surface) the formality torsor is sometimes
presented as a $\mathrm{GRT}_1^{\mathrm{super}}(\Q)$-torsor because the
bracket on $\HH^\bullet(\cC)$ has odd degree and the associator carries
super-sign conventions (remark~\ref{rem:motivic-grt1-torsor-k3};
Etingof--Kazhdan). At $d = 3$ the bracket has degree $-2$, an even
integer, so the Grothendieck--Teichmüller torsor is the non-super
$\mathrm{GRT}_1(\Q)$. The super-refinement at $d = 2$ is the Koszul sign
shift induced by odd-degree operators; at $d = 3$ this shift is trivial.
Both torsors share the same Lie algebra $H^0(\mathrm{GC}_2)$ of graph
cocycles; the super-decoration affects the representation, not the
underlying group.
\end{remark}

\begin{definition}[Space of $E_3$-formality quasi-isomorphisms]
\label{def:e3-formality-space}
\ClaimStatusDefinitional
The space of $E_3$-formality quasi-isomorphisms over $\Q$ is
\[
 \mathrm{Form}_3(\Q)
 \;:=\;
 \bigl\{
  F \colon C_\bullet(D_3; \Q) \xrightarrow{\;\sim\;} \mathrm{Ger}_3
  \mid
  F \text{ operadic quasi-isomorphism in } \mathrm{Op}(\mathrm{Ch}(\Q))
 \bigr\}
 /
 \text{operadic homotopy}.
\]
By Tamarkin $2003$ and Fresse--Willwacher $2020$ this is non-empty.
\end{definition}

\begin{theorem}[The formality space as a $\mathrm{GRT}_1(\Q)$-torsor]
\label{thm:form3-grt1-torsor}
\ClaimStatusTheorem
The set $\mathrm{Form}_3(\Q)$ carries a natural free and transitive
action of $\mathrm{GRT}_1(\Q)$: if $F, F' \in \mathrm{Form}_3(\Q)$, there
exists a unique $g \in \mathrm{GRT}_1(\Q)$ with $F' \simeq g \cdot F$ as
operadic homotopy classes. Equivalently,
\[
 \mathrm{Form}_3(\Q) \;\simeq\; \mathrm{GRT}_1(\Q)\text{-torsor},
\]
(Tamarkin $2003$ §$3$: each associator $\Phi \in \mathrm{Assoc}(\Q)$
constructs an $E_2$-formality quasi-isomorphism and the map is
$\mathrm{GRT}_1$-equivariant; Fresse--Willwacher $2020$ Theorem $B$
extends this to $n \geq 3$). The torsor is non-trivial: the
$\mathrm{GRT}_1$-action has free orbits.
\end{theorem}

\begin{proof}[Proof outline at $n = 3$]
From an associator $\Phi \in \mathrm{Assoc}(\Q)$ one constructs a
formality quasi-isomorphism $F_\Phi$ via the Kontsevich graph-weighted
integrals: each graph $\Gamma$ in $\mathrm{Graphs}_3(k)$ receives a
weight $w_\Phi(\Gamma) \in \Q$ read off the expansion of $\Phi$ in the
universal associator variable, and $F_\Phi$ sends a cohomology class to
a sum $\sum_\Gamma w_\Phi(\Gamma) \cdot \mathrm{KontInt}(\Gamma)$ on
configuration space. The association $\Phi \mapsto F_\Phi$ descends to a
bijection on torsor orbits by Fresse--Willwacher $2020$
Theorem $3.1$: $\pi_0(\mathrm{Form}_3) = \mathrm{Assoc}(\Q) /
\mathrm{inner} = \mathrm{GRT}_1(\Q)$-torsor. The freeness is Willwacher
$2015$ Corollary $1.3$: the identity of $\mathrm{GRT}_1$ is the only
inner automorphism of $\mathrm{Ger}_3$ commuting with every formality map.
\end{proof}

\subsection*{The chain-level $E_3$-algebra on $\HH^\bullet(\cC)$ at $d = 3$}

\begin{theorem}[Chain-level $E_3$-algebra on $\HH^\bullet(\cC)$; Stage-$1$ upgrade]
\label{thm:chain-level-e3-algebra-hochschild-d3}
\ClaimStatusTheorem
Let $\cC \in \CY_3\text{-}\Cat^{\mathrm{dg}}$ and $F \in
\mathrm{Form}_3(\Q)$ a chosen $E_3$-formality quasi-isomorphism. Then
$F$ determines an $E_3$-algebra structure on the Hochschild cochain
complex $\CC^\bullet(\cC, \cC)$ in $\mathrm{Ch}(\Q)$, whose underlying
graded-commutative product is the cup product and whose underlying Lie
bracket in degree $-2$ is the Gerstenhaber bracket. The induced
$E_3$-algebra structure on $\HH^\bullet(\cC) := H^\bullet
(\CC^\bullet(\cC, \cC))$ is the Gerstenhaber operad of shift $-2$:
\[
 \HH^\bullet(\cC) \;\in\; \mathrm{Ger}_3\text{-}\mathrm{Alg}(\mathrm{Ch}(\Q)),
\]
with the symmetric pairing of CY$_3$-degree $-3$ making it into a
cyclic $\mathrm{Ger}_3$-algebra (Ginzburg $2006$ \textit{arXiv:math/0612139};
Kontsevich--Soibelman $2008$ \textit{arXiv:0806.0107}). Two different
formality quasi-isomorphisms $F, F' \in \mathrm{Form}_3(\Q)$ give
quasi-isomorphic $E_3$-algebra structures on $\CC^\bullet(\cC, \cC)$
related by a $\mathrm{GRT}_1(\Q)$-element; the induced cohomological
$\mathrm{Ger}_3$-structure on $\HH^\bullet(\cC)$ is identical.
\end{theorem}

\begin{proof}[Proof]
The chain-level Gerstenhaber bracket $\{-, -\}_G$ on
$\CC^\bullet(\cC, \cC)$ is Gerstenhaber $1963$, with degree $-1$ on
$\CC^\bullet$; after the Serre shift by $d = 3$ on the CY$_3$ side it
becomes degree $-2$ in the convention where $\CC^\bullet(\cC, \cC)$
has been reindexed by $[-d] = [-3]$ so that $\HH^\bullet$ is
concentrated in degrees $0, 1, 2, 3$ and the Yoneda bracket on
$\Ext^\bullet(\cC, \cC)$ is of degree $-2 = 1 - 3$. The cup product is
Gerstenhaber's $\cup$; its commutativity up to $\{-,-\}_G$-correction is
the Gerstenhaber $1963$ Proposition $1$. Deligne's conjecture at
$n = 3$ (Tamarkin $2003$; Dolgushev--Tamarkin--Tsygan $2010$
\textit{arXiv:$0807.5117$}: formality of the homotopy calculus of
Hochschild cochains) lifts these to a homotopy-compatible
$E_3$-algebra structure $F \cdot (\cup, \{-,-\}_G)$ on the cochain
complex, with $F$ the chosen formality element. The $\mathrm{GRT}_1$-action
on $F$ modifies higher operations (arity $\geq 3$ chain operations) but
preserves the arity-$2$ binary operations, hence preserves the
cohomological $\mathrm{Ger}_3$-structure on $\HH^\bullet(\cC)$ as an
un-upgraded graded-commutative-plus-bracket pair.
\end{proof}

\begin{corollary}[Stage-$1$ pinning on $\PhiFA_3$]
\label{cor:phifa3-stage-1-pinning-torsor}
\ClaimStatusTheorem
Stage-$1$ of $\PhiFA_3$ on $\cC \in \CY_3\text{-}\Cat^{\mathrm{dg}}$
is determined up to a contractible choice parametrised by
$\mathrm{Form}_3(\Q) \simeq \mathrm{GRT}_1(\Q)$-torsor. Different
points of the torsor produce quasi-isomorphic but not equal
$E_3$-factorisation algebras on $X$:
\[
 \PhiFA_3(\cC)_F \;\simeq\; \PhiFA_3(\cC)_{F'} \;\text{ via the } \mathrm{GRT}_1(\Q)\text{-element }
 g \text{ with } F' = g \cdot F,
\]
with the equivalence an $E_3$-quasi-isomorphism in
$\EdHolFA(X)^{(\infty,1)}$, not a strict equality of factorisation
algebras.
\end{corollary}

\begin{proof}
Combine Theorem~\ref{thm:chain-level-e3-algebra-hochschild-d3} with
step $(\mathrm{b})$ of Proposition~\ref{prop:phi-fa-three-step-assembly}
(Costello--Gwilliam locality): the $E_3$-algebra on $\HH^\bullet(\cC)$
assembles into an $E_3$-factorisation algebra on the underlying
topological space of $X$ by the factorisation-envelope construction,
which is $(\infty,1)$-functorial in the $E_3$-algebra input. A
$\mathrm{GRT}_1(\Q)$-element $g$ acts on the $E_3$-algebra input by the
formality-torsor action, hence on the assembled factorisation algebra
by a $\mathrm{GRT}_1(\Q)$-induced automorphism of
$\EdHolFA(X)^{(\infty,1)}$. Step $(\mathrm{c})$
(Costello--Li holomorphic twist) is $\mathrm{GRT}_1$-equivariant by the
holomorphic-twist-commuting-with-formality lemma of Costello--Li $2016$
\textit{arXiv:$1605.09124$} §$4.3$.
\end{proof}

\begin{remark}[Contractibility of the choice on fixed $X$]
\label{rem:form3-contractibility-on-fixed-x}
Fix $X$ a smooth complex CY$_3$. The space of $E_3$-holomorphic
factorisation algebras on $X$ arising as Stage-$1$ of $\PhiFA_3(\cC)$ for
$\cC$ a CY$_3$ category is, as a space of choices, contractible: at
each $F \in \mathrm{Form}_3(\Q)$ we get an $E_3$-hFA
$\PhiFA_3(\cC)_F \in \EdHolFA(X)^{(\infty,1)}$, and the
$\mathrm{GRT}_1(\Q)$-action connects any two such choices. This is the
precise meaning of ``Stage-$1$ is pinned up to contractible choice'' in
Theorem~\ref{thm:phi-two-stage-factorisation}: the choice is a point in
a $\mathrm{GRT}_1(\Q)$-torsor, which is contractible as a pro-simplicial
set because $\mathrm{GRT}_1(\Q)$ is pro-unipotent.
\end{remark}

\subsection*{The CY$_3$ self-duality $\omega_X \cong \cO_X$ does not rigidify}

\begin{proposition}[CY$_3$ self-duality and the torsor: non-rigidification]
\label{prop:cy3-self-duality-torsor-not-rigidified}
\ClaimStatusTheorem
The CY$_3$ self-duality $\omega_X \cong \cO_X$ (trivialisation of the
canonical bundle by $\Omega_X$) provides a degree-$(-3)$ pairing on
$\HH^\bullet(\cC)$ compatible with the $\mathrm{Ger}_3$-structure but does not
select a point in $\mathrm{Form}_3(\Q) \simeq
\mathrm{GRT}_1(\Q)$-torsor. Every formality quasi-isomorphism
$F \in \mathrm{Form}_3(\Q)$ is compatible with the cyclic pairing; the
torsor of such compatible formalities is still a full
$\mathrm{GRT}_1(\Q)$-torsor.
\end{proposition}

\begin{proof}
The cyclic $E_3$-algebra structure of
Theorem~\ref{thm:chain-level-e3-algebra-hochschild-d3} is
$F$-independent in degree $-3$: the CY pairing is the Serre-duality
trace $\mathrm{tr} \colon \HH_3(\cC) \to \Q$, a cohomological datum
read from the trivialisation of the Serre functor; this datum is
fixed by $\Omega_X$ and does not depend on the formality choice.
The $\mathrm{GRT}_1$-action moves the higher-arity operations of $F$
but fixes the pairing-degree $\mu \colon \HH^\bullet \otimes
\HH^\bullet \to \HH^\bullet[-d]$, because the cyclic structure lives
one degree below the torsor-visible layer. Hence the
pairing-preserving subtorsor is still $\mathrm{GRT}_1(\Q)$: the cyclic
constraint is a parallel-transport datum on the torsor, not a reduction
of its structure group (Kontsevich--Soibelman $2008$
\textit{arXiv:$0806.0107$} Proposition $3.5$: cyclic $L_\infty$-formality
is a $\mathrm{GRT}_1^{\mathrm{cyc}}$-torsor isomorphic to
$\mathrm{GRT}_1$ at every even $d$).
\end{proof}

\begin{corollary}[Universality of the torsor]
\label{cor:torsor-universality-across-cy3}
\ClaimStatusTheorem
The $\mathrm{GRT}_1(\Q)$-torsor $\mathrm{Form}_3(\Q)$ on Stage-$1$ of
$\PhiFA_3$ is independent of the CY$_3$ category $\cC$ and of the CY$_3$
variety $X$: it is the universal torsor on
$C_\bullet(D_3; \Q) \xrightarrow{\sim} \mathrm{Ger}_3$. Every
$\PhiFA_3(\cC)$ on every $X$ carries a free $\mathrm{GRT}_1(\Q)$-action;
two different $\cC, \cC'$ land on the same orbit structure of the same
torsor.
\end{corollary}

\begin{remark}[Strict-vs-canonical: the equals sign is the theorem]
\label{rem:equals-sign-at-cy3-torsor}
The association $\cC \mapsto \PhiFA_3(\cC)$ is not a functor valued in
strict factorisation algebras: it is a functor valued in the homotopy
category $\mathrm{Ho}(\EdHolFA(X))$, and within each homotopy class
there is a $\mathrm{GRT}_1(\Q)$-torsor of strict representatives. The
equality $\PhiFA_3(\cC) = \cA$ for $\cA$ a specific factorisation algebra
on $X$ is the theorem $[\PhiFA_3(\cC)] = [\cA]$ in the homotopy
category: after choosing any $F \in \mathrm{Form}_3(\Q)$ the equality
holds strictly. The torsor non-triviality is compatible with the equals
sign because the torsor orbit is a single homotopy class.
\end{remark}

\subsection*{Side-by-side correspondence with CFG $2026$ topological Chern--Simons}

\begin{theorem}[Universal $E_3$ under $\Phi$: the CFG $2026$ correspondence]
\label{thm:cfg-phi-e3-correspondence}
\ClaimStatusConditional
Let $M^3$ be a compact orientable topological $3$-manifold and
$\frakg$ a finite-dimensional reductive Lie algebra with non-degenerate
invariant pairing (the CFG $2026$ setting). The classical topological
Chern--Simons BV observables
$\Obs^{\mathrm{CS}}_{\mathrm{top,cl}}(M^3, \frakg) \in
E_3\text{-}\mathrm{Alg}^{\mathrm{top}}(\mathrm{Ch}(\Q))$ carry a canonical $E_3$-algebra
structure, pinned up to $\mathrm{GRT}_1(\Q)$-torsor by the same
formality quasi-isomorphism $F \in \mathrm{Form}_3(\Q)$ used for
Stage-$1$ of $\PhiFA_3$. The CY-to-chiral Stage-$1$ assembly
\[
 \PhiFA_3(\cC) \in \EdHolFA(X)
\]
is the holomorphic-twisted counterpart of the CFG $2026$ output:
replacing (topological $3$-manifold $M^3$, reductive $\frakg$) with
(complex CY$_3$ variety $X$, polyvector dg-Lie $\HH^\bullet(\cC)$),
and (topological $E_3$-FA) with (holomorphic $E_3$-FA) via the
Costello--Li holomorphic twist. The $\mathrm{GRT}_1(\Q)$-torsor is the
same torsor.
\end{theorem}

\subsection*{Why the $\mathrm{GRT}_1(\Q)$-ambiguity does not contaminate physical invariants}

\begin{proposition}[$\mathrm{GRT}_1(\Q)$-invariance of $\kch$, $\kBKM$, $\kfibre$]
\label{prop:grt1-invariance-of-kappa-invariants}
\ClaimStatusTheorem
The four $\kappa$-invariants of the CY$_3$ programme are
$\mathrm{GRT}_1(\Q)$-invariant, i.e.\ depend on the $\mathrm{GRT}_1$-orbit
but not on the individual formality element $F \in \mathrm{Form}_3(\Q)$:
\begin{itemize}
\item $\kch(\Phi_3(\cC))$ is the Heisenberg--Cartan rank of the
Stage-$2$ shadow, read from generator-counting on $\HH^{\bullet+1}(\cC)$;
generator count is $\mathrm{GRT}_1$-invariant because the formality
map preserves cohomology.
\item $\kcat(\cC) = \chi(\cO_X)$ is the Hodge supertrace, a
topological invariant of $X$; $\mathrm{GRT}_1$-invariant because the
graph-complex action affects higher-arity operations, not Hodge
numbers.
\item $\kBKM(\Phi_N) = c_N(0)/2$ is read from the Borcherds-lift
Fourier expansion, a modular-arithmetic datum;
$\mathrm{GRT}_1$-invariant because the Borcherds lift is
geometric-and-automorphic, not chain-level.
\item $\kfibre(X) = \mathrm{rank}(\widetilde\Lambda_{\mathrm{Muk}})$ is
the Mukai-lattice rank; $\mathrm{GRT}_1$-invariant because the Mukai
lattice is derived-categorical, not chain-level-formality-dependent.
\end{itemize}
\end{proposition}

\begin{proof}
All four invariants factor through the $\mathrm{GRT}_1$-orbit, not the
individual torsor point, because they are read off cohomological
or derived-categorical data: $\kch$ from $H^\bullet(\HH^{\bullet+1})$;
$\kcat$ from $H^\bullet(X, \cO_X)$; $\kBKM$ from Fourier coefficients
of automorphic forms; $\kfibre$ from the Mukai lattice of $X$ in
$D^b(\Coh(X))$. The $\mathrm{GRT}_1$-torsor action is on the
pre-cohomological chain-level data; taking cohomology factors out the
action.
\end{proof}

\begin{corollary}[Physical well-definedness of $\Phi_3$-outputs]
\label{cor:phi3-physical-well-definedness}
\ClaimStatusTheorem
The $\Phi_3$-output $A_\cC^{(\Sigma_2, C)} = \SpCh_{\Sigma_2, C}
(\PhiFA_3(\cC))$ is well-defined up to $\mathrm{GRT}_1(\Q)$-isomorphism.
All physically measurable invariants of $A_\cC^{(\Sigma_2, C)}$ (the
four $\kappa$-invariants, the central character, the modular trace,
the Mukai-Heisenberg Cartan, the Borcherds-root multiplicities, the
Gritsenko $\Delta_5$ weight) are $\mathrm{GRT}_1$-invariant and do not
depend on the formality choice.
\end{corollary}

\subsection*{Connection to the derived automorphism group $\mathrm{Aut}^{E_3}$}

\begin{proposition}[The derived automorphism group of an $E_3$-algebra]
\label{prop:aut-e3-derived}
\ClaimStatusTheorem
For $A \in E_3\text{-}\mathrm{Alg}(\mathrm{Ch}(\Q))$ a connected unital $E_3$-algebra,
the derived automorphism group $\mathrm{Aut}^{E_3}(A) \in
\mathrm{Grp}(\mathrm{Top})$ fits into an exact sequence
\[
 1 \;\to\; \mathrm{Inn}(A) \;\to\; \mathrm{Aut}^{E_3}(A) \;\to\;
 \mathrm{GRT}_1(\Q) \;\to\; 1
\]
where $\mathrm{Inn}(A)$ is the inner automorphism group
(conjugation by invertible elements of $A$) and
$\mathrm{GRT}_1(\Q)$ is the pro-unipotent Grothendieck--Teichmüller
group acting on the $E_3$-structure via the formality torsor
(Tamarkin $2003$ Corollary $4.2$; Willwacher $2015$ Theorem $1.1$
extended to $n = 3$). On $A = \HH^\bullet(\cC)$ for
$\cC \in \CY_3\text{-}\Cat^{\mathrm{dg}}$, the $\mathrm{Inn}$-piece is
determined by the derived Picard group of $\cC$; the
$\mathrm{GRT}_1$-piece is universal.
\end{proposition}

\subsection*{What the $\mathrm{GRT}_1(\Q)$-torsor does not do}

\begin{remark}[Non-trivialisations]
\label{rem:grt1-torsor-non-trivialisations-d3}
The $\mathrm{GRT}_1(\Q)$-torsor on Stage-$1$ of $\PhiFA_3$ does not:
\begin{itemize}
\item \emph{Provide a canonical associator.} No choice of
$\Phi \in \mathrm{Assoc}(\Q)$ is more natural than any other from the
operadic point of view. Number-theoretic constraints (multiple zeta
values, Drinfeld's integrality) select the KZ associator over $\C$, but
over $\Q$ the set $\mathrm{Assoc}(\Q)$ has no single distinguished
element.
\item \emph{Rigidify the chain-level $[m_3, B^{(2)}]$ obstruction.}
The chain-level $\bS^3$-framing obstruction (Chapter~\ref{ch:m3-b2-obstruction})
is a separate datum from the $\mathrm{GRT}_1$-torsor. A choice of
$F \in \mathrm{Form}_3(\Q)$ fixes the $E_3$-formality but does not
automatically select an $\bS^3$-framing; the framing is an additional
homotopy datum.
\item \emph{Commute with Stage-$2$ specialisation.} The
$\mathrm{GRT}_1$-action on Stage-$1$ does not in general descend to a
$\mathrm{GRT}_1$-action on $\SpCh_{\Sigma_2, C}(\PhiFA_3(\cC))$: the
specialisation destroys higher-arity $E_3$-operations by restriction to
$E_1$, so most of the $\mathrm{GRT}_1$-orbit collapses at Stage-$2$.
What survives is an action on the chosen chiral-algebra presentation,
not on the underlying quantum group structure.
\end{itemize}
\end{remark}

\subsection*{Explicit computation: the KZ associator and the K3$\times E$ Stage-$1$}

\begin{construction}[KZ associator $\Phi_{\mathrm{KZ}}$ over $\C$]
\label{constr:kz-associator-explicit}
\ClaimStatusDefinitional
The KZ associator $\Phi_{\mathrm{KZ}}$ is the monodromy of the
Knizhnik--Zamolodchikov connection on $\C \setminus \{0, 1\}$ with
values in the infinitesimal pure-braid Lie algebra $\mathfrak{t}_3$.
Its explicit expansion in the variables $t_{12}, t_{23}$ is
\[
 \Phi_{\mathrm{KZ}} = 1 + \zeta(2) [t_{12}, t_{23}] +
 \zeta(3) \bigl([t_{12}, [t_{12}, t_{23}]] + [[t_{12}, t_{23}], t_{23}]\bigr)
 + \sum_{w \in \mathrm{MZV}} c_w \cdot w \cdot T_w,
\]
where the sum runs over Lyndon words $w$ of weight $\geq 4$, $c_w$ is
the associated multiple zeta value, and $T_w$ the corresponding
commutator word in $t_{12}, t_{23}$ (Drinfeld $1991$ Theorem $A$;
Brown $2012$ \textit{Ann. Math.} $175$: all $c_w$ are
$\Q$-linear combinations of MZVs, and MZVs are a motivic Galois module
under the motivic fundamental group of $\mathbb{P}^1 \setminus
\{0, 1, \infty\}$). The KZ associator is defined over $\C$, not over
$\Q$; any associator over $\Q$ must differ from $\Phi_{\mathrm{KZ}}$ by
a $\mathrm{GRT}_1(\C)$-element, and the transcendence of
$\zeta(2n+1)$ forces the difference to be non-trivial.
\end{construction}

\begin{example}[$\PhiFA_3(D^b(\Coh(K3 \times E)))$ and the KZ associator]
\label{ex:k3xe-kz-stage-1}
\ClaimStatusConditional
For $\cC = D^b(\Coh(K3 \times E))$, the Hochschild cohomology
\[
 \HH^\bullet(\cC) \;\cong\; \bigoplus_{p+q=\bullet} H^q(X, \wedge^p T_X)
 \;=\; H^\bullet(X, \mathrm{PV}^\bullet)
\]
is a polyvector dg-Lie algebra with Schouten--Nijenhuis bracket of
degree $-1$, shifted to degree $-2$ on the CY$_3$ side. The chain-level
$E_3$-algebra $\CC^\bullet(\cC, \cC)$ depends on a choice of
$F \in \mathrm{Form}_3(\Q)$; the cohomological $\mathrm{Ger}_3$-structure on
$\HH^\bullet(\cC)$ is independent of the choice. Specifying the KZ
associator $F = F_{\Phi_{\mathrm{KZ}}}$ (after tensoring to $\C$) pins
the $E_3$-algebra structure to a canonical representative with MZV
Feynman integrals; specifying any other $F' \in \mathrm{Form}_3(\Q)$
gives a $\mathrm{GRT}_1(\C)$-equivalent presentation. The Borcherds
image $\PhiFA_3(D^b(\Coh(K3 \times E)))$ after $\SpCh_{K3, E}$
specialisation, namely the $\Delta_5$-Heisenberg chiral algebra with
$\kBKM(\Delta_5) = 5$, is $\mathrm{GRT}_1$-invariant by
Proposition~\ref{prop:grt1-invariance-of-kappa-invariants}.
\end{example}

\subsection*{Principal theorem}

\begin{theorem}[Chain-level Stage-$1$ pinning of $\PhiFA_3$]
\label{thm:kt-e3-stage-1-pinning-master}
\ClaimStatusTheorem
Let $\cC \in \CY_3\text{-}\Cat^{\mathrm{dg}}$ and $X$ a smooth complex
CY$_3$ variety with $\Omega_X \in H^0(X, K_X)$. The Stage-$1$ assembly
\[
 \PhiFA_3 \colon \CY_3\text{-}\Cat^{\mathrm{dg}} \to \EdHolFA(X)^{(\infty,1)},
 \quad
 \cC \mapsto \PhiFA_3(\cC),
\]
factors through a $\mathrm{GRT}_1(\Q)$-torsor of chain-level $E_3$-algebra
refinements of $\HH^\bullet(\cC)$, with the following three properties.
\begin{enumerate}[label=\textup{(\roman*)}]
\item The space of strict chain-level representatives is a torsor under
the pro-unipotent group $\mathrm{GRT}_1(\Q)$ acting by Willwacher's
graph-cocycle action, and the torsor is independent of $\cC$ and $X$:
it is the universal formality torsor $\mathrm{Form}_3(\Q)$ of
Theorem~\ref{thm:form3-grt1-torsor}.
\item The CY$_3$ self-duality $\omega_X \cong \cO_X$ does not rigidify
the torsor; the $\mathrm{GRT}_1(\Q)$-action preserves the
CY$_3$-shifted cyclic pairing on $\HH^\bullet(\cC)$.
\item The four $\kappa$-invariants
$\kch, \kcat, \kBKM, \kfibre$ are $\mathrm{GRT}_1(\Q)$-invariant and
do not depend on the torsor point.
\end{enumerate}
The CFG $2026$ identification of $E_3$-structure on topological
Chern--Simons observables realises the same $\mathrm{GRT}_1(\Q)$-torsor
on its topological side; the holomorphic twist carries the torsor
across to Stage-$1$ of $\PhiFA_3$.
\end{theorem}
