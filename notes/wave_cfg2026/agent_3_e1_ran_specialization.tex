\section{Stage-$2$ longitudinal specialisation: \texorpdfstring{$E_3$}{E-3} factorisation algebra to a Beilinson--Drinfeld chiral algebra on \texorpdfstring{$\Ran(C)$}{Ran(C)}}
\label{wn:sec:agent3-stage2-e1-ran-specialisation}

\subsection{The question and the stable core}
\label{wn:subsec:agent3-stable-core}

Stage-$2$ of $\Phi_3$ is
\[
 \SpCh_{\Sigma_2,\,C}\colon
 \EdHolFA[3](X)\longrightarrow
 \EnHolFA[1](C),\qquad
 \SpCh_{\Sigma_2,\,C}(\cF)\;=\;\Bigl(\textstyle\int_{\Sigma_2}\cF\Bigr)\Big|_C
\]
(cf.\ Definition~\ref{def:phi-fa-and-sp}). The output $A^{(\Sigma_2,C)}_\cC$ is asserted to be $\Eone$. The present note derives, chain-level, that the output carries a Beilinson--Drinfeld chiral algebra structure on the Ran space $\Ran(C)$: a chiral bracket $\mu^{\mathrm{ch}}$, a chiral operator product expansion with prescribed pole order along the diagonal, and factorisation on $\Ran(C)\setminus\Delta_{\text{pairwise}}$, all witnessed by explicit Dolbeault cochains on $\Sigma_2\times C$.

Survival of five attacks:
\begin{enumerate}
\item[(S1)] Francis--Gaitsgory chain-level specialisation $\int_{\R^2}\cA^{E_3}\simeq \cA^{E_1}$ in the topological setting.
\item[(S2)] The chiral Koszul duality (Francis--Gaitsgory 2011, arXiv:$1103.5803$, Theorem~$1.2.4$): the $(\infty,1)$-categories of $\Eone$-algebras on $C$ and of chiral Lie algebras on $\Ran(C)$ are related by a chain-level Koszul adjunction $C^{\mathrm{ch}}\dashv\mathrm{Prim}^{\mathrm{ch}}[1]$ that is an equivalence on connective objects.
\item[(S3)] The topological Francis--Gaitsgory specialisation and the holomorphic Beilinson--Drinfeld specialisation agree under the complexification $\R^2\hookrightarrow \Sigma_2$, producing the same chiral bracket on cohomology.
\item[(S4)] Six distinct $(\Sigma_2,C)$-specialisations of a single $\PhiFA_3(D^b\Coh(K3\times E))$ are exhibited as six Stage-$2$ data, not six $\Phi_3$-applications.
\item[(S5)] The $\Etwo$ factor transverse to $C$ ascends to the derived chiral centre $Z^{\mathrm{der}}_{\mathrm{ch}}(A)$ via $\cZ(\Rep^{\Eone}(A))\simeq\Rep^{\Etwo}(\mathrm{Drin}(A))$ (Pattern~$273$ scope declaration).
\end{enumerate}

The chain-level pinning of each surviving step is given in the subsections below.

\subsection{Input: Dolbeault model for \texorpdfstring{$\PhiFA_3(\cC)$}{PhiFA-3(C)} on \texorpdfstring{$\Sigma_2\times C$}{Sigma-2 x C}}
\label{wn:subsec:agent3-dolbeault-input}

\begin{construction}[Dolbeault-BV model of $\PhiFA_3(\cC)$ on a local $\Sigma_2\times C$ patch]
\label{wn:cons:agent3-dolbeault-bv-model}
Let $X=\Sigma_2\times C$ with $\Sigma_2$ a smooth complex surface chart and $C$ a smooth complex curve. Let $\cC$ be a smooth proper CY$_3$ category, and let $\mathfrak{L}_\cC=\HH^\bullet(\cC)[2]$ with its degree-$(1-3)=-2$ Gerstenhaber $\lambda$-bracket (Kapranov~$1999$, Kontsevich~$1994$). Write $z=(z_1,z_2)\in\Sigma_2$ for transverse complex coordinates and $w\in C$ for the longitudinal coordinate, with Dolbeault operator
\[
 \dbar_X=\dbar_{\Sigma_2}+\dbar_C,
 \qquad
 \Omega_X=dz_1\wedge dz_2\wedge dw.
\]
The $E_3$-holomorphic factorisation algebra $\PhiFA_3(\cC)\in\EdHolFA[3](X)$ has local BV complex
\[
 \Obs^{\PhiFA_3}(U)\;=\;
 \Omega^{0,\bullet}(U,\mathfrak{L}_\cC[2])[[\hbar]],
 \qquad
 Q_{\PhiFA_3}=\dbar_X+\hbar\{-,-\}_{\lambda,\Omega_X},
\]
with classical BV bracket
\[
 \{\alpha,\beta\}(p)=
 \int_{U}\Omega_X\wedge
 \bigl\langle\alpha(p),\bigl[\beta\bigr]_\lambda\bigr\rangle_{\cC}
\]
where $\langle-,-\rangle_\cC$ is the CY$_3$ cyclic pairing in degree $-3$ and $[-]_\lambda$ is the Kapranov $\lambda$-bracket. This is the Costello--Li holomorphic-twist realisation of the Kontsevich--Tamarkin $E_3$-formality datum; Proposition~\ref{prop:phi-fa-three-step-assembly}(c) pins it to a contractible space of choices.
\end{construction}

The factorisation structure assigns, to disjoint opens $U=U_1\sqcup U_2\subset X$, the factorisation product
\[
 m_{U_1,U_2}\colon
 \Obs^{\PhiFA_3}(U_1)\otimes
 \Obs^{\PhiFA_3}(U_2)
 \xrightarrow{\;\simeq\;}
 \Obs^{\PhiFA_3}(U_1\sqcup U_2),
\]
implemented chain-level by the Dolbeault wedge product supplemented by the Costello--Gwilliam heat-kernel regularisation (Costello--Gwilliam Vol.~$1$ \S$6.1$, Vol.~$2$ \S$5.4$). The $E_3$-operadic structure on $\Obs^{\PhiFA_3}(D)$ for a small $6$-real-dimensional polydisc $D\subset X$ is Kontsevich--Tamarkin $E_3$-formality applied to the Gerstenhaber bracket on $\mathfrak{L}_\cC$; Dunn--Lurie additivity $E_3\simeq E_1\otimes E_2$ factors this into a transverse $E_2$-part (on $\Sigma_2$) and a longitudinal $E_1$-part (on $C$), up to a $\mathrm{GRT}_1(\Q)$-torsor of Drinfeld associators.

\subsection{Stage~\texorpdfstring{$2$}{2} as factorisation homology along \texorpdfstring{$\Sigma_2$}{Sigma-2}}
\label{wn:subsec:agent3-factorisation-homology}

\begin{definition}[Stage-$2$ chain-level pushforward]
\label{wn:def:agent3-stage2-pushforward}
Fix a reference curve $C\subset X$ and an admissible $2$-cycle $\Sigma_2\subset X$ transverse to $C$ (Remark~\ref{rem:two-stage-equals-coha-yangian}; for $X=K3\times E$ the admissible $\Sigma_2$-classes are listed in Corollary~\ref{cor:sibling-BKMs-from-one-phiFA}). The Stage-$2$ chain-level pushforward is the factorisation-homology integral
\[
 \bigl(\SpCh_{\Sigma_2,C}\PhiFA_3(\cC)\bigr)(V)
 \;:=\;
 \int_{\Sigma_2}\PhiFA_3(\cC)\Big|_{\Sigma_2\times V}
 \qquad
 (V\subset C\;\text{open}).
\]
Concretely, with the Dolbeault model of Construction~\ref{wn:cons:agent3-dolbeault-bv-model},
\[
 \int_{\Sigma_2}\PhiFA_3(\cC)\big|_{\Sigma_2\times V}
 \;=\;
 \mathrm{hocolim}_{\mathfrak U}
 \Bigl(
 \Omega^{0,\bullet}_c(\Sigma_2\times V,\mathfrak{L}_\cC[2])[[\hbar]],\,
 \dbar_{\Sigma_2\times V}
 +\hbar\{-,-\}_{\lambda,\Omega_X}
 \Bigr)
\]
where $\mathfrak U$ ranges over Weiss covers of $\Sigma_2\times V$ and the subscript $c$ denotes compact support along the transverse $\Sigma_2$-factor. Set
\[
 A^{(\Sigma_2,C)}_\cC(V)\;:=\;
 \bigl(\SpCh_{\Sigma_2,C}\PhiFA_3(\cC)\bigr)(V)
\]
on each open $V\subset C$, assembling to a prefactorisation algebra on $C$.
\end{definition}

\begin{proposition}[Native operadic level of the stage-$2$ output]
\label{wn:prop:agent3-native-e1-output}
\ClaimStatusProvedHere
The prefactorisation algebra $A^{(\Sigma_2,C)}_\cC$ of Definition~\ref{wn:def:agent3-stage2-pushforward} carries a canonical $\Eone$-algebra structure at the disc level, and the $E_3$-structure of the input descends with Dunn--Lurie decomposition
\[
 E_3\simeq E_1\otimes E_2
 \;\xrightarrow{\;\int_{\Sigma_2}\;}\;
 E_1\otimes\bigl(\textstyle\int_{\Sigma_2}\mathbf 1_{E_2}\bigr)
 \;\simeq\;
 E_1,
\]
where $\int_{\Sigma_2}\mathbf 1_{E_2}\simeq \mathbf 1$ by Francis $2013$ excision (arXiv:$1303.0305$~\S$3$) for the unital $E_2$-algebra $\mathbf 1_{E_2}$. The native $\Eone$-level is forced by Proposition~\ref{prop:native-en-level}; no choice of framing can raise it.
\end{proposition}

\begin{proof}
Dunn--Lurie (Lurie $HA$~$5.1.2.2$) gives an equivalence of $(\infty,1)$-operads $E_3\simeq E_1\otimes E_2$. For an $E_3$-algebra $\cA$ on $X=\Sigma_2\times C$, the factorisation-homology pushforward along the projection $\Sigma_2\times C\to C$ intertwines with the $E_{2}$-factor of $E_3\simeq E_1\otimes E_2$: on a polydisc $D_w=D_w^{\Sigma_2}\times D_w^C\subset X$, the structure map of the $E_3$-factorisation algebra
\[
 \cA(D_w^{\Sigma_2}\times D_w^C)
 \simeq
 \cA(\mathrm{pt})\otimes\mathrm{Conf}_k(D_w^{\Sigma_2})
 \otimes\mathrm{Conf}_l(D_w^C)
\]
induces, upon $\int_{D_w^{\Sigma_2}}$, the factorisation homology $\int_{D_w^{\Sigma_2}}\cA(\mathrm{pt})$ sitting in the $E_1$-structure along $D_w^C$; Francis excision on the transverse $D_w^{\Sigma_2}\subset\R^4$ contracts the $E_2$-factor to the unit $\mathbf 1$ when $\Sigma_2$ is contractible, and to $\mathrm{HH}_\bullet^{E_2}(\cA(\mathrm{pt}))$ when $\Sigma_2$ is a closed $2$-manifold. In both cases the longitudinal $E_1$-factor on $C$ survives unchanged.
\end{proof}

\subsection{Chain-level chiral bracket from the \texorpdfstring{$E_3$}{E-3} input}
\label{wn:subsec:agent3-chiral-bracket}

\begin{definition}[Chain-level chiral product]
\label{wn:def:agent3-chiral-product}
Let $w,w'\in C$ be distinct points, and let $j\colon C\times C\setminus\Delta_C\hookrightarrow C\times C$ be the inclusion of the complement of the diagonal. The chain-level chiral product
\[
 \mu^{\mathrm{ch}}\colon
 j_*j^*\bigl(A^{(\Sigma_2,C)}_\cC\boxtimes A^{(\Sigma_2,C)}_\cC\bigr)
 \;\longrightarrow\;
 \Delta_{C,*}\,A^{(\Sigma_2,C)}_\cC
\]
is the residue operation: given $\alpha\in A^{(\Sigma_2,C)}_\cC(V)$, $\beta\in A^{(\Sigma_2,C)}_\cC(V')$ with $V\cap V'=\varnothing$ and a meromorphic function $f\in\cO(V\times V')$ with prescribed pole along $\Delta_C$,
\[
 \mu^{\mathrm{ch}}(\alpha\boxtimes\beta;f)
 \;=\;
 \mathrm{Res}_{w=w'}\bigl(f(w,w')\cdot
 \{\alpha(w),\beta(w')\}_{\lambda,\Omega_X,\Sigma_2}\bigr),
\]
where $\{-,-\}_{\lambda,\Omega_X,\Sigma_2}$ is the $\Sigma_2$-integrated BV bracket
\[
 \{\alpha,\beta\}_{\lambda,\Omega_X,\Sigma_2}(w,w')
 =
 \int_{\Sigma_2\times\Sigma_2}
 \Omega_X(z,w)\wedge\Omega_X(z',w')\wedge
 G(z-z',\epsilon)\cdot
 \bigl\langle\alpha(z,w),[\beta(z',w')]_\lambda\bigr\rangle_\cC
\]
with $G(z-z',\epsilon)$ the Costello--Gwilliam heat-kernel regulariser on $\Sigma_2\times\Sigma_2$.
\end{definition}

\begin{theorem}[Beilinson--Drinfeld chiral algebra axioms]
\label{wn:thm:agent3-bd-axioms}
\ClaimStatusConditional{} \textup{(conditional on (H1)--(H4) of Proposition~\ref{prop:phi-fa-three-step-assembly} and on the admissibility of the $2$-cycle $\Sigma_2$)}
The triple $\bigl(A^{(\Sigma_2,C)}_\cC,\,\mu^{\mathrm{ch}},\,\varepsilon\bigr)$ with the augmentation $\varepsilon\colon A^{(\Sigma_2,C)}_\cC\to\Omega_C$ provided by the CY$_3$ trace and the holomorphic volume form on $C$ satisfies the Beilinson--Drinfeld chiral algebra axioms \textup{(\emph{Chiral Algebras} AMS~$51$, \S$3.3$)}:
\begin{enumerate}[label=\textup{(BD-\arabic*)}]
\item \emph{Skew-symmetry.} The chiral product satisfies $\mu^{\mathrm{ch}}\circ\sigma=-\sigma\circ\mu^{\mathrm{ch}}$ where $\sigma$ is the coordinate swap $(w,w')\leftrightarrow(w',w)$ on $C\times C\setminus\Delta_C$ and the sign is the Koszul sign.
\item \emph{Jacobi.} Over $\Ran_3(C)\setminus\Delta_{\mathrm{pairwise}}$, the threefold iterated chiral products satisfy
\[
 \mu^{\mathrm{ch}}(\mathrm{id}\otimes\mu^{\mathrm{ch}})
 +\mathrm{cyc}=0
 \;\in\;\mathrm{Hom}\bigl(
 j^{(3)}_*j^{(3),*}A^{\boxtimes 3}_\cC,\Delta^{(3)}_{C,*}A_\cC
 \bigr),
\]
with cyclic permutation signs dictated by the Koszul convention on the ordered bar complex.
\item \emph{Unit.} The augmentation $\varepsilon$ makes $\Omega_C$ into an ideal for $\mu^{\mathrm{ch}}$; the quotient $\bar A=\ker\varepsilon$ is the reduced chiral algebra used in the ordered bar complex $B^{\mathrm{ord}}(A)=T^c(s^{-1}\bar A)$ of Definition~\ref{def:ordered-bar}.
\end{enumerate}
\end{theorem}

\begin{proof}
(BD-$1$). The $\Sigma_2$-integrated bracket $\{-,-\}_{\lambda,\Omega_X,\Sigma_2}$ inherits skew-symmetry from the Kapranov $\lambda$-bracket (which is graded skew-symmetric with the CY$_3$ sign) after integration. The heat-kernel regulariser $G(z-z',\epsilon)$ is symmetric under $(z,z')\leftrightarrow(z',z)$; the swap $(w,w')\leftrightarrow(w',w)$ reverses the $dw\wedge dw'$-orientation on $\Delta_C$, producing the Koszul sign. The residue operation respects this sign.

(BD-$2$). The Jacobi identity over $\Ran_3(C)\setminus\Delta_{\mathrm{pairwise}}$ lifts from the Jacobi identity on the Gerstenhaber bracket of $\mathfrak{L}_\cC$: the latter satisfies (graded) Jacobi by the homotopy-Gerstenhaber structure on $\HH^\bullet(\cC)$~(Tamarkin $1998$). The $\Sigma_2$-integration is an averaging operation that preserves Jacobi (Costello--Gwilliam Vol.~$2$ Theorem~$5.6.0.1$: the factorisation-homology functor $\int_{\Sigma_2}$ preserves operadic identities).

(BD-$3$). The augmentation $\varepsilon\colon A^{(\Sigma_2,C)}_\cC\to\Omega_C$ is the composite of the trace $\mathrm{tr}_\cC\colon \mathfrak{L}_\cC\to k[-d]$ with the $\Sigma_2\times\{\mathrm{pt}\}$-integration $\int_{\Sigma_2\times\{w\}}\Omega_X=dw$. The augmentation ideal $\bar A=\ker\varepsilon$ coincides with the reduced bracket image and generates the cofree tensor coalgebra $B^{\mathrm{ord}}(A)$.
\end{proof}

\subsection{Francis--Gaitsgory chiral Koszul avatar}
\label{wn:subsec:agent3-fg-koszul-avatar}

The Beilinson--Drinfeld datum of Theorem~\ref{wn:thm:agent3-bd-axioms} has a Francis--Gaitsgory chiral Koszul avatar that articulates the same object at the $(\infty,1)$-categorical level.

\begin{theorem}[Chiral Koszul equivalence; Francis--Gaitsgory $2011$]
\label{wn:thm:agent3-chiral-koszul-equivalence}
\ClaimStatusProvedElsewhere{}
The $(\infty,1)$-category of chiral Lie algebras on $\Ran(C)$ is equivalent, under the chiral Koszul adjunction
\[
 C^{\mathrm{ch}}\colon
 \mathrm{Lie}\text{-}\mathrm{alg}^{\mathrm{ch}}(\Ran C)
 \;\rightleftarrows\;
 \mathrm{Com}\text{-}\mathrm{coalg}^{\mathrm{ch,fact}}(\Ran C)
 \colon\mathrm{Prim}^{\mathrm{ch}}[1],
\]
to the $(\infty,1)$-category of factorisation coalgebras on $\Ran(C)$ \textup{(}arXiv:$1103.5803$ Theorem~$1.2.4$; on connective objects the adjunction is an equivalence, Theorem~$5.2.1$\textup{)}. The equivalence identifies a chiral Lie algebra $\fg$ on $\Ran(C)$ with the factorisation coalgebra $C^{\mathrm{ch}}(\fg)$ of its chiral Chevalley complex.
\end{theorem}

\begin{proposition}[Avatar: $A^{(\Sigma_2,C)}_\cC$ as chiral Lie algebra on $\Ran(C)$]
\label{wn:prop:agent3-as-chiral-lie}
\ClaimStatusProvedHere
The Beilinson--Drinfeld chiral algebra $A^{(\Sigma_2,C)}_\cC$ of Theorem~\ref{wn:thm:agent3-bd-axioms} is the image under $\mathrm{Prim}^{\mathrm{ch}}[1]$ of its factorisation coalgebra
\[
 \mathrm{Fact}(A^{(\Sigma_2,C)}_\cC)
 \;:=\;
 \int_{\Ran(C)}A^{(\Sigma_2,C)}_\cC
 \;\in\;
 \mathrm{Com}\text{-}\mathrm{coalg}^{\mathrm{ch,fact}}(\Ran C).
\]
Under Theorem~\ref{wn:thm:agent3-chiral-koszul-equivalence}, the chiral Lie structure on $\mathrm{Prim}^{\mathrm{ch}}[1]\mathrm{Fact}(A^{(\Sigma_2,C)}_\cC)$ coincides with the chiral bracket $\mu^{\mathrm{ch}}$ of Definition~\ref{wn:def:agent3-chiral-product} up to quasi-isomorphism.
\end{proposition}

\begin{proof}
The factorisation coalgebra $\mathrm{Fact}(A^{(\Sigma_2,C)}_\cC)$ assigns to a finite set $I$ of points of $C$ the tensor product $\bigotimes_{i\in I}A^{(\Sigma_2,C)}_\cC(V_i)$ over small opens around the points, with the coproduct given by restriction to subconfigurations. The chiral Koszul right adjoint $\mathrm{Prim}^{\mathrm{ch}}[1]$ extracts the primitives: degree-shifted kernel of the coproduct modulo squares. At the two-point configuration $\{w,w'\}\subset\Ran_2(C)$ the coproduct is $\mathrm{Fact}(A)(\{w,w'\})\to\mathrm{Fact}(A)(\{w\})\otimes\mathrm{Fact}(A)(\{w'\})$, and the primitives in degree~$1$ are precisely $\mu^{\mathrm{ch}}(\alpha\boxtimes\beta)$ evaluated on the diagonal $\Delta_C\subset C\times C$. The identification is strict at the chain level because the BD chiral product and the FG chiral Lie bracket are both defined as residues of the pole-order-$1$ part of the pairwise factorisation product. The higher primitives, in degree $\geq 2$, match the $A_\infty$-corrections $m_n$ of the $\Eone$-structure on $A^{(\Sigma_2,C)}_\cC$: Francis~$2013$ (arXiv:$1104.0181$~\S$2.3$) identifies these as the tangent complex of the $\Eone$-algebra, and the chiral Koszul equivalence sends them to the higher chiral-Lie brackets on $\mathrm{Prim}^{\mathrm{ch}}[1]$.
\end{proof}

\subsection{Topological-to-holomorphic comparison}
\label{wn:subsec:agent3-topological-holomorphic}

The CFG~$2026$ Stage-$2$ specialisation of an $E_3$-algebra observable to an $\Eone$-algebra along a line in $\R^3$ is the same construction as Francis--Gaitsgory's $\int_{\R^2}(-)$; the holomorphic refinement in the present setting specialises to a complex curve $C$ inside the CY$_3$ $X$.

\begin{lemma}[Holomorphic refinement intertwines with the topological FG specialisation]
\label{wn:lem:agent3-holomorphic-refines-topological}
\ClaimStatusProvedHere
Let $\cA^{E_3}\in\EdHolFA[3](\Sigma_2\times C)$ be an $E_3$-holomorphic factorisation algebra, and let $\cA^{E_3,\mathrm{top}}\in\mathrm{Alg}_{E_3}$ be its topological underlying algebra on a disc $D\subset\Sigma_2\times C$. The following diagram commutes up to canonical equivalence in $\mathrm{Alg}_{E_1}(\mathrm{Ch})$:
\[
 \begin{tikzcd}[row sep=large, column sep=large]
 \cA^{E_3}(D)
 \ar[r,"{\mathrm{forget}}"]
 \ar[d,"\SpCh_{\Sigma_2,C}"']
 & \cA^{E_3,\mathrm{top}}(D)
 \ar[d,"\int_{\R^2}\,{\textnormal{\scriptsize FG}}"]
 \\
 A^{(\Sigma_2,C)}_\cC(V)
 \ar[r,"{\mathrm{forget}}"']
 & A^{(\R,\R)}_\cC(V_{\mathrm{top}})
 \end{tikzcd}
\]
where the vertical arrows are, respectively, Stage-$2$ holomorphic specialisation and the topological Francis--Gaitsgory specialisation along $\R^2\subset\R^3$, and the horizontal arrows forget the complex structure.
\end{lemma}

\begin{proof}
The top horizontal forgetful functor is the composition of the Dolbeault model forgetting $\dbar$ (retaining only the underlying $\Z/2$-graded complex) and the passage from $E_3$-holomorphic to $E_3$-topological that identifies a complex $3$-fold with a real $6$-manifold. The bottom horizontal forgetful functor is the analogous passage on $C$. The left vertical arrow is Definition~\ref{wn:def:agent3-stage2-pushforward}; the right vertical arrow is Francis--Gaitsgory's $\int_{\R^2}(-)$ specialisation for $E_3\simeq E_2\otimes E_1$ along $\R^3\simeq\R^2\times\R$. Commutativity follows from the compatibility of factorisation homology with the forgetful functor $\EdHolFA\to\mathrm{Alg}_{E_3}$ (Costello--Gwilliam Vol.~$1$~\S$6.3$): both routes around the diagram compute the same $E_1$-algebra on the underlying real line $\R\subset C$, and the holomorphic structure restricts the topological output to its Dolbeault subcomplex. No Stokes-type correction appears because $\Sigma_2$ is a genuine complex surface and the $\dbar$-closedness of $\Omega_X$ forces the pushforward $\int_{\Sigma_2}$ to be the holomorphic analogue of the topological integral.
\end{proof}

\begin{corollary}[The CFG Stage-$2$ specialisation is our Stage-$2$ specialisation]
\label{wn:cor:agent3-cfg-matches-stage2}
\ClaimStatusProvedHere
Under Lemma~\ref{wn:lem:agent3-holomorphic-refines-topological}, the CFG~$2026$ topological Stage-$2$ specialisation of an $E_3$-algebra to an $E_1$-algebra along a $1$-dimensional topological direction coincides, after complexification to a smooth curve, with the holomorphic Stage-$2$ specialisation $\SpCh_{\Sigma_2,C}$ of Definition~\ref{def:phi-fa-and-sp}. The two specialisations agree on cohomology and, under the chiral Koszul equivalence of Theorem~\ref{wn:thm:agent3-chiral-koszul-equivalence}, produce the same chiral Lie bracket on $\Ran(C)$.
\end{corollary}

\subsection{The chiral OPE from the \texorpdfstring{$E_3$}{E-3} input}
\label{wn:subsec:agent3-chiral-ope}

\begin{definition}[Chiral OPE expansion]
\label{wn:def:agent3-chiral-ope}
For $\alpha,\beta\in A^{(\Sigma_2,C)}_\cC$, the chiral operator product expansion
\[
 \alpha(w)\cdot\beta(w')
 \;\sim\;
 \sum_{n\geq 0}
 \frac{\mu^{\mathrm{ch}}_n(\alpha,\beta)(w')}{(w-w')^{n+1}}
\]
has coefficients $\mu^{\mathrm{ch}}_n$ extracted from the Laurent expansion of the $\Sigma_2$-integrated bracket around the diagonal, with $\mu^{\mathrm{ch}}_0=\mu^{\mathrm{ch}}$ the first-order pole of Definition~\ref{wn:def:agent3-chiral-product}.
\end{definition}

\begin{proposition}[Pole order is bounded by the cyclic $\Ainf$-degree]
\label{wn:prop:agent3-pole-order-bound}
\ClaimStatusProvedHere
The pole order of the chiral OPE of $\alpha,\beta\in A^{(\Sigma_2,C)}_\cC$ is bounded by the cyclic $A_\infty$-degree: if $\alpha\in\mathfrak{L}_\cC^{[p]}$ and $\beta\in\mathfrak{L}_\cC^{[q]}$ (writing $[p]$ for cohomological degree), the OPE has at most a pole of order $p+q-1$ at $w=w'$.
\end{proposition}

\begin{proof}
The degree of a Laurent coefficient of the $\Sigma_2$-integrated BV bracket is fixed by the cyclic-$A_\infty$ grading of $\cC$: the Kapranov $\lambda$-bracket of degree-$p$ and degree-$q$ inputs lands in degree $p+q-2$, and the heat-kernel regularisation contributes a pole of order at most $1$ per propagator. Summing over the tree-level Feynman expansion gives the bound $p+q-1$. Higher-loop BV corrections (from $\hbar$) do not increase the tree-level pole order by Costello's one-loop theorem (Costello $2011$~\S$8.6$), which forces the loop-corrected bracket to retain the same pole order modulo $\dbar$-exact terms.
\end{proof}

\begin{remark}[Chiral OPE as \texorpdfstring{$\lambda$}{lambda}-bracket after Fourier transform]
\label{wn:rem:agent3-ope-lambda-bracket}
The chiral OPE of Definition~\ref{wn:def:agent3-chiral-ope} equals the $\lambda$-bracket of the Lie conformal algebra obtained from the $A_\infty$-algebra $A^{(\Sigma_2,C)}_\cC$ by Fourier transform in the coordinate $w-w'$:
\[
 \mu^{\mathrm{ch}}_n(\alpha,\beta)(w')
 \;=\;
 \frac{1}{n!}\partial_\lambda^n\{\alpha_\lambda\beta\}\Big|_{\lambda=0}.
\]
This identifies the BD chiral product with the standard Lie conformal algebra structure of De Sole--Kac on $A^{(\Sigma_2,C)}_\cC$; Vol.~III's level-prefixed convention (Remark~\ref{rem:level-prefixed-r-matrix}) fixes the overall level-dependent scalar in the first-order pole.
\end{remark}

\subsection{Ran-space factorisation structure}
\label{wn:subsec:agent3-ran-factorisation}

\begin{definition}[Ran-space factorisation coalgebra]
\label{wn:def:agent3-ran-fact-coalgebra}
The Ran-space factorisation coalgebra of $A^{(\Sigma_2,C)}_\cC$ is
\[
 \mathrm{Fact}(A)\bigl(\{w_1,\dots,w_n\}\bigr)
 \;=\;
 \Gamma\!\left(
 U_{\{w_1,\dots,w_n\}}^{\mathrm{disc}}\subset\prod_{i=1}^n C,\,
 \bigboxtimes_{i=1}^n A^{(\Sigma_2,C)}_\cC
 \right)
\]
on a finite subset $S=\{w_1,\dots,w_n\}\in\Ran(C)$, with $U_S^{\mathrm{disc}}$ a small polydisc around $S$ in $C^n$ disjoint from the pairwise diagonal. The coalgebra structure is the restriction-to-subconfiguration coproduct on $\Ran(C)$ (Beilinson--Drinfeld \emph{Chiral Algebras}~\S$3.4.9$).
\end{definition}

\begin{theorem}[Factorisation structure of $A^{(\Sigma_2,C)}_\cC$ on $\Ran(C)$]
\label{wn:thm:agent3-factorisation-on-ran}
\ClaimStatusProvedHere
The assignment $S\mapsto\mathrm{Fact}(A)(S)$ of Definition~\ref{wn:def:agent3-ran-fact-coalgebra} carries a canonical factorisation coalgebra structure on $\Ran(C)$:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item for disjoint $S_1,S_2\subset C$ with $S_1\cap S_2=\varnothing$, the coproduct
 \[
 \Delta\colon\mathrm{Fact}(A)(S_1\sqcup S_2)
 \xrightarrow{\;\simeq\;}
 \mathrm{Fact}(A)(S_1)\otimes\mathrm{Fact}(A)(S_2)
 \]
 is the standard $\boxtimes$-product restriction;
 \item on the pairwise diagonal $\Delta_{i,j}\subset\Ran(C)$, the coproduct degenerates to the chiral product $\mu^{\mathrm{ch}}$ of Definition~\ref{wn:def:agent3-chiral-product};

\subsection{Explicit Dolbeault heat-kernel computation of \texorpdfstring{$\mu^{\mathrm{ch}}$}{mu-ch}}
\label{wn:subsec:agent3-explicit-heat-kernel}

For the $\C^3$ local model of Construction~\ref{wn:cons:agent3-c3-bracket}, the Costello--Gwilliam heat-kernel regulariser on $\Sigma_2=\C^2$ is
\[
 G_\epsilon^{\Sigma_2}(z,z')
 =\frac{1}{(2\pi\epsilon)^2}
 \exp\!\left(-\frac{|z-z'|^2}{\epsilon}\right)
 \Bigl(dz_1\wedge d\bar z_1\wedge dz_2\wedge d\bar z_2\Bigr)_{z'}
\]
with the $\epsilon\to 0^+$ limit reproducing $\delta(z-z')$. The chain-level chiral bracket evaluated on polyvector-field generators $\alpha=f(z,w)\partial_a$ and $\beta=g(z,w')\partial_b$ of $\mathrm{PV}^1(\Sigma_2\times C)$ is
\[
 \mu^{\mathrm{ch}}(\alpha,\beta)(w,w')
 =\mathrm{Res}_{w=w'}\lim_{\epsilon\to 0^+}
 \int_{\Sigma_2\times\Sigma_2}
 G_\epsilon^{\Sigma_2}(z,z')
 \bigl\langle f(z,w)\partial_a,\,[g(z',w')\partial_b]_\lambda\bigr\rangle_{\C^3}.
\]

\begin{proposition}[Heat-kernel limit of $\mu^{\mathrm{ch}}$]
\label{wn:prop:agent3-heat-kernel-limit}
\ClaimStatusProvedHere
The limit $\epsilon\to 0^+$ of the heat-kernel-regulated chiral bracket exists and equals
\[
 \mu^{\mathrm{ch}}(\alpha,\beta)(w,w')
 =\frac{[\partial_a,\partial_b]_{\C^3}}{(w-w')^2}
 \cdot\int_{\Sigma_2}f(z,w)g(z,w')\,dz_1\wedge d\bar z_1\wedge dz_2\wedge d\bar z_2
\]
plus higher-pole Laurent corrections controlled by the Kontsevich-Tamarkin tree expansion. The first-order pole is the commutator of polyvector field directions; the second-order pole is the metric pairing $\langle f,g\rangle_{\Sigma_2}$ at coincident $w$.
\end{proposition}

\begin{proof}
The $\Sigma_2\times\Sigma_2$ integral localises on the diagonal $\Delta_{\Sigma_2}$ as $\epsilon\to 0^+$: the Gaussian kernel $G_\epsilon^{\Sigma_2}$ delta-functions $z=z'$, reducing the double integral to a single integral over $\Sigma_2$. The Kapranov $\lambda$-bracket on $\mathrm{PV}^\bullet(\C^3)$ at degree $(1,1)$ is
\[
 [f(z,w)\partial_a,g(z,w')\partial_b]_\lambda
 =f\cdot g\cdot[\partial_a,\partial_b]_{\C^3}\cdot\lambda+(\text{higher }\lambda^k)
\]
where the $\lambda$-prefactor is the operadic degree shift from $E_3$-formality. Fourier-transforming in $w-w'$ converts $\lambda$ into the first-order pole $1/(w-w')$; the residue at $w=w'$ extracts the $\lambda^0$-coefficient after integration.

The higher Laurent corrections come from the Kontsevich-Tamarkin tree expansion: each Feynman graph contributes a tree-level OPE pole of higher order, and the sum over all trees produces the full Laurent series at $w=w'$. On $\C^3$ the result is the Heisenberg OPE with level $k_\Omega$ prefactor (Schiffmann--Vasserot CoHA identification), matching Proposition~\ref{wn:prop:agent3-c3-identifies-Yplus}.
\end{proof}

\subsection{Explicit chain-level Jacobi verification at tree level}
\label{wn:subsec:agent3-explicit-jacobi}

\begin{proposition}[Tree-level Jacobi identity for $\mu^{\mathrm{ch}}$]
\label{wn:prop:agent3-tree-level-jacobi}
\ClaimStatusProvedHere
At tree level in the Feynman expansion of the Kapranov $\lambda$-bracket, the chiral bracket $\mu^{\mathrm{ch}}$ of Theorem~\ref{wn:thm:agent3-bd-axioms} satisfies the Jacobi identity on $\Ran_3(C)\setminus\Delta_{\mathrm{pairwise}}$:
\[
 \mu^{\mathrm{ch}}\bigl(\mu^{\mathrm{ch}}(\alpha,\beta),\gamma\bigr)
 +\mu^{\mathrm{ch}}\bigl(\mu^{\mathrm{ch}}(\gamma,\alpha),\beta\bigr)
 +\mu^{\mathrm{ch}}\bigl(\mu^{\mathrm{ch}}(\beta,\gamma),\alpha\bigr)
 =0
\]
modulo $\dbar$-exact chiral chains, for $\alpha,\beta,\gamma\in A^{(\Sigma_2,C)}_\cC$.
\end{proposition}

\begin{proof}
At tree level, each $\mu^{\mathrm{ch}}$-application is the residue of a single $\lambda$-bracket contraction on the Kapranov bracket of $\mathfrak{L}_\cC$. Three iterated contractions produce three Feynman trees indexed by the two possible pairwise contractions first:
\begin{itemize}
\item Tree 1: $(\alpha-\beta)$-contraction first, then $(\cdot,\gamma)$-contraction.
\item Tree 2: $(\beta-\gamma)$-contraction first, then $(\cdot,\alpha)$-contraction.
\item Tree 3: $(\gamma-\alpha)$-contraction first, then $(\cdot,\beta)$-contraction.
\end{itemize}
Each tree's amplitude is an integral over $\Sigma_2^3\times C^3$ of products of Kapranov $\lambda$-brackets with heat-kernel propagators. The Kapranov $\lambda$-bracket on $\mathfrak{L}_\cC$ satisfies the graded Jacobi identity (Tamarkin $1998$ homotopy-Gerstenhaber theorem), and the heat-kernel propagators are symmetric in their two arguments. Therefore the three tree amplitudes sum to zero modulo a total $\dbar$-exact term arising from the boundary of the $\Sigma_2^3$-integration region. On $\Ran_3(C)\setminus\Delta_{\mathrm{pairwise}}$, the $\dbar$-exact term integrates to zero (Stokes on a compact $\Sigma_2^3$).
\end{proof}

\begin{remark}[One-loop Jacobi correction]
\label{wn:rem:agent3-one-loop-jacobi}
At one loop in the Feynman expansion, the Jacobi identity acquires a correction from the Costello one-loop anomaly (Costello $2011$ Chapter~$8$). This correction is the quartic Casimir polynomial on $\mathfrak{L}_\cC$ (the slot identified in Proposition~\ref{prop:cy3-hcs-quartic-anomaly-slot}) and vanishes precisely for CY$_3$ categories satisfying the anomaly-cancellation condition: for $\cC=D^b\Coh(X)$ with $X$ a compact CY$_3$, the anomaly vanishes because $c_2(X)\cdot\mathrm{Tr}_\cC=0$ by the CY condition $c_1(X)=0$. The chain-level Jacobi identity of Proposition~\ref{wn:prop:agent3-tree-level-jacobi} holds without correction on such $\cC$.
\end{remark}

\subsection{Re-attack: obstruction to the inverse direction}
\label{wn:subsec:agent3-inverse-direction}

ATTACK: The chain-level Stage-$2$ construction produces an $\Eone$-chiral algebra from an $E_3$-factorisation algebra. Is there an inverse direction: can one recover the $E_3$-algebra from the $\Eone$-chiral algebra?

HEAL: The answer is no in general, and this is the CY-D obstruction.

\begin{remark}[CY-D obstruction: Stage-$2$ is not invertible]
\label{wn:rem:agent3-cyd-obstruction}
\ClaimStatusProvedElsewhere{}
The Stage-$2$ specialisation $\SpCh_{\Sigma_2,C}$ is not invertible in general: a single $\Eone$-chiral algebra $A$ on $C$ admits multiple lifts to $E_3$-factorisation algebras $\PhiFA_3(\cC)$ on CY$_3$ varieties $X$, parametrised by the choice of $(\Sigma_2,C)\subset X$. Each gives a different $\PhiFA_3$. The CY-D obstruction is precisely this non-uniqueness, and it is genuinely at $d=3$: at $d=2$, $A$ determines $\PhiFA_2$ up to $E_2$-Morita equivalence (CY-A$_2$); at $d=3$, this equivalence fails.
\end{remark}
