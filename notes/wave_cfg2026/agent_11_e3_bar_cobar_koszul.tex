\section*{$E_n$-operadic Koszul self-duality up to shift, the chain-level $E_3$ bar-cobar, and the $\PhiFA_3$ self-dual pairing on CY$_3$}
\label{sec:agent-11-e3-bar-cobar-koszul-cfg2026}

Fix the ambient as in Stage-$1$ of Proposition~\ref{prop:phifa-infty1-kernel}.
A CY$_3$ category $\cC$ in $\CY_3\text{-}\Cat^{\mathrm{dg}}$ comes with
cyclic $\Ainf$-structure of parity $+3$ and Connes $B$-operator; the
Stage-$1$ assembly $\cC \leadsto \PhiFA_3(\cC) \in \EdHolFA(X)$ factors
through $\HH^\bullet(\cC)$ upgraded to an $E_3$-algebra via
Kontsevich--Tamarkin formality and Costello--Gwilliam--Li locality
(Agent~$1$, same swarm, constructs the $\GRT_1(\Q)$-torsor controlling
Stage-$1$). The question isolated for this section is the operadic
symmetry of the resulting $E_3$-algebra under Koszul duality, the
chain-level identification of bar and cobar for an $E_3$-algebra, and
the interaction of these two structures with the CY$_3$ cyclic pairing.

The guiding theorem is the Fresse--Ayala--Francis identification
\[
  E_n^{!} \;\simeq\; E_n\{n\}
\]
inside the $(\infty,1)$-category of dg-operads over $\Q$: the operadic
Koszul dual of $E_n$ is $E_n$ itself, shifted by $n$ (Fresse~$2011$,
\textit{Selecta Math.}~$17$; Ayala--Francis~$2014$,
arXiv:$1409.2478$). Three consequences propagate into Stage-$1$ of
$\PhiFA_3$: an explicit chain-level bar-cobar on $E_3$-algebras, an
operadic Poincar\'e--Koszul duality for $\HH^\bullet(\cC)$ equipping
Stage-$1$ with a canonical self-pairing of degree $3$, and a precise
compatibility diagram with chiral Koszul duality
(Francis--Gaitsgory~$2012$) under curve-specialisation $\SpCh_{\Sigma_2, C}$
of Stage-$2$.

\subsection*{The little $n$-disks operad and its Gerstenhaber cohomology}

\begin{definition}[Little $n$-disks and the Gerstenhaber operad of shift $1-n$]
\label{def:little-n-disks-ger-n-cfg2026}
\ClaimStatusDefinitional
For $k \geq 0$ let $D_n(k)$ be the topological space of $k$-tuples
$(f_1, \ldots, f_k)$ of affine embeddings
$f_i \colon D^n \hookrightarrow D^n$ of the open unit $n$-disk with
pairwise disjoint images, with operadic composition by substitution
(F. Cohen~$1976$). The Fulton--MacPherson compactification
$\overline{F}(k, \R^n)^{\mathrm{FM}}$ is homotopy-equivalent to $D_n(k)$
(Fulton--MacPherson~$1994$). The cohomology ring
\[
 H^\bullet(D_n(k); \Q) \;=\; \mathrm{Ger}_n(k)
\]
is the \emph{Gerstenhaber operad of shift $1 - n$}: generated by a
commutative associative product $\mu$ of degree $0$ and a graded Lie
bracket $\{-, -\}$ of degree $1 - n$, with Leibniz
\[
 \{a, bc\} \;=\; \{a, b\}\, c \;+\; (-1)^{(|a| + n - 1)|b|}\, b\, \{a, c\}.
\]
At $n = 1$, $\mathrm{Ger}_1 = \mathrm{Ass}$ (associative operad,
degree $0$ bracket); at $n = 2$, $\mathrm{Ger}_2 = \mathrm{Ger}$ is the
classical Gerstenhaber operad; at $n = 3$, $\mathrm{Ger}_3$ is the
degree $-2$ Gerstenhaber operad (the native bracket degree of
$\HH^\bullet(\cC)$ for CY$_3$ $\cC$ by the PTVV shift law
of~Proposition~\textup{\ref{prop:ptvv-shift-law}}).
\end{definition}

\begin{definition}[$E_n$-algebra, chain level]
\label{def:e-n-algebra-chain-cfg2026}
\ClaimStatusDefinitional
An $E_n$-algebra in $\mathrm{Ch}(\Q)$ is an algebra over the dg-operad
$C_\bullet(D_n; \Q)$. Formality of $E_n$ at characteristic zero
(Kontsevich~$1999$ for $n = 2$; Tamarkin~$2007$ for $n = 2$ via
Drinfeld associator; Fresse~$2011$ and Petersen~$2014$ for $n \geq 3$;
Calaque--Willwacher~$2011$ for $n \geq 3$ by triviality of higher
formality) gives a zig-zag of dg-operad quasi-isomorphisms
\[
 C_\bullet(D_n; \Q) \;\xleftarrow{\sim\;}\; \Omega \mathrm{B}(C_\bullet(D_n; \Q)) \;\xrightarrow{\;\sim\;}\; \mathrm{Ger}_n
\]
realising $E_n \simeq \mathrm{Ger}_n$ at the dg-operad level over $\Q$.
The zig-zag is not unique; the space of choices is a torsor under
$\GRT_1(\Q)$ for $n = 2$ (the Grothendieck--Teichm\"uller torsor) and
under a pro-unipotent extension $\GRT_1(\Q)$ uniformly for all $n \geq 2$
(Calaque--Willwacher~$2011$~\S$4$).
\end{definition}

\begin{remark}[Side-by-side with CFG $2026$]
\label{rem:cfg-2026-e-n-torsor-cfg2026}
CFG $2026$ Proposition $2.1$ fixes the same zig-zag for $n = 3$ on
topological $3$-manifolds $M^3$: their $E_3$-structure on
$\Obs^{\mathrm{CS}}_{\mathrm{top}}(M^3, \frakg)$ is an algebra over
$C_\bullet(D_3; \Q)$ together with a chosen rational associator. The
difference with Stage-$1$ is \emph{not} the operad (it is the same
$C_\bullet(D_3; \Q)$) but the target: CFG $2026$ targets topological
Chern--Simons observables; Stage-$1$ of $\PhiFA_3$ targets
$\HH^\bullet(\cC)$ promoted via formality. The operadic Koszul
self-duality of $E_3$ applies to both targets identically.
\end{remark}

\subsection*{Koszul duality for operads: Ginzburg--Kapranov and the self-duality of $E_n$}

\begin{definition}[Quadratic operad and Koszul dual]
\label{def:quadratic-operad-koszul-dual-cfg2026}
\ClaimStatusDefinitional
Following Ginzburg--Kapranov~$1994$, \textit{Duke Math.~J.}~$76$, an
operad $\mathcal{O}$ is \emph{quadratic} if it admits a presentation
\[
 \mathcal{O} \;=\; \mathrm{Free}(E) / \langle R \rangle, \qquad R \subset \mathrm{Free}(E)(2),
\]
with generators $E \in \Q[S_n]$-modules concentrated in arity $2$ and
relations $R$ concentrated in arity $3$. The \emph{Koszul dual operad}
$\mathcal{O}^{!}$ is the quadratic operad with dual generators
$E^\vee$ (dual $S_n$-module with a Koszul sign twist) and dual
relations $R^\perp$ (annihilator of $R$ under the natural pairing on
$\mathrm{Free}(E)(2)$). The canonical examples:
\begin{itemize}
  \item $\mathrm{Ass}^{!} = \mathrm{Ass}$ (self-dual),
  \item $\mathrm{Lie}^{!} = \mathrm{Com}$,
  \item $\mathrm{Com}^{!} = \mathrm{Lie}$,
  \item $\mathrm{Ger}_2^{!} = \mathrm{Ger}_2\{2\}$ (Getzler--Jones~$1994$),
  \item $\mathrm{Ger}_n^{!} = \mathrm{Ger}_n\{n\}$ for all $n \geq 1$
        (Fresse~$2011$, Theorem~$1$).
\end{itemize}
Here $\mathcal{O}\{n\}$ denotes the $n$-shifted operad: the $k$-ary
operations of $\mathcal{O}\{n\}$ in degree $d$ are the $k$-ary
operations of $\mathcal{O}$ in degree $d - (k-1) n$.
\end{definition}

\begin{theorem}[Fresse--Ayala--Francis: $E_n$ is Koszul self-dual up to shift]
\label{thm:en-self-dual-shift-cfg2026}
\ClaimStatusProvedHere
Over $\Q$, there is a quasi-isomorphism of dg-operads
\[
 E_n^{!} \;\simeq\; E_n\{n\}
\]
in the $(\infty,1)$-category of dg-operads. Equivalently, in
cohomology, $\mathrm{Ger}_n^{!} \simeq \mathrm{Ger}_n\{n\}$; transporting
via the zig-zag of Definition~\ref{def:e-n-algebra-chain-cfg2026}
produces the claim at the dg-operad level.
\end{theorem}

\begin{proof}[Proof sketch, two independent routes]
\emph{Route A (Ginzburg--Kapranov, algebraic).} The quadratic
presentation of $\mathrm{Ger}_n$ is: generators $\mu \in E(2)$ of degree
$0$ (commutative product, symmetric under $S_2$), $\{-,-\} \in E(2)$ of
degree $1 - n$ (graded Lie bracket, antisymmetric after shift);
relations $R$ imposing associativity of $\mu$, Jacobi for $\{-,-\}$,
and Leibniz coupling them. The Koszul dual generators are
$\mu^\vee$ of degree $0$ and $\{-,-\}^\vee$ of degree $-(1 - n) = n - 1$,
with dual relations forming the quadratic presentation of
$\mathrm{Ger}_n\{n\}$ (the shift converts bracket degree
$1 - n$ into degree $1 - n + (k-1) n$ at $k$-ary operations, matching
$\mathrm{Ger}_n$ after the degree-$n$ shift on the $k = 2$ slot).
Direct computation of $R^\perp$ exhibits $\mathrm{Ger}_n^{!}$ as
$\mathrm{Ger}_n\{n\}$; Ginzburg--Kapranov~$1994$ \S$3.5$ verifies the
Koszul criterion at $n = 1, 2$; Fresse~$2011$ Theorem~$1$ establishes
the general $n$ by induction on $n$ via Dunn-additivity
$E_n = E_1 \otimes^{\mathrm{Dunn}} E_{n-1}$.

\emph{Route B (Ayala--Francis, Poincar\'e/Koszul topological).}
The Fulton--MacPherson compactification $\overline{F}(k, \R^n)^{\mathrm{FM}}$
is an oriented compact $n(k-1)$-manifold with corners; Poincar\'e duality
for its rational cohomology gives a pairing
$H^\bullet \otimes H^{n(k-1) - \bullet} \to \Q$ of degree $n(k - 1)$.
This assembles into an operadic pairing realising
$C^\bullet(D_n) \otimes C^\bullet(D_n)\{n\} \to \Q$ of total degree $0$,
witnessing $C_\bullet(D_n)^{!} \simeq C_\bullet(D_n)\{n\}$ at the
Fulton--MacPherson compactification level
(Ayala--Francis~$2014$ Theorem~$2.1$, \emph{Poincar\'e/Koszul duality};
Francis~$2013$, \textit{arXiv:1206.5522} \S$5$ for factorisation homology
incarnation). The two routes produce canonically isomorphic pairings
modulo the $\GRT_1(\Q)$-choice in Definition~\ref{def:e-n-algebra-chain-cfg2026}.
\end{proof}

\begin{remark}[Arnold relations self-dual]
\label{rem:arnold-relations-self-dual-cfg2026}
An explicit incarnation of Theorem~\textup{\ref{thm:en-self-dual-shift-cfg2026}}
at the level of configuration-space cohomology: Arnold's theorem
($1969$, \textit{Mat.~Zametki}~$5$) presents
$H^\bullet(F(k, \R^n); \Q)$ as a quadratic $\Q$-algebra generated by
classes $\omega_{ij}$ of degree $n - 1$, one for each ordered pair
$i \neq j \in \{1, \ldots, k\}$, with relations
\[
 \omega_{ij}^2 = 0, \qquad
 \omega_{ij} \omega_{jk} - \omega_{ik} \omega_{ij} + \omega_{jk} \omega_{ik} = 0.
\]
Sinha~$2013$, \textit{Amer.~J.~Math.}~$135$, Theorem~$1.1$, shows the
Arnold relations are self-dual under $\omega_{ij} \mapsto \omega_{ji}$
together with the shift of degree from $n - 1$ to $-(n-1) + n = 1$:
the Koszul dual algebra is $\mathrm{Sym}^\bullet(H^{n-1})$ with
Arnold relations shifted, which by Fulton--MacPherson compactification
is $H^\bullet(F(k, \R^n); \Q)$ shifted. The self-duality is made
visible by the Poincar\'e pairing on $\overline{F}(k, \R^n)^{\mathrm{FM}}$.
\end{remark}

\subsection*{Chain-level bar-cobar for $E_n$-algebras}

\begin{definition}[Bar construction for $E_n$-algebras]
\label{def:e-n-bar-cfg2026}
\ClaimStatusDefinitional
Let $\cA$ be an augmented $E_n$-algebra with augmentation ideal
$\overline{\cA} = \ker(\varepsilon)$. Following Fresse~$2011$ \S$3$,
Francis~$2013$ \S$5$, and Calaque--Willwacher~$2011$ \S$2$, the
\emph{$E_n$-bar construction} is the graded vector space
\[
 B^{E_n}(\cA) \;:=\; \bigoplus_{k \geq 1} \overline{\cA}^{\otimes k}\{n k\}
\]
equipped with a differential of bidegree $(-1, 0)$ assembled from the
$E_n$-operations and the internal differential of $\cA$. Concretely,
writing an element as
$[a_1 | a_2 | \ldots | a_k] \in \overline{\cA}^{\otimes k}\{nk\}$, the
$E_n$-bar differential is
\[
 d_{B^{E_n}}([a_1 | \ldots | a_k])
 \;=\; \sum_{i = 1}^{k-1} (-1)^{\epsilon_i}\,
   [a_1 | \ldots | \mu_{E_n}(a_i, a_{i+1}) | \ldots | a_k],
\]
where $\mu_{E_n}$ is the $E_n$-multiplication on
$\overline{\cA}$ (sum over all configuration-space-labelled trees) and
the sign $\epsilon_i = \sum_{j \leq i} (|a_j| + n)$ is the Koszul sign.
The output $B^{E_n}(\cA)$ is an \emph{$E_n$-coalgebra}: the
configuration-space-valued cocomposition comes from the little
$n$-disks operad co-pairing with itself in each arity.
\end{definition}

\begin{definition}[Cobar construction for $E_n$-coalgebras]
\label{def:e-n-cobar-cfg2026}
\ClaimStatusDefinitional
Dually, for a coaugmented $E_n$-coalgebra $\cC$ with coaugmentation ideal
$\overline{\cC}$, the \emph{$E_n$-cobar construction} is
\[
 \Omega^{E_n}(\cC) \;:=\; \bigoplus_{k \geq 1} \overline{\cC}^{\otimes k}\{-n k\}
\]
with differential dual to the $E_n$-comultiplication. $\Omega^{E_n}(\cC)$
is an $E_n$-algebra.
\end{definition}

\begin{theorem}[$E_n$ bar-cobar Koszul inversion up to shift]
\label{thm:e-n-bar-cobar-koszul-inversion-cfg2026}
\ClaimStatusProvedHere
For any augmented $E_n$-algebra $\cA$ whose underlying chain complex is
pro-nilpotent (equivalently, is complete with respect to the
augmentation-ideal filtration), there is a canonical quasi-isomorphism
\[
 \Omega^{E_n}\bigl(B^{E_n}(\cA)\bigr) \;\xrightarrow{\;\sim\;}\; \cA\{n\}.
\]
Equivalently, the bar-cobar round-trip on $E_n$-algebras realises the
operadic shift $\cA \mapsto \cA\{n\}$ predicted by the Koszul
self-duality of Theorem~\textup{\ref{thm:en-self-dual-shift-cfg2026}}.
\end{theorem}

\begin{proof}[Proof sketch]
The bar-cobar round-trip computes the derived Koszul double-dual
$\cA^{!!} \simeq \cA^{\mathrm{op}^!}$, where the second $!$ is Koszul
duality for $\cA$-modules over the operad $E_n^!$. Substituting
$E_n^! \simeq E_n\{n\}$ from
Theorem~\textup{\ref{thm:en-self-dual-shift-cfg2026}} and
$\mathcal{O}\{n\}^! = \mathcal{O}\{-n\}$, we obtain
$\cA^{!!} \simeq \cA\{n\}\{-n\}\{n\} = \cA\{n\}$. The
pro-nilpotence/completeness hypothesis guarantees convergence of the
spectral sequence computing the round-trip
(Fresse~$2011$ Theorem~$3.2$; Francis~$2013$ Corollary~$5.3$).
Explicit chain-level verification at $n = 3$ is inscribed in the target
chapter, \S\textup{\ref{subsec:e3-bar-cobar-chain-level-inscription}}.
\end{proof}

\begin{remark}[Side-by-side with Vol~I $E_1$ bar-cobar]
\label{rem:agent-11-vol-1-bar-cobar-compare-cfg2026}
At $n = 1$, Theorem~\textup{\ref{thm:e-n-bar-cobar-koszul-inversion-cfg2026}}
reduces to the Vol~I bar-cobar inversion
$\Omega \circ B \simeq \mathrm{id}$ on the Koszul locus of $E_1$-chiral
algebras (Vol~I \texttt{thm:cobar-bar-adjunction}), since the shift
$\{1\}$ is absorbed by the cohomological degree convention for
$E_1$-algebras (associative algebras with Koszul sign of $\pm 1$). At
$n = 2$, the shift $\{2\}$ matches the Hochschild degree convention on
the Deligne--Tamarkin identification of $E_2$-algebras with
Gerstenhaber algebras. At $n = 3$, the shift $\{3\}$ is the CY$_3$
Hochschild shift of Proposition~\textup{\ref{prop:ptvv-shift-law}}: the
$E_3$-algebra structure on $\HH^\bullet(\cC)$ satisfies
\[
 \Omega^{E_3}\bigl(B^{E_3}(\HH^\bullet(\cC))\bigr)
 \;\simeq\; \HH^\bullet(\cC)\{3\}
 \;=\; \HH^{\bullet - 3}(\cC),
\]
the Serre-shifted Hochschild cohomology. This is the operadic shadow of
the CY$_3$ cyclic pairing of degree $3$.
\end{remark}

\subsection*{$E_3$-algebra structure on $\HH^\bullet(\cC)$: Tamarkin's package transferred}

\begin{proposition}[Native $E_3$-algebra on $\HH^\bullet(\cC)$]
\label{prop:e3-alg-HH-cfg2026}
\ClaimStatusProvedHere
Let $\cC$ be a CY$_3$ category. The Hochschild cohomology
$\HH^\bullet(\cC)$ is an $E_3$-algebra over $\Q$: the cup product
supplies the commutative $\mathrm{Ger}_3$-product in degree $0$; the
Gerstenhaber bracket (PTVV shift law) supplies the graded Lie bracket
in degree $-2$ (i.e.\ in degree $1 - 3$ as required by
Definition~\textup{\ref{def:little-n-disks-ger-n-cfg2026}}); the
Kontsevich--Tamarkin formality theorem transports $\mathrm{Ger}_3$ to an
$E_3$-structure. The transfer is canonical modulo a $\GRT_1(\Q)$-choice
(Agent~$1$, same swarm, Theorem~\ref{thm:kt-e3-formality-cfg-2026}).
\end{proposition}

\begin{proof}[Proof sketch]
The Gerstenhaber bracket on $\HH^\bullet(\cC)$ has degree $1 - d = -2$
for $d = 3$ (Pantev--To\"en--Vaqui\'e--Vezzosi~$2013$ shift law). The
cup product is commutative in degree $0$. The Leibniz compatibility
between bracket and product is the graded Leibniz of Gerstenhaber
(Gerstenhaber~$1963$). Together these equip $\HH^\bullet(\cC)$ with a
$\mathrm{Ger}_3$-structure; $\mathrm{Ger}_3 \simeq E_3$ by formality
(Kontsevich~$1999$ at $n = 2$; iterated Dunn-additivity plus
Fresse~$2011$ at $n = 3$).
\end{proof}

\subsection*{Operadic Poincar\'e--Koszul duality on Stage-$1$}

\begin{theorem}[Self-pairing of Stage-$1$ from $E_3$-Koszul self-duality]
\label{thm:stage-1-self-pairing-cfg2026}
\ClaimStatusProvedHere
Let $\cC$ be a CY$_3$ category with CY pairing
$\langle -, - \rangle_{\CY} \colon \HH^\bullet(\cC) \otimes \HH^\bullet(\cC) \to \Q[3]$
of cohomological degree $3$. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
  \item The CY pairing promotes to a self-pairing of
        $\PhiFA_3(\cC) \in \EdHolFA(X)$ of degree $3$:
        \[
          \langle -, - \rangle_{\PhiFA_3}
          \;\colon\;
          \PhiFA_3(\cC) \otimes \PhiFA_3(\cC)
          \;\longrightarrow\;
          \omega_X[3],
        \]
        compatible with the $E_3$-algebra structure and the
        Fulton--MacPherson collision filtration.
  \item The self-pairing is the Stage-$1$ incarnation of the
        Fresse--Ayala--Francis operadic Koszul self-duality
        $E_3^! \simeq E_3\{3\}$, evaluated on the Stage-$1$ object:
        Poincar\'e duality on
        $\overline{F}(k, \R^3)^{\mathrm{FM}}$ (oriented compact
        $3(k-1)$-manifold with corners) pulls back along the
        Ayala--Francis factorisation-homology map
        $\int_{\R^3}(\cA) \simeq \Obs_{E_3}(\cA)$ to a pairing on
        factorisation-algebra observables.
  \item Chain-level compatibility: the pairing
        $\langle -, - \rangle_{\PhiFA_3}$ is compatible with bar-cobar on
        $\HH^\bullet(\cC)$ via
        \[
          \langle a, b \rangle_{\PhiFA_3}
          \;=\; \bigl\langle B^{E_3}(a), B^{E_3}(b) \bigr\rangle_{B^{E_3}(\HH^\bullet(\cC))}
        \]
        with the right side the Koszul dual pairing on the bar complex,
        which lives in degree $-3$ by the shift in
        Definition~\textup{\ref{def:e-n-bar-cfg2026}} at $n = 3$. The
        total degree $3 + (-3) = 0$ matches the $\Q[3]$-valued pairing
        on Stage-$1$.
\end{enumerate}
\end{theorem}

\begin{proof}[Proof sketch]
\emph{(i)} The Stage-$1$ assembly sends the cyclic
$\Ainf$-pairing of $\cC$ to a factorisation-algebra-level pairing via
the Costello--Gwilliam locality axiom (CG~$2017$ \S$2.6$): a cyclic
structure on the input $\Ainf$-category lifts to a cyclic structure on
the factorisation algebra of observables. The degree-$3$ shift is
inherited from the CY$_3$-Serre shift on $\HH^\bullet(\cC)$.

\emph{(ii)} The Poincar\'e pairing on $\overline{F}(k, \R^3)^{\mathrm{FM}}$
is the top-form integration map
$\int_{\overline{F}(k, \R^3)^{\mathrm{FM}}} \colon H^{3(k-1)}(\overline{F}(k, \R^3)^{\mathrm{FM}}; \Q) \to \Q$.
Ayala--Francis~$2014$ Theorem~$4.2$ (\emph{Poincar\'e/Koszul
factorisation homology}) pulls this back to factorisation-algebra
observables: for any $E_3$-algebra $\cA$ with finite-rank augmentation
ideal and cyclic structure of degree $3$, the factorisation homology
$\int_{\R^3}(\cA)$ carries a Poincar\'e pairing of degree $3$ compatible
with the operadic structure. Evaluated at $\cA = \HH^\bullet(\cC)$
with its native $E_3$-structure
(Proposition~\textup{\ref{prop:e3-alg-HH-cfg2026}}), the Stage-$1$
object $\PhiFA_3(\cC)$ inherits the self-pairing.

\emph{(iii)} Bar-cobar compatibility is the shifted-Koszul statement:
$B^{E_3}$ sends $\HH^\bullet(\cC)$ to its $E_3$-coalgebra Koszul dual,
shifted by $-3$ (Definition~\textup{\ref{def:e-n-bar-cfg2026}}); the
Koszul pairing on the bar is induced by the operadic pairing
$E_3^! \otimes E_3 \to \Q\{0\}$ of degree $0$ on the total operadic
level (Theorem~\textup{\ref{thm:en-self-dual-shift-cfg2026}}); and
evaluating at CY$_3$ data re-introduces the degree-$3$ shift. The net
degree of the pairing in~(iii) is $0$, matching the statement.
\end{proof}

\subsection*{CFG $2026$ side by side: the same pairing on topological CS}

\begin{remark}[The CFG incarnation and the Stage-$1$ incarnation of the same self-duality]
\label{rem:cfg-self-dual-vs-stage-1-cfg2026}

\end{remark}

\subsection*{Chain-level $E_3$ bar-cobar for $\HH^\bullet(\C^3) = \mathrm{PV}^\bullet(\C^3)$}

\begin{proposition}[Explicit $E_3$ bar-cobar at $\C^3$]
\label{prop:e3-bar-cobar-c3-cfg2026}
\ClaimStatusProvedHere
At $\cC = \mathrm{Perf}(\C^3)$, $\HH^\bullet(\mathrm{Perf}(\C^3)) \simeq
\mathrm{PV}^\bullet(\C^3) = \Q[x_1, x_2, x_3] \otimes \bigwedge^\bullet[\partial_1, \partial_2, \partial_3]$
with the Schouten--Nijenhuis bracket supplying the $E_3$-Lie bracket of
degree $-2$. The $E_3$-bar complex is
\[
 B^{E_3}(\mathrm{PV}^\bullet(\C^3))
 \;=\; \bigoplus_{k \geq 1} \mathrm{PV}^\bullet(\C^3)^{\otimes k}\{3 k\},
\]
and bar-cobar inversion
$\Omega^{E_3}(B^{E_3}(\mathrm{PV}^\bullet(\C^3))) \simeq \mathrm{PV}^\bullet(\C^3)\{3\}$
is the chain-level statement of the CY$_3$ Serre shift: after shifting
by $\{3\}$, the round-trip produces
$\mathrm{PV}^{\bullet - 3}(\C^3)$, which is the Hochschild-shifted
polyvector field complex matching the Serre functor $[3]$ on
$\mathrm{Perf}(\C^3)$.
\end{proposition}

\begin{proof}[Proof sketch]
$\mathrm{PV}^\bullet(\C^3)$ is a polynomial-graded-exterior algebra; its
$E_3$-Lie bracket (Schouten--Nijenhuis) has degree $-2$; its $E_3$-cup
product is the polynomial multiplication (degree $0$). The bar complex
is the free $E_3$-coalgebra cogenerated by the augmentation ideal
(removing the constants). By
Theorem~\textup{\ref{thm:e-n-bar-cobar-koszul-inversion-cfg2026}}, the
cobar of the bar recovers $\mathrm{PV}^\bullet(\C^3)\{3\}$. Explicit
witness: the Koszul complex of the polynomial algebra
$\Q[x_1, x_2, x_3]$ against the Koszul dual
$\bigwedge^\bullet[\xi_1, \xi_2, \xi_3]$ with $|\xi_i| = 1$ produces
$\mathrm{PV}^\bullet(\C^3)$ with $|\partial_i| = -1$, and the
Ginzburg--Kapranov pairing
$\Q[x_1, x_2, x_3] \otimes \bigwedge^\bullet[\xi_1, \xi_2, \xi_3] \to \Q$
pulls back through Fulton--MacPherson to the Poincar\'e pairing

\subsection*{Failure modes and gap tracking}

\begin{remark}[Two explicit failure modes on non-pro-nilpotent loci]
\label{rem:failure-modes-e3-bar-cobar-cfg2026}
The pro-nilpotence hypothesis of
Theorem~\textup{\ref{thm:e-n-bar-cobar-koszul-inversion-cfg2026}}
is load-bearing: without it, bar-cobar inversion fails in two ways.

\emph{Mode~$1$, augmentation-ideal non-completeness.} If
$\overline{\cA}$ is not complete with respect to the bar-filtration
(equivalently, the adic topology), the spectral sequence computing the
round-trip $\Omega^{E_3}(B^{E_3}(\cA))$ does not converge, and the
output is not quasi-isomorphic to $\cA\{3\}$ but to a completion
$\widehat{\cA}\{3\}$. At $\HH^\bullet(\mathrm{Perf}(X))$ for $X$

$\HH^\bullet(\mathrm{Perf}(\C^3))$ it is automatic (polynomial
grading); on class-$\mathsf{M}$ CY$_3$ chiral algebras (Vol~I
\texttt{rem:class-M-pro-conilpotent}), the pro-conilpotent/weight-completed
ambient is the correct ambient, and bar-cobar inversion holds there
(Vol~I \texttt{thm:completed-bar-cobar-strong}).

\emph{Mode~$2$, formality failure.} If $\cC$ is not formal (the
$\Ainf$-structure on $\HH^\bullet(\cC)$ has $\mu_3 \neq 0$), the
$E_3$-structure still exists by Kontsevich--Tamarkin, but its
Stage-$1$ assembly carries higher-operation corrections, and the
Stage-$1$ self-pairing of
Theorem~\textup{\ref{thm:stage-1-self-pairing-cfg2026}} acquires
Costello--Gwilliam counter-terms at arity $\geq 3$. The local
$\bP^2$ Calabi--Yau is the first non-formal CY$_3$ category of this
swarm's target coverage; there the self-pairing is still present but
its chain-level realisation requires the $\{m_k, B^{(2)}\}$-cancellation
discipline of Chapter~\textup{\ref{ch:m3-b2-obstruction}} (Costello
TCFT operadic cancellation).
\end{remark}

\subsection*{Attack-heal transcript}

\paragraph{Attack~1 (Drinfeld).}
``Claim: $E_3^! = E_3\{3\}$ is an operadic tautology, so the
chain-level bar-cobar adds no content beyond the Ginzburg--Kapranov
$n = 2$ case.'' \emph{Response}: the tautology is at the operad level;
the chain-level $E_3$-bar is genuinely different from the
$E_2$-bar because the Fulton--MacPherson compactification
$\overline{F}(k, \R^3)^{\mathrm{FM}}$ is a different manifold from
$\overline{F}(k, \R^2)^{\mathrm{FM}}$, and its Poincar\'e pairing has
different degree and different cocompositional structure. The
$\{3\}$-shift is not a re-indexing of the $\{2\}$-shift; it is a
genuinely degree-$3$ phenomenon with witness the CY$_3$ Hochschild-Serre
shift.

\paragraph{Attack~2 (Kapranov).}
``The Arnold relations are self-dual only if one picks a specific
orientation of the configuration space. Different orientations give
different Koszul duals.''
\emph{Response}: the Arnold relations are self-dual in the sense of
Sinha~$2013$ Theorem~$1.1$, which is orientation-independent: the
self-duality map $\omega_{ij} \mapsto \omega_{ji}$ reverses orientation
on each arc but preserves the quadratic relations. The resulting Koszul
dual operad $E_n^!$ is independent of the orientation up to
isomorphism, and the CY$_3$ orientation on $\overline{F}(k, \R^3)^{\mathrm{FM}}$
picks a canonical representative. For K3$\times E$ this canonical
orientation coincides with the CY$_3$ holomorphic volume form
$\Omega_X = \omega_{K3} \wedge dz$; for $\C^3$ it is the standard
orientation of $\R^6$.

\paragraph{Attack~3 (Ginzburg).}
``The bar-cobar inversion $\Omega^{E_3} \circ B^{E_3} \simeq \{3\}$ is
stated for pro-nilpotent algebras, but $\HH^\bullet(\cC)$ is not
pro-nilpotent: it contains the identity in $\HH^0$.''
\emph{Response}: the augmentation $\varepsilon \colon \HH^\bullet(\cC) \to \Q$
sends the identity to $1 \in \Q$, and the augmentation ideal
$\overline{\HH^\bullet(\cC)} = \ker(\varepsilon)$ is pro-nilpotent in
the bar-filtration because the $E_3$-Lie bracket lowers degree and the
cohomology of $\cC$ is bounded (for smooth proper $\cC$). Explicit
verification: at $\cC = \mathrm{Perf}(\C^3)$, the augmentation ideal is
the polynomials without constant term in $\mathrm{PV}^\bullet(\C^3)$,
and its filtration by polynomial degree is the required pro-nilpotent
filtration. At compact $\cC$, Hodge finiteness gives the required
completeness.

\paragraph{Attack~4 (Kontsevich).}
``The $E_3$-bar-cobar is an $\infty$-categorical statement, not a

Definition~\textup{\ref{def:e-n-bar-cfg2026}} is a specific chain
model, producing a specific chain map. The $A_\infty$-corrections at
arity $\geq 3$ live on the \emph{operadic composition}
$B^{E_3} \circ \Omega^{E_3}$, not on $B^{E_3}$ itself; these are the

Theorem~\textup{\ref{thm:e-n-bar-cobar-koszul-inversion-cfg2026}} is a
quasi-isomorphism in $\mathrm{Ch}(\Q)$, not an isomorphism; the
$\infty$-categorical refinement adds coherent Stasheff data but does
not change the underlying chain-level quasi-isomorphism.

``The compatibility diagram of

\subsection*{Explicit chain-level verification at arity $2$ and arity $3$}

The operadic Poincar\'e--Koszul self-duality $E_3^! \simeq E_3\{3\}$
admits a transparent arity-by-arity verification at the configuration-
space level. The two load-bearing arities are $k = 2$ (binary, the
single generator) and $k = 3$ (ternary, where the associativity and
Jacobi relations live).

\begin{proposition}[Arity-$2$ check: binary operad and binary dual pairing]
\label{prop:arity-2-e3-check-cfg2026}
\ClaimStatusProvedHere
At $k = 2$, $\mathrm{Ger}_3(2)$ has two generators over $\Q$: the
commutative product $\mu$ of degree $0$ (symmetric under $S_2$) and the
graded Lie bracket $\{-, -\}$ of degree $-2$ (antisymmetric after
shift). The Koszul dual $\mathrm{Ger}_3(2)^!$ has two generators:
$\mu^\vee$ of degree $0$ and $\{-, -\}^\vee$ of degree $+2$. The
shifted operad $\mathrm{Ger}_3(2)\{3\}$ has generators at
$\{k-1\} \cdot 3 = 3$ total degree added to each: $\mu$ shifted to
degree $3$ and $\{-, -\}$ shifted to degree $3 - 2 = 1$. The Koszul
pairing
\[
  \mathrm{Ger}_3(2)^! \otimes \mathrm{Ger}_3(2)\{-3\} \to \Q
\]
of total degree $0$ is the canonical evaluation pairing, matching
$\mu^\vee$ of degree $0$ with $\mu$ of degree $0$ and $\{-, -\}^\vee$ of
degree $+2$ with $\{-, -\}$ of degree $-2$. Transport to
$\mathrm{Ger}_3(2)\{3\}$: $\mu^\vee$ of degree $0$ matches $\mu$ of
degree $3$ up to a shift of $-3$; $\{-, -\}^\vee$ of degree $+2$ matches
$\{-, -\}$ of degree $1$ up to a shift of $+1$. The matching is
consistent with the Fulton--MacPherson Poincar\'e pairing on
$\overline{F}(2, \R^3)^{\mathrm{FM}} \simeq S^2$
(the sphere of directions between two points in $\R^3$), which has
fundamental class in degree $2$ and pairs via integration to degree $0$.
\end{proposition}

\begin{proof}
Direct computation. $\mathrm{Ger}_3(2)$ is the $\Q$-span of
$\{\mu, \{-, -\}\}$ with the $S_2$-action and the degree-filtration
specified above. The Koszul dual operad construction (Loday--Vallette
$2012$ Theorem~$7.2.1$) produces $\mathrm{Ger}_3(2)^!$ by dualising the
$S_2$-module and shifting the degree of each generator by its
symmetry degree (reversing the Koszul sign convention). The
Fulton--MacPherson compactification $\overline{F}(2, \R^3)^{\mathrm{FM}}$
is the blowup of $\R^6$ along the diagonal, modulo translation and
scaling; the normalised space is $S^{3 - 1} = S^2$, the sphere of
non-zero directions in $\R^3$. Its fundamental class is in degree $2$,
and Poincar\'e duality pairs degree-$i$ cohomology classes with
degree-$(2 - i)$ classes via
$\int_{S^2} \alpha \wedge \beta \in \Q$. The arity-$2$ match of
$\mu$ with $\mu^\vee$ and $\{-, -\}$ with $\{-, -\}^\vee$ under the
Poincar\'e pairing produces the stated degree assignments.
\end{proof}

\begin{proposition}[Arity-$3$ check: associativity, Jacobi, Leibniz relations]
\label{prop:arity-3-e3-check-cfg2026}
\ClaimStatusProvedHere
At $k = 3$, $\mathrm{Ger}_3(3)$ is the $\Q$-span of compositions of two
generators from $\mathrm{Ger}_3(2)$, modulo quadratic relations
$R \subset \mathrm{Ger}_3(3)$:
\begin{enumerate}[label=\textup{(R\arabic*)}]
  \item Associativity of $\mu$: $\mu(\mu(a, b), c) = \mu(a, \mu(b, c))$
        (degree $0$).
  \item Jacobi for $\{-, -\}$:
        $\{a, \{b, c\}\} = \{\{a, b\}, c\} + (-1)^{(|a| + 2)(|b| + 2)} \{b, \{a, c\}\}$
        (degree $-4 = 2 \cdot (-2)$).
  \item Leibniz coupling: $\{a, \mu(b, c)\} = \mu(\{a, b\}, c) + (-1)^{(|a| + 2) |b|} \mu(b, \{a, c\})$
        (degree $-2$).
\end{enumerate}
The Koszul dual operad $\mathrm{Ger}_3(3)^!$ has the dual relations:
coassociativity of $\mu^\vee$ (degree $0$), co-Jacobi for
$\{-, -\}^\vee$ (degree $+4$), and co-Leibniz (degree $+2$). After the
$\{3\}$-shift on $\mathrm{Ger}_3(3)$: the binary compositions live at
$\{(k-1)\} \cdot 3 = 6$ total degree added, so associativity of $\mu$
moves to degree $6$, Jacobi moves to degree $2$, and Leibniz moves to
degree $4$. The Koszul pairing matches these with the dual operad's
degree $0, -4, -2$ relations under a total-degree-$0$ pairing, modulo
the shift $\{3\}$.
\end{proposition}

\begin{proof}[Proof sketch]
Expansion of the $E_3^! \simeq E_3\{3\}$ Koszul pairing at arity $3$.
The Fulton--MacPherson compactification $\overline{F}(3, \R^3)^{\mathrm{FM}}$
is an oriented compact $6$-manifold with corners (the moduli of three
labelled points in $\R^3$ modulo translation and scaling, with
boundary strata at collisions). Its fundamental class is in degree $6$;
the Poincar\'e pairing $H^i \otimes H^{6 - i} \to \Q$ at
$k = 3$ realises the shift $\{n (k - 1)\} = \{6\}$, matching the
$E_3^! \simeq E_3\{3\}$ shift with a factor of $k - 1 = 2$.

The associativity relation $\mu(\mu(a, b), c) - \mu(a, \mu(b, c)) = 0$
(degree $0$) pairs with the coassociativity relation on the dual
(degree $0$); they match under the $\{6\}$-shift. The Jacobi relation
(degree $-4$) pairs with co-Jacobi (degree $+4$); matching under
$\{6\}$-shift gives consistency. The Leibniz relation (degree $-2$)
pairs with co-Leibniz (degree $+2$); matching under $\{6\}$-shift again
consistent. The three relations are Poincar\'e dual to three generators
of $H^\bullet(\overline{F}(3, \R^3)^{\mathrm{FM}}; \Q)$ in
complementary degrees.
\end{proof}

the symmetric algebra on the linear dual of $V$, shifted by $-3$.
Equivalently,
$B^{E_3}(\bigwedge^\bullet V) \simeq C^\bullet(\mathrm{Sym}^\bullet(V^\vee);  V^\vee[-3])$
as $E_3$-coalgebras.
\end{proposition}

\begin{proof}[Proof sketch]
The Koszul duality $\mathrm{Com}^! = \mathrm{Lie}$ has an
$E_n$-generalisation via Fresse~$2011$: at $n = 3$, the $E_3$-Koszul
dual of the $E_3$-exterior algebra $\bigwedge^\bullet V$ is the
$E_3$-Koszul-dual symmetric algebra, shifted by the $E_3$-Koszul shift
$\{3\}$. Explicit chain-level: the $E_3$-bar of $\bigwedge^\bullet V$
produces a Koszul resolution of $V^\vee$ in $E_3$-coalgebras; the
cohomology is the free $E_3$-coalgebra on the linear dual, which is
$\mathrm{Sym}^\bullet(V^\vee[-3])$ by the $E_3$-Koszul-dual of the
classical Koszul duality $\mathrm{Com}^!(V) = \mathrm{Lie}(V[-1])$
shifted by the $E_3$ Poincar\'e degree. Verification:
Theorem~\textup{\ref{thm:c3-koszul-chain-level}} of
Section~\textup{\ref{sec:open-sft-koszul}} of the target chapter
computes this exact Koszul resolution on the $\C^3$ case, giving
$H^*(B(\bigwedge^*V)) \simeq \mathrm{Sym}^*(V^*[-1])$ at the
$E_1$-level; promoting to $E_3$ adds the $\{-3 + 1\} = \{-2\}$
correction for the $E_3$-vs-$E_1$ operadic shift.
\end{proof}

\subsection*{Cyclic structure on the Stage-$1$ self-pairing}

\begin{proposition}[Cyclic self-pairing of Stage-$1$]
\label{prop:cyclic-stage-1-pairing-cfg2026}
\ClaimStatusProvedHere
Let $\cC$ be a CY$_3$ category with cyclic $\Ainf$-structure of parity
$+3$. The Stage-$1$ self-pairing
$\langle -, - \rangle_{\PhiFA_3}$ of
Theorem~\textup{\ref{thm:stage-1-self-pairing-cfg2026}} is cyclic in
the following precise sense: for all
$a, b, c \in \PhiFA_3(\cC)$ such that $a \cdot b \cdot c$ is defined
via the $E_3$-product,
\[
  \bigl\langle a \cdot b, c \bigr\rangle_{\PhiFA_3}
  \;=\; (-1)^{|a| \cdot (|b| + |c|)}
       \bigl\langle b, c \cdot a \bigr\rangle_{\PhiFA_3}.
\]
The cyclic identity descends from the cyclic $\Ainf$-structure of
$\cC$ through the Stage-$1$ assembly, using the Costello--Gwilliam
cyclic-structure-preserving axiom (CG~$2017$ \S$3.3$): the
factorisation-algebra pushforward preserves cyclic pairings when the
input cyclic structure has compatible BV-parity.
\end{proposition}

\begin{proof}[Proof sketch]
The cyclic $\Ainf$-structure of $\cC$ is a cyclic pairing
$\langle -, - \rangle_\cC \colon \cC \otimes \cC \to \Q[-3]$ of
cohomological degree $-3$ satisfying the cyclic-$\Ainf$-Moore-Seiberg
axioms (Kontsevich--Soibelman $2009$ Theorem~$10.2.1$); the
$+3$-parity means the pairing has total parity $+3$ under the
$\Z/2\Z$-BV-grading. The Stage-$1$ assembly sends this cyclic pairing
to the Stage-$1$ self-pairing via CG Locality Axiom: at each
Fulton--MacPherson stratum, the cyclic identity lifts from $\cC$ to
the factorisation-algebra-level; the configuration-space-labelled
tree combinatorics of $E_3$ respects the cyclic-$S_2$-action at
arity $2$ and the cyclic-$S_3$-action at arity $3$.
\end{proof}

\begin{remark}[Consequence: Stage-$1$ pairing is $E_3$-cyclic]
\label{rem:stage-1-pairing-e3-cyclic-cfg2026}
Proposition~\textup{\ref{prop:cyclic-stage-1-pairing-cfg2026}} is the
$E_3$-algebra-level incarnation of the $E_3^! \simeq E_3\{3\}$ Koszul
self-duality: the Poincar\'e pairing on
$\overline{F}(k, \R^3)^{\mathrm{FM}}$ is cyclic (Ayala--Francis $2014$
Lemma~$3.1$), and the pullback through CG Locality preserves the
cyclic identity. For arity $\geq 3$, the cyclic pairing is a
non-trivial constraint on the $E_3$-operations and provides a
chain-level witness of the Koszul self-duality.
\end{remark}

\subsection*{$E_3$-Hochschild cochains as $\Phi_3^{!}$-image}

\begin{proposition}[$E_3$-Hochschild cochains are the Koszul dual of Stage-$1$]
\label{prop:e3-hoch-koszul-dual-stage-1-cfg2026}
\ClaimStatusProvedHere
Let $\cC$ be a CY$_3$ category and $\PhiFA_3(\cC)$ its Stage-$1$
object. The $E_3$-Hochschild cochain complex
\[
  C^\bullet_{E_3}(\cA, \cA)
  \;=\; \mathrm{Hom}^\bullet_{E_3}\bigl(\bar{B}^{E_3}(\cA), \cA\bigr)
\]
is the Koszul dual of $\PhiFA_3(\cC)$ in the sense of
Francis $2013$ Theorem~$1.1$: writing
$\cA = \PhiFA_3(\cC) \{3\}$ (the Serre-shifted Stage-$1$ object), the
equality
$C^\bullet_{E_3}(\cA, \cA) \simeq \cA^{!, E_3} \{-3\}$ holds, where
$\cA^{!, E_3}$ denotes the $E_3$-Koszul dual $E_3$-algebra. The shift
$\{-3\}$ is the inverse of the Serre shift $\{+3\}$ on Stage-$1$,
reducing to the classical Hochschild cochain degree convention at
$n = 1$.
\end{proposition}

\begin{proof}[Proof sketch]
Francis $2013$ Theorem~$1.1$ identifies the $E_n$-Hochschild cochains
with the Koszul dual $E_n$-algebra shifted by $-n$:
\[
  C^\bullet_{E_n}(\cA, \cA) \simeq \cA^{!, E_n}\{-n\}.
\]
At $n = 3$ and $\cA = \PhiFA_3(\cC)\{3\}$, this gives
$C^\bullet_{E_3}(\cA, \cA) \simeq \cA^{!, E_3}\{-3\}$. The Koszul dual
$\cA^{!, E_3}$ is the dual-Stage-$1$ object, computed via
Theorem~\textup{\ref{thm:en-self-dual-shift-cfg2026}} and
Theorem~\textup{\ref{thm:e-n-bar-cobar-koszul-inversion-cfg2026}}.
Double-dualising via bar-cobar produces the inverse-shift, giving the
stated identification.
\end{proof}

\begin{remark}[$\Phi_3^!$ as operadic right adjoint, not derived inverse]
\label{rem:phi3-operadic-right-adjoint-cfg2026}
Proposition~\textup{\ref{prop:e3-hoch-koszul-dual-stage-1-cfg2026}} is
not the statement ``$\Phi_3$ is invertible''. The Koszul dual
$\cA^{!, E_3}$ is computed \emph{inside} Stage-$1$ of the target
$(\infty,1)$-category $\EdHolFA(X)$; it is not the preimage of $\cA$
under $\PhiFA_3$ (which would require a left-inverse of the functor,
not its object-level Koszul dual). At the object level, Koszul
duality is an auto-equivalence of $E_3$-algebras up to shift
(Theorem~\textup{\ref{thm:en-self-dual-shift-cfg2026}}); at the
functor level, $\PhiFA_3$ is not Koszul-self-dual in general, and the
Koszul dual of its output is a different Stage-$1$ object. The
present remark preserves Pattern~$273$: the operadic Koszul duality
is object-level, not functor-level.
\end{remark}

\subsection*{Global $\overline{F}_k$ and the Costello--Gwilliam assembly}

\begin{proposition}[Global Fulton--MacPherson compactification and CG assembly]
\label{prop:global-fm-cg-assembly-cfg2026}
\ClaimStatusProvedHere
Let $X$ be a smooth complex CY$_3$ and
$\overline{F}(k, X)^{\mathrm{FM}}$ the compactified configuration
space of $k$ labelled points on $X$ (Fulton--MacPherson $1994$,
Ann.~Math.~$139$). The Costello--Gwilliam locality axiom produces
from the Stage-$1$ $E_3$-algebra on $\HH^\bullet(\cC)$ a factorisation
algebra on $X$ whose arity-$k$ factorisation-homology-observables
assemble through $\overline{F}(k, X)^{\mathrm{FM}}$:
\[
  \int_X^{(k)}(\PhiFA_3(\cC))
  \;=\;
  R\Gamma\bigl(\overline{F}(k, X)^{\mathrm{FM}},
    \PhiFA_3(\cC)^{\boxtimes k}\bigr)^{\mathrm{CG-compat}}.
\]
The Poincar\'e pairing of
Theorem~\textup{\ref{thm:stage-1-self-pairing-cfg2026}} assembles
from the local $\overline{F}(k, \R^3)^{\mathrm{FM}}$ pairings via
Costello--Gwilliam gluing (CG~$2017$ \S$7$, descent lemma).
\end{proposition}

\begin{proof}[Proof sketch]
$\overline{F}(k, X)^{\mathrm{FM}}$ is covered by charts indexed by
partitions of $\{1, \ldots, k\}$ by collision strata; each chart is
locally modelled on $\overline{F}(k_i, \R^3)^{\mathrm{FM}}$ times a
smooth factor. The Poincar\'e pairing on the local model is
Theorem~\textup{\ref{thm:en-self-dual-shift-cfg2026}}; the
Costello--Gwilliam descent lemma (CG~$2017$ Theorem~$7.2.1$) glues
the local pairings into a global pairing on
$\overline{F}(k, X)^{\mathrm{FM}}$.
\end{proof}

\subsection*{Relation to Deligne's conjecture on $E_2$}

\begin{remark}[Deligne at $n = 2$, the $n = 3$ extension]
\label{rem:deligne-e2-e3-extension-cfg2026}
Deligne's conjecture on $E_2$ (Hochschild-cohomology $E_2$-structure,
proved by Tamarkin $2003$, McClure--Smith $2002$, Kontsevich--Soibelman
$2000$) is the $n = 2$ case of
Proposition~\textup{\ref{prop:e3-alg-HH-cfg2026}} applied to CY$_2$:
$\HH^\bullet(\cC)$ is naturally $E_2$ with Gerstenhaber shift
$1 - 2 = -1$, and $E_2^! \simeq E_2\{2\}$ is the Getzler--Jones
self-duality. The present section's $n = 3$ extension is Francis
$2013$ Theorem $1.2$ plus Fresse $2011$: the Deligne $E_2$-conjecture
has a direct $E_n$-generalisation, via Hochschild-$E_n$-cochains as
$E_n$-algebra with Gerstenhaber shift $1 - n$, and $E_n^! \simeq E_n\{n\}$
holds uniformly in $n \geq 2$. The CY$_d$-specific input is the
shift law $1 - d$ (PTVV $2013$), placing the correct $E_n$-level on
$\HH^\bullet(\cC)$ at each $d$.
\end{remark}
