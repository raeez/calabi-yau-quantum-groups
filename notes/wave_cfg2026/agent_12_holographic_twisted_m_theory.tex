\section*{Standing data}

Fix once and for all:

\begin{itemize}
\item $k = \C$.
\item $X$ a smooth projective Calabi--Yau threefold with holomorphic
volume form $\Omega_X \in H^0(X, \omega_X)$, $\omega_X \simeq \cO_X$.
Two running examples: \textbf{(CY3-local)} $X_0 = \C^3$ with
$\Omega_{\C^3} = dz_1 \wedge dz_2 \wedge dz_3$, and
\textbf{(CY3-compact)} $X = K3 \times E$ with
$\Omega_{K3 \times E} = \Omega_{K3} \wedge dz_E$.
\item $\frakg$ a finite-dimensional complex reductive Lie algebra.
The ``physical'' gauge algebra of twisted 11d supergravity in the
$\Omega$-background of Costello 2016 is $\widehat{\fgl}_1$ at
transverse rank $1$ and $\widehat{\fgl}_\infty$ at the infinite-rank
bulk; mathematically the source is finite-dimensional and the
infinite-rank structure appears only after Stage-1 globalisation.
\item $Y = X \times \R_t^{\mathrm{top}}$ the 5-manifold carrying 5d
holomorphic Chern--Simons (hCS) of Costello 2013/2017, with
$\R_t^{\mathrm{top}}$ the topological direction surviving the
holomorphic twist.
\item $(\epsilon_1, \epsilon_2, \epsilon_3) \in (\C^\times)^3$ the
equivariant parameters of the torus $T^3 \subset \mathrm{Aut}(X_0)$
on the local model $X_0 = \C^3$, restricted to the Calabi-Yau
locus $\epsilon_1 + \epsilon_2 + \epsilon_3 = 0$.
\item $C \subset X$ a holomorphic curve, serving as the chosen
Stage-2 specialisation datum. On $K3 \times E$ the canonical choice
is $C = E$.
\end{itemize}

The target is the two-stage chain-level chain
\[
 \{\text{twisted M-theory on } Y \times \R^5\}
 \;\xrightarrow{\;Q_{\mathrm{top/hol}}\;}\;
 \Obs^{5\mathrm{d\,hCS}}(Y, \frakg)
 \;\xrightarrow{\;\int_{\R_t}\;}\;
 \PhiFA_3(\Perf X)
 \;\xrightarrow{\;\SpCh_{\Sigma_2, C}\;}\;
 A_\cC,
\]
with each arrow chain-level and every $E_n$-level read off the
factorisation homology cochains of the preceding factor.

\section*{Programme: six ATTACK-HEAL cycles}

\begin{enumerate}
\item[(A1)] \textbf{Attack:} ``Twisted M-theory is a physics shorthand,
not a BV-rigorous field theory; the `reduction to 5d hCS' is
heuristic.''
\emph{Heal:} Costello 2016 arXiv:1610.04144 gives a rigorous
twisted-supergravity ground state on $\C^3 \times \R^5$ in the
Omega-background; its perturbative observables form a BV factorisation
algebra (Costello-Li arXiv:1606.00365), and compactification along the
$\R^2$-plane reduces to 5d hCS on $X \times \R_t^{\mathrm{top}}$ as a
theorem.
\item[(A2)] \textbf{Attack:} ``$E_3$ bulk observables on $X \times \R^2$
are not a theorem chain-level; they are a dimensional-analysis slogan
based on Dunn-Lurie additivity.''
\emph{Heal:} the holomorphic-topological tensor-factorisation of
Costello-Gwilliam Vol II \S4.10 (arXiv:2210.13036) promotes Dunn-Lurie
$E_3 \simeq E_2 \otimes E_1$ to a chain-level BV factorisation on
$X \times \R^2 = X \times \R_t \times \R_s$: one topological copy of
$E_1$ per $\R$ and the $E_2$ holomorphic factor on $\C^2$ sections of
$X$. This realises $\PhiFA_3(\Perf X)$ as the holomorphic factor of
the 5d hCS observable FA.
\item[(A3)] \textbf{Attack:} ``Twisted holography (Costello-Gaiotto 2018)
is a $d=2$ phenomenon; dragging it to $d=3$ on $K3 \times E$ is the
ghost of the absent $E_2$-chiral boundary.''
\emph{Heal:} Costello-Paquette 2020 (arXiv:2001.02177) extends twisted
holography to $E_1$-chiral boundaries of $3$d HT gauge theories, which
is the precise correct framework at $d=3$. The boundary of
$Y^{\mathrm{HT}}_3(X)$ is $E_1$-chiral, not $E_2$-chiral; VOA is
forbidden and the mode algebra is associative, with
$\Omega$-parameter reading the elliptic period.
\item[(A4)] \textbf{Attack:} ``The 5d hCS $\to$ Yangian VOA theorem
(Costello-Yagi arXiv:1810.01970) is only about $\C^2 \times \R_t$; it
says nothing about compact CY$_3$.''
\emph{Heal:} the theorem is a statement about the holomorphic factor,
and its chain-level proof uses only KT formality on $\C^2$. Under the
factorisation $\Phi_3 = \SpCh \circ \PhiFA_3$ with Stage-1 canonical up

\item[(A5)] \textbf{Attack:} ``$\kappa_{\mathrm{ch}}$ as a central charge of

$\kappa_{\mathrm{ch}}(A_\cC) = \sum_q (-1)^q h^{0,q}(X)$ combined with

\item[(A6)] \textbf{Attack:} ``The CFG 2026 envelope uses observables of

bulk-boundary reading is grafting different physical theories.''
\emph{Heal:} CFG 2026 is the topological-CS $E_3$ local-observables
envelope on a $3$-manifold $M^3$; our twisted-M-theory reading places

\end{enumerate}

\section{Cycle 1 --- Twisted M-theory as a rigorous BV field theory}
\label{sec:agent12-twisted-m-theory-rigorous}

\subsection*{Reduction datum}

Costello 2016 arXiv:1610.04144 \S1--\S3 constructs twisted 11d
supergravity on $\C^3 \times \R^5$ in the Omega-background with
rotation on a $\C \subset \R^5$ plane. The twisting operator is a
combination of topological and holomorphic supercharges
$Q = Q_{\mathrm{hol},\C^3} + Q_{\mathrm{top},\R^2}$; the cohomology
$H^\bullet(11\text{d SUGRA}, Q)$ collapses supergravity to a
perturbatively rigorous BV theory on $\C^3 \times \R^5$ whose gauge
algebra is $\widehat{\fgl}_\infty$ (Costello 2016 Thm.~4.1, the
algebra controlling M2/M5 perturbative reductions).

\begin{theorem}[Perturbative reduction to 5d hCS]
\label{thm:agent12-A1-perturbative-reduction}
The perturbative BV factorisation algebra of twisted 11d supergravity
on $\C^3 \times \R^5$ restricted to the slab $\C^3 \times \R_t$
($\R^2$-rotation plane compactified out by the Omega-background) is
equivalent, as a BV factorisation algebra, to 5d hCS on
$\C^3 \times \R_t$ with gauge $\widehat{\fgl}_\infty$.
\end{theorem}

\begin{proof}[Proof sketch (following Costello 2016 \S4--\S5)]
The Omega-background localises the $\R^2$-rotation plane to its fixed
point: the torus action $T \simeq \C^\times$ on $\R^2 \simeq \C$ with
generic equivariant parameter $\epsilon$ has unique fixed point $0$,
and equivariant localisation reduces the BV integral over fields
supported on $\R^2$ to the residue at the fixed point. The residual
5d hCS action on $\C^3 \times \R_t$ is the Costello 2013
arXiv:1303.2632 action~\eqref{eq:5d-hcs-bv-action-companion} below
with gauge algebra fixed by the $\widehat{\fgl}_\infty$-content of
the 11d parent:
\[
 S_{5\mathrm{d\,hCS}}[\cA]
 \;=\;
 \int_{\C^3 \times \R_t} \Omega_{\C^3}
 \wedge \operatorname{tr}\!\left(
 \cA \wedge \bar\partial_{\C^3} \cA
 + \cA \wedge d_t \cA
 + \tfrac{2}{3} \cA \wedge \cA \wedge \cA
 \right).
\]
The gauge algebra is $\widehat{\fgl}_\infty$; restriction to rank $r$
corresponds to compactifying on a 4-manifold carrying $r$ M5-branes.
This is the main theorem of Costello 2016.
\end{proof}

\begin{remark}[Holographic datum: one-loop output is paramodular weight $5$]
\label{rem:agent12-one-loop-paramodular-5}
The compactification of Theorem~\ref{thm:agent12-A1-perturbative-reduction}
on $X = K3 \times E$ is not the local $\C^3$ reduction but the global
CY$_3$ reduction with $24$ M5-branes wrapping the $I_1$ Kodaira fibres
of the elliptic $K3$. The one-loop partition function of the resulting
5d hCS theory on $(K3 \times E) \times \R_t$ contributes the
paramodular weight
\[
 \mathrm{wt}(\Delta_5)
 \;=\;
 \chi(\cO_{K3}) + \sum_{\text{24 fibres}} \Res(I_1)
 \;=\;
 2 + 3
 \;=\;
 5
\]
as the one-loop forced output of paramodular anomaly cancellation in
twisted 11d supergravity on $K3 \times T^2$ with $24$ M5-branes
(Harvey-Moore 1996 heterotic threshold; DVV 1996--97; DMVV 1997;
Witten 1995 M on $K3 \times T^3$). The $\chi(\cO_{K3}) = 2$ summand
is Mukai doubling on the K3-fibre; the $+3$ is the orbifold residue
of the $I_1$ Kodaira stratum summed over the 24-section of the
elliptic fibration.
\end{remark}

\section{Cycle 2 --- $E_3$ bulk observables chain-level via
holomorphic-topological factorisation}
\label{sec:agent12-E3-chain-level}

\subsection*{Target theorem}

The bulk local observables of twisted M-theory on $X \times \R^2$
form a chain-level $E_3$-algebra (equivalently, $E_3$-holomorphic
factorisation algebra on the holomorphic factor). The chain-level
construction factors as
\[
 \Obs^{\mathrm{tw-M}}_{\mathrm{loc}}(X \times \R^2)
 \;\simeq\;
 \Obs^{4\mathrm{d\,hol}}(X, \frakg)
 \;\boxtimes\;
 \Obs^{1\mathrm{d\,top}}(\R_t)
 \;\boxtimes\;
 \Obs^{1\mathrm{d\,top}}(\R_s),
\]
where the two topological $\R$-factors and the 4d holomorphic $X$-factor
are tensored by the Costello-Gwilliam holomorphic-topological tensor
theorem (Costello-Gwilliam Vol II \S4.10 arXiv:2210.13036).

\begin{theorem}[$E_3$ chain-level realisation of $\PhiFA_3(\Perf X)$]
\label{thm:agent12-A2-E3-chain-level}
Under the factorisation above, the holomorphic factor
$\Obs^{4\mathrm{d\,hol}}(X, \frakg)$ is a chain-level
$E_4$-algebra via Kontsevich-Tamarkin 4d formality on the Gerstenhaber
bracket of degree $-3$ on $\HH^\bullet(\Perf X)$
(Kontsevich 1999; Tamarkin 2007; Costello-Li 2016 arXiv:1606.00365;
Costello-Gwilliam-Li locality). The two topological factors contribute
$E_1 \otimes E_1 \simeq E_2$ by Dunn-Lurie additivity (Lurie HA 5.1.2.2).
Tensoring, the total bulk observable algebra is
$E_4 \otimes E_2 \simeq E_6$, but the Stage-1 output
$\PhiFA_3(\Perf X)$ carries only the $E_3$-subfactor obtained by
compactifying one of the $\R$-factors and restricting to the
$\R_s$-transverse slab. This gives the chain-level equivalence
\[
 \PhiFA_3(\Perf X)
 \;\simeq\;
 \int_{\R_s} \Obs^{\mathrm{tw-M}}_{\mathrm{loc}}(X \times \R^2)
 \;\in\;
 \EdHolFA(X).
\]
The Stage-1 output is the canonical $E_3$-holomorphic factorisation
algebra on the CY$_3$ datum.
\end{theorem}

\begin{proof}[Proof sketch]
The 5d hCS action of Theorem~\ref{thm:agent12-A1-perturbative-reduction}
is linear in the $\R_t$-direction: $\cA \wedge d_t \cA$ is a first-order
$t$-derivative. The classical BV complex on $X \times \R_t$ is therefore
a tensor product of complexes on the factors:
\[
 (\Omega^{0,\bullet}(X) \otimes \frakg[1], \bar\partial_X)
 \;\otimes\;
 (\Omega^\bullet(\R_t), d_t)
 \;\otimes\;
 (\text{antibracket}, S_{\mathrm{int}}).
\]
Costello-Gwilliam Vol II \S4.10 promotes this tensor decomposition to a
holomorphic-topological tensor product at the level of BV
factorisation algebras on the product stratification of $X \times \R_t$.
The holomorphic factor on $X$ is Costello-Li 4d hCS
(arXiv:1606.00365), with underlying $E_4$-algebra from Kontsevich
$E_d$-formality at $d = 4$. The topological factor on $\R_t$ is the
$E_1$-algebra $C^*_{\mathrm{Lie}}(\frakg[1])$ (Chevalley-Eilenberg
cochains); factorisation homology $\int_{\R_t}$ of $E_1$-algebras is the
associative enveloping algebra, giving
$\int_{\R_t} C^*_{\mathrm{Lie}}(\frakg[1]) \simeq U(\frakg[\![\hbar]\!])$
after BD quantisation (Francis 2013 arXiv:1303.0305 Theorem 2.29).
Restricting to Stage-1 (before $\SpCh$-specialisation) reads only the
holomorphic factor; this is $\PhiFA_3(\Perf X)$.
\end{proof}

\section{Cycle 3 --- Costello-Paquette $E_1$-chiral twisted holography
at $d = 3$}
\label{sec:agent12-costello-paquette-E1}

\subsection*{Why the VOA is absent at $d = 3$}

Costello-Gaiotto 2018 arXiv:1812.09257 twisted holography produces a
VOA on the boundary of a $4$d HT theory on $K3 \times \R_+^2$ (\emph{two}
topological directions). The boundary inherits both a holomorphic and
a topological direction from the bulk, acquiring an $E_2$-chiral
structure with full sphere-OPE.

At $d = 3$ on $X = K3 \times E$, the HT bulk is $3$d (one holomorphic
curve, $E$, plus one topological direction $\R_+$), and the boundary
factorisation algebra is $E_1$-chiral (locally-constant on a single
complex curve $E$), not $E_2$-chiral. Direct transplantation of
Costello-Gaiotto would produce a VOA that does not exist, as the
sphere-OPE datum lives in $H^\bullet(\mathrm{Ran}(\mathbb{A}^1),
\mathrm{Fact}_{E_2})$ with a one-dimensional $E$-factor whose
$H^1(E, \C)$ encodes the Omega-deformation parameter rather than the
sphere-OPE.

\begin{proposition}[Costello-Paquette bulk-boundary at $d = 3$]
\label{prop:agent12-A3-costello-paquette-d3}
Costello-Paquette 2020 arXiv:2001.02177 Theorem~4.7 extends twisted
holography to $E_1$-chiral boundaries of $3$d HT gauge theory. The
boundary datum of $Y^{\mathrm{HT}}_3(X)$ is a locally-constant
factorisation algebra $\cA^{\Eone}_\partial$ on a complex curve
$C \subset X$; its mode algebra $\mathrm{Modes}(\cA^{\Eone}_\partial)$
is an associative algebra (not a vertex algebra), whose pairing is the
linearised $E_1$-commutator, not the full Borcherds OPE. Via this
extension,
\[
 \Phi_3\bigl(D^b\mathrm{Coh}(X)\bigr)
 \;\simeq\;
 \cA^{\Eone}_\partial\bigl(Y^{\mathrm{HT}}_3(X)\bigr)
 \quad\text{on}\quad
 C = E \subset X,
\]
with $\Omega$-deformation parameter $\varepsilon$ set by the elliptic
period $\oint_\gamma \omega_E \in H^1(E, \C)$. The identification is
$E_1$-monoidal, not $E_2$-braided; the $r$-matrix $r_{\mathrm{CY}}$ is
an $E_1$-commutator shadow, not a braiding coefficient of a modular
tensor category.
\end{proposition}

\begin{remark}[Where the $E_2$-data hides]
\label{rem:agent12-E2-data-hides-centre}
The $E_2$-chiral lift of $\Phi_3(D^b\mathrm{Coh}(X))$ lives on the
Drinfeld centre $\cZ(\Rep^{\Eone}(\Phi_3(D^b\mathrm{Coh}(X))))$, one

$Y^{\mathrm{HT}}_3(X)$ on $X \times S^1$, not the character of
$\Phi_3(D^b\mathrm{Coh}(X))$ itself. Genus-$2$ $\Delta_5(Z) \in
M_5(\mathrm{Sp}_4(\Z))$ is the genus-$2$ shadow of the $E_1$-mode
character:
\[
 \Delta_5(Z)
 \;=\;
 \mathrm{ch}^{\mathrm{modes}}_{g=2}(\cA^{\Eone}_\partial)(Z)
 \;=\;
 \mathrm{Tr}^{\Eone}_{\cA^{\Eone}_\partial}\bigl(
 e^{2\pi i(\tau L_0 + z J_0 + \sigma \bar L_0)}\bigr),
\]
with genus-$2$ period matrix $Z = \begin{pmatrix} \tau & z\\ z & \sigma
\end{pmatrix} \in \mathbb{H}_2$ encoding the $\Omega_{K3 \times E}$-
expansion directions and the coupling $z$.
\end{remark}

\section{Cycle 4 --- The 5d hCS $\to$ Yangian VOA theorem as
$(\Sigma_2, C)$-specialisation of $\PhiFA_3$}
\label{sec:agent12-5d-hcs-yangian}

\subsection*{Costello-Yagi two-author result}

Costello-Yagi 2018 arXiv:1810.01970 Theorem~4.1: the perturbative BV
quantisation of 5d hCS on $\R_t \times \C^2$ with gauge $\frakg$
simply-laced bosonic produces the Yangian VOA $Y^{\mathrm{VOA}}(\frakg)$
to all orders in $\hbar$, with convergence (not merely asymptotic) by
Kontsevich-Tamarkin $E_d$-formality on the holomorphic $\C^2$-factor.
The three-author Costello-Gaiotto-Yagi 2021 arXiv:2103.01835 is a
different theorem (twisted supergravity dualities); not interchangeable
(AP-CY319 attribution discipline).

\begin{proposition}[Costello-Yagi as $(\Sigma_2, C)$-specialisation]
\label{prop:agent12-A4-costello-yagi-as-spch}
Let $X^{\mathrm{germ}} = \C^3 = \C^2 \times \C$ and fix the
specialisation datum $(\Sigma_2, C) = (\C^2\text{-germ}, \C\text{-germ})$.
Under the two-stage factorisation $\Phi_3 = \SpCh_{\Sigma_2, C} \circ
\PhiFA_3$:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $\PhiFA_3(\Perf \C^3)$ is realised as the 5d hCS BV observable
algebra on $\C^3 \times \R_t$ with gauge $\frakg$
(Proposition~\ref{prop:5d-hcs-bv-factorisation-companion} in the
master chapter).
\item $\SpCh_{\C^2, \C}$ applied to this Stage-1 object is the
$\R_t$-compactification ($\int_{\R_t} \Obs^{1\mathrm{d\,top}}
\simeq U(\frakg[\![\hbar]\!])$ by Francis 2013 Thm.~2.29) together with
specialisation to a holomorphic disc $\C \subset \C^2$.
\item The composite $\SpCh_{\C^2, \C}(\PhiFA_3(\Perf \C^3))$ is the
Yangian VOA $Y^{\mathrm{VOA}}(\frakg)$ for $\frakg$ simply-laced
bosonic.
\end{enumerate}
The Costello-Yagi theorem is thus the
$(\Sigma_2, C) = (\C^2\text{-germ}, \C\text{-germ})$-specialisation of
$\PhiFA_3$ on $X^{\mathrm{germ}} = \C^3$.
\end{proposition}

\begin{proof}[Proof sketch]
$(i)$ By Theorem~\ref{thm:agent12-A2-E3-chain-level}, the holomorphic
factor $\Obs^{4\mathrm{d\,hol}}(\C^3, \frakg)$ is the $E_4$-algebra
structure induced by Kontsevich-Tamarkin $E_d$-formality at $d = 4$ on
the polyvector-field dg-Lie $\HH^\bullet(\Perf \C^3) = \PV^\bullet(\C^3)$
via HKR. Tensoring with the topological $E_1$-factor on $\R_t$ and
reading the $E_3$-subfactor of Proposition~\ref{prop:5d-hcs-bv-factorisation}
gives $\PhiFA_3(\Perf \C^3) \in \EdHolFA(\C^3)$.

$(ii)$ $\SpCh_{\C^2, \C}$ is factorisation homology over
$\R_t \times \C^2$-fibre followed by restriction to a holomorphic
disc $\C \subset \C^2$. The $\R_t$-integral is $U(\frakg[\![\hbar]\!])$
by Francis 2013 Thm.~2.29. The $\C^2$-to-$\C$ restriction is
factorisation homology over a holomorphic disc, which is the
$E_2$-to-$E_1$ Dunn-Lurie step.

$(iii)$ The composite is the Costello-Yagi theorem
(arXiv:1810.01970 Thm.~4.1): at $\frakg$ simply-laced bosonic
($\frakg \in \{A_n, D_n, E_6, E_7, E_8\}$), the one-loop wheel
$\mathrm{Tr}(A\,\partial A\,\partial A\,\partial A)$ anomaly is
proportional to $d^{abc}_\frakg$, which vanishes by the cyclic-symmetric
trace identity on the adjoint. The Kontsevich integral bound on Feynman
weights and exponential propagator decay on $\C^2$ give convergence
to all orders. The $E_2$-structure on $Y^{\mathrm{VOA}}(\frakg)$ comes
from the holomorphic $\C^2$-directions after $\R_t$-compactification.
\end{proof}
