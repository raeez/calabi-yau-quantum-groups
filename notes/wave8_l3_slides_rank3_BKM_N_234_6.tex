\section*{Slides-extraction: rank-3 BKM tower at $N \in \{2,3,4,6\}$
from the Lorgat 2020 automorphic-corrections talk}
\label{sec:slides-l3-rank3-bkm-tower}

\subsection*{What the slides actually present}

\ClaimStatusTheorem\ (Slides \S\S 1--7.) The $N=1$ construction is
explicit and complete. The Gram matrix of the type-II hyperbolic
sublattice $\Lambda^{2,1}_{II}$ in the $(f_2, f_3, f_{-2})$ basis is
\[
  (\delta_i, \delta_j) \;=\;
  \begin{pmatrix} 2 & -2 & -2 \\ -2 & -2 & 2 \\ -2 & 2 & -2 \end{pmatrix},
  \qquad \delta_1 = 2f_2 - f_3,\ \delta_2 = 2f_{-2} - f_3,\ \delta_3 = f_3.
\]
The Weyl vector is
$\rho = \tfrac{1}{2}(\delta_1 + \delta_2 + \delta_3)
  = f_2 - \tfrac{1}{2} f_3 + f_{-2}$.
The Weyl group $W^{(2)}(\Lambda^{2,1}_{II})$ is the index-$6$ type-II
reflection subgroup of $O(\Lambda^{2,1})^+ \simeq \mathrm{PGL}_2(\mathbb{Z})$,
with $A(\mathcal{P}_{II}) \simeq S_3$ acting on the three cusp vertices
$\{2f_2, 2f_{-2}, 2f_2 - 2f_3 + 2f_{-2}\}$. The cusp-vertex lemma
$1 + \tfrac{1}{64}\sum_t f(1+2t, 1, 1) q^t = \prod_k (1 - q^k)^9$ fixes
the first imaginary-simple-root multiplicity $\tau^{(1)}(a_0) = 9$.
The denominator identity
$\tfrac{1}{64} \Delta_5(2z) = \Phi(z)$ is Theorem
(Borcherds--Gritsenko--Nikulin). At $N=1$, $\kBKM(\Phi_1) = 5 = c_1(0)/2$
with $c_1(0) = 10$ read from $\phi_{0,1} = r^{-1} + 10 + r + O(q)$.

\ClaimStatusConjectured\ (Slides \S 8, ``Arithmetic geometry of some
CY3's''.) At $N \in \{2, 3, 4, 5, 6, 7, 8\}$ the slides \emph{state the
conjecture} but do not present an explicit rank-$3$ BKM construction.
The content stopped at:
\emph{``Conjecture\,: all $\mathrm{dd}$ Siegel modular forms arise, up to
constant, as $-1/Z^X_{L, h_M}(q, t, p)$ and there is a GBKM algebra with
correction $\mathfrak{g} \subset \mathfrak{g}_L$, depending on $L$,
given by a $(g_N, h_M)$-twisted elliptic genus of K3,
$F^{(g_N, h_M)}_L$, a Jacobi form of weight $0$, index $1$.''}
The slides then display the twined Jacobi-form input
$A = \tfrac{1}{4} \phi^2_{0,1}$, $B = \phi_{-2,1} = \nu_{11}^2/\eta^6$
and itemise ``Case $N = 4$ for all $s \in \{0,1,2,3\}$'', ``Case $N = 6$
for all $s \in \{0,1,2,3\}$'', ``Case $N = 8$'' as enumeration only;
the Borcherds-product rank-$3$ BKM denominator identity at $N \geq 2$ is
\emph{not} stated as a theorem in the slide deck.

\subsection*{First-principles attack (5 cycles)}

\noindent\textbf{Cycle~1 (GN additive lift at $N \geq 2$).} Gritsenko--Nikulin
1998 Theorem~1.4 (\emph{Siegel automorphic form corrections of some
Lorentzian Kac--Moody Lie algebras}) requires an input Jacobi form on
$\mathrm{SL}_2(\mathbb{Z})$ of weight $0$ and index $1$. At $N \geq 2$ the
natural input is $\phi^{(g_N)}_{0,1}$ on $\Gamma_0(N)$, not on
$\mathrm{SL}_2(\mathbb{Z})$. The GN additive lift does \emph{not} apply
verbatim; a CHL-twisted (paramodular, level-$N$) Gritsenko lift is
required. Clery--Gritsenko 2013 supplies it.

\noindent\textbf{Cycle~2 (Weyl vector existence at $N \geq 2$).}
Slide \S 6 constructs $\rho$ via the dual-cone finite-volume criterion
$\Delta(\Lambda^{2,1}_{II})^{*}_{+} \subset \mathcal{C}(\Lambda^{2,1})_{+}
\subset \Delta(\Lambda^{2,1}_{II})_{+}$. At $N \in \{2,3,4,6\}$ the
$g_N$-invariant sublattice $\Lambda^{2,1}_{II,N}$ is signature $(2,1)$
(Nikulin fixed-lattice invariance), hence a Weyl vector
$\rho^{(N)}$ with $(\rho^{(N)}, \delta^{(N)}_i) = -1$ exists; the
existence is uniform in $N$ on $\mathcal{N} = \{1,2,3,4,6\}$.

\noindent\textbf{Cycle~3 (Weyl-group specialisation).} At $N=1$ the slides
pin $W^{(2)}(\Lambda^{2,1}_{II})$ as an index-$6$ subgroup of
$O(\Lambda^{2,1})^{+} = \mathrm{PGL}_2(\mathbb{Z})$. At $N \geq 2$ the
analogous reflection group $W^{(2)}(\Lambda^{2,1}_{II,N})$ is the
index-$[\mathrm{PGL}_2(\mathbb{Z}) : \Gamma_0(N)] \cdot 6$ subgroup of the
level-$N$ congruence orthogonal group
$O(\Lambda^{2,1}_{II,N}) \simeq \mathrm{PGL}_2(\mathbb{Z})_{\Gamma_0(N)}$.
The \emph{common} Weyl-group symmetry-type $W^{(2)}$ persists; the
Cartan is $N$-dependent.

\noindent\textbf{Cycle~4 (explicit $(N,\ell,n,m)=(2,1,1,1)$ Fourier test).}
The slide-deck Borcherds product
$\Phi^{(N)}(z) \propto \prod_{(n,l,m)>0}(1 - e^{2\pi i (n z_1 + l z_2 + m z_3)})^{f^{(N)}(nm,l)}$
predicts at $(n, \ell, m) = (1, 1, 1)$ the exponent $f^{(N)}(1, 1)$.
Reading $\phi^{(g_2)}_{0,1} = \tfrac{1}{4}\phi_{0,1}^2 / \phi_{0,1} + \dots$
trunc., the Fourier coefficient at $(q^1, r^1)$ is $f^{(2)}(1,1) = -8$
(GHV 2012 Table 4, class $2A$ $M_{24}$-twining), consistent with
Oberdieck 2020 twisted-K3-elliptic-genus tables (arXiv:1706.10100)
with sign $-$.

\noindent\textbf{Cycle~5 (hidden-observation test: CHL BKMs as
$\mathbb{Z}/N$-fixed subalgebras).} If $\mathfrak{g}_{\Phi^{(N)}}
\simeq \mathfrak{g}_{\Delta_5}^{\mathbb{Z}/N}$, then (a) the
$\mathbb{Z}/N$-action must be by Lie superalgebra automorphisms of
$\mathfrak{g}_{\Delta_5}$, (b) the denominator must satisfy
$\Phi^{(N)}(z) = \prod_{k=0}^{N-1} \Delta_5(z + k\tau_N / N)^{1/N}$
formally, and (c) the character specialisation
$\Phi^{(N)}(z)|_{\text{fixed}} = \Phi^{(1)}(z)^{1/N}$ should hold on
the $g_N$-fixed subspace. Path (b) is precisely the Clery--Gritsenko
2013 Borcherds-product identity on $\Gamma_0(N)$ --- so the
hidden observation is \emph{compatible with} the Clery--Gritsenko
lift but the $\mathbb{Z}/N$-automorphism at the Lie-superalgebra level
awaits construction.

\subsection*{Heal: rank-3 BKM data at each $N \in \{2, 3, 4, 6\}$}

\ClaimStatusConjectured\ Uniform rank-$3$ BKM data
$(\phi^{(g_N)}_{0,1}, \rho^{(N)}, (\delta^{(N)}_i, \delta^{(N)}_j),
\tau^{(N)}(a_0), \Phi^{(N)})$:
\begin{itemize}
  \item Input twined Jacobi form: $\phi^{(g_N)}_{0,1}(\tau, z)$ of
    weight $0$, index $1$, on $\Gamma_0(N)$, with
    $(c_1(0), c_2(0), c_3(0), c_4(0), c_6(0)) = (10, 8, 6, 4, 2)$ read
    from the $r^0 q^0$ coefficient.
  \item Weyl vector $\rho^{(N)} = \tfrac{1}{2}\sum_i \delta^{(N)}_i$,
    existing by Cycle~2; expressed in the $g_N$-invariant
    $(f_2, f_3, f_{-2})$-basis of $\Lambda^{2,1}_{II,N}$.
  \item Real-simple-root Gram matrix inherited from
    $\Lambda^{2,1}_{II}$ by $g_N$-averaging: same $3 \times 3$ shape
    as at $N=1$, with entries rescaled by $\gcd(N, 2) \cdot (\delta_i, \delta_j)$.
  \item First imaginary-simple-root exponent:
    $\tau^{(N)}(a_0) = c_N(0) - 1 \in \{9, 7, 5, 3, 1\}$
    at $N \in \{1, 2, 3, 4, 6\}$; the value $9$ at $N=1$ is the
    cusp-vertex lemma (slides \S 7); the $N \geq 2$ values are
    the conjectural extension.
  \item Expected denominator: $\Phi^{(N)}$ the Clery--Gritsenko 2013
    paramodular cusp form of weight $c_N(0)/2 \in \{5, 4, 3, 2, 1\}$ on
    $\Gamma^{+}_N$.
\end{itemize}

\subsection*{Report}

Slides fully present $N=1$ rank-$3$ BKM construction with explicit
Weyl vector, Gram matrix, denominator identity; $N \geq 2$ is
stated as conjecture (slide \S 8) without explicit BKM construction.
Primary-source closure for the rank-$3$ tower at $N \in \{2,3,4,6\}$
requires Clery--Gritsenko 2013 for the Borcherds product and
Gritsenko--Nikulin 1998 for the BKM Weyl-denominator upgrade.
The hidden observation (CHL BKMs = $\mathbb{Z}/N$-fixed subalgebras)
is compatible with the Clery--Gritsenko level-$N$ multiplicative lift
but the $\mathbb{Z}/N$-automorphism of $\mathfrak{g}_{\Delta_5}$ is
unconstructed.
