\title{Equivariant factorisation algebra on $K3\times E$:
symplectic $\Aut(K3)\times E$-translation action and the $N$-twisted
chiral BKM}
\author{Wave-14 I5/20}
\date{2026-04-22}

\begin{document}
\maketitle

Wave-13 G4 introduced the candidate chiral BKM factorisation algebra
$\cF^{\chir}_{\mathrm{BKM}}(X) := \CE^*_{\dbar,\chir}(\cE_X,\cO_X)$ on
the compact CY$_3$ $X = K3\times E$; on the total space
$\kappach(\cF^{\chir}_{\mathrm{BKM}}) = \chi(\cO_X) = 2\cdot 0 = 0$ by
K\"unneth, distinct from the fibre $\kappacat(K3)=2$,
from $\kappaBKM(\Phi_1) = c_1(0)/2 = 5$, and from the fibre-rank $24$.
The symplectic automorphism group $\Aut_{\mathrm{s}}(K3)$ acts on
$D^b(\mathrm{Coh}(K3))$ preserving $\Omega_{K3}$; together with
$N$-torsion translation on $E$ (Mukai 1988, Nikulin 1980), five
conjugacy classes of finite symplectic automorphisms
$g_N\in\Aut_{\mathrm{s}}(K3)$ of orders $N\in\{1,2,3,4,6\}$ survive as
Gritsenko--Nikulin frame-shape representatives. Five attack-heal cycles
globalise G4's local construction to an $\Aut_{\mathrm{s}}(K3)\times
E[N]$-equivariant factorisation algebra, and identify the
$g_N$-twisted sector.

\section{Cycle 1: $\C^3$-charts and global gluing}

\begin{construction}[Chart cover]\label{con:chart-cover}
Fix an open cover $\{U_\alpha\}$ of $X = K3\times E$ by polydiscs
$U_\alpha\cong \Delta^3\subset\C^3$ adapted to the product structure:
each $U_\alpha = V_\alpha\times W_\alpha$ with $V_\alpha\subset K3$ a
Calabi--Yau polydisc (trivialising $\Omega_{K3}$) and $W_\alpha\subset
E$ an arc trivialising $\Omega_E$. On each $U_\alpha$, Wave-13 G4
Cycle~2 produces a local factorisation algebra
$\Obs^{\mathrm{hCS}}_\alpha := \Obs^{\mathrm{hCS}}(U_\alpha)\simeq
\CE^*_{\dbar,\chir}(\cE|_{U_\alpha},\cO_{U_\alpha})$.
\end{construction}

\begin{proposition}[Global gluing]\label{prop:global-gluing}
The local data $\{\Obs^{\mathrm{hCS}}_\alpha\}$ glue to a factorisation
algebra $\cF^{\chir}_{\mathrm{BKM}}(X)$ on $X$. For overlaps
$U_\alpha\cap U_\beta$ the Bochner--Martinelli propagator $P_\alpha$
transforms under the biholomorphic transition $\varphi_{\alpha\beta}$
as $\varphi_{\alpha\beta}^*P_\beta = P_\alpha + \dbar\eta_{\alpha\beta}$
for a smooth $\eta_{\alpha\beta}\in\Omega^{0,0}(U_\alpha\cap U_\beta)$;
the $\eta_{\alpha\beta}$ assemble into a \v{C}ech $1$-cocycle
trivialising the obstruction to a global BV Laplacian.
\end{proposition}

\begin{proof}[Sketch]
Costello--Gwilliam Vol~II Theorem~5.5.4 (holomorphic locally constant
descent) gives a Weiss cosheaf on the polydisc topos; Weiss descent to
the \v{C}ech cover of $X$ is Costello--Gwilliam Vol~I Theorem~6.5.4
applied to the sheafification functor. The $\dbar$-exactness of
$P_\alpha - P_\beta$ on the overlap is the standard transition
behaviour of Bochner--Martinelli under biholomorphism, Henkin--Leiterer
1984, Proposition~2.3.1.
\end{proof}

\section{Cycle 2: equivariant structure in the Kapranov--Vasserot sense}

\begin{definition}[$G$-equivariant factorisation algebra]\label{def:equivariant-fact}
Let $G$ act holomorphically on $X$ preserving $\Omega_X$. A
\emph{$G$-equivariant factorisation algebra} on $X$ is a factorisation
algebra $\cF$ together with, for each $g\in G$, a quasi-isomorphism
$\rho_g\colon g^*\cF\xrightarrow{\sim}\cF$ intertwining the
factorisation structure maps and satisfying the cocycle
$\rho_{gh}=\rho_g\cdot g^*\rho_h$, in the sense of Kapranov--Vasserot
2018 (arXiv:1901.07641) \S 6 for holomorphic factorisation on complex
surfaces and threefolds.
\end{definition}

\begin{construction}[$\Aut_{\mathrm{s}}(K3)\times E[N]$-equivariant
refinement]\label{con:equivariant}
The product group $H := \Aut_{\mathrm{s}}(K3)\times E$ acts
holomorphically on $X = K3\times E$ preserving $\Omega_X =
\Omega_{K3}\wedge\Omega_E$. The chart cover
$\{U_\alpha\}$ of Construction~\ref{con:chart-cover} may be chosen
$H$-equivariantly by averaging over $g\in \Aut_{\mathrm{s}}(K3)$ and
translating by $t\in E$; the local $\Obs^{\mathrm{hCS}}_\alpha$ carry
$H$-action through the pullback $(g,t)^*$ on
$\cE = \Omega^{0,\bullet}(X,\gBPS)$. The gluing cocycle
$\eta_{\alpha\beta}$ of Proposition~\ref{prop:global-gluing} is
$H$-invariant because $P_\alpha$ transforms trivially under
biholomorphisms preserving $\Omega_X$; hence
$\cF^{\chir}_{\mathrm{BKM}}$ is $H$-equivariant.
\end{construction}

\section{Cycle 3: $g_N$-twisted sector and the CHL orbifold}

Nikulin 1980 classifies finite symplectic automorphisms $g_N\in
\Aut_{\mathrm{s}}(K3)$: five conjugacy classes survive in cyclic
orders $N\in\{1,2,3,4,6\}$, each embedding in Mathieu $M_{23}$.
Pairing $g_N$ with translation by an $N$-torsion point
$t_N\in E[N]\cong(\Z/N\Z)^2$ yields a fixed-point-free $\Z/N\Z$-action
on $X$; the quotient $X_N := X/(\Z/N\Z)$ is a
Chaudhuri--Hockney--Lykken (CHL) CY$_3$ orbifold (Chaudhuri--Hockney--Lykken
1995 \emph{Phys.\ Rev.\ Lett.}~75).

\begin{definition}[$g_N$-twisted chiral CE]\label{def:twisted-CE}
For $N\in\{1,2,3,4,6\}$, the \emph{$g_N$-twisted chiral CE complex} is
\[
\CE^*_{\dbar,\chir,g_N}(\cE_X,\cO_X) :=
\bigl(\CE^*_{\dbar,\chir}(\cE_X,\cO_X)\bigr)^{\Z/N\Z}
\]
where the $\Z/N\Z$-action is generated by $(g_N,t_N)^*$ on
$\cE_X = \Omega^{0,\bullet}(X,\gBPS)$. Untwisted: $N=1$.
\end{definition}

\begin{proposition}[CHL identification]\label{prop:chl}
There is a quasi-isomorphism of factorisation algebras on $X_N$:
\[
\CE^*_{\dbar,\chir,g_N}(\cE_X,\cO_X)
 \;\simeq\;
\cF^{\chir}_{\mathrm{BKM}}(X_N),
\]
the right side being the chiral BKM factorisation algebra on the CHL
quotient $X_N = (K3\times E)/\langle g_N\times t_N\rangle$.
\end{proposition}

\begin{proof}[Sketch]
The $\Z/N\Z$-action on $X$ is free (Nikulin); $X_N$ is smooth CY$_3$.
$\dbar$- and chiral-CE differentials commute with $\Z/N\Z$-pullback;
taking invariants commutes with the Weiss cosheaf limit by finiteness
of the group. Kapranov--Vasserot 2018 Proposition~6.3.1 (equivariant
factorisation homology equals factorisation homology on the quotient,
for free finite group actions) identifies the invariant complex with
the factorisation algebra on $X_N$.
\end{proof}

\section{Cycle 4: twined elliptic genus and BKM root multiplicities}

Cheng--Duncan--Harvey 2014 (arXiv:1204.2779, \S 3.2) introduce the
twined elliptic genus
\[
\phi_{0,1}^{(g_N)}(\tau,z) = \mathrm{Tr}_{H_{RR}(K3)}\!\bigl(g_N\cdot
(-1)^{F_L+F_R} y^{J_L} q^{L_0-c/24}\bar q^{\bar L_0-\bar c/24}\bigr)
\in J_{0,1}(\Gamma_0(N)),
\]
a weak Jacobi form of weight $0$, index $1$, with level structure
$\Gamma_0(N)$. The Gritsenko multiplicative lift sends
$\phi_{0,1}^{(g_N)}$ to a Siegel paramodular form
$\Delta_5^{(N)}:= \mathrm{Lift}(\phi_{0,1}^{(g_N)})$ of weight
$\mathrm{wt}(\Delta_5^{(N)}) = c_N(0)/2$, with
\[
(c_N(0))_{N\in\{1,2,3,4,6\}} = (10,4,2,2,2),
\]
Gritsenko--Nikulin 1995 (the rank-$3$ sequel, \emph{Amer.\ J.\ Math.}~\textbf{119}) Theorem~2.1.

\begin{conjecture}[$N$-twisted chiral BKM]\label{conj:n-twisted-bkm}
For $N\in\{1,2,3,4,6\}$:
\[
\kappaBKM(\cF^{\chir}_{\mathrm{BKM}}(X_N))
 \;=\; \frac{c_N(0)}{2}
 \;=\; \mathrm{wt}(\Delta_5^{(N)}),
\]
and the root multiplicities of the $g_N$-twisted sector are the Fourier
coefficients of $\phi_{0,1}^{(g_N)}$ via the Borcherds product expansion
of $\Delta_5^{(N)}$.
\end{conjecture}

\textbf{Status.} ClaimStatusConjectured. The $N=1$ identification
$\kappaBKM(\Phi_1) = 5$ is the Wave-13 baseline
(\texttt{cy\_d\_kappa\_stratification.tex}, Theorem~\texttt{thm:borcherds-weight-kappa-BKM-universal}).
The $N\geq 2$ cases depend on (a) Oberdieck--Pixton twisted DT at $N\geq 2$
(open beyond $N=1$; see \texttt{audit\_red6\_k3xe.md} R6-3), (b)
existence of the BKM $\gDelta^{(N)}$ for $N\geq 2$ (Gritsenko--Nikulin
1998 construct the Lie superalgebra; the chiral envelope is not
constructed). Mathematical content: the $g_N$-twined frame shape
matches the twined bar-Euler of M_{24} (CDH 2014 \S 4), fixing the
numerator of $\Delta_5^{(N)}$.

\section{Cycle 5: scope of $\gDelta^{(N)}$ from equivariant $\cF^{\chir}_{\mathrm{BKM}}$}

\begin{proposition}[Scope statement]\label{prop:scope-n}
The equivariant factorisation algebra
$\cF^{\chir}_{\mathrm{BKM}}(X_N)$ is constructed unconditionally for
$N\in\{1,2,3,4,6\}$ via Proposition~\ref{prop:chl} and
Construction~\ref{con:equivariant}. Its identification with
$U^{\chir}(\gDelta^{(N)})$ (a putative chiral envelope of the BKM
$\gDelta^{(N)}$) is conjectural and factors through four open
ingredients: (i) the $d=3$ functor $\Phi$ (AP-CY6; extending
\texttt{chapters/theory/cy\_to\_chiral.tex} to CY$_3$);
(ii) Davison--Meinhardt BPS integrality on $X_N$
(\texttt{k3e\_cy3\_programme.tex}, Proposition~\texttt{prop:k3e-cy3-motivic-dt-js-integrality-wc});
(iii) the equivariant version of the Oberdieck--Pixton identification
$Z^{DT}_{X_N} \propto 1/\Delta_5^{(N),2}$ (open); (iv)
morphism-preservation of $\Phi$ on $\Aut_{\mathrm{s}}$-equivariant data
(Pattern 273).
\end{proposition}

\begin{proof}
Construction: immediate from Proposition~\ref{prop:chl}.
Identification: the chain of implications at (i)--(iv) is the scope
column of \texttt{k3e\_cy3\_programme.tex} Conjecture~\texttt{conj:eight-qvcg}.
Current status: for $N=1$ all four ingredients are theorems or
established (Oberdieck--Pixton; Gritsenko $\Delta_5$); for $N\in\{2,3,4,6\}$
ingredient~(iii) remains open.
\end{proof}

\begin{remark}[Subscripted-$\kappa$ discipline across the twisted tower]
\label{rem:kappa-n-tower}
Across $N\in\{1,2,3,4,6\}$: $\kappacat(X_N) = 0$ (K\"unneth on the free
quotient gives $\chi(\cO_{X_N}) = \chi(\cO_X) = 0$); $\kappach(X_N)=0$
matching the Hodge-supertrace identity of the $E_n$-hierarchy (Part
\ref{part:en-hierarchy} of the main text); $\kappafib(K3/g_N)
= 20/N$ at generic level (Nikulin 1980); $\kappaBKM(\Phi_N) =
c_N(0)/2\in\{5,2,1,1,1\}$. The four invariants remain distinct: no
$N$-index formula of the shape $\kappaBKM =\kappach + \kappafib$ holds
(\texttt{CLAUDE.md} key facts, $N=1$: $5 \neq 0+0$; $N=2$: $2\neq 0 + \kappafib(K3/g_2)=10$).
\end{remark}
