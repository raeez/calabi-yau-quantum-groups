\begin{abstract}
For the five CM-compatible CHL orders
$N\in\{1,2,3,4,6\}$ on $K3\times E$, the Borcherds denominator
$\PhiN$ has modular weight
\[
  \kBKM(\PhiN)=\frac{c_N(0)}{2}.
\]
The primitive CHL constants are
\[
  (c_1(0),c_2(0),c_3(0),c_4(0),c_6(0))
  =(10,8,6,4,2),
\]
and hence
\[
  (\kBKM(\Phi_1),\kBKM(\Phi_2),\kBKM(\Phi_3),
    \kBKM(\Phi_4),\kBKM(\Phi_6))
  =(5,4,3,2,1).
\]
This is the Borcherds-weight invariant of the denominator. It is
separate from $\kch$, $\kcat$, and $\kfib$. The Gritsenko-Cl\'ery
eight-row catalogue is a different family and is named separately.
\end{abstract}

\section{Objects and normalisation}
\label{sec:objects-normalisation}

Let $S$ be a K3 surface with a symplectic automorphism $g_N$ of order
$N\in\{1,2,3,4,6\}$, and let $E$ be an elliptic curve with translation
by an $N$-torsion point. The diagonal quotient
\[
  X_N=(S\times E)/\langle g_N\times\sigma_N\rangle
\]
is a smooth compact CY$_3$ in the CHL family. At $N=1$ this is
$K3\times E$.

The BKM denominator attached to the CHL input is denoted $\Phi_N$.
Its modular characteristic is the automorphic weight
\[
  \kBKM(\Phi_N):=\operatorname{wt}(\Phi_N).
\]
This invariant is not
\[
  \kcat(X_N)=\chi(\mathcal O_{X_N}),\qquad
  \kch(\cA_{X_N}),\qquad
  \kfib(S).
\]
For the compact CY$_3$ total space, $\chi(\mathcal O_{X_N})=0$.
The Borcherds characteristic is read from the input Jacobi or
vector-valued modular form, not from a Hodge supertrace on $X_N$.

\begin{definition}[CHL Borcherds-input constant]
\label{def:chl-borcherds-input}
For the primitive CHL Borcherds input attached to $g_N$, let $c_N(0)$ be the
constant term of the zero component at the cusp used in the singular
theta lift. In the Gritsenko-Nikulin CHL normalisation,
\[
  (c_1(0),c_2(0),c_3(0),c_4(0),c_6(0))
  =(10,8,6,4,2).
\]
\end{definition}

The $N=1$ entry is the Gritsenko form $\DeltaFive$ of weight $5$.
For $N=2,3,4,6$ the forms are the CHL paramodular cusp forms of
weights $4,3,2,1$. The scalar-square convention doubles these weights;
it is not the $\kBKM$ convention used here.

At level one the monic normalisation of the primitive denominator and
the two square conventions are
\[
  \Dfive(Z)=64^{-1}\DeltaFive(Z),\qquad
  \Phiun(Z)=\DeltaFive(Z)^2,\qquad
  \PhiOP(Z)=\Dfive(Z)^2=4096^{-1}\DeltaFive(Z)^2.
\]
The monic product $\Dfive$ and the unnormalised product $\DeltaFive$
have the same Borcherds weight. The Oberdieck-Pixton and
Oberdieck-Pandharipande reduced-DT scalar convention uses the square
$\PhiOP$:
\[
  \DTred(K3\times E)=-(\PhiOP)^{-1}
  =-\Dfive^{-2}=-4096\,\DeltaFive^{-2}.
\]
This scalar square is not the primitive BKM denominator and does not
alter $\kBKM(\DeltaFive)=5$.

\section{Borcherds weight theorem}
\label{sec:borcherds-weight-theorem}

\begin{theorem}[Universal CHL Borcherds weight identity]
\label{thm:universal-chl-borcherds-weight}
For $N\in\{1,2,3,4,6\}$,
\[
  \kBKM(\Phi_N)=\operatorname{wt}(\Phi_N)=\frac{c_N(0)}{2},
\]
with
\[
  (c_N(0))_{N=(1,2,3,4,6)}=(10,8,6,4,2)
\]
and
\[
  (\kBKM(\Phi_N))_{N=(1,2,3,4,6)}=(5,4,3,2,1).
\]
\end{theorem}

\begin{proof}
Borcherds' singular theta correspondence states that the automorphic
product $\Psi(f)$ attached to a vector-valued weakly holomorphic
modular form $f$ has weight
\[
  \operatorname{wt}(\Psi(f))=\frac{c_0(0)}{2},
\]
where $c_0(0)$ is the constant term of the zero component of $f$ at the
chosen cusp. This is Borcherds 1998, Theorem 13.3.

The CHL products $\Phi_N$ are the Gritsenko-Nikulin Borcherds products
attached to the five CM-compatible symplectic frame shapes. Their
constant terms are the values in Definition~\ref{def:chl-borcherds-input}.
Substituting them into Borcherds' formula gives
\[
  \frac{1}{2}(10,8,6,4,2)=(5,4,3,2,1).
\]
These are exactly the weights of the CHL denominator forms
\[
  \Delta_5,\quad \Delta_4^{(2)},\quad
  \Delta_3^{(3)},\quad \Delta_2^{(4)},\quad
  \Delta_1^{(6)} .
\]
\end{proof}

\begin{remark}[Additive decompositions fail]
\label{rem:additive-decomposition-fails}
The formula is not
\[
  \kBKM=\kch+\chi(\mathcal O_{\mathrm{fiber}})
\]
and it is not a fibre correction to $\kcat$. At $N=1$ the Borcherds
side is $5$, while
\[
  \kch(\cA_{K3\times E})+\chi(\mathcal O_E)=0+0=0.
\]
At $N=2$ the Borcherds side is $4$, while the same additive reading
gives $1$ in the quotient convention. The invariant $\kBKM$ is the
weight of an automorphic denominator.
\end{remark}

\section{The five CHL entries}
\label{sec:five-chl-entries}

\[
\begin{array}{c|c|c|c|c}
N & \text{frame shape} & c_N(0) & \kBKM(\Phi_N) &
\text{denominator} \\
\hline
1 & 1^{24} & 10 & 5 & \Delta_5 \\
2 & 1^8 2^8 & 8 & 4 & \Delta_4^{(2)} \\
3 & 1^6 3^6 & 6 & 3 & \Delta_3^{(3)} \\
4 & 1^4 2^2 4^4 & 4 & 2 & \Delta_2^{(4)} \\
6 & 1^2 2^2 3^2 6^2 & 2 & 1 & \Delta_1^{(6)}
\end{array}
\]

The table is a primitive CHL table. It is not identified componentwise
with the Gritsenko-Cl\'ery eight-row catalogue of reflective
diagonal-divisor paramodular products. That catalogue is indexed by
pairs $(t,N)$ and
has twice-weight tuple
\[
  (10,4,6,2,4,1,3,2),
\]
namely weights
\[
  (5,2,3,1,2,\tfrac12,\tfrac32,1).
\]
The common entry is the level-one $\Delta_5$ row. The other rows belong
to different automorphic groups, multiplier systems, Borcherds inputs,
and denominator normalisations. Any comparison at
$N\in\{2,3,4,6\}$ requires a stated row map.

\section{Fourier and fixed-point checks}
\label{sec:fourier-fixed-point-checks}

The $N=1$ input is the weak Jacobi form
\[
  \phi_{0,1}(\tau,z)
  =(\zeta^{-1}+10+\zeta)+O(q),
\]
so $c_1(0)=10$ and $\operatorname{wt}(\Delta_5)=5$.

For $N=2,3,4,6$ the CHL frame shapes give constant terms
\[
  c_2(0)=8,\qquad c_3(0)=6,\qquad
  c_4(0)=4,\qquad c_6(0)=2.
\]
The same constants are read from the Gritsenko-Nikulin paramodular
tower by the denominator weights
\[
  4,\qquad 3,\qquad 2,\qquad 1.
\]
Mukai's fixed-point classification supplies the geometric K3 side of
the same order set: the symplectic classes of orders $2,3,4,6$ have
fixed loci and invariant lattices compatible with the frame shapes
above. The fixed-point list is a geometric check on the CHL inputs,
not a replacement for the Borcherds constant term.

\begin{proposition}[The $N=6$ separation]
\label{prop:n6-separation}
At $N=6$, the CHL and umbral Niemeier anchor is the global
rank-$24$ lattice with root system $6D_4$ and umbral group
$3.\mathrm{Sym}_6$. The cyclic quotient surface
$K3/\langle g_6\rangle$ has local Du Val type
\[
  2A_5+2A_2+2A_1 .
\]
These labels live in different categories.
\end{proposition}

\begin{proof}
The Niemeier label is a positive-definite even unimodular rank-$24$
lattice label. It controls the umbral mock modular anchor and the
global character match. The quotient singularity type is computed from
the stabiliser orbits of the order-$6$ symplectic K3 automorphism:
two points with stabiliser order $6$, two orbits with stabiliser order
$3$, and two orbits with stabiliser order $2$. Hence the local
singularities are $2A_5+2A_2+2A_1$. This local ADE decomposition is
not the Niemeier root system.
\end{proof}

\section{Extended rows}
\label{sec:extended-rows}

The CHL ladder is the five-order family
$N\in\{1,2,3,4,6\}$ for diagonal quotients of $K3\times E$. Two
extensions must be kept separate.

\begin{enumerate}
\item The Nikulin symplectic orders $N\in\{5,7,8\}$ have Borcherds
weights $2,1,1$ in the extended K3 automorphism list, with constants
$4,2,2$. They do not arise from the same smooth diagonal
$K3\times E$ CHL quotient because the elliptic factor has no
origin-fixing automorphism of those orders.
\item The Gritsenko-Cl\'ery reflective catalogue is an eight-row
classification indexed by paramodular data $(t,N)$, not by CHL cyclic
orders. Its weights are
\[
  (5,2,3,1,2,\tfrac12,\tfrac32,1).
\]
\end{enumerate}

Both extensions obey the same Borcherds weight theorem in their own
normalisations. Neither changes the CHL constants
$(10,8,6,4,2)$.

\section{Chiral output scope}
\label{sec:chiral-output-scope}

For the compact CY$_3$ family $X_N$, the CY-to-chiral specialization is
an $E_1$-chiral algebra on the elliptic curve. The six-dimensional
holomorphic Chern-Simons local-observable model carries an $E_3$
structure before compactification. The specialized output for
$d\geq3$ is $E_1$. An $E_2$ structure belongs on a centre or
Drinfeld double after the hypotheses constructing that centre or double
are in place.

This scope distinction is independent of the Borcherds weight theorem.
The equality
\[
  \kBKM(\Phi_N)=c_N(0)/2
\]
is a theorem about automorphic denominators; it is not a construction
of the full chiral algebra.

\section{Convention separations}
\label{sec:convention-separations}

\begin{enumerate}
\item Bruinier's coefficient $c_3^{\mathrm{Br}}=-8$ is a
Heegner-discriminant coefficient in a weight-$1/2$ vector-valued input.
It is not the CHL constant $c_3(0)=6$.
\item The primitive scalar-square convention uses weights
$(10,8,6,4,2)$. The BKM denominator convention here uses half those weights:
$(5,4,3,2,1)$.
\item At $N=1$, $\Dfive=64^{-1}\DeltaFive$ and
$\PhiOP=\Dfive^2=4096^{-1}\DeltaFive^2$. The reduced scalar
$-(\PhiOP)^{-1}=-4096\,\DeltaFive^{-2}$ is an OP square convention,
not the primitive Borcherds denominator.
\item The total-space value $\kcat(K3\times E)=0$ follows from
K\"unneth multiplicativity and $\chi(\mathcal O_E)=0$. The fibre value
$\kfib(K3)=2$ is a different invariant.
\item The Fake-Monster denominator $\Phi_{12}^{\mathrm{FM}}$ has
weight $12$ from $\eta^{-24}$ and belongs to the orthogonal
rank-$26$ sibling lane. It is not one of the five CHL entries above.
\end{enumerate}

\section{Boundary questions}
\label{sec:boundary-questions}

\begin{enumerate}
\item Direct CHL Fourier expansions for each $\Phi_N$ in the same
normalisation as the Gritsenko-Nikulin denominator tower.
\item Relation of the Costello-Gaiotto hCS partition function to the CHL
Borcherds products without changing the $E_1$ output scope of the
CY-to-chiral specialization.
\item Construction of the higher Yetter-Drinfeld tower on
$\fg_{\Phi_N}$ for $N\geq2$.
\item Transitivity or non-transitivity of the Grothendieck-Teichm\"uller
action on the five CHL BKM algebras.
\end{enumerate}
