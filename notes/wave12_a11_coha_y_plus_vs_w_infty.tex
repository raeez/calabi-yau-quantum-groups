\begin{abstract}
The critical cohomological Hall algebra of the Jordan three-loop quiver
with potential $W=\operatorname{tr}(X[Y,Z])$ is the positive half
$\Yp(\gglone)$ of the affine Yangian. Schiffmann-Vasserot prove this as
an equality of associative Hall algebras. The vertex algebra
$\Wilpha[\lambda]$ enters after adjoining the Cartan and negative
halves, completing the Drinfeld double, choosing an evaluation or
corner-VOA representation, and applying state-field correspondence. The
native CY$_3$ output in the two-stage CY-to-chiral construction is
$\Eone$; braided $\Etwo$ structure belongs to the centre, double, or
representation layer.
\end{abstract}

\section{Objects}

\begin{definition}[The five structures]
\label{def:five-structures}
On the $\mathbb{C}^3$ Hall side five algebraic structures occur.
\[
\begin{array}{l|l|l}
\text{object} & \text{category} & \text{generators or fields} \\\hline
\CoHA(\mathbb{C}^3) &
\Eone\text{-associative Hall algebra} &
\text{critical fixed-point classes} \\
\Yp(\gglone) &
\Eone\text{-associative shuffle algebra} &
E_n,\ n\geq 0 \\
\YFull(\gglone)=\Ym\bowtie\Yz\bowtie\Yp &
\text{completed Hopf algebra} &
F_n,\ \psi_n^\pm,\ E_n \\
\Ecal_{q_1,q_2,q_3}(\gglone) &
\text{quantum toroidal algebra} &
\text{vertical and horizontal currents} \\
\Wilpha[\lambda] &
\text{vertex algebra} &
J^{(s)}(z)=\sum_{n\in\mathbb{Z}}J^{(s)}_n z^{-n-s}
\end{array}
\]
\end{definition}

The first two rows are positive Hall halves. The third row is obtained
from a Hall-Serre pairing and a completion. The fourth row is the
$K$-theoretic, trigonometric lift. The last row is a chiral object with
state-field correspondence and locality. Equalities between different
rows require a named construction.

\section{The Positive Half}

\subsection{The Calabi-Yau torus}

Let $T=(\mathbb{C}^{\times})^3$ act on
$\mathbb{C}^3=\operatorname{Spec}k[X,Y,Z]$ by
$(t_1,t_2,t_3)\cdot(x,y,z)=(t_1x,t_2y,t_3z)$. Let
$\epsilon_i=c_1(t_i)$ be the additive equivariant weights. The
holomorphic volume form $\Omega_{\mathbb{C}^3}=dX\wedge dY\wedge dZ$
has character $t_1t_2t_3$, equivalently additive weight
$\epsilon_1+\epsilon_2+\epsilon_3$. The Calabi-Yau subtorus is
\[
T^{\CY}:=\{(t_1,t_2,t_3)\in T:\ t_1t_2t_3=1\},
\]
and all equivariant Hall statements below are over
\[
\mathbb{F}_{\CY}
=\mathbb{Q}(\epsilon_1,\epsilon_2,\epsilon_3)/
(\epsilon_1+\epsilon_2+\epsilon_3).
\]
The three equivariant parameters are coordinates on the ambient torus;
the Calabi-Yau condition restricts them to a two-parameter slice.

\subsection{The three-loop critical chart}

Let $Q_{\mathbb{C}^3}$ be the one-vertex quiver with loops $X,Y,Z$ and
cyclic potential
\[
W=\operatorname{tr}(X[Y,Z])=\operatorname{tr}(XYZ-XZY).
\]
The cyclic derivatives are
\[
\partial_XW=[Y,Z],\qquad
\partial_YW=[Z,X],\qquad
\partial_ZW=[X,Y],
\]
hence
\[
J(Q_{\mathbb{C}^3},W)
=k\langle X,Y,Z\rangle/([X,Y],[Y,Z],[Z,X])
\cong k[X,Y,Z]
=\mathcal{O}_{\mathbb{C}^3}.
\]
The word ``three-loop'' records the quiver. No Jordan triple system is
part of this Jacobi-algebra computation.

\subsection{The shuffle presentation}

\begin{definition}[Feigin-Odesskii shuffle product]
\label{def:fo-shuffle}
Set $\Lambda_n=\mathbb{F}_{\CY}[z_1,\ldots,z_n]^{S_n}$ and
$\Lambda=\bigoplus_{n\geq 0}\Lambda_n$. For
$F\in\Lambda_m$ and $G\in\Lambda_n$ define
\[
(F\star G)(x_1,\ldots,x_{m+n})
=\frac{1}{m!\,n!}\sum_{\sigma\in S_{m+n}}\sigma\cdot
\left[
F(x_1,\ldots,x_m)G(x_{m+1},\ldots,x_{m+n})
\prod_{\substack{1\leq i\leq m\\ m<j\leq m+n}}
\omega(x_j,x_i)
\right],
\]
where
\[
\omega(z,w)
=\frac{\prod_{i=1}^{3}(z-w-\epsilon_i)}{(z-w)^3}.
\]
\end{definition}

\begin{theorem}[Schiffmann-Vasserot positive-half theorem]
\label{thm:sv-positive-half}
The $T^{\CY}$-equivariant critical cohomological Hall algebra of
$(Q_{\mathbb{C}^3},W)$ is isomorphic, as an
$\mathbb{F}_{\CY}$-associative algebra, to the positive half of the
affine Yangian:
\[
\CoHA_{T^{\CY}}(\mathbb{C}^3)
\cong
\Yp(\gglone).
\]
Under the Schiffmann-Vasserot shuffle realisation
(Publ.\ Math.\ IHES 118, 2013, Theorem A), the charge-$n$
summand maps to the SV shuffle subalgebra of $\Lambda_n$, and the
critical Hall convolution corresponds to the product $\star$.
Kapranov-Vasserot and Rap\v{c}ak-Soibelman-Yang-Zhao give the parallel
factorisation and toric-Hall formulations.
\end{theorem}

\begin{proposition}[Generator dictionary]
\label{prop:generator-dictionary}
The positive half $\Yp(\gglone)$ has shuffle generators given by the
charge-one classes
\[
f_{1,k}(z)=z^k\in\Lambda_1,\qquad k\geq 0.
\]
The SV isomorphism sends the Drinfeld current generators to these
classes:
\[
E_k\longmapsto f_{1,k}=z^k .
\]
Higher charge power sums in $\Lambda_n$ are produced by shuffle
products and the kernel $\omega$. The elements $F_k$ and $\psi_k^\pm$
belong to the negative and Cartan layers of the completed double; they
are absent from the raw CoHA.
\end{proposition}

\begin{proof}
The Hall product is the pull-push product along the extension
correspondence for the critical stack. Schiffmann-Vasserot identify this
correspondence with the Feigin-Odesskii kernel in
Definition~\ref{def:fo-shuffle}. Charge one is the one-point sector, so
the functions $z^k$ give the positive Drinfeld current coefficients.
The Hall product of charge-one sectors generates the positive shuffle
subalgebra. The Serre-dual negative sector and Cartan current require a
Hopf pairing and a double, hence have no source in
$\bigoplus_n\Lambda_n$ alone.
\end{proof}

\section{Modes and Vertex Algebras}

\begin{definition}[$\Wilpha[\lambda]$]
\label{def:winfty}
The Kac-Radul and Frenkel-Kac-Radul-Wang algebra
$\Wilpha[\lambda]$ is a vertex algebra with fields
\[
J^{(s)}(z)=\sum_{n\in\mathbb{Z}}J^{(s)}_n z^{-n-s},
\qquad s\geq 1,
\]
with $J^{(1)}$ the Heisenberg field. Its structure includes a vacuum,
translation, locality, and a state-field map
\[
Y:V\to\End(V)[[z,z^{-1}]].
\]
\end{definition}

\begin{lemma}[The positive half has no state-field structure]
\label{lem:positive-half-no-voa}
$\Yp(\gglone)$ is the creation half of the affine-Yangian mode algebra.
It has the generators $E_k$ for $k\geq 0$, an associative Hall product,
and no intrinsic annihilation modes, Cartan field, vacuum axiom,
Borcherds identity, or locality operator. The vertex algebra
$\Wilpha[\lambda]$ is therefore a different object.
\end{lemma}

\begin{proof}
A vertex algebra field has modes indexed by all of $\mathbb{Z}$, and
locality is expressed by the Borcherds identity for these two-sided
Laurent modes. The Hall algebra $\Yp$ contains the positive
Drinfeld current coefficients. The opposite current $F(u)$ and the
Cartan current $\psi^\pm(u)$ are produced by adjoining the
negative half and the Cartan part. Thus the data defining a state-field
map enters at the doubled and represented layer.
\end{proof}

\begin{proposition}[Completed double and evaluation]
\label{prop:completed-double-evaluation}
After a non-degenerate Hall-Serre pairing and the required completion
are fixed, the Drinfeld double has triangular decomposition
\[
D_{\mathrm{Drin}}(\Yp)
=\Ym\widehat{\bowtie}\Yz\widehat{\bowtie}\Yp
=\YFull(\gglone).
\]
Completed evaluation representations of $\YFull(\gglone)$ act on
$\Wilpha[\lambda]$ vacuum modules:
\[
\Yp(\gglone)
\hookrightarrow
\YFull(\gglone)
\xrightarrow{\ \mathrm{ev}_{\lambda}\ }
\End\bigl(\Wilpha[\lambda]\text{-}\mathrm{vac}\bigr).
\]
The image is an algebra of operators on the chosen vacuum module. The
vertex algebra is carried by that module with its state-field
correspondence; the positive Hall half supplies creation operators.
\end{proposition}

The standard mode comparison in Arbesfeld-Schiffmann-Vasserot and
Schiffmann-Vasserot sends $\Yp$ to creation operators, $\Ym$ to
annihilation operators, and $\Yz$ to the Heisenberg-Cartan generating
series. The equality lives at the completed mode or
endomorphism-algebra level after central charge and representation data
are fixed.

\section{Native Operadic Level}

\begin{theorem}[Two-stage CY-to-chiral level]
\label{thm:native-level}
For a specialisation datum $(\Sigma_{d-1},C)$ the constructed
CY-to-chiral assignment has the form
\[
\Phi_d^{(\Sigma_{d-1},C)}
=\operatorname{Sp}^{\mathrm{ch}}_{\Sigma_{d-1},C}\circ\Phi_d^{FA},
\]
with residual operadic level
\[
n_{\mathrm{native}}(d)=
\begin{cases}
\infty,& d=1,\\
2,& d=2,\\
1,& d\geq 3.
\end{cases}
\]
Thus a specialised CY$_3$ output is natively $\Eone$-chiral. The braided
$\Etwo$ structure is carried by a centre, double, or representation
category:
\[
\mathcal{Z}\bigl(\Rep^{\Eone}(A)\bigr)
\simeq
\Rep^{\Etwo}\bigl(D(A)\bigr),
\]
on the locus where the double, completion, pairing, and half-braiding
hypotheses are supplied.
\end{theorem}

\begin{proof}
This is the PTVV and Dunn-Lurie shift calculation used in
Chapter~\texttt{chapters/theory/cy\_to\_chiral.tex}. Stage one produces
an $E_d$ holomorphic factorisation algebra. Stage two integrates over a
complex $(d-1)$-dimensional specialisation submanifold and restricts to the
reference curve. The operadic level drops by $d-1$, hence $E_d$ becomes
$E_1$ for every $d\geq 3$. The nonsymmetric braiding required by
Yangian representation theory is the half-braiding in the Drinfeld
centre; the raw Hall product remains ordered and associative.
\end{proof}

\section{The Bridge to $\Wilpha[\lambda]$}

\begin{theorem}[Typed bridge]
\label{thm:typed-bridge}
The passage from $\CoHA(\mathbb{C}^3)$ to $\Wilpha[\lambda]$ is the
following sequence of typed constructions:
\[
\CoHA(\mathbb{C}^3)
\xrightarrow[\cong]{\mathrm{SV}}
\Yp(\gglone)
\xhookrightarrow{\ D_{\mathrm{Drin}}\ }
\YFull(\gglone)
\xleftarrow[\mathrm{rat.\ deg.}]{\cong}
\Ecal^{\mathrm{rat}}(\gglone)
\xrightarrow{\ T_{\mathrm{Miki}}\ }
\Ecal^{\mathrm{hor}}(\gglone)
\xrightarrow{\ \mathrm{ev/corner}\ }
\End\bigl(\Wilpha[\lambda]\text{-}\mathrm{vac}\bigr).
\]
The steps have different mathematical types.
\begin{enumerate}
\item Schiffmann-Vasserot gives an equality of positive associative Hall
algebras.
\item The Drinfeld double adjoins $\Ym$ and $\Yz$ through a Hopf pairing.
\item Rational degeneration identifies the completed affine Yangian
with the rational limit of the Ding-Iohara-Miki quantum toroidal
algebra.
\item Miki's automorphism exchanges vertical and horizontal toroidal
directions.
\item A vacuum or corner-VOA representation gives an action on the
$\Wilpha[\lambda]$ vacuum module; state-field correspondence is part of
the vertex algebra carried by that module.
\end{enumerate}
The composite is an action chain between distinct layers. It is not a
direct identification of the Hall algebra with the vertex algebra.
\end{theorem}

\begin{remark}[The Heisenberg fibre]
\label{rem:heisenberg-fibre}
At the special point $\epsilon_3=0$, $\epsilon_1=-\epsilon_2$, the
shuffle kernel becomes
\[
\omega(z,w)
=\frac{(z-w-\epsilon_1)(z-w+\epsilon_1)(z-w)}{(z-w)^3}
=\frac{(z-w)^2-\epsilon_1^2}{(z-w)^2}.
\]
The corresponding vacuum evaluation of the completed double degenerates
to the level-one Heisenberg vertex algebra $H_1$, the
$\Wilpha[\lambda]$ fibre usually denoted $\Wilpha[0]$ in the
Gaiotto-Rap\v{c}ak parametrisation. This is a point on the
Calabi-Yau slice, distinct from the generic $\mathbb{C}^3$ Hall
algebra.
\end{remark}

\section{Holomorphic Chern-Simons Reading}

\begin{proposition}[Bulk, line, and Hall sector]
\label{prop:hcs-reading}
Let $X=\mathbb{C}^3$ with holomorphic volume form
$\Omega=dz_1\wedge dz_2\wedge dz_3$. Six-dimensional holomorphic
Chern-Simons theory has action
\[
S_{\hCS}(A)
=\int_X\Omega\wedge
\operatorname{tr}\left(A\bar{\partial}A+\frac{2}{3}A\wedge A\wedge A\right).
\]
Costello-Gwilliam locality gives a holomorphic factorisation algebra of
observables on $X$, hence a holomorphic $E_3$ bulk object. Restriction
to a chosen complex spectral line $\mathbb{C}_z\subset\mathbb{C}^3$
with the relevant defect data gives a factorisation algebra on the
line, equivalently a vertex algebra. In the Costello-Paquette and
Gaiotto-Rap\v{c}ak corner realisations this vertex algebra is
$\Wilpha[\lambda]$.

The Hall identification is a separate comparison: after the
hCS-to-critical-CoHA map is fixed on the toric chart, the positive BPS
charge sector is $\CoHA(\mathbb{C}^3)=\Yp(\gglone)$. The boundary
vertex algebra is obtained from the doubled and evaluated mode algebra,
as in Theorem~\ref{thm:typed-bridge}.
\end{proposition}

\section{Conifold and $K3\times E$}

\subsection{Conifold}

For the resolved conifold
\[
\mathbb{Y}_{\mathrm{con}}
=\operatorname{Tot}\bigl(\mathcal{O}_{\mathbb{P}^1}(-1)\oplus
\mathcal{O}_{\mathbb{P}^1}(-1)\bigr),
\]
the Klebanov-Witten quiver has two vertices, arrows
$a_1,a_2:\circ\to\bullet$ and $b_1,b_2:\bullet\to\circ$, and potential
\[
W_{\mathrm{con}}
=\operatorname{tr}(a_1b_1a_2b_2-a_1b_2a_2b_1).
\]
Davison's critical CoHA, together with the
Rap\v{c}ak-Soibelman-Yang-Zhao toric theorem, identifies the conifold
Hall algebra with the positive half of the corresponding affine
super-Yangian or quantum toroidal algebra. It is an ordered
$\Eone$-algebra. A conifold defect VOA requires the doubled toroidal
data and the defect-line OPE construction. Its generators and OPEs are
conifold-specific; the $\gglone$ algebra $\Wilpha[\lambda]$ is the
rank-one $\mathbb{C}^3$ corner.

\subsection{$K3\times E$}

Let $Y=K3\times E$. This compact threefold is non-toric and carries no
$T^3$ localisation package. The invariant readings are:
\[
\begin{array}{c|c|l}
\text{invariant} & \text{value} & \text{construction}\\\hline
\kappa_{\mathrm{cat}}(Y) & 0 &
\chi(\mathcal{O}_{K3})\chi(\mathcal{O}_{E})=2\cdot 0\\
\kappa_{\mathrm{ch}}(Y) & 0 &
\sum_q(-1)^qh^{0,q}(Y)=1-1+1-1\\
\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(Y) & 3 &
K3\text{-fibre Heisenberg specialisation}\\
\kappa_{\mathrm{BKM}}(\mathfrak{g}_{\Delta_5}) & 5 &
c_1(0)/2=10/2=\operatorname{wt}(\Delta_5)\\
\kappa_{\mathrm{fiber}}(K3) & 24 &
\operatorname{rk}\widetilde{H}(K3,\mathbb{Z})
\end{array}
\]
The automorphic normalisations are
\[
D_5=64^{-1}\Delta_5,\qquad
\Phi_{10}^{\mathrm{OP}}=D_5^2=4096^{-1}\Delta_5^2,\qquad
\Phi_{10}^{\mathrm{un}}=\Delta_5^2.
\]
Oberdieck-Pandharipande reduced DT theory uses the OP scalar
normalisation. The primitive Borcherds denominator algebra uses
$\Delta_5$, with
$\kappa_{\mathrm{BKM}}(\mathfrak{g}_{\Delta_5})=5$. The OP scalar
normalisation changes the reciprocal DT scalar and leaves the
Borcherds weight unchanged.

\begin{proposition}[Compact Hall-Borcherds scope]
\label{prop:k3e-scope}
Davison and Davison-Meinhardt supply PBW and integrality theorems for
critical CoHAs under their critical-chart and orientation hypotheses.
For $K3\times E$, primitive reduced DT dimensions match the
Gritsenko-Nikulin and Oberdieck-Pandharipande automorphic product
coefficients. This proves a graded-dimension comparison with the
Borcherds algebra $\mathfrak{g}_{\Delta_5}$.

A Lie-algebra isomorphism
\[
\mathfrak{g}_{\mathrm{BPS}}(K3\times E)
\cong
\mathfrak{g}_{\Delta_5}
\]
requires bracket compatibility, compact Hall pairing, coproduct,
Cartan, completion, and Borcherds-Serre quotient data. Those data form
the compact Hall-Drinfeld-Borcherds construction, separate from the
$\mathbb{C}^3$ affine-Yangian theorem.
\end{proposition}

\section{KZ and KZB}

KZ and KZB connections act on conformal blocks. Through the
Drinfeld-Kohno and Cherednik correspondences they produce monodromy,
associators, and braidings on representation categories. They furnish no
morphism
\[
\CoHA(\mathbb{C}^3)\longrightarrow\Wilpha[\lambda].
\]
In the present dictionary they control the horizontal toroidal action
and the $R$-matrix on $\Rep(\YFull(\gglone))$. The vertex algebra enters
after evaluation and state-field correspondence.

\section{Consolidated Dictionary}

\begin{theorem}[Typed equalities and constructions]
\label{thm:consolidated-dictionary}
Over
$\mathbb{F}_{\CY}=\mathbb{Q}(\epsilon_1,\epsilon_2,\epsilon_3)/
(\epsilon_1+\epsilon_2+\epsilon_3)$:
\begin{enumerate}
\item $\CoHA_{T^{\CY}}(\mathbb{C}^3)\cong\Yp(\gglone)$ as
associative $\Eone$ Hall algebras.
\item $D_{\mathrm{Drin}}(\Yp)=\Ym\widehat{\bowtie}\Yz
\widehat{\bowtie}\Yp=\YFull(\gglone)$ after the Hall pairing,
Cartan part, negative half, and completion are fixed.
\item $\YFull(\gglone)$ is the completed affine-Yangian mode algebra
acting on $\Wilpha[\lambda]$ vacuum and corner-VOA modules by
evaluation.
\item $\Wilpha[\lambda]$ is a vertex algebra, distinct from the positive
shuffle algebra $\Yp$.
\item For $d\geq 3$, the CY-to-chiral output is natively $\Eone$.
The $\Etwo$ braided structure belongs to
$\mathcal{Z}(\Rep^{\Eone}(-))$, to a Drinfeld double, or to a module
category with half-braidings.
\item BKM algebras are Borcherds denominator objects whose weights are
computed by $\kappa_{\mathrm{BKM}}(\Phi_N)=c_N(0)/2$ after the
relevant automorphic product is fixed.
\item Affine Yangians, quantum toroidal algebras, VOAs, and KZ/KZB flat
connections are distinct structures connected by the constructions
above.
\end{enumerate}
\end{theorem}

\section{Comparison Boundaries}

\begin{proposition}[Independent comparison data]
\label{prop:comparison-boundaries}
The following comparisons require data beyond the
$\mathbb{C}^3$ positive-half theorem and beyond the
$\Wilpha[\lambda]$ evaluation chain.
\begin{enumerate}
\item \emph{$K3\times E$ bracket comparison.}
Primitive reduced DT dimensions match $\Delta_5$ root multiplicities.
A bracket-preserving isomorphism between the compact BPS Lie algebra and
$\mathfrak{g}_{\Delta_5}$ additionally requires the compact Hall
pairing, coproduct, Cartan, completion, and Borcherds-Serre quotient
data.
\item \emph{Conifold defect OPEs.}
The positive conifold Hall algebra is constructed. The defect vertex
algebra requires explicit generators, OPEs, and comparison with the
horizontal quantum toroidal presentation.
\item \emph{General compact CY$_3$ critical charts.}
Davison-style PBW statements require oriented critical atlases,
orientation data, and integrality hypotheses case by case.
\item \emph{Geometric negative half beyond toric surfaces.}
For $\mathbb{C}^3$, stable envelopes and Hall pairings construct
$\Ym$. For compact non-toric geometries, the Serre-dual pairing and
completion are prerequisites for any Drinfeld double or centre-level
claim.
\end{enumerate}
\end{proposition}
