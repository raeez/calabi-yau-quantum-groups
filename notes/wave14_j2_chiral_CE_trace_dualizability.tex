Let $X$ be a Calabi--Yau threefold with holomorphic volume form
$\Omega_X$, and let $\g$ be a finite-dimensional quadratic Lie algebra.
For $U\subset X$ write
\[
 L_U=\Omega_c^{0,\bullet}(U,\g)[1],
 \qquad
 \Obs_{\hCS}^{\mathrm{cl}}(U)=\CE^\bullet_{\dbar,\chir}(L_U,\cO_U).
\]
The bracket on $L_U$ is the pointwise Lie bracket on $\g$ multiplied by
the wedge product of Dolbeault forms. The notation
$\CE^\bullet_{\dbar,\chir}$ denotes completed chiral
Chevalley--Eilenberg cochains. It contains the differential and the
factorisation product; a trace is an additional continuous chain map or
cyclic pairing. The BV pairing is
\[
 \langle\alpha,\beta\rangle_{\BV}
 =
 \int_U \Omega_X\wedge\langle\alpha,\beta\rangle_{\g}.
\]
Quantum observables mean the $\hb$-adically complete Costello--Gwilliam
renormalised complex after a gauge fixing, a solution of the quantum
master equation, and the required counterterms have been supplied.

\section{Diagonal Residue}

\begin{proposition}[Local residue pairing]\label{prop:local-residue-pairing}
On the formal polydisc $D^3\subset\C^3$, the real oriented boundary of
the Fulton--MacPherson compactification of two ordered points has fibre
$S^5$ over the diagonal. Integration over this fibre defines a local BV
residue pairing
\[
 \tau_{\mathrm{res}}:
 \Obs_{\hCS}(D^3)\widehat\otimes \Obs_{\hCS}(D^3)
 \longrightarrow \C[[\hb]]
\]
on renormalised observables whose kernels have controlled singularity
along the diagonal.
\end{proposition}

\begin{proof}
The real blow-up of the diagonal in $D^3\times D^3$ replaces the normal
space $\C^3$ by its unit sphere $S(\C^3)=S^5$. The Bochner--Martinelli
kernel represents the fundamental class for $\dbar$ on the punctured
normal space:
\[
 \dbar P=\delta_\Delta,
 \qquad
 P\in\Omega^{0,2}\bigl((D^3\times D^3)\setminus\Delta\bigr).
\]
For two observables represented by distributional kernels $K_1,K_2$,
their product has a normal restriction to the boundary after
renormalisation. The residue is
\[
 \tau_{\mathrm{res}}(K_1,K_2)
 =
 \int_{S^5}
 \bigl(K_1K_2\bigr)_{\partial\cConf_2(D^3)}\wedge P\wedge\Omega_{\C^3}.
\]
Stokes' theorem gives independence of the representative, because
$\dbar P=\delta_\Delta$ is exactly the boundary term inserted in the BV
Laplacian. Nondegeneracy holds on the finite jet quotient at the
diagonal. On the full noncompact formal disk it is a continuous pairing
after choosing the distributional completion.
\end{proof}

\begin{remark}[Residue and framed trace]\label{rem:residue-trace-scope}
The sphere $S^5$ is the real link of the complex diagonal. A framed
$E_3$ trace in the cobordism-hypothesis sense is a map out of the
factorisation homology controlled by the framed $3$-disk operad. The
residue above becomes such a trace only after the holomorphic theory is
strictified to the framed $E_3$ topological skeleton. The strictification
requires a chain-level $S^3$ framing and compatibility of the BV
orientation with the Costello--Gwilliam factorisation product.
\end{remark}

\section{Chiral Chevalley--Eilenberg Form}

\begin{proposition}[CE model for hCS observables]\label{prop:ce-model-hcs}
The classical local observables of 6D holomorphic Chern--Simons theory
are the completed chiral Chevalley--Eilenberg cochains of $L_U$:
\[
 \Obs_{\hCS}^{\mathrm{cl}}(U)
 =
 \CE^\bullet_{\dbar,\chir}
 \bigl(\Omega_c^{0,\bullet}(U,\g)[1],\cO_U\bigr).
\]
The quantum observables are an $\hb$-adic deformation of this CE complex
by the renormalised BV Laplacian and the effective higher brackets.
\end{proposition}

\begin{proof}
The underlying graded vector space of local polynomial observables is
the completed symmetric algebra on $L_U^\vee[-1]$, which is the CE
cochain space of $L_U$. The linear part of the CE differential is
$\dbar$. The quadratic part is dual to the bracket
$[\ ,\ ]_{\g}$ combined with wedge product. Quantisation adds the
second-order BV operator determined by the propagator and local
counterterms. Costello--Gwilliam renormalisation packages the resulting
operations as an effective $L_\infty$ structure on the same completed
Dolbeault field space, hence as a deformed CE differential.
\end{proof}

\section{Dualizability}

\begin{theorem}[Dualizability under compactness hypotheses]\label{thm:dualizability-scope}
Let $A_X=\Obs_{\hCS}(X,\g)$ be the renormalised quantum observable
algebra in a Morita $(\infty,3)$-category of $E_3$-algebras whose trace
levels are required to be compact.
\begin{enumerate}
\item Finite jet truncations of the abelian formal-disk theory are
dualizable in the corresponding finite-dimensional $E_3$ Morita
category.
\item For nonabelian $\g$, the full theory on $\C^3$ lies outside this
compact Morita target. Its trace level has infinite rank.
\item For compact CY$_3$ targets, full framed $E_3$ dualizability
requires compact BV quantisation, smoothness and properness of the
resulting $E_3$ algebra as a bimodule object, a chain-level $S^3$
framing or TCFT carrier, anomaly cancellation, orientation data, and a
completed hCS-to-Hall comparison.
\end{enumerate}
\end{theorem}

\begin{proof}
For a finite jet quotient, the abelian field complex is a finite
perfect complex and the Gaussian BV integral gives a perfect pairing.
This is the standard finite-dimensional duality argument.

For the full noncompact nonabelian theory on $\C^3$, the $E_3$ Hochschild
cohomology retains infinitely many translation descendants. In the
Gwilliam--Williams computation for higher Kac--Moody observables,
\[
 \HH^0_{E_3}\bigl(\Obs_{\hCS}(\C^3,\mathfrak{gl}_n)\bigr)
 \cong \C[[\tau_1,\tau_2,\tau_3]],
\]
an infinite-rank formal power series algebra. A fully dualizable object
in the compact Morita target has perfect trace levels. The displayed
trace algebra has infinite rank over $\C$, which violates that
compactness requirement.

On compact $X$, Dolbeault cohomology is finite-dimensional, so the
Hodge-theoretic input needed for properness is present. Full framed
dualizability also requires smoothness of the algebra as a bimodule
object. The remaining data are the compact BV quantum master equation,
the framed strictification or TCFT carrier, orientation of the
determinant line, and the completion needed to compare hCS observables
with the Hall algebra.
\end{proof}

\section{The K3 Product}

\begin{proposition}[K3$\times E$ numerical checks]\label{prop:k3e-numerical-checks}
For $X=K3\times E$,
\[
 \kcat(X)=0,\qquad
 \kch(A_X)=0,\qquad
 \kch^{\mathrm{Heis}}(X)=3,\qquad
 \kBKM(\Delta_5)=5,\qquad
 \kfib(X;K3)=24.
\]
The set of numerical values is $\{0,3,5,24\}$.
\end{proposition}

\begin{proof}
K\"unneth gives
\[
 \kcat(X)=\chi(\cO_{K3})\chi(\cO_E)=2\cdot0=0.
\]
The compact Hodge supertrace gives
\[
 \kch(A_X)=\sum_{q=0}^3(-1)^q h^{0,q}(X)=1-1+1-1=0.
\]
The Heisenberg specialisation contributes the rank sum $2+1=3$. The
Borcherds denominator $\Delta_5$ has weight $5$, equivalently
$\kBKM(\Delta_5)=c_1(0)/2=10/2=5$. The fibre invariant is the Mukai rank
$\mathrm{rk}\,\widetilde H(K3,\Z)=24$. These are four constructions:
total-space Euler characteristic, compact chiral Hodge supertrace and
Heisenberg specialisation on the chiral side, Borcherds weight, and K3
fibre Mukai rank.
\end{proof}

\section{Operadic Level}

\begin{proposition}[Native output at $d=3$]\label{prop:d3-native-output-j2}
The hCS factorisation algebra on a CY$_3$ is an $E_3$ holomorphic
factorisation object before factorisation-homology specialisation. For
a fixed admissible pair $(\Sigma_2,C)$, the specialised CY-to-chiral
output $\Phi_3^{(\Sigma_2,C)}$ is native $E_1$-chiral. An $E_2$
structure belongs to a Drinfeld centre, double, or representation
category layer after the relevant hypotheses are supplied.
\end{proposition}

\begin{proof}
The PTVV shift law gives Hochschild cochains of a CY$_3$ category an
$E_3$ structure. Factorisation homology integrates over the complex
surface $\Sigma_2$ and restricts to the curve $C$, hence the residual
holomorphic operadic level is
\[
 E_{3-2}=E_1.
\]
The centre construction is a separate Morita operation on the
representation category:
\[
 \mathrm{Rep}^{E_1}(A)\longmapsto
 Z\bigl(\mathrm{Rep}^{E_1}(A)\bigr).
\]
The resulting centre may carry an $E_2$ structure. That structure is a
Morita-layer product external to the specialised algebra.
\end{proof}

\section{Two Formulations}

The chain-level statement consists of the Dolbeault local Lie algebra,
the renormalised BV differential, the finite-jet perfect pairing, and
the $S^5$ residue on the diagonal link. The $(\infty,1)$-categorical
statement concerns compactness, smoothness, properness, and
evaluation--coevaluation data for an $E_3$ algebra in a Morita target.
The comparison between the two formulations is part of the data: a
compact BV quantisation, a compatible framing, and an orientation of the
determinant line.
