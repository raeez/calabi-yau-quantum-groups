\paragraph{Objects and hypotheses.}
The vertex-operator-algebra statements in this note are theorems of
Frenkel--Lepowsky--Meurman, Borcherds, and Carnahan. The CY\(_3\)
realisation below is conditional. Its input is a smooth proper CY\(_3\)
category \(\mathcal C_B\) equipped with a framed Leech
\(\Z/2\)-orbifold sector and a second-stage specialization compatible
with the FLM twisted sector. The symbol \(X_B^{\mathrm{form}}\) names
only this formal Leech CFT datum together with the elliptic curve
\(E_\tau\).

For any constructed \(d=3\) input, Vol III's two-stage rule
\[
  \Phi_3=\SpCh_{\Sigma_2,C}\circ\PhiFA_3
\]
places the specialized chiral output at the native \(E_1\)-level. Any
\(E_2\)-braiding belongs to the Drinfeld centre
\(\mathcal Z(\Rep^{E_1}(-))\).

\section*{1. The FLM moonshine module}

\begin{theorem}[Frenkel--Lepowsky--Meurman]
\label{thm:flm-vnatural}
Let \(\Leech\) be the Leech lattice, let \(\theta\) be the involution
induced by \(-1\) on \(\Leech\), and let \(\VLam^T\) be the unique
irreducible \(\theta\)-twisted \(\VLam\)-module. The orbifold
\[
  \Vnat=\VLam^{+}\oplus(\VLam^T)^{+}
\]
is a holomorphic vertex operator algebra of central charge \(24\), with
\[
  \Tr_{\Vnat}\bigl(q^{L_0-1}\bigr)
  =J(\tau)=j(\tau)-744
  =q^{-1}+196884q+21493760q^2+\cdots ,
\]
\(\Vnat_1=0\), and \(\Aut(\Vnat)=\Mon\). Its weight-two algebra is
\(\C\omega\oplus\mathfrak B\), where \(\omega\) is the Virasoro vector
and \(\mathfrak B\) is the \(196883\)-dimensional Griess algebra.
\end{theorem}

\begin{proof}[Source]
FLM construct the Leech lattice VOA, the \(\theta\)-twisted module, and
the fixed-sector orbifold in Chapters 10--12 of \emph{Vertex Operator
Algebras and the Monster}. The character identity follows from central
charge \(24\), the vacuum \(q^{-1}\)-term, and \(\Vnat_1=0\). FLM
Theorem 12.3.1 identifies the full automorphism group with the Monster.
\end{proof}

\begin{construction}[Conditional two-stage host]
\label{cons:vnat-two-stage}
A CY\(_3\) host for \(\Vnat\) consists of the following data.
\begin{enumerate}
\item A smooth proper CY\(_3\) category \(\mathcal C_B\).
\item A first-stage factorisation algebra
\[
  \PhiFA_3(\mathcal C_B)\in\ETHolFA .
\]
\item A formal Leech-orbifold specialization datum
\[
  s_B=(\Leech/\{\pm1\},E_\tau,\alpha_{\mathrm{FLM}})
\]
where \(\alpha_{\mathrm{FLM}}\) records the FLM twisted sector and its
gluing with the untwisted fixed sector.
\item A specialization functor \(\SpCh_{s_B}\) whose factorisation
homology over \(s_B\) recovers both summands
\(\VLam^+\) and \((\VLam^T)^+\).
\end{enumerate}
Given these data, set
\[
  A_B(\tau):=\SpCh_{s_B}\bigl(\PhiFA_3(\mathcal C_B)\bigr).
\]
\end{construction}

\begin{conjecture}[Moonshine module as a second-stage output]
\label{conj:vnat-two-stage}
If Construction~\ref{cons:vnat-two-stage} exists, then there is an
isomorphism of \(E_1\)-chiral algebras
\[
  A_B(\tau)\cong\Vnat .
\]
Consequently
\[
  Z_{E_\tau}(A_B)=\Tr_{\Vnat}\bigl(q^{L_0-1}\bigr)=J(\tau).
\]
The mathematical input still to be supplied is the geometric category
\(\mathcal C_B\), the Leech-orbifold specialization datum \(s_B\), and
the comparison between \(\SpCh_{s_B}\PhiFA_3(\mathcal C_B)\) and the
FLM orbifold complex.
\end{conjecture}

\section*{2. McKay--Thompson series and the Monster denominator}

For \(g\in\Mon\), define the McKay--Thompson series
\[
  \Tg(\tau)
  =
  \Tr_{\Vnat}\bigl(g\,q^{L_0-1}\bigr)
  =
  \sum_{n\geq -1}c_g(n)q^n,
  \qquad q=e^{2\pi i\tau}.
\]
The identity class gives \(T_e=J\).

\begin{theorem}[Borcherds' Conway--Norton theorem]
\label{thm:hauptmodul-borcherds}
For every \(g\in\Mon\), \(\Tg\) is the normalized Hauptmodul for the
genus-zero group assigned to \(g\) by Conway--Norton. The identity-class
denominator is the Koike--Norton--Zagier product
\[
  J(\sigma)-J(\tau)
  =
  p^{-1}
  \prod_{\substack{m>0\\ n\in\Z}}
  \bigl(1-p^m q^n\bigr)^{c_e(mn)},
  \qquad p=e^{2\pi i\sigma}.
\]
For general \(g\), Borcherds' equivariant denominator is the Adams
operation form
\[
  \Tg(\sigma)-\Tg(\tau)
  =
  p^{-1}
  \exp\!\left(
    -\sum_{r\geq1}\frac{1}{r}
     \sum_{\substack{m>0\\ n\in\Z}}
     c_{g^r}(mn)p^{mr}q^{nr}
  \right).
\]
\end{theorem}

\begin{proof}[Source]
Borcherds constructs the Monster Lie algebra \(\mMon\) from
\(\Vnat\otimes V_{\mathrm{II}_{1,1}}\), identifies its root
multiplicities with the shifted coefficients of \(J\), and proves the
denominator identity in \emph{Inventiones} 109 (1992), Theorems 1.1,
6.1, and 10.1. The equivariant identity is the corresponding twined
denominator formula: the \(r\)-th Adams operation sees \(g^r\), which
explains the appearance of \(c_{g^r}\). The resulting replication
relations prove the Conway--Norton genus-zero assertion.
\end{proof}

\begin{conjecture}[Twined sectors in the CY\(_3\) host]
\label{conj:g-twisted-spec}
Assume Construction~\ref{cons:vnat-two-stage} and an action of
\(\Mon\) on \(\PhiFA_3(\mathcal C_B)\) whose second-stage specialization
is the FLM action on \(\Vnat\). For each \(g\in\Mon\), the
\(g\)-twisted specialization
\[
  \SpCh_{s_B(g)}\bigl(\PhiFA_3(\mathcal C_B)\bigr)
\]
is the irreducible \(g\)-twisted \(\Vnat\)-module. Its \(h=1\) graded
trace is \(\Tg(\tau)\).
\end{conjecture}

\section*{3. Commuting-pair sectors}

Let \(g,h\in\Mon\) commute. Generalized moonshine attaches to the pair
an irreducible \(g\)-twisted \(\Vnat\)-module \(V^\natural(g)\), a
projective action of \(C_{\Mon}(g)\), and a trace function
\[
  \Tgh(\tau)
  =
  \Tr_{V^\natural(g)}
  \bigl(\widetilde h\,q^{L_0-1}\bigr),
\]
where \(\widetilde h\) is a lift of \(h\) to the twisted module. Changing
the lift rescales \(\Tgh\) by a nonzero scalar.

\begin{theorem}[Carnahan's generalized moonshine theorem]
\label{thm:carnahan-generalized-moonshine}
For the functions \(\Tgh\) attached to \(V^\natural\):
\begin{enumerate}
\item \(\Tgh\) is invariant under simultaneous conjugacy of
\((g,h)\), up to scalar.
\item For
\(\gamma=\begin{psmallmatrix}a&b\\ c&d\end{psmallmatrix}\in
\mathrm{SL}_2(\Z)\),
\[
  T_{(g^a h^c,g^b h^d)}(\tau)
  \quad\text{is proportional to}\quad
  \Tgh\!\left(\frac{a\tau+b}{c\tau+d}\right).
\]
\item Each \(\Tgh\) is constant or a Hauptmodul.
\item \(\Tgh=J\) exactly for \((g,h)=(1,1)\).
\end{enumerate}
\end{theorem}

\begin{proof}[Source]
Carnahan proves the theorem for twisted modules of \(\Vnat\) by
constructing the required \(g\)-twisted modules, building the associated
monstrous Lie algebras, proving the Fricke cases by twisted denominator
identities, and reducing the remaining cases through
\(\mathrm{SL}_2(\Z)\)-covariance. The genus-zero criterion used in the
last step is Carnahan 2010, \emph{Generalized Moonshine I}, Theorem 4.1:
weak Hecke-monic functions with algebraic integer coefficients and a
pole at infinity are genus-zero or degenerate. The full generalized
moonshine statement is Carnahan 2016, \emph{Generalized Moonshine IV},
Theorem ``Generalized Moonshine for twisted modules of \(V^\natural\)''.
\end{proof}

\begin{conjecture}[Commuting-pair second stage]
\label{conj:two-stage-hosting-generalized}
Under Conjecture~\ref{conj:g-twisted-spec}, a commuting pair
\((g,h)\) is the monodromy of a flat \(\Mon\)-bundle on \(E_\tau\).
The specialization over this bundle,
\[
  \SpCh_{s_B(g,h)}\bigl(\PhiFA_3(\mathcal C_B)\bigr),
\]
is \(V^\natural(g)\) equipped with the projective \(h\)-action, and its
graded trace is \(\Tgh(\tau)\). The scalar in the modular covariance is
the holonomy of the anomaly line of this finite orbifold. Identifying
that anomaly line with the transgression of a Dijkgraaf--Witten
\(3\)-cocycle requires an additional construction beyond Carnahan's
Hauptmodul theorem.
\end{conjecture}

\section*{4. The identity sector and \(j(\tau)\)}

\begin{conjecture}[\(J(\tau)=j(\tau)-744\) as the identity-sector character]
\label{conj:j-universal-witness}
Assume a CY\(_3\) host satisfying
Construction~\ref{cons:vnat-two-stage}. Then
\[
  Z_{E_\tau}\!\left(
  \SpCh_{s_B}\PhiFA_3(\mathcal C_B)
  \right)
  =
  J(\tau)
  =
  j(\tau)-744.
\]
The same first-stage datum hosts, after the additional twisted-sector
and flat-bundle data of Conjectures~\ref{conj:g-twisted-spec} and
\ref{conj:two-stage-hosting-generalized}, the family
\[
  J,\qquad \Tg,\qquad \Tgh,\qquad
  \Tg(\sigma)-\Tg(\tau).
\]
These are respectively the identity trace, ordinary twined trace,
commuting-pair trace, and Borcherds denominator trace.
\end{conjecture}

The constant \(744\) is the normalization passing from the Klein
\(j\)-function to the Monster character \(J\) with vanishing constant
term. It carries no \(\kappa_\bullet\)-content and has no additive
relation to the \(K3\times E\) invariants below.

\section*{5. Separation from the \(K3\times E\) Borcherds constants}

The Monster denominator \(J(\sigma)-J(\tau)\) has bi-weight \((0,0)\).
In the Monster Lie algebra convention this gives
\[
  \kBKM(\mMon)=0.
\]
The \(K3\times E\) Borcherds object is the separate
Gritsenko--Nikulin form \(\Delta_5\). For \(Y=K3\times E\), the
canonical constants are
\[
\begin{array}{c|c|l}
\text{invariant} & \text{value} & \text{source} \\
\hline
\kcat(Y) & 0 &
\chi(\mathcal O_{K3})\chi(\mathcal O_E)=2\cdot0 \\
\kch^{\mathrm{compact}}(Y) & 0 &
\sum_q(-1)^q h^{0,q}(Y)=1-1+1-1 \\
\kch^{\mathrm{Heis}}(Y) & 3 &
\text{Heisenberg--Mukai specialization} \\
\kBKM(\Delta_5) & 5 &
c_1(0)/2=10/2 \\
\kfib(Y) & 24 &
\operatorname{rk}\widetilde H(K3,\Z)=1+22+1 .
\end{array}
\]
The primitive CHL/BKM denominator ladder has
\[
  N\in\{1,2,3,4,6\},\qquad
  c_N(0)=(10,8,6,4,2),\qquad
  \kBKM(\Phi_N)=(5,4,3,2,1).
\]
The Gritsenko--Clery diagonal-divisor catalogue is a different
eight-row list. Its twice-weight tuple is
\[
  (10,4,6,2,4,1,3,2),
\]
with the overlap with the CHL ladder only at the common
\(\Delta_5\) row.

In the theta-normalized Igusa convention,
\[
  [q^{1/2}r^{1/2}s^{1/2}]\Delta_5=64,
  \qquad
  D_5=64^{-1}\Delta_5.
\]
Thus \(D_5\) is the monic Borcherds product, while \(\Delta_5\) is the
theta-normalized Siegel cusp form of weight \(5\).
