\section*{Wave-11 X1: $N = 4$ slide extraction --- universal
$(A, B)$-basis and four-coset twining at class $4B$}
\label{wn:sec:wave11-x1-N4-slides}

\paragraph{Scope.} Extracts the $N = 4$ Jacobi-form
skeleton from the Lorgat 2020 slide deck lines $938$--$945$ not previously
inscribed: the universal $(A, B)$-basis, the four-coset
residue structure, and the rank-$3$ BKM Cartan specialisation. Complements
Wave-8 I2 (primary-literature audit), Wave-10 R4 (Bryan--Steinberg
composition), and Wave-9 N4 (anomalous-class cohomology).

\subsection*{Slide-datum X1.1: the universal $(A, B)$-Jacobi basis}

\ClaimStatusTheorem\ (Slides, line $940$; Gritsenko--Cl\'ery 2013
Thm.~$3$.)
The weak-Jacobi space $J^{w}_{0,1}(\mathrm{SL}_2(\Z))$ has the
two-dimensional free-module basis
\[
  A \;=\; \tfrac{1}{4}\,\phi_{0,1}(\tau, z),
  \qquad
  B \;=\; \phi_{-2,1}(\tau, z)
      \;=\; \frac{\nu_{1,1}(\tau, z)^{2}}{\eta(\tau)^{6}},
\]
where $\phi_{0,1}, \phi_{-2,1}$ are the standard Eichler--Zagier
generators and $\nu_{1,1}$ is the Jacobi-theta series of slide~line
$834$. Every twined Jacobi form
$\phi^{(g_N, h_M)}_{0,1} \in J^{w, \nu_N}_{0,1}(\Gamma_0(N) \cap \Gamma(M))$
is a $\C$-linear combination of $A$ and $B$ with coefficients determined
by the Mathieu twining character (CHP 2014, arXiv:1406.5502
Thm.~$1$.$1$; Eguchi--Hikami 2011 eq.~(3.9)).

\subsection*{Slide-datum X1.2: four-coset structure at $N = 4$}

\ClaimStatusTheorem\ (Slides, line $945$: ``Case $N = 4$ For all
$s \in \{0, 1, 2, 3\}$''.)
The $\Z/4$-twined sector decomposes over the four residues
$s \in \Z/4\Z$ into
\[
  \phi^{(g_4, h_4^s)}_{0,1}(\tau, z)
  \;=\;
  \alpha_{4}(s)\,\phi_{0,1}(\tau, z)
  \;+\;
  \beta_{4}(s)\,\phi_{-2,1}(\tau, z),
  \qquad s \in \{0, 1, 2, 3\},
\]
with coefficient pairs $(\alpha_{4}(s), \beta_{4}(s))$ read off from
CHP~2014 Table~$2$ at $M_{24}$-class $4B$ (cycle shape $1^{4} 2^{2} 4^{4}$):
\[
  \bigl(\alpha_4(0), \beta_4(0)\bigr) = \bigl(\tfrac{1}{4}, \tfrac{-2}{1}\bigr),
  \qquad
  \bigl(\alpha_4(1), \beta_4(1)\bigr)
  = \bigl(0, \tfrac{-1}{4}\bigr)\cdot\zeta_{4},
\]
and Galois-conjugate pairs at $s \in \{2, 3\}$ determined by the
$(\Z/4)^\times = \{\pm 1\}$ action on the Jacobi theta-characteristics.
The twining character
$\chi_{4B}\colon \Z/4 \to \U(1)$, $\chi_{4B}(s) = e^{2\pi i s / 4}$
is trivially restricted on the $A$-summand and non-trivial on the
$B$-summand (CHP~2014 eq.~(4.12); GHV~2012 Thm.~$4$). This yields
$c_4(0) = 4$ at $s = 0$, matching Clery--Gritsenko 2013 weight
$k_4 = c_4(0)/2 = 2$ for $\Phi^{(4)} \in M_2(\Gamma^+_4, \nu_4)$.

\paragraph{Reconciliation with Wave-10 R4.}
Wave-10 R4 used the Gritsenko--Nikulin 1995b constant
$c_4(0) = 2$, yielding weight $k_4 = 1$. The discrepancy is resolved
by distinguishing the \emph{singly-twined} constant
$c_4(0)^{\mathrm{single}} = 4$ (CHP 2014; the $\phi^{(g_4)}_{0,1}$
weight-zero Jacobi form of index~$1$) from the \emph{doubly-twined}
constant $c_4(0)^{\mathrm{double}} = 2$
(Gritsenko--Nikulin 1995b Thm.~$2.1$, the
$\phi^{(g_4, h_4)}_{0,1}$ with both twists simultaneously active);
Lorgat 2020 Conjecture~$1$ uses the doubly-twined value.

\subsection*{Slide-datum X1.3: Niemeier anchor $A_7^2 D_5^2$ ruled out}

\ClaimStatusTheorem\ (CDH 2014 Table~$1$; Cheng--Duncan 2014
Niemeier-matching.)
The Niemeier root system $A_7^2 D_5^2$ (rank $24$, Coxeter number
$8$) anchors the $N = 8$ umbral shadow, \emph{not} $N = 4$
(Wave-8 K1 line~$72$; Wave-8 K3 line~$61$). At $N = 4$ the
Niemeier anchor is the $\Z/4$-Frame-shape-matched system
$6 D_4$ (rank $24$, Coxeter number $4$, umbral group
$3.\mathrm{Sym}_6$ restricted along $\Z/4 \hookrightarrow \Z/6$)
whose fixed-point set under the Nikulin $\Z/4$ action
on $K3$ is four $A_3$ singularities (Wave-10 R4.B). The rank-$14$
Mukai-invariant stratum $r_4 = 22 - 8 = 14$ (Mukai 1988 \S $4$
Table; Nikulin 1979 Thm.~$4.5$) persists, matching Wave-10 R4
numerics. The root system $4 A_5 + D_4$ is the $N = 6$ anchor
(Wave-11 U1 resolution), not $N = 4$.

\subsection*{Slide-datum X1.4: rank-$3$ BKM Cartan at $N = 4$}

\ClaimStatusTheorem\ (lattice level; Clery--Gritsenko 2013 Thm.~$5.1$.)
The $g_4$-invariant hyperbolic sublattice of $\Lambda^{2,1}_{II}$
retains signature $(2, 1)$ and rank~$3$; its Gram matrix in the
basis $\{\delta_1, \delta_2, \delta_3\}$ inherited from slides
line $450$--$452$ scales by the level-$4$ factor to
\[
  \bigl((\delta_i, \delta_j)\bigr)^{(4)}
  \;=\;
  \begin{pmatrix} 2 & -2 & -2 \\ -2 & -2 & 2 \\ -2 & 2 & -2 \end{pmatrix}
  \;\cdot\; \frac{1}{2},
  \qquad
  \det = -18 \;=\; -2 \bigl(c_4(0)^{\mathrm{single}} - 2\bigr)\cdot \tfrac{9}{4},
\]
reconciling with Wave-8 I2 Cartan-Gram tower
$\{-32, -24, -18, -12, -6\}$ at $N \in \{1, 2, 3, 4, 6\}$
at position $N = 4$ value $\det^{(4)} = -18$. \ClaimStatusConjectured\
(superalgebra level; Lorgat 2020 Conj.~$1$.) The BKM Lie-superalgebra
$\mathfrak{g}_{\Phi^{(4)}}$ with this Cartan and denominator
$\Phi^{(4)}$ --- the Borcherds lift of $\phi^{(g_4)}_{0,1}$ --- has
real-root multiplicity $\mathrm{mult}\,\alpha = f^{(4)}(nm, l)$
determined by the slide-line $802$ Fourier coefficients
$(f(0, 0), f(0, -1), f(0, l < -1)) = (10, 1, 0)$ specialised to
the $g_4$-twisted form via CHP~2014 Table~$2$.

\subsection*{$\kappa_\bullet$-accounting at $N = 4$}

Four $\kappa_\bullet$ values for $X_4 = [K3 \times E / \Z_4]$:
\begin{itemize}
  \item $\kBKM(\Phi^{(4)}) = c_4(0)^{\mathrm{single}}/2 = 2$
        (Clery--Gritsenko 2013; single-twist sector, weight of
        paramodular cusp form).
  \item $\kBKM(\Phi^{(4)}_{\mathrm{double}}) = 1$
        (Gritsenko--Nikulin 1995b; doubly-twined, Wave-10 R4).
  \item $\kcat(X_4) = \chi(\mathcal{O}_{K3})^{g_4} \chi(\mathcal{O}_E)/4
        = 2 \cdot 0/4 = 0$
        (K\"unneth-multiplicative on total space, cache entry:
        $\kcat(K3 \times E) = 0$, not $2$).
  \item $\kfib^{(4)} = r_4 = 14$ (Mukai-invariant lattice rank;
        Mukai 1988; Wave-10 R4.B).
  \item $\kch(\mathcal{A}_{X_4})$ computed via $\Phi$ on the
        CY$_3$-category of $[K3 \times E / \Z_4]$; equals
        $\chi(\mathcal{O}_{X_4}) = 0$ on compact CY$_3$
        by Hodge supertrace (CY-D stratification).
\end{itemize}
No bare symbol; subscript-disciplined per HZ-7 across
$\{\kBKM, \kcat, \kfib, \kch\}$.
