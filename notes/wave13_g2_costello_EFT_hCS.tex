\title{Costello effective-field-theory quantisation of abelian 6D hCS on $\C^3$}
\author{Raeez Lorgat}
\date{2026-04-22}

\begin{document}
\maketitle

Fix $X=\C^3$ with holomorphic volume form
\[
  \Omega=dz_1\wedge dz_2\wedge dz_3
\]
and abelian coefficients $\g=\C$. The field complex is
$\cE=\Omega^{0,\bullet}(\C^3)$ with BV differential $Q=\dbar$. The cubic
hCS vertex vanishes for $\g=\C$, and the action is the quadratic
functional
\[
  S_2(\cA)=\frac12\int_{\C^3}\Omega\wedge\cA\,\dbar\cA .
\]
This note records the Costello effective-field-theory construction of
the quantum factorisation algebra $\Obs^q$ in the abelian affine model.
The non-abelian compact CY$_3$ problem requires a separate
anomaly-cancellation and completion datum.

\begin{remark}[$\hbar$ convention]\label{rmk:hbar}
Throughout, $\hbar$ is the single Costello deformation parameter:
weight one on Feynman loops, with
$S_\eff=\sum_{g\geq0}\hbar^gS_\eff^{(g)}$, and $\Obs^q$ a
$\C[[\hbar]]$-module. The OPE coefficient in
Proposition~\ref{prop:Fock}, the QME coefficient in
Theorem~\ref{thm:QME}, and the Fock quantisation parameter in
Construction~\ref{cons:Obs-q} are the same $\hbar$.
\end{remark}

\section{Classical BV datum}

\begin{construction}[Classical free BV theory on $\C^3$]
\label{cons:classical-BV}
The abelian hCS BV datum is
\[
  (\cE,Q,S_\cl,\omega_\BV)
\]
with:
\begin{itemize}
\item $\cE=\Omega^{0,\bullet}(\C^3)$ and cohomological degree
$\deg=(\text{Dolbeault})-1\in\{-1,0,1,2\}$.
\item $Q=\dbar:\Omega^{0,q}\to\Omega^{0,q+1}$ of degree $+1$.
\item $S_\cl(\cA)=\tfrac12\int_{\C^3}\Omega\wedge\cA\,\dbar\cA$.
\item The $(-1)$-shifted symplectic pairing
\[
  \omega_\BV(\cA_1,\cA_2)
  =
  \int_{\C^3}\Omega\wedge\cA_1\cA_2,
\]
non-degenerate after imposing compact support or Schwartz decay, by
Serre duality pairing
$\Omega^{0,q}\otimes\Omega^{0,3-q}\to\C$.
\end{itemize}
The classical master equation $\{S_\cl,S_\cl\}=0$ is the identity
$\dbar^2=0$ written through the inverse BV pairing
(Costello 2011, \S2.6; Costello--Gwilliam 2017 Vol.~I, \S5.3).
\end{construction}

\section{Gauge fixing and effective actions}

\begin{construction}[Propagator via heat kernel]\label{cons:propagator}
Choose the $L^2$-adjoint $\dbar^\ast$ for the flat metric on $\C^3$.
The Laplacian is $\Delta_{\dbar}=[\dbar,\dbar^\ast]$. In a
translation-invariant trivialisation the scalar part of the heat kernel
of $e^{-t\Delta_{\dbar}}$ is
\[
  (4\pi t)^{-3}e^{-|z-w|^2/4t}.
\]
The Dolbeault kernel $K_t\in\cE\widehat{\otimes}\cE$ is this Gaussian
factor multiplied by the canonical antisymmetric Dolbeault volume factor
pairing bidegrees $(0,q)$ and $(0,3-q)$. The scale propagator is
\[
  P(\varepsilon,\Lambda)
  =
  \int_\varepsilon^{1/\Lambda}(\dbar^\ast\otimes 1)K_t\,dt
  \in\cE\widehat{\otimes}\cE,
\]
with UV cutoff $\varepsilon$ and IR scale $\Lambda^{-1}$.
\end{construction}

\begin{definition}[Scale-$\Lambda$ effective action]\label{def:S-eff}
Following Costello 2011 Def.~2.5.1, for interaction $I$ one sets
\[
S_\eff[\Lambda](\cA)
=
\hbar\log\!\int_{y\in\cE_{>\Lambda}}
\exp\!\Bigl(\hbar^{-1}\bigl(S_\cl(\cA+y)+I(\cA+y)\bigr)\Bigr),
\]
where $\cE_{>\Lambda}$ denotes the eigenspaces of $\Delta_{\dbar}$ with
eigenvalue $>\Lambda^2$. Equivalently, $S_\eff[\Lambda]$ is the sum over
connected Feynman graphs with internal edges labelled by $P(0,\Lambda)$
and external legs $\cA$. In the abelian branch $I=0$.
\end{definition}

\begin{proposition}[Renormalisation-group flow]\label{prop:RG}
$S_\eff[\Lambda]$ and $S_\eff[\Lambda']$ are related by Costello's RG
equation
\[
  S_\eff[\Lambda']
  =
  W(P(\Lambda,\Lambda'),S_\eff[\Lambda]),
\]
where $W$ is the connected-graph sum with propagator
$P(\Lambda,\Lambda')$
(Costello 2011, \S2.5; Costello--Gwilliam 2017 Vol.~I,
Ch.~8, Thm.~8.2.2.1).
\end{proposition}

\begin{proof}
Gaussian integration over the intermediate modes gives the propagator
$P(\Lambda,\Lambda')$. The logarithm keeps only connected graphs. For
$\g=\C$, the interaction vanishes, so the only remaining graph
contribution is the field-independent vacuum determinant, removed by
the standard normalisation of observables.
\end{proof}

\section{Quantum master equation}

\begin{definition}[BV Laplacian at scale $\Lambda$]\label{def:BV-Lap}
The operator $\Delta_\Lambda:\cO(\cE)\to\cO(\cE)$ contracts two
functional derivatives with the regularised kernel
	$K_{1/\Lambda}\in\cE\widehat{\otimes}\cE$. It is a second-order
differential operator on polynomial functionals and vanishes on linear
and constant functionals.
\end{definition}

\begin{theorem}[Quantum master equation, abelian 6D hCS]\label{thm:QME}
For abelian 6D hCS on $\C^3$, the normalised effective action satisfies
\[
\{S_\eff[\Lambda],S_\eff[\Lambda]\}
\;+\;\hbar\,\Delta_\Lambda S_\eff[\Lambda]
\;=\;0
\qquad\text{for all }\Lambda>0 .
\]
\end{theorem}

\begin{proof}
The bracket term is the classical master equation for the free complex:
$\{S_2,S_2\}=0$ is equivalent to $\dbar^2=0$. The BV Laplacian is
compatible with $Q=\dbar$ because the heat kernel commutes with
$\dbar$ and $\dbar^\ast$. With $I=0$ there are no interaction vertices
from which a local obstruction cocycle can be formed. The possible
field-independent determinant is removed by the Costello normalisation
of the connected generating functional. Thus the renormalised QME holds
with zero anomaly in the abelian affine model.
\end{proof}

\section{Factorisation algebra of quantum observables}

\begin{construction}[Quantum observables]\label{cons:Obs-q}
For an open $U\subset\C^3$,
\[
\Obs^q(U)
=
\bigl(\Sym(\cE_c(U)^\vee[1])[[\hbar]],\ Q+\hbar\Delta_\Lambda\bigr),
\]
where $\cE_c(U)$ denotes compactly supported fields and the
$\Lambda\to\infty$ limit is defined by homotopy transfer along the RG
flow (Costello--Gwilliam 2017 Vol.~I, Ch.~8, \S8.6,
Def.~8.6.0.1). A disjoint inclusion
$U_1\sqcup U_2\hookrightarrow V$ induces
\[
  \Obs^q(U_1)\otimes\Obs^q(U_2)\longrightarrow\Obs^q(V)
\]
by multiplication of polynomial observables supported on disjoint
subsets.
\end{construction}

\begin{proposition}[Free holomorphic boson Fock module]\label{prop:Fock}
At $\g=\C$, $\Obs^q(U)$ is the Fock module of the free holomorphic
Dolbeault boson on $U$. Its basic contraction is
\[
\cA(z)\cA^\vee(w)\sim\hbar\,G_{\dbar}(z,w)+\mathrm{regular},
\]
where the Bochner--Martinelli kernel is characterised by
\[
  \dbar_zG_{\dbar}(z,w)=\delta_\Delta(z,w)\,\Omega_z .
\]
In flat coordinates it has the form
\[
  G_{\dbar}(z,w)
  =
  c_3\,
  \frac{\sum_{j=1}^3(-1)^{j-1}\overline{(z_j-w_j)}
  d\bar z_1\wedge\cdots\widehat{d\bar z_j}\cdots\wedge d\bar z_3}
  {|z-w|^6}
  \wedge dz_1\wedge dz_2\wedge dz_3
\]
for the Costello normalisation constant $c_3$.
\end{proposition}

\section{Existence and uniqueness}

\begin{theorem}[Costello 2011; Costello--Gwilliam 2017 Vol.~I, Ch.~8]
\label{thm:existence}
Let $(\cE,Q,S_\cl,\omega_\BV)$ be a free BV theory with quadratic action,
compatible elliptic gauge fixing, and heat-kernel propagator. Then:
\begin{enumerate}
\item $\Obs^q$ exists as a factorisation algebra of cochain complexes on
open subsets of the spacetime.
\item $\Obs^q$ is unique up to contractible gauge-fixing choice.
\item The classical limit $\Obs^q/\hbar\simeq\Obs^\cl$ recovers the
commutative factorisation algebra of classical observables.
\end{enumerate}
For abelian 6D hCS on $\C^3$, this is the holomorphic factorisation
algebra of the free Dolbeault boson with propagator $G_{\dbar}$.
\end{theorem}

\section{Quantisation chain}

The construction is:
\[
(\cE,\dbar,S_\cl,\omega_\BV)
\xrightarrow{\ \dbar^\ast\ }
\{S_\eff[\Lambda]\}_\Lambda
\xrightarrow{\ \QME\ }
(Q+\hbar\Delta_\Lambda)^2=0
\xrightarrow{\ \Sym,\ \Lambda\to\infty\ }
\Obs^q=\mathrm{Fock}(\dbar\text{-boson}).
\]

The abelian affine vacuum sector has one holomorphic generator; this is
the local witness behind the toric value
$\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(\C^3)=1$. It is not the compact
Hodge-supertrace theorem. For non-abelian hCS the cubic term
\[
  S_3(\cA)=\frac13\int_{\C^3}\Omega\wedge\tr(\cA[\cA,\cA])
\]
produces interaction graphs. On compact CY$_3$ targets, cancelling their
one-loop obstruction is part of the Costello--Li or Green--Schwarz
anomaly datum and of the analytic completion required before a global
curve-specialised $E_1$ chiral algebra is asserted.
