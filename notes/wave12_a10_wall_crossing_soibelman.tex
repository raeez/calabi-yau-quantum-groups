\begin{abstract}
Kontsevich-Soibelman wall-crossing is the invariance of an ordered
quantum-dilogarithm product in a completed quantum torus.  The data are a
3-Calabi-Yau category, a stability chamber, a strict active sector, fixed
orientation data, a completed motivic Hall algebra, and an integration map
to the completed quantum torus.  The Maurer-Cartan formulation is the
Baker-Campbell-Hausdorff logarithm of the same sector product.  The
conifold pentagon is the rank-two primitive case.  In the
Calabi-Yau-to-chiral application, the constructed $d\geq 3$ output of
$\Phi_d$ is $\Eone$; $\Etwo$ structure belongs to centre, double, and
module layers.
\end{abstract}

\section{Stability Chamber and Hall Completion}

\begin{definition}[Wall-crossing datum]
\label{def:wc-datum}
A Kontsevich-Soibelman wall-crossing datum consists of:
\[
(\cC,\sigma,V,o,\widehat{\Hall}_{\cC,V,o,\sigma},\mathcal I_{V,o,\sigma}).
\]
Here $\cC$ is a 3-Calabi-Yau triangulated category, $\sigma=(Z,\cA)$ is a
Bridgeland stability condition in a chamber of $\Stab(\cC)$, and
\[
 \Gamma=K_{\mathrm{num}}(\cC),\qquad
 \langle\alpha,\beta\rangle
 =
 \chi(\alpha,\beta)-\chi(\beta,\alpha)
\]
is the numerical charge lattice with its skew Euler form.  The sector
$V\subset\C^\times$ is strict and convex, with no active central-charge
ray on its boundary.  The orientation datum $o$ is the quadratic
refinement of the determinant-line gerbe required by the motivic
integration map; it is fixed throughout the comparison.

The completed Hall algebra
$\widehat{\Hall}_{\cC,V,o,\sigma}$ is the Harder-Narasimhan completion
of stack functions supported on semistable objects with
$Z_\sigma(\gamma)\in V$.  The map
\[
 \mathcal I_{V,o,\sigma}\colon
 \widehat{\Hall}_{\cC,V,o,\sigma}
 \longrightarrow
 \widehat{\mathbb T}_{\Gamma,V}
\]
is the Kontsevich-Soibelman motivic integration map in the same
orientation normalization.
\end{definition}

\begin{definition}[Completed quantum torus]
\label{def:quantum-torus}
The active cone of the sector is
\[
 \Gamma_V(\sigma)=\{\gamma\in\Gamma: Z_\sigma(\gamma)\in V\}.
\]
The completed quantum torus is the topological
$\Q(q^{1/2})$-algebra
\[
 \widehat{\mathbb T}_{\Gamma,V}
 =
 \widehat{\bigoplus}_{\gamma\in\Gamma_V(\sigma)}
 \Q(q^{1/2})\,\hat x_\gamma,
 \qquad
 \hat x_\alpha\hat x_\beta
 =
 q^{\langle\alpha,\beta\rangle/2}\hat x_{\alpha+\beta}.
\]
The completion is the monoid completion along $\Gamma_V(\sigma)$,
equivalently the topology in which every bounded central-charge sector
contains only finitely many active classes by the support property.  Its
semiclassical Lie algebra has basis $e_\gamma$ and bracket
\[
 [e_\alpha,e_\beta]
 =
 (-1)^{\langle\alpha,\beta\rangle}
 \langle\alpha,\beta\rangle e_{\alpha+\beta}.
\]
\end{definition}

\begin{definition}[Ray factors]
\label{def:ray-factors}
Let $\Omega_{\sigma,o}(\gamma)\in\Q$ be the rational DT/BPS invariant
obtained from $\mathcal I_{V,o,\sigma}$ for the fixed orientation data.
For an active ray $\ell\subset V$, define
\[
 A_\ell(\sigma,o)
 =
 \prod_{Z_\sigma(\gamma)\in\ell}
 \Ad_{\mathbb E_q(\hat x_\gamma)^{\Omega_{\sigma,o}(\gamma)}} .
\]
The product over one ray is taken in increasing central-charge norm
inside the completed ray subalgebra.  The chosen quantum dilogarithm
$\mathbb E_q$ is the one attached to the motivic normalization of
$\mathcal I_{V,o,\sigma}$.  At $q=1$ the corresponding Poisson
automorphism is
\[
 T_{\gamma,\Omega}(x_\beta)
 =
 x_\beta(1-x_\gamma)^{\Omega\langle\gamma,\beta\rangle}.
\]
\end{definition}

\section{Ordered Product}

\begin{theorem}[Kontsevich-Soibelman sector product]
\label{thm:ks-ordered-product}
Fix the datum of Definition~\ref{def:wc-datum}.  For a stability
condition $\sigma$ in a chamber define
\[
 A_V(\sigma,o)
 =
 \prod_{\ell\subset V}^{\curvearrowleft} A_\ell(\sigma,o),
\]
with rays ordered by decreasing argument $\Arg(\ell)$.  Let
$\sigma_0$ and $\sigma_1$ be joined by a path in $\Stab(\cC)$ along
which no active ray meets $\partial V$, and transport the same
orientation datum $o$ along the path.  Then
\[
 A_V(\sigma_0,o)=A_V(\sigma_1,o)
 \quad\text{in}\quad
 \Aut(\widehat{\mathbb T}_{\Gamma,V}).
\]
The individual invariants $\Omega_{\sigma,o}(\gamma)$ and ray factors
$A_\ell(\sigma,o)$ depend on the chamber; the ordered product over the
strict sector is invariant.
\end{theorem}

\begin{proof}
The stack of objects in the fixed sector has a Harder-Narasimhan
factorisation in the completed motivic Hall algebra
$\widehat{\Hall}_{\cC,V,o,\sigma}$.  When the path in stability space
does not move an active ray through the boundary of $V$, the total Hall
class of the sector remains the same and only its ordered ray
factorisation changes.  The motivic integration map
$\mathcal I_{V,o,\sigma}$ sends Hall convolution to the quantum-torus
product, with the skew Euler form producing the
$q^{\langle\alpha,\beta\rangle/2}$ commutation factor.  Applying
$\mathcal I_{V,o,\sigma}$ to the two Harder-Narasimhan
factorisations gives the displayed equality.  The support property gives
the pronilpotent convergence required by the sector completion, and the
fixed orientation datum keeps the square-root normalization constant.
\end{proof}

\begin{remark}[Variables]
The central charge $Z_\sigma$ determines rays, the function
$\Omega_{\sigma,o}$ determines multiplicities, $A_\ell(\sigma,o)$ is the
automorphism attached to one ray, and $A_V(\sigma,o)$ is the invariant
sector product.  A physical marginal-stability wall becomes a theorem
only after the category, stability condition, strict sector, orientation
data, Hall algebra, integration map, and completion have been fixed.
\end{remark}

\section{Maurer-Cartan Logarithm}

\begin{definition}[Logarithmic sector element]
Let $\fg_{\Gamma,V}$ be the completed Lie algebra of the pro-unipotent
subgroup $1+\widehat{\mathbb T}_{\Gamma,V}^{>0}$.  The logarithmic
wall-crossing element is
\[
 \Theta_V(\sigma,o)=\log A_V(\sigma,o)\in\widehat{\fg}_{\Gamma,V},
\]
where the logarithm is the Baker-Campbell-Hausdorff logarithm in the
completed group.
\end{definition}

\begin{proposition}[BCH gauge form]
\label{prop:mc-interpretation}
Under the hypotheses of Theorem~\ref{thm:ks-ordered-product},
\[
 \Theta_V(\sigma_0,o)=\Theta_V(\sigma_1,o).
\]
Equivalently, two ray-factorisation charts of the same sector product
are related by Baker-Campbell-Hausdorff gauge transformations in
$\widehat{\fg}_{\Gamma,V}$.
\end{proposition}

\begin{proof}
Theorem~\ref{thm:ks-ordered-product} gives equality of the two
pro-unipotent group elements.  The logarithm is defined on the completed
group by the same cone filtration; applying it gives the equality of
logarithms.  The chamber-dependent raw sum
$\sum_\gamma\Omega_{\sigma,o}(\gamma)e_\gamma$ is not the invariant
object.  Its ordered quantum-dilogarithm exponential is the object
controlled by wall-crossing.
\end{proof}

\section{Conifold Pentagon}

Let
\[
 Y=\operatorname{Tot}(\cO_{\bP^1}(-1)\oplus\cO_{\bP^1}(-1))
\]
be the resolved conifold.  Use the Klebanov-Witten Jacobi algebra
$J(Q_{\mathrm{con}},W_{\mathrm{con}})$, the Szendroi noncommutative DT
heart, and the adjacent large-volume heart inside the same completed
Hall algebra.  The elementary wall has active lattice
\[
 \Gamma_W=\Z\gamma_1\oplus\Z\gamma_2,\qquad
 \langle\gamma_1,\gamma_2\rangle=1,
\]
where $\gamma_1,\gamma_2$ are the two simple primitive charges.

\begin{proposition}[Quantum pentagon]
\label{prop:conifold-pentagon}
Fix a strict sector $V$ containing the rays of
$\gamma_1,\gamma_2,\gamma_1+\gamma_2$ and no active boundary ray.
Suppose the central-charge order crosses from
\[
 \Arg Z_{\sigma_-}(\gamma_1)>\Arg Z_{\sigma_-}(\gamma_2)
\]
to the opposite order.  In the completed quantum torus with
\[
 \hat x_{\gamma_1}\hat x_{\gamma_2}
 =
 q^{1/2}\hat x_{\gamma_1+\gamma_2},
 \qquad
 \hat x_{\gamma_2}\hat x_{\gamma_1}
 =
 q^{-1/2}\hat x_{\gamma_1+\gamma_2},
\]
the invariant sector product is
\[
 \mathbb E_q(\hat x_{\gamma_1})\,
 \mathbb E_q(\hat x_{\gamma_2})
 =
 \mathbb E_q(\hat x_{\gamma_2})\,
 \mathbb E_q(\hat x_{\gamma_1+\gamma_2})\,
 \mathbb E_q(\hat x_{\gamma_1}).
\]
In the chamber $\sigma_+$ the primitive classes in this rank-two cone
have
\[
 \Omega_{\sigma_+,o}(\gamma_1)=
 \Omega_{\sigma_+,o}(\gamma_2)=
 \Omega_{\sigma_+,o}(\gamma_1+\gamma_2)=1 .
\]
\end{proposition}

\begin{proof}
The two simple conifold charges form the $A_2$ scattering diagram.  At
the wall their central charges align; consistency of the Hall product
around the joint inserts the bound-state ray
$\R_{>0}Z(\gamma_1+\gamma_2)$.  The Hall identity is the
Harder-Narasimhan factorisation before and after the simple tilt of the
heart.  Applying the fixed oriented integration map gives precisely the
quantum dilogarithm identity displayed above.  The specialization
$q=1$ gives the equality of Poisson automorphisms
$T_{\gamma,\Omega}$.
\end{proof}

\begin{remark}[Flop and stability wall]
The conifold flop is the derived equivalence
$D^b(Y)\simeq D^b(Y^+)$.  It transports the charge lattice between two
birational models.  The pentagon is a stability-wall identity inside a
single completed Hall algebra after a heart, central charge, sector, and
orientation datum are fixed.  The conifold chiral invariant is
\[
 \kch(Y)=1,
\]
computed from the Klebanov-Witten NCCR model; it is not a local-surface
Hodge computation.
\end{remark}

\section{Calabi-Yau-to-Chiral Consequences}

\begin{proposition}[Native output for $d\geq 3$]
\label{prop:d3-output}
On constructed Calabi-Yau-to-chiral loci in dimension $d\geq 3$, the
specialised output of $\Phi_d$ is an $\Eone$-chiral algebra.  The
$\Etwo$ structures used in Yangian, $R$-matrix, and braided
representation statements live after passage to a Drinfeld centre,
Drinfeld double, representation category, or evaluation module.  In
particular
\[
 \CoHA(\C^3)\simeq \Yplus(\widehat{\mathfrak{gl}}_1)
\]
is the Hall positive half.  The $\mathcal W_{1+\infty}$ algebra is
obtained after passage to the full Yangian or Drinfeld double, central
completion, and Fock/evaluation realisation.
\end{proposition}

\begin{proof}
The toric critical CoHA computation for $\C^3$ identifies the Hall
algebra with the positive half of the affine Yangian.  The braided
structure required by the representation-theoretic $R$-matrix is
produced after passing to centre, double, or module category.  This is
the same separation as in wall-crossing: the Hall algebra supplies the
product identity, and the representation layer supplies the braided
realisation.
\end{proof}

\section{K3 x E and Borcherds Constants}

\begin{proposition}[Four invariant slots for $K3\times E$]
\label{prop:k3xe-four-slots}
The compact threefold $K3\times E$ carries four distinct invariant
slots:
\[
 \kcat(K3\times E)=0,\qquad
 \kch(K3\times E)=0,\qquad
 \kch^{\mathrm{Heis}}(K3\times E)=3,
\]
\[
 \kBKM(\fg_{\Delta_5})=5,\qquad
 \kfib(K3)=24.
\]
The values arise from different constructions.
\end{proposition}

\begin{proof}
Kunneth multiplicativity gives
\[
 \kcat(K3\times E)
 =
 \chi(\cO_{K3})\chi(\cO_E)
 =
 2\cdot0
 =
 0.
\]
The compact total-space chiral Hodge supertrace is
\[
 \kch(K3\times E)
 =
 \sum_q(-1)^q h^{0,q}(K3\times E)
 =
 1-1+1-1
 =
 0.
\]
The Heisenberg-Mukai specialisation has
$\kch^{\mathrm{Heis}}(K3\times E)=3$.  The Borcherds slot is the weight
of Gritsenko's paramodular form:
\[
 \kBKM(\fg_{\Delta_5})=c_1(0)/2=10/2=5.
\]
The fibre slot is the Mukai-lattice rank
\[
 \kfib(K3)=\operatorname{rk}H^\bullet(K3,\Z)=24.
\]
The fibre value $\chi(\cO_{K3})=2$ is not a total-space invariant.
\end{proof}

\begin{proposition}[CHL ladder and Gritsenko-Clery separation]
\label{prop:chl-gc-separation}
For the primitive compact CHL ladder
$N\in\{1,2,3,4,6\}$, the Borcherds input constants and BKM
characteristics are
\[
\begin{array}{c|ccccc}
N&1&2&3&4&6\\ \hline
c_N(0)&10&8&6&4&2\\
\kBKM(\Phi_N)=c_N(0)/2&5&4&3&2&1 .
\end{array}
\]
The Gritsenko-Clery diagonal-divisor catalogue is a separate eight-row
classification.  Its triples and weights are
\[
 (1,1;5),\ (2,1;2),\ (1,2;3),\ (3,1;1),\
 (1,3;2),\ (4,1;\tfrac12),\ (1,4;\tfrac32),\ (2,2;1),
\]
with twice-weight tuple
\[
 (10,4,6,2,4,1,3,2).
\]
The two lists coincide only at the common $\Delta_5$ row
$(t,N;k)=(1,1;5)$.
\end{proposition}

\begin{proof}
Borcherds' weight theorem gives
$\mathrm{wt}(\Phi_N)=c_N(0)/2$.  Gritsenko-Nikulin construct the CHL
paramodular forms for $N\in\{1,2,3,4,6\}$ with weights
$(5,4,3,2,1)$, hence
$c_N(0)=(10,8,6,4,2)$ and
$\kBKM(\Phi_N)=(5,4,3,2,1)$.  The $N=1$ form is $\Delta_5$, and
$\mathrm{wt}(\Delta_5)=5=c_1(0)/2$.

Gritsenko-Clery classify genus-two Siegel modular forms whose divisor is
the diagonal Humbert divisor with multiplicity one.  Their eight
triple-indexed rows have the displayed weights, so their twice-weight
tuple is $(10,4,6,2,4,1,3,2)$.  This tuple is not the CHL constant table
$(10,8,6,4,2)$.  Componentwise identification away from $\Delta_5$
therefore has no numerical content without an explicit row map, group,
cover, character, and denominator datum.
\end{proof}

\section{CY4 Centre-Layer Statement}

\begin{conjecture}[Local CY4 centre-layer enhancement]
\label{conj:cy4-centre-layer}
Let $\cC$ be a local CY4 Jacobi category equipped with the Costello-Li
holomorphic Chern-Simons factorisation algebra, a compatible Hall
integration map, fixed orientation data, and a completion compatible with
the CY4 charge cone.  Then the associated centre or factorisation-module
layer carries an $\Etwo$ structure.  The specialised $\Phi_d$ rule still
assigns an $\Eone$ output for $d\geq 3$.
\end{conjecture}

\begin{remark}[Compact CY4]
For compact examples such as $K3\times K3$, an $\Etwo$ conclusion
requires a theorem identifying the relevant Hall category with a
centre-layer or tensor-decomposed factorisation module.  The local CY4
Jacobi statement and the compact hyperkahler statement have different
input categories and different completion problems.
\end{remark}
