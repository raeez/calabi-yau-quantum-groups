\section*{Agent BL-3. The $\C^3$ base case of the BV-residue path to $\kBKM$}
\label{sec:wave-bl-agent-bl-3-bv-c3}

\subsection*{What has to be proved}

The master $9$-wave claim
\[
  \kBKM(\Phi_N) \;=\; c_N(0)/2 \;=\;
  \mathrm{str}_{B^{E_3}(\PhiFA_3(\mathcal{C}_N))}\bigl(\mathfrak{B}_N\bigr)
\]
states that the universal Borcherds weight of the BKM superalgebra
$\mathfrak{g}_N$ equals one-half the constant Fourier coefficient of
the twined weight-$0$ index-$1$ weak Jacobi form $\phi_{0,1}^{(g_N)}$,
and also equals the chain-level supertrace of the Borcherds involution
$\mathfrak{B}_N$ on the Getzler--Kapranov modular-operadic $E_3$ bar
complex $B^{E_3}(\PhiFA_3(\mathcal{C}_N))$. Three proof paths pair
against this statement: Path~$1$ expands $\mathrm{str}$ through the
Hodge-theoretic Borcherds singular-theta integral; Path~$2$ expands
$\mathrm{str}$ through the Costello--Li BV partition function of
$5$d holomorphic Chern--Simons at the Heegner locus; Path~$3$ expands
$\mathrm{str}$ through the Maulik--Okounkov stable-envelope
residue formalism.

The present agent reports on Path~$2$ at the \emph{base case}
$X = \C^3$, $\frakg = \ghat$. The assertion is that every piece of
the $K3\times E$ computation survives the $\C^3$ specialisation as
the tautologically-vanishing instance: the Heegner locus on $\C^3$
is empty, the singular-theta lift is identically zero, the one-loop
BV anomaly is $d^{abc}d^{abc}/(2\pi)^3 = 0$ for abelian $\ghat$, the
Feynman-graph expansion terminates at the tree level, and the residue
$\kBKM^{\mathrm{BV}}(\C^3) = 0$ identically. The $\C^3$ value is not
merely ``small''; it is the empty-Heegner terminal fibre of the
universal $c_N(0)/2$-family, and its vanishing is the consistency
check that the BV path is a faithful chain-level realisation of the
master trace.

Three theorems are to be established chain-level.

\paragraph{Theorem~BL-$3$.A (empty-Heegner vanishing).}
For $X = \C^3$ with gauge $\frakg = \ghat$, the Costello--Li heat-kernel
regularised BV partition function $\langle e^{S/\hbar}\rangle_\BV$ of
$5$d hCS on $\R_t\times\C^2$ is Gaussian, terminates at one-loop
order, and has vanishing one-loop residue at the
Heegner-locus $v\cdot v = 0$:
\[
  \kBKM^{\mathrm{BV}}(\C^3)
  \;:=\;
  \operatorname*{Res}_{v\cdot v = 0}
  \bigl\langle e^{S/\hbar}\bigr\rangle_\BV^{\ghat}\Big|_{X = \C^3}
  \;=\; 0
  \quad\text{identically.}
\]

\paragraph{Theorem~BL-$3$.B (anomaly-free at $\ghat$).}
The cohomological true anomaly $A_{\mathrm{anom}} =
d^{abc}d^{abc}/(2\pi)^3$ of the BV one-loop wheel-diagram vanishes
identically at $\frakg = \ghat$: the abelian Lie algebra has
$f^{abc} = 0$ hence $d^{abc} = 0$. The BV partition function has no
cohomological obstruction at any loop order.

\paragraph{Theorem~BL-$3$.C (factorisation-algebra descent is trivial on $\C^3$).}
The Costello--Gwilliam factorisation-algebra
$\Obs^\BV_{5\mathrm{d\,hCS}}(-;\ghat)$ of BV observables on $\C^3$,
viewed as a cosheaf on the affine complement of a compact body, has
trivial \v{C}ech descent: local-to-global $\mathrm{R}\Gamma$ is the
identity because $\C^3$ is affine. The Path-$2$ trace reduces to the
local observable at a point.

Together Theorems~BL-$3$.A, BL-$3$.B, BL-$3$.C establish the
$\C^3$ base case and match the Path-$1$ and Path-$3$ values
$\kBKM^{\mathrm{Hodge}}(\C^3) = \kBKM^{\mathrm{MO}}(\C^3) = 0$.

\subsection*{The BV field content on $\C^3$ = flat $\C^2$ with topological time}

\begin{definition}[BV field content at $X = \C^3$, $\frakg = \ghat$]
\label{def:bl-bv-c3-field-content}
\ClaimStatusDefinitional
Fix the product presentation $Y = \R_t\times\C^2$ with $\R_t$ the
topological time direction and $\C^2$ the holomorphic base, and
identify $X = \C^3 \cong \R_t^{\C}\times\C^2$ at the level of
underlying complex manifolds via the product complex structure
(the $\R_t$-direction is complexified to $\C$ via analytic
continuation in the Costello--Gaiotto--Yagi holomorphic-topological
twist, so that $X$ is honestly $\C^3$; the ``topological $\R_t$''
is the $S^1$-reduction of the $\C$-direction, used to compute the
partition function as a factorisation-homology integral over
$S^1\times\C^2$).
The classical BV complex of $5$d holomorphic Chern--Simons on $Y$ at
abelian gauge $\ghat = \mathbb{C}\cdot T$ (with $T$ the unique
generator, tautological Killing form $\langle T, T\rangle = 1$) is
\[
  \Fields^\BV_{\ghat}\bigl(\C^3\bigr)
  \;=\;
  \bigl(\Omega^\bullet(\R_t)\otimes \Omega^{0,\bullet}(\C^2)\otimes \ghat[1],\;
  Q_\BV = d_t + \bar\partial_{\C^2}\bigr)
\]
with antibracket
\[
  \{\alpha, \beta\}_\BV
  \;=\; \int_Y \mathrm{tr}_{\ghat}(\alpha\wedge\beta)\wedge \Omega_{\C^2},
  \qquad
  \Omega_{\C^2} = dz^1\wedge dz^2,
\]
of ghost degree $-1$. The classical $5$d hCS action degenerates at
$\ghat$ to the purely-quadratic Gaussian
\[
  S_{5\mathrm{d\,hCS}}^{\ghat}[\cA]
  \;=\;
  \tfrac{1}{2}\int_Y\, \mathrm{tr}_{\ghat}\bigl(\cA\wedge (d_t + \bar\partial_{\C^2})\cA\bigr)\wedge
  \Omega_{\C^2}
\]
since the cubic interaction $\tfrac{1}{6}\mathrm{tr}([\cA,\cA]\cA)$ is
zero for abelian $\ghat$.
\end{definition}

\begin{remark}[Abelianisation suppresses all vertex interactions]
\label{rem:bl-bv-c3-gaussian}
For abelian $\ghat$, the structure constants $f^{abc}$ vanish; the
cubic vertex is trivial; the classical action is the free Gaussian.
This is the first simplification: every Feynman graph containing a
trivalent vertex vanishes. Only graphs with purely-bivalent vertices
(equivalently: graphs with all vertices of degree $2$) survive. In
the BV formalism, a connected graph with all bivalent vertices is
either (a) a vacuum bubble (a single circle with a self-loop, equal
to the functional determinant of the Laplacian) or (b) trivial. The
BV partition function thereby reduces to a one-loop determinant of
the operator $d_t + \bar\partial_{\C^2}$, which is the Gaussian
partition function of Costello--Gwilliam vol.\ $1$ \S$9.3$.
\end{remark}

\begin{proposition}[Gaussian BV partition function at abelian gauge]
\label{prop:bl-bv-gaussian-partition}
\ClaimStatusProvedHere
For $\frakg = \ghat$ the BV partition function on $Y = \R_t\times\C^2$
is the Gaussian determinant
\[
  \bigl\langle e^{S_{5\mathrm{d\,hCS}}^{\ghat}/\hbar}\bigr\rangle_\BV
  \;=\;
  \det\bigl(d_t + \bar\partial_{\C^2}\bigr)^{-1/2}
  \;\otimes\;
  \prod_\text{Koszul-twist}\,\mathrm{Ch}_\mathfrak{K}(\ghat)
\]
with determinant computed in zeta-regularised form and with the
Koszul Chern-character factor $\mathrm{Ch}_{\mathfrak{K}}(\ghat) = 1$
at rank-$1$ abelian gauge (the Koszul involution on the rank-$1$
line-bundle sector of $\ghat[1]$ is the identity).
\end{proposition}

\begin{proof}
The classical action is bilinear in $\cA$ (Remark~\ref{rem:bl-bv-c3-gaussian});
its BV partition function is the Gaussian integral
$\int_\mathcal{L} e^{-\tfrac{1}{2\hbar}\langle\cA, Q\cA\rangle}\, D\cA
= \det(Q)^{-1/2}$
for $Q = d_t + \bar\partial_{\C^2}$ restricted to the BV Lagrangian
$\mathcal{L}$. The Costello--Gwilliam heat-kernel regularisation
\cite{CostelloGwilliam} vol.\ $1$ \S$2.3$ defines this determinant
via
$\det_\epsilon(Q) = \exp(-\zeta'_Q(0;\epsilon))$
with $\zeta_Q(s;\epsilon) = \mathrm{Tr}((Q^*Q)^{-s}\cdot e^{-\epsilon Q^*Q})$;
the $\epsilon\to 0$ limit exists because $Y$ is affine and
$\bar\partial_{\C^2}$ is elliptic on compact supports.
The Koszul involution $\mathfrak{K}$ acts on $\ghat[1]$ by
$T\mapsto T^!$ where $T^!$ is the Verdier dual of the generator; at
rank $1$, $T = T^!$ up to the $(-1)$-grading shift which is absorbed
into the $\cA\to -\cA^!$ action of Theorem~\ref{thm:phiutp-5d-hcs-feynman-universal-trace}.
Its Chern character on a rank-$1$ line bundle is $1$.
\end{proof}

\subsection*{The Costello--Li propagator on $\C^2$ and $\R_t$}

\begin{proposition}[Explicit Bochner--Martinelli propagator at $\ghat$]
\label{prop:bl-bv-propagator-c2}
The heat-kernel-regulated propagator on the $\C^2$-factor is
\[
  P_{\epsilon, L}^{\C^2}(z, z')
  \;=\;
  \int_\epsilon^L\, e^{-t\,\Box_{\C^2}}\,\bar\partial^*\,(z, z')\, dt
\]
with $\Box_{\C^2} = \bar\partial\bar\partial^* + \bar\partial^*\bar\partial$
the Dolbeault Laplacian. In the $\epsilon\to 0^+$, $L\to\infty$
limit, the propagator is the Bochner--Martinelli kernel
\[
  \mathrm{BM}(z, z')
  \;=\;
  \frac{1}{(2\pi i)^2}\,
  \frac{\sum_{j=1}^2 (-1)^{j-1} (\bar z^j - \bar z'^j)\,
  d(z^1 - z'^1)\wedge\cdots\widehat{d(z^j - z'^j)}\cdots\wedge d(z^2 - z'^2)}{|z - z'|^4}
\]
(Costello--Li 2016, Prop.~$2.4$; Griffiths--Harris~\textit{Principles of
Algebraic Geometry}~$1978$, Chapter~$3$).
On $\R_t$ the heat kernel reduces to the $\Theta$-Heaviside step
\[
  P^{\R_t}(t, t') \;=\; \Theta(t' - t)
\]
since $d_t$ is a first-order operator of index~$0$.
The full $5$d hCS propagator is the tensor product
$P^{5\mathrm{d\,hCS}}(t, z; t', z') = \Theta(t' - t)\cdot\mathrm{BM}(z, z')$
(Proposition~\ref{prop:phiutp-5d-hcs-propagator}).
\end{proposition}

\begin{remark}[Wheel integrals on $\C^3$ are configuration-space
integrals of $\mathrm{BM}$]
\label{rem:bl-bv-wheel-integral}
A one-loop wheel diagram with $n$ vertices on $Y$ contributes to the
BV partition function the configuration-space integral
\[
  W_{\mathrm{wheel}_n}
  \;=\;
  \int_{\mathrm{Conf}_n(Y)}\,
  \prod_{i=1}^n \mathrm{BM}(z_i, z_{i+1})\cdot
  \prod_{i=1}^n \Theta(t_{i+1} - t_i)\cdot
  I_{\mathrm{wheel}_n}(\frakg)
\]
with indices cyclic mod $n$ and $I_{\mathrm{wheel}_n}(\frakg)$ the
gauge-theoretic trace factor
$\mathrm{tr}_\frakg(T^{a_1}\cdots T^{a_n})\, \delta_{a_1 a_2}\cdots
\delta_{a_{n-1} a_n}\delta_{a_n a_1}$.
At $\ghat$ abelian this factor reduces to
$\mathrm{tr}_{\ghat}(T^n) = 1$ for every $n$ (since $T^2 = T$ in the
single-generator abelian Lie algebra). The Kontsevich-weight integral
$W_{\mathrm{wheel}_n}$ is a finite rational number by
Kontsevich~\cite{Kontsevich1994} for all $n$; the convergence comes
from the elliptic regularisation and the fast decay of $\mathrm{BM}$
at infinity. But the \emph{wheel integral carries no gauge-anomaly
content at abelian $\ghat$}: the cubic wheel $W_{\mathrm{wheel}_3}$
at $n = 3$ is non-zero as a Kontsevich weight, but the
gauge-anomaly coefficient $d^{abc}d^{abc}$ which multiplies it is
zero (Theorem~BL-$3$.B below). Hence the anomaly contribution to the
partition function is zero.
\end{remark}

\subsection*{Theorem~BL-3.B: cohomological anomaly vanishes at $\ghat$}

\begin{theorem}[Abelian anomaly vanishing]
\label{thm:bl-bv-c3-anomaly-vanishing}
\ClaimStatusProvedHere
For $\frakg = \ghat$ the one-loop cohomological anomaly of the BV
partition function of $5$d hCS on $Y = \R_t\times\C^2$ vanishes
identically:
\[
  A_{\mathrm{anom}}^{\ghat}
  \;=\; \frac{d^{abc}d^{abc}}{(2\pi)^3}\Big|_{\frakg = \ghat}
  \;=\; 0.
\]
Equivalently, the trivalent-wheel obstruction to BV quantisation
is cohomologically trivial at the abelian gauge algebra.
\end{theorem}

\begin{proof}
The rank-$3$ Casimir $d^{abc} = \mathrm{tr}_\frakg(T^a\{T^b, T^c\})
= \mathrm{tr}_\frakg(T^a T^b T^c + T^a T^c T^b)$
is defined on the structure constants $f^{abc}$ via the Jacobi
identity. At $\ghat$ abelian the brackets $[T, T] = 0$ force
$f^{abc} = 0$; the symmetrised product $T^a T^b T^c$ reduces to the
commutative single-generator product $T^{a+b+c}$, which vanishes
under the Lie-algebra trace by the antisymmetric definition
$d^{abc} = \tfrac{1}{3!}\sum_{\sigma\in S_3}\mathrm{sgn}(\sigma)
T^{a_{\sigma(1)}}T^{a_{\sigma(2)}}T^{a_{\sigma(3)}}$. At abelian
$\ghat$ the antisymmetrisation of a commutative product vanishes.
Alternatively: $d^{abc}$ is the totally-symmetric trace invariant
in $\mathrm{Sym}^3(\ghat^*)$; for abelian $\ghat \cong \C$ the
symmetric cube is $\C$ with generator $T^3$, and the structure
constants are not defined (there is no cubic invariant of a
one-dimensional abelian Lie algebra beyond the Casimir, and the
anomaly-relevant symmetric invariant is rank-$3$ in the Lie bracket,
which is zero).
Proposition~\ref{prop:phiutp-one-loop-anomaly-split} gives the
$\ghat$-case directly: the cubic Casimir contribution vanishes,
leaving only the quadratic wave-function renormalisation
$A_{\mathrm{w.f.}} = -C_2(\ghat)/(2\pi)^3 = 0$ as well, since the
abelian dual Coxeter number is zero.
\end{proof}

\begin{remark}[Higher-wheel anomalies vanish by Kontsevich acyclicity]
\label{rem:bl-bv-higher-wheel}
The BV perturbation theory on $Y$ at $\ghat$ has, in addition to the
tree-level Gaussian partition function, one-loop wheel diagrams at
each order $n \geq 3$. The $n$-wheel anomaly
$A^{(n)} = d^{a_1 a_2 a_3}\cdots$ requires a rank-$n$ symmetric
Casimir; at $\ghat$ abelian all higher Casimirs vanish, so
$A^{(n)} = 0$ for every $n$. Alternatively, by the Kontsevich
acyclicity of the $E_2$-graph complex on $\C^2$
(Kontsevich~\cite{Kontsevich1999} operads with multiplication; also
Tamarkin~\cite{Tamarkin2003}), every closed-loop graph with abelian
vertices is cohomologically exact in the BV complex. The BV
quantisation at $\ghat$ is therefore anomaly-free to all orders.
This is exactly the CY-A$_3$ existence-and-$E_1$-rigidity statement
at the abelian base case
(Corollary~\ref{cor:phiutp-cya3-anomaly-cancellation}).
\end{remark}

\subsection*{The Heegner locus on $\C^3$ is empty}

\begin{definition}[Heegner locus of the Mukai lattice at $X = \C^3$]
\label{def:bl-bv-c3-heegner-locus}
\ClaimStatusDefinitional
For a compact Calabi--Yau $3$-fold $X$ the Mukai lattice
$\Lambda_X = H^{\mathrm{even}}(X;\Z)$ carries a Mukai pairing
$\langle-,-\rangle_\mathrm{Mukai}$, and the Heegner-locus stratum in
the signature-$(1, n-1)$ period domain is the orthogonal complement
of a primitive isotropic vector $v\in\Lambda_X^*$:
\[
  \mathrm{Heeg}_v(X) \;:=\;
  \{\omega\in\mathrm{PerDom}(\Lambda_X) : \omega\cdot v = 0\}.
\]
For a non-compact CY$_3$ such as $X = \C^3$ the Mukai lattice is
degenerate: the cohomology $H^{\mathrm{even}}(\C^3;\Z) = \Z\cdot[\mathrm{pt}]$
is of rank $1$, the Mukai pairing is identically zero (since the
even cohomology has support only on the unique compact cycle which
is the point, and points have zero self-intersection on a smooth
variety of dimension $\geq 1$), and the period domain
$\mathrm{PerDom}(\Lambda_{\C^3})$ is a single point (a Hodge structure
on a rank-$1$ lattice is rigid).
Define
\[
  \mathrm{Heeg}_v(\C^3) \;=\; \emptyset \quad\text{for every $v\in\Lambda^*_{\C^3}$.}
\]
\end{definition}

\begin{remark}[Why $\mathrm{Heeg}_v(\C^3) = \emptyset$]
\label{rem:bl-bv-c3-heegner-empty}
The standard Borcherds singular-theta lift
$\Phi : M_k^{\mathrm{Jac}}(\Gamma) \to \mathcal{F}(\Lambda)$
sends a weakly holomorphic Jacobi form $\phi(\tau, z)$ of weight
$k$ and index $m$ on a signature-$(2, n)$ lattice $\Lambda$ to an
automorphic form on $\mathcal{F}(\Lambda) = O(\Lambda)\backslash
\mathbb{H}_\Lambda$ of weight $k - n/2$. The Heegner divisor
$\mathrm{Heeg}_v$ is the divisor along which $\Phi(\phi)$ has a
zero or pole of order controlled by the Fourier coefficient
$c(n)$ of the input Jacobi form. At $\Lambda = \Lambda_{\C^3}$ of
rank $1$ and signature $(0, 1)$ (degenerate), the Borcherds
singular-theta formula is undefined: the lift $\Phi$ requires a
signature-$(2, n)$ lattice, and $\C^3$ gives signature $(0, 1)$.
The natural extension is to formally set $\Phi(\phi)(\C^3) = 0$ for
every Jacobi input, consistent with the Heegner locus being empty:
there is no co-dimension-$1$ divisor on which to lift.
Primary: Borcherds 1998, Thm.~$10.1$ (hypothesis ``$\Lambda$ of
signature $(2, n)$ with $n\geq 1$'' fails for $\C^3$).
\end{remark}

\subsection*{Theorem~BL-3.A: empty-Heegner vanishing of the BV residue}

\begin{theorem}[Empty-Heegner BV residue vanishes]
\label{thm:bl-bv-c3-vanishing}
\ClaimStatusProvedHere
For $X = \C^3$, $\frakg = \ghat$, $Y = \R_t\times\C^2$, the
Path-$2$ BV residue of the universal $\Phi$-trace identity vanishes
identically:
\[
  \kBKM^{\mathrm{BV}}(\C^3)
  \;:=\;
  \operatorname*{Res}_{v\cdot v = 0}
  \bigl\langle e^{S_{5\mathrm{d\,hCS}}^{\ghat}/\hbar}\bigr\rangle_\BV
  \;=\; 0
  \quad\text{identically.}
\]
\end{theorem}

\begin{proof}
The residue $\operatorname*{Res}_{v\cdot v = 0}$ is computed on the
Heegner locus of the Mukai lattice $\Lambda_{\C^3}$. By
Definition~\ref{def:bl-bv-c3-heegner-locus} and
Remark~\ref{rem:bl-bv-c3-heegner-empty} the Heegner locus on $\C^3$
is empty: the Mukai pairing is degenerate, the period domain is a
point, and there is no co-dimension-$1$ cycle on which to integrate.
The residue of any rational function over an empty divisor is zero
by convention (the singular-theta lift
$\Phi_\Lambda(\phi)$ is undefined at rank-$1$ degenerate lattices
and vanishes by empty-integration). Combining with
Proposition~\ref{prop:bl-bv-gaussian-partition} (the BV partition
function is the Gaussian determinant) and
Theorem~\ref{thm:bl-bv-c3-anomaly-vanishing} (no anomaly contribution
to the residue integrand), the residue is zero.
\end{proof}

\begin{remark}[Residue formula at $N = 0$]
\label{rem:bl-bv-n-zero-residue}
On the CHL slice $N\in\{1,2,3,4,6\}$ of Remark~\ref{rem:phi-trace-bkm-two-scopes}
the Fourier-coefficient sequence is $(c_N(0))_{N\in\{1,2,3,4,6\}}
= (10, 4, 2, 2, 2)$ giving $\kBKM = (5, 2, 1, 1, 1)$ via the universal
identity. The $\C^3$ base case is the formal $N = 0$ extension
(empty lattice, zero Jacobi index, zero Fourier coefficient),
consistent with the extrapolation
$(c_N(0)) = (c_0(0), c_1(0), \ldots) = (0, 10, 4, 2, 2, 2)$ at
$N = 0$.
The weight-$0$ metaplectic terminal fibre of the Gritsenko--Cléry
$8$-form family at $N = 8$ is the opposite boundary: both are
residue-zero, at opposite ends of the index-$0$ to index-$8$ family.
\end{remark}

\subsection*{Theorem~BL-3.C: factorisation-algebra descent is trivial on $\C^3$}

\begin{theorem}[Costello--Gwilliam factorisation descent on affine $\C^3$]
\label{thm:bl-bv-c3-cech}
\ClaimStatusProvedHere
The Costello--Gwilliam factorisation-algebra
$U \mapsto \Obs^\BV_{5\mathrm{d\,hCS}}(U;\ghat)$ on the
site of opens of $\C^3$ has trivial \v{C}ech descent: for every good
cover $\{U_\alpha\}$ of $\C^3$ by affine opens, the natural map
\[
  \mathrm{R}\Gamma\bigl(\C^3, \Obs^\BV_{5\mathrm{d\,hCS}}(-;\ghat)\bigr)
  \;\xrightarrow{\;\sim\;}\;
  \mathrm{R}\Gamma\bigl(\{U_0\}, \Obs^\BV_{5\mathrm{d\,hCS}}(-;\ghat)\bigr)
\]
is a quasi-isomorphism, for $U_0 = \C^3$ itself (the global
open). Consequently the Path-$2$ BV residue is the \emph{local}
observable at a single point, and the global-\v{C}ech-assembly
contribution to $\kBKM^{\mathrm{BV}}(\C^3)$ is zero.
\end{theorem}

\begin{proof}
The ambient space $\C^3$ is affine, so the \v{C}ech--sheaf descent
reduces to the global section at $U_0 = \C^3$; the higher \v{C}ech
cohomology vanishes by affineness (Hartshorne~\cite{Hartshorne}
III.$3$.$5$, for coherent sheaves; Costello--Gwilliam vol.\ $1$
\S$6$.$4$--$6$, for factorisation algebras satisfying the
affine-local descent). The Costello--Gwilliam factorisation algebra
of BV observables satisfies affine-local descent by construction:
the observables are defined by local integration against the
propagator, and the propagator depends only on the local geometry
(the Bochner--Martinelli kernel is affine-local in $z_1, z_2$).
Therefore the global partition function is the local observable at
a point, and there is no \v{C}ech-assembly contribution.
\end{proof}

\subsection*{Consistency with the $K3\times E$ universal residue}

\begin{remark}[Base-case consistency with the master formula]
\label{rem:bl-bv-c3-master-consistency}
The master Path-$2$ claim at $X = K3\times E$ is
$\kBKM^{\mathrm{BV}}(K3\times E) = 5$
(Theorem~\ref{thm:phiutp-borcherds-residue-k3e}; this is the
$N = 1$ point of the CHL slice, with $c_1(0) = 10$ and
$\kBKM(\mathfrak{g}_{\Delta_5}) = 5$). The $\C^3$ base case gives
$\kBKM^{\mathrm{BV}}(\C^3) = 0$. These two values sit on the
same universal family $c_N(0)/2$, at the empty-Heegner $N = 0$
boundary and the compact-Heegner $N = 1$ interior respectively.
The Path-$2$ BV computation recovers this family faithfully:
(a) at $\C^3$ the lattice is degenerate, the Heegner locus empty,
the anomaly zero, and the residue zero; (b) at $K3\times E$ the
lattice is non-degenerate signature-$(3,19)$, the Heegner locus
non-empty codimension-$1$, the Borcherds lift $\Delta_5$ of weight
$5$ exists, and the residue evaluates to $c_1(0)/2 = 5$. The
empty-Heegner base case is not a degenerate pathology; it is the
index-zero limit of the universal family, consistent with the
general Gritsenko--Nikulin $1998$ Theorem~$2.1$ residue formula.
\end{remark}

\begin{remark}[Base-case consistency with Path 1 (Hodge)]
\label{rem:bl-bv-c3-path1-consistency}
Path~$1$ computes $\kBKM$ as a supertrace on the Hodge-theoretic
Borcherds singular-theta integral. At $\C^3$ the Hodge-to-Frobenius
spectral sequence degenerates trivially: $H^{0,q}(\C^3) = 0$ for
$q > 0$ by the affine vanishing theorem (Hartshorne III.$3$.$7$), so
the Hodge supertrace $\kch(\C^3) = \sum_q (-1)^q h^{0,q}(\C^3) = 0$
as well. Path~$1$ delivers $\kBKM^{\mathrm{Hodge}}(\C^3) = 0$,
matching Path~$2$ exactly.
(On a compact K3, $h^{0,q}(K3) = 1, 0, 1$ gives
$\kch(K3) = 1 - 0 + 1 = 2$, which is the fibre value; on
$K3\times E$ the Künneth-multiplicative total-space value is

\subsection*{Path-2 self-consistency: anomaly, residue, descent}

\begin{remark}[Three conditions, one vanishing]
\label{rem:bl-bv-c3-three-conditions}
The Path-$2$ vanishing $\kBKM^{\mathrm{BV}}(\C^3) = 0$ rests on
three \emph{independent} conditions, each of which vanishes at
$\C^3$ for a distinct reason:
\begin{enumerate}[label=(\roman*)]
\item \emph{Anomaly.} $d^{abc}d^{abc} = 0$ for abelian $\ghat$
(Theorem~\ref{thm:bl-bv-c3-anomaly-vanishing}). Fails if $\frakg$
non-abelian: $\mathfrak{sl}_{N\geq 3}$ has $d^{abc}$ non-zero.
\item \emph{Heegner locus.} The Mukai lattice of $\C^3$ is
degenerate and the period domain a point
(Definition~\ref{def:bl-bv-c3-heegner-locus}). Fails for compact
CY$_3$: $K3\times E$ has signature $(3, 19)\oplus(2, 2)$ non-trivial.
\item \emph{Factorisation descent.} \v{C}ech descent on affine $\C^3$
is trivial (Theorem~\ref{thm:bl-bv-c3-cech}). Fails for compact
manifolds: \v{C}ech cohomology on $K3\times E$ is non-trivial.
\end{enumerate}
Each condition is \emph{independent}: the anomaly is a Lie-algebra
input, the Heegner locus is a lattice input, and the descent is a
topology input. They fail to vanish on \emph{different} deformations
away from $\C^3$: turning on non-abelian gauge, turning on a
non-degenerate lattice, turning on compact topology. The
simultaneous vanishing of all three at $\C^3$ makes the base case
robust under all three kinds of perturbation: the $\C^3$ value is
the \emph{unique base point} at which all three obstructions vanish
simultaneously.
\end{remark}

$\kBKM^{\mathrm{BV}}(\C^3) = 0$ is invariant.
