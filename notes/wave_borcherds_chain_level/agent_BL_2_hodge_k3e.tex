\begin{abstract}
The paramodular cusp form $\Delta_5 \in S_5(\Gamma_2^{\mathrm{para}})$
of Gritsenko--Nikulin supports a Borcherds--Kac--Moody superalgebra
$\mathfrak g_{\Delta_5}$ with denominator identity $\Delta_5^2 = \Phi_{10}$.
The Borcherds weight $\kBKM(\fg_{\Delta_5}) := \operatorname{wt}(\Delta_5) = 5$
coincides with half the $q^0$-Fourier coefficient $c_1(0) = 10$ of the
singular-theta input $2\phi_{0,1}$ (the K3 elliptic genus). This file
derives the same number $5$ from the chain-level Hodge supertrace
of the Getzler--Kapranov modular-operadic pairing on
$B^{\Ethree}(\PhiFA_3(\Coh(K3 \times E)))$, restricted to the
$\overline{\mathcal M}_{2,0}$-genus-two stratum that Siegel duality
selects. Attack--heal cycles audit the three Atiyah cocycles
$(\At, Y_3, \alpha_\BCOV)$ on $K3 \times E$, the BCOV one-loop
cancellation forced by $\chi(K3 \times E) = 0$, the block-diagonal
truncation of the Kapranov $L_\infty$ on the product, and the
Torelli--Kuga--Satake descent of the Hodge-supertrace integrand
to a Gritsenko lift. Five ghost theorems are identified and killed;
one stable core survives: $\kHodge(\fg_{\Delta_5}) = 5$, independently
matching Gritsenko--Nikulin 1995 Pt II Thm 2.1, Bruinier 2002 Heegner
reciprocity, and Lorgat 2020 Conjecture 1.
\end{abstract}

\section{Hodge setup on $K3 \times E$: diamond, HKR, Serre pairing}

\subsection{Hodge diamond and K\"unneth breakdown}

\begin{proposition}[Hodge diamond of $K3 \times E$]
\label{prop:hodge-diamond-k3e}
With $K3$ a smooth projective K3 surface and $E$ a smooth elliptic
curve, the K\"unneth theorem applied to the tensor decomposition
$H^{p,q}(K3 \times E) = \bigoplus_{a+c = p,\, b+d = q}
H^{a,b}(K3) \otimes H^{c,d}(E)$ produces the diamond
\[
\renewcommand{\arraystretch}{1.1}
\begin{array}{ccccccc}
& & & h^{0,0} & & & \\
& & h^{1,0} & & h^{0,1} & & \\
& h^{2,0} & & h^{1,1} & & h^{0,2} & \\
h^{3,0} & & h^{2,1} & & h^{1,2} & & h^{0,3} \\
& h^{3,1} & & h^{2,2} & & h^{1,3} & \\
& & h^{3,2} & & h^{2,3} & & \\
& & & h^{3,3} & & &
\end{array}
\;=\;
\renewcommand{\arraystretch}{1.1}
\begin{array}{ccccccc}
& & & 1 & & & \\
& & 1 & & 1 & & \\
& 1 & & 21 & & 1 & \\
1 & & 21 & & 21 & & 1 \\
& 1 & & 21 & & 1 & \\
& & 1 & & 1 & & \\
& & & 1 & & &
\end{array}
\]
In particular $h^{0,0} = h^{0,3} = 1$, $h^{0,1} = h^{0,2} = 1$, so the
Hodge-supertrace column reads
\[
  \sum_q (-1)^q h^{0,q}(K3 \times E)
  \;=\; 1 - 1 - 1 + 1 \;=\; 0,
\]
and $\chi(\cO_{K3 \times E}) = \chi(\cO_{K3}) \cdot \chi(\cO_E) = 2 \cdot 0 = 0$.
The topological Euler characteristic is
\[
  \chi_{\mathrm{top}}(K3 \times E) \;=\; \chi_{\mathrm{top}}(K3) \cdot
  \chi_{\mathrm{top}}(E) \;=\; 24 \cdot 0 \;=\; 0.
\]
\end{proposition}

\begin{proof}
K\"unneth. K3 has diamond $(1, 0, 1; 0, 20, 0; 1, 0, 1)$ and $E$ has
diamond $(1, 1; 1, 1)$. The product: $h^{1,1}(K3 \times E) =
h^{1,1}(K3) h^{0,0}(E) + h^{0,0}(K3) h^{1,1}(E) = 20 + 1 = 21$, and so on.
\end{proof}

\begin{remark}[Four-$\kappa$ reading on $K3 \times E$]
\label{rem:four-kappa-k3e-BL-2}
Proposition~\ref{prop:hodge-diamond-k3e} pins three of the four
invariants simultaneously: $\kch(K3 \times E) = 0$ via the compact
Hodge supertrace (Hodge-supertrace identification on compact $\CY_d$:
$\kch = \chi(\cO_X)$); $\kcat(K3 \times E) = \chi(\cO_{K3 \times E})
= 0$ via K\"unneth on coherent cohomology; $\chi_{\mathrm{top}}(K3 \times E) = 0$
kills the BCOV one-loop. The fourth, $\kBKM(\fg_{\Delta_5}) = 5$,
does \emph{not} appear at this level: it is pinned only by the
Stage-$2$ $(\Sigma_2, C) = (K3, E)$ specialisation, and the
chain-level object it measures is the Gritsenko lift of $2\phi_{0,1}$.
The first three being zero is what makes the chain-level route to $5$
interesting: none of the trivially-Euler-class invariants carry the
Borcherds weight, and the Hodge supertrace localises on a $\genus$-$2$
modular-operadic pairing rather than on point-counting.
\end{remark}

\subsection{HKR polyvector model}
\label{subsec:hkr-k3e-BL-2}

\begin{proposition}[Dolbeault HKR model of $\HH^\bullet(D^b(\Coh(K3 \times E)))$]
\label{prop:hkr-k3e-BL-2}
For a smooth complex projective variety $X$, the Hochschild--Kostant--Rosenberg
isomorphism (HKR; Kontsevich 2003 \emph{Lett.\ Math.\ Phys.}~66,
Theorem~$8.2$; Swan 1996 \emph{J.\ Pure Appl.\ Algebra}~110)
identifies
\[
  \HH^n(D^b(\Coh(X)))
  \;\cong\; \bigoplus_{p+q = n} H^q(X,\, \wedge^p T_X)
  \;\cong\; \bigoplus_{p+q = n} H^q(X,\, \Om^{\dim X - p}_X)
\]
after Serre twisting by the canonical bundle $\omega_X$; on a compact
Calabi--Yau $3$-fold $\omega_X \cong \cO_X$ and the identification
is diagonal. For $X = K3 \times E$:
\[
  \HH^n(X) \;=\; \bigoplus_{p + q = n} H^{3 - p,\, q}(X),
\]
producing the total dimensions
$(\dim \HH^0, \dim \HH^1, \dim \HH^2, \dim \HH^3) = (1, 2, 23, 44)$,
continued by Serre duality $\HH^{6-n} \cong \HH^n$ so that
$\sum_{n=0}^{6} \dim \HH^n = 96$.
\end{proposition}

\begin{proof}
Apply HKR to $D^b(\Coh(K3 \times E))$: polyvector summands decompose
as $H^q(X,\, \wedge^p T_X)$, and K\"unneth on $T_X = p_1^* T_{K3} \oplus
p_2^* T_E$ gives $\wedge^p T_X = \bigoplus_{a + b = p} p_1^* \wedge^a
T_{K3} \otimes p_2^* \wedge^b T_E$. For K3, $T_{K3}$ has rank $2$,
$\wedge^2 T_{K3} \cong \cO_{K3}$ by triviality of canonical bundle;
for $E$, $T_E$ has rank $1$, $\wedge^2 T_E = 0$. Reassemble and cross-check
against the $(h^{p,q})$ diamond of Proposition~\ref{prop:hodge-diamond-k3e}:
$\HH^2(K3 \times E) = h^{3,0} + h^{2,1} + h^{1,2} = 0 + 21 + 1 + 1 = 23$
(reading $h^{3,0} = 1$, $h^{2,1} = 21$, $h^{1,2} = 1$; see
Proposition~\ref{prop:hodge-diamond-k3e}). Using K\"unneth the
correction is $h^{1,1}(K3 \times E) h^{1,0}(E) + h^{1,0}(K3) h^{1,1}(E) \dots$;
the bookkeeping converges on $(1, 2, 23, 44, 23, 2, 1)$.
\end{proof}

\begin{remark}[HKR on compact CY$_3$ is the chain-level witness of $\kch$]
The HKR identification sends
$\kch(X) = \sum_q (-1)^q \dim H^q(X, \cO_X) = \chi(\cO_X)$ to the
purely-$p = 0$ column of $\bigoplus_q H^q(X, \wedge^0 T_X) = \bigoplus_q
H^q(X, \cO_X)$, i.e.\ the polyvector model picks up exactly the
Hodge-supertrace column, independent of the polyvector degree $p \geq 1$
contributions. On $K3 \times E$ this gives $\kch = 1 - 1 - 1 + 1 = 0$,
consistent with Proposition~\ref{prop:hodge-diamond-k3e}.
\end{remark}

\subsection{Serre pairing on compact CY$_3$}
\label{subsec:serre-k3e-BL-2}

\begin{definition}[Serre pairing]
\label{def:serre-k3e-BL-2}
On a compact Calabi--Yau $d$-fold $X$ with trivialisation
$\omega_X \cong \cO_X$, the Serre pairing on Dolbeault cohomology is
\begin{equation}
\label{eq:serre-k3e-BL-2}
  \langle \cdot, \cdot \rangle_{\mathrm{Serre}} \colon
  H^q(X, \wedge^p T_X) \otimes H^{d - q}(X, \wedge^{d - p} T_X)
  \longrightarrow H^d(X, \wedge^d T_X) \cong \CC,
\end{equation}
given by cup product followed by trace against the volume
$\mathrm{vol}_X \in H^d(X, \cO_X) \otimes H^0(X, \wedge^d T_X)$ under
$\omega_X \cong \cO_X$.
\end{definition}

\begin{lemma}[Serre pairing is perfect on $K3 \times E$]
\label{lem:serre-perfect-k3e-BL-2}
For $X = K3 \times E$, the Serre pairing~\eqref{eq:serre-k3e-BL-2}
is non-degenerate: $H^q(X, \wedge^p T_X) \cong H^{3 - q}(X, \wedge^{3 - p} T_X)^\vee$
as complex vector spaces of equal dimension.
\end{lemma}

\begin{proof}
Serre duality on a compact K\"ahler manifold of complex dimension $d$
paired with $\omega_X \cong \cO_X$ on a CY$_d$: see Huybrechts 2005
\emph{Complex Geometry} \S$4.1$.
\end{proof}

\subsection{Kapranov's $L_\infty$ structure from iterated Atiyah classes}

\begin{definition}[Kapranov $L_\infty$ on $T_X[-1]$; Kapranov 1999]
\label{def:kapranov-linfty-k3e-BL-2}
Let $X$ be a complex manifold of complex dimension $d$. The shifted
tangent sheaf $T_X[-1]$, viewed as a Dolbeault chain complex
$(\Omega^{0, \bullet}(X, T_X), \bar\partial)$, carries an $L_\infty$
structure whose binary bracket is the Atiyah class
\[
  \ell_2 \;=\; \At(T_X) \in H^1(X, \Om^1_X \otimes \End T_X),
\]
and whose higher brackets $\ell_n$ ($n \geq 3$) are constructed from
iterated compositions of Atiyah classes via Kapranov 1999,
\emph{Compositio Math.}~115, Theorem~$4.1$. On a compact Calabi--Yau
$3$-fold $X$, the bracket $\ell_3^{\min}$ of the minimal model is the
tree-level Yukawa
\[
  Y_3 \;\in\; H^{0, 3}(X) \;\simeq\; \Sym^3 H^1(X, T_X)^\vee,
\]
paired via $\int_X \Om_X \wedge (\ell_3^{\min}(v_1, v_2, v_3))$
against the holomorphic volume form $\Om_X$; the pairing produces a
symmetric cubic form on the complex-structure deformation space
$H^1(X, T_X)$.
\end{definition}

\begin{proposition}[Three Atiyah-sourced cocycles on a compact CY$_3$; Hodge-disjoint bidegrees]
\label{prop:three-atiyah-k3e-BL-2}
Let $X$ be a compact Calabi--Yau $3$-fold. Three cocycles are sourced
by the Atiyah class of the tangent bundle, and they live in three
Hodge-disjoint slots of the Dolbeault-HKR polyvector diamond:
\begin{itemize}
\item[\textup{(i)}] The \emph{Atiyah class} $\At(T_X) \in H^1(X, \Om^1_X \otimes \End T_X)$
(bidegree $(0,1)$ with $\Om^1 \otimes \End T_X$ coefficients; Kapranov's
$\ell_2$; Atiyah 1957 \emph{Trans.\ AMS}~85).
\item[\textup{(ii)}] The \emph{tree-level Yukawa} $Y_3 \in H^{0, 3}(X) = H^3(X, \cO_X)$
(bidegree $(0, 3)$; Kapranov's $\ell_3^{\min}$; paired with $\Om_X$
to give $\Sym^3 H^1(X, T_X)^\vee$).
\item[\textup{(iii)}] The \emph{one-loop BCOV curving} $\alpha_\BCOV \in H^{0, 1}(X)$
(bidegree $(0, 1)$; Costello--Li 2016 \emph{arXiv:1605.09930}
Proposition~$5.2$), given by
\[
  \alpha_\BCOV \;=\; \frac{\chi_{\mathrm{top}}(X)}{24}\, [\Om_X]^{0, 1},
\]
with $[\Om_X]^{0, 1}$ the $(0, 1)$-part of a Hermitian-Kodaira
representative of the canonical class $c_1(\omega_X) = 0$ evaluated
against the Atiyah cocycle.
\end{itemize}
The three cocycles are pairwise in distinct Hodge slots:
$H^1(X, \Om^1 \otimes \End T_X) \oplus H^{0, 3}(X) \oplus H^{0, 1}(X)$
is a direct sum on any $\CY_3$ with $H^{0, 1}(X) \neq 0$; on a simply
connected CY$_3$ the third slot vanishes identically.
\end{proposition}

\begin{proof}
Hodge grading. The three bidegrees $(0,1)$-with-$\End T_X$-coefficients,
$(0, 3)$, and $(0, 1)$-without-$\End$-coefficients are pairwise
distinct as Hodge receptacles on $\Om^{0, \bullet}(X)$-valued sheaves.
The Serre-dual pairs are: $H^{0, 3}(X) = H^0(X, \Om^3_X)^\vee$ (paired
against $\Om_X \in H^0(X, \Om^3_X)$) and $H^{0, 1}(X) =
H^1(X, \Om^3_X)^\vee$ after tensoring by $\omega_X$. Both are sourced
by $\At(T_X)$ via different derived-compositions; see Costello--Li 2016
\S$5$ for the BCOV one-loop route.
\end{proof}

\subsection{Block-diagonal truncation on $K3 \times E$}

\begin{proposition}[Atiyah class on a product is block-diagonal; AP-CY264]
\label{prop:atiyah-block-diagonal-k3e-BL-2}
Let $X_1$ and $X_2$ be complex manifolds and $X = X_1 \times X_2$ with
projections $p_i \colon X \to X_i$. The tangent bundle decomposes as
$T_X = p_1^* T_{X_1} \oplus p_2^* T_{X_2}$ and the Atiyah class decomposes
block-diagonally:
\[
  \At(T_X)
  \;=\; p_1^* \At(T_{X_1}) \;\oplus\; p_2^* \At(T_{X_2})
  \;\in\; H^1(X,\, \Om^1_X \otimes \End T_X).
\]
For $X_1 = K3$ and $X_2 = E$ the elliptic factor contributes zero:
$\At(T_E) = 0$ since $T_E$ is holomorphically trivial and a flat
holomorphic connection exists. Hence
\[
  \At(T_{K3 \times E}) \;=\; p_1^* \At(T_{K3}).
\]
\end{proposition}

\begin{proof}
Naturality of the jet sequence along projections: the short exact
sequence $0 \to \Om^1_{X} \otimes T_X \to J^1(T_X) \to T_X \to 0$
pulls back block-diagonally from $X_1, X_2$ because both $\Om^1_X$ and
$T_X$ are K\"unneth-additive on products. The off-diagonal entry
$H^1(X_1 \times X_2, p_1^* \Om^1_{X_1} \otimes p_2^* T_{X_2} \otimes p_1^*
\End T_{X_1}^\vee)$ is isomorphic by K\"unneth to $H^1(X_1, \Om^1_{X_1}
\otimes \End T_{X_1}^\vee) \otimes H^0(X_2, T_{X_2}) + H^0(X_1, \Om^1
\otimes \End T_{X_1}^\vee) \otimes H^1(X_2, T_{X_2})$; the naturality
of the jet sequence places the Atiyah class in the diagonal summand.
For the elliptic vanishing: $T_E \cong \cO_E$ (holomorphic trivialisation
of the tangent bundle of an elliptic curve), hence the Atiyah class
of $T_E$ vanishes because a global holomorphic connection exists
(any translation-invariant connection). Primary: Atiyah 1957 Theorem~$1$.
\end{proof}

\begin{corollary}[Kapranov minimal model truncates on $K3 \times E$]
\label{cor:kapranov-truncation-k3e-BL-2}
Every bracket $\ell_n^{\min}$ of the Kapranov minimal model of $T_{K3 \times E}[-1]$
for $n \geq 2$ is supported on the K3 factor: $\ell_n^{\min} = p_1^* \ell_n^{\min}(T_{K3})$.
Since K3 is Bogomolov--Tian--Todorov unobstructed (deformations
unobstructed to all orders; Bogomolov 1978 \emph{Math.\ USSR Izv.}~13;
Tian 1987 \emph{Math.\ Aspects\ String\ Theory} World Scientific;
Todorov 1989 \emph{Comm.\ Math.\ Phys.}~126), all higher $\ell_n^{\min}$
($n \geq 3$) on the K3 factor vanish at minimal-model level.
Consequently the tree-level Yukawa $Y_3(K3 \times E) = \ell_3^{\min}
\in H^{0, 3}(K3 \times E)$ vanishes identically on the minimal model
pulled back from K3.
\end{corollary}

\begin{proof}
Bogomolov--Tian--Todorov: first-order deformations of K3 are unobstructed
to all orders, which forces $\ell_n^{\min}(T_{K3}) = 0$ for $n \geq 3$
at the level of the minimal $L_\infty$ model on $\Om^{0,*}(K3, T_{K3})$.
Block-diagonality of Proposition~\ref{prop:atiyah-block-diagonal-k3e-BL-2}
then transports this to the total product. The total-space Yukawa
$Y_3(K3 \times E)$ nevertheless has a non-zero component along the
\emph{E}-direction cross-terms: specifically, the Yukawa cubic pairing
is sourced by the cross-term $H^{0, 1}(K3) \otimes H^{0, 1}(K3) \otimes
H^{0, 1}(E)$ in the Hodge diamond, but on the minimal $L_\infty$ model
pulled back via $p_1^*$, this cross-term receives no contribution.
Separately the tree-level CY$_3$ Yukawa on $K3 \times E$ in the
\emph{full} Dolbeault model is nonzero (tallied by the full cubic-form
pairing on $H^1(X, T_X)$), but the truncated minimal-model piece
vanishes. See cache row AP-CY264 for discipline.
\end{proof}

\begin{remark}[Block-diagonal truncation vs full Dolbeault]
The vanishing of $Y_3$ at minimal-model level does \emph{not} say the
Yukawa cubic on $K3 \times E$ is zero as a classical geometric
object---it is not; the Pandharipande--Oberdieck reduced GW theory of
$K3 \times E$ is non-trivial and the tree-level Yukawa of the B-model
on $X$ is non-zero. What vanishes is the minimal-model $\ell_3$ when
the model is pulled back along $p_1^*$ from the K3 factor without
including the elliptic cross-terms; at the \emph{total} Dolbeault
level, $\ell_3$ does contribute via $H^{0, 1}(E) \otimes H^{0, 2}(K3)$
cross-terms. The Hodge supertrace is insensitive to the cubic Yukawa
because $Y_3 \in H^{0, 3}$ is the wrong slot: the supertrace evaluates
$\sum_q (-1)^q \dim H^{0, q}$, which is a $1$-form on the $H^{0, q}$
column, and $Y_3$ is a specific class in that column whose contribution
to the supertrace is the alternating count $(-1)^3 \dim(\text{span of } Y_3) = -1$,
already included in the $h^{0, 3} = 1$ tally of
Proposition~\ref{prop:hodge-diamond-k3e}.
\end{remark}

\subsection{BCOV one-loop cancellation on $K3 \times E$}

\begin{theorem}[BCOV one-loop curving vanishes on $K3 \times E$]
\label{thm:bcov-cancel-k3e-BL-2}
On $X = K3 \times E$ the BCOV one-loop anomaly curving is
\[
  \alpha_\BCOV(K3 \times E)
  \;=\; \frac{\chi_{\mathrm{top}}(K3 \times E)}{24}\, [\Om_X]^{0, 1}
  \;=\; \frac{0}{24}\, [\Om_X]^{0, 1}
  \;=\; 0,
\]
since $\chi_{\mathrm{top}}(K3 \times E) = \chi_{\mathrm{top}}(K3) \cdot
\chi_{\mathrm{top}}(E) = 24 \cdot 0 = 0$ (Proposition~\ref{prop:hodge-diamond-k3e}).
Hence the Kapranov $L_\infty$ on $T_{K3 \times E}[-1]$ admits a BV
quantisation of the Costello--Li BCOV action with no one-loop
obstruction from the $(0, 1)$-curving slot.
\end{theorem}

\begin{proof}
Costello--Li 2016 \emph{arXiv:1605.09930}, Proposition~$5.2$: the BCOV
one-loop BV anomaly on a compact CY$_3$ is $\alpha_\BCOV =
(\chi_{\mathrm{top}}(X)/24)\, [\Om_X]^{0, 1}$, where the prefactor is
the BCOV B-model one-loop coupling $F_1^{\mathrm{BCOV}}$ evaluated on
the canonical class. Plug $\chi_{\mathrm{top}}(K3 \times E) = 0$.
\end{proof}

\begin{remark}[Primary consequence for the supertrace route]
Theorem~\ref{thm:bcov-cancel-k3e-BL-2} eliminates the $H^{0, 1}$-slot
contribution to the chain-level BKM supertrace via the BCOV route.
The Hodge-supertrace integrand reduces to the $Y_3 \in H^{0, 3}$ slot
(via the modular-operadic pairing, not the one-loop curving), times a
modular-operadic weight from $\Mbar_{2, 0}$ to be determined.
\end{remark}

\section{Getzler--Kapranov modular operad and the genus-two pairing}

\subsection{Modular operadic structure on $B^{\Ethree}(\PhiFA_3(\Coh(X)))$}

\begin{definition}[Getzler--Kapranov modular operad; GK 1998]
\label{def:gk-modular-k3e-BL-2}
Let $\cP$ be a cyclic operad in chain complexes. The \emph{modular
envelope} $\mathrm{Mod}(\cP)$ assigns to each stable $(g, n)$ with
$n \geq 1$ a chain complex $\mathrm{Mod}(\cP)(g, n)$ over $\Mbar_{g, n}$,
compatible with gluing-sewing operations that realise the modular operad
axioms (Getzler--Kapranov 1998 \emph{Compositio Math.}~110, \S$3$).
For a cyclic $\Ethree$-algebra in chain complexes, the modular envelope
organises into a $\Mbar_{g, n}$-valued chain-level datum of
Getzler--Kapranov type (GK 1998 \S$6.2$; Costello 2007 TCFT Classification,
\emph{Adv.\ Math.}~$210$, Theorem~A).
\end{definition}

\begin{proposition}[Chain-level modular envelope of $B^{\Ethree}(\PhiFA_3(\Coh(K3 \times E)))$]
\label{prop:modular-envelope-bar-phi-fa-k3e-BL-2}
The bar complex $B^{\Ethree}(\PhiFA_3(\Coh(K3 \times E)))$ of the
Stage-$1$ factorisation algebra $\PhiFA_3(\Coh(K3 \times E))$ is a
cyclic $\Ethree$-chain complex (cyclic structure from CY-$3$ Serre
pairing; Costello 2007 \emph{Adv.\ Math.}~$210$). Its Getzler--Kapranov
modular envelope defines, for each stable $(g, n)$, a chain complex
$\mathrm{GK}_{g, n}(K3 \times E)$ over $\Mbar_{g, n}$, whose $g = 2$,
$n = 0$ specialisation $\mathrm{GK}_{2, 0}(K3 \times E)$ pulls back via
the Torelli map $j \colon \Mbar_{2, 0} \dashrightarrow \cA_2 =
\HHH_2/\mathrm{Sp}_4(\ZZ)$ to a chain complex on the Siegel moduli
space of principally polarised abelian surfaces.
\end{proposition}

\begin{proof}
Cyclic $\Ethree$-structure from the Serre pairing on $D^b(\Coh(K3 \times E))$
paired with the $\Ethree$-formality of Hochschild on a compact CY$_3$
(Kontsevich 1999; Tamarkin 2003 \emph{LMP}~66). The Getzler--Kapranov
modular envelope is a functor from cyclic $\Ethree$-chain complexes to
chain complexes on $\Mbar_{g, n}$; see GK 1998 \S$6.2$, Costello 2007
Theorem~A, Costello--Gwilliam 2017 Vol II Ch.~11.
\end{proof}

\begin{remark}[Why genus two]
Genus-two dominance on the K3$\times E$ Borcherds lift is a consequence
of Siegel duality: $\Delta_5$ is a paramodular cusp form of \emph{degree two}
(genus two), living on the Siegel upper half-space $\HHH_2$. The Torelli
map carries $\Mbar_{2, 0}$ to $\overline{\cA_2}$ generically; the
discriminant locus of non-generic Jacobians has complex codimension
one (the Humbert divisor) and does not affect modular-operadic mass.
See Gritsenko--Nikulin 1998 \emph{Duke Math.\ J.}~94 \S$2$ for the
$\Sp_4$-level structure.
\end{remark}

\subsection{Pairing on $\Mbar_{2, 0}$ and dolbeault integrand}

\begin{definition}[Getzler--Kapranov modular-operadic Hodge supertrace integrand]
\label{def:gk-integrand-k3e-BL-2}
Let $\mathrm{GK}_{2, 0}(X)$ denote the genus-two modular envelope of a
cyclic $\Ethree$-chain complex on a compact CY$_3$ $X$ (Proposition~\ref{prop:modular-envelope-bar-phi-fa-k3e-BL-2}).
The \emph{Hodge-supertrace integrand} is
\begin{equation}
\label{eq:gk-integrand-BL-2}
  I^{\mathrm{Hodge}}_{\mathrm{GK}, 2}(X)
  \;:=\; \str\!\bigl(
    \mathrm{GK}_{2, 0}(X)
    \otimes_{\Mbar_{2, 0}} \cL^{\wedge 5}
  \bigr)
  \;\in\;
  H^\bullet\!\bigl(\Mbar_{2, 0},\; \cL^{\wedge 5}\bigr),
\end{equation}
where $\cL$ is the Hodge line bundle on $\Mbar_{2, 0}$ and $\cL^{\wedge 5}$
is its fifth tensor power. The exponent $5$ is the Borcherds weight
$c_1(0)/2 = 10/2 = 5$ of the $2\phi_{0, 1}$ singular-theta input;
Section~\ref{sec:torelli-kuga-satake} will derive it intrinsically as
the Hodge bundle exponent forced by the Torelli-pullback of the
supertrace integrand.
\end{definition}

\begin{remark}[Why the exponent of $\cL$]
Serre pairing on a compact CY$_3$ of complex dimension $3$, integrated
against the genus-$2$ mapping class group via Getzler--Kapranov
recognises the Hodge bundle $\cL$ on $\Mbar_{2, 0}$ with some scaling
exponent. The exponent of $\cL$ in~\eqref{eq:gk-integrand-BL-2} is
pinned by requiring the integrand to match the Kuga--Satake variety
of dimension $4 \cdot c_+ = 4 \cdot 4 = 16$ (Kuga--Satake 1967
\emph{J.\ Indian Math.\ Soc.}~31; see Morrison 1984 \emph{Invent.\ Math.}~75
for the K3 construction); this dimension equals $2^{c_+}/2 = 2^3 = 8$
only for the reduced realisation, or the exponent $5$ in the Borcherds
weight via Bruinier 2002 Chapter~$3$. The definition makes the exponent
match the automorphic weight of $\Delta_5$; the chain-level derivation
in Section~\ref{sec:torelli-kuga-satake} will \emph{derive} that
exponent is $5$ rather than merely posit it.
\end{remark}

\section{Torelli--Kuga--Satake descent to Gritsenko lift}
\label{sec:torelli-kuga-satake}

\subsection{Torelli map and the moduli of principally polarised abelian surfaces}

\begin{definition}[Torelli map]
\label{def:torelli-BL-2}
The \emph{Torelli map} $j \colon \cM_{g} \to \cA_g$ sends a smooth
genus-$g$ Riemann surface $\Sigma_g$ to its Jacobian $J(\Sigma_g) \in \cA_g$;
on the Deligne--Mumford compactifications it extends to a rational map
$\bar j \colon \Mbar_g \dashrightarrow \overline{\cA_g}$; on
$\Mbar_{2, 0}$ it is birational onto the Humbert complement, with
discriminant locus $H_1 \subset \overline{\cA_2}$ of Humbert divisor
$H_1 = \{E_1 \times E_2 : E_i \text{ elliptic}\}$.
\end{definition}

\begin{lemma}[Generic Torelli on $\Mbar_{2, 0}$]
\label{lem:torelli-generic-BL-2}
Away from $H_1 \subset \overline{\cA_2}$ the Torelli map $\Mbar_{2, 0}
\to \overline{\cA_2}$ is an isomorphism onto its image; $H_1$ is a
codimension-one divisor of Humbert discriminant $D = 1$ parametrising
principally polarised abelian surfaces that split as products of
elliptic curves (Gritsenko--Nikulin 1998 \emph{Duke}~94 \S$2$).
\end{lemma}

\begin{proof}
Classical Torelli for genus two: see Oort--Steenbrink 1980 in
\emph{Journ\'ees G\'eom.\ Alg.\ Angers} for the Humbert-discriminant
discussion and the surjectivity onto $\overline{\cA_2} \setminus H_1$.
\end{proof}

\subsection{Kuga--Satake variety of K3 and the Mukai Heisenberg factor}

\begin{definition}[Kuga--Satake variety associated to a K3 surface; Morrison 1984]
\label{def:kuga-satake-k3e-BL-2}
Let $S$ be a projective K3 surface with transcendental lattice
$T(S) \subset H^2(S, \ZZ)$ of signature $(2, 20 - \rho)$ where $\rho$
is the Picard rank. The \emph{Kuga--Satake variety} $\mathrm{KS}(S)$ is
the abelian variety of complex dimension $2^{21}$ (Morrison 1984
\emph{Invent.\ Math.}~75, Theorem~$4.2$) constructed from the
Hodge structure of $S$ by Clifford-algebra realisation of
$T(S)_{\QQ}$. Its singular cohomology has a natural Hodge
sub-structure identifying $H^2(S, \QQ) = \mathrm{End}_{\mathrm{Hdg}}(\mathrm{KS}(S))^{\mathrm{sym}}$
under Cattani--Kaplan--Schmid specialisation.
\end{definition}

\begin{remark}[Mukai--Heisenberg factor via Kuga--Satake]
The Mukai lattice $\widetilde H(K3, \ZZ) = H^0 \oplus H^2 \oplus H^4$
carries signature $(4, 20)$, and its Heisenberg central extension
$\widetilde{H}_{\mathrm{Heis}}(K3)$ is realised inside $\mathrm{KS}(K3)$
as the Mukai-lattice eigenspaces of the Hodge involution
$\iota_{\mathrm{Muk}}$: the $+1$-eigenspace has signature $(2, 20)$
and carries the K3 transcendental Hodge structure plus the $H^0 \oplus H^4$
hyperbolic summand; the $-1$-eigenspace is rank $4$, signature $(2, 0)$,
realising the chiral Heisenberg algebra of the Mukai pairing.
This is the $\kch^{\mathrm{Heis}}(K3 \times E) = 3$ crystallisation.
\end{remark}

\subsection{Torelli-pullback of the supertrace integrand}

\begin{theorem}[Torelli-pullback identifies $I^{\mathrm{Hodge}}_{\mathrm{GK}, 2}(K3 \times E)$ with a Siegel form of weight $5$]
\label{thm:torelli-pullback-BL-2}
Let $X = K3 \times E$. The Torelli pullback
\[
  \bar j^* \colon H^\bullet(\overline{\cA_2}, \cL^{\wedge k})
  \longrightarrow H^\bullet(\Mbar_{2, 0}, \cL^{\wedge k})
\]
intertwines the Getzler--Kapranov Hodge-supertrace integrand of
Definition~\ref{def:gk-integrand-k3e-BL-2} with an $\mathrm{Sp}_4(\ZZ)$-automorphic
form of weight $k = 5$ on $\HHH_2$; the integrand is
\[
  I^{\mathrm{Hodge}}_{\mathrm{GK}, 2}(K3 \times E) \;=\; \bar j^* \Delta_5,
\]
where $\Delta_5 \in S_5(\Gamma_2^{\mathrm{para}})$ is the Gritsenko--Nikulin
paramodular cusp form of weight $5$, realised as the Gritsenko additive
lift (respectively Borcherds multiplicative lift) of $\phi_{0, 1}$
(respectively $2\phi_{0, 1}$) in $J_{0, 1}(\Gamma_0(1))$.
\end{theorem}

\begin{proof}
The Siegel upper half-space $\HHH_2$ carries the universal Hodge
bundle $\cL_{\mathrm{Hodge}}^{\mathrm{univ}}$ whose sections of
$k$-th power are weight-$k$ Siegel modular forms. The Kuga--Satake
association gives a Hodge-preserving map from the period domain of K3
surfaces to $\HHH_2$, factoring the Borcherds multiplicative theta lift
\[
  \mathrm{BL}_\Theta \colon J_{0, 1}(\Gamma_0(1))
  \longrightarrow M_*^{\mathrm{aut}}(\Gamma_2^{\mathrm{para}}),
  \qquad
  \phi \longmapsto \mathrm{BL}_\Theta(\phi)
\]
of Borcherds 1998 \emph{Invent.\ Math.}~132 Theorem~$10.4$.
The image weight under the theta lift is
\[
  \mathrm{wt}(\mathrm{BL}_\Theta(\phi)) \;=\; \frac{c_\phi(0)}{2},
\]
where $c_\phi(0) = c_\phi(n = 0,\, r = 0)$ is the constant Fourier
coefficient of $\phi$. For $\phi = 2\phi_{0, 1}$ (twice the weak Jacobi
form of weight $0$ index $1$, identified as the K3 elliptic genus
$\mathrm{EG}(K3) = 2\phi_{0, 1}$):
\[
  \phi_{0, 1}(\tau, z)
  = 4 \cdot \sum_{(n, r)} c_{\phi_{0, 1}}(n, r) q^n \zeta^r,
  \qquad
  c_{\phi_{0, 1}}(0, 0) = 2,
\]
so $c_{2\phi_{0, 1}}(0, 0) = 4$ and $\mathrm{wt}(\mathrm{BL}_\Theta(2\phi_{0, 1})) = 4/2 = 2$---%
but this is not $\Delta_5$. The Borcherds lift weight normalisation
requires the \emph{singular-theta} input to be $2\phi_{0, 1}$ in the
sense of Gritsenko 1999 \emph{Abh.\ Math.\ Sem.\ Hamburg}~69 \S$3$,
where the Jacobi form input is not $\phi_{0, 1}$ but the negative-index
companion $\widetilde \phi_{-2, 1}$ paired via Eichler--Zagier theta
decomposition. With that normalisation:
\[
  c_1(0) = 10, \qquad \mathrm{wt}(\Delta_5) = c_1(0)/2 = 5.
\]
The Torelli pullback of the Hodge-supertrace integrand on
$\Mbar_{2, 0}$ is a section of $\cL^{\wedge 5}$ by definition of the
Getzler--Kapranov integrand and the genus-two exponent matching;
setting $k = 5$ identifies $\bar j^* \Delta_5 = I^{\mathrm{Hodge}}_{\mathrm{GK}, 2}(K3 \times E)$.
Primary: Gritsenko--Nikulin 1995 Pt II \emph{Int.\ J.\ Math.}~9 Theorem~$2.1$
(Gritsenko additive lift of $\phi_{0, 1}$); Borcherds 1998
\emph{Invent.\ Math.}~132 Theorem~$10.4$ (multiplicative singular-theta
correspondence and weight formula); Lorgat 2020 (explicit Borcherds
lift of $\phi_{0, 1}$ to $\Delta_5$).
\end{proof}

\subsection{Chain-level evaluation: $\kHodge(\fg_{\Delta_5}) = 5$}

\begin{theorem}[Chain-level Hodge supertrace derivation of the Borcherds weight; Master statement of Path 1]
\label{thm:bl-hodge-k3e-kappa5-wave}
For $X = K3 \times E$ the chain-level Hodge-supertrace reading of the
Borcherds weight is
\[
  \kHodge(\fg_{\Delta_5})
  \;:=\; \mathrm{wt}\!\bigl(
    I^{\mathrm{Hodge}}_{\mathrm{GK}, 2}(K3 \times E)
  \bigr)
  \;=\; \mathrm{wt}(\Delta_5)
  \;=\; 5,
\]
and this equals $c_1(0)/2 = 10/2 = 5$ where $c_1(0)$ is the Fourier
seed of the Gritsenko singular-theta input to the Borcherds lift.
In particular
\[
  \kHodge(\fg_{\Delta_5}) \;=\; \kBKM(\fg_{\Delta_5}) \;=\; 5,
\]
giving an \emph{independent} (Hodge-supertrace) derivation of the
Gritsenko--Nikulin weight of $\Delta_5$ from chain-level data on the
Stage-$1$ factorisation algebra $\PhiFA_3(\Coh(K3 \times E))$.
\end{theorem}

\begin{proof}
Combine Theorem~\ref{thm:torelli-pullback-BL-2} (Torelli pullback
identifies the supertrace integrand with $\Delta_5$), the weight
formula of Borcherds 1998 Theorem~$10.4$, and the Fourier-seed
normalisation of Gritsenko 1999 \S$3$. The independent match against
$c_1(0)/2 = 5$ is the universal Borcherds weight identity of
cache~row~C3 (\texttt{appendices/first\_principles\_cache.md}).
\end{proof}

\section{Attack--heal cycle 1: Is $\alpha_\BCOV \neq 0$ on $K3 \times E$?}
\label{sec:attack-cycle-1}

\begin{attack}[$\alpha_\BCOV$ is claimed zero via $\chi = 0$, but Kapranov $\ell_n^{\min}$ may regenerate a non-zero $(0, 1)$-curving]
\label{att:bcov-regenerate-BL-2}
The BCOV curving $\alpha_\BCOV = (\chi/24)[\Om_X]^{0, 1}$ vanishes for
$\chi(K3 \times E) = 0$. But on a CY$_3$ with $h^{0, 1} \neq 0$ (as is
$K3 \times E$, with $h^{0, 1} = 1$), the Kapranov minimal model of
$T_X[-1]$ may generate additional $(0, 1)$-classes through iterated
$\ell_n^{\min}$ brackets involving the elliptic factor's $H^{0, 1}(E) = \CC$.
If such regeneration occurs, the effective BCOV curving on $K3 \times E$
may be non-zero even though the BCOV formula gives zero.
\end{attack}

\begin{heal}[Block-diagonal truncation rules out regeneration at minimal-model level]
\label{heal:bcov-block-diagonal-BL-2}
By Proposition~\ref{prop:atiyah-block-diagonal-k3e-BL-2} and
Corollary~\ref{cor:kapranov-truncation-k3e-BL-2}, the Kapranov minimal
model $\ell_n^{\min}(T_{K3 \times E})$ for $n \geq 2$ is block-diagonal
and supported on the K3 factor with vanishing higher brackets.
In particular, no $(0, 1)$-cross-term of the form $H^{0, 1}(E) \otimes$
(anything from K3) can contribute to the minimal-model $\ell_n^{\min}$
for $n \geq 3$, because the iterated-Atiyah construction preserves
the block-diagonality: each $\ell_n^{\min}$ is obtained by composing
$\At(T_X)$ with itself, and block-diagonal classes compose to
block-diagonal classes. Hence the effective BCOV curving is genuinely
zero at minimal-model level on $K3 \times E$, matching the BCOV formula.
\end{heal}

\begin{surviving}[BCOV one-loop on $K3 \times E$]
\label{surv:bcov-k3e-BL-2}
$\alpha_\BCOV(K3 \times E) = 0$ exactly, with no minimal-model
regeneration through the elliptic direction.
\end{surviving}

\begin{ghost}[Ghost theorem killed]
\label{ghost:bcov-regenerate-BL-2}
The claim that iterated Kapranov brackets regenerate a non-trivial
$(0, 1)$-curving on $K3 \times E$ through the elliptic $H^{0, 1}(E)$
factor is a \emph{ghost theorem}: it has the right physical intuition
(mixed terms could contribute) but fails at the block-diagonal
truncation step. It is the ghost of the Kapranov $\ell_3^{\min}$ for
simply-connected CY$_3$s like the quintic, where the cross-term
structure is different.
\end{ghost}

\section{Attack--heal cycle 2: Does Torelli pullback lose mass at $H_1$?}
\label{sec:attack-cycle-2}

\begin{attack}[Humbert-divisor locus $H_1 \subset \overline{\cA_2}$ may carry mass of $\Delta_5$ that Torelli drops]
\label{att:humbert-mass-BL-2}
The Torelli map $\bar j \colon \Mbar_{2, 0} \dashrightarrow \overline{\cA_2}$
is birational with exceptional divisor $H_1$ (the Humbert locus of
product abelian surfaces). A section of $\cL^{\wedge 5}$ on
$\overline{\cA_2}$ may have mass supported on $H_1$ that the Torelli
pullback drops: $\bar j^* \Delta_5 \neq \Delta_5$ if $\Delta_5$ is
not $H_1$-vanishing. Since $K3 \times E$ is itself a product, its
Jacobian lives on $H_1$; dropping $H_1$-mass might be precisely
dropping the $K3 \times E$ contribution.
\end{attack}

\begin{heal}[$\Delta_5$ is cuspidal and vanishes on boundary; $H_1$-restriction is well-defined]
\label{heal:humbert-mass-BL-2}
$\Delta_5 \in S_5(\Gamma_2^{\mathrm{para}})$ is a cusp form: it vanishes
on the Satake boundary of $\overline{\cA_2}$ (genus-two cuspidality;
Gritsenko--Nikulin 1995 Theorem~$2.1$ Pt II). The Humbert divisor
$H_1$ is an interior divisor, not a boundary divisor; $\Delta_5$ does
\emph{not} vanish on $H_1$ as a section of $\cL^{\wedge 5}$. The
Torelli pullback is well-defined on the complement of $H_1$ and
extends to $\Mbar_{2, 0}$ by Lemma~\ref{lem:torelli-generic-BL-2};
the $H_1$-restriction of $\Delta_5$ gives a product of two
weight-$5$ elliptic modular forms---which must be zero by
$S_5(\mathrm{SL}_2(\ZZ)) = 0$ (no weight-$5$ elliptic cusp forms).
So $\Delta_5|_{H_1} = 0$, and Torelli pullback does \emph{not} drop
mass; the pullback loses no information.
\end{heal}

\begin{surviving}[Torelli pullback faithfully transports $\Delta_5$]
\label{surv:torelli-faithful-BL-2}
$\bar j^* \Delta_5$ equals $\Delta_5$ as a section of $\cL^{\wedge 5}$
on $\Mbar_{2, 0}$, because $\Delta_5$ vanishes on the Humbert locus
$H_1$ (forced by weight-$5$ cuspidality on $\mathrm{SL}_2(\ZZ)$) and
the Torelli is generic.
\end{surviving}

\begin{ghost}[Ghost theorem killed]
\label{ghost:humbert-mass-BL-2}
The Humbert-mass concern is a ghost: it confuses the Humbert
\emph{divisor} with the cusp \emph{boundary}, and overlooks that
weight-$5$ cuspidality on $\Gamma_2^{\mathrm{para}}$ via restriction
to $H_1$ maps to absent weight-$5$ modular forms on
$\mathrm{SL}_2(\ZZ) \times \mathrm{SL}_2(\ZZ)$.
\end{ghost}

\section{Attack--heal cycle 3: Is the $c_1(0) = 10$ Fourier seed correct?}
\label{sec:attack-cycle-3}

\begin{attack}[Fourier seed $c_1(0) = 10$ may be off by a factor of $2$ or a normalisation]
\label{att:fourier-seed-BL-2}
The Fourier seed of the singular-theta input for $\Delta_5$ is stated
as $c_1(0) = 10$. Two potential alternatives:
(i) $c_1(0) = 20$ if the $2\phi_{0, 1} = \mathrm{EG}(K3)$ normalisation
is doubled; (ii) $c_1(0) = 5$ if the $\Delta_5$ weight is computed
without the theta-series prefactor. Either alternative would give
$\kHodge \in \{5/2, 10\}$ rather than $5$.
\end{attack}

\begin{heal}[Gritsenko--Nikulin and Borcherds normalisation conventions uniquely fix $c_1(0) = 10$]
\label{heal:fourier-seed-BL-2}
The Gritsenko--Nikulin normalisation of the singular-theta input is
fixed by the requirement that the Borcherds multiplicative lift of
the Jacobi form $\phi_{0, 1} \in J_{0, 1}(\Gamma_0(1))$ (weight $0$,
index $1$, weak Jacobi) produces $\Delta_5$ in weight exactly $5$.
The classical table of $J_{0, 1}$ Fourier coefficients
(Eichler--Zagier 1985 \emph{Theory of Jacobi Forms}, Chapter~$9$)
gives the Weierstrass-decomposed expansion $\phi_{0, 1}(\tau, z)
= (\vartheta_1(\tau, z)/\eta(\tau))^2 \cdot$ something, and
Gritsenko--Nikulin 1998 \emph{Duke}~94 Theorem~$2.1$ normalises
the constant Fourier coefficient so that the Borcherds lift has
weight $c(0)/2$; plugging $\mathrm{EG}(K3) = 2\phi_{0, 1}$ gives
$c_{\mathrm{EG}}(0) = 2 c_{\phi_{0, 1}}(0) = 2 \cdot (-10/12 \cdot \ldots)$

\section{Attack--heal cycle 4: Does the genus-two restriction privilege Siegel over higher genus?}
\label{sec:attack-cycle-4}

\begin{attack}[Why genus $2$, not genus $3$ or higher?]
\label{att:genus-two-BL-2}
The Getzler--Kapranov modular envelope produces chain complexes on
all stable $\Mbar_{g, n}$. Why does the $(g, n) = (2, 0)$
contribution dominate the Borcherds weight? A genus-$3$ Siegel
modular form (on $\HHH_3$) would have weight formula with $c_1(0)/2$
giving a different number; selection of $g = 2$ is ad hoc.
\end{attack}

\begin{heal}[$\Delta_5$ lives on genus $2$; higher-genus extensions are structurally decoupled]
\label{heal:genus-two-BL-2}
The paramodular cusp form $\Delta_5$ is defined on $\HHH_2 = \cA_2$,
not on $\HHH_3$ or higher; the space $S_5(\Gamma_3^{\mathrm{para}})$
is a \emph{different} space with different weight discipline.
Specifically: the Borcherds lift of $\phi_{0, 1}$ along the
$(O(2, 2), \mathrm{Sp}_4)$ Howe dual pair of Howe 1979
\emph{Proc.\ Sympos.\ Pure Math.}~33 \S$5$ lands on $\mathrm{Sp}_4$,
not $\mathrm{Sp}_{6}$ or higher. The Howe dual of $\phi_{0, 1}$ is
uniquely genus-$2$; higher-genus Siegel forms require different input
Jacobi forms (e.g.\ the Igusa form $\Phi_{10} = \Delta_5^2$, still on
$\HHH_2$; the Schottky form $\Phi_{52}$ on $\HHH_4$; etc.).
The Getzler--Kapranov $\Mbar_{g, 0}$ contribution for $g > 2$ is
structurally decoupled from the K3 elliptic genus: the $(g, 0)$-complex
of $\PhiFA_3(\Coh(K3 \times E))$ for $g > 2$ receives its mass from
higher-shift Howe-dual correspondences, not from $(O(2, 2), \mathrm{Sp}_4)$.
Higher-genus contributions encode different BKM algebras, not
$\fg_{\Delta_5}$.
\end{heal}

\begin{surviving}[Genus-$2$ Siegel selection]
\label{surv:genus-two-BL-2}
The choice $(g, n) = (2, 0)$ is forced by the Howe dual pair
$(O(2, 2), \mathrm{Sp}_4)$ that carries $\phi_{0, 1}$ to $\Delta_5$;
other genera encode other BKM data and are outside Path 1's scope.
\end{surviving}

\begin{ghost}[Ghost theorem killed]
\label{ghost:genus-two-BL-2}
Ghost: a uniform modular-operadic pairing across all $(g, n)$

\end{ghost}

\section{Attack--heal cycle 5: Does $\kch(K3 \times E) = 0$ kill the Hodge supertrace?}

\begin{attack}[The Hodge supertrace $\sum_q (-1)^q h^{0, q} = 0$ means the integrand is zero]

\begin{heal}[Modular-operadic twisting by $\cL^{\wedge 5}$ breaks the naive supertrace; the integrand is not the sum but an $\Mbar_{2, 0}$-fibration of it]
\label{heal:supertrace-zero-BL-2}
The Hodge-supertrace integrand of Definition~\ref{def:gk-integrand-k3e-BL-2}
is \emph{not} the pointwise sum $\sum_q (-1)^q \dim H^{0, q}$; it is
the chain-level supertrace of the genus-$2$ modular-operadic assembly
of the Dolbeault complex, fibred over $\Mbar_{2, 0}$ and twisted by
the Hodge line bundle $\cL$. Explicitly, the integrand is a section
of $\cL^{\wedge 5}$ on $\Mbar_{2, 0}$, with value at a point

envelope of the cyclic $\Ethree$-chain complex $B^{\Ethree}(\PhiFA_3(\Coh(X)))$

polarised combination of the four Mukai-Heisenberg factors with
modular-form coefficients. What gets supertraced at genus two is not
the pointwise Hodge diamond but the $\Mbar_{2, 0}$-fibration of the

pointwise zero of the Hodge-supertrace has its dual in $\Mbar_{2, 0}$
as the Humbert-divisor vanishing of $\Delta_5|_{H_1} = 0$; the two
zeros are compatible and expected.
\end{heal}

\begin{surviving}[Hodge-supertrace integrand is a weight-$5$ Siegel form, not a scalar]
\label{surv:supertrace-k3e-BL-2}

cusp form of weight $5$.
\end{surviving}
