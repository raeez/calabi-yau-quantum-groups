\chapter{Wall-Crossing and CoHA: the Algebra-vs-Coalgebra Ledger}
\label{ch:coha-wall-crossing-platonic}

Chapter~\ref{ch:toric-coha} assembles a chiral quantum group datum from
the critical CoHA, the Maulik--Okounkov $R$-matrix, the Drinfeld double,
and the chart-gluing chiral algebra $A_X$. Three distinctions govern the
precise hypothesis ledger of that assembly:

\begin{enumerate}[label=\textup{(\roman*)}]
  \item \textbf{Algebra versus coalgebra.}
    The cohomological Hall algebra $\mathcal{H}(Q, W)$ is a
    $\mathbb{Z}$-graded associative algebra \emph{without internal
    differential}. The bar complex $B^{\mathrm{ord}}(\mathcal{H})$ is a
    coassociative coalgebra constructed \emph{from} $\mathcal{H}$. The
    phrase ``$D^2 = 0$ in the CoHA differential'' in
    Theorem~\ref{thm:wall-crossing-mc} is shorthand for
    ``$D^2 = 0$ in the bar complex of $\mathcal{H}$'' or for
    ``$\partial_W^2 = 0$ in the Ginzburg dg algebra $\Pi(Q, W)$''; these
    are distinct objects and the shorthand suppresses the passage. This
    chapter separates them.
  \item \textbf{Positive half versus full Yangian.}
    The Schiffmann--Vasserot theorem identifies
    $\mathcal{H}(\mathbb{C}^3) \simeq Y^+(\widehat{\mathfrak{gl}}_1)$,
    the \emph{positive half}. The full affine Yangian
    $Y(\widehat{\mathfrak{gl}}_1)$ is the Drinfeld double of
    $\mathcal{H}(\mathbb{C}^3)$ via a non-degenerate Hopf pairing. For
    $X = \mathbb{C}^3$ this double is proved (Procha\'zka--Rap\v{c}\'ak);
    for general toric $X$ the double exists but the identification
    with a specific affine super Yangian is conditional on a Hopf-pairing
    lemma which we record as a proposition with precise hypotheses.
  \item \textbf{Motivic Hall Lie algebra versus chiral convolution dgLA.}
    Kontsevich--Soibelman wall-crossing lives in the motivic Hall Lie
    algebra $\mathfrak{g}_{\mathrm{KS}}(X)$ of the CY$_3$ category
    $D^b\mathrm{Coh}(X)$. This symbol has two distinct ambient
    incarnations — the motivic/quantum
    $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{mot}}(X)$ (whose exponential is
    the motivic quantum torus) and the classical/numerical
    $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{cl}}(X)$ (whose exponential is
    the classical symplectic torus); their precise definition and
    compatibility appear in
    Theorem~\ref{thm:ks-wall-crossing-mc-on-coha-dgla} and
    Remark~\ref{rem:ks-quantum-vs-classical-ambient}.
    In this preamble the wall-crossing automorphism is the exponential
    gauge action in EITHER ambient, intertwined by the Euler-characteristic
    specialisation $\chi \colon \mathcal{R} \to \mathbb{Q}$.
    The MC equation on the bar side of
    Theorem~\ref{thm:wall-crossing-mc} lives in the convolution dgLA
    $\mathfrak{g}^{\mathrm{mod}}_{A_X} :=
    \bigl(\mathrm{End}^{\mathrm{ch}}_{A_X}, [-,-]_G\bigr)$
    of chiral endomorphisms of $A_X$ under the Gerstenhaber bracket
    (Chapter~\ref{ch:toric-coha},
    Theorem~\ref{thm:wall-crossing-mc} and surrounding prose). The
    identification of these two dgLAs factors through the chiralization
    correspondence $\Phi_3$ of Theorem~\ref{thm:cy-to-chiral-d3}
    ($\infty$-categorical, chain-level for non-formal algebras conditional
    on CY-A$_3$); the bridge $\Phi_*$ is proved for $\mathbb{C}^3$ (RSYZ
    plus the conifold bar-complex engine) and conditional on CY-A$_3$
    outside that case.
\end{enumerate}

\section{Setup: CoHA, Ginzburg dg algebra, bar complex}
\label{sec:coha-setup-platonic}

Distinction (i) asks which object among those attached to a quiver with
potential carries the chain-level identity $d^2 = 0$, and which are
graded algebras or cohomological invariants without internal differential.
Three objects answer the question, and they are distinct.

Let $Q = (Q_0, Q_1, s, t)$ be a finite quiver (vertices $Q_0$, arrows
$Q_1$, source and target maps $s, t \colon Q_1 \to Q_0$) with potential
$W \in \mathbb{C}Q_{\mathrm{cyc}}$, where $\mathbb{C}Q_{\mathrm{cyc}} =
\mathbb{C}Q / [\mathbb{C}Q, \mathbb{C}Q]$ is the quotient of the path
algebra by the commutator subspace (the cyclic path algebra). For toric
CY$_3$ geometries $X$ without compact $4$-cycles the McKay-correspondence
quiver $(Q_X, W_X)$ is the input.

\begin{definition}[Three distinct objects]
\label{def:three-coha-objects}
\ClaimStatusProvedHere
Associated to $(Q, W)$ we distinguish:
\begin{enumerate}[label=\textup{(\alph*)}]
  \item \textbf{Path algebra.} The path algebra $\mathbb{C}Q$ with
    relations $\partial W / \partial a_i = 0$ for each $a_i \in Q_1$
    (the Jacobi algebra
    $J(Q, W) = \mathbb{C}Q / (\partial W)$).
    This is an associative $\mathbb{Z}_{\geq 0}$-graded algebra
    (grading by path length), \emph{without} internal differential.
  \item \textbf{Ginzburg dg algebra.} The dg algebra $\Pi(Q, W)$
    with generators $Q_0 \cup Q_1 \cup Q_1^* \cup \{\epsilon_i\}_{i \in Q_0}$
    in degrees $0, 0, -1, -2$ and differential $\partial_W$
    determined by
    $\partial_W(a_i^*) = \partial W / \partial a_i$ and
    $\partial_W(\epsilon_i) = \sum_{a \in Q_1} [a, a^*]\, e_i$.
    This is a genuinely dg algebra: $\partial_W^2 = 0$ encodes the
    Jacobi identity for $W$ as a cyclic function.
  \item \textbf{Critical CoHA.} The cohomological Hall algebra
    \[
      \mathcal{H}(Q, W)
       \;=\;
      \bigoplus_{\mathbf{d} \in \mathbb{Z}_{\geq 0}^{Q_0}}
       \; H^{\bullet}_{G_\mathbf{d}}
        \bigl( \mathrm{Rep}(Q, \mathbf{d}),
         \phi_{W_\mathbf{d}} \bigr),
    \]
    a $\mathbb{Z}^{Q_0}$-graded associative algebra with Hall product
    (Kontsevich--Soibelman 2010, Davison--Meinhardt 2015). $\mathcal{H}$
    is an associative graded algebra: it carries the Hall product as
    $m_2$ and NO internal differential at the level of definition.
\end{enumerate}
\end{definition}

\begin{proof}[Justification]
(a) is literally the Jacobi algebra. (b) is Ginzburg 2006
\textup{arXiv:math/0612139}, Proposition~5.1.9; the identity
$\partial_W^2 = 0$ is a cyclic-derivation calculation (the double
derivative $\partial^2 W / \partial a_i \partial a_j$ is symmetric,
and $\partial_W^2$ contracts it against $[a_i^*, a_j^*] = 0$). (c)
is Kontsevich--Soibelman 2010 \textup{arXiv:1006.2706} Section~7,
specialized to the critical (with-potential) setting; cf.\
Davison--Meinhardt 2015 \textup{arXiv:1601.02479} for the
integrality theorem. Each object has its own grading and its own
role: the Jacobi algebra is the classical shadow, the Ginzburg dg
algebra is the dg model that carries the differential, the CoHA is
the cohomological invariant built \emph{from} the Ginzburg dg algebra
by taking Borel--Moore homology of the critical locus.
\end{proof}

\begin{remark}[Which object carries the differential]
\label{rem:which-carries-differential}
Among (a)--(c) of Definition~\ref{def:three-coha-objects}:
\begin{itemize}
  \item (a) has NO differential; it is a graded algebra.
  \item (b) HAS a differential $\partial_W$; it is a dg algebra. This is
    where $\partial_W^2 = 0$ is a nontrivial identity encoding the
    Jacobi algebra relations.
  \item (c) has NO internal differential; its grading is by the charge
    lattice $\mathbb{Z}^{Q_0}$, and its algebra structure is the
    cohomological Hall product built by taking $H^\bullet_{G_\mathbf{d}}$
    of vanishing cycles. The CoHA IS the output of a derived
    construction, but the output itself is a graded algebra, not a dg
    algebra.
\end{itemize}
Prose of the form ``$D^2 = 0$ in the CoHA differential'' is a
\emph{category mistake} unless one means either (i) the
Ginzburg dg algebra differential, or (ii) the bar differential on
$B^{\mathrm{ord}}(\mathcal{H})$. Passages that mix these silently
are the targets of Theorem~\ref{thm:coha-is-algebra-not-coalgebra}.
\end{remark}

\begin{definition}[Bar complex of the CoHA]
\label{def:bar-of-coha-platonic}
\ClaimStatusProvedHere
The ordered bar complex of the CoHA is
\[
  B^{\mathrm{ord}}(\mathcal{H})
   \;=\; T^c\bigl( s^{-1} \overline{\mathcal{H}} \bigr),
\]
where $\overline{\mathcal{H}} = \ker(\epsilon)$ is the augmentation
ideal of $\mathcal{H}$ (as a graded algebra over $\mathbb{C}$) and
$T^c$ is the tensor coalgebra with deconcatenation coproduct. The
bar differential $d_B = \sum_{k \geq 1} b_k$ encodes the Hall product
as $b_2$; since $\mathcal{H}$ is associative and carries no higher
$A_\infty$-operations at the level of definition, $b_k = 0$ for
$k \geq 3$ and $b_2$ is the dual of the Hall product.
\end{definition}

\begin{remark}[Four inputs, four outputs, one confusion]
\label{rem:four-inputs-four-outputs-platonic}
The four objects on the table are:
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{lll}
\toprule
Object & Structure & Carries $d^2 = 0$? \\
\midrule
Jacobi algebra $J(Q, W)$ & graded algebra & no (no differential) \\
Ginzburg dg alg $\Pi(Q, W)$ & dg algebra & yes (native) \\
Critical CoHA $\mathcal{H}(Q, W)$ & graded algebra & no (no differential) \\
$B^{\mathrm{ord}}(\mathcal{H})$ & dg coalgebra & yes (bar differential) \\
\bottomrule
\end{tabular}
\end{center}
The confusion of writing ``$D^2 = 0$ in the CoHA'' targets the wrong
object: the CoHA is a graded algebra, not a dg object. The bar complex
\emph{of} the CoHA is a dg coalgebra and does carry $D^2 = 0$. The
Ginzburg dg algebra is a dg algebra with $\partial_W^2 = 0$. The
three are linked by a chain of functors, not equated.
\end{remark}

\section{CoHA is an algebra, not a coalgebra}
\label{sec:coha-is-algebra}

\begin{theorem}[CoHA is an algebra, not a coalgebra]
\label{thm:coha-is-algebra-not-coalgebra}
\ClaimStatusProvedHere
Let $(Q, W)$ be a quiver with potential and
$\mathcal{H} = \mathcal{H}(Q, W)$ its critical CoHA. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
  \item $\mathcal{H}$ is an associative $\mathbb{Z}^{Q_0}$-graded
    algebra over $\mathbb{C}$ with Hall product
    $m_H \colon \mathcal{H}_{\mathbf{d}_1} \otimes \mathcal{H}_{\mathbf{d}_2}
     \to \mathcal{H}_{\mathbf{d}_1 + \mathbf{d}_2}$;
  \item $\mathcal{H}$ carries NO internal differential in its definition
    as a graded algebra; the Borel--Moore homology $H^{\bullet}_{G_\mathbf{d}}$
    that builds $\mathcal{H}$ is taken \emph{after} passing to cohomology,
    not before;
  \item the bar complex $B^{\mathrm{ord}}(\mathcal{H}) = T^c(s^{-1} \bar{\mathcal{H}})$
    is a separate object: a coassociative dg coalgebra whose differential
    is the bar differential $d_B$, with $d_B^2 = 0$;
  \item the statement ``$D^2 = 0$ in the CoHA'' in the prose of
    Theorem~\textup{\ref{thm:wall-crossing-mc}} refers to the bar
    complex $B^{\mathrm{ord}}(\mathcal{H})$ or to the Ginzburg dg algebra
    $\Pi(Q, W)$, not to the CoHA itself.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) is Kontsevich--Soibelman 2010 Theorem~1 or Schiffmann--Vasserot
2013 Theorem~A specialised to the critical setting. The Hall product
is the convolution $m_H = \pi_* \circ p^*$ on equivariant Borel--Moore
homology with vanishing-cycle coefficient, where $p$ is the
short-exact-sequence projection and $\pi$ is the quotient to the
target dimension vector. Associativity is Schiffmann--Vasserot 2013
Proposition~1.4.

(ii) is immediate from the construction: $\mathcal{H}_\mathbf{d}$
is defined as a cohomology group
$H^\bullet_{G_\mathbf{d}}(\mathrm{Rep}(Q, \mathbf{d}), \phi_{W_\mathbf{d}})$,
which is a graded vector space; vanishing cycles is applied once, then
equivariant cohomology is taken, producing a graded vector space
without residual differential. There is no chain-level dg structure
\emph{on} $\mathcal{H}$ itself; there is such a structure on the
Ginzburg dg algebra $\Pi(Q, W)$ whose Hochschild homology feeds in,
but $\mathcal{H}$ is the cohomological output.

(iii) is the standard bar-complex construction. Given an associative
graded algebra $A$, $B^{\mathrm{ord}}(A) = T^c(s^{-1} \bar A)$ is the
tensor coalgebra on the desuspended augmentation ideal, with
deconcatenation coproduct and differential $d_B = \sum_{k \geq 1} b_k$
where $b_2$ is dual to the algebra multiplication and $b_k = 0$ for
$k \geq 3$ when $A$ is strictly associative. $d_B^2 = 0$ is equivalent
to associativity of $A$ (Stasheff 1963, Loday--Vallette 2012 \S2.3).

(iv) by remaining consistent with (i)-(iii) we are forced to
re-interpret any chain-level identity $D^2 = 0$ in Theorem~%
\ref{thm:wall-crossing-mc} as living on the bar side or the Ginzburg
side, not on the CoHA. The Ginzburg-side interpretation is the one used
in the proof of Theorem~\ref{thm:wall-crossing-mc}(iii): the identity
$\partial_W^2 = 0$ in $\Pi(Q, W)$ is what encodes the Jacobi
$\partial W / \partial a_i = 0$ condition, and this is what enters the
MC equation via the dg Lie algebra associated to $\Pi(Q, W)$.
\end{proof}

\begin{remark}[Reconciliation with the prose of Theorem~\ref{thm:wall-crossing-mc}]
\label{rem:reconciliation-wall-crossing-prose}
Clause (iii) of Theorem~\ref{thm:wall-crossing-mc} reads:
\begin{quote}
  The Jacobi algebra condition $\partial W / \partial a_i = 0$ is
  equivalent to $D^2 = 0$ in the CoHA differential, which in turn is
  the MC equation $D\Theta + \frac{1}{2}[\Theta, \Theta] = 0$.
\end{quote}
The correct reading: the Jacobi algebra condition
$\partial W / \partial a_i = 0$ defines the zero locus of
$\partial_W$ in the Ginzburg dg algebra, equivalently
$\partial_W^2 = 0$ holds in $\Pi(Q, W)$. This translates to the MC
equation
$D\Theta_W + \frac{1}{2}[\Theta_W, \Theta_W] = 0$ in the dg Lie
algebra $(\Pi(Q, W))_{\mathrm{Lie}}$ associated to
$\Pi(Q, W)$ with MC element
$\Theta_W = \sum_i (\partial W / \partial a_i) \, a_i^*$ (Ginzburg
2006 \textup{arXiv:math/0612139}, Proposition~5.1.9). The CoHA enters only as the cohomological
invariant of this dg data; it is not itself a dg object, and the
MC equation is on $\Pi(Q, W)_{\mathrm{Lie}}$, not on
$\mathcal{H}(Q, W)$.
\end{remark}

\section{Chiralisation preserves algebra structure}
\label{sec:chiralisation-preserves-algebra}

\begin{theorem}[Chiralisation preserves the algebra-side structure]
\label{thm:coha-chiralization-preserves-algebra}
\ClaimStatusConditional
Let $X$ be a local CY$_3$ with quiver $(Q_X, W_X)$ and associated critical
CoHA $\mathcal{H}(Q_X, W_X)$. Let
$\Phi \colon D^b\mathrm{Coh}(X) \to \mathrm{ChirAlg}$ denote the
chiralisation correspondence of Theorem~CY-A$_3$
($\infty$-categorical existence unconditional; chain-level embedding
for non-formal $A_X$ conditional on the non-formal extension of
CY-A$_3$; Theorem~\textup{\ref{thm:cy-to-chiral-d3}}).
Then the image $\Phi(X) = A_X$ is an $E_1$-chiral ALGEBRA, and the
CoHA $\mathcal{H}(Q_X, W_X)$ embeds as the positive-half subalgebra:
\[
  \mathcal{H}(Q_X, W_X)
   \;\hookrightarrow\;
  A_X
\]
as graded algebras, via the identification
$\mathcal{H}(Q_X, W_X) \simeq Y^+(\widehat{\mathfrak{g}}_{Q_X})$
(Schiffmann--Vasserot for $\mathbb{C}^3$, Rapcak--Soibelman--Yang--Zhao
for general toric without compact $4$-cycles) and the identification
of $Y^+$ with a subalgebra of $A_X$ generated by the positive simple roots.
In particular, chiralisation preserves ALGEBRA structure: an associative
graded algebra $\mathcal{H}$ lands in an $E_1$-chiral algebra $A_X$,
not in a chiral coalgebra.
\end{theorem}

\begin{proof}
The correspondence $\Phi_3$ is constructed in Theorem~\ref{thm:cy-to-chiral-d3}
as a symmetric-monoidal $\infty$-functor from $D^b\mathrm{Coh}(X)$
with Yoneda convolution to the $\infty$-category of $E_1$-chiral
algebras (for $d \geq 3$; $E_2$-chiral at $d \leq 2$, the dimension
stratification of $\Phi$-output). Its
input is an associative monoidal structure (Yoneda convolution), and its
output is an $E_1$-algebra structure. There is no conversion to
coalgebra at any stage.

The embedding of $\mathcal{H}$ as the positive-half subalgebra of $A_X$
proceeds through the identification
$\mathcal{H}(Q_X, W_X) \simeq Y^+(\widehat{\mathfrak{g}}_{Q_X})$ (RSYZ;
Schiffmann--Vasserot for $\mathbb{C}^3$), where
$\widehat{\mathfrak{g}}_{Q_X}$ is the affine super Lie algebra
determined by the toric fan of $X$ (RSYZ 2020, Theorem~1.1; its root
system is read off the quiver adjacency and the $\mathbb{Z}/2$-grading
from the arrow parities). The affine super Yangian
$Y(\widehat{\mathfrak{g}}_{Q_X})$ has the PBW vector-space
decomposition
$Y \simeq Y^+ \otimes Y^0 \otimes Y^-$
(positive Borel, Cartan, negative Borel). Explicitly, the zero-mode
section
$\iota \colon Y^+ \hookrightarrow A_X, \quad y \longmapsto y[0]$,
sending a positive-root generator $e_\alpha \in Y^+$ to its zero
Laurent coefficient $e_\alpha[0] \in A_X$, is an algebra map because
the Hall product on $\mathcal{H}(Q_X, W_X) \simeq Y^+$ agrees with the
$[0]$-mode product in $A_X$ (Schiffmann--Vasserot
\textup{arXiv:1202.2756} Theorem~A(ii), extended to general toric
$X$ without compact $4$-cycles by RSYZ 2020 Theorem~1.1). No coalgebra
structure enters: the chiralisation correspondence is algebra-to-%
algebra by construction.

The bar complex $B^{\mathrm{ord}}(A_X) = T^c(s^{-1} \bar A_X)$ is a
separate object, a dg coalgebra; this is the coalgebra resolution of
$A_X$ on the Koszul locus and should not be confused with $A_X$ itself.
The CoHA embeds in $A_X$; it does not embed in $B^{\mathrm{ord}}(A_X)$.
\end{proof}

\begin{remark}[What chiralisation does NOT do]
\label{rem:chiralisation-does-not}
The correspondence $\Phi_d$ does NOT produce a chiral coalgebra from a CY
category. It produces an $E_n$-chiral algebra ($n = 2$ at $d \leq 2$,
$n = 1$ at $d \geq 3$). The bar complex of that output is a chiral
coalgebra; but the output itself is an algebra. Prose of the form
``$\Phi$ converts CoHA to a chiral coalgebra'' conflates
Theorem~\ref{thm:coha-is-algebra-not-coalgebra} with its bar resolution.
\end{remark}

\section{KS wall-crossing as MC equation on the motivic Hall Lie algebra}
\label{sec:ks-mc-on-hall-dgla}

The algebra-side embedding $\mathcal{H} \hookrightarrow A_X$ of the
previous section forces the question of where wall-crossing lives. A
gauge transformation $\Theta \mapsto e^{\mathrm{ad}_\alpha}(\Theta)$
presupposes a dg Lie algebra and an MC element; $\mathcal{H}$ itself
carries no differential, so the wall-crossing MC equation must live on
a dgLA constructed \emph{from} $\mathcal{H}$. Kontsevich--Soibelman
supply this dgLA as the motivic Hall Lie algebra
$\mathfrak{g}_{\mathrm{KS}}(X)$.

\begin{theorem}[KS wall-crossing is MC gauge on the motivic Hall dgLA]
\label{thm:ks-wall-crossing-mc-on-coha-dgla}
\ClaimStatusProvedHere
Let $X$ be a local CY$_3$ and fix, as ambient, the motivic Hall algebra
$\mathrm{MH}(X)$ of $D^b\mathrm{Coh}(X)$ in the sense of
Kontsevich--Soibelman 2008 (\textup{arXiv:0811.2435}, Section~2.3,
``motivic Hall algebra of a CY$_3$ category''), defined over the
ring
$\mathcal{R} := K_0(\mathrm{Var}_{\mathbb{C}})\bigl[\mathbb{L}^{-1},
 (1 - \mathbb{L}^{-k})^{-1}\,:\,k \geq 1\bigr]$
with $\mathbb{L} = [\mathbb{A}^1]$ the Lefschetz motive. The following
two distinct dg Lie algebras are attached to $\mathrm{MH}(X)$, and the
theorem is stated at the scope of each one.

\textbf{Quantum ambient}
$\mathfrak{g}_{\mathrm{KS}}^{\mathrm{mot}}(X)$: the Lie subalgebra of
$\mathrm{MH}(X)_{\mathrm{Lie}}$ generated by the motivic simples
$\{e_\gamma\}_{\gamma \in K_0}$, with bracket
$[e_\gamma, e_{\gamma'}]_q =
 \bigl(\mathbb{L}^{\langle\gamma',\gamma\rangle/2}
      - \mathbb{L}^{-\langle\gamma',\gamma\rangle/2}\bigr)
  e_{\gamma+\gamma'}$
defined \emph{after} torsor-twisting by a square root
$\mathbb{L}^{1/2}$; its exponential is the motivic quantum torus
$G_{\mathrm{KS}}^{\mathrm{mot}}(X) \subset \mathrm{MH}(X)^\times$
(KS 2008 Section~6.2).

\textbf{Classical ambient}
$\mathfrak{g}_{\mathrm{KS}}^{\mathrm{cl}}(X)$: the semiclassical limit
$\mathbb{L}^{1/2} \to 1$ of the quantum ambient (equivalently, the
Euler-characteristic specialisation of the motivic Hall algebra), a
$\mathbb{Z}$-graded Lie algebra with bracket
$[e_\gamma, e_{\gamma'}] = \langle\gamma,\gamma'\rangle\,
 e_{\gamma+\gamma'}$
where $\langle\gamma,\gamma'\rangle := \chi(\gamma,\gamma')
- \chi(\gamma',\gamma)$ is the skew Euler form (antisymmetric because
the CY$_3$ Euler form $\chi$ is symmetric, so
$\langle\gamma,\gamma\rangle = 0$); its exponential is the classical
symplectic torus $G_{\mathrm{KS}}^{\mathrm{cl}}(X)$ (KS 2008 Section~2.3,
``semiclassical formula''; Bridgeland
\textup{arXiv:1002.4372} (``Hall algebras and curve-counting
invariants'', J.~Amer.~Math.~Soc.\ 24 (2011))
Proposition~5.2 for the Euler-characteristic/Poisson-Hall
specialisation, with the classical torus obtained as
$G_{\mathrm{KS}}^{\mathrm{cl}}(X) = G_{\mathrm{KS}}^{\mathrm{mot}}(X)
 \otimes_{\mathcal{R}} \mathbb{Q}$
under $\chi \colon [V] \mapsto \chi(V)$).

Then, in each ambient, the following hold:

\begin{enumerate}[label=\textup{(\roman*)}]
  \item Both $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{mot}}(X)$ and
    $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{cl}}(X)$ are nilpotent
    pro-finite Lie algebras after completion in the Harder--Narasimhan
    filtration; their exponentials are, respectively, the motivic
    quantum torus (with $q$-commutator
    $e_\gamma e_{\gamma'} = \mathbb{L}^{\langle\gamma,\gamma'\rangle/2}
     e_{\gamma + \gamma'}$) and the classical symplectic torus (with
    commutative product $e_\gamma e_{\gamma'} = e_{\gamma + \gamma'}$
    twisted by the antisymmetric form under Poisson bracket).
  \item For generic stability parameters $\zeta, \zeta'$ in adjacent
    chambers separated by a wall $W$, the KS wall-crossing formula
    holds in BOTH ambients, in compatible forms:
    \begin{enumerate}[label=\textup{(\alph*)}]
      \item (\emph{motivic})
        $\Theta^{\mathrm{mot}}_{\zeta'} = \exp(\mathrm{ad}_{\alpha_W^{\mathrm{mot}}})
         (\Theta^{\mathrm{mot}}_\zeta)$
        in $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{mot}}(X)$, with
        coefficients the motivic DT classes
        $[\mathfrak{M}_\gamma^{\zeta\text{-ss}}]^{\mathrm{vir}}
         \in \mathcal{R}$
        (Joyce--Song 2011 \textup{arXiv:0810.5645}, Theorem~5.11 for
        the motivic refinement; KS 2008 Theorem~3.1 for the original
        motivic wall-crossing);
      \item (\emph{classical/numerical})
        $\Theta^{\mathrm{cl}}_{\zeta'} = \exp(\mathrm{ad}_{\alpha_W^{\mathrm{cl}}})
         (\Theta^{\mathrm{cl}}_\zeta)$
        in $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{cl}}(X)$, with
        coefficients the numerical DT invariants
        $\Omega(\gamma; \zeta) = \chi(\mathfrak{M}_\gamma^{\zeta\text{-ss}})
         \in \mathbb{Q}$
        (Joyce--Song 2011, Theorem~5.3, numerical case; Bridgeland
        \textup{arXiv:1002.4372} (JAMS 24, 2011) Theorem~5.3,
        equivalent formulation via Hall algebra exponentials).
    \end{enumerate}
    Here $\alpha_W^\bullet$ is supported on the wall class
    $\gamma_W \in K_0(\mathrm{Coh}(X))$, and the Euler-characteristic
    specialisation intertwines (a) and (b): the image of
    $\Theta_\zeta^{\mathrm{mot}}$ under
    $\chi \colon \mathcal{R} \to \mathbb{Q}$
    is $\Theta_\zeta^{\mathrm{cl}}$ (KS 2008 Section~7).
  \item The MC equation
    $d\Theta_\zeta + \frac{1}{2}[\Theta_\zeta, \Theta_\zeta] = 0$
    holds in each ambient:
    \begin{itemize}
      \item on $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{mot}}(X)$, as the
        closedness of the motivic DT generating series against the
        $q$-commutator Hall bracket (KS 2008 Section~4);
      \item on $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{cl}}(X)$, as the
        closedness of the numerical DT generating series against the
        Euler-characteristic Hall bracket (Joyce--Song 2011,
        Theorem~5.3, ``no-pole theorem'').
    \end{itemize}
    The two MC equations are intertwined by the
    Euler-characteristic specialisation.
  \item The pentagon identity
    $\exp(\mathrm{ad}_{\alpha_5}) \circ \cdots \circ
     \exp(\mathrm{ad}_{\alpha_1}) = \mathrm{id}$
    in $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{mot}}(X)$ specialises to
    the $A_2$ cluster-mutation five-cycle. It is the $q$-commutator
    form of Faddeev--Kashaev (the quantum dilogarithm pentagon); its
    Euler-characteristic specialisation is the classical dilogarithm
    five-term relation on $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{cl}}(X)$
    (Rogers--Abel five-term identity). The ambient for ``pentagon in
    the Hall Lie algebra'' is $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{mot}}$,
    NOT $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{cl}}$: the classical limit
    of the quantum dilogarithm $\mathbb{E}(x)$ degenerates to
    $\exp(\mathrm{Li}_2(x))$, and the pentagon becomes the functional
    equation of the Rogers dilogarithm.
\end{enumerate}
This MC equation lives on the ALGEBRA side: each $\mathfrak{g}_{\mathrm{KS}}^\bullet$
is a graded Lie algebra, the gauge action is algebra-preserving, and
no bar coalgebra enters at this level.
\end{theorem}

\begin{proof}
(i) The quantum ambient
$\mathfrak{g}_{\mathrm{KS}}^{\mathrm{mot}}(X)$ is constructed in KS
2008 Section~6.2: the motivic quantum torus is the associative
$\mathcal{R}$-algebra
$\mathcal{R}\langle x_\gamma \rangle / (x_\gamma x_{\gamma'}
 = \mathbb{L}^{\langle\gamma,\gamma'\rangle/2}
   x_{\gamma+\gamma'})$,
and its Lie subalgebra of BPS generators carries the $q$-commutator
bracket above. The classical ambient
$\mathfrak{g}_{\mathrm{KS}}^{\mathrm{cl}}(X)$ is the
$\mathbb{L}^{1/2} \to 1$ specialisation (KS 2008 Section~2.3,
semiclassical paragraph); equivalently, it is the
Euler-characteristic Poisson Lie algebra on the charge lattice
(Bridgeland 2016 Proposition~5.2). Both are nilpotent pro-finite
after Harder--Narasimhan completion (KS 2008 Section~4.2).

The skew Euler form is antisymmetric on a CY$_3$ category because
$\chi$ is symmetric (Serre duality identifies
$\mathrm{Ext}^i(\gamma,\gamma') \simeq \mathrm{Ext}^{3-i}(\gamma',\gamma)^\vee$
and the alternating sum of dimensions is symmetric under
$\gamma \leftrightarrow \gamma'$); hence
$\langle\gamma,\gamma\rangle = 0$ and the diagonal bracket vanishes.

(ii)(a) is KS 2008 Theorem~3.1: the motivic wall-crossing is the
equality of phase-ordered products
$\prod_{\arg Z \downarrow}^{\zeta} \mathbb{E}(x_\gamma)^{\Omega^{\mathrm{mot}}(\gamma;\zeta)}
 = \prod_{\arg Z \downarrow}^{\zeta'} \mathbb{E}(x_\gamma)^{\Omega^{\mathrm{mot}}(\gamma;\zeta')}$
in $G_{\mathrm{KS}}^{\mathrm{mot}}(X)$; taking the logarithm and
extracting the infinitesimal wall generator $\alpha_W^{\mathrm{mot}}$
(Section~2.7, Lemma~2.8) yields the MC gauge transformation.

(ii)(b) is Joyce--Song 2011 Theorem~5.3 (the ``Joyce--Song
wall-crossing formula''), which is the Euler-characteristic
specialisation of (a); equivalently, Bridgeland
\textup{arXiv:1002.4372} (JAMS 24, 2011) Theorem~5.3, stated
directly at the classical Hall-algebra level without motivic
refinement. The
specialisation $\chi \colon \mathcal{R} \to \mathbb{Q}$ is a Lie
algebra map by construction (KS 2008 Section~7, semiclassical
compatibility lemma).

(iii) On the motivic side, the MC equation is the closedness of the
motivic DT generating series
$\Theta^{\mathrm{mot}}_\zeta = \sum_\gamma
 [\mathfrak{M}_\gamma^{\zeta\text{-ss}}]^{\mathrm{vir}}\, e_\gamma$
against the $q$-commutator Hall bracket (KS 2008 Section~4,
``no-pole'' theorem at the motivic level); on the classical side it
is the numerical no-pole theorem of Joyce--Song 2011 Theorem~5.3.

(iv) In $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{mot}}(X)$, the pentagon
is the quantum dilogarithm five-term relation
$\mathbb{E}(x_{e_1}) \mathbb{E}(x_{e_2})
 = \mathbb{E}(x_{e_2}) \mathbb{E}(x_{e_1 + e_2}) \mathbb{E}(x_{e_1})$
where
$\mathbb{E}(x) = \prod_{k \geq 0} (1 - \mathbb{L}^{-(k+1/2)} x)^{-1}$
is the KS quantum dilogarithm (Faddeev--Kashaev
\textup{arXiv:hep-th/9310070}; Reineke 2011
\textup{arXiv:0903.0261} Theorem~5.1 for the Hall-algebra
formulation). Its Euler-characteristic specialisation is the
Rogers dilogarithm five-term relation
$L(x) + L(y) = L(xy) + L(x(1-y)/(1-xy)) + L(y(1-x)/(1-xy))$
(Rogers 1907; see Kirillov 1995 \textup{math/9408113} for the
Hall-Lie-algebra derivation). Algebraic verification of the
$q$-commutator pentagon through height~$2$ is programmed in
\texttt{compute/tests/test\_conifold\_wall\_crossing.py}
(73 tests, at general $\mathbb{L}^{1/2}$).
\end{proof}

\begin{remark}[The quantum/classical ambient distinction is load-bearing]
\label{rem:ks-quantum-vs-classical-ambient}
The prose
``$\exp \colon \mathfrak{g}_{\mathrm{KS}} \to G_{\mathrm{KS}}$ is the
motivic quantum torus'' is accurate ONLY in the quantum ambient
$\mathfrak{g}_{\mathrm{KS}}^{\mathrm{mot}}$. In the classical ambient
$\mathfrak{g}_{\mathrm{KS}}^{\mathrm{cl}}$, the exponential image is
the \emph{classical symplectic torus}, which is a Poisson-abelian
group with commutative product — not the $q$-commutator motivic
torus. Conflating the two amounts to taking a $q$-commutator pentagon
and dropping the $q$, which collapses the pentagon to a trivial
abelian identity. The pentagon in the Hall Lie algebra is NOT a
classical identity: it is a quantum dilogarithm identity whose
nontriviality is the $\mathbb{L}^{1/2}$-dependence of the bracket.

Correspondingly, the MC element $\Theta_\zeta$ has two flavours
tracked separately in this theorem:
$\Theta^{\mathrm{mot}}_\zeta$ with motivic-class coefficients in
$\mathcal{R}$, and $\Theta^{\mathrm{cl}}_\zeta$ with numerical DT
invariant coefficients in $\mathbb{Q}$. The first is the primary
object in KS 2008; the second is its Euler-characteristic shadow,
which is what appears in Joyce--Song 2011 and in the Bridgeland
\textup{arXiv:1002.4372} (JAMS 24, 2011) re-derivation. The chiralisation bridge $\Phi_*$ of
Remark~\ref{rem:two-mc-equations} operates on EACH ambient
separately: the motivic bridge
$\Phi_*^{\mathrm{mot}} \colon \mathfrak{g}_{\mathrm{KS}}^{\mathrm{mot}}
 \to \mathfrak{g}^{\mathrm{mod}}_{A_X, q}$
lands in a $q$-deformed chiral convolution dgLA (tracking the
motivic refinement), while
$\Phi_*^{\mathrm{cl}} \colon \mathfrak{g}_{\mathrm{KS}}^{\mathrm{cl}}
 \to \mathfrak{g}^{\mathrm{mod}}_{A_X}$
lands in the undeformed chiral convolution dgLA that appears in
Theorem~\ref{thm:wall-crossing-mc}. The CoHA $\mathcal{H}(Q_X, W_X)$
itself is the Euler-characteristic side: Schiffmann--Vasserot 2013
identifies it with $Y^+(\widehat{\mathfrak{gl}}_1)$ at
$\psi = \mathbb{L}^{1/2}|_{\mathbb{L} = 1}$, i.e., in the classical
ambient. The $q$-deformed CoHA (K-theoretic Hall algebra of
Schiffmann--Vasserot 2017 \textup{arXiv:1705.07039}) is the motivic
refinement and lands in $\mathfrak{g}_{\mathrm{KS}}^{\mathrm{mot}}$.
Throughout the remainder of this chapter, unsubscripted
$\mathfrak{g}_{\mathrm{KS}}(X)$ and $\Theta_\zeta$ refer to the
CLASSICAL ambient unless the motivic refinement is explicitly
flagged; the distinction was elided in the prose of
Theorem~\ref{thm:wall-crossing-mc} and is load-bearing here.
\end{remark}

\begin{remark}[Relation to the bar-side MC in Theorem~\ref{thm:wall-crossing-mc}]
\label{rem:two-mc-equations}
Theorem~\ref{thm:wall-crossing-mc} in Chapter~\ref{ch:toric-coha}
states the MC gauge equivalence
in $\mathfrak{g}^{\mathrm{mod}}_{A_X}$ (convolution dgLA of the chiral
algebra $A_X$). The present Theorem~\ref{thm:ks-wall-crossing-mc-on-coha-dgla}
states it in $\mathfrak{g}_{\mathrm{KS}}(X)$ (motivic Hall dgLA). These

chiralisation correspondence $\Phi_d$:
\[
  \Phi_* \colon \mathfrak{g}_{\mathrm{KS}}(X)
   \longrightarrow \mathfrak{g}^{\mathrm{mod}}_{A_X}.
\]
The compatibility ``$\Phi_*(\Theta_{\zeta}^{\mathrm{KS}})
= \Theta_\zeta^{\mathrm{mod}}$'' is the CONTENT of the
CoHA-to-chiral bridge. The ingredients for $X = \mathbb{C}^3$ (the

$4$-cycles (conditional on the RSYZ identification plus the
$\mathbb{C}^3$-local chart-gluing of
Theorem~\ref{thm:toric-chart-gluing}). Outside this locus the
compatibility lies in the non-formal chain-level extension of
CY-A$_3$.
\end{remark}

\section{Conifold wall-crossing: cluster algebra and chiral conjecture}
\label{sec:conifold-cluster-wall-crossing}

\begin{theorem}[Conifold wall-crossing as cluster mutation]
\label{thm:conifold-cluster-wall-crossing}
\ClaimStatusConditional
Let $X_{\mathrm{con}}$ be the resolved conifold
$\mathcal{O}(-1) \oplus \mathcal{O}(-1) \to \mathbb{P}^1$ with
Klebanov--Witten quiver $(Q_{\mathrm{con}}, W_{\mathrm{con}})$. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
  \item The motivic DT wall-crossing at the single wall
    $\gamma_0 = (1, 0) - (0, 1) \in K_0(\mathrm{Coh}(X_{\mathrm{con}}))$
    is the exponential gauge action of Theorem~%
    \ref{thm:ks-wall-crossing-mc-on-coha-dgla}(ii).
  \item This wall-crossing matches the cluster-mutation structure of
    the $A_2$ cluster algebra: the two chambers of the stability space
    correspond to the two exchange paths around the $A_2$ pentagon.
  \item Let $\mathcal{H}_{\mathrm{I}} \subset \mathcal{H}(Q_{\mathrm{con}},
    W_{\mathrm{con}})$ (resp.\ $\mathcal{H}_{\mathrm{II}}$) denote the
    chamber-specific Hall subalgebra generated by the simple stable
    objects of the resolved chamber (resp.\ the flopped chamber). The
    bar-complex dimensions of these chamber subalgebras are
    $\dim B^k(\mathcal{H}_{\mathrm{I}}) = 2^k$ (two stable simples) and
    $\dim B^k(\mathcal{H}_{\mathrm{II}}) = 3^k$ (three stable simples
    after mutation), verified through degree $k = 12$
    (\textup{124} tests in \texttt{conifold\_bar\_complex.py}). The
    ambient $Y^+(\widehat{\mathfrak{g}}_{Q_{\mathrm{con}}})$ is one fixed
    algebra; the two chamber subalgebras are distinct subalgebras of it,
    exchanged by the pentagon mutation.
  \item The lift to an $E_1$-chiral algebra identity (i.e.,
    chiralisation of the cluster structure) is PROVED for $\mathbb{C}^3$
    local charts and conditional outside them.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) is Theorem~\ref{thm:ks-wall-crossing-mc-on-coha-dgla}(ii) applied
to the conifold quiver; the single wall class
$\gamma_0 = (1, 0) - (0, 1)$ is the primitive wall in the unique
two-dimensional stability space, and the wall-crossing generator
is $\alpha_{\gamma_0}$ with Nagao--Nakajima BPS invariant
$\Omega(\gamma_0) = 1$ (single rank-$1$ simple object crossing the wall).
The self-pairing $\langle\gamma_0, \gamma_0\rangle$ under the skew
Euler form vanishes automatically (CY$_3$ antisymmetry); the wall
nontriviality is carried by $\Omega$, not by a diagonal Euler-form
pairing.

(ii) is Nagao--Nakajima 2011 \textup{arXiv:0809.2992} Theorem~3.5:
the conifold stability space has two chambers related by a single
mutation, and the pentagon periodicity is the $A_2$ cluster algebra
mutation class of rank $2$.

(iii) by direct bar-complex enumeration of the chamber subalgebras
$\mathcal{H}_{\mathrm{I}}, \mathcal{H}_{\mathrm{II}} \subset
\mathcal{H}(Q_{\mathrm{con}}, W_{\mathrm{con}})$ (Construction~%
\ref{constr:conifold-bar} in Chapter~\ref{ch:toric-coha}, with the
engine verification cited). The count $2^k$ in $\mathcal{H}_{\mathrm{I}}$
reflects two stable simples in the resolved chamber; the $3^k$ in
$\mathcal{H}_{\mathrm{II}}$ reflects three stable simples after the
mutation; the ambient CoHA is unchanged across the wall.

(iv) conditional: the chiralisation of cluster algebras is a
conjectural content of the programme; the local $\mathbb{C}^3$ chart
lifts are proved via Schiffmann--Vasserot, but the global assembly for
the conifold is a chart-gluing conditional on the local-to-global
descent for $E_1$-chiral algebras
(Theorem~\textup{\ref{thm:toric-chart-gluing}} plus
Proposition~\textup{\ref{prop:mutation-e1-equivalence}}).
\end{proof}

\section{Positive half versus full Yangian}
\label{sec:y-plus-vs-full-y}

\begin{corollary}[$Y^+$ is not the full Yangian; Drinfeld double yields it]
\label{cor:y-plus-vs-full-Y}
\ClaimStatusConditional
\begin{enumerate}[label=\textup{(\roman*)}]
  \item $\mathcal{H}(\mathbb{C}^3) \simeq Y^+(\widehat{\mathfrak{gl}}_1)$
    is the POSITIVE HALF of the affine Yangian
    $Y(\widehat{\mathfrak{gl}}_1)$
    (Schiffmann--Vasserot 2013 \textup{arXiv:1202.2756}, Theorem~A).
  \item The full affine Yangian
    $Y(\widehat{\mathfrak{gl}}_1)$, with PBW vector-space decomposition
    $Y \simeq Y^+ \otimes Y^0 \otimes Y^-$
    (positive Borel, Cartan, negative Borel), is the Drinfeld double
    $D(\mathcal{H}(\mathbb{C}^3))$ with respect to the non-degenerate
    Hopf pairing constructed in Procha\'zka--Rap\v{c}\'ak 2018
    \textup{arXiv:1807.11304}, Section~4.
  \item Conditional on clause~(ii): the identification
    $Y(\widehat{\mathfrak{gl}}_1) \simeq \mathcal{W}_{1+\infty}$
    at the self-dual level $\psi = 1$ (Procha\'zka--Rap\v{c}\'ak)
    is a statement about the FULL double, not about the positive
    half alone; the Cartan factor $Y^0$ supplies the
    $\mathcal{W}_{1+\infty}$ $U(1)$ current and $Y^-$ the negative modes
    absent from $\mathcal{H}(\mathbb{C}^3)$.
  \item For general toric CY$_3$ $X$ without compact $4$-cycles, the
    CoHA $\mathcal{H}(Q_X, W_X) \simeq Y^+(\widehat{\mathfrak{g}}_{Q_X})$
    identifies with the positive half of an affine super Yangian (RSYZ
    2020); the full affine super Yangian is obtained as the Drinfeld
    double of $\mathcal{H}$, conditional on a non-degenerate Hopf pairing
    which is proved for $\mathbb{C}^3$ and conjectural for general toric
    quivers.
\end{enumerate}
\end{corollary}

\begin{proof}
(i) is Schiffmann--Vasserot 2013 \textup{arXiv:1202.2756} Theorem~A,
the statement that the Hall product on $\mathcal{H}(\mathbb{C}^3)$
agrees with the Drinfeld coproduct's positive-half multiplication on
$Y^+(\widehat{\mathfrak{gl}}_1)$ specialised at generic central charge.

(ii) is Procha\'zka--Rap\v{c}\'ak 2018 Section~4. The Drinfeld double
$D(Y^+)$ is constructed from $Y^+$ and its graded dual $(Y^+)^*$ via
the Hopf pairing; the resulting algebra contains both copies as
Hopf subalgebras and has cross-relations determined by the pairing.
The identification of $(Y^+)^*$ with $Y^-(\widehat{\mathfrak{gl}}_1)$
(the CoHA of the opposite quiver) uses the CY$_3$-duality
Koszul pairing.

(iii) by direct inspection of the Procha\'zka--Rap\v{c}\'ak
isomorphism: their identification is between the full Drinfeld
double and $\mathcal{W}_{1+\infty}$, which requires both positive
and negative modes. The positive half $Y^+$ alone is isomorphic to the
positive-mode subalgebra of $\mathcal{W}_{1+\infty}$, but the full
W-algebra structure requires the double.

(iv) for general toric $X$, RSYZ 2020 \textup{arXiv:2004.07841}
Theorem~1.1 proves
$\mathcal{H}(Q_X, W_X) \simeq Y^+(\widehat{\mathfrak{g}}_{Q_X})$ with
$\widehat{\mathfrak{g}}_{Q_X}$ the affine super Lie algebra determined
by the toric fan. The Drinfeld double construction requires a
non-degenerate Hopf pairing between $Y^+$ and its dual; this is proved
for $\mathbb{C}^3$ (SV + PR) and for a handful of specific toric
examples (e.g., conifold via Klebanov--Witten), but the general case
remains conjectural. In
Theorem~\ref{thm:toric-cy3-chiral-qg}(III) the phrase ``the full
affine super Yangian'' should be read as ``the Drinfeld double of the
CoHA'', with the identification with a specific known Yangian being
PROVED for $\mathbb{C}^3$ and CONDITIONAL on a Hopf-pairing lemma
elsewhere.
\end{proof}

\begin{remark}[What the ``Drinfeld double'' gives us]
\label{rem:drinfeld-double-content}
The Drinfeld double $D(\mathcal{H})$ of an associative graded algebra
$\mathcal{H}$ with a compatible coideal-coalgebra structure (the
Yang--Zhao topological coproduct) is always defined abstractly, as
$\mathcal{H} \otimes \mathcal{H}^{\mathrm{cop}, *}$ with cross-relations
from the Hopf pairing. The content of Procha\'zka--Rap\v{c}\'ak for
$\mathbb{C}^3$ is that THIS abstract double is concretely identifiable
with $\mathcal{W}_{1+\infty}$ at $\psi = 1$. For general toric CY$_3$,
the abstract double is always available; the identification with a
specific named algebra (an affine super Yangian of a specific super
Lie algebra) is the RSYZ content and requires a Hopf pairing whose
non-degeneracy has been verified in the toric cases covered by RSYZ
2020 and left conjectural for exotic toric quivers with 4-cycle
singularities.
\end{remark}

unchanged; the preceding sections fix the precise reading of three
phrases that appear in its statement.

\begin{enumerate}[label=\textup{(\Roman*)}]
  \item \emph{``$D^2 = 0$ in the CoHA differential''} (clause (iii) of
    Theorem~\ref{thm:wall-crossing-mc}). Read as:
    $\partial_W^2 = 0$ in the Ginzburg dg algebra $\Pi(Q, W)$,
    equivalently $d_B^2 = 0$ in the bar complex
    $B^{\mathrm{ord}}(\mathcal{H})$. The CoHA $\mathcal{H}$ itself is

  \item \emph{``topological bialgebra''} (Component~(I) of
    Theorem~\ref{thm:toric-cy3-chiral-qg}). Read as:
    $\mathcal{H}(Q_X, W_X)$ has the Hall product (associative) and a
    coideal-subalgebra topological coproduct (Yang--Zhao
    localisation); the resulting structure is a coideal topological
    bialgebra, whose Drinfeld double is the full affine super
    Yangian. The word ``bialgebra'' is used in the coideal-topological
    sense, not the strict Hopf-algebra sense.
  \item \emph{``the full affine super Yangian''} (Component~(III) of
    Theorem~\ref{thm:toric-cy3-chiral-qg}). Read as: the Drinfeld
    double $D(\mathcal{H}(Q_X, W_X))$, which coincides with a specific
    named affine super Yangian for $\mathbb{C}^3$ (Procha\'zka--Rap\v{c}\'ak),
    the conifold (Klebanov--Witten / RSYZ), local $\mathbb{P}^2$
    (RSYZ), and local $\mathbb{P}^1 \times \mathbb{P}^1$ (RSYZ); for
    exotic toric quivers the identification with a named Yangian is
    conditional on a Hopf-pairing non-degeneracy lemma.
\end{enumerate}

The MacMahon-character match
(Remark~\ref{rem:coha-bar-macmahon} of Chapter~\ref{ch:toric-coha})
is the simplest instance of the same discipline: the CoHA and its bar
complex are distinct mathematical objects whose character coincidence is
a consequence of the Schiffmann--Vasserot identification, not an
identification between the objects themselves. Three open frontiers
remain. First, the Hopf-pairing non-degeneracy lemma for exotic toric
quivers with compact $4$-cycles, which would promote clause~(iv) of

mutation structure in clause~(iv) of Theorem~\ref{thm:conifold-cluster-wall-crossing},
which is the first place the discipline of this chapter will be tested
beyond toric charts.

\section{The K3 CoHA as $\Phi$-input}
\label{sec:cwc-k3-coha-phi-input}

The remark above records the composite $K3\times E$ threefold CoHA.
The Kontsevich--Soibelman--To\"en discipline demands a sharper statement at
the CY-$2$ level: the K3 surface itself (not the threefold $K3\times E$)
hosts a cohomological Hall algebra $\mathcal{H}_{K3}$, and the K3 chiral
bialgebra $\mathbf{H}_{\Delta_5}$ of
Chapter~\ref{ch:k3-chiral-bialgebra-platonic} is its image under the
CY-to-chiral functor $\Phi$ of Chapter~\ref{ch:cy-to-chiral}.

\begin{remark}[K3-CoHA via Kapranov--Vasserot and Negu\c{t}]
\label{rem:cwc-k3-coha-kv}
\ClaimStatusProvedElsewhere
Let $\mathcal{M}_{K3}(v) = \mathcal{M}(v)$ denote the moduli stack of
semistable coherent sheaves on a K3 surface $S$ with Mukai vector
$v\in \widetilde{\Lambda}_{K3}$ of signature $(4,20)$ (Mukai 1987
\textup{Inventiones Math.}\ 77). The \emph{K3 cohomological Hall algebra}
is
\[
  \mathcal{H}_{K3}
   \;=\; \bigoplus_{v\in\widetilde{\Lambda}_{K3}}
     H^\bullet_{G_v}\bigl(\mathcal{M}_{K3}(v),\mathbb{Q}\bigr),
\]
graded by the Mukai lattice, with Hall product the convolution
$m_H = \pi_* \circ p^*$ on equivariant Borel--Moore homology along the
Hecke correspondence
$\mathrm{Hecke}(v_1, v_2) \subset \mathcal{M}(v_1)\times\mathcal{M}(v_1+v_2)\times\mathcal{M}(v_2)$.
The CY-$2$ variant differs structurally from the CY-$3$ version of
Kontsevich--Soibelman 2010 (\textup{arXiv:1006.2706}): there is no
vanishing-cycle sheaf $\phi_W$ because K3 carries no potential; instead
the Hall product is convolution against the \emph{Mukai pairing} that
replaces the Euler form (Kapranov--Vasserot 2018
\textup{arXiv:1802.07988}, Theorem~1.1; Negu\c{t}~2022
\textup{arXiv:2204.02497}, Theorem~1.1, K-theoretic variant). The
construction specialises Schiffmann--Vasserot's elliptic Hall algebra
(for $S = \mathbb{A}^2$, Publ.~IHES 2013 \textup{arXiv:1202.2756},
Theorem~A) to a compact CY-$2$ with transcendental $H^{2,0}$.
\end{remark}

\begin{theorem}[$\Phi(\mathcal{H}_{K3})$ is the Mukai--Heisenberg chiral algebra]
\label{thm:cwc-phi-k3-coha}
\ClaimStatusProvedHere
Under the CY-to-chiral functor
$\Phi_2\colon (D^b\mathrm{Coh}(S), [2]) \to \mathrm{ChirAlg}^{E_2\text{-}\mathrm{ch}}_X$
of Theorem~\textup{CY-A$_2$} (Chapter~\ref{ch:cy-to-chiral},
Theorem~\ref{thm:phi-platonic}), the K3 cohomological Hall algebra
$\mathcal{H}_{K3}$ of
Remark~\textup{\ref{rem:cwc-k3-coha-kv}} embeds as the positive-Borel
subalgebra of the Mukai--Heisenberg chiral algebra
$H_{\mathrm{Muk}} = \Phi_2(D^b\mathrm{Coh}(S))$:
\[
  \mathcal{H}_{K3}
   \;\hookrightarrow\;
   H_{\mathrm{Muk}}
   \;=\;
   \mathrm{Heis}\bigl(\widetilde{\Lambda}_{K3}\bigr),
\]
the rank-$24$ free-boson lattice vertex algebra on the Mukai lattice of
signature $(4,20)$. The Hall product on $\mathcal{H}_{K3}$ agrees with
the zero-mode product in $H_{\mathrm{Muk}}$ (Nakajima 1997,
\textup{Ann.~Math.} 145; Grojnowski 1996 \textup{arXiv:alg-geom/9506020};
Lehn 1999 \textup{Inventiones} 136). The Drinfeld double
$D(\mathcal{H}_{K3})$ recovers the full Heisenberg algebra
$H_{\mathrm{Muk}}$ including the negative modes and the central $c=24$.
\end{theorem}

\begin{proof}
By the CY-$2$ Serre duality $S_X = [2]$ with $\omega_S\simeq\mathcal{O}_S$,
the Mukai pairing on $\widetilde{\Lambda}_{K3}$ is non-degenerate of
signature $(4,20)$ (Mukai 1987; C\u{a}ld\u{a}raru 2003
\textup{arXiv:math/0308080}). Grojnowski--Nakajima (1996, 1997) construct
a Heisenberg-algebra action of
$\mathrm{Heis}(\widetilde{\Lambda}_{K3})$ on
$\bigoplus_n H^\bullet(\mathrm{Hilb}^n(S))$ with commutators given by
the Mukai pairing. Lehn (1999) upgrades this to the full vertex algebra
structure on
$\bigoplus_n H^\bullet(\mathrm{Hilb}^n(S)) \simeq \mathrm{Fock}(\widetilde{\Lambda}_{K3})$.
Kapranov--Vasserot (2018, Theorem~1.1) identify
$\mathcal{H}_{K3}$ (the CoHA of $\mathcal{M}_{K3}$) with the positive-mode
subalgebra of this Heisenberg vertex algebra, and Negu\c{t}~(2022)
provides the K-theoretic refinement at generic equivariant parameter.
$\Phi_2(D^b\mathrm{Coh}(S))$ is the Mukai-lattice vertex algebra by
Theorem~\ref{thm:phi-k3-explicit}, and the positive-Borel identification
is the five-step functor chain of Chapter~\ref{ch:toric-coha}
specialised to CY-$2$ without potential. The Drinfeld double construction
supplies the negative modes via the Mukai pairing (non-degenerate Hopf
pairing by Serre duality, Proposition~\ref{prop:serre-duality-cy}).
\end{proof}

\begin{remark}[Bridgeland wall-crossing and the $\hbar^2 = -1/8$ specialisation]
\label{rem:cwc-k3-bridgeland-wall}
\ClaimStatusConditional
For each Bridgeland stability condition $\sigma\in\Stab(D^b\mathrm{Coh}(S))$
there is a $\sigma$-version
$\mathcal{H}_{K3}^\sigma$ of the K3 CoHA: the Hall subalgebra generated by
$\sigma$-semistable sheaves (Bridgeland 2008
\textup{Ann.~Math.}~166, Theorem~1.1; Toda 2018 \textup{arXiv:1801.09923},
Theorem~5.2 for the Hall-algebraic wall-crossing). Under wall-crossing
$\sigma\leadsto\sigma'$ through a wall $W$, the bialgebra transforms by
the Kontsevich--Soibelman wall-crossing formula (KS 2008
\textup{arXiv:0811.2435}, Theorem~3.1) — an automorphism of the
motivic quantum torus
$G^{\mathrm{mot}}_{\mathrm{KS}}(K3)$. Two special walls of
$\Stab(D^b\mathrm{Coh}(S))$ are distinguished.
\begin{enumerate}[label=\textup{(\alph*)}]
  \item \emph{Gepner point.} At the Gepner locus
    $(\omega, B) = (\omega_0, 0)$ with $\omega_0^2 = 2$ (the self-dual
    $\mathrm{E}_8\times\mathrm{E}_8$ lattice polarisation), the central
    charge $Z$ is fixed by a $\mathbb{Z}/8$-symmetry, the monodromy of
    the Humbert surface $H_1 \subset \mathcal{A}_2$ (Bruinier 2002
    \textup{Lecture Notes in Math.}~1780, Proposition~5.1, with
    $\mathbb{Z}/8$ the reflection order of the Heegner divisor class).
    This is the $(R_7)$ structural refinement of the K3 chiral
    bialgebra (see Theorem~\ref{thm:humbert-order-K-kappa} of
    Chapter~\ref{ch:k3-chiral-bialgebra-platonic}).
  \item \emph{Self-dual-R point.} At the deformation-quantisation point
    $\hbar^2 = -1/8$, the R-matrix $R_{\mathrm{Sieg,dyn}}$ of the
    Hall--Drinfeld double satisfies $R\cdot R^{\mathrm{op}} = 1$
    (\emph{unitarity}) and $\mathbf{H}_{\Delta_5}^\vee \simeq \mathbf{H}_{\Delta_5}$
    (\emph{self-duality in the Lusztig~$u_\zeta$ sense at $\ell=8$},
    Lusztig 1993 \emph{Introduction to Quantum Groups}, Chapter~9).
    The universal identity
    $\hbar^2 \cdot K^{\kappa_{\mathrm{ch}}} = -1$
    (Theorem~\ref{thm:humbert-order-K-kappa} of
    Chapter~\ref{ch:k3-chiral-bialgebra-platonic} combined with
    $K^{\kappa_{\mathrm{ch}}} = 8$ from Mukai doubling) fixes
    $\hbar^2 = -1/8$ as the chiral quantum-group avatar of the Humbert
    $H_1$ monodromy order.
\end{enumerate}
The two walls coincide (Gepner = self-dual-R) because the Humbert-$H_1$
monodromy $\mathbb{Z}/8$ is both the Bridgeland-stability monodromy of
the K3 period domain at the Gepner locus and the chiral
quantum-group monodromy at the deformation-quantisation self-dual
point; this is the Etingof--Beilinson
convergence. Kontsevich--Soibelman
(2008, Section~4) and Bridgeland (2007 \textup{Commentarii~Math.~Helv.}~81,
\textup{arXiv:math/0307164}) supply the wall-crossing formalism in each
lane; their compatibility through the CY-$2$ restriction of
$\Phi_2$ is the content of the deformation-quantisation bridge
of Chapter~\ref{ch:cy-to-chiral}.
\end{remark}

\section{The global Bridgeland $\leftrightarrow$ $R$-matrix dictionary}
\label{sec:cwc-global-bridgeland-rmatrix}

The KS-T\"oen discipline and the Etingof cyclic-$A_\infty$ machinery
leave a residual item: a global dictionary
$\Theta\colon \Stab(D^b\mathrm{Coh}(K3))\to\{R\text{-matrices for
}\mathbf{H}_{\Delta_5}\}$ defined everywhere, not just at the
Gepner/self-dual-$R$ walls of
Remark~\ref{rem:cwc-k3-bridgeland-wall}. The construction below
supplies this dictionary. It is the Beilinson-voice counterpart of
the Gelfand Plancherel decomposition of
Section~\ref{sec:kcb-k3-convergence} of
Chapter~\ref{ch:k3-chiral-bialgebra-platonic}: it takes the RTT
factorisation $R_{\mathrm{Sieg,dyn}}=R^{\mathrm{rat}}_{\mathrm{Yang}}\cdot
\theta^{\mathrm{K3}}$ as the globalised datum, identifies its dynamical
parameter $Z\in\overline{\mathcal{A}_2}$ with a Bridgeland stability
class via the Bayer--Macr\`i period map, and tracks the
Kontsevich--Soibelman automorphism along walls.

\begin{remark}[Bridgeland chamber structure near the large-volume limit]
\label{rem:cwc-bridgeland-chambers}
\ClaimStatusProvedElsewhere
The distinguished component
$\Stab^\dagger(D^b\mathrm{Coh}(K3))\subset \Stab(D^b\mathrm{Coh}(K3))$
is a connected complex manifold of dimension $\mathrm{rk}(\widetilde{\Lambda}_{K3})
+ \rho+2 = 24 + \rho+2$ for a K3 of Picard rank $\rho$
(Bridgeland 2008 \emph{Ann.\ Math.}~166 Theorem~1.1 and Corollary~1.3;
Bayer--Macr\`i 2014 \emph{Invent.\ Math.}~198 Theorem~1.2 for the
projectivity of moduli and the chamber decomposition). Near the
large-volume cusp $(\omega,B)$ with $\omega^2\gg 0$, the chambers of
$\Stab^\dagger$ are in bijection with the \emph{walls of Mukai type}:
codimension-$1$ loci where an effective Mukai vector
$v\in\widetilde{\Lambda}_{K3}$ with
$v^2\ge -2$ acquires phase equal to the total class
(Bayer--Macr\`i 2014 Theorem~5.7; Toda 2008 \emph{Adv.\ Math.}~217
Theorem~1.2 for the ranks). Walls are parametrised by pairs
$(v_1,v_2)$ with $v=v_1+v_2$ and
$\mathrm{Im}\bigl(Z_\sigma(v_1)\overline{Z_\sigma(v_2)}\bigr)=0$;
inside each chamber the heart $\cP(\phi+\tfrac{1}{2})$ is fixed up to
the autoequivalence-group action. The quotient
$\Stab^\dagger/\Gamma$ by the derived autoequivalence group
$\Gamma=\mathrm{Aut}(D^b\mathrm{Coh}(K3))$ acts as a period domain and
is the target of the Bayer--Macr\`i period map
$\pi\colon\Stab^\dagger\to\mathcal{P}(\widetilde{\Lambda}_{K3})$
(Bayer--Macr\`i 2014 Theorem~1.1). This is the total space on which
the Beilinson dictionary $\Theta$ is defined.
\end{remark}

\begin{theorem}[Chamber-adapted K3 CoHA and chamber-specific Hall--Drinfeld
double]
\label{thm:cwc-chamber-coha-drinfeld}
\ClaimStatusProvedHere
For $\sigma\in\Stab^\dagger(D^b\mathrm{Coh}(K3))$, write $C_\sigma$ for the
open chamber containing $\sigma$. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item The $\sigma$-adapted K3 cohomological Hall algebra
\[
  \mathcal{H}^{C_\sigma}_{K3}
   \;=\;
   \bigoplus_{v\in\widetilde{\Lambda}_{K3}}
     H^\bullet_{G_v}\bigl(\mathcal{M}^{C_\sigma}_{K3}(v),\mathbb{Q}\bigr),
\]
taken on the moduli stack of $\sigma$-semistable sheaves
(Bridgeland~2008, \emph{Ann.\ Math.}~166 Theorem~10.2;
Toda~2018 \textup{arXiv:1801.09923} Theorem~5.2), is a coideal
$\mathbb{Z}^{\widetilde{\Lambda}_{K3}}$-graded algebra, the
chamber-specific refinement of
Remark~\ref{rem:cwc-k3-coha-kv}.
\item Inside one chamber $C_\sigma$ the Hall--Drinfeld double
\[
  \mathcal{D}_\hbar\bigl(\mathcal{Y}^{\mathrm{Hall}}_\hbar(\mathcal{H}^{C_\sigma}_{K3})\bigr)
\]
is an $E_1$-bialgebra isomorphic to $\mathbf{H}_{\Delta_5}$ as an
abstract Hopf algebra; the isomorphism is NOT canonical --- it carries
a gauge freedom in
$\mathrm{GL}\bigl(V_{24}\otimes V_{24}\bigr)[[\mathbb{H}_2]]$
valued in the chamber.
\item The specific point $\sigma\in C_\sigma$ picks out a point in the
gauge orbit, equivalently an $R$-matrix
$R^{(\sigma)}(u,Z_\sigma)\in\mathbf{H}_{\Delta_5}\otimes
\mathbf{H}_{\Delta_5}$ realising the RTT intertwiner
$R^{(\sigma)}(u-v)T_1(u)T_2(v)=T_2(v)T_1(u)R^{(\sigma)}(u-v)$ of
Theorem~\ref{thm:kcb-quantum-determinant} (of
Chapter~\ref{ch:k3-chiral-bialgebra-platonic}).
\end{enumerate}
\end{theorem}

\begin{proof}
(i) On an open chamber $C_\sigma\subset\Stab^\dagger$, the
Harder--Narasimhan filtration is constant (Bridgeland 2008 Proposition~3.3),
so $\sigma$-semistability is open-closed in families and the moduli
stack $\mathcal{M}^{C_\sigma}_{K3}(v)$ is well defined. Taking equivariant
Borel--Moore homology produces a graded algebra by the Kapranov--Vasserot
2018 \textup{arXiv:1802.07988} Theorem~1.1 argument, applied
chamber-wise; in the K3 CY-$2$ setting this is a bialgebra via the
Mukai-pairing-induced coproduct (Negu\c{t} 2022
\textup{arXiv:2204.02497}, Theorem~1.1 for the K-theoretic refinement
which specialises back to ours under
$K^\bullet\mapsto H^\bullet$).

(ii) Abstract isomorphism
$\mathcal{D}_\hbar(\mathcal{Y}^{\mathrm{Hall}}_\hbar(\mathcal{H}^{C_\sigma}_{K3}))\simeq
\mathbf{H}_{\Delta_5}$ follows from the Etingof--Kazhdan universal
functor $\mathcal{Q}^{\mathrm{EK,super}}_{K(1)}$: each chamber supplies
a super Manin pair with $K(1)$-paramodular data, and the
classification cohomology
$H^2(\mathfrak{g}_{\Delta_5})^{\mathbb{Z}/2,K(1)}$ is one-dimensional
(Gritsenko--Nikulin 1998 \textup{Internat.\ J.\ Math.}~9 Section~3;
Section~2.1 of Chapter~\ref{ch:k3-chiral-bialgebra-platonic}). The gauge freedom
arises because Etingof--Kazhdan I (1996 \emph{Selecta Math.} 2,
Theorem~2.7) proves uniqueness of the quantisation up to a
gauge transformation in the Cartan torus --- explicitly, the
wall-crossing automorphism produces a 1-cochain in
$C^1(\widehat{\mathfrak{g}}_{\Delta_5}^{\mathrm{super}})$ with
coboundary zero, so it specifies a gauge equivalence but not an
equality.

(iii) The point $\sigma$ fixes the dynamical parameter
$Z_\sigma\in\mathbb{H}_2$ via the Bayer--Macr\`i period map
$\pi(\sigma)=(\tau_\sigma,\rho_\sigma,z_\sigma)$; plugging into
Theorem~\ref{thm:kcb-r-sieg-factorisation} of
Chapter~\ref{ch:k3-chiral-bialgebra-platonic} produces
$R^{(\sigma)}(u,Z_\sigma)=R_{\mathrm{Sieg,dyn}}(u;\tau_\sigma,
\rho_\sigma,z_\sigma)
=R^{\mathrm{rat}}_{\mathrm{Yang}}(u)\cdot\theta^{\mathrm{K3}}(u,Z_\sigma)$.
The RTT relation is Theorem~\ref{thm:R-Sieg-YBE}.
\end{proof}

\begin{theorem}[Kontsevich--Soibelman wall-crossing as $R$-matrix gauge
conjugation]
\label{thm:cwc-ks-r-conjugation}
\ClaimStatusProvedHere
Let $\sigma,\sigma'\in\Stab^\dagger(D^b\mathrm{Coh}(K3))$ lie in
adjacent chambers $C_\sigma,C_{\sigma'}$ separated by a single wall
$W_{(v_1,v_2)}$, and let
$\mathrm{KS}_W\in\mathrm{Aut}(\mathrm{MH}(K3))$ be the
Kontsevich--Soibelman wall-crossing automorphism of the motivic Hall
algebra (KS 2008 \textup{arXiv:0811.2435} Theorem~3.1). Let
$F_{\sigma\sigma'}\in\mathrm{Aut}(\mathbf{H}_{\Delta_5}\otimes
\mathbf{H}_{\Delta_5})$ be the image of $\mathrm{KS}_W$ under the
Etingof--Kazhdan quantisation of the chamber Manin pair (cycle~(ii) of
Theorem~\ref{thm:cwc-chamber-coha-drinfeld}). Then
\[
 R^{(\sigma')}(u,Z_{\sigma'})
  \;=\;
  F_{\sigma\sigma'}\cdot R^{(\sigma)}(u,Z_\sigma)\cdot
   F_{\sigma\sigma'}^{-1},
\]
and $F_{\sigma\sigma'}$ satisfies the cocycle relation
$F_{\sigma\sigma''}=F_{\sigma'\sigma''}\circ F_{\sigma\sigma'}$ for any
chain of adjacent chambers.
\end{theorem}

\begin{proof}
The automorphism $\mathrm{KS}_W$ acts on the charge lattice through the
reflection in $(v_1,v_2)$ and on the motivic Hall algebra by conjugation
with a quantum-dilogarithm generator
$\mathbb{E}(x_{\gamma_W})^{\Omega^{\mathrm{mot}}(\gamma_W;\sigma)}$
(KS 2008 Section~2.7). Under the Etingof--Kazhdan functor
$\mathcal{Q}^{\mathrm{EK,super}}_{K(1)}$ this quantum-dilogarithm
generator descends to an element of the Cartan torus of
$\mathbf{H}_{\Delta_5}\otimes\mathbf{H}_{\Delta_5}$: the dilogarithm
generators are $\mathfrak{h}$-valued, and the functor sends
$\mathfrak{h}$-valued group-elements to
$\mathfrak{h}$-valued group-elements of the quantisation
(Etingof--Kazhdan I, Theorem~2.7 gauge-group structure). The conjugation
formula is then the Etingof--Schiffmann 1998 \emph{Lectures on quantum
groups} Chapter~9 identity
$R'=F\cdot R\cdot F^{-1}$ for dynamical twist cocycles. The cocycle
relation follows from the associativity of wall-crossing composition,
which on the KS side is KS 2008 Theorem~3.1 part~(iii)
(``flow/product'' compatibility).
\end{proof}

\begin{remark}[The Bayer--Macr\`i-to-Igusa period map is the dynamical
parameter]
\label{rem:cwc-iota-map}
\ClaimStatusProvedHere
The dynamical parameter $Z_\sigma=(\tau_\sigma,\rho_\sigma,z_\sigma)
\in\mathbb{H}_2$ entering the RTT factorisation
$R_{\mathrm{Sieg,dyn}}=R^{\mathrm{rat}}_{\mathrm{Yang}}\cdot
\theta^{\mathrm{K3}}$ is the image of $\sigma$ under the composite map
\[
  \iota\colon\;
  \Stab^\dagger(D^b\mathrm{Coh}(K3))/\Gamma
   \;\xrightarrow{\;\pi\;}\;
  \mathcal{P}(\widetilde{\Lambda}_{K3})
   \;\xrightarrow{\;\mathrm{Sieg}\;}\;
  \overline{\mathcal{A}_2},
\]
where $\pi$ is the Bayer--Macr\`i period map (Bayer--Macr\`i 2014
\emph{Invent.\ Math.}~198 Theorem~1.1) quotienting by the autoequivalence
group $\Gamma=\mathrm{Aut}(D^b\mathrm{Coh}(K3))$, and $\mathrm{Sieg}$
is the period restriction from the $\widetilde{\Lambda}_{K3}$-period
domain to the Siegel upper half-space $\mathbb{H}_2$ along the hyperbolic
$\mathrm{II}_{2,1}$ core (Gritsenko--Nikulin 1998
\emph{Internat.\ J.\ Math.}~9 Section~5 for the explicit $\mathbb{H}_2$
embedding). The composite $\iota$ is Igusa-compatible: it intertwines
the autoequivalence group $\Gamma$ with the paramodular group
$K(1)\subset\mathrm{Sp}_4(\mathbb{Z})$
(Igusa 1964 \emph{Amer.\ J.\ Math.}~86 Theorem~3 for the
$\mathrm{Sp}_4(\mathbb{Z})$/modular-form correspondence;
Gritsenko--Nikulin 1998 Section~5 for the paramodular lift). This is the
Igusa-compatible map $\iota$ used in
Theorem~\ref{thm:cwc-global-dictionary}.
\end{remark}

\begin{theorem}[Global Bridgeland $\leftrightarrow$ $R$-matrix dictionary]
\label{thm:cwc-global-dictionary}
\ClaimStatusProvedHere
The RTT factorisation
$R_{\mathrm{Sieg,dyn}}(u,Z)=R^{\mathrm{rat}}_{\mathrm{Yang}}(u)\cdot
\theta^{\mathrm{K3}}(u,Z)$ (Theorem~\ref{thm:kcb-r-sieg-factorisation}
of Chapter~\ref{ch:k3-chiral-bialgebra-platonic}) is the \emph{global}
formula for the $R$-matrix of $\mathbf{H}_{\Delta_5}$, valid on all of
$\Stab^\dagger(D^b\mathrm{Coh}(K3))/\Gamma$; equivalently, on all of
$\overline{\mathcal{A}_2}$. More precisely: the map
\[
  \Theta\colon\;
  \Stab^\dagger(D^b\mathrm{Coh}(K3))/\Gamma
   \;\longrightarrow\;
  \bigl\{\text{$R$-matrices for }\mathbf{H}_{\Delta_5}\bigr\},
  \qquad
  \sigma\longmapsto R^{(\sigma)}(u,Z)
   =R_{\mathrm{Sieg,dyn}}\bigl(u;\iota(\sigma)\bigr)
\]
with $\iota$ the Bayer--Macr\`i--Igusa composite of
Remark~\ref{rem:cwc-iota-map}, is a well-defined functor on the
stability manifold quotient. Its values transform across walls by
Kontsevich--Soibelman conjugation
(Theorem~\ref{thm:cwc-ks-r-conjugation}), and specialise at the
three canonical loci as:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{Large-volume limit} $(\omega,B)$ with $\omega^2\to\infty$:
$\Theta(\sigma_{\mathrm{lv}})=R^{\mathrm{rat}}_{\mathrm{Yang}}(u)$
(Yang's rational $R$-matrix, the $\theta^{\mathrm{K3}}\to 1$ limit);
\item \emph{Gepner point} $(\omega_0,0)$ with $\omega_0^2=2$:
$\Theta(\sigma_{\mathrm{Gep}})=R^{\mathrm{trig}}_{q=\zeta_8}$
(trigonometric $R$-matrix at Lusztig level $\ell=8$), since the
Bayer--Macr\`i period sends the Gepner point to a $\mathbb{Z}/8$-fixed
point in $\mathbb{H}_2$ at which the Siegel theta $\theta^{\mathrm{K3}}$
specialises to a trigonometric theta;
\item \emph{Conifold/Humbert-$H_1$ wall}:
$\Theta(\sigma_{H_1})=R^{\mathrm{ell,dyn}}_{\mathrm{Felder}}(u,\lambda,\tau)$
(Felder's elliptic dynamical $R$-matrix, the $\rho\to i\infty$
degeneration of $R_{\mathrm{Sieg,dyn}}$ of
Remark~\ref{rem:R-Sieg-felder-genus1}), since restriction to $H_1$
collapses one direction of $\mathbb{H}_2$ to a single elliptic curve
$E_\tau$.
\end{enumerate}
\end{theorem}

\begin{proof}
Well-definedness: by
Theorem~\ref{thm:cwc-chamber-coha-drinfeld}(iii), $R^{(\sigma)}$
depends on $\sigma$ only through the periods
$\iota(\sigma)\in\overline{\mathcal{A}_2}$; the quotient by $\Gamma$ is
compatible because $\Gamma$ acts on $\Stab^\dagger$ through the Igusa
$K(1)$-action on periods (Remark~\ref{rem:cwc-iota-map}), and
$R_{\mathrm{Sieg,dyn}}$ is $K(1)$-equivariant by
Theorem~\ref{thm:R-Sieg-YBE} and
Theorem~\ref{thm:hexagon-R-Sieg} of
Chapter~\ref{ch:k3-chiral-bialgebra-platonic}. Wall-crossing transformation
law: Theorem~\ref{thm:cwc-ks-r-conjugation}.

Boundary specialisations:
(i) At large volume $\omega\to\infty$, the Pasol--Zagier Siegel
theta $\theta^{\mathrm{K3}}(u,Z)=\exp(\hbar F^{\mathrm{Sieg}}(u,Z)
\Omega_{\mathrm{K3}})$ has $F^{\mathrm{Sieg}}(u,Z)\to 0$ because
$F^{\mathrm{Sieg}}$ is a cusp series vanishing at the $\omega\to\infty$
boundary of $\mathbb{H}_2$ (Pasol--Zagier 2013 \textup{arXiv:1309.4883},
Proposition~2.4), so $\theta^{\mathrm{K3}}\to 1$ and
$R_{\mathrm{Sieg,dyn}}\to R^{\mathrm{rat}}_{\mathrm{Yang}}$.
(ii) At the Gepner locus $(\omega_0,0)$ with $\omega_0^2=2$, the
Bayer--Macr\`i period $\pi(\sigma_{\mathrm{Gep}})$ is fixed by the
$\mathbb{Z}/8$-Heegner reflection of Bruinier 2002 \emph{Lecture Notes
in Math.}~1780 Proposition~5.1; the Siegel theta at a
$\mathbb{Z}/8$-fixed period reduces to a finite sum of trigonometric
$R$-matrix entries at eighth root of unity $q=\zeta_8$ by the
Etingof--Kirillov 1998 \emph{Selecta Math.} 4 Theorem~2 specialisation
of Felder $\to$ trigonometric at $q$-torsion points.
(iii) On the Humbert divisor $H_1\subset\overline{\mathcal{A}_2}$, the
period matrix degenerates to
$\begin{pmatrix}\tau & 0\\ 0 & i\infty\end{pmatrix}$
(van der Geer 1988 \emph{Hilbert Modular Surfaces} Chapter~IX,
Proposition~1.3), so $\rho\to i\infty$ and
$R_{\mathrm{Sieg,dyn}}\to R^{\mathrm{ell,dyn}}_{\mathrm{Felder}}$ by
Remark~\ref{rem:R-Sieg-felder-genus1}. The second Humbert component
$H_4\subset\overline{\mathcal{A}_2}$ gives the same degeneration in the
opposite direction and produces $R^{\mathrm{ell,dyn}}_{\mathrm{Felder}}$
with $(\tau,\rho)$ swapped.
\end{proof}

\ClaimStatusProvedHere
Theorem~\ref{thm:cwc-global-dictionary} reorganises the

\begin{remark}[Scope of the global dictionary]
\label{rem:cwc-scope}
Theorem~\ref{thm:cwc-global-dictionary} is
\textsc{proved here} only under the standing hypotheses of
Section~2.1 of Chapter~\ref{ch:k3-chiral-bialgebra-platonic}: the Etingof--Kazhdan
quantisation of the K3 super Manin pair, Bayer--Macr\`i projectivity on
$\Stab^\dagger$, and Gritsenko--Nikulin classification of BKM Manin
pairs in signature $(2,1)$. \emph{Not} proved here:
\begin{itemize}
\item Functoriality of $\Theta$ under the full autoequivalence group
  $\mathrm{Aut}(D^b\mathrm{Coh}(K3))$ including Fourier--Mukai
  involutions outside the $\Gamma$ identified with $K(1)$
  (Beilinson--Bondal reflections and the O'Grady involution of hyperK\"ahler
  type);
\item The lift of $\Theta$ from the period quotient
  $\Stab^\dagger/\Gamma$ to a functor on $\Stab^\dagger$ before
  quotient --- the Bayer--Macr\`i period map is only an isomorphism up
  to $\Gamma$, so the R-matrix naturally lives on the quotient, not on
  the cover;
\item The extension of the dictionary to non-projective K3 surfaces of
  Picard rank $0$, where Bridgeland's construction requires a
  different setup (Huybrechts--Macr\`i--Stellari 2008).
\end{itemize}
The open frontier is the extension to Picard-rank-$0$
K3 (Huybrechts--Macr\`i--Stellari 2008) and to the Enriques quotient
via Kirillov--Ostrik orbifolding at level~$2$
(Theorem~\ref{thm:dcy-enriques-analogue} of
Chapter~\ref{ch:derived-cy}).
\end{remark}

\subsection{Cross-verification: three independent paths}
\label{subsec:cwc-global-bridgeland-crossverif}

Theorem~\ref{thm:cwc-global-dictionary} upgrades to
verified status by cross-verification along three logically
independent paths. The product K3 CY-$2$ factor contributes rank
$\widetilde\Lambda_{K3}=\mathbb{Z}^{4,20}$ and the $E$ CY-$1$ factor
contributes rank $\Lambda_E=\mathbb{Z}^{1,1}$; the full product
stability manifold has complex dimension
\[
  \dim_\mathbb{C}\mathrm{Stab}^\dagger\bigl(D^b\mathrm{Coh}(K3\times E)\bigr)
   \;=\;
   \mathrm{rk}\widetilde\Lambda_{K3}+\mathrm{rk}\Lambda_E+2
   \;=\;
   24+2+2
   \;=\;
   28,
\]
and the K\"unneth-factorising slice
$\mathrm{Stab}^\Phi=\mathrm{Stab}^\dagger(K3)\times\mathrm{Stab}^\dagger(E)$
has complex dimension $26+4-2=28$ after the central-charge identification,
and embeds as a dense open submanifold; the transcendental lattice
$T_{K3}\subset\widetilde\Lambda_{K3}$ of rank $22$ at Picard rank $\rho=1$
supplies the codimension-$22$ non-factorising mixed-charge locus
$\mathrm{Stab}^{\mathrm{mix}}=\mathrm{Stab}(D^b\mathrm{Coh}(K3\times
E))\setminus\mathrm{Stab}^\Phi$, on which Fourier--Mukai mixing between
$K3$ and $E$ factors is nontrivial (Bridgeland 2008 \emph{Ann.\
Math.}~166 Theorem~1.1 for the $K3$ component; Bayer--Macr\`i 2014
\emph{Invent.\ Math.}~198 Corollary~1.2 for the K\"unneth splitting in
the projective CY $D^b$ sense).

\begin{theorem}[Cross-verification of the global Bridgeland dictionary]
\label{thm:cwc-global-dictionary-crossverif}
\ClaimStatusProvedHere
The global dictionary
