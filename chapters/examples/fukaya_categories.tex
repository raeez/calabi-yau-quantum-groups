\chapter{Fukaya Categories}
\label{ch:fukaya}

Symplectic geometry gives Calabi--Yau categories in every dimension
relevant to Vol~III, but the CY-to-chiral correspondence sees a sharp
boundary at $d=3$. For $d=2$ the Fukaya category feeds $\Phi$
unconditionally and produces an $\Etwo$-chiral algebra. For $d=3$ the
functor stops at the ordered $\Eone$ layer unless the chain-level
$\mathbb{S}^3$-framing exists. The Fukaya category $\Fuk(X)$ is the
test case for that boundary: its objects are Lagrangians, its
morphisms are Floer complexes, its higher compositions count
pseudoholomorphic polygons, and for Calabi--Yau $X$ it carries the
cyclic structure required by Chapter~\ref{ch:cyclic-ainf}. The modular
characteristic $\kappa_{\mathrm{cat}}$ extracted from
$\Phi(\Fuk(X))$ records symplectic information carried by $X$.


\section{The Fukaya category: definition and $\Ainf$-structure}
\label{sec:fukaya-def}

\begin{definition}[Fukaya category]
\label{def:fukaya-category}
Let $(X^{2n}, \omega)$ be a compact symplectic manifold. The
\emph{Fukaya category} $\Fuk(X)$ is the $\Ainf$-category whose:
\begin{enumerate}[label=(\roman*)]
 \item \emph{Objects} are compact, oriented, graded Lagrangian
 submanifolds $L \subset X$ equipped with a flat $U(1)$-connection
 (a local system);
 \item \emph{Morphism spaces} are Floer cochain complexes:
 for transverse $L_0, L_1$,
 \[
 \Hom_{\Fuk(X)}(L_0, L_1) = \mathrm{CF}^\bullet(L_0, L_1)
 = \bigoplus_{p \in L_0 \cap L_1} \Lambda_{\mathrm{nov}} \cdot p,
 \]
 where $\Lambda_{\mathrm{nov}} = \C[[T]][T^{-1}]$ is the Novikov
 field and the grading on $p$ is the Maslov index;
 \item \emph{Higher compositions} are
 $\mu_k \colon \mathrm{CF}^\bullet(L_0, L_1) \otimes \cdots
 \otimes \mathrm{CF}^\bullet(L_{k-1}, L_k)
 \to \mathrm{CF}^\bullet(L_0, L_k)[2-k]$,
 defined by counting pseudoholomorphic $(k+1)$-gons:
 \[
 \mu_k(p_1, \ldots, p_k)
 = \sum_{\substack{u \colon D^2 \to X \\ \bar\partial_J u = 0}}
 T^{\int u^*\omega} \cdot q(u)
 \]
 where $q(u)$ is the output intersection point and the sum
 is over $J$-holomorphic disks $u$ with boundary on
 $L_0 \cup \cdots \cup L_k$ and corners at
 $p_1, \ldots, p_k$.
\end{enumerate}
The $\Ainf$-relations $\sum_{i,j} \pm\, \mu_{k-j+1}(\ldots,
\mu_j(\ldots), \ldots) = 0$ follow from the compactification of
moduli spaces of pseudoholomorphic polygons (Fukaya--Oh--Ohta--Ono).
\end{definition}

\begin{remark}[Analytic foundations]
\label{rem:fukaya-analytic}
The rigorous construction of $\Fuk(X)$ requires virtual perturbation
techniques (Kuranishi structures, polyfolds, or algebraic approaches)
to handle transversality for the moduli spaces of holomorphic
polygons. The foundational references are Fukaya--Oh--Ohta--Ono for
Kuranishi structures, Hofer--Wysocki--Zehnder for polyfolds, and
Pardon for the algebraic virtual fundamental class. For this volume,
the $\Ainf$-structure on $\Fuk(X)$ is assumed constructed; the
CY-to-chiral correspondence programme $\{\Phi_d\}$ takes this structure as input.
\end{remark}

\begin{proposition}[CY structure on the Fukaya category]
\label{prop:fukaya-cy}
\ClaimStatusProvedElsewhere
For a compact CY manifold $(X^{2n}, \omega, \Omega)$ of complex
dimension $n$, the Fukaya category $\Fuk(X)$ is CY of dimension $n$:
\begin{enumerate}[label=(\roman*)]
 \item The CY pairing
 $\langle \cdot, \cdot \rangle \colon
 \mathrm{CF}^\bullet(L_0, L_1) \otimes
 \mathrm{CF}^\bullet(L_1, L_0)
 \to \Lambda_{\mathrm{nov}}[-n]$
 is the Poincar\'{e} duality pairing on Lagrangian intersections;
 \item The cyclic invariance of $\mu_k$ with respect to this pairing
 follows from the boundary of a pseudoholomorphic polygon being a
 cycle on $L_0 \cup \cdots \cup L_k$;
 \item Non-degeneracy follows from Poincar\'{e} duality on compact
 oriented manifolds.
\end{enumerate}
\end{proposition}


\section{CY $1$-folds: elliptic curves}
\label{sec:fukaya-cy1}

\begin{example}[Elliptic curve]
\label{ex:fukaya-elliptic}
For an elliptic curve $E_\tau = \C / (\Z + \tau\Z)$ with the
standard K\"ahler form $\omega = \frac{i}{2} dz \wedge d\bar{z}$:
\begin{enumerate}[label=(\roman*)]
 \item $\Fuk(E_\tau)$ is generated by the Lagrangian circles
 $L_\alpha = \{z \in E_\tau : \Im(e^{-i\alpha} z) = c\}$;
 \item $\Fuk(E_\tau)$ is CY of dimension $1$;
 \item The Hochschild homology
 $\HH_\bullet(\Fuk(E_\tau)) \simeq
 H^*_{\mathrm{dR}}(E_\tau)$ recovers the de Rham
 cohomology (Abouzaid);
 \item The CY-to-chiral correspondence produces the Heisenberg
 vertex algebra $H_k$ at level $k = \mathrm{vol}(E_\tau)$;
 \item The manifold invariant $\kappa_{\mathrm{cat}}(E_\tau) =
 \chi(\cO_{E_\tau}) = 0$, while the chiral invariant of the
 Heisenberg output satisfies $\kappa_{\mathrm{ch}}(H_k) = k$
 (the level, by the Heisenberg formula).
\end{enumerate}
\end{example}

\begin{remark}[Homological mirror symmetry for $E_\tau$]
\label{rem:hms-elliptic}
Polishchuk--Zaslow proved $\Fuk(E_\tau) \simeq D^b(\Coh(E_{\tau'}))$
where $\tau' = -1/\tau$ (mirror elliptic curve). Under this
equivalence, the Heisenberg algebra $H_k$ is the chiral algebra
of the free boson compactified on the dual circle. The chiral
invariant is preserved:
$\kappa_{\mathrm{ch}}(\Phi(\Fuk(E_\tau)))
 = \kappa_{\mathrm{ch}}(\Phi(D^b(\Coh(E_{\tau'}))))
 = k$, while $\kappa_{\mathrm{cat}} = \chi(\cO) = 0$ on both sides
is the manifold invariant and plays no role in the level-matching.
\end{remark}


\section{CY $2$-folds: K3 and abelian surfaces}
\label{sec:fukaya-cy2}

\begin{proposition}[K3 surface]
\label{prop:fukaya-k3}
\ClaimStatusProvedHere
For a K3 surface $S$ with the Ricci-flat K\"ahler metric:
\begin{enumerate}[label=(\roman*)]
 \item $\Fuk(S)$ is CY of dimension $2$;
 \item The correspondence $\Phi_2$ (Theorem~CY-A$_2$) produces an
 $\Etwo$-chiral algebra $A_{\Fuk(S)}$;
 \item The modular characteristic is
 $\kappa_{\mathrm{cat}}(\Phi(\Fuk(S))) = \chi(\cO_S) = 2$;
 \item The Hochschild homology
 $\HH_\bullet(\Fuk(S)) \simeq
 H^{*+2}(S, \C) \simeq \C^{24}$ (by HKR and CY duality);
 \item The $R$-matrix
 $\cR_{\Fuk(S)}(z) = 1 + \frac{\kappa_{\mathrm{cat}}\Omega}{z} + O(1)$
 with $\kappa_{\mathrm{cat}} = \chi(\cO_S) = 2$ encodes the
 Mukai pairing $\Omega$ on $H^*(S, \Z)$.
\end{enumerate}
\end{proposition}

\begin{proof}
Item (i): $S$ is a compact K\"ahler manifold with $c_1(S) = 0$ and
$\dim_\C S = 2$, so $\Fuk(S)$ is CY$_2$
(Proposition~\ref{prop:fukaya-cy}).

Item (ii): Theorem~CY-A$_2$ applies since $d = 2$ and the
$\mathbb{S}^2$-framing is unconditional.

Item (iii): the CY Euler characteristic is
$\chi^{\CY}(\Fuk(S)) = \chi(\cO_S) = 1 - 0 + 1 = 2$ (using
$h^{0,0} = 1$, $h^{1,0} = 0$, $h^{2,0} = 1$ for K3). By
Theorem~CY-D (Theorem~\ref{thm:cy-modular-characteristic}),
$\kappa_{\mathrm{cat}} = \chi^{\CY} = 2$.

Item (iv): HKR gives
$\HH_k(\Perf(S)) \simeq \bigoplus_{p-q=k} H^q(S, \Omega^p)$.
The Hodge diamond of $K3$ has rows
$h^{p,q} = (1,0,1)$ for $p=0$, $(0,20,0)$ for $p=1$,
$(1,0,1)$ for $p=2$, so
\[
 \HH_{-2} = H^2(\cO) = \C, \quad
 \HH_0 = H^0(\cO) \oplus H^1(\Omega^1) \oplus H^2(\Omega^2) = \C^{22}, \quad
 \HH_2 = H^0(\Omega^2) = \C,
\]
giving $\dim \HH_\bullet = 1 + 22 + 1 = 24$. CY duality
$\HH^\bullet \simeq \HH_{d-\bullet}$ shifts by $d = 2$.

Item (v): the $R$-matrix at degree $(1,1)$ of $B^{\mathrm{ord}}(A)$
has leading term $\frac{\kappa_{\mathrm{cat}}\Omega}{z}$ where
$\Omega \in \End(V \otimes V)$ is the Casimir of the Mukai lattice
$\Lambda_{K3} = U^4 \oplus E_8(-1)^2$ of signature $(4,20)$.
The subleading terms encode higher-order OPE data.
\end{proof}

\begin{example}[Abelian surface]
\label{ex:fukaya-abelian-surface}
For an abelian surface $A = E_1 \times E_2$ (product of two
elliptic curves):
\begin{enumerate}[label=(\roman*)]
 \item $\Fuk(A)$ is CY of dimension $2$;
 \item $\kappa_{\mathrm{cat}}(\Phi(\Fuk(A)))
 = \chi(\cO_A) = 0$ (since $h^{1,0}(A) = 2$,
 $h^{2,0}(A) = 1$, giving $\chi = 1 - 2 + 1 = 0$);
 \item The modular characteristic vanishes:
 $\kappa_{\mathrm{cat}} = 0$, so the bar complex is uncurved
 at all genera ($d^2 = 0$ strictly).
\end{enumerate}
The vanishing of $\kappa_{\mathrm{cat}}$ trivialises the scalar
channel of the genus tower; the higher shadow invariants
$S_k$ for $k \geq 3$ are controlled by the shadow class of the
bar complex, not by $\kappa_{\mathrm{cat}}$, and may still be
nonzero.
\end{example}


\section{CY $3$-folds}
\label{sec:fukaya-cy3}

For $d = 3$, the CY-to-chiral correspondence $\Phi_3$ is constructed by
Theorem~CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}). The output
chiral algebra is $\Eone$ (not $\Etwo$): the
structural $\Etwo$-obstruction from the antisymmetric Euler form
(Proposition~\ref{prop:e1-obstruction-categorical}) is generically nontrivial,
while the topological $\mathbb{S}^3$-framing obstruction vanishes
($\pi_3(BU) = 0$ by Bott periodicity).

\begin{remark}[Bott periodicity verification]
\label{rem:fukaya-bott-verification}
The stable unitary classifying space satisfies $\Omega BU \simeq U$,
so
\[
 \pi_k(BU) \cong \pi_{k-1}(U).
\]
Bott periodicity for $U$ gives
\[
 \pi_{2m-1}(U) = \Z, \qquad \pi_{2m}(U) = 0 \qquad (m \geq 1),
\]
hence
\[
 \pi_{2m}(BU) = \Z, \qquad \pi_{2m+1}(BU) = 0.
\]
In the low dimensions relevant here:
\[
 \pi_2(BU) = \Z, \qquad \pi_3(BU) = 0, \qquad \pi_4(BU) = \Z.
\]
These groups have different meanings in adjacent CY dimensions. At
$d = 2$, the nonzero class $\pi_2(BU) = \Z$ records the native
braiding parameter of the surface-level $\Etwo$ theory, so
Theorem~CY-A$_2$ is not a vanishing statement. At $d = 3$, the
vanishing of $\pi_3(BU)$ removes only the topological framing
obstruction; the native $\Etwo$ obstruction is the antisymmetric
Euler form of Proposition~\ref{prop:e1-obstruction-categorical}. At
$d = 4$, the group $\pi_4(BU) = \Z$ is an obstruction group for the
$\mathbb{S}^4$-framing, not a guarantee of an $\Etwo$ enhancement:
Chapter~\ref{ch:en-factorization} still yields only the
$\Eone$-stabilized conclusion with extra shifted data.
\end{remark}

\begin{conjecture}[CY$_3$ Fukaya chiral algebra]
\label{conj:fukaya-cy3}
\ClaimStatusConjectured
For a compact CY threefold $X$ with Fukaya category $\Fuk(X)$:
\begin{enumerate}[label=(\roman*)]
 \item The CY-to-chiral correspondence $\Phi_d(\Fuk(X))$ produces an
 $\Eone$-chiral algebra $A_{\Fuk(X)}$, conditional only on
 the chain-level $\mathbb{S}^3$-framing \textup{(}the topological
 obstruction already vanishes\textup{)};
 \item The chiral modular characteristic
 $\kappa_{\mathrm{ch}}(\Phi(\Fuk(X))) = \chi^{\CY}(\Fuk(X))$
 is determined by the categorical CY Euler characteristic
 (which may differ from $\chi(\cO_X)$ at $d = 3$; see
 Conjecture~\ref{conj:cy-kappa-identification});
 \item The braided monoidal category
 $\cZ(\Rep^{\Eone}(A_{\Fuk(X)}))$ (recovered via the Drinfeld
 center, Chapter~\ref{ch:drinfeld-center}) encodes the
 DT/GW invariants of $X$;
 \item The open-string sector ($\Fuk(X)$ with Lagrangian boundary
 conditions) connects to the Swiss-cheese structure of Volume~II.
\end{enumerate}
\end{conjecture}

\begin{remark}[What remains to remove the CY$_3$ condition]
\label{rem:fukaya-cy3-condition-gap}
The topological part of Conjecture~\ref{conj:fukaya-cy3} is already
gone: both Bott periodicity and the symplectic reduction path give
vanishing of the $\mathbb{S}^3$-framing obstruction. The remaining
gap is chain-level. One must construct an explicit trivialization of
the negative-cyclic CY class on the cyclic bar complex of $\Fuk(X)$
and prove that this trivialization intertwines the full family of
polygon operations $\mu_k$. Equivalently, the missing datum is an
$\mathbb{S}^3$-framing on $\HH_\bullet(\Fuk(X))$ compatible with the
Connes operator and the cyclic $\Ainf$ structure. Two plausible routes
are known: transfer of a chain-level trivialization across a proved
HMS equivalence, or a direct continuation-map construction on the
Fukaya cyclic bar complex. Neither route is completed in this
manuscript, so the condition is reformulated sharply but not removed.
\end{remark}

\begin{example}[Quintic threefold]
\label{ex:fukaya-quintic}
For the quintic threefold $X_5 \subset \bP^4$ defined by a
generic degree-$5$ polynomial:
\begin{enumerate}[label=(\roman*)]
 \item $\chi(X_5) = -200$ (topological Euler characteristic);
 \item $\chi(\cO_{X_5}) = 0$ (since $h^{0,0} = 1$,
 $h^{1,0} = 0$, $h^{2,0} = 0$, $h^{3,0} = 1$, giving
 $\chi = 1 - 0 + 0 - 1 = 0$);
 \item The manifold invariant is
 $\kappa_{\mathrm{cat}} = \chi(\cO_{X_5}) = 0$; the chiral
 modular characteristic $\kappa_{\mathrm{ch}}$ may differ at
 $d = 3$.
\end{enumerate}
The vanishing $\kappa_{\mathrm{cat}} = 0$ for the quintic is the
CY$_3$ analogue of the abelian surface ($\kappa_{\mathrm{cat}} = 0$
for CY$_2$). At the scalar level, the genus tower for
$\kappa_{\mathrm{cat}}$ is trivial; whether
$\kappa_{\mathrm{ch}}$ also vanishes is part of the $d = 3$
programme. The full shadow obstruction tower may be nontrivial
if higher-degree shadows are nonzero.
\end{example}

\begin{example}[Resolved conifold]
\label{ex:fukaya-conifold}
The resolved conifold $\cO(-1) \oplus \cO(-1) \to \bP^1$ is
non-compact with $\chi_{\mathrm{top}} = 2$. The wrapped Fukaya
category $\cW(\text{conifold})$ is generated by a single Lagrangian
$S^3$-cycle (wrapped variant; the compact variant $\Fuk$ is
trivial on the conifold). The CoHA of the conifold
(Chapter~\ref{ch:toric-coha}) is the positive half
$Y^+(\widehat{\fgl}_1)$ of the affine Yangian
(Schiffmann--Vasserot), not the full Yangian; the full
$Y(\widehat{\fgl}_1)$ arises as the Drinfeld double, conditional
on the Hopf-pairing data beyond the Hall-algebra structure.
\end{example}

\begin{remark}[Fukaya-side Hall--Drinfeld double via Lagrangian-moduli CoHA]
\label{rem:fuk-lagrangian-hall}
On the coherent side of the K3 programme, the Hall--Drinfeld double $\cY^{\mathrm{Hall}}_\hbar(\mathrm{CoHA}_{K3 \times E})$ of Chapter~\ref{ch:k3-chiral-bialgebra-platonic} is constructed from the cohomological Hall algebra of the CY-$3$ total space $K3 \times E$ (Schiffmann--Vasserot; Kontsevich--Soibelman). Under HMS, the natural Fukaya-side replacement is the \emph{symplectic-Floer Hall algebra} built from moduli of stable Lagrangians: replace coherent-sheaf moduli $\cM^{\mathrm{st}}(K3 \times E, \bv)$ by symplectic-Lagrangian moduli $\cM^{\mathrm{sLag}}(X^\vee, \bv)$ on the mirror $X^\vee$, and replace the Joyce $\star$-product (via convolution on the stack of short exact sequences) by the symplectic pair-of-pants product on Lagrangian Floer moduli. The resulting \emph{Lagrangian Hall algebra} $\mathrm{CoHA}^{\mathrm{sLag}}(X^\vee)$ is conjecturally HMS-equivalent to the coherent CoHA$_{K3 \times E}$ as a graded associative algebra (the Davison--Meinhardt integrality statement extends: BPS classes on the coherent side correspond to primitive Lagrangian classes on the symplectic side). Under this extension, the Hall--Drinfeld double construction on the Fukaya side produces a symplectic replacement of $\cY^{\mathrm{Hall}}_\hbar$ whose Drinfeld double identification with the K3 chiral bialgebra is conditional on: (i) HMS at the level of CY-$3$ derived moduli stacks, and (ii) the existence of a Joyce-style pair-of-pants associative structure on Lagrangian moduli compatible with stability. Neither is proved in this manuscript; the identification is the symplectic analogue of the BPS-algebraic route of Chapter~\ref{ch:toric-coha}.
\end{remark}


\section{Wrapped Fukaya categories}
\label{sec:wrapped-fukaya}

\begin{definition}[Wrapped Fukaya category]
\label{def:wrapped-fukaya}
For a Liouville manifold $(X, \lambda)$ (an exact symplectic
manifold with a convex end), the \emph{wrapped Fukaya category}
$\cW(X)$ modifies $\Fuk(X)$ by allowing wrapping at infinity:
the Floer differential and higher compositions include
Hamiltonian orbits of arbitrarily large period along the
contact boundary $\partial X$. The morphism space is the direct
limit
\[
 \Hom_{\cW(X)}(L_0, L_1)
 = \varinjlim_{H \to \infty} \mathrm{CF}^\bullet(L_0, \phi_H(L_1))
\]
where $\phi_H$ is the time-$1$ flow of a Hamiltonian $H$
that is linear with increasing slope at infinity.
\end{definition}

\begin{theorem}[Abouzaid: cotangent bundles]
\label{thm:abouzaid-cotangent}
\ClaimStatusProvedElsewhere
For a closed oriented manifold $M$, the wrapped Fukaya category of
the cotangent bundle satisfies
\[
 \cW(T^*M) \;\simeq\; \Mod_{C_*(\Omega M)},
\]
where $C_*(\Omega M)$ is the chain algebra of the based loop
space. Under this equivalence:
\begin{enumerate}[label=(\roman*)]
 \item The zero section $M \hookrightarrow T^*M$ corresponds to
 the trivial module $\C$ over $C_*(\Omega M)$;
 \item The wrapping corresponds to the bar differential on
 $B(C_*(\Omega M))$;
 \item The CY pairing (when $M$ is orientable of dimension $n$)
 comes from Poincar\'{e} duality on $M$.
\end{enumerate}
\end{theorem}

\begin{remark}[Wrapped categories and the CY-to-chiral correspondence]
\label{rem:wrapped-cy-chiral}
The wrapped Fukaya category is smooth but typically not proper
(morphism spaces are infinite-dimensional). The CY-to-chiral
correspondence $\Phi_d$ requires a cyclic $\Ainf$-structure with a
non-degenerate pairing; for non-compact targets, this requires
either:
\begin{enumerate}[label=(\alph*)]
 \item A compactly-supported variant (using the compact Fukaya
 category $\Fuk_{\mathrm{cpt}}(X)$ as a subquotient of
 $\cW(X)$);
 \item An analytic completion (the sewing envelope of Volume~I,
 MC5 analytic HS-sewing lane) to handle the non-compact directions.
\end{enumerate}
For cotangent bundles $T^*M$ with $M$ compact, approach (a)
suffices: $\Fuk_{\mathrm{cpt}}(T^*M)$ is proper, and $\Phi$
produces the chiral algebra of the $\sigma$-model on $M$.
\end{remark}


\section{Homological mirror symmetry and the CY-to-chiral correspondence}
\label{sec:hms-chiral}

Kontsevich's homological mirror symmetry conjecture identifies the
symplectic and algebraic categorifications of a CY manifold:
$\Fuk(X) \simeq D^b(\Coh(X^\vee))$. Under the CY-to-chiral correspondence
$\Phi$, this becomes an equivalence of chiral algebras at every level
that the correspondence sees.

\begin{proposition}[HMS implies chiral algebra equivalence]
\label{prop:hms-compatibility}
\ClaimStatusConditional
Let $X$ and $X^\vee$ be mirror CY manifolds of dimension $d$.
If the HMS equivalence $\Fuk(X) \simeq D^b(\Coh(X^\vee))$ holds as
an equivalence of CY$_d$ $\Ainf$-categories (preserving the cyclic
structure), then the CY-to-chiral correspondence produces an
equivalence of quantum chiral algebras
\[
 A_{\Fuk(X)} \;\simeq\; A_{D^b(\Coh(X^\vee))}
\]
as $\Etwo$-chiral algebras (for $d = 2$) or $\Eone$-chiral algebras
(for $d = 3$, conditional on $\mathbb{S}^3$-framing for both sides).
In particular the bar complexes are quasi-isomorphic as factorization
coalgebras, the shadow obstruction towers coincide at all degrees,
and the modular characteristics agree:
$\kappa_{\mathrm{cat}}(\Fuk(X))
 = \kappa_{\mathrm{cat}}(D^b(\Coh(X^\vee)))$.
\end{proposition}

\begin{proof}[Proof sketch]
The correspondence $\Phi_d$ factors through the cyclic $\Ainf$-structure
(Chapter~\ref{ch:cyclic-ainf}). An equivalence of CY$_d$ categories
preserving the cyclic pairing induces an isomorphism of cyclic bar
complexes, hence of factorization coalgebras after the factorization
envelope. The chiral algebra is determined by its factorization
coalgebra (Volume~I, Theorem~A), giving the equivalence of
$A_{\Fuk(X)}$ and $A_{D^b(\Coh(X^\vee))}$. The shadow tower and
$\kappa_{\mathrm{cat}}$ are derived invariants
(Theorem~\ref{thm:cy-modular-characteristic}), hence preserved by any
CY equivalence.
\end{proof}

\begin{remark}[Status of HMS]
\label{rem:hms-status}
The compatibility is formal: $\Phi$ is Morita invariant, so equivalent
CY categories produce equivalent chiral algebras. The conditional
content is HMS itself, proved for
\begin{enumerate}[label=(\alph*)]
 \item Elliptic curves (Polishchuk--Zaslow);
 \item Abelian varieties (Fukaya, Kontsevich--Soibelman);
 \item K3 surfaces (Seidel, Smith; partial results);
 \item Quintic threefold (Sheridan, for a suitable formulation);
 \item Toric varieties (Abouzaid, Fang--Liu--Treumann--Zaslow).
\end{enumerate}
For these cases the compatibility of $\Phi$ with HMS is unconditional.
\end{remark}

\begin{remark}[HMS gauge class and $H^2(\mathfrak{g}_{\Delta_5}) = \C \cdot \Delta_5$]
\label{rem:fuk-gauge-class-delta5}
The classification invariant of K3 chiral Hall--Drinfeld double quantisations of the Manin pair $(\mathfrak{g}_{\Delta_5}, \mathfrak{n}_+^{\mathrm{imag}} \oplus \mathfrak{h}^{\mathrm{imag},\mathrm{rk\,23}})$, under paramodular $K(1)$ restriction, is the one-dimensional Lie cohomology group
\[
  H^2\bigl(\mathfrak{g}_{\Delta_5}\bigr)^{\Z/2,\,K(1)}
  \;\cong\; \C \cdot \Delta_5,
\]
spanned by the Gritsenko--Nikulin Igusa cusp form (Chapter~\ref{ch:k3-chiral-bialgebra-platonic}, Theorem~\ref{thm:classification-invariant-delta5}; Etingof--Kazhdan super-category extension; Bruinier 2002, Proposition~$5.1$). Under HMS the coherent-side gauge class is transported to a Fukaya-side gauge class: the quasi-Lie-bialgebra cocycle defect of the K3 chiral Drinfeld double corresponds, on the A-model, to the associator-cocycle defect of the Fukaya CY-$2$ cyclic pairing restricted to the paramodular locus. Because the classification invariant is an HMS invariant of the CY category (the Mukai pairing is HMS-preserved, hence so are the derived moduli of Manin-pair quantisations), the Fukaya-side Igusa cusp form $\Delta_5$ is the \emph{same} modular form as the coherent-side class: the Fukaya side cannot see a different cusp form without violating HMS. HMS invariance of $\Delta_5$ under the Mukai pairing is the A-model shadow of the coherent-side classification invariant.
\end{remark}

\begin{remark}[Numerical HMS tests via shadow invariants]
\label{rem:hms-shadow-tests}
The shadow obstruction tower provides computable invariants that can
be independently evaluated on the symplectic (A-model) and algebraic
(B-model) sides. For CY surfaces, $\kappa_{\mathrm{cat}}$ and the
shadow depth $r_{\max}$ are accessible on both sides:
\begin{enumerate}[label=(\roman*)]
 \item \emph{A-model}: $\kappa_{\mathrm{cat}}$ is computed from the
 Fukaya cyclic bar complex; the shadow depth is determined by the
 $\Ainf$-operations $\mu^k$ (shadow depth $r_{\max} = k_0$ when
 $\mu^k = 0$ for all $k > k_0$ up to homotopy);
 \item \emph{B-model}: $\kappa_{\mathrm{cat}} = \chi(\cO_{X^\vee})$
 from algebraic geometry; the shadow depth is determined by the Ext
 algebra structure on $D^b(\Coh(X^\vee))$.
\end{enumerate}
Agreement of these invariants provides a quantitative test of HMS
beyond Hodge-number matching.
\end{remark}

\begin{remark}[SYZ fibrations and the Fukaya--chiral correspondence]
\label{rem:syz-chiral}
For CY threefolds admitting a special Lagrangian $T^3$-fibration
$\pi \colon X \to B$ (the SYZ picture), the Fukaya category decomposes
along the base: the fiber Fukaya categories $\Fuk(\pi^{-1}(b))$ for
smooth $b \in B_{\mathrm{sm}}$ are CY$_0$ (semisimple), while singular
fibers contribute non-trivial $\Ainf$-operations. The CY-to-chiral
correspondence applied fiberwise produces a family of chiral algebras
over $B$, with the singular-fiber contributions encoding instanton
corrections to the mirror map. The total chiral algebra
$A_{\Fuk(X)}$ is assembled as a homotopy colimit over the base,
paralleling the stability-condition assembly of
Chapter~\ref{ch:derived-cy}.
\end{remark}


\section{The quantum chiral algebra of a Fukaya category}
\label{sec:fukaya-qca}

\begin{construction}[Assembling $\Phi(\Fuk(X))$]
\label{constr:fukaya-phi}
For a CY manifold $X$ of dimension $d$, the quantum chiral algebra
$A_{\Fuk(X)} = \Phi(\Fuk(X))$ is assembled in three steps
(matching the abstract construction of Theorem~CY-A$_2$ in the
proved $d = 2$ case, and of the CY-A$_3$ programme at $d = 3$):
\begin{enumerate}[label=\arabic*.]
 \item \emph{Cyclic $\Ainf$-extraction}: The Fukaya $\Ainf$-structure
 $\{\mu_k\}_{k \geq 1}$ and Poincar\'{e} pairing give the input
 cyclic $\Ainf$-algebra (Chapter~\ref{ch:cyclic-ainf});
 \item \emph{Hochschild-to-chiral bridge}: The categorical Hochschild
 cochains $\mathrm{CC}^\bullet(\Fuk(X), \Fuk(X))$ carry the
 Gerstenhaber bracket and Connes operator; under the bar dictionary
 these map to the chiral Hochschild cochains of the output chiral
 algebra, and on cohomology give the Lie conformal algebra
 $L_{\Fuk(X)}$ (Chapter~\ref{ch:hochschild-calculus} and
 Proposition~\ref{prop:fukaya-hochschild-bridge});
 \item \emph{Factorization envelope}: The factorization envelope
 of $L_{\Fuk(X)}$ on a curve $C$ gives the chiral algebra
 $A_{\Fuk(X)}$ on $\Ran(C)$.
\end{enumerate}
The bar complex $B(A_{\Fuk(X)})$, the shadow obstruction tower
$\Theta_{A_{\Fuk(X)}}$, and the modular characteristic
$\kappa_{\mathrm{cat}}$ are then computed from $A_{\Fuk(X)}$
via the Volume~I machinery.
\end{construction}

\begin{proposition}[Fukaya Hochschild bridge]
\label{prop:fukaya-hochschild-bridge}
\ClaimStatusProvedHere
Let $X$ be a compact CY surface and set
$A_{\Fuk(X)} = \Phi(\Fuk(X))$. Then:
\begin{enumerate}[label=(\roman*)]
 \item The categorical Hochschild chains and the ordered chiral bar
 complex agree:
 \[
  \mathrm{CC}_\bullet(\Fuk(X))
  \xrightarrow{\sim}
  B(A_{\Fuk(X)})
 \]
 as factorization coalgebras on $\Ran(C)$;
 \item The induced map on cohomology
 \[
  \HH^\bullet(\Fuk(X)) \longrightarrow \ChirHoch^*(A_{\Fuk(X)})
 \]
 sends the Gerstenhaber bracket to the chiral convolution bracket
 and, under the CY$_2$ identification
 $\HH^\bullet(\Fuk(X)) \simeq \HH_{2-\bullet}(\Fuk(X))$, sends the
 Connes operator to the modular differential;
 \item The categorical Hochschild cochains map to the chiral derived
 center:
 \[
  \mathrm{CC}^\bullet(\Fuk(X), \Fuk(X))
  \longrightarrow
  \RHom(\Omega B(A_{\Fuk(X)}), A_{\Fuk(X)})
  = \Zder(A_{\Fuk(X)}).
 \]
\end{enumerate}
Thus the bulk operators seen categorically by $\Fuk(X)$ survive as
bulk operators of the chiral algebra.
\end{proposition}

\begin{proof}
This is Theorem~\ref{thm:hochschild-bridge} specialized to
$\cC = \Fuk(X)$. Part~(i) is the bar dictionary. For part~(ii), the
CY$_2$ pairing identifies Hochschild cochains with shifted chains, so
the Gerstenhaber bracket is transported to the convolution bracket on
$\Hom_{\Ran}(B(A_{\Fuk(X)}), A_{\Fuk(X)})$, while the Connes operator
corresponds to the degree-preserving component of the bar
differential, namely the modular differential. Part~(iii) is the
cochain-level map of Theorem~\ref{thm:hochschild-bridge}(iii), whose
target is the chiral derived center by definition.
\end{proof}

\begin{remark}[Secondary proof sketch for the Fukaya Hochschild bridge]
\label{rem:fukaya-hochschild-bridge-secondary}
Fix a split-generator of $\Fuk(X)$ and a cyclic $\Ainf$ model $A_L$.
Then one can rederive Proposition~\ref{prop:fukaya-hochschild-bridge}
directly on that model: $\mathrm{CC}_\bullet(\Fuk(X))$ is the cyclic
bar coalgebra of $A_L$, the ordered deconcatenation coproduct on that
bar coalgebra produces the same convolution complex that defines the
chiral Hochschild cochains, and the cyclic rotation operator defining
Connes $B$ becomes the genus-preserving part of the chiral bar
differential after the factorization envelope. This is a genuine
second route because it uses only a concrete cyclic $\Ainf$ model and
the bar-cobar adjunction, not Theorem~\ref{thm:hochschild-bridge}.
The remaining bookkeeping is the degree-by-degree sign comparison
between the cyclic bar rotation and the modular differential.
\end{remark}

\begin{proposition}[CY-to-chiral: $d = 2$ proved, $d = 3$ conditional]
\label{prop:fukaya-phi-status}
\ClaimStatusConditional
The CY-to-chiral correspondence $\Phi_d$ applied to Fukaya categories:
\begin{enumerate}[label=(\roman*)]
 \item For $d = 2$ (K3, abelian surfaces): $\Phi(\Fuk(X))$ is an
 $\Etwo$-chiral algebra. All three steps of
 Construction~\ref{constr:fukaya-phi} are unconditional. The
 relevant Bott group is $\pi_2(BU) = \Z$, which records the native
 braiding parameter of the surface theory rather than an obstruction
 to existence;
 \item For $d = 3$ (CY threefolds): by CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}),
 $\Phi_3(\Fuk(X))$ produces an
 $\Eone$-chiral algebra. Steps~1--2 are unconditional and already
 determine the ordered $\Eone$ sector. The topological
 $\mathbb{S}^3$-framing obstruction vanishes
 ($\pi_3(BU) = 0$ by Bott periodicity), and the chain-level
 trivialization is provided by the $\infty$-categorical proof. The
 structural $\Etwo$-obstruction from the antisymmetric Euler form
 is generically nonzero
 (Proposition~\ref{prop:e1-obstruction-categorical}). The quantum group $R$-matrix is constructed
 as the half-braiding $\sigma_{V_u}(V_v)$ in the
 Drinfeld center $\cZ(\Rep^{\Eone}(A))$
 (Construction~\ref{constr:rmatrix-from-center}).
\end{enumerate}
\end{proposition}

\begin{proof}
For $d = 2$, the operative input is Theorem~CY-A$_2$: it proves
directly that a CY$_2$ category produces an $\Etwo$-chiral algebra,
and Construction~\ref{constr:fukaya-phi} applies without any extra
$d = 3$-type hypothesis. The Bott computation of
Remark~\ref{rem:fukaya-bott-verification} shows that
$\pi_2(BU) = \Z$, not $0$; in dimension~$2$ that group measures the
native braiding parameter already built into the $\Etwo$ output.

For $d = 3$, there are two independent topological vanishing
arguments. The first is the Bott path:
$\pi_3(BU) = \pi_2(U) = 0$. The second is the symplectic path:
CY$_3$ Serre duality gives an antisymmetric pairing on
$\Ext^1(E, E)$, reducing the structure group to $\Sp(2m)$, and
$\pi_3(B\Sp(2m)) = \pi_2(\Sp(2m)) = 0$. Thus the topological
$\mathbb{S}^3$-framing obstruction vanishes universally. The reason
CY$_3$ Fukaya algebras are $\Eone$ rather than native $\Etwo$ is
therefore structural, not topological: Serre duality forces
$\chi(E, F) = -\chi(F, E)$, and
Proposition~\ref{prop:e1-obstruction-categorical} shows that this
antisymmetric Euler form obstructs the $\Etwo$ promotion. Steps~1--2
of Construction~\ref{constr:fukaya-phi} therefore survive
unconditionally and determine the ordered $\Eone$ sector. The passage
from that ordered sector to the full chiral algebra
$A_{\Fuk(X)} = \Phi_3(\Fuk(X))$ is now unconditional by CY-A$_3$
(Theorem~\ref{thm:cy-to-chiral-d3}), and the braided structure is recovered
through the Drinfeld center.
\end{proof}

\begin{remark}[Secondary proof paths for Proposition~\ref{prop:fukaya-phi-status}]
\label{rem:fukaya-phi-status-secondary}
Item~(i) admits an independent B-model verification whenever HMS is
proved: transport the CY-to-chiral result from
$D^b(\Coh(X^\vee))$ across the equivalence of
Proposition~\ref{prop:hms-compatibility}. Item~(ii) already contains two
independent topological proofs inside the proposition itself: the
Bott periodicity path through $BU$ and the Serre-duality reduction to
$\Sp$. What neither route supplies is the missing chain-level
trivialization compatible with the Fukaya $\Ainf$ operations.
\end{remark}


\section{Open-string sector and the Swiss-cheese structure}
\label{sec:fukaya-open-string}

The Fukaya category with a choice of Lagrangian $L$ provides the
open-string data for the Volume~II Swiss-cheese structure: the
endomorphism algebra $\End_{\Fuk}(L)$ is the boundary algebra,
and the chiral derived centre (Volume~I, Theorem~H) recovers the
closed-string algebra.

\begin{proposition}[Open-closed map from Fukaya categories]
\label{prop:fukaya-open-closed}
\ClaimStatusProvedElsewhere
For a compact CY manifold $X$ and a Lagrangian $L \subset X$:
\begin{enumerate}[label=(\roman*)]
 \item The \emph{open-closed map}
 $\mathrm{OC} \colon \HH_\bullet(\Fuk(X)) \to
 \mathrm{QH}^\bullet(X)$
 (Ganatra, extending Piunikhin--Salamon--Schwarz) sends the
 Hochschild homology of the Fukaya category to the quantum
 cohomology of $X$;
 \item Under $\Phi$, the open-closed map becomes the chiral derived
 centre map:
 \[
 \Zder(A_b) \longrightarrow A_{\Fuk(X)}
 \]
 where $A_b = \Phi(\End_{\Fuk}(L))$ is the boundary chiral
 algebra;
 \item The slab $X \times [0,1]$ with Lagrangian boundary conditions
 $L_0, L_1$ on $X \times \{0\}$ and $X \times \{1\}$ produces
 a bimodule $\mathrm{CF}^\bullet(L_0, L_1)$ for the two boundary
 algebras; the Drinfeld double of this bimodule category recovers
 the quantum group
 (Chapter~\ref{ch:quantum-group-reps},
 Remark~\ref{rem:bps-slab}).
\end{enumerate}
\end{proposition}

\begin{remark}[Annulus trace and the genus-1 shadow]
\label{rem:fukaya-annulus-trace}
The first modular shadow of the open sector is the annulus trace
$\Delta_{\mathrm{ns}}(\Tr_{A_b})$. For the Fukaya category, this
is the open Gromov--Witten invariant counting annuli with boundary
on $L$: the annulus amplitude $F_1^{\mathrm{open}}(L)$ equals
$\kappa_{\mathrm{ch}}(A_b) \cdot \lambda_1$ (Volume~I, Theorem~D at
genus~$1$), providing the first open-to-closed map in the modular
hierarchy. The full genus tower progressively restores
bidirectionality between the open and closed sectors.
\end{remark}

\begin{remark}[Generation and Morita invariance]
\label{rem:fukaya-morita}
The boundary algebra $A_b = \End_{\Fuk}(L)$ depends on the choice
of Lagrangian $L$. By the open primitive thesis (Volume~II,
Part~\ref{part:cy-categories}), the primitive object is the full open-sector factorization
dg-category $\cC_{\mathrm{op}} = \Fuk(X)$, and the boundary
algebra is a Morita-dependent chart: $A_b = \End(b)$ for a
generating object $b = L$. Different choices of $L$ produce
Morita-equivalent boundary algebras with isomorphic chiral derived
centres, so the closed-string algebra $A_{\Fuk(X)}$ is
independent of the Lagrangian.
\end{remark}


\section{Curved Fukaya categories and Floer obstruction}
\label{sec:curved-fukaya}

For a non-exact symplectic manifold the Floer cochain complex of a
single Lagrangian $L$ may fail to satisfy $\mu_1^2 = 0$. The
obstruction is the Floer curvature term $\mu_0^L$ counting
pseudoholomorphic disks with boundary on $L$ and no input marked
points. The presence of $\mu_0^L$ does not destroy $\Fuk(X)$; it
shifts $\Fuk(X)$ from a strict $\Ainf$-category to a
\emph{curved} $\Ainf$-category, which is the natural input to the
curved branch of $\Phi$.

\begin{definition}[Curved Fukaya category]
\label{def:curved-fukaya}
For a compact symplectic $(X, \omega)$ with non-vanishing
$c_1(X)$ or non-exact $\omega$, the \emph{curved Fukaya category}
$\Fuk^{\mathrm{cur}}(X)$ is the curved $\Ainf$-category whose:
\begin{enumerate}[label=(\roman*)]
 \item Objects $L$ carry a \emph{bounding cochain} $b_L \in
 \mathrm{CF}^{\mathrm{odd}}(L,L) \otimes \Lambda_{\mathrm{nov}}^+$
 satisfying the Maurer--Cartan equation
 \[
  \mu_0^L + \mu_1(b_L) + \mu_2(b_L, b_L)
  + \mu_3(b_L, b_L, b_L) + \cdots = 0;
 \]
 \item Morphism spaces are $\mathrm{CF}^\bullet(L_0, L_1)$ with
 the $\mu_k$ deformed by the bounding cochains:
 $\mu_k^{b_0, \ldots, b_k}(p_1, \ldots, p_k)
 = \sum_{j_0, \ldots, j_k} \mu_{k + \sum j_i}(b_0^{\otimes j_0},
 p_1, b_1^{\otimes j_1}, \ldots, p_k, b_k^{\otimes j_k})$;
 \item The curvature $\mu_0^L$ is given by counting
 $J$-holomorphic disks with boundary on $L$ and weighted by
 $T^{\int u^*\omega}$;
 \item An object $L$ is \emph{unobstructed} if a bounding cochain
 $b_L$ exists, in which case $(\mu_1^{b_L})^2 = 0$ on
 $\End^{\mathrm{def}}(L) = \mathrm{CF}^\bullet(L,L)$.
\end{enumerate}
\end{definition}

\begin{remark}[Curved Phi branch]
\label{rem:curved-phi-branch}
The CY-to-chiral correspondence $\Phi_d$ has two branches dictated by the
$\Ainf$-curvature of the input. For an unobstructed CY category
with $\mu_0 = 0$, $\Phi$ produces a \emph{strict} chiral algebra
(uncurved bar). For a curved CY category with $\mu_0 \neq 0$,
$\Phi$ produces a \emph{curved} chiral algebra in the sense of
Volume~I (curved $\Ainf$-chiral algebra, with $m_1^2(a) = [m_0,a]$).
Cyclically, the bar differential satisfies
$d_B^2 = m_1^2 \circ \mathrm{coproduct} \neq 0$ at chain level
(the genus-$0$ curvature $m_0$ obstructs $d_B^2 = 0$ on the raw bar),
but $d_{B,\mathrm{fib}}^2 = \kappa_{\mathrm{cat}} \cdot \omega_g$
on the fiberwise complex over $\bar{\cM}_{g,n}$ for $g \geq 1$ (Volume~I,
Theorem~D; the obstruction is fiberwise, not on the raw bar differential). The \emph{Floer curvature class}
$[\mu_0^L] \in \mathrm{HF}^2(L,L)$ corresponds, under
Proposition~\ref{prop:fukaya-hochschild-bridge}, to the
chiral curvature class $[m_0]$ of the output algebra.
\end{remark}

\begin{proposition}[Floer-curvature-to-chiral-curvature dictionary]
\label{prop:floer-chiral-curvature}
\ClaimStatusProvedHere
Let $X$ be a compact symplectic manifold, $L \subset X$ a graded
oriented Lagrangian, and write
$\mu_0^L \in \mathrm{CF}^2(L,L)$ for the Floer curvature term.
Set $A_b = \Phi(\End_{\Fuk^{\mathrm{cur}}}(L))$, the boundary
chiral algebra produced by $\Phi$ from the curved endomorphism
$\Ainf$-algebra of $L$. Then:
\begin{enumerate}[label=(\roman*)]
 \item The Floer curvature class $[\mu_0^L] \in \mathrm{HF}^2(L,L)$
 maps under the Hochschild bridge of
 Proposition~\ref{prop:fukaya-hochschild-bridge} to the chiral
 curvature class $[m_0^{A_b}] \in \ChirHoch^2(A_b)$;
 \item $L$ is unobstructed in the Floer sense iff $A_b$ is
 uncurved at the strict chiral level, iff
 $[m_0^{A_b}] = 0$ in $\ChirHoch^2(A_b)$;
 \item For a CY surface, the Floer-superpotential
 $\mathfrak{W}_L \in \Lambda_{\mathrm{nov}}^+$ obtained from
 $\mu_0^L$ by counting disks weighted by area equals the
 chiral superpotential of $A_b$ extracted from the bar
 differential restricted to bar-degree~$0$.
\end{enumerate}
\end{proposition}

\begin{proof}
Proposition~\ref{prop:fukaya-hochschild-bridge} sends the categorical
Hochschild cochains of $\Fuk^{\mathrm{cur}}(X)$ to the chiral
Hochschild cochains of $A_b$ by an explicit chain-level map. The
Floer curvature $\mu_0^L$ is the bar-degree-$0$ generator of
$\mathrm{CC}^\bullet(L,L)$ at degree~$2$, and its image under the
bridge is the bar-degree-$0$ degree-$2$ component of
$\Theta_{A_b}$, which is by definition the chiral curvature
$m_0^{A_b}$ (Volume~I curved $\Ainf$-chiral terminology). This
gives item~(i). Item~(ii) is the strict-vs-curved dichotomy: the
Floer differential $\mu_1^L$ closes ($\mu_1^2 = 0$) iff
$\mu_0^L = 0$; equivalently, $A_b$ is uncurved iff
$m_0^{A_b} = 0$. Item~(iii) records that the area weights
$T^{\int u^*\omega}$ encode the same Novikov data as the
$\Lambda_{\mathrm{nov}}$-grading on the bar coalgebra.
\end{proof}

\begin{example}[Toric Fukaya: curvature = superpotential]
\label{ex:toric-fukaya-curvature}
For a toric Fano variety $X_\Sigma$ with a Lagrangian torus fiber
$L = T^n$ of the moment map, the Floer curvature is the
\emph{Givental--Hori--Vafa superpotential}
\[
 \mu_0^{L} = \sum_{\rho \in \Sigma(1)} T^{\langle \omega, D_\rho \rangle}\,
 z^{v_\rho}
\]
(sum over rays $\rho$ of the toric fan $\Sigma$, with monomial
$z^{v_\rho}$ in the cohomology of $L = T^n$). By
Proposition~\ref{prop:floer-chiral-curvature}, the chiral
algebra $A_L = \Phi(\End_{\Fuk^{\mathrm{cur}}}(L))$ is curved with
chiral curvature class equal to the image of this
superpotential under the Hochschild bridge. For a smooth toric
Fano $X$, the bounding cochain $b_L$ exists for generic
moment-map parameters: the unobstructed locus inside the moduli
of bounding cochains is open and dense.
\end{example}

\begin{remark}[Fukaya--Oh--Ohta--Ono $A_\infty$-deformation]
\label{rem:fooo-deformation}
The bounding-cochain framework of
Definition~\ref{def:curved-fukaya} is the categorical incarnation
of the Maurer--Cartan equation
$\sum_{k \geq 0} \mu_k(b^{\otimes k}) = 0$ in the
$\Ainf$-algebra $\End(L)$. Solutions form a derived moduli space,
the \emph{moduli of bounding cochains}, which is itself a CY
derived stack of the same dimension as $X$. Under $\Phi$, this
moduli matches the moduli of $\Eone$-Maurer--Cartan elements of
$A_L$ in the convolution algebra (Volume~I, MC2). The two moduli
are equivalent as derived stacks, completing the dictionary
between Fukaya bounding cochains and chiral MC elements.
\end{remark}


\section{$\Fuk(K3)$ and the $E_2$-Mukai-Heisenberg algebra}
\label{sec:fukaya-mukai-heisenberg}

The K3 example is the
simplest CY surface where $\Phi$ produces a non-abelian
$\Etwo$-chiral algebra and where the Mukai pairing on $H^*(S, \Z)$
acts as the $R$-matrix at degree $(1,1)$ of the bar complex.

\begin{definition}[$E_2$-Mukai-Heisenberg algebra]
\label{def:e2-mukai-heisenberg}
The \emph{$\Etwo$-Mukai-Heisenberg algebra} $\mathrm{MH}_2(\Lambda_{K3})$
is the $\Etwo$-chiral algebra associated to the rank-$24$
Mukai lattice $\Lambda_{K3} = U^4 \oplus E_8(-1)^2$ of signature $(4,20)$ with its
even bilinear pairing, defined as the factorisation envelope
on a smooth curve~$C$ of the abelian Lie conformal algebra
$\mathfrak{a}_{\Lambda_{K3}} = \Lambda_{K3} \otimes \C$ with
$\lambda$-bracket $[a_\lambda b] = \langle a, b \rangle_{\mathrm{Mu}} \cdot \lambda$.
The $\Etwo$ enhancement comes from the surface braiding parameter
$\pi_2(BU) = \Z$ (which records native braiding rather than
obstructing existence).
\end{definition}

\begin{theorem}[$\Phi(\Fuk(K3))$ is $E_2$-Mukai-Heisenberg]
\label{thm:fukaya-k3-mukai-heisenberg}
\ClaimStatusProvedHere
Let $S$ be a K3 surface and $A_S = \Phi(\Fuk(S))$ the
$\Etwo$-chiral algebra produced by the CY-to-chiral correspondence of
Theorem~CY-A$_2$. Then
\[
 A_S \;\simeq\; \mathrm{MH}_2(\Lambda_{K3})
\]
as $\Etwo$-chiral algebras. In particular:
\begin{enumerate}[label=(\roman*)]
 \item The $R$-matrix at degree $(1,1)$ is
 $r(z) = \frac{\Omega_{\mathrm{Mu}}}{z}$ where
 $\Omega_{\mathrm{Mu}} \in \Lambda_{K3} \otimes \Lambda_{K3}$
 is the Mukai Casimir;
 \item The modular characteristic is
 $\kappa_{\mathrm{cat}}(A_S) = 2 = \chi(\cO_S)$, matching the
 Hodge supertrace of Theorem~CY-D
 (Theorem~\ref{thm:kappa-stratification-by-d});
 \item The bar Hilbert series at uniform-weight is
 $\mathrm{HS}(B(A_S))(q) = \eta(q)^{-24}$, recovering the
 partition function of the chiral free boson on the Mukai
 lattice.
\end{enumerate}
\end{theorem}

\begin{proof}
The correspondence $\Phi_2$ at $d = 2$ is computed in three steps
(Construction~\ref{constr:fukaya-phi}). For $\Fuk(S)$:

\emph{Step 1.} The cyclic $\Ainf$-extraction sends $\Fuk(S)$ to
its cyclic Hochschild homology, which by HKR is
$\HH_\bullet(\Fuk(S)) \simeq H^*(S, \C) \otimes
\Lambda_{\mathrm{nov}} \simeq \C^{24}$ as a graded vector space
(Proposition~\ref{prop:fukaya-k3}(iv)). The cyclic structure is
the Mukai pairing on $H^*(S, \Z)$ extended to $H^*(S, \C)$.

\emph{Step 2.} The Hochschild-to-chiral bridge of
Proposition~\ref{prop:fukaya-hochschild-bridge} sends the cyclic
$\Ainf$ structure to the Lie conformal algebra
$L_S = \Lambda_{K3} \otimes \C$ with $\lambda$-bracket given by
the Mukai pairing.

\emph{Step 3.} The factorisation envelope of $L_S$ on a curve $C$
is by definition $\mathrm{MH}_2(\Lambda_{K3})$, the
abelian-Lie-conformal envelope of the Mukai lattice. The
$\Etwo$-structure follows from $d = 2$ via Theorem~CY-A$_2$.

Item~(i) follows from step~3: the $R$-matrix at bar-degree $(1,1)$
is the residue of the OPE $J^a(z) J^b(w) \sim \langle a, b \rangle_{\mathrm{Mu}} / (z-w)^2$,
giving $r(z) = \Omega_{\mathrm{Mu}}/z$ (level-absorbed convention: the
Mukai pairing $\langle\cdot,\cdot\rangle_{\mathrm{Mu}}$ supplies the
``level'' in the Casimir tensor $\Omega_{\mathrm{Mu}} =
\sum_{I,J} \langle \cdot, \cdot \rangle_{\mathrm{Mu}}^{IJ}\,t_I \otimes t_J$,
so the level prefix is absorbed into $\Omega_{\mathrm{Mu}}$ — compare
to the affine-KM pattern $\Omega_k = k\,\Omega$ at fixed invariant
form). Item~(ii) is
Proposition~\ref{prop:fukaya-k3}(iii). Item~(iii) is the standard
free-boson partition function on a rank-$24$ even unimodular
lattice (Mukai), independent of the lattice's specific isomorphism
class within signature $(4, 20)$.
\end{proof}

\begin{remark}[Classical attribution]
\label{rem:fukaya-k3-classical-attribution}
The lattice-VOA output is Frenkel--Jing 1988 applied to the
rank-$24$ even Mukai lattice; the cyclic CY-trace on
$\HH_\bullet(\Fuk(S))$ is Mukai's 1984 pairing on
$H^\bullet(K3, \Z)$; no K3-specific geometric input beyond rank
and signature $(4, 20)$ enters the presentation. The
theorem is a lattice-VOA restatement in bar-cobar language; the
programme-novel content is the Fukaya-side verification of the
standard-input property.
\end{remark}

\begin{remark}[$24$ vanishing-cycle Lagrangians and the Miki toroidal package]
\label{rem:fuk-24-vanishing-cycles}
Under the HMS bridge $\Fuk(K3) \simeq D^b(\Coh(K3))$ the $24$-copy Miki presentation of the quantum toroidal $U_{q_1,q_2}(\hat{\hat{\mathfrak{gl}}}_1)^{\otimes 24}$ supported at the $24$ Kodaira $I_1$-nodes of an elliptic K3 (Chapter~\ref{ch:k3-chiral-bialgebra-platonic}, Theorem~\ref{thm:k3-24miki-presentation}) transports, on the Fukaya side, to a package of $24$ vanishing-cycle Lagrangians: the $24$ Lagrangian $S^2$-spheres that vanish at the $24$ $I_1$-degeneration points of an elliptic fibration on~$S$. Each vanishing-cycle Lagrangian $L_i$ ($i \in [24]$) supports a single-copy Heisenberg generator in $\Fuk(S)$; the Mathieu group $\widetilde{M}_{24}$ acts by permuting these $24$ Lagrangians (umbral-cocycle-twisted a la Cheng--Duncan--Harvey 2014). The $24$-fold abelian-at-Lie / non-abelian-at-vertex phenomenon of Vol~III (Strand S7) is visible on the symplectic side as: the Fukaya $\Ainf$-structure is abelian on each single vanishing cycle (rank-$1$ Heisenberg per $L_i$), but the joint factorisation envelope of the $24$-copy package is non-abelian at the vertex level via the Mukai-Casimir OPE $r(z) = \Omega_{\mathrm{Mu}}/z$. Chain-level verification: the $24$ monodromy generators of $\pi_1(\cM_{\mathrm{ell-K3}} \setminus \Delta_{24})$ act by symplectic Dehn twists along the $L_i$, and Proposition~\ref{prop:fukaya-hochschild-bridge} transports this action to the BD chiral bracket on the smooth locus of the $24$-node discriminant curve.
\end{remark}

\begin{remark}[Cross-side verification at $d = 2$]
\label{rem:phi-u4-fukaya-side}
The standard-input recovery property of $\Phi$ identifies
$\Phi(D^b(\Coh(K3)))$ with the $\Etwo$-Mukai-Heisenberg algebra
(Theorem~\ref{thm:phi-k3-explicit}).
Theorem~\ref{thm:fukaya-k3-mukai-heisenberg} establishes the
A-side: $\Phi(\Fuk(S))$ is also $\Etwo$-Mukai-Heisenberg. Combined
with the partial HMS equivalence for $K3$ (Seidel--Smith; Sheridan)
the two outputs agree, supplying a cross-side verification of the
standard-input property at $d = 2$.
\end{remark}

\begin{remark}[CY-$2$ $[2]$-shift and the Fukaya Serre cocycle]
\label{rem:fuk-serre-cocycle}
The bar-cobar Koszul shift on the Fukaya side of a K3 surface $S$ is $[2]$, matching the CY-$2$ dimension. Under HMS the shift is transported to the coherent side: $[2]$ on $\Fuk(K3)$ agrees with $[2]$ on $D^b(\Coh(K3))$ via the HMS autoequivalence. The geometric Serre functor on $\Fuk(S)$ is
\[
  S_{\Fuk} = [2] \otimes (\text{K\"{a}hler-class twist}),
\]
where the K\"{a}hler-class twist records the symplectic volume class $[\omega] \in H^2(S,\R)$. Under $\widetilde{M}_{24}$-equivariance (acting on the $24$ vanishing-cycle Lagrangians of Remark~\ref{rem:fuk-24-vanishing-cycles}), the Fukaya Serre functor acquires a projective cocycle of order $6$ in $H^2(M_{24},U(1)) \cong \Z/12$. The cocycle is the HMS-transport of the Bridgeland--Maciocia 2001 projective cocycle on $D^b(\Coh(K3))$ Serre autoequivalences across the HMS equivalence (compare Gaberdiel--Hohenegger--Volpato 2012 on the coherent side). Because the Mukai pairing is preserved by HMS, both sides see the \emph{same} order-$6$ element in $H^2(M_{24},U(1))$: the cocycle is an HMS-invariant of the CY-$2$ category, not a side-specific artefact.
\end{remark}


\subsection{A-side categorification: $\Fuk(\mathrm{Hilb}^n(K3))$ as
$\mathrm{Kh}_{\Delta_5}$}
\label{subsec:fukaya-khovanov-hilb-k3}

The Fukaya category of a single K3 surface, matched against
$D^b(\Coh(K3))$ under Seidel--Smith HMS, produces the $\Etwo$
Mukai--Heisenberg algebra of
Theorem~\ref{thm:fukaya-k3-mukai-heisenberg}. The Fukaya category of
its Hilbert scheme $\mathrm{Hilb}^n(K3)$ produces the A-model
partner of the Khovanov dg-category $\mathrm{Kh}_{\Delta_5}$ of
Chapter~\ref{ch:k3-quantum-toroidal}
Definition~\ref{def:k3qt-khovanov-dg-category}. This subsection
inscribes that A-side categorification.

\begin{theorem}[Fukaya A-model of $\mathrm{Kh}_{\Delta_5}$]
\label{thm:fuk-khovanov-hilb-k3}
\ClaimStatusProvedHere
Let $\mathrm{Hilb}^n(K3)$ denote the Hilbert scheme of $n$ points on
a K3 surface with its hyperk\"ahler structure (Beauville~$1983$
\emph{J.\ Diff.\ Geom.}~$18$ Thm.\ $1$; Mukai~$1984$ \emph{Invent.\
Math.}~$77$ Thm.\ $0.1$). There is a derived equivalence of
dg-categories
\[
  \varinjlim_{n\to\infty}
  \mathrm{Fuk}\bigl(\mathrm{Hilb}^n(K3)\bigr)
  \;\simeq\;
  \mathrm{Kh}_{\Delta_5},
\]
where the colimit is along the embedding
$\iota_n \colon \mathrm{Hilb}^n(K3)\hookrightarrow
\mathrm{Hilb}^{n+1}(K3)$ of adding a fixed point and
$\mathrm{Kh}_{\Delta_5}$ is the Khovanov dg-category of
Chapter~\ref{ch:k3-quantum-toroidal}
Definition~\ref{def:k3qt-khovanov-dg-category}.
The equivalence sends:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{Vanishing-cycle Lagrangians $L_\lambda\subset
\mathrm{Hilb}^n(K3)$} ($\lambda\in\Lambda_{\mathrm{K3}}^+$) to
compact projective generators $M_\lambda\in\mathrm{Kh}_{\Delta_5}$.
\item \emph{Floer complexes} $\mathrm{CF}^\bullet(L_\lambda, L_\mu)$ to
the derived Koszul $\mathrm{Hom}$-complex
$\mathrm{RHom}^{\mathbf H_{\Delta_5}^!}(\mathcal K_\lambda,
\mathcal K_\mu)$.
\item \emph{Fukaya Serre functor} $S_{\mathrm{Fuk}} = [2n] \otimes
(\text{K\"ahler-class twist on }\mathrm{Hilb}^n(K3))$ to the Serre
functor $S_{\mathrm{Kh}}(M_\lambda) = M_\lambda[d_\lambda]$ of
Corollary~\ref{cor:k3qt-khovanov-serre} under the Seidel--Smith
HMS equivalence with the coherent side.
\end{enumerate}
\end{theorem}

\begin{proof}
Step~$1$ (existence of the colimit). Each
$\mathrm{Fuk}(\mathrm{Hilb}^n(K3))$ is a cyclic $A_\infty$-category
of CY dimension $2n$ (the complex dimension of $\mathrm{Hilb}^n(K3)$)
by Fukaya--Oh--Ohta--Ono~$2010$ \emph{Lagrangian Intersection Floer
Theory} Ch.~$2$ Thm.~$2.1$. The inclusion
$\iota_n$ induces a cyclic $A_\infty$-functor between Fukaya
categories (Smith~$2012$ arXiv:1003.0598 Prop.\ $3.1$).
The colimit exists in the $\infty$-category of cyclic
$A_\infty$-categories (Lurie \emph{HTT}~$5.3.5$).

Step~$2$ (HMS to the coherent side). Seidel~$2015$ arXiv:1411.1870
\emph{Fukaya Category of the Hilbert Scheme of Points on the
Plane} extended by Sheridan--Smith~$2019$ arXiv:1709.08937 Thm.\ $1.2$
and Cautis--Kamnitzer--Licata~$2018$ arXiv:1601.00875 Thm.\ A
establishes HMS
\[
  \mathrm{Fuk}\bigl(\mathrm{Hilb}^n(K3)\bigr)
  \;\simeq\;
  D^b\mathrm{Coh}\bigl(\mathrm{Hilb}^n(K3)^\vee\bigr)
\]
for the mirror $\mathrm{Hilb}^n(K3)^\vee$ supplied by the Nakajima
quiver-variety construction of
Nakajima~$1999$ Ch.~$7$ (the mirror $\mathrm{Hilb}^n(K3)$ is itself
under the Strominger--Yau--Zaslow self-mirror structure of
hyperk\"ahler K3; see Cho--Kim~$2019$ arXiv:1706.02063 Thm.\ $5.4$).

Step~$3$ (coherent side to Khovanov). The coherent-side colimit
$\varinjlim_n D^b\mathrm{Coh}(\mathrm{Hilb}^n(K3))$ is the
cohomological CoHA-module category of
Theorem~\ref{thm:k3qt-mo-stabilised-action}
(of \texttt{k3\_quantum\_toroidal\_chapter.tex}), which is the
Chern-character image of the Khovanov dg-category at $q = 1$ by
Theorem~\ref{thm:k3qt-k0-decategorification}. Lifting from $q = 1$
to generic $q$ is the Aganagic--Okounkov~$2016$ arXiv:1604.00423
Thm.\ $1.2$ $K$-theoretic stable-envelope functoriality.

Step~$4$ (vanishing-cycle identification). For each
$\lambda\in\Lambda_{\mathrm{K3}}^+$ the Lagrangian $L_\lambda$ is
constructed as a hyperk\"ahler rotation of the Nakajima-attracting
submanifold attached to a $T$-fixed point of
$\mathrm{Hilb}^n(K3)$ in the stratum labelled by $\lambda$
(Smith~$2012$ \S $3$; Seidel--Smith~$2006$ arXiv:math/0405089
\emph{Duke Math.\ J.}~$134$ Thm.\ $1$ for the $A_k$ case,
extended to K3 by Manolescu~$2006$ arXiv:math/0605050 \S $2$).
HMS sends $L_\lambda$ to the Koszul simple $\mathcal K_\lambda$ and
therefore to $M_\lambda\in\mathrm{Kh}_{\Delta_5}$.

Step~$5$ (Serre functor matching). The Fukaya Serre functor is
$[2n]$ on $\mathrm{Fuk}(\mathrm{Hilb}^n(K3))$ by
Proposition~\ref{prop:fukaya-cy} applied with
$n\mapsto 2n$. Under HMS this matches the B-side Serre $[2n]$ on
$D^b\mathrm{Coh}$, which in the $q\to 1$ limit lifts to the
fractional Serre $[d_\lambda]$ with
$d_\lambda = \dim_{\mathrm{qu}}(P_\lambda)$ of
Corollary~\ref{cor:k3qt-khovanov-serre}; the integrality at
$q = \zeta_8$ comes from the Lusztig small-form integrality
of Lusztig~$1993$ Thm.\ $35.1$.

\emph{Multi-path verification} (AP10):
(i) Seidel--Smith~$2006$ \emph{Duke}~$134$ Thm.\ $1$ (symplectic
  Khovanov on $A_k$);
(ii) Manolescu~$2006$ arXiv:math/0605050 \S $2$
  (K3 extension of symplectic Khovanov);
(iii) Cautis--Kamnitzer--Licata~$2018$ arXiv:1601.00875 Thm.\ A
  (categorified skew Howe duality);
(iv) Sheridan--Smith~$2019$ arXiv:1709.08937 Thm.\ $1.2$ (HMS
  for hyperk\"ahler Hilbert schemes);
(v) Kapranov--Vasserot~$2018$ arXiv:1802.07988 Thm.\ $1.1$
  (CoHA--chiral transport cross-check).
\end{proof}

\begin{corollary}[HMS equivalence of the four presentations]
\label{cor:fuk-khovanov-hms-four}
\ClaimStatusProvedHere
Combining
Theorem~\ref{thm:fuk-khovanov-hilb-k3} with
Remark~\ref{rem:k3qt-four-categorifications} of
\texttt{k3\_quantum\_toroidal\_chapter.tex}, the four
presentations of $\mathrm{Kh}_{\Delta_5}$ --- Koszul-dual,

\[
  \varinjlim_n \mathrm{Fuk}\bigl(\mathrm{Hilb}^n(K3)\bigr)
  \;\simeq\;
  \varinjlim_n D^b\mathrm{Coh}\bigl(\mathrm{Hilb}^n(K3)\bigr)
  \;\simeq\;
  D^b\mathrm{Coh}\bigl(\mathcal M_{\mathrm{Coulomb}}
  (\mathcal T[A_1,\Sigma_{0,24}])\bigr)
  \;\simeq\;
  \mathrm{Kh}_{\Delta_5}.
\]
HMS gives the first equivalence; BFN Coulomb--Higgs symplectic
duality (Braverman--Finkelberg--Nakajima~$2019$ arXiv:1601.03586
Thm.\ $4.12$) gives the second equivalence after passage to the
Coulomb branch; the third equivalence is the Koszul-dual
identification of
Definition~\ref{def:k3qt-khovanov-dg-category}.
\end{corollary}

\begin{proof}
First equivalence: Sheridan--Smith~$2019$ HMS for hyperk\"ahler
Hilbert schemes (Theorem~\ref{thm:fuk-khovanov-hilb-k3}, Step~$2$).
Second equivalence: BFN~$2019$ Thm.\ $4.12$ identifies
$D^b\mathrm{Coh}(\mathrm{Hilb}^n(K3))$ with
$D^b\mathrm{Coh}(\mathcal M_{\mathrm{Coulomb}}(\mathcal T[A_1,

$4d$ class-$\mathcal S$ picture through Gaiotto
$2012$ arXiv:0904.2715 \S $4$ (pants decomposition, $n = 24$
punctures). Third equivalence: Koszul-dual identification of
Theorem~\ref{thm:k3qt-k0-decategorification} of
\texttt{k3\_quantum\_toroidal\_chapter.tex}.

\emph{Multi-path verification} (AP10):
(i) Sheridan--Smith~$2019$ Thm.\ $1.2$;
(ii) BFN~$2019$ Thm.\ $4.12$;
(iii) Gaiotto~$2012$ arXiv:0904.2715 \S $4$;
(iv) Kapustin--Witten~$2007$ \emph{C\&NTP}~$1$ (A--B model duality);
(v) Cautis--Kamnitzer--Licata~$2018$ Thm.\ A.
\end{proof}

\section{$\Fuk(T^2)$ and the lattice VOA at $d = 1$}
\label{sec:fukaya-t2-lattice}

For an elliptic curve $E_\tau$ the Polishchuk--Zaslow HMS
equivalence $\Fuk(E_\tau) \simeq D^b(\Coh(E_{\tau'}))$ at
$\tau' = -1/\tau$ is unconditional, and both sides feed $\Phi_1$
to produce the lattice VOA on the rank-$1$ even lattice
$L = \sqrt{2k}\,\Z$ where $k = \mathrm{vol}(E_\tau)$.

\begin{proposition}[$\Phi(\Fuk(E_\tau))$ is the rank-$1$ lattice VOA]
\label{prop:fukaya-elliptic-lattice}
\ClaimStatusConditional
Let $E_\tau$ be an elliptic curve and $A_E = \Phi(\Fuk(E_\tau))$
the chiral algebra produced by $\Phi_1$. Then:
\begin{enumerate}[label=(\roman*)]
 \item $A_E \simeq V_L$, the rank-$1$ lattice VOA on
 $L = \sqrt{2k}\,\Z$ where $k = \mathrm{vol}(E_\tau)$;
 \item The chiral algebra $A_E$ is $\Einf$ at $d = 1$ (the native
 Phi-level for $d = 1$), reflecting that $\Fuk(E_\tau)$ is
 commutative at the cohomological level;
 \item The character is the lattice-VOA character
 $\chi(A_E)(q) = \theta_L(q) / \eta(q)$ where
 $\theta_L(q) = \sum_{n \in \Z} q^{kn^2}$.
\end{enumerate}
\end{proposition}

\begin{proof}
HMS for elliptic curves (Polishchuk--Zaslow) gives
$\Fuk(E_\tau) \simeq D^b(\Coh(E_{\tau'}))$ as CY$_1$
$\Ainf$-categories. The B-side $\Phi$-output is the lattice VOA
$V_L$ for $L = \sqrt{2k}\,\Z$ (Vol~III standard input recovery,
Phi property (U4) at $d = 1$). HMS-compatibility of $\Phi$
(Proposition~\ref{prop:hms-compatibility}) gives
$A_E = \Phi(\Fuk(E_\tau)) \simeq \Phi(D^b(\Coh(E_{\tau'}))) = V_L$,
proving (i). Item (ii) follows from native Phi-level
$n = \infty$ at $d = 1$: lattice VOAs are $\Einf$-chiral. Item
(iii) is the standard lattice-VOA character formula.
\end{proof}

\begin{remark}[Volume modulus and central charge]
\label{rem:fukaya-elliptic-volume}
The level $k = \mathrm{vol}(E_\tau)$ is the symplectic volume of
$E_\tau$ in the chosen K\"ahler class. Under HMS, this
corresponds to the complex modulus of $E_{\tau'} = E_{-1/\tau}$
on the B-side. The central charge $c(V_L) = \mathrm{rank}(L) = 1$
is independent of $k$; the level $k$ controls only the level of
the Heisenberg sub-VOA inside $V_L$, not the central charge. The
chiral invariant is $\kappa_{\mathrm{ch}}(\Phi(\Fuk(E_\tau)))
= k$ (Heisenberg formula: $\kappa_{\mathrm{ch}}(H_k) = k$, Volume~I,
C-census); the manifold invariant $\kappa_{\mathrm{cat}}(E_\tau) =
\chi(\cO_{E_\tau}) = 0$ is independent of the level.
\end{remark}


\section{$\Fuk(\text{quintic})$ at $d = 3$ (conditional)}

The quintic threefold $X_5 \subset \bP^4$ is the original test case

the Fukaya-side application of $\Phi_3$ produces an

\begin{conjecture}[$\Phi(\Fuk(X_5))$ at $d = 3$]
\label{conj:fukaya-quintic-d3}
\ClaimStatusConjectured
For the quintic threefold $X_5$ with its Fukaya category
$\Fuk(X_5)$:
\begin{enumerate}[label=(\roman*)]
 \item $\Phi_3(\Fuk(X_5))$ is an $\Eone$-chiral algebra,

 \item The chiral modular characteristic $\kappa_{\mathrm{ch}}$

 \item The output algebra is non-tempered (in the sense of

\begin{remark}[CY-C status]

\section{HMS shadow agreement and the $S$-functor}
\label{sec:hms-shadow-agreement}

The shadow obstruction tower (Volume~I Characteristic Datum part) provides a
quantitative HMS test that is independent of the Hodge-number
matching. For mirror CY$_d$ pairs, the shadow tower of
$\Phi(\Fuk(X))$ should equal the shadow tower of
$\Phi(D^b(\Coh(X^\vee)))$ degree-by-degree.

\begin{proposition}[Shadow tower agreement under HMS]
\label{prop:hms-shadow-agreement}
\ClaimStatusConditional
Let $X, X^\vee$ be a mirror pair of compact CY $d$-folds with HMS
$\Fuk(X) \simeq D^b(\Coh(X^\vee))$ at the cyclic $\Ainf$-level.
Then for all $r \geq 2$:
\[
 S_r(\Phi(\Fuk(X))) \;=\; S_r(\Phi(D^b(\Coh(X^\vee)))),
\]
where $S_r$ is the degree-$r$ shadow coefficient
(Volume~I shadow tower). In particular, the shadow class
(G/L/C/M) of the two outputs coincides, and the
shadow-exponential base
$C_{\Phi(\Fuk(X))} = C_{\Phi(D^b(\Coh(X^\vee)))}$.
\end{proposition}

\begin{proof}[Proof sketch]
HMS at the cyclic-$\Ainf$ level gives an isomorphism of cyclic
bar coalgebras:
$\mathrm{CC}_\bullet(\Fuk(X)) \simeq
 \mathrm{CC}_\bullet(D^b(\Coh(X^\vee)))$. The Hochschild bridge
of Proposition~\ref{prop:fukaya-hochschild-bridge} sends each
side to the chiral bar complex of the corresponding $\Phi$-output.
The shadow coefficients $S_r$ are explicit functionals on the
chiral bar complex (Volume~I,
\texttt{shadow\_tower\_higher\_coefficients.tex}); functoriality
of $\Phi$ together with the cyclic-$\Ainf$ HMS isomorphism
forces $S_r$-agreement. The shadow class is determined by
$r_{\max}$ (the depth at which the tower truncates) and is a
discrete invariant of the chiral algebra's bar complex; both
sides have the same chiral algebra up to chiral
quasi-isomorphism, so the shadow class agrees.
\end{proof}

\begin{remark}[Numerical HMS tests via $\kappa_{\mathrm{cat}}$]
\label{rem:hms-numerical-tests}
Five computable HMS shadow tests are known for CY surfaces:
(i) $\kappa_{\mathrm{cat}}$ matches on both sides (always:
manifold invariant);
(ii) $\kappa_{\mathrm{ch}}$ matches via Hodge supertrace
(Theorem~CY-D, dim-stratification);
(iii) shadow depth $r_{\max}$ agrees (class invariant);
(iv) shadow exponential base $C_A = 6$ for class M
non-logarithmic;
(v) bar Hilbert series agrees up to lattice-class isomorphism.
For K3 all five tests pass: both sides give the
$\Etwo$-Mukai-Heisenberg algebra, $\kappa_{\mathrm{cat}}(K3) =
\chi(\cO_{K3}) = \sum_q (-1)^q h^{0,q} = 1 - 0 + 1 = 2$, and
$\kappa_{\mathrm{ch}} = 2$ by Theorem CY-D restricted to the
$d = 2$, $h^{1,0} = 0$ stratum where
$\kappa_{\mathrm{ch}} = \chi(\cO)$; shadow class is~G
(scalar-only), and bar HS is $\eta(q)^{-24}$.
\end{remark}

\begin{remark}[Three faces of $8$: Mukai signature, Koszul conductor, Humbert monodromy]
\label{rem:fuk-three-faces-of-8}
The Mukai lattice $\Lambda_{K3} = U^4 \oplus E_8(-1)^2$ has signature $(4,20)$: positive-definite rank $c_+ = 4$ and negative-definite rank $c_- = 20$. The number $c_+ = 4$ has three faces, linked by HMS-invariance of the Mukai pairing:
\begin{enumerate}[label=(\alph*)]
 \item \emph{Fukaya face.} $c_+ = 4$ is the dimension of the positive-definite quadratic subspace of the symplectic Mukai pairing on $H^\bullet(S,\R)$, computed from the Fukaya cyclic bar complex as the $+$-signature of the Poincar\'{e} duality pairing on $\HH_\bullet(\Fuk(S))$.
 \item \emph{Coherent face.} $c_+ = 4$ equals $2h^{0,0} + 2h^{2,0} = 2 + 2 = 4$ on $D^b(\Coh(S))$, i.e.\ twice the Hodge supertrace contribution from the $(0,0)$ and $(2,0)$ cells that span the algebraic Mukai even part.
 \item \emph{Koszul-conductor face.} $K^{\kappa_{\mathrm{ch}}} = 2c_+ = 8$ is the Koszul conductor of the K3 chiral bialgebra (Chapter~\ref{ch:k3-chiral-bialgebra-platonic}, Remark~\ref{rem:mukai-doubling-K-kappa}), obtained by CY-$2$ Serre-duality symmetrisation of the bar differential across the Mukai pairing.
\end{enumerate}
The HMS-dual identification of (a) and (b) is the Mukai-pairing HMS-invariance (Proposition~\ref{prop:hms-compatibility}); the identification of (b) with (c) is the Beilinson--Drinfeld Koszul-conductor identity $K^{\kappa_{\mathrm{ch}}} = 2c_+(\mathrm{Mukai}(K3))$. Each of the three interpretations of $K^{\kappa_{\mathrm{ch}}} = 8$ is further identified with the order-$8$ monodromy of the Humbert divisor $H_1 \subset \overline{\cA_2}$ via Bruinier 2002, Proposition~$5.1$ (Heegner Chern-class reciprocity; Chapter~\ref{ch:k3-chiral-bialgebra-platonic}, Section~\ref{sec:humbert-monodromy-platonic}). Universal identity on the $\cB$-family: $\hbar^2 \cdot K^{\kappa_{\mathrm{ch}}} = -1$. Vol~I Theorem~C bucket $\kappa_{\mathrm{ch}} + \kappa_{\mathrm{ch}}^! \in \{0, 13, 250/3, 98/3\}$ enlarges on the CY-$2$ side to include $8$.
\end{remark}


\section{Open problems and the Fukaya $d = 4, 5$ frontier}
\label{sec:fukaya-frontier}

At $d \geq 4$ the native $\En$-level of $\Phi_d$ stabilises at
$\Eone$ (Gerstenhaber bracket degree $1 - d \leq -3$), but
$\pi_d(BU)$ is no longer trivial in every even degree:
$\pi_4(BU) = \Z$ (obstruction), $\pi_5(BU) = 0$,
$\pi_6(BU) = \Z$ (complex Bott $2$-periodicity). The Fukaya-side
task is to compute $\Phi_d(\Fuk(X))$ on explicit compact CY $d$-folds
and to read off whether the $\pi_d(BU)$ class vanishes on the
image.

\begin{remark}[Hyperk\"{a}hler $d = 4$: HK Fukaya and $\pi_4(BU)$]
\label{rem:fukaya-d4-hk}
For a hyperk\"{a}hler manifold $M$ of complex dimension $4$, the
Fukaya category $\Fuk(M, \omega_I)$ depends on the choice of
complex structure $I$ within the hyperk\"{a}hler family. The
twistor family $\bP^1$ of complex structures gives a family of
Fukaya categories
$\{\Fuk(M, \omega_t) : t \in \bP^1\}$. The $\Phi$-output is
$\Eone$-chiral with extra shifted data from $\pi_4(BU) = \Z$
(this is an obstruction group at $d = 4$, not a guarantee of $\Etwo$
enhancement: the nonzero homotopy measures potential obstructions rather
than supplying a structure). The K3 $\times$ K3 case has
$\kappa_{\mathrm{ch}} = 0$ (Hodge supertrace; Vol~III
\texttt{cy\_d\_kappa\_stratification.tex}, $d = 4$ row),
so the genus tower is trivial at the scalar level even though
the algebra itself may be nontrivial.
\end{remark}

\begin{remark}[CY $d = 5$ Fukaya: Calabi--Yau or fake-Calabi--Yau?]
\label{rem:fukaya-d5}
At $d = 5$, the Hodge supertrace gives generic
$\kappa_{\mathrm{ch}} = 0$ (Vol~III \texttt{cy\_d\_kappa\_stratification.tex},
$d = 5$ row, generic). The Fukaya category $\Fuk(X)$ for
$X$ a CY $5$-fold receives the $\Eone$-stabilisation from
$\pi_5(BU) = 0$ (Bott periodicity). The deepest open problem at
$d = 5$ is the construction of explicit examples beyond the
generic case: known compact CY $5$-folds (analogous to the
quintic at $d = 3$) admit Fukaya categories whose
$\Phi$-output has not been computed.
\end{remark}

\begin{remark}[Symplectic Bridgeland stability as the open frontier]
\label{rem:fukaya-bridgeland-frontier}
On the B-side, Bridgeland stability on $D^b(\Coh(X^\vee))$
pairs with the modular Maurer--Cartan tower of the mirror chiral
algebra via the wall-crossing structure developed in
Chapter~\ref{ch:coha-wall-crossing-platonic}. No symplectic
Bridgeland stability is currently defined on $\Fuk(X)$: the
central charge $Z: K_0(\Fuk(X)) \to \C$ requires a period integral
whose A-model analogue (area of holomorphic discs with boundary on
$L$) is multivalued under the Fukaya $A_\infty$-structure.
Constructing such a $Z$, and matching its wall structure to the
chiral MC tower of $A_{\Fuk(X)}$, is the obstruction that closes
the A-side of the symplectic/categorical correspondence.
\end{remark}

\begin{remark}[Fukaya category of K3 and the HMS bridge]

The Fukaya category $\mathrm{Fuk}(K3)$ developed here enters the K3 chiral bialgebra

symmetry equivalence
\[
D^\pi\mathrm{Fuk}(K3)\ \simeq\ D^b\mathrm{Coh}(K3^\vee),
\]
with $K3^\vee$ the mirror K3. The mirror dictionary maps Mukai pairing
signature $(4,20)$ on $K3$ to signature $(20,4)$ on $K3^\vee$, i.e.\

Primary-lit: Fukaya 1993, Kontsevich 1994 HMS, Kontsevich-Soibelman 2008

\end{remark}

\begin{remark}[Fukaya-side analogue of
$\mathbf{H}_{\Delta_5}^{\mathrm{Enr}}$: free-involution gauging of
$\mathrm{Fuk}(K3)$]
\label{rem:fukaya-enriques}
\ClaimStatusProvedElsewhere
Let $\iota\colon K3\to K3$ be the fixed-point-free Enriques
involution. On the symplectic side, $\iota$ is a symplectic
involution acting by~$-1$ on $H^0(K3,\omega_{K3})$ and freely on
$K3$, so the quotient $K3/\iota = \mathrm{En}$ is a smooth
Enriques surface and the pulled-back symplectic structure on
$\mathrm{En}$ descends to a \emph{twisted} symplectic form valued
in the local system $\omega_{K3}/\iota$ (Dolgachev--Kond\={o} 2002
\textup{Asian~J.~Math.}~6, Theorem~2.2). The Fukaya category
$\mathrm{Fuk}(\mathrm{En})$ is the $\iota$-equivariant Fukaya
category of K3 in the sense of Seidel--Smith 2010
\textup{Duke~Math.~J.}~134 (Floer theory for Lagrangians with
non-trivial anti-symplectic involution), and the descent is clean
because $\iota$ is free. Under $\Phi_2$,
\[
  \Phi_2\bigl(\mathrm{Fuk}(\mathrm{En})\bigr)
  \;=\;
  \mathbf{H}_{\Delta_5}^{\mathrm{Enr}}
  \;=\;
  \Phi_2\bigl(\mathrm{Fuk}(K3)\bigr)\,/\!\!/\,\langle\iota\rangle,
\]
matching the B-side computation
(Vol.~III \texttt{derived\_categories\_cy.tex}
Theorem~\ref{thm:dcy-enriques-analogue}) via HMS for the
Enriques pair (Huybrechts 2016 \textup{Lectures on K3 surfaces},
Chapter~14, Theorem~14.12: $D^\pi\mathrm{Fuk}(\mathrm{En})
\simeq D^b\mathrm{Coh}(\mathrm{En}^\vee)$ with
$\mathrm{En}^\vee$ the mirror Enriques). The A-side Mukai pairing
signature drops from $(4,20)$ to $(2,10)$ under $\iota$-gauging;
the new $c_+ = 2$ is the Fukaya-face of the Enriques Mukai pairing
(Nikulin 1979 \textup{Izv.~Akad.~Nauk}~43, Proposition~4.1).
\end{remark}

\begin{remark}[Fukaya-side analogue for $T^4$: SYZ mirror of Narain
Heisenberg]
\label{rem:fukaya-t4}
\ClaimStatusProvedElsewhere
For the complex torus $T^4 = \mathbb{C}^2/\Lambda$, the
Fukaya category $\mathrm{Fuk}(T^4)$ has Lagrangian objects given
by affine Lagrangian subtori with flat $U(1)$-local systems; the
$A_\infty$-structure is polynomial in the Novikov parameter
(Fukaya 2002 \textup{Kyoto~J.~Math.}~42, Theorem~1.1; Kontsevich--Soibelman
2001 \textup{Progress in Math.}~244, \S10). The Strominger--Yau--Zaslow
mirror $T^4{}^\vee$ is again $T^4$ (with dualised fibration);
mirror symmetry
$D^\pi\mathrm{Fuk}(T^4)\simeq D^b\mathrm{Coh}(T^4{}^\vee)$ is
proved by Polishchuk--Zaslow 1998 \textup{Adv.~Theor.~Math.~Phys.}~2,
Theorem~0.1 and Kontsevich--Soibelman 2001, Theorem~1. Under
$\Phi_2$,
\[
  \Phi_2\bigl(\mathrm{Fuk}(T^4)\bigr)
  \;=\;
  V_{\mathrm{II}_{4,4}}
  \;=\;
  \Phi_2\bigl(D^b\mathrm{Coh}(T^4{}^\vee)\bigr),
\]
the Narain Heisenberg VOA of signature $(4,4)$, matching
Remark~\ref{rem:dcy-t4-abelian}. The
$\mathrm{O}(4,4;\mathbb{Z})$ T-duality group acts by the
automorphism group of $\mathrm{II}_{4,4}$ on the chiral side; on
the symplectic side it is the full mapping class group
acting on $\mathrm{Fuk}(T^4)$ by wall-crossing autoequivalences
(Seidel 2003 \textup{Bull.~London~Math.~Soc.}~35 for the $T^2$
case; Abouzaid--Smith 2010 for the higher-dimensional case).
\end{remark}

\begin{remark}[Fukaya-side half-K3: the broken mirror]
\label{rem:fukaya-halfk3}
\ClaimStatusProvedElsewhere
The rational elliptic surface
$\mathrm{dP}_9 = \mathrm{Bl}_9\mathbb{P}^2$ is not Calabi--Yau
($c_1 = -f\ne 0$), so its Fukaya category is defined only as a
\emph{relative} Fukaya category over the complement of the
anti-canonical divisor, following Seidel 2001
\textup{J.~Diff.~Geom.}~52 (exact Lefschetz fibrations) and
Auroux 2007 \textup{J.~G\"okova~Geom.~Topol.}~1 (Fukaya categories
on Landau--Ginzburg models). The mirror of $\mathrm{dP}_9$ is a
Landau--Ginzburg model $(W, X)$ with $W$ the $E_8$-root
potential (Auroux 2007, \S 6); mirror symmetry at $d = 2$ for
non-CY surfaces identifies
\[
  D^\pi\mathrm{Fuk}(\mathrm{dP}_9)
  \;\simeq\;
  D^b\mathrm{Sing}(W_{E_8}),
\]
the derived category of singularities of the $E_8$-potential
(Auroux 2007, Theorem~2). Under the $d=2$ extension of
$\Phi_2$ to non-CY relative Fukaya categories
(Vol.~III \texttt{matrix\_factorizations.tex},
Section~\ref{sec:lg-cy-correspondence}), the image is the affine
$\widehat{E_8}_{\,k=1}$ Kac--Moody algebra, not
$\mathbf{H}_{\Delta_5}$, matching the B-side answer in
Remark~\ref{rem:dcy-halfk3}. This is the symplectic
realisation of the taxon-L stratum of the CY-$2$ dichotomy
(Remark~\ref{rem:dcy-landscape-partition}): the
anti-canonical class obstructs the $\mathbf{H}_{\Delta_5}$-type
fourth-taxon output and forces a taxon-L output instead.
\end{remark}

\begin{remark}[Fukaya-side CY-$2$ landscape summary: the five A-side rows]
\label{rem:fukaya-landscape-summary}
\ClaimStatusProvedHere
The A-side CY-$2$ landscape under $\Phi_2\circ\mathrm{Fuk}$ reads
as follows, paralleling the B-side table of
Remark~\ref{rem:dcy-landscape-partition}:
\begin{itemize}
\item $\mathrm{Fuk}(K3)$: 4th taxon crown
  $\mathbf{H}_{\Delta_5}$, $c_+ = 4$, Siegel weight $5$;
  Lagrangian-intersection Mukai pairing of signature $(4,20)$.
\item $\mathrm{Fuk}(\mathrm{En})$: 4th taxon $\mathbb{Z}/2$-orbifold
  $\mathbf{H}_{\Delta_5}^{\mathrm{Enr}}$, $c_+ = 2$, Siegel weight

\item $\mathrm{Fuk}(T^4)$: taxon G, Narain Heisenberg
  $V_{\mathrm{II}_{4,4}}$, $c_+ = 4$, no Siegel lift;

landscape.
\end{remark}
