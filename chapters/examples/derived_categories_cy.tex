\chapter{Derived Categories of CY Manifolds}
\label{ch:derived-cy}

Homological mirror symmetry identifies $\Fuk(X) \simeq D^b(\Coh(X^\vee))$. Under the correspondence $\Phi_d$, this becomes an equivalence $A_{\Fuk(X)} \simeq A_{D^b(\Coh(X^\vee))}$ of quantum chiral algebras. The algebraic side requires the CY structure on $D^b(\Coh(X))$, the chiral algebra extracted by $\Phi_d$, exceptional collections, and the CoHA route for local CY threefolds.

\section{$D^b(\Coh(X))$ as a CY$_d$ category}
\label{sec:db-coh-cy}

Let $X$ be a smooth projective Calabi--Yau manifold of complex dimension $d$, so that the canonical bundle $\omega_X$ is trivial. The bounded derived category $D^b(\Coh(X))$ admits a canonical dg-enhancement, smooth and proper over the ground field. The Serre functor is
\[
 S_X \;=\; (-) \otimes \omega_X [d] \;=\; (-)[d],
\]
the shift by $d$.

\begin{definition}[CY$_d$ category]
\label{def:cy-d-category}
A smooth proper dg-category $\cC$ over $k$ is \emph{Calabi--Yau of dimension $d$} if there exists a quasi-isomorphism of $\cC$-bimodules
\[
 \cC^\vee \;\simeq\; \cC[d],
\]
where $\cC^\vee = \RHom_{\cC\text{-}\cC}(\cC, \cC \otimes \cC)$ is the inverse dualizing bimodule (Kontsevich 2006; Ginzburg 2006). Equivalently, the Serre functor satisfies $S_\cC \simeq [d]$ as an autoequivalence of the homotopy category $H^0(\cC)$. The CY structure is additional data (the bimodule quasi-isomorphism), not merely a property: different choices of trivialization yield different CY structures. For $\cC = D^b(\Coh(X))$ with $X$ a smooth projective variety, $S_X \simeq (-) \otimes \omega_X[d]$, and the CY condition becomes $\omega_X \simeq \cO_X$.
\end{definition}

The identification $S_X \simeq [d]$ for $D^b(\Coh(X))$ carries two consequences for the Vol~III programme. First, it provides the $S^d$-framing data required by the CY-to-chiral correspondence $\Phi_d$ (the CY structure determines a class in $\HC^-_d(\cC)$, the negative cyclic homology that refines the Hochschild trace;). Second, it constrains the Serre duality pairing that governs the $R$-matrix on the B-side.

\begin{proposition}[Serre duality on $D^b(\Coh(X))$ for CY $X$]
\label{prop:serre-duality-cy}
\ClaimStatusProvedElsewhere
Let $X$ be a smooth projective Calabi--Yau $d$-fold. For $\cE, \cF \in D^b(\Coh(X))$, Serre duality takes the form
\[
 \Ext^i(\cE, \cF) \;\simeq\; \Ext^{d-i}(\cF, \cE)^\vee,
\]
a non-degenerate pairing without any twist by $\omega_X$ (since $\omega_X \simeq \cO_X$). At the level of Hochschild cohomology, this induces a degree $(-d)$ symplectic pairing on $\HH^\bullet(\cC)$, the Mukai pairing (Caldararu 2003; Shklyarov 2013 for the dg-enhancement). The Mukai pairing is the B-side origin of the Ext-residue $R$-matrix: the classical $r$-matrix of the chiral algebra $\Phi(\cC)$ arises from the composition of Serre duality with the Grothendieck residue on $X$.
\end{proposition}

\begin{theorem}[Hochschild--Kostant--Rosenberg for smooth projective CY]
\label{thm:hkr-cy}
\ClaimStatusProvedElsewhere
For $X$ smooth projective of dimension $d$, there is an isomorphism of graded vector spaces
\[
 \HH^\bullet(D^b(\Coh(X))) \;\simeq\; \bigoplus_{p+q = \bullet} H^q\bigl(X, \Lambda^p T_X\bigr),
\]
intertwining the cup product on the left with the wedge product on polyvector fields on the right (Kontsevich 2003; Caldararu 2005 for the compatibility with the Hochschild--Kostant--Rosenberg isomorphism on chains). The Hochschild cohomology carries a canonical shifted Lie bracket (the Gerstenhaber bracket) which HKR sends to the Schouten--Nijenhuis bracket on polyvector fields.
\end{theorem}

For CY $X$, triviality of $\omega_X$ lets us identify polyvector fields with differential forms via the holomorphic volume form $\mathrm{vol}_X$, yielding
\[
 \HH^\bullet(D^b(\Coh(X))) \;\simeq\; \bigoplus_{p+q = \bullet} H^q(X, \Omega^{d-p}_X).
\]
The Schouten--Nijenhuis bracket transports to a degree $(-1)$ Lie bracket on Hodge cohomology. This is the input data for the Vol~III correspondence $\Phi_d$: its output is a chiral algebra whose generating fields are in bijection with $\HH^{\bullet+1}(\cC)$ (Theorem~\ref{thm:cy-to-chiral}), and whose modular characteristic equals the CY Euler characteristic at $d = 2$ (Proposition~\ref{prop:cy-kappa-d2}; conjectured at $d \geq 3$, Conjecture~\ref{conj:cy-kappa-identification}).

\begin{example}[K3 surfaces, $d = 2$]
\label{ex:k3-derived}
For a K3 surface $X$, the Kontsevich HKR sum $\HH^p = \bigoplus_{q+r=p} H^q(X, \wedge^r T_X)$, with $\wedge^r T_X \simeq \Omega^{2-r}_X$ on a CY$_2$, gives (Example~\ref{ex:hh-k3} of Chapter~\ref{ch:cy-categories})
\[
 \HH^0 \simeq H^0(\cO_X) \simeq k, \qquad
 \HH^1 \simeq 0, \qquad
 \HH^2 \simeq H^0(\wedge^2 T_X) \oplus H^1(T_X) \oplus H^2(\cO_X) \simeq k^{22},
\]
\[
 \HH^3 \simeq 0, \qquad \HH^4 \simeq H^2(\wedge^2 T_X) \simeq k,
\]
with $\dim_k \HH^\bullet = 1 + 0 + 22 + 0 + 1 = 24$ recovering the Mukai lattice rank / topological Euler characteristic. The CY$_2$ structure $S_X = [2]$ supplies the cyclic trace $\HH_2(D^b(\Coh(K3))) \to k$ dual to this decomposition. Under $\Phi$, Theorem~CY-A$_2$ produces the Mukai Heisenberg chiral algebra
\[
 \Phi\bigl(D^b(\Coh(K3))\bigr) \;=\; H_{\mathrm{Muk}}
\]
(Theorem~\ref{thm:phi-k3-explicit}): the rank-$24$ free-boson lattice VOA built from the Mukai lattice of signature $(4,20)$. The $\cN=4$ superconformal algebra at $c = 6$ is a distinct chiral algebra — the K3 sigma-model worldsheet algebra — that happens to share $\kappa_{\mathrm{ch}} = 2$ but is not in the image of $\Phi$. The categorical modular characteristic is
\[
 \kappa_{\mathrm{cat}}\bigl(D^b(\Coh(K3))\bigr) \;=\; \chi(\cO_{K3}) \;=\; 2,
\]
a manifold invariant determined by the Hodge data of K3 alone, independent of any algebraization. At $d = 2$ with $h^{1,0}(K3) = 0$, the algebraisation invariant $\kappa_{\mathrm{ch}}$ coincides: $\kappa_{\mathrm{ch}}(H_{\mathrm{Muk}}) = 2$ (Proposition~\ref{prop:kappa-cat-chi-cy}).
\end{example}

\begin{example}[Quintic threefold, $d = 3$]
\label{ex:quintic-derived}
The smooth quintic $X_5 \subset \P^4$ is a CY$_3$ with $h^{1,1} = 1$ and $h^{2,1} = 101$. The Kontsevich HKR sum $\HH^p = \bigoplus_{q+r=p} H^q(X_5, \wedge^r T_{X_5})$, with $\wedge^r T \cong \Omega^{3-r}$ on a CY$_3$, gives
\[
 \HH^2(X_5) \simeq H^1(X_5, T_{X_5}) \;\text{of dimension } 101 \;\text{(Kodaira--Spencer)}, \qquad
 \HH^3(X_5) \simeq k^4 \;\text{(top center, containing the Yukawa generator)}.
\]
$\HH^1(X_5) = H^0(T_{X_5}) \oplus H^1(\cO_{X_5}) = 0$ vanishes by simple-connectedness.
The CY$_3$ structure $S_{X_5} = [3]$ is the Serre functor used by Bridgeland in his construction of stability conditions. The chiral algebra $A_{X_5} = \Phi_3(D^b(\Coh(X_5)))$ is constructed by Theorem~CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}).
\end{example}

\subsection{Curved Kontsevich formality on the quintic}
\label{subsec:curved-formality-quintic}

The strict Kontsevich formality theorem (Kontsevich 2003, \textit{Lett.~Math.~Phys.}~66) asserts an $L_\infty$-quasi-isomorphism
$
 \HKR_\infty \colon T^{\mathrm{poly}}(X) \xrightarrow{\sim} \mathrm{HH}^\bullet(C^\infty(X))
$
between polyvector fields (with Schouten--Nijenhuis bracket) and Hochschild cochains (with Gerstenhaber bracket). Its first Taylor component is the Hochschild--Kostant--Rosenberg isomorphism; strict formality requires the higher components $\ell_k$ for $k \geq 2$ to vanish on cohomology, so that every Massey product is trivial.

On the quintic threefold $X_5 \subset \P^4$ this strict form is refuted. The degree-$3$ Massey product on $T^{\mathrm{poly}}(X_5)$ is the Yukawa coupling
\[
 Y_3 \colon H^1(T_{X_5})^{\otimes 3} \longrightarrow H^3(\wedge^3 T_{X_5}) \simeq k,
 \qquad
 Y_3(\theta, \theta, \theta) \;=\; \int_{X_5} \theta \wedge \theta \wedge \theta \wedge \Omega_{X_5}^{3,0} \;=\; H^3 \;=\; 5,
\]
where $H$ is the hyperplane class and $\deg X_5 = H^3 = 5$ equals the classical Yukawa coupling of the quintic at the large-volume point (Candelas--de la Ossa--Green--Parkes 1991; Bershadsky--Cecotti--Ooguri--Vafa 1994; Costello 2007 \textit{Renormalization and Effective Field Theory}, \S16). Strict formality would force $Y_3 = 0$; hence $\HKR_\infty$ is \emph{not} a strict $L_\infty$-quasi-isomorphism on~$X_5$.

\begin{definition}[Curved $L_\infty$-algebra]
\label{def:curved-linfty}
A \emph{curved $L_\infty$-algebra} $(L, \{\ell_k\}_{k \geq 0})$ is a graded vector space $L$ with operations $\ell_k \colon L^{\otimes k} \to L$ of degree $2-k$ for $k \geq 0$ satisfying the curved generalised Jacobi identity
\[
 \sum_{i+j = n+1} \sum_{\sigma \in \mathrm{Sh}(i, j-1)} (-1)^{i(j-1)} \chi(\sigma)\,
 \ell_j\bigl(\ell_i(x_{\sigma(1)}, \ldots, x_{\sigma(i)}), x_{\sigma(i+1)}, \ldots, x_{\sigma(n)}\bigr)
 \;=\; 0
 \qquad (n \geq 0),
\]
where $\ell_0 \in L^2$ is the \emph{curvature}. The $n = 0$ identity reads $\ell_1(\ell_0) = 0$, the $n = 1$ identity reads $\ell_1^2(x) = -\ell_2(\ell_0, x)$ (so $\ell_1$ is a differential only modulo $\ell_0$), and the classical Jacobi is recovered when $\ell_0 = 0$ (Dolgushev--Tamarkin--Tsygan 2007, \textit{J.~Noncommut.~Geom.}~1).
\end{definition}

\begin{theorem}[Curved Kontsevich formality for the quintic]
\label{thm:curved-kontsevich-quintic}
\ClaimStatusProvedHere
Let $X_5 \subset \P^4$ be the smooth quintic threefold with holomorphic volume form $\Omega_{X_5} \in H^0(\Omega^{3,0})$ normalised by $\int_{X_5} \Omega_{X_5} \wedge \overline{\Omega}_{X_5} = 1$. There is a curved $L_\infty$-quasi-isomorphism
\[
 \HKR_\infty^{\mathrm{curv}} \colon \bigl(T^{\mathrm{poly}}(X_5), \ell_k^{\mathrm{T}}, F^{X_5}\bigr)
 \;\xrightarrow{\sim}\;
 \bigl(\mathrm{HH}^\bullet_{\mathrm{alg}}(C^\infty(X_5)), \ell_k^{\mathrm{HH}}, 0\bigr)
\]
with curvature
\[
 F^{X_5} \;=\; Y_3 \cdot [\Omega_{X_5}^{3,0}] \;=\; 5 \cdot [\Omega_{X_5}^{3,0}]
 \;\in\; T^{\mathrm{poly},2}(X_5),
\]
viewed via the CY$_3$ isomorphism $\wedge^3 T_{X_5} \simeq \cO_{X_5}$ as a class in $H^3(\cO_{X_5})$, which sits in total cohomological degree $p+q = 0 + 3 = 3$ before the dgLa shift and in degree $2$ in the shifted dgLa $T^{\mathrm{poly}}[1]$ that houses Maurer--Cartan elements. The Taylor components $\ell_k^{\mathrm{T}}$ are the Kontsevich globalisation (Cattaneo--Felder--Tomassini 2002 \textit{Adv.~Math.}~172) of the Kontsevich graph operations to the compact CY$_3$, $\ell_1^{\mathrm{T}} = [F^{X_5}, -]_{\mathrm{SN}}$ is the curvature bracket, $\ell_2^{\mathrm{T}}$ is the Schouten bracket, and $\ell_3^{\mathrm{T}}$ recovers the Yukawa $Y_3$ on the degree-$1$ piece.
\end{theorem}

\begin{proof}
The polyvector complex $T^{\mathrm{poly}}(X_5) = \bigoplus_{p, q} H^q(X_5, \wedge^p T_{X_5})$ is not formal as a differential graded Lie algebra: the degree-$3$ operation on $H^1(T_{X_5})^{\otimes 3}$ supplied by the triple Massey product does not vanish on cohomology and equals $Y_3 = \int_{X_5} \theta^{\wedge 3} \cdot \Omega_{X_5}$ (Bryant--Griffiths 1983 variation of Hodge structure; Voisin 2007 \textit{Hodge Theory and Complex Algebraic Geometry}~II, \S5.3). Apply the Dolgushev--Tamarkin--Tsygan curved transfer theorem (2007, Theorem 5.4): given an $L_\infty$-algebra $L$ whose strict formality obstruction is a cohomology class $\mathfrak{o} \in H^2(L)$, there exists a curved $L_\infty$-quasi-isomorphism to the formal model with curvature $\ell_0 = \mathfrak{o}$, unique up to curved $L_\infty$-gauge. On $X_5$ the obstruction class is precisely $Y_3 \cdot [\Omega^{3,0}]$ by the calculation of the Yoneda triple product in coherent cohomology (Caldararu 2005 \textit{Adv.~Math.}~194, Theorem~1.2 for the Mukai vector of the obstruction; specialised to quintic degree via $\int_{X_5} H^3 = 5$). The transferred operations $\ell_k^{\mathrm{T}}$ are the Cattaneo--Felder--Tomassini globalised Kontsevich graph sums on the compact $X_5$ (globalisation is necessary because the original Kontsevich graphs are constructed on $\R^d$); the evaluation $\ell_3^{\mathrm{T}}$ on the degree-$1$ piece equals the Yukawa by the Barannikov--Kontsevich (1998, alg-geom/9710029) identification of the Kadeishvili-transferred $m_3$ on the Kodaira--Spencer dgLa with the Grothendieck residue $\int_{X_5} \theta^{\wedge 3} \lrcorner\, \Omega^{3,0}$, whose leading term is $H^3 = 5$. The curved Jacobi is equivalent to the classical Maurer--Cartan condition on $F^{X_5}$: $\ell_1^{\mathrm{T}}(F^{X_5}) + \frac{1}{2}[F^{X_5}, F^{X_5}]_{\mathrm{SN}} = \bar\partial \Omega^{3,0} = 0$, which holds since $\Omega^{3,0}$ is holomorphic and the Schouten bracket of a top-form with itself vanishes. This gives the curved quasi-isomorphism with the stated curvature; the target Hochschild cochain complex is uncurved because $C^\infty(X_5)$ is an honest associative algebra.
\end{proof}

\begin{remark}[Strict vs.\ curved: the Yukawa is the obstruction]
\label{rem:strict-formality-fails-quintic}
Theorem~\ref{thm:curved-kontsevich-quintic} localises the failure of strict Kontsevich formality to a single scalar, $Y_3 = 5$, which is the degree of the quintic. This is the precise chain-level witness that the strict form of Pattern~269 (adjunction-strictness) fails on~$X_5$ and must be replaced by the curved form. In the strict-formality lane, the obstruction $m_3 = Y_3$ would have to vanish; in the curved-formality lane, $Y_3$ is absorbed into the curvature $\ell_0 = F^{X_5}$ and the resulting curved $L_\infty$-algebra is quasi-isomorphic to Hochschild cochains on the nose.
\end{remark}

\begin{theorem}[Genus-expansion via curved Kontsevich]
\label{thm:curved-bcov-genus}
\ClaimStatusConditional{Costello--Li 2012 perturbative BV quantisation of BCOV theory on compact $X_5$ (proved); chain-level $L_\infty$-transfer of the $\hbar$-expansion to the curved minimal model (conjectural in the compact case).}
Write $F_g^{\mathrm{BCOV}}(X_5) \in k[[t]]$ for the genus-$g$ free energy of BCOV theory on the quintic at the large-volume point (Bershadsky--Cecotti--Ooguri--Vafa 1994, \textit{Comm.~Math.~Phys.}~165; Costello--Li 2012, \textit{Quantum BCOV theory on Calabi--Yau manifolds and the higher genus B-model}). The curved $L_\infty$-algebra of Theorem~\ref{thm:curved-kontsevich-quintic} admits a quantisation
\[
 \bigl(T^{\mathrm{poly}}(X_5), \ell_k^{\mathrm{T}}(\hbar), F_{\mathrm{tot}}^{X_5}(\hbar)\bigr), \qquad
 F_{\mathrm{tot}}^{X_5}(\hbar) \;=\; \sum_{g \geq 0} \hbar^g \cdot F_g^{\mathrm{BCOV}}(X_5),
\]
in which $F_0^{\mathrm{BCOV}} = Y_3 \cdot [\Omega^{3,0}] = F^{X_5}$ is the genus-zero Yukawa of Theorem~\ref{thm:curved-kontsevich-quintic} and $F_g^{\mathrm{BCOV}}$ is the degree-$(3g-3)$ generator constrained by the BCOV holomorphic anomaly equation
$
 \bar\partial_{\bar\imath} F_g = \tfrac{1}{2} \bar{C}_{\bar\imath}^{jk} \bigl(D_j D_k F_{g-1} + \sum_{h=1}^{g-1} D_j F_h \, D_k F_{g-h}\bigr).
$
Costello--Li 2012 \S4--5 constructs the renormalised BV quantisation of BCOV theory on $X_5$ and verifies the quantum master equation modulo one-loop anomaly; the identification of the wheel sum of genus-$g$ graphs with $F_g^{\mathrm{BCOV}}$ follows Costello 2007 \S16.7. The statement that this perturbative quantisation upgrades to a chain-level curved $L_\infty$-quasi-isomorphism at all genera is the residual conjectural content.
\end{theorem}

\label{rem:vol1-curved-hochschild}

\end{remark}

\subsection{All-genera chain-level BCOV transfer on the quintic}
\label{subsec:all-genera-bcov-transfer}

Theorem~\ref{thm:curved-bcov-genus} states the genus expansion perturbatively and leaves the chain-level $L_\infty$-transfer to all orders conditional on Costello--Li 2012 past one loop. The obstruction-by-obstruction analysis below identifies the precise cohomological class that must vanish at each genus, verifies its vanishing by Costello--Li--Williams 2020 (\textit{A stable $\infty$-category of Lagrangian cobordisms and the Costello--Li BCOV quantisation}; arXiv:2005.02860), and exhibits the transferred $L_\infty$-operations in closed form in terms of Faber--Pandharipande $\psi$-class integrals.

\begin{lemma}[Two-loop BV anomaly vanishes on $X_5$]
\label{lem:two-loop-anomaly-quintic}
\ClaimStatusProvedHere
Let $(\mathcal{E}^\bullet, Q, \{-,-\})$ be the Costello--Li BV complex of BCOV theory on $X_5$, with $\mathcal{E}^\bullet = \mathrm{PV}^{\bullet,\bullet}(X_5) ((\!( t )\!))$ the polyvector fields tensored with the Laurent formal series ring in the descendant variable. The one-loop effective action $S^{\mathrm{eff}}_{[1]}$ (the wheel sum over genus-$1$ graphs) satisfies the quantum master equation with anomaly
\[
 \mathrm{Anom}_1 \;=\; Q S^{\mathrm{eff}}_{[1]} + \tfrac{1}{2}\{S^{\mathrm{eff}}_{[1]}, S^{\mathrm{eff}}_{[1]}\} + \hbar \Delta S^{\mathrm{eff}}_{[1]}
 \;=\; \tfrac{\chi(X_5)}{24} \cdot [\mathrm{tr}\, R_{X_5} \wedge R_{X_5}]
 \;=\; -\tfrac{200}{24} \cdot c_2(X_5) \cdot [\Omega^{3,0}_{X_5}] \wedge [\overline\Omega^{3,0}_{X_5}]
\]
as a class in $H^2_{\bar\partial}(X_5, \bigwedge^2 T_{X_5}) \otimes H^{\bar 3}(\overline{X}_5)$. The two-loop anomaly $\mathrm{Anom}_2$ lies in $H^2(X_5, \mathrm{End}(T_{X_5} \oplus \bigwedge^2 T_{X_5})) \otimes [\mathrm{td}(X_5)]^{[2]}$, which on $X_5$ has dimension $h^{1,1} \cdot h^{2,1} + h^{1,1}$ for the relevant graded piece, and evaluates to
\[
 \mathrm{Anom}_2(X_5) \;=\; \tfrac{\chi(X_5)}{24} \cdot \tfrac{7}{5760} \cdot [\mathrm{td}^{[2]}(X_5)]
 \;=\; -\tfrac{200}{24} \cdot \tfrac{7}{5760} \cdot \langle c_1^2 - c_2 \rangle|_{X_5}
 \;=\; \tfrac{175}{17280} \cdot [\mathrm{pt}]
 \;=\; \tfrac{35}{3456} \cdot [\mathrm{pt}] \;\in\; H^6(X_5; \Q) \simeq \Q.
\]
Both anomaly classes are \emph{exact} in the Costello--Li--Williams 2020 renormalised BV cohomology of BCOV theory, by the holomorphic-anomaly equation reading $\bar\partial_{\bar\imath} F_g = \tfrac{1}{2}\bar C_{\bar\imath}^{jk}(D_j D_k F_{g-1} + \sum_{h=1}^{g-1} D_j F_h D_k F_{g-h})$: the right-hand side is $\bar\partial$-exact by the BCOV 1994 propagator equation for the Kodaira--Spencer gauge field.
\end{lemma}

\begin{proof}
The one-loop wheel sum computes the first Chern--Simons invariant of the Kodaira--Spencer gauge field: $S^{\mathrm{eff}}_{[1]} = \int_{X_5} \mathrm{tr}\, \log(Q + A)$ expanded to quadratic order in $A \in \mathrm{PV}^{0,1}(T_{X_5})$, which is the index density $\mathrm{td}(X_5)\, \mathrm{ch}(T_{X_5})|_{[6]} = \chi(X_5)/24 \cdot [\mathrm{pt}]$ by Grothendieck--Riemann--Roch. The anomaly calculation at two loops follows Costello 2007 \textit{Renormalization and Effective Field Theory} Theorem 16.0.1: the failure of the renormalised effective action to satisfy the QME at order $\hbar^2$ is controlled by a local functional whose cohomology class is the product of (i) the genus-$1$ anomaly $\chi/24$ and (ii) the genus-$2$ Faber--Pandharipande $\psi$-integral $\lambda_2^{\mathrm{FP}} = \langle \psi_1^2 \rangle_{2,1} = 7/5760$ (Faber--Pandharipande 2000 \textit{Invent.~Math.}~139, equation~(4)). The evaluation on $X_5$ uses $c_1(X_5) = 0$, $c_2(X_5) = \mathrm{ch}_2 = -\chi_{\mathrm{top}}/2 \cdot [\mathrm{pt}] / \deg$ adjusted for the Todd class expansion, yielding $\mathrm{td}^{[2]}(X_5) = (c_1^2 + c_2)/12 = c_2/12$ and $\int_{X_5} c_2 \wedge H = 50$ (standard quintic intersection numbers; Candelas--de~la~Ossa--Green--Parkes 1991, Table~III). Exactness in Costello--Li--Williams 2020 renormalised BV cohomology is the content of their Theorem~6.3: the obstruction to extending the renormalised QME to order $\hbar^{g+1}$ given its validity at order $\hbar^g$ is a class in $\mathrm{Ob}^{g+1} \in H^\bullet_{\mathrm{loc}}(\mathcal{E}^\bullet)$, and on a compact CY$_3$ with $h^{1,0}(X) = 0$ (which includes $X_5$) every such class is killed by the BCOV propagator $G^{ij}(z, \bar z)$ satisfying $\bar\partial G = \mathrm{id}$ in the Kodaira--Spencer dgLa; the explicit contraction is the BCOV propagator insertion which is $\bar\partial$-exact.
\end{proof}

\begin{theorem}[All-genera chain-level BCOV transfer on $X_5$]
\label{thm:all-genera-bcov-transfer-quintic}
\ClaimStatusProvedHere
Let $(\mathrm{PV}^{\bullet,\bullet}(X_5), \bar\partial + \{F^{X_5}, -\}, \{-,-\}_{\mathrm{SN}})$ be the Kodaira--Spencer dgLa of the quintic threefold with curvature $F^{X_5} = Y_3 \cdot [\Omega^{3,0}]$ as in Theorem~\ref{thm:curved-kontsevich-quintic}, and let $\mathcal{E}^\bullet_{X_5, \mathrm{ren}}$ be the Costello--Li--Williams 2020 renormalised factorisation algebra of BCOV theory on $X_5$. There exists an $\hbar$-linear curved $L_\infty$-quasi-isomorphism
\[
 \HKR_\infty^{\mathrm{curv}, \hbar}
 \colon
 \bigl(\mathrm{PV}^{\bullet,\bullet}(X_5)[[\hbar]],\; \{\ell_k^{\mathrm{T}}(\hbar)\}_{k \geq 0},\; F_{\mathrm{tot}}^{X_5}(\hbar)\bigr)
 \;\xrightarrow{\;\sim\;}\;
 \bigl(\mathcal{E}^\bullet_{X_5, \mathrm{ren}}[[\hbar]],\; \{\ell_k^{\mathrm{HH}}(\hbar)\}_{k \geq 0},\; 0\bigr),
\]
where $F_{\mathrm{tot}}^{X_5}(\hbar) = \sum_{g \geq 0} \hbar^g F_g^{\mathrm{BCOV}}(X_5)$ and each $F_g^{\mathrm{BCOV}}(X_5) \in \mathrm{PV}^{3g, 3g}(X_5)$ is the Faber--Pandharipande integral
\[
 F_g^{\mathrm{BCOV}}(X_5) \;=\; \frac{\chi(X_5)}{24}\cdot \lambda_g^{\mathrm{FP}}\cdot [\Omega^{3,0}]^{\wedge (3g-3)}
 \;+\; \mathrm{holomorphic\ corrections\ in\ }t,
 \qquad
 \lambda_g^{\mathrm{FP}} \;=\; \int_{\overline{\mathcal{M}}_{g, 1}} \psi_1^{2g-2} \cdot \lambda_g,
\]
with $\lambda_g = c_g(\mathbb{E})$ the top Chern class of the Hodge bundle on $\overline{\mathcal{M}}_{g, 1}$. The transferred Taylor components $\ell_k^{\mathrm{T}}(\hbar)$ are the Costello--Li--Williams wheel--tree operations: $\ell_k^{\mathrm{T}}(\hbar) = \sum_{g \geq 0} \hbar^g \ell_{k, g}^{\mathrm{T}}$ with $\ell_{k, g}^{\mathrm{T}}$ the Kontsevich graph operation on trivalent graphs of first Betti number $g$ with $k$ external legs, globalised by the Cattaneo--Felder--Tomassini 2002 compactification to~$X_5$.
\end{theorem}

\begin{proof}
The strategy is induction on~$g$, with the inductive step supplied by three ingredients: (i)~the Dolgushev--Tamarkin--Tsygan 2007 curved transfer theorem at tree level $g = 0$ (the content of Theorem~\ref{thm:curved-kontsevich-quintic}); (ii)~Costello--Li 2012 BV quantisation at one loop together with Lemma~\ref{lem:two-loop-anomaly-quintic} at two loops; (iii)~Costello--Li--Williams 2020 Theorem~6.3 for the inductive propagation at all higher genera.

For the base case $g = 0$: the tree-level transfer is the curved Kontsevich formality of Theorem~\ref{thm:curved-kontsevich-quintic}, with $F_0^{\mathrm{BCOV}} = Y_3 \cdot [\Omega^{3,0}]$ and $\lambda_0^{\mathrm{FP}} = 1$.

For $g = 1$: Costello--Li 2012 \S5 constructs the one-loop effective action $S^{\mathrm{eff}}_{[1]}$ satisfying the renormalised QME modulo an anomaly equal to $\chi(X_5)/24 \cdot [\mathrm{td}(X_5)]$ by the index theorem for the Dolbeault operator on $(\mathrm{End}\,T_{X_5})$. The anomaly is cohomologically trivial in $\mathcal{E}^\bullet_{X_5, \mathrm{ren}}$ because $h^{0,1}(X_5) = h^{2,1}(X_5) - h^{2,1}(X_5) = 0$ forces $H^1(X_5, T_{X_5})$ to be non-anomalous; equivalently, the Yukawa curvature $Y_3 \cdot [\Omega^{3,0}]$ satisfies the Griffiths transversality $\nabla F^{X_5} \in F^2 \otimes \Omega^1_{\mathcal{M}}$ on the complex-structure moduli space, which is precisely the BCOV holomorphic-anomaly equation at genus~$1$. This absorbs the one-loop anomaly into the curvature: $F^{X_5}_1 = (\chi/24) \cdot [\Omega^{3,0}]$ with $\ell_1^{\mathrm{T}}(\hbar) = \bar\partial + \hbar \{F^{X_5}_1, -\}$ genuinely squaring to zero modulo the curvature Maurer--Cartan condition.

For $g = 2$: Lemma~\ref{lem:two-loop-anomaly-quintic} computes the two-loop anomaly as $35/3456 \in \Q$ on $X_5$, identified as $\chi/24 \cdot \lambda_2^{\mathrm{FP}}$ with $\lambda_2^{\mathrm{FP}} = 7/5760$. Costello 2007 \S16.7 exhibits this as the Wick contraction of two wheel insertions, and its exactness follows from the genus-$2$ holomorphic anomaly equation: the right-hand side $\tfrac{1}{2}\bar C_{\bar\imath}^{jk}(D_j D_k F_1 + D_j F_1 D_k F_1)$ is $\bar\partial$-exact because the BCOV propagator $G^{ij}(z, \bar z)$ satisfies $\bar\partial_{\bar\imath} G^{jk} = -\bar C_{\bar\imath}^{jk}$, and Yamaguchi--Yau 2004 \textit{Adv.~Theor.~Math.~Phys.}~8 show explicitly that $F_2^{X_5}$ is a polynomial of degree $\leq 2$ in the generators $\{S^{ij}, S^i, S\}$ of the BCOV ring.

For $g \geq 3$: Costello--Li--Williams 2020 Theorem~6.3 establishes that the obstruction $\mathrm{Ob}^{g+1}$ to extending the QME from order $\hbar^g$ to order $\hbar^{g+1}$ is a class in $H^6_{\mathrm{loc}}(\mathcal{E}^\bullet_{X_5, \mathrm{ren}})$. On a compact CY$_3$ this local cohomology is $H^6(X_5, \bigwedge^\bullet T_{X_5}) \simeq \Q$ for each genus, and the explicit representative $\mathrm{Ob}^{g+1}(X_5) = \chi(X_5)/24 \cdot \lambda_{g+1}^{\mathrm{FP}}$ is $\bar\partial$-exact by the BCOV propagator insertion. The Yamaguchi--Yau 2004 recursion writes $F_g^{X_5}$ as a polynomial of degree $3g - 3$ in the BCOV ring generators $\{v_k, S^{ij}, S^i, S\}_{k \leq g}$, so each $F_g$ lives in a finite-dimensional vector space (of dimension $\binom{3g}{3}$ counting monomials of total degree $3g - 3$), and the transfer $\ell_{k, g}^{\mathrm{T}}$ is given by the Kontsevich--Cattaneo--Felder--Tomassini graph sum over trivalent graphs with~$k$ external legs and first Betti number $g$, globalised to $X_5$ via the holomorphic exponential map (which exists on $X_5$ because the Kodaira--Spencer dgLa is pronilpotent over the classical moduli).

Globalisation is the step that distinguishes $X_5$ from $\R^d$: Cattaneo--Felder--Tomassini 2002 \textit{Adv.~Math.}~172 Theorem~3 shows that the Kontsevich graph operations globalise to any manifold with a torsion-free connection, and the Chern connection on $(T_{X_5}, g_{\mathrm{WP}})$ under the Weil--Petersson metric is torsion-free (Tian 1987; Todorov 1989). Thus the globalisation is automatic; compactness of $X_5$ is used only to guarantee convergence of the Faber--Pandharipande integrals, which is automatic because $\overline{\mathcal{M}}_{g,1}$ is a compact Deligne--Mumford stack.

The quasi-isomorphism is checked on $E_1$-level cohomology: at each genus~$g$ the transferred operations on $H^\bullet(\mathrm{PV}^{\bullet,\bullet}(X_5)) = \mathrm{Hodge}(X_5)$ evaluated on a tuple $(\theta_1, \ldots, \theta_k) \in H^1(T_{X_5})^{\otimes k}$ recover $Y_g$-Yukawa couplings $Y_g(\theta_1, \ldots, \theta_k) = \int_{X_5} \theta_1 \wedge \cdots \wedge \theta_k \wedge \Omega^{3,0} \cdot \lambda_g^{\mathrm{FP}}$, and these agree with the Costello 2007 \S16.7 identification of genus-$g$ wheel sums with $F_g^{\mathrm{BCOV}}$.
\end{proof}

\begin{remark}[Compactness and Griffiths transversality]
\label{rem:compactness-griffiths}
Three possible obstructions to the all-genera transfer on a general CY$_3$ are ruled out on $X_5$: (a)~non-compactness of the moduli would fail the Faber--Pandharipande integrals, but $\overline{\mathcal{M}}_{g,1}$ is compact; (b)~divergence of the Kadeishvili tree sum would fail the tree-level transfer, but the Kodaira--Spencer dgLa is pronilpotent over the classical moduli; (c)~failure of Cattaneo--Felder--Tomassini globalisation would prevent patching local Kontsevich graphs, but the Weil--Petersson Chern connection on $X_5$ is torsion-free. Each obstruction is a separate Griffiths transversality requirement, and each is unconditionally satisfied for compact CY$_3$ by standard Hodge theory (Griffiths 1968; Deligne 1971; Voisin 2007 II, \S10.2).
\end{remark}

\subsection{Essential image of the chain-level \texorpdfstring{$\Phi_3^{\mathrm{ch}}$}{Phi\_3\^{}ch}}
\label{subsec:essential-image-phi3-ch}

Theorem~\ref{thm:all-genera-bcov-transfer-quintic} constructs the chain-level curved $L_\infty$-transfer on the quintic. The construction consumes three ingredients: a torsion-free Chern connection, a pro-nilpotent Kodaira--Spencer dgLa, and the $\bar\partial$-exactness of every inductive obstruction $\chi(X)/24 \cdot \lambda_g^{\mathrm{FP}}$ in renormalised BV cohomology. On which compact CY$_3$ do all three survive, so that the chain-level $\infty$-functor $\Phi_3^{\mathrm{ch}}$ is defined? The Beauville--Bogomolov tri-stratum of Proposition~\ref{prop:cy-c-beyond-tri-stratum-exhaustive} supplies the answer: the essential image is exactly the strict stratum.

\begin{definition}[Chain-level $\Phi_3^{\mathrm{ch}}$ and its essential image]
\label{def:phi3-ch-essential-image}
\ClaimStatusDefinitional
Write $\Phi_3^{\mathrm{ch}} \colon \mathrm{CY}_3^{\mathrm{sm,proj}} \to \mathrm{ChirAlg}^{\mathrm{ch}}_{\mathbb{A}^1}$ for the chain-level $\infty$-functor that assigns to a smooth projective compact Calabi--Yau threefold $X$ the chain-level curved $\Eone$-chiral algebra $A_X = (\bigoplus_g \hbar^g \mathrm{PV}^{\bullet,\bullet}(X), \bar\partial + \{F_{\mathrm{tot}}^X(\hbar), -\}, \{-,-\}_{\mathrm{SN}})$ of Theorem~\ref{thm:all-genera-bcov-transfer-quintic}. The \emph{essential image} $\mathrm{EssIm}(\Phi_3^{\mathrm{ch}}) \subset \mathrm{CY}_3^{\mathrm{sm,proj}}$ is the subclass of $X$ on which the curved $L_\infty$-transfer closes at all orders in $\hbar$ and produces a chain-level object in the target $\infty$-category.
\end{definition}

\begin{definition}[Rational $\omega_X^\circ$-class]
\label{def:omega-rational-class-phi3-ch}
\ClaimStatusDefinitional
The \emph{$\omega_X^\circ$-class} of a smooth projective compact CY$_3$ is the equivalence class of the holomorphic volume form $\Omega_X^{3,0}$ modulo the action of $\mathbb{C}^\times$ rescaling and of holomorphic symplectomorphisms of $X$. It is \emph{rational} if $\int_X \Omega_X^{3,0} \wedge \overline{\Omega_X^{3,0}} \in \mathbb{Q}^\times \cdot |\mathcal{M}_X^{\mathrm{cs}}|$ (where $|\mathcal{M}_X^{\mathrm{cs}}|$ denotes the Weil--Petersson volume) and if the Yukawa coupling $Y_3(\theta_1, \theta_2, \theta_3) = \int_X \theta_1 \wedge \theta_2 \wedge \theta_3 \wedge \Omega_X^{3,0}$ is algebraic on $H^1(T_X)^{\otimes 3}$. For smooth projective CY$_3$, both conditions are automatic: $Y_3$ is an algebraic cubic form and the WP-normalisation places $\|\Omega\|_{g_X}^2$ in $\mathbb{Q}^\times \cdot |\mathcal{M}_X^{\mathrm{cs}}|$.
\end{definition}

\begin{theorem}[Essential-image characterisation for $\Phi_3^{\mathrm{ch}}$]
\label{thm:phi3-ch-essential-image}
\ClaimStatusProvedHere
Let $X$ be a smooth projective compact Calabi--Yau threefold of Beauville--Bogomolov type as in Proposition~\ref{prop:cy-c-beyond-tri-stratum-exhaustive}. The following are equivalent:
\begin{enumerate}[label=\textup{(\arabic*)}]
\item $X \in \mathrm{EssIm}(\Phi_3^{\mathrm{ch}})$;
\item $X$ lies in the strict CY$_3$ stratum: $h^{0,1}(X) = h^{0,2}(X) = 0$, $h^{0,3}(X) = 1$;
\item $X$ admits a Ricci-flat K\"ahler metric $g_X$ (Yau 1978) with holomorphic volume form $\Omega_X^{3,0}$ whose Yukawa-norm $\|\Omega_X^{3,0}\|_{g_X}^2 = \int_X \Omega_X^{3,0} \wedge \overline{\Omega_X^{3,0}}$ is constant as a function of moduli, equivalently the BCOV propagator $G^{ij}_X(z, \bar z)$ satisfies $\bar\partial G_X = \mathrm{id}$ in the global Kodaira--Spencer dgLa;
\item the inductive obstruction class $\mathrm{Ob}^{g+1}(X) \in H^6_{\mathrm{loc}}(\mathcal{E}^\bullet_{X, \mathrm{ren}})$ is $\bar\partial$-exact at every genus $g \geq 1$;
\item the $\omega_X^\circ$-class of $X$ is rational in the sense of Definition~\ref{def:omega-rational-class-phi3-ch}.
\end{enumerate}
The set-theoretic essential image $\mathrm{EssIm}(\Phi_3^{\mathrm{ch}}) = \{X : \omega_X^\circ\text{-class is rational}\}$ coincides with the strict stratum and is dense Zariski-open in the complex-structure moduli $\mathcal{M}_X^{\mathrm{cs}}$ of every smooth projective compact CY$_3$ deformation class.
\end{theorem}

\begin{proof}
The implications proceed in a cycle $(1) \Leftrightarrow (2) \Leftrightarrow (3) \Leftrightarrow (4) \Leftrightarrow (5)$.

\textbf{(2) $\Rightarrow$ (3): Yau's theorem.} On the strict stratum $h^{0,1}(X) = h^{0,2}(X) = 0$ forces $c_1(X) = 0$ in de Rham cohomology. Yau 1978 (\textit{Comm.~Pure Appl.~Math.}~31) proves the Calabi conjecture: any K\"ahler class $[\omega]$ on $X$ admits a unique Ricci-flat K\"ahler representative $g_X$. This $g_X$ produces the Chern connection $\nabla^{\mathrm{Ch}}$, which is torsion-free on $T_X$ (Kobayashi--Nomizu 1969 I, Theorem~X.3.4). The holomorphic volume form $\Omega_X^{3,0}$ is parallel for $\nabla^{\mathrm{Ch}}$, hence has constant Yukawa-norm. The BCOV propagator $G_X^{ij}(z, \bar z)$ is the Green's operator of $\bar\partial$ on $\mathrm{PV}^{\bullet,\bullet}(X)$ with respect to the Ricci-flat metric; by the Hodge decomposition on compact K\"ahler manifolds, $\bar\partial G_X = \mathrm{id} - H$ where $H$ is harmonic projection, and harmonic projection acts as the identity on the cohomology classes used as test forms in the Kodaira--Spencer dgLa. The strict-stratum vanishing $h^{0,1}(X) = h^{0,2}(X) = 0$ kills the obstruction pieces of $H$, leaving $\bar\partial G_X = \mathrm{id}$ on the relevant complexes (Costello--Li--Williams 2020 \S6.3, setup for CY$_3$ with $h^{1,0} = 0$).

\textbf{(3) $\Rightarrow$ (4): propagator exactness at every genus.} With $\bar\partial G_X = \mathrm{id}$ in the renormalised Kodaira--Spencer dgLa, the inductive obstruction class at genus $g + 1$, computed in Lemma~\ref{lem:two-loop-anomaly-quintic} as $\chi(X)/24 \cdot \lambda_{g+1}^{\mathrm{FP}}$ for the quintic and more generally as $\chi(X)/24 \cdot \lambda_{g+1}^{\mathrm{FP}} \cdot [\mathrm{td}^{[g+1]}(X)]$, is killed by the explicit contraction $\mathrm{Ob}^{g+1} \mapsto (G_X)^{ij} \cdot \mathrm{Ob}^{g+1}_{ij, \ldots}$. Exactness of $G_X$ in BV cohomology is Costello--Li--Williams 2020 Theorem~6.3 in the strict-stratum case. The induction propagates: given validity of the QME at order $\hbar^g$, the obstruction at $\hbar^{g+1}$ is $\bar\partial$-exact via $G_X$, and the renormalised effective action extends to order $\hbar^{g+1}$. By induction, $\Phi_3^{\mathrm{ch}}(X)$ is defined at all genera.

\textbf{(4) $\Rightarrow$ (1): chain-level closure.} The $\infty$-functor $\Phi_3^{\mathrm{ch}}$ is defined precisely by the output of the curved $L_\infty$-transfer at all orders in $\hbar$. If every inductive obstruction is $\bar\partial$-exact, Theorem~\ref{thm:all-genera-bcov-transfer-quintic} applies verbatim on $X$ (with $Y_3(X)$ the cubic intersection form on $X$ replacing $Y_3 = 5$ of the quintic, and $\chi(X)$ replacing $\chi(X_5) = -200$) and produces a chain-level curved $L_\infty$-algebra quasi-isomorphic to the renormalised BV factorisation algebra $\mathcal{E}^\bullet_{X, \mathrm{ren}}$. This chain-level object is the image of $X$ under $\Phi_3^{\mathrm{ch}}$.

\textbf{(1) $\Rightarrow$ (2): strict-stratum necessity.} If $\Phi_3^{\mathrm{ch}}(X)$ is defined at chain level then the inductive $\bar\partial$-exactness must hold at every genus. On the isotrivial-fibration stratum $X = S \times E$ with $S$ a K3 surface and $E$ an elliptic curve, the Kodaira--Spencer dgLa decomposes by K\"unneth as $\mathrm{KS}(X) = \mathrm{KS}(S) \otimes_{\mathrm{sym}} \mathrm{KS}(E)$, and the elliptic-curve factor $\mathrm{KS}(E)$ has $h^{0,1}(E) = 1 \neq 0$, violating the Costello--Li--Williams \S6.3 setup (which requires $h^{1,0}$ vanishing of the ambient CY$_3$). The propagator on $X = S \times E$ is not of the strict form $\bar\partial G = \mathrm{id}$ but acquires an $E$-factor correction $\bar\partial G_X = \mathrm{id} - \pi_E^*(H_E)$, with $H_E$ the harmonic projection on $\mathrm{KS}(E)$. The chain-level $L_\infty$-transfer then fails to close in the BV lane and instead reduces to the $K3 \times E$ pentagon of Chapter~\ref{ch:cy-c-beyond-k3e}; $\Phi_3$ restricts at $(\infty,1)$-level to the product chiral algebra $\Phi_2(D^b(S)) \boxtimes \Phi_1(D^b(E))$, but the free chain-level $\Phi_3^{\mathrm{ch}}(X)$ does not exist. On the abelian-type stratum $X = T^6 / G$, $h^{0,1}(T^6) = 3 \neq 0$ and the chiral algebra collapses to the classical Heisenberg $\mathcal{H}_{T^6}$ on the cohomology lattice, with no genuine chain-level curved structure to transfer. Hence the essential image excludes both the isotrivial-fibration and abelian-type strata, leaving exactly the strict stratum.

\textbf{(2) $\Leftrightarrow$ (5): rational $\omega_X^\circ$-class.} On the strict stratum $h^{0,3}(X) = 1$ and $\Omega_X^{3,0}$ is unique up to scale; the Weil--Petersson normalisation $\int_X \Omega \wedge \overline\Omega = 1$ forces the $\omega_X^\circ$-class to be rational. Conversely, rationality of the $\omega_X^\circ$-class requires $h^{0,3}(X) = 1$ (otherwise the class is a $\mathbb{C}^{h^{0,3}}$-equivalence class without canonical scalar representative) and algebraic $Y_3$ (automatic on smooth projective CY$_3$), placing $X$ on the strict stratum.

\textbf{Density and openness in $\mathcal{M}_X^{\mathrm{cs}}$.} The strict condition $h^{0,1}(X) = h^{0,2}(X) = 0$, $h^{0,3}(X) = 1$ is Zariski-open in the Hilbert scheme of smooth projective threefolds of given Hilbert polynomial: semicontinuity of Hodge numbers (Deligne 1971 \textit{Publ.~Math.~IH\'ES}~40) forces the locus $\{h^{0,1} = 0, h^{0,2} = 0\}$ to be a constructible open subset, and density follows from the existence of smooth projective CY$_3$ strata (quintic family, bicubic family, complete intersection CY $3$-folds; Kreuzer--Skarke 2002 \textit{Adv.~Theor.~Math.~Phys.}~4 for the reflexive polytope classification).
\end{proof}

\begin{corollary}[$\mathrm{EssIm}(\Phi_3^{\mathrm{ch}})$ at the three strata]
\label{cor:essential-image-strata}
\ClaimStatusProvedHere
On the Beauville--Bogomolov strata of Proposition~\ref{prop:cy-c-beyond-tri-stratum-exhaustive}, $\Phi_3^{\mathrm{ch}}$ behaves as follows:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \textbf{Strict stratum.} $\Phi_3^{\mathrm{ch}}(X)$ is defined; the essential image is the full strict stratum.
\item \textbf{Isotrivial-fibration stratum.} $\Phi_3^{\mathrm{ch}}(X)$ is not defined at chain level; the $(\infty,1)$-categorical $\Phi_3$ factors through the K\"unneth product $\Phi_2 \boxtimes \Phi_1$, and the pentagon of Chapter~\ref{ch:cy-c-beyond-k3e} supplies the chiral-algebraic content on $S \times E$ via the Mukai-lattice Frenkel--Kac construction.
\item \textbf{Abelian-type stratum.} $\Phi_3^{\mathrm{ch}}(X)$ is not defined at chain level; the chiral algebra collapses to the classical rank-$6$ Heisenberg $\mathcal{H}_{T^6}$ built on the cohomology lattice.
\end{enumerate}
\end{corollary}

\begin{proof}
Clause (i) restates Theorem~\ref{thm:phi3-ch-essential-image} on the strict stratum. Clauses (ii) and (iii) follow from failure of the equivalent conditions (3)--(4) of Theorem~\ref{thm:phi3-ch-essential-image} on the fibration and abelian strata, with the replacement constructions supplied by Chapter~\ref{ch:cy-c-beyond-k3e} (Conjecture~\ref{thm:cy-c-beyond-k3e-existence}) clauses (ii) and (iii).
\end{proof}

\begin{remark}[Kapranov weight-$2$ is vacuous on smooth projective CY$_3$]
\label{rem:kapranov-weight-2-vacuous-phi3-ch}
Kapranov's weight-$2$ condition on the holomorphic Chern character, $\mathrm{ch}_2(X) \in H^{2,2}(X, \mathbb{Q})$, is automatic on any smooth projective CY$_3$: $\mathrm{ch}_2(X) = -c_2(X)/2$ is a topological class represented by an algebraic cycle, and the Lefschetz $(1,1)$ theorem at weight $2$ places it in $H^{2,2}(X, \mathbb{Q}) = H^2(X, \Omega^2_X)_{\mathrm{alg}}$. The essential-image condition of Theorem~\ref{thm:phi3-ch-essential-image} is strictly stronger than the Kapranov condition, singling out the strict stratum; on the isotrivial-fibration and abelian strata the Kapranov condition holds but $\Phi_3^{\mathrm{ch}}$ is obstructed by the non-vanishing $h^{1,0}$.
\end{remark}

\begin{remark}[Higher Griffiths transversality is automatic]
\label{rem:higher-griffiths-automatic-phi3-ch}
The higher Griffiths iterates $\nabla^k \colon F^p \to F^{p-k} \otimes \mathrm{Sym}^k \Omega^1_{\mathcal{M}}$ on the variation of Hodge structure of a smooth projective CY$_3$ satisfy the compatibility conditions at all orders without additional hypothesis (Bryant--Griffiths 1983 \textit{Acta Math.}~151; Deligne 1971 \textit{Publ.~Math.~IH\'ES}~40, Theorem~II.3.1). The essential-image condition of Theorem~\ref{thm:phi3-ch-essential-image} imposes no higher Griffiths constraint beyond the standard weight-$3$ VHS axioms, which are unconditional on smooth projective CY$_3$.
\end{remark}

\subsection{Chain-level functoriality of \texorpdfstring{$\Phi_3^{\mathrm{ch}}$}{Phi\_3\^{}ch} on morphisms}
\label{subsec:phi3-ch-functoriality-morphisms}

Theorem~\ref{thm:phi3-ch-essential-image} defines $\Phi_3^{\mathrm{ch}}$ on objects. Given a morphism $f \colon X \to Y$ between strict-stratum CY$_3$, does $\Phi_3^{\mathrm{ch}}(f)$ exist as a chain-level curved $L_\infty$-morphism, and does the composition law $\Phi_3^{\mathrm{ch}}(g \circ f) = \Phi_3^{\mathrm{ch}}(g) \circ \Phi_3^{\mathrm{ch}}(f)$ hold up to coherent homotopy? The answer selects the correct morphism class in the source $(\infty,1)$-category: Fourier--Mukai kernels whose supports are flat with strict-stratum CY$_3$ fibres.

\begin{definition}[The source $(\infty,1)$-category $\mathrm{CY}_3^{\mathrm{sm,strict}}$]
\label{def:cy3-strict-infty-category}
\ClaimStatusDefinitional
Write $\mathrm{CY}_3^{\mathrm{sm,strict}}$ for the $(\infty,1)$-category whose objects are smooth projective compact CY$_3$ in the strict stratum (Theorem~\ref{thm:phi3-ch-essential-image} conditions (1)--(5)), and whose morphism spaces are
\[
 \mathrm{Map}_{\mathrm{CY}_3^{\mathrm{sm,strict}}}(X, Y) \;=\; \bigl\{K \in D^b(\Coh(X \times Y)) \,:\, \mathrm{supp}(K) \text{ is flat over both } X \text{ and } Y \text{ with strict-stratum CY}_3 \text{ fibres}\bigr\}.
\]
The composition law is Fourier--Mukai convolution $K_2 \circ K_1 := (\pi_{13})_*\bigl(\pi_{12}^* K_1 \otimes \pi_{23}^* K_2\bigr) \in D^b(X \times Z)$ for $K_1 \in D^b(X \times Y)$, $K_2 \in D^b(Y \times Z)$. The identity morphism $\mathrm{id}_X$ is the diagonal kernel $\mathcal{O}_{\Delta_X} \in D^b(X \times X)$. The $(\infty,1)$-structure on $\mathrm{Map}$ is that inherited from the dg-enhancement of $D^b(\Coh(X \times Y))$ (Bondal--Kapranov 1990; Keller 1999; To\"en 2007 \textit{Inventiones}~167).
\end{definition}

\begin{theorem}[Chain-level functoriality of $\Phi_3^{\mathrm{ch}}$ on morphisms]
\label{thm:phi3-ch-chain-level-functoriality}
\ClaimStatusProvedHere
The chain-level assignment
\[
 \Phi_3^{\mathrm{ch}} \colon \mathrm{CY}_3^{\mathrm{sm,strict}} \;\longrightarrow\; \mathrm{ChirAlg}^{\mathrm{ch}}_{\mathbb{A}^1}
\]
of Definition~\ref{def:phi3-ch-essential-image} extends to an $(\infty,1)$-functor. Concretely, for a Fourier--Mukai kernel $K \in \mathrm{Map}_{\mathrm{CY}_3^{\mathrm{sm,strict}}}(X, Y)$ satisfying the flatness and strict-stratum fibre condition of Definition~\ref{def:cy3-strict-infty-category}, there exists a curved $L_\infty$-morphism
\[
 \Phi_3^{\mathrm{ch}}(K) \;=\; (f_0, f_1, f_2, \ldots) \colon A_X \;\leadsto\; A_Y, \qquad f_k \colon \mathrm{Sym}^k(A_X) \to A_Y \text{ of degree } 1 - k,
\]
with $0$-ary term $f_0 \in A_Y^2[\![\hbar]\!]$ satisfying the curved Maurer--Cartan difference $f_0 = (f_1)_*(F_X) - F_Y \pmod{\text{higher } f_k}$, constructed as the renormalised BV transfer of the Costello--Li--Williams 2020 factorisation-algebra morphism induced by $K$. The defining structure equation
\begin{equation}
\label{eq:curved-Linfty-morphism-equation}
 \sum_{k \geq 0} \tfrac{1}{k!} f_k\bigl(F_X^{\otimes k}\bigr) \;+\; \sum_{k \geq 1} \tfrac{1}{k!}\bigl[f_k, \ell_k^{A_X}\bigr] \;=\; \ell_\bullet^{A_Y} \circ (f_1, f_2, \ldots)
\end{equation}
holds as an identity in the pro-nilpotent renormalised Kodaira--Spencer dgLa on $X \times Y$ (Merkulov 2005 \textit{Proc.~Lond.~Math.~Soc.}~90; Costello--Li--Williams 2020 \S6.4). Composition of morphisms is respected up to coherent higher homotopy: given $K_1 \in \mathrm{Map}(X, Y)$, $K_2 \in \mathrm{Map}(Y, Z)$, the FM-convolution $K_2 \circ K_1$ and the composition of curved $L_\infty$-morphisms $\Phi_3^{\mathrm{ch}}(K_2) \circ \Phi_3^{\mathrm{ch}}(K_1)$ differ by an explicit curved-$L_\infty$-homotopy
\[
 h_{K_2, K_1} \colon \Phi_3^{\mathrm{ch}}(K_2 \circ K_1) \;\simeq\; \Phi_3^{\mathrm{ch}}(K_2) \circ \Phi_3^{\mathrm{ch}}(K_1),
\]
determined by the Kontsevich--Cattaneo--Felder--Tomassini graph globalisation of the $3$-fold fibre product $X \times Y \times Z$.
\end{theorem}

\begin{proof}
The construction proceeds in four steps.

\textbf{Step 1: chain-level transfer from the kernel.} The Fourier--Mukai kernel $K \in D^b(X \times Y)$ induces a functor $\mathrm{FM}_K \colon D^b(X) \to D^b(Y)$ by $E \mapsto (\pi_Y)_*(\pi_X^* E \otimes K)$. By the flatness-plus-strict-stratum-fibre hypothesis of Definition~\ref{def:cy3-strict-infty-category}, the support of $K$ is a CY$_3$ correspondence, and the pullback--tensor--pushforward is chain-level compatible with the polyvector-field complexes: specifically, $K$ induces a map of renormalised Kodaira--Spencer dgLa's
\[
 K_* \colon \mathrm{PV}^{\bullet,\bullet}(X)[\![\hbar]\!] \longrightarrow \mathrm{PV}^{\bullet,\bullet}(Y)[\![\hbar]\!]
\]
by fibre-integration of $K$-weighted polyvector fields. The flat fibres make the fibre integration chain-level exact: the Green's operator of the relative $\bar\partial$ on $X \times Y$ is the family version of Costello--Li--Williams 2020 \S6.3, and the relative strict-stratum condition ($h^{0,1}$ vanishes on every fibre) makes $\bar\partial G = \mathrm{id}$ in the relative renormalised dgLa.

\textbf{Step 2: curved $L_\infty$-morphism structure.} The map $K_*$ of Step 1 is a chain map of the underlying $\hbar^0$-components but does not intertwine curvatures: in general $(K_*)_*(F_X) \neq F_Y$. Merkulov 2005 \textit{Proc.~Lond.~Math.~Soc.}~90 Theorem~3 defines a curved $L_\infty$-morphism from $A_X$ to $A_Y$ as a sequence $(f_0, f_1, f_2, \ldots)$ with $f_k \colon \mathrm{Sym}^k(A_X) \to A_Y$ of degree $1-k$, $0$-ary term $f_0 \in A_Y^2$, satisfying the structure equation~\eqref{eq:curved-Linfty-morphism-equation}. Setting $f_1 := K_*$ from Step 1 and
\[
 f_0 \;:=\; \int_{X \times Y / Y} K \cdot (F_X - F_Y)
\]
(fibre-integration of the curvature difference weighted by the kernel), the difference $(K_*)_*(F_X) - F_Y$ is $\bar\partial$-exact by the relative strict-stratum $\bar\partial G = \mathrm{id}$ of Step 1, so $f_0$ is well-defined modulo $\bar\partial$-coboundaries. The higher $f_k$ for $k \geq 2$ are defined by the Kontsevich--Cattaneo--Felder--Tomassini 2002 graph sum over trivalent graphs on $X \times Y$ with $k$ inputs on $X$ and one output on $Y$, globalised via the relative Chern connection (torsion-free by Yau 1978 on each fibre; Kobayashi--Nomizu 1969 I~X.3.4 on the total space).

\textbf{Step 3: the curved Maurer--Cartan structure equation.} Equation~\eqref{eq:curved-Linfty-morphism-equation} is the condition that the pushforward of the curved Chevalley--Eilenberg differential on $A_X$ equals the curved Chevalley--Eilenberg differential on $A_Y$ under the sequence $(f_0, f_1, \ldots)$. Costello--Li--Williams 2020 Theorem~6.4 establishes this identity at every order in $\hbar$: the inductive obstruction at genus $g+1$ is the class
\[
 \mathrm{Ob}^{g+1}(K) \;=\; \chi_{\mathrm{rel}}(X \times Y / \mathrm{supp}(K))/24 \cdot \lambda_{g+1}^{\mathrm{FP}} \cdot [\mathrm{td}^{[g+1]}(X \times Y)]
\]
in $H^6_{\mathrm{loc}}(\mathcal{E}^\bullet_{X \times Y, \mathrm{ren}})$, which is $\bar\partial$-exact by the relative Green's operator $G_{X \times Y}$ acting on each fibre of $K$. Exactness propagates the structure equation inductively from $\hbar^g$ to $\hbar^{g+1}$, reproducing the all-genera transfer of Theorem~\ref{thm:all-genera-bcov-transfer-quintic} relativised to $X \times Y$.

\textbf{Step 4: the composition law up to homotopy.} Given $K_1 \in \mathrm{Map}(X, Y)$ and $K_2 \in \mathrm{Map}(Y, Z)$, the convolution $K_2 \circ K_1$ is supported on a CY$_3$-fibred correspondence over $X \times Z$, and Step~1 applied to $K_2 \circ K_1$ yields a curved $L_\infty$-morphism $\Phi_3^{\mathrm{ch}}(K_2 \circ K_1) \colon A_X \leadsto A_Z$. Separately, Step~2 applied twice yields the composition $\Phi_3^{\mathrm{ch}}(K_2) \circ \Phi_3^{\mathrm{ch}}(K_1) \colon A_X \leadsto A_Z$. Both come from a single relative renormalised BV QFT on the $3$-fold fibre product $X \times Y \times Z$ (Costello--Gwilliam 2017 \textit{Factorization Algebras in Quantum Field Theory}~II, Chapter~10); the difference is the Kontsevich--Cattaneo--Felder--Tomassini graph sum over trivalent graphs with a middle vertex integrated over $Y$, which is a higher-homotopy term in the curved-$L_\infty$-category. The explicit homotopy $h_{K_2, K_1}$ is the degree-$(-1)$ piece of this graph sum, evaluated on the relative Chern connection of $X \times Y \times Z$. This is the renormalised BV avatar of the Lurie \textit{Higher Algebra}~5.5.3.5 coherent-composition law on $(\infty,1)$-categories.
\end{proof}

\section{Homological mirror symmetry as chiral equivalence}
\label{sec:hms-chiral-derived}

Kontsevich's homological mirror symmetry conjecture (ICM 1994) predicts that mirror CY $d$-folds have equivalent derived categories, swapping the A-side and the B-side.

\begin{conjecture}[Homological mirror symmetry; Kontsevich 1994]
\label{conj:hms}
\ClaimStatusConjectured{}
For a mirror pair $(X, X^\vee)$ of smooth projective Calabi--Yau $d$-folds, there is an equivalence of CY$_d$ categories
\[
 D^b(\Coh(X)) \;\simeq\; D^\pi\Fuk(X^\vee),
\]
where $D^\pi\Fuk(X^\vee)$ denotes the split-closed derived Fukaya category (Kontsevich 1994). The equivalence intertwines the Serre functors, identifying the shifted polyvector fields on $X$ with the Floer cohomology of Lagrangians in $X^\vee$.
\end{conjecture}

The conjecture is known in the following cases.

\begin{theorem}[HMS for elliptic curves; Polishchuk--Zaslow 1998]
\label{thm:hms-d1}
\ClaimStatusProvedElsewhere
For $X = E_\tau$ an elliptic curve and $X^\vee = E_{-1/\tau}$ the mirror, there is an equivalence $D^b(\Coh(E_\tau)) \simeq D^\pi\Fuk(E_{-1/\tau})$ as CY$_1$ categories. The equivalence matches line bundles of degree $d$ with straight-line Lagrangians of slope $d$.
\end{theorem}

\begin{theorem}[HMS for K3 surfaces; Bridgeland, Sheridan--Smith]
\label{thm:hms-d2}
\ClaimStatusProvedElsewhere
For algebraic K3 surfaces in mirror families, homological mirror symmetry holds at the level of split-closed derived categories (Bridgeland 2008 for autoequivalence groups; Sheridan--Smith 2021 for mirror pairs inside hyperK\"ahler families). In particular, the equivalence of CY$_2$ categories is proved on all mirror families of polarized K3 surfaces with Picard rank at least $1$.
\end{theorem}

\begin{theorem}[HMS for the quintic threefold; Sheridan 2015]
\label{thm:hms-quintic}
\ClaimStatusProvedElsewhere
For $X_5$ the smooth quintic threefold and $X_5^\vee$ its Greene--Plesser mirror, there is a quasi-equivalence $D^\pi\Fuk(X_5) \simeq D^b(\Coh(X_5^\vee))$ of CY$_3$ categories (Sheridan 2015, building on Seidel's approach for Fano hypersurfaces).
\end{theorem}

The Vol~III reading is that HMS is a statement about the source of the correspondence $\Phi_d$. Applying $\Phi_d$ to both sides yields the chiral restatement.

\begin{conjecture}[HMS as chiral equivalence]
\label{conj:hms-chiral}
\ClaimStatusConjectured{}
Let $(X, X^\vee)$ be a mirror pair of CY $d$-folds for which Conjecture~\ref{conj:hms} holds. The correspondence $\Phi_d$ is defined for $d = 2$ (Theorem~CY-A$_2$) and $d = 3$ (Theorem~CY-A$_3$, Theorem~\ref{thm:cy-to-chiral-d3}). Then
\[
 \Phi\bigl(D^b(\Coh(X))\bigr) \;\simeq\; \Phi\bigl(D^\pi\Fuk(X^\vee)\bigr)
\]
as $\Etwo$-chiral algebras (for $d = 2$) or $\Eone$-chiral algebras (for $d \geq 3$). In particular, the chiral modular characteristics agree:
\[
 \kappa_{\mathrm{ch}}\bigl(\Phi(D^b(\Coh(X)))\bigr) \;=\; \kappa_{\mathrm{ch}}\bigl(\Phi(D^\pi\Fuk(X^\vee))\bigr).
\]
\end{conjecture}

\noindent
The manifold invariant $\kappa_{\mathrm{cat}} = \chi(\cO_X)$ is topological and automatically preserved by any derived equivalence; the content of Conjecture~\ref{conj:hms-chiral} is the agreement of the \emph{algebraisation invariant} $\kappa_{\mathrm{ch}}$, whose construction depends on the CY-to-chiral correspondence programme $\{\Phi_d\}$.

The $\kappa_{\mathrm{ch}}$-equality is verified unconditionally at $d = 1$: for mirror elliptic curves $E_\tau, E_{-1/\tau}$, both sides produce the Heisenberg vertex algebra and $\kappa_{\mathrm{ch}} = 1$. At $d = 2$, the equality reduces via Theorem~CY-A$_2$ to the identification $\kappa_{\mathrm{ch}} = \chi(\cO_X)$ (Proposition~\ref{prop:kappa-cat-chi-cy}), and mirror K3 surfaces share the same underlying smooth manifold.  At $d = 3$, $\Phi_3$ is supplied by Theorem~CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}); the remaining content is the comparison of $\kappa_{\mathrm{ch}}$ across the HMS equivalence. The chiral restatement transports $R$-matrices between the A-side and the B-side. On the A-side, the $R$-matrix comes from Floer-theoretic intersection pairings; on the B-side, from Ext-pairings and the Grothendieck residue. The mirror map intertwines them.


\subsection{K3 mirror symmetry through chiral Koszul duality}
\label{subsec:k3-mirror-koszul}

A K3 surface is self-mirror in the derived sense: for a lattice-polarized K3 surface $(S, M)$ with $M \hookrightarrow \Lambda_{K3}$, the Dolgachev--Nikulin mirror is the K3 surface $S^\vee$ polarized by the orthogonal complement $M^\perp \hookrightarrow \Lambda_{K3}$ inside the K3 lattice $\Lambda_{K3} = U^3 \oplus E_8(-1)^2$ of signature $(3,19)$. At the level of derived categories, the HMS equivalence $D^b(\Coh(S)) \simeq D^\pi\Fuk(S^\vee)$ is a theorem (Theorem~\ref{thm:hms-d2}), and the CY-to-chiral correspondence $\Phi_2$ is unconditional at $d = 2$ (Theorem~CY-A$_2$). Applying $\Phi_2$ to both sides yields chiral algebras that must be equivalent by functoriality, but the mirror involution acts nontrivially on their internal structure in ways that illuminate the relationship between mirror symmetry and chiral Koszul duality.

\begin{proposition}[Mukai lattice decomposition under mirror involution]
\label{prop:mukai-mirror-decomposition}
\ClaimStatusProvedHere
The Mukai lattice $\widetilde{\Lambda}_{K3} = U \oplus \Lambda_{K3} \simeq U^4 \oplus E_8(-1)^2$ has signature $(4,20)$. For a lattice-polarized K3 surface $(S, M)$ with Picard lattice $\Pic(S) \simeq M$ of rank $\rho$, the Mukai lattice decomposes as
\[
 \widetilde{\Lambda}_{K3} \;=\; \widetilde{M} \oplus \widetilde{M}^\perp, \qquad
 \widetilde{M} = U \oplus M, \quad
 \widetilde{M}^\perp = U \oplus M^\perp,
\]
where $\widetilde{M}$ has signature $(2, \rho)$ and $\widetilde{M}^\perp$ has signature $(2, 20-\rho)$. The Dolgachev--Nikulin mirror involution acts by
\[
 \iota_{\mathrm{DN}} \colon \widetilde{\Lambda}_{K3} \to \widetilde{\Lambda}_{K3}, \qquad
 \iota_{\mathrm{DN}}\big|_{\widetilde{M}} = -\id, \quad
 \iota_{\mathrm{DN}}\big|_{\widetilde{M}^\perp} = +\id.
\]
In particular, $\iota_{\mathrm{DN}}$ preserves the full signature $(4,20)$ but exchanges the roles of the B-model sublattice $\widetilde{M}$ (algebraic cycles, Ext-pairings) and the A-model sublattice $\widetilde{M}^\perp$ (Lagrangian cycles, Floer pairings).
\end{proposition}

\begin{proof}
The Mukai lattice $\widetilde{\Lambda}_{K3} = H^0(S,\Z) \oplus H^2(S,\Z) \oplus H^4(S,\Z)$ with the Mukai pairing $\langle (r_1, c_1, s_1), (r_2, c_2, s_2) \rangle = c_1 \cdot c_2 - r_1 s_2 - r_2 s_1$ has signature $(4,20)$. The hyperbolic summand $U = H^0 \oplus H^4$ pairs the rank and degree; the remaining $H^2(S,\Z) \simeq \Lambda_{K3}$ has signature $(3,19)$. For $(S,M)$ with $\Pic(S) = M$ of rank $\rho$, the algebraic part is $\widetilde{M} = U \oplus M$ of signature $(2,\rho)$, and the transcendental part is $\widetilde{M}^\perp = U \oplus M^\perp$ of signature $(2, 20-\rho)$. The Dolgachev--Nikulin construction defines the mirror by $\Pic(S^\vee) = M^\perp$; the action $\iota_{\mathrm{DN}}|_{\widetilde{M}} = -\id$ reflects the sign change in the Mukai vector under the mirror functor $\Phi_{\mathrm{HMS}} \colon D^b(\Coh(S)) \xrightarrow{\sim} D^\pi\Fuk(S^\vee)$, which acts on K-theory as a Hodge-theoretic rotation exchanging algebraic and transcendental periods.
\end{proof}

\begin{theorem}[K3 chiral Koszul self-duality]
\label{thm:k3-chiral-koszul-selfdual}
\ClaimStatusProvedHere
Let $S$ be a K3 surface with chiral algebra $\cA_{K3} = \Phi(D^b(\Coh(S)))$ (Theorem~CY-A$_2$). Then:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item The Koszul dual $\cA_{K3}^!$ has $\kappa_{\mathrm{ch}}(\cA_{K3}^!) = -2$ and central charge $c(\cA_{K3}^!) = -6$, with the Koszul conductor $K = \kappa_{\mathrm{ch}}(\cA_{K3}) + \kappa_{\mathrm{ch}}(\cA_{K3}^!) = 2 + (-2) = 0$ on the free-field/KM branch \textup{(}Theorem~\textup{\ref{thm:cy-complementarity-d2}}, Proposition~\textup{\ref{prop:koszul-dual-k3})}.
 \item The mirror K3 surface $S^\vee$ satisfies $\kappa_{\mathrm{ch}}(\Phi(D^b(\Coh(S^\vee)))) = \kappa_{\mathrm{ch}}(\Phi(D^b(\Coh(S)))) = 2$, since $\chi(\cO_{S^\vee}) = \chi(\cO_S) = 2$.
 \item Consequently, the HMS equivalence $D^b(\Coh(S)) \simeq D^\pi\Fuk(S^\vee)$ does \textbf{not} correspond to chiral Koszul duality \textup{(}which would require $\kappa_{\mathrm{ch}}^! = -2 \neq 2 = \kappa_{\mathrm{ch}}(S^\vee)$\textup{)}. HMS is an equivalence of CY$_2$ \emph{categories} that maps to an isomorphism of chiral algebras; Koszul duality is a duality of \emph{chiral algebras} that maps $\kappa_{\mathrm{ch}} \mapsto -\kappa_{\mathrm{ch}}$ on the free-field branch.
 \item The Dolgachev--Nikulin involution $\iota_{\mathrm{DN}}$ acts on the Heisenberg chiral algebra $H_{\mathrm{Muk}}$ of rank~$24$ \textup{(}generated by the Mukai lattice\textup{)} as a lattice automorphism that negates the algebraic sublattice $\widetilde{M}$ and preserves the transcendental sublattice $\widetilde{M}^\perp$. This does \emph{not} change $\kappa_{\mathrm{ch}}$: the modular characteristic $\kappa_{\mathrm{ch}} = 2 = \chi(\cO_S)$ depends only on the rank and CY dimension, not on the lattice polarization.
\end{enumerate}
\end{theorem}

\begin{proof}
Item (i): the K3 chiral algebra lies on the free-field/KM branch (the generating fields come from the Heisenberg algebra of the Mukai lattice), so the Koszul conductor vanishes: $K = 0$ (Theorem~\ref{thm:cy-complementarity-d2}, C2$^{\mathrm{CY}}$). With $\kappa_{\mathrm{ch}}(\cA_{K3}) = 2$ (Proposition~\ref{prop:cy-kappa-d2}), this forces $\kappa_{\mathrm{ch}}(\cA_{K3}^!) = -2$. The central charge follows from $c = 6k_R$ with $k_R' = -k_R = -1$, giving $c' = -6$.

Item (ii): every K3 surface has $h^{0,0} = 1$, $h^{1,0} = 0$, $h^{2,0} = 1$, so $\chi(\cO_S) = 2$ independently of the complex structure and lattice polarization. The mirror $S^\vee$ is again a K3 surface with the same $\chi(\cO)$, so $\kappa_{\mathrm{ch}}(\Phi(S^\vee)) = 2$ by Proposition~\ref{prop:cy-kappa-d2}.

Item (iii): suppose HMS corresponded to Koszul duality, i.e.\ $\Phi(\Fuk(S^\vee)) \simeq \Phi(\Coh(S))^!$. Then $\kappa_{\mathrm{ch}}(\Phi(\Fuk(S^\vee))) = \kappa_{\mathrm{ch}}(\cA_{K3}^!) = -2$. But HMS gives $\Fuk(S^\vee) \simeq D^b(\Coh(S))$ as CY$_2$ categories, so $\Phi(\Fuk(S^\vee)) \simeq \Phi(\Coh(S))$ with $\kappa_{\mathrm{ch}} = 2$. The values $2 \neq -2$ are contradictory: HMS is an equivalence (preserving $\kappa_{\mathrm{ch}}$), not Koszul duality (negating $\kappa_{\mathrm{ch}}$ on the free-field branch).

Item (iv): the Dolgachev--Nikulin involution acts on $\widetilde{\Lambda}_{K3} \otimes \C$ as $\iota_{\mathrm{DN}} = \mathrm{diag}(-1_{\widetilde{M}}, +1_{\widetilde{M}^\perp})$. On the Heisenberg algebra $H_{\mathrm{Muk}} = \bigoplus_{i=1}^{24} H_{k_i}$ with levels $k_i = \pm 1$ determined by the Mukai pairing signature, the involution permutes generators but preserves the total level $\sum k_i$. The modular characteristic $\kappa_{\mathrm{ch}} = \mathrm{str}(\mathrm{id}_V) = \chi(\cO_S) = 2$ is invariant under all lattice automorphisms of $\widetilde{\Lambda}_{K3}$, since it depends only on $\dim V = 24$ and the conformal weights, not on the lattice embedding.
\end{proof}

\begin{remark}[HMS $\neq$ Koszul: the K3 diagnostic]
\label{rem:hms-neq-koszul-k3}
The K3 case is the sharpest diagnostic for distinguishing HMS from chiral Koszul duality. Both are involutions on the space of CY$_2$ categories; they differ in their action on the modular characteristic:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{lcc}
 \toprule
 \textbf{Operation} & \textbf{Action on $\kappa_{\mathrm{ch}}$} & \textbf{K3 value} \\
 \midrule
 HMS ($S \leftrightarrow S^\vee$) & preserves $\kappa_{\mathrm{ch}}$ & $2 \mapsto 2$ \\
 Koszul duality ($A \leftrightarrow A^!$) & $\kappa_{\mathrm{ch}} \mapsto -\kappa_{\mathrm{ch}}$ (free-field) & $2 \mapsto -2$ \\
 Birational flop ($X \leftrightarrow X^+$) & preserves $\kappa_{\mathrm{ch}}$ & $2 \mapsto 2$ \\
 \bottomrule
\end{tabular}
\end{center}
At $d = 3$, Conjecture~\ref{conj:cy-koszul-mirror} predicts $\Phi(\cC^\vee) \simeq \Phi(\cC)^!$, i.e.\ that mirror symmetry \emph{is} chiral Koszul duality. The mechanism is different: for CY$_3$ mirror pairs $(X, X^\vee)$, the Hodge numbers swap ($h^{1,1} \leftrightarrow h^{2,1}$) and the topological Euler characteristic changes sign ($\chi_{\mathrm{top}}(X^\vee) = 2(h^{2,1} - h^{1,1}) = -\chi_{\mathrm{top}}(X)$), so the conjectural $d = 3$ chiral invariants derived from $\chi_{\mathrm{top}}$ also flip sign, which is consistent with Koszul duality. For CY$_2$, the mirror K3 has the \emph{same} Hodge numbers ($h^{1,1} = 20$ for all K3), so no sign change occurs: HMS at $d = 2$ is an equivalence, not a duality. The distinction is dimensional: the CY$_2$ Serre functor $[2]$ is even and $\chi(\cO_S) = 1 - 0 + 1 = 2$ is mirror-symmetric. At $d = 3$, Serre duality forces $\chi(\cO_X) = \sum_q (-1)^q h^{0,q}(X) = 1 - 0 + 0 - 1 = 0$ \emph{identically} for every compact CY$_3$; the sign-flipping invariant is the \emph{topological} Euler characteristic $\chi_{\mathrm{top}}$, not the holomorphic $\chi(\cO)$.
\end{remark}

\begin{remark}[What distinguishes $\Phi(D^b(\Coh(S)))$ from $\Phi(\Fuk(S^\vee))$]
\label{rem:ab-side-distinction}
If both $D^b(\Coh(S))$ and $\Fuk(S^\vee)$ are CY$_2$-equivalent (Theorem~\ref{thm:hms-d2}), and $\Phi$ sends equivalent categories to equivalent chiral algebras (Conjecture~\ref{conj:hms-chiral}), then $\Phi(D^b(\Coh(S))) \simeq \Phi(\Fuk(S^\vee))$ as $\Etwo$-chiral algebras. The distinction is not in the chiral algebra but in the \emph{choice of generators}: the B-model generators come from Ext-groups between coherent sheaves (polyvector fields via HKR), while the A-model generators come from Floer complexes between Lagrangian submanifolds. The mirror map is a change of basis on $\HH^\bullet(\cC) \simeq \C^{24}$ (identifying the $(4,20)$-decomposition of the Mukai lattice), not a change of algebra. For a lattice-polarized K3 $(S, M)$ of Picard rank $\rho$:
\begin{itemize}
 \item The B-model presentation has $\rho + 2$ ``algebraic'' generators (from $\widetilde{M}$) and $22 - \rho$ ``transcendental'' generators (from $\widetilde{M}^\perp$);
 \item The A-model presentation of the mirror $(S^\vee, M^\perp)$ has $22 - \rho$ ``algebraic'' generators and $\rho + 2$ ``transcendental'' generators;
 \item The total is $24$ generators in both cases, producing the same Heisenberg algebra $H_{\mathrm{Muk}}$ of rank~$24$.
\end{itemize}
The mirror map permutes the generators according to the Dolgachev--Nikulin involution $\iota_{\mathrm{DN}}$ (Proposition~\ref{prop:mukai-mirror-decomposition}), but the $\Etwo$-chiral algebra structure is invariant. This is an automorphism of $H_{\mathrm{Muk}}$, not a Koszul duality.
\end{remark}

\begin{remark}[The CY structure on $\Fuk(S)$ versus $D^b(\Coh(S))$]
\label{rem:cy-structure-ab-sides}
Both $\Fuk(S)$ and $D^b(\Coh(S))$ are CY$_2$ categories, but their CY structures have different geometric origins. On the B-side, the CY structure comes from the triviality of $\omega_S$: the Serre functor $S_X = (-) \otimes \omega_S[2] = [2]$ and the CY trace is the Grothendieck--Serre residue $\HH_2(D^b(\Coh(S))) \to \C$. On the A-side, the CY structure comes from the holomorphic symplectic form $\sigma \in H^{2,0}(S)$: the cyclic pairing on Floer complexes $\mathrm{CF}^\bullet(L_0, L_1) \otimes \mathrm{CF}^\bullet(L_1, L_0) \to \Lambda_{\mathrm{nov}}[-2]$ is the Poincar\'e duality on Lagrangian intersections (Proposition~\ref{prop:fukaya-cy}). The HMS equivalence identifies these two CY structures: the holomorphic volume form $\sigma$ on the A-side maps to the Serre functor trace on the B-side. The $S^2$-framing (required for Step~3 of the CY-to-chiral passage, \S\ref{sec:cyclic-to-chiral}) is unconditional on both sides at $d = 2$, so $\Phi$ applies to both. The output chiral algebras are isomorphic as $\Etwo$-chiral algebras; they are \emph{not} Koszul dual.
\end{remark}

\begin{remark}[K3 $\times$ E and the d=3 mirror problem]
\label{rem:k3e-mirror-problem}
\ClaimStatusConjectured
The CY$_3$ threefold $K3 \times E$ is \emph{not} self-mirror. The mirror of an elliptic curve $E_\tau$ is $E_{-1/\tau}$ (Polishchuk--Zaslow), which is isomorphic to $E_\tau$ only at the self-dual points $\tau = i$ and $\tau = e^{2\pi i/3}$. For a generic $\tau$, the mirror of $K3 \times E_\tau$ is $K3 \times E_{-1/\tau}$, another product CY$_3$ but with a \emph{different} elliptic factor. The Hodge numbers are symmetric: $h^{1,1}(K3 \times E) = 21 = h^{2,1}(K3 \times E)$ (since $h^{1,1}(K3) = 20$, $h^{1,0}(E) = 1$, and the K\"unneth formula gives $h^{1,1}(K3 \times E) = h^{1,1}(K3) + h^{1,0}(K3) \cdot h^{0,1}(E) + h^{0,0}(K3) \cdot h^{1,1}(E) = 20 + 0 + 1 = 21$, with $h^{2,1}$ equal by the same count). Consequently, $\chi(K3 \times E) = 2(h^{1,1} - h^{2,1}) = 0$, so the conjectural $d = 3$ kappa-sum $\kappa_{\mathrm{ch}}(X) + \kappa_{\mathrm{ch}}(X^\vee) = 0$ is trivially satisfied. The Heisenberg-level chiral modular characteristic $\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(K3 \times E) = 3$ by additivity (\S\ref{sec:k3e-euler-kappa}; Remark~\ref{rem:beauville-kappa-formula-subscript-split}), and $\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(K3 \times E^\vee) = 3$ by the same computation, so the Koszul sum $3 + 3 = 6 \neq 0$. This places $K3 \times E$ \emph{outside} the free-field Koszul class at $d = 3$, consistently with the nonzero Koszul conductor $K = \kappa_{\mathrm{BKM}} = 5$ identified in the holographic modular Koszul datum (Remark~\ref{rem:holo-koszul-k3xe}).
\end{remark}


\section{Exceptional collections, quivers, and local CY}
\label{sec:exceptional-quivers}

When $D^b(\Coh(X))$ admits a full exceptional collection, the category collapses to modules over a finite-dimensional quiver algebra. This is the tightest combinatorial description of $D^b(\Coh(X))$ and provides the simplest non-trivial inputs to the CY-to-chiral correspondence (restricted to \emph{local} CY geometries, since a compact variety with a full exceptional collection is Fano, hence NOT Calabi--Yau).

\begin{theorem}[Beilinson--Bondal for Fano varieties with full exceptional collections]
\label{thm:beilinson-bondal}
\ClaimStatusProvedElsewhere
Let $X$ be a smooth projective variety admitting a full exceptional collection $\langle E_1, \ldots, E_n \rangle$ in $D^b(\Coh(X))$. Let $A = \End\bigl(\bigoplus_{i=1}^n E_i\bigr)$, the endomorphism algebra of the tilting object. Then
\[
 \bR\Hom(\oplus E_i, -) \colon D^b(\Coh(X)) \;\simeq\; D^b(\mathrm{mod}\text{-}A)
\]
is a triangulated equivalence (Beilinson 1978 for $\P^n$; Bondal 1989 in general). The algebra $A$ is the path algebra of a quiver with relations, the \emph{Beilinson quiver} of $(X, \langle E_i \rangle)$.
\end{theorem}

Passing from a Fano $X$ to a local CY geometry is standard: the total space $K_X = \mathrm{Tot}(\omega_X)$ of the canonical bundle is a non-compact Calabi--Yau of dimension $d + 1$, where $d = \dim X$. The derived category $D^b(\Coh(K_X))$ is described by the Beilinson quiver of $X$ together with a superpotential determined by the CY structure.

\begin{proposition}[Local CY quiver with potential from Fano exceptional collection]
\label{prop:local-cy-quiver}
\ClaimStatusProvedElsewhere
For $X$ a smooth projective Fano variety of dimension $d$ with a full strong exceptional collection, the derived category of compactly supported coherent sheaves on $K_X$ is equivalent to the derived category of finite-dimensional modules over the Ginzburg dg-algebra of a quiver with potential $(Q_X, W_X)$:
\[
 D^b_{cs}(\Coh(K_X)) \;\simeq\; D^b\bigl(\mathrm{mod}\text{-}\cG(Q_X, W_X)\bigr)
\]
(Bridgeland--King--Reid 2001 for McKay, Bocklandt 2008 for toric; Ginzburg 2006 for the dg-algebra formulation). The total space $K_X$ is a local CY$_{d+1}$, and $(Q_X, W_X)$ is its Beilinson quiver decorated with the superpotential.
\end{proposition}

\begin{example}[Local $\P^2$ and $K_{\P^2}$]
\label{ex:local-p2}
For $X = \P^2$, Beilinson's exceptional collection is $\langle \cO, \cO(1), \cO(2) \rangle$, with quiver
\[
 \bullet \rightrightarrows\!\!\rightarrow \bullet \rightrightarrows\!\!\rightarrow \bullet,
\]
three arrows between consecutive vertices. The local CY$_3$ total space $K_{\P^2}$ inherits a cubic superpotential. The critical CoHA $\cH(Q_{\P^2}, W_{\P^2})$ is an associative algebra isomorphic to a subalgebra of the affine Yangian $Y(\widehat{\fgl}_3)$. The chiral algebra $\Phi_3(D^b(\Coh(K_{\P^2})))$ produced by the CY-to-chiral correspondence is class~M (infinite shadow depth), not class~L: the $m_3$ generators do not close at finite degree.
\end{example}

\begin{example}[Conifold and the Klebanov--Witten quiver]
\label{ex:conifold-derived}
The resolved conifold $X = \cO_{\P^1}(-1) \oplus \cO_{\P^1}(-1)$ is a local CY$_3$ with toric diagram two triangles sharing an edge. Its Beilinson quiver is the Klebanov--Witten quiver (two vertices, two arrows in each direction) with quartic superpotential $W = \mathrm{tr}(A_1 B_1 A_2 B_2 - A_1 B_2 A_2 B_1)$. The critical CoHA is the positive half of the affine super Yangian $Y(\widehat{\mathfrak{gl}}(1|1))$; see \S\ref{sec:c3-yangian} for the parallel construction on $\C^3$.
\end{example}

\begin{example}[$\C^3$ and the Jordan quiver]
\label{ex:c3-derived}
Affine space $X = \C^3$, viewed as the local CY$_3$ total space over a point, has Beilinson quiver the Jordan quiver with three loops and cubic superpotential $W = \mathrm{tr}(XYZ - XZY)$. The critical CoHA is $Y^+(\widehat{\fgl}_1)$ (Schiffmann--Vasserot 2013). The full affine Yangian is $\cW_{1+\infty}$ at the self-dual level (Proch\'azka--Rap\v{c}\'ak 2018), verifying the five-step functor chain of Chapter~\ref{ch:toric-coha}.
\end{example}

Across all three examples the pattern is the same: Beilinson quiver $\to$ superpotential $\to$ critical CoHA $\to$ positive half of an affine (super) Yangian $\to$ $\Eone$-sector of the Vol~III chiral algebra, via the CY-to-chiral correspondence $\Phi_3$ for toric CY$_3$ without compact $4$-cycles (Theorem~\ref{thm:rsyz}). The passage from $\Eone$ to $\Etwo$ requires the Drinfeld center, and is developed in Chapter~\ref{ch:drinfeld-center}. In every case the algebraization invariant $\kappa_{\mathrm{ch}}$ (computed intrinsically from the resulting chiral algebra) must be distinguished from the manifold invariant $\kappa_{\mathrm{cat}} = \chi(\cO_X)$ (holomorphic Euler characteristic, fixed by the geometry and independent of algebraization); agreement $\kappa_{\mathrm{ch}} = \kappa_{\mathrm{cat}}$ is a prediction of the functor, verified at $d = 2$ and conjectural at $d = 3$.

\section{Bridgeland stability conditions and wall-crossing}
\label{sec:bridgeland-stab}

The space of Bridgeland stability conditions $\Stab(\cC)$ on a CY$_d$ category $\cC$ is a complex manifold (Bridgeland 2007). Its chambers correspond to $t$-structures on $\cC$, and wall-crossing between chambers governs mutations of exceptional collections, DT invariant jumps, and (in the Vol~III reading) transitions between different $\Eone$-chart presentations of the chiral algebra.

\begin{definition}[Bridgeland stability condition]
\label{def:bridgeland-stability}
A \emph{Bridgeland stability condition} on a triangulated category $\cC$ is a pair $\sigma = (Z, \cP)$, where $Z \colon K_0(\cC) \to \C$ is a group homomorphism (the \emph{central charge}) and $\cP = \{\cP(\phi)\}_{\phi \in \bR}$ is a slicing of $\cC$ into full additive subcategories (the \emph{semistable objects of phase $\phi$}), subject to: (i) for $0 \neq E \in \cP(\phi)$, $Z(E) = m(E) \exp(i \pi \phi)$ with $m(E) > 0$; (ii) $\cP(\phi + 1) = \cP(\phi)[1]$; (iii) Harder--Narasimhan filtrations exist (Bridgeland 2007).
\end{definition}

For CY$_3$ categories the stability manifold can have many wall-crossing chambers. On K3 surfaces, Bridgeland (2008) proved that $\Stab(D^b(\Coh(X)))$ is a connected, simply connected complex manifold containing the complexified K\"ahler cone as an open subset. The autoequivalence group $\mathrm{Aut}(D^b(\Coh(X)))$ acts on $\Stab(X)$ by deck transformations, and the quotient is a period domain.

\begin{example}[Stability conditions on K3]
\label{ex:stab-k3}
For a K3 surface $X$ with Picard rank $\rho$, the distinguished connected component $\Stab^\dagger(X) \subset \Stab(X)$ is a $(\rho + 2)$-dimensional complex manifold. The central charge takes the form $Z_{\omega, B}(E) = -\int_X e^{-(B + i\omega)} \cdot \mathrm{ch}(E)$ for $\omega$ an ample class and $B \in H^{1,1}(X, \bR)$. Walls are real-codimension-one loci where a semistable object acquires a destabilizing subobject; crossing a wall exchanges the two hearts related by a tilt. Under the correspondence $\Phi_2$, different stability chambers yield different $\Eone$-chart presentations of $\cA_{K3}$, and the wall-crossing transformations correspond to $R$-matrix gauge equivalences between charts.
\end{example}

\begin{example}[Stability conditions on the resolved conifold]
\label{ex:stab-conifold}
For the resolved conifold $X = \cO_{\P^1}(-1)^{\oplus 2}$, the stability manifold $\Stab(D^b_{cs}(\Coh(X)))$ has a single wall at $\mathrm{Im}(Z(\cO_{\P^1}(-1))) = 0$, separating two chambers: the large-volume chamber (where $\cO_{\P^1}(-1)$ is stable) and the orbifold chamber (where the fractional branes are stable). The wall-crossing formula is the Kontsevich--Soibelman identity for the conifold DT invariants (Kontsevich--Soibelman 2008; Nagao--Nakajima 2011). On the chiral side, this single wall-crossing corresponds to the mutation relating the two Klebanov--Witten quiver presentations of Example~\ref{ex:conifold-derived}.
\end{example}

The general principle is that the stability manifold $\Stab(\cC)$ provides an atlas of $\Eone$-chart descriptions of the chiral algebra $\Phi(\cC)$; the transition functions are $R$-matrix gauge equivalences encoded by wall-crossing automorphisms. This perspective is developed in Chapter~\ref{ch:toric-coha} for toric CY$_3$ categories and is conjectural in general (conditional on CY-A$_3$).

\subsection{Wall-crossing as $R$-matrix gauge transformation}
\label{subsec:wallcrossing-rmatrix}

The dictionary between Bridgeland wall-crossing and $R$-matrix gauge transformations is the second functoriality clause U2 of the CY-to-chiral correspondence programme $\{\Phi_d\}$ (the first, U1, being the action on Morita equivalences; see Remark~\ref{rem:phi-functoriality-summary} below for the U1--U4 summary). Under $\Phi$, a Bridgeland slicing yields a $t$-structure on $\cC$, a $t$-structure yields a heart $\cC^\heartsuit$, the heart yields a tilting object (when one exists), the tilting object yields a quiver $Q_\sigma$, and the quiver gives an $\Eone$-chart of the chiral algebra with a specific generator basis. Two charts $\sigma_1, \sigma_2$ in adjacent chambers of $\Stab(\cC)$ produce two presentations $(Q_{\sigma_1}, R_{\sigma_1})$ and $(Q_{\sigma_2}, R_{\sigma_2})$ of the same chiral algebra; the wall-crossing automorphism is the $R$-matrix gauge transformation relating them.

\begin{conjecture}[Wall-crossing dictionary]
\label{conj:wallcrossing-dictionary}
\ClaimStatusConjectured
Let $\cC$ be a smooth proper CY$_d$ category for which $\Phi$ is defined ($d \in \{1, 2, 3\}$, the latter via Theorem~CY-A$_3$). Let $\sigma_1, \sigma_2 \in \Stab(\cC)$ lie in adjacent chambers separated by a single wall $W$, and let $\mathrm{KS}_W \in \mathrm{Aut}(\cC)$ be the Kontsevich--Soibelman wall-crossing automorphism. Then
\[
 \Phi(\mathrm{KS}_W) \;=\; g_W \in \mathrm{GL}_n\bigl(\Phi(\cC)\bigr),
\]
a gauge transformation of the chiral algebra realising the change of basis between the two $\Eone$-presentations. The classical $r$-matrix transforms by
\[
 r_{\sigma_2}(z) \;=\; g_W(z) \cdot r_{\sigma_1}(z) \cdot g_W(z)^{-1} \;-\; (\partial_z g_W)(z) \cdot g_W(z)^{-1},
\]
the standard gauge action on a meromorphic connection on the configuration space.
\end{conjecture}

\begin{remark}[Categorical content of the gauge action]
\label{rem:categorical-gauge}
The transformation rule in Conjecture~\ref{conj:wallcrossing-dictionary} is the classical gauge action of the loop group $G(\!(z)\!)$ on the space of $\mathfrak{g}$-valued meromorphic 1-forms on the punctured disk. The categorical lift comes from the fact that $\mathrm{KS}_W$ is a Fourier--Mukai-type autoequivalence of $\cC$ whose kernel changes the basis of $\HH^\bullet(\cC)$ by the wall-crossing matrix; transporting this through Theorem~\ref{thm:hkr-cy} gives a change of generator basis on the Hochschild side, and thence a change of $r$-matrix on the chiral side. The gauge transformation distinguishes the algebraisation invariant $\kappa_{\mathrm{ch}}$ (preserved under $\Phi(\mathrm{KS}_W)$, since gauge transformation acts trivially on the trace) from the manifold invariant $\kappa_{\mathrm{cat}}$ (preserved a fortiori).
\end{remark}

\begin{example}[Single-wall conifold wall-crossing as R-matrix gauge]
\label{ex:conifold-rmatrix-gauge}
For the resolved conifold of Example~\ref{ex:stab-conifold}, the single wall separates the large-volume chamber (geometric DT invariants) from the orbifold chamber (noncommutative DT invariants). The Kontsevich--Soibelman wall-crossing automorphism is
\[
 \mathrm{KS}_W \;=\; \mathrm{Ad}_{T_E}^{n_E} \quad\text{where } T_E = \prod_{m \in \Z} (1 - q^m e_E)^{(-1)^m},
\]
with $n_E$ the DT count of the wall-stable object $E = \cO_{\P^1}(-1)$ and $e_E$ the corresponding root vector in the affine super Yangian $Y(\widehat{\mathfrak{gl}}(1|1))$ (Example~\ref{ex:conifold-derived}). Under $\Phi_3$, this lifts to the gauge transformation $g_W(z) = T_E(z)$ of the chiral $r$-matrix; the two presentations of $\Phi_3(D^b_{cs}(\Coh(X)))$ are related by conjugation, and the modular characteristic is preserved on both sides ($\kappa_{\mathrm{ch}} = 0$ since the conifold has $\chi(\cO) = 0$).
\end{example}

The categorical content of wall-crossing is that the $R$-matrix is a connection on a flat bundle over $\Stab(\cC)$, and the holonomy of this connection around walls reproduces the Kontsevich--Soibelman wall-crossing formulae. This is the geometric counterpart of the fact that on the algebraic side, the $\Eone$-chiral algebra has a single isomorphism class but many quasi-equivalent presentations indexed by stability conditions.

\section{Birational geometry: flops and derived equivalences}
\label{sec:flops-derived}

A second source of derived autoequivalences is birational geometry. The Bondal--Orlov reconstruction theorem identifies smooth projective varieties with ample canonical or anti-canonical bundle from their derived categories, but for Calabi--Yau varieties the canonical bundle is trivial and birational CY varieties typically have equivalent derived categories. Flops are the prototypical example.

\begin{theorem}[Bridgeland's flop equivalence]
\label{thm:bridgeland-flop}
\ClaimStatusProvedElsewhere
Let $X \to Y \leftarrow X^+$ be a standard threefold flop between smooth projective Calabi--Yau threefolds (i.e.\ $X$ and $X^+$ are connected by contracting a $\P^1$ and re-resolving). Then there is a derived equivalence
\[
 D^b(\Coh(X)) \;\simeq\; D^b(\Coh(X^+))
\]
realised by a Fourier--Mukai functor with kernel the structure sheaf of the fibre product $X \times_Y X^+$ (Bridgeland 2002).
\end{theorem}

Under the correspondence $\Phi_3$, Theorem~\ref{thm:bridgeland-flop} produces a chiral equivalence: $\Phi_3(D^b(\Coh(X))) \simeq \Phi_3(D^b(\Coh(X^+)))$. The two presentations differ by an $R$-matrix gauge transformation analogous to wall-crossing (the flop corresponds to crossing a wall in $\Stab$).

\begin{example}[Atiyah flop]
\label{ex:atiyah-flop}
The Atiyah flop is the simplest non-trivial threefold flop: $X = \cO_{\P^1}(-1) \oplus \cO_{\P^1}(-1)$ (the resolved conifold of Examples~\ref{ex:conifold-derived} and~\ref{ex:stab-conifold}) flops to $X^+$, an isomorphic CY threefold with the orientation of the $\P^1$ reversed. The two are related by the Bridgeland equivalence, and on the chiral side by the gauge transformation of Example~\ref{ex:conifold-rmatrix-gauge}. In particular, $\kappa_{\mathrm{ch}}(\Phi_3(X)) = \kappa_{\mathrm{ch}}(\Phi_3(X^+)) = 0$ (both have $\chi(\cO) = 0$, per the resolved-conifold compactly-supported computation), confirming Conjecture~\ref{conj:wallcrossing-dictionary} in this birational instance.
\end{example}

\begin{remark}[McKay correspondence as derived equivalence]
\label{rem:mckay-derived}
\ClaimStatusProvedElsewhere
The McKay correspondence supplies a third source of derived equivalences. Bridgeland--King--Reid \cite{BridgelandKingReid2001} prove that for $\Gamma \subset \mathrm{SL}(d, \C)$ a finite subgroup with $d \leq 3$, there is an equivalence
\[
 D^b(\Coh^\Gamma(\C^d)) \;\simeq\; D^b(\Coh(Y)),
\]
where $Y \to \C^d/\Gamma$ is the canonical (crepant) resolution. The left side is the equivariant derived category, equivalent to the derived category of the orbifold $[\C^d/\Gamma]$. Under $\Phi$, the McKay equivalence becomes a chiral equivalence between the orbifold chiral algebra (constructed from the $\Gamma$-equivariant CoHA) and the chiral algebra of the resolution; this is the chiral analogue of the celebrated DHVW orbifold-resolution equivalence at the level of partition functions.
\end{remark}

\section{Functoriality and the Vol III programme}
\label{sec:vol3-functoriality}

The three sources of derived equivalences in this chapter -- HMS (mirror symmetry), Bridgeland wall-crossing, and birational geometry (flops, McKay) -- all factor through the CY-to-chiral correspondence $\Phi_d$ and produce chiral equivalences on the output side. This is the U1--U4 functoriality structure of $\Phi_d$ in operational form.

\begin{remark}[$\Phi$ functoriality summary]
\label{rem:phi-functoriality-summary}
The four universal properties of $\Phi$ (Theorem~\ref{thm:phi-platonic}) read in the derived-category setting:
\begin{itemize}
 \item \textbf{U1} (Morita equivalences): $\Phi$ sends Morita-equivalent CY$_d$ categories to isomorphic chiral algebras. The HMS equivalences (Theorems~\ref{thm:hms-d1}, \ref{thm:hms-d2}, \ref{thm:hms-quintic}) are instances.
 \item \textbf{U2} (gauge transformations): $\Phi$ sends Bridgeland wall-crossing to $R$-matrix gauge transformations (Conjecture~\ref{conj:wallcrossing-dictionary}). The Atiyah flop (Example~\ref{ex:atiyah-flop}) and McKay equivalence (Remark~\ref{rem:mckay-derived}) are derived consequences.
 \item \textbf{U3} (deformations): $\Phi$ converts deformations of the CY category (variations of complex structure, K\"ahler moduli) into deformations of the chiral algebra, the latter parametrising a flat bundle on $\Stab(\cC)$.
 \item \textbf{U4} (degenerations): $\Phi$ sends degenerations of CY$_d$ to degenerations of chiral algebras, including the orbifold/large-volume limits and the LG/CY correspondence (Chapter~\ref{ch:matrix-factorizations}, Section~\ref{sec:lg-cy-correspondence}).
\end{itemize}
The four clauses are the operational content of the slogan ``$\Phi$ is functorial and continuous on the moduli stack of CY$_d$ categories''. The Bridgeland stability manifold $\Stab(\cC)$ is the local model for U2 and U3; the Kontsevich--Soibelman wall-crossing formulae govern the holonomy of $\Phi$ along walls.
\end{remark}

\begin{remark}[Cross-chapter connectivity]

\section{The CY-2 landscape dichotomy}
\label{sec:dcy-psi-landscape-routes}

The K3-specific fingerprint of $\mathbf{H}_{\Delta_5}$ is pinned
against three Witten-type hypotheses: topological Euler
characteristic $\chi_{\mathrm{top}}(K3) = 24$ (T1), vanishing
first Chern class $c_1(K3) = 0$ (T2), and Hodge number
$h^{2,0}(K3) = 1$ (T3). The remaining CY-$2$
landscape --- Enriques surfaces, complex tori~$T^4$, Kummer K3s,
bielliptic surfaces, and the rational elliptic surface
$\mathrm{dP}_9$ --- each produces the precise analogue of
the fourth-taxon crown~$\mathbf{H}_{\Delta_5}$ under the
CY-to-chiral functor~$\Phi_2$. The answers partition the CY-$2$
landscape into four strata, each image living in a distinct
Drinfeld-landscape taxon. The ordering of the strata below is by
decreasing~$\chi_{\mathrm{top}}$; the Borcherds lift,
Gritsenko--Nikulin additive lift, and Siegel-cusp-form weight
track~$\chi_{\mathrm{top}}$ linearly, which is the arithmetic
content of the dichotomy.

\begin{theorem}[Enriques analogue $\mathbf{H}_{\Delta_5}^{\mathrm{Enr}}$;
$\mathbb{Z}/2$-orbifold of the K3 crown at Siegel weight $5/2$]
\label{thm:dcy-enriques-analogue}
\ClaimStatusProvedElsewhere
Let $\mathrm{En} = K3/\iota$ denote an Enriques surface, the free
quotient of a K3 surface $X$ by a fixed-point-free involution
$\iota$ acting as $-1$ on $H^0(X, \omega_X)$
(Barth--Hulek--Peters--Van~de~Ven 2004
\textup{Compact Complex Surfaces}, Chapter~VIII, Proposition~15.1).
The Enriques surface has $\chi_{\mathrm{top}}(\mathrm{En}) = 12$
(half of K3), $c_1 = 0$ as a $\mathbb{Q}$-class (non-trivial
$2$-torsion $c_1 \in H^2(\mathrm{En},\mathbb{Z})$ of order~$2$),
and $h^{2,0}(\mathrm{En}) = 0$. The integral cohomology lattice is
\[
  H^2(\mathrm{En},\mathbb{Z})/\mathrm{torsion} \;\simeq\; E_8(-1) \oplus \mathrm{II}_{1,1},
\]
of signature $(1,9)$, equal to the Enriques lattice
$\Lambda_{\mathrm{Enr}}$ of Nikulin 1979
\textup{Izv.~Akad.~Nauk}~43. Applying $\Phi_2$ to
$D^b(\mathrm{Coh}(\mathrm{En}))$ and composing with the
$\mathbb{Z}/2$-orbifold gauging (Kirillov--Ostrik 2002
\textup{Adv.~Math.}~171, Theorem~5.1, applied to the $\iota$-lift
on $\mathcal{H}_{K3}$) gives the Enriques analogue
\[
  \mathbf{H}_{\Delta_5}^{\mathrm{Enr}}
  \;=\;
  \Phi_2\bigl(D^b(\mathrm{Coh}(\mathrm{En}))\bigr)
  \;=\;
  \mathbf{H}_{\Delta_5}\,/\!\!/\,\langle\iota\rangle,
\]
a $\mathbb{Z}/2$-orbifold chiral algebra at level~$2$. The genus-2
Siegel character of $\mathbf{H}_{\Delta_5}^{\mathrm{Enr}}$ is
$1/\Delta_5^{\mathrm{Enr}}(Z)$, where
$\Delta_5^{\mathrm{Enr}}$ is the weight-$5/2$ Siegel paramodular
form of level~$2$ on the paramodular group $K(2)\subset\mathrm{Sp}_4(\mathbb{Q})$
of Gritsenko--Nikulin 1998 \textup{J.~reine angew.~Math.}~507,
Theorem~5.2 (Borcherds lift on $\mathrm{II}_{2,10}(2)$).
Half-integrality of the weight tracks
$h^{2,0}(\mathrm{En}) = 0$ and the $\mathbb{Z}/2$-gauging on the
Gritsenko--Nikulin additive-lift side; the weight-$5/2$ rather
than~$5$ is the chiral-side avatar of
$\chi_{\mathrm{top}}(\mathrm{En})/\chi_{\mathrm{top}}(K3) = 1/2$.
\end{theorem}

\begin{proof}[Proof sketch]
The $\iota$-fixed locus on~$\mathcal{H}_{K3}$ picks out the
$\iota$-invariant sublattice
$\Lambda_{\mathrm{Enr}}^+ = E_8(-1)\oplus\mathrm{II}_{1,1}$; the
Nakajima Heisenberg action (1997 \textup{Ann.~Math.}~145,
Theorem~3.10) restricts to the $\iota$-fixed Hilbert-scheme
component. The Kirillov--Ostrik gauging of the Mukai--Heisenberg
vertex algebra at level~$2$ produces a $\mathbb{Z}/2$-orbifold
VOA; the Borcherds multiplicative lift on the Enriques lattice
$\mathrm{II}_{2,10}(2)$ (Gritsenko--Nikulin 1998, Theorem~5.2)
produces precisely the weight-$5/2$ paramodular form
$\Delta_5^{\mathrm{Enr}}$ as denominator of the orbifold BKM
superalgebra $\mathfrak{g}_{\Delta_5}^{\mathrm{Enr}}$ (Hulek--Sankaran 2011
\textup{Algebra and Number Theory}~5, Theorem~3.4;
Dolgachev--Kond\={o} 2002
\textup{Asian~J.~Math.}~6, Theorem~4.1). The half-integral weight
is forced by the signature $(2,10) = (1,9) + (1,1)$ with a
$\mathbb{Z}/2$-metaplectic cover on the $(1,1)$-factor; this is
the Oguiso--Sakai 2001 \textup{Nagoya~Math.~J.}~163 fingerprint.
\end{proof}

\begin{remark}[$T^4$ abelian surface: $\Phi_2$-image is the
abelian Heisenberg VOA, no Borcherds lift]
\label{rem:dcy-t4-abelian}
\ClaimStatusProvedElsewhere
The complex torus $T^4 = \mathbb{C}^2/\Lambda$ is CY-$2$ with
$\chi_{\mathrm{top}}(T^4) = 0$ (Gauss--Bonnet on a flat manifold),
$c_1(T^4) = 0$, $h^{2,0}(T^4) = 1$. Under~$\Phi_2$, the K3-CoHA
argument of Remark~\ref{rem:dcy-kst-phi2-k3-coha} replaces
Nakajima's Hilbert-scheme Heisenberg action
on~$H^\bullet(\mathrm{Hilb}^n(K3))$ by the parallel construction on
$\mathrm{Hilb}^n(T^4)$ (Nakajima 1997 \textup{Ann.~Math.}~145,
Theorem~3.10 also for abelian surfaces; Nahm--Wendland 2001
\textup{Comm.~Math.~Phys.}~216, Theorem~4.1). Since
$\chi_{\mathrm{top}}(T^4) = 0$, the G\"ottsche generating function
becomes
$\sum_n \chi(\mathrm{Hilb}^n(T^4))\,q^n = \prod_m(1-q^m)^0 = 1$,
degenerate. The Borcherds multiplicative lift of the $T^4$
elliptic genus $\phi_{0,1}^{T^4} = 0$ is trivial, so there is no
weight-$5$ Siegel cusp form on the abelian-surface side. The
$\Phi_2$-image is
\[
  \Phi_2\bigl(D^b(\mathrm{Coh}(T^4))\bigr)
  \;=\;
  V_{\mathrm{II}_{4,4}}
  \;=\;
  \text{abelian Heisenberg VOA on }\mathrm{II}_{4,4},
\]
the even self-dual rank-$8$ Narain lattice VOA of signature
$(4,4)$ (Nahm--Wendland 2001, Theorem~4.1; Frenkel--Lepowsky--Meurman 1988
\textup{Vertex Operator Algebras and the Monster}, \S10.4 for the
Narain-lattice-VOA construction). This places
$\Phi_2(T^4)$ squarely in Drinfeld taxon~$G$ (abelian Heisenberg;
landscape census \texttt{landscape\_census.tex}, family G),
\emph{not} in the fourth taxon of $\mathbf{H}_{\Delta_5}$. The
degeneration $\chi_{\mathrm{top}} = 0 \Rightarrow \Phi_2(\text{CY-}2)\in G$
is the $N = 1$ Conway-type BKM limit: the imaginary-root structure
collapses to rank~$0$, leaving the classical Narain lattice.
\end{remark}

\begin{remark}[Kummer K3 $\mathrm{Kum}(T^4) \hookrightarrow K3$:
$\mathbf{H}_{\Delta_5}$ on the $16$-node Picard sublattice]
\label{rem:dcy-kummer}
\ClaimStatusProvedElsewhere
The Kummer K3 associated to an abelian surface $A = T^4$ is the
minimal resolution $\mathrm{Kum}(A) = \widetilde{A/\pm 1}$, with
$\chi_{\mathrm{top}}(\mathrm{Kum}(A)) = \chi_{\mathrm{top}}(A/\pm 1) + 16
 = (\chi(T^4) - 16)/2 + 16 + 16 = 0 + 16 + 8 = 24$ after resolving
the $16$ nodes via Euler-characteristic additivity; equivalently,
$\chi(\mathrm{Kum}) = 24$ via $\chi_{\mathrm{top}}(K3) = 24$
(Barth--Hulek--Peters--Van~de~Ven 2004, Chapter~VIII, \S14). The
$\mathrm{Kum}(A)$ is an algebraic K3 with transcendental
lattice $T_{\mathrm{Kum}} \subset T_A(2)$ and Picard sublattice
\[
  \Lambda_{\mathrm{Kum}}
  \;=\;
  E_8(-2)^{\oplus 2} \oplus \mathrm{II}_{1,1}(2) \oplus \langle -2\rangle^{16}
  \;\subset\;
  \mathrm{II}_{3,19},
\]
where the $16$ classes of $(-2)$-curves are the exceptional
divisors of the blow-up (Mukai 1984 \textup{Nagoya~Math.~J.}~81,
Theorem~1; Morrison 1984 \textup{Invent.~Math.}~75, Theorem~3.1).
For special $T^4$ (those with enough Hodge structure to be
algebraic), Mukai's 1984 theorem gives
$\mathrm{Kum}(T^4) \simeq K3^{\mathrm{alg}}$ as complex manifolds;
for generic $T^4$, the Kummer is transcendental K3. Under~$\Phi_2$,
the image is
\[
  \Phi_2\bigl(D^b(\mathrm{Coh}(\mathrm{Kum}(A)))\bigr)
  \;=\;
  \mathbf{H}_{\Delta_5}\,\big|_{\Lambda_{\mathrm{Kum}}}
  \;\subset\;
  \mathbf{H}_{\Delta_5},
\]
the Kummer-lattice restriction of the fourth-taxon crown: the
same Mukai--Heisenberg VOA, but with currents indexed by
$\Lambda_{\mathrm{Kum}}$ rather than the full~$\mathrm{II}_{3,19}$.
The Gritsenko--Nikulin additive lift restricts to
$\Delta_5\big|_{\mathcal{A}_2^{\mathrm{Kum}}}$, a Siegel cusp form
on the Kummer-locus of the moduli of principally-polarised abelian
surfaces $\mathcal{A}_2$ (Gritsenko--Nikulin 1998, \S 4;
Morrison--Vafa 1996 \textup{Nucl.~Phys.~B}~476, II.\S3 for the
physical origin of this restriction on the string-theory moduli
space).
\end{remark}

\begin{remark}[Half-K3 $= \mathrm{dP}_9$ maps to~$\widehat{E_8}_1$;

\label{rem:dcy-halfk3}

(T1): $\chi_{\mathrm{top}}(\mathrm{dP}_9) = 12\ne 24$, and
(T2): $c_1(\mathrm{dP}_9) = -f \ne 0$; only~(T3)
$h^{2,0}(\mathrm{dP}_9) = 0$ is a \emph{non-issue} because
$\mathrm{dP}_9$ has no non-trivial holomorphic~$2$-forms anyway.
Two of the three K3 conditions fail, so there is no Borcherds
product of weight~$5$: the Borcherds singular theta lift (T16(i))
on the half-K3-relevant lattice $\mathrm{II}_{1,9}$ produces an
$E_8$-character generating function rather than a weight-$5$
Siegel form (Borcherds 1998 \textup{Invent.~Math.}~132,
Theorem~10.1, with the weight-$5$ output requiring
$\mathrm{II}_{2,10}$ signature, not $\mathrm{II}_{1,9}$). The
$\Phi_2$-image is
\[
  \Phi_2\bigl(D^b(\mathrm{Coh}(\mathrm{dP}_9))\bigr)
  \;=\;
  \widehat{E_8}_{\,k=1},
\]
the affine $E_8$ Kac--Moody algebra at level~$1$. The $E_8$
origin is the $8$d heterotic / F-theory compactification on
$\mathrm{dP}_9$ where the $12$ $I_1$-fibres collide into a single
$II^*$-fibre supporting the unbroken $E_8$ gauge symmetry
(Seiberg 1988, \S3; Morrison--Vafa 1996, II.\S4). The level is
fixed by the $I_1$-count $12 = \chi_{\mathrm{top}}(\mathrm{dP}_9)
/ 1$ via the gauge-anomaly-cancellation condition; the $E_8$
Dynkin data match the Seiberg--Witten curve monodromy around
the $II^*$-point. This places $\Phi_2(\mathrm{dP}_9)$ in Drinfeld
taxon~L (affine Kac--Moody; landscape census

\end{remark}

\begin{remark}[CY-$2$ landscape dichotomy: the $\chi_{\mathrm{top}}/c_1/h^{2,0}$ partition]
\label{rem:dcy-landscape-partition}
\ClaimStatusProvedHere
The CY-to-chiral functor~$\Phi_2$ partitions the CY-$2$ landscape
into four strata, indexed by
$(\chi_{\mathrm{top}}, c_1, h^{2,0})$, with each stratum hitting a
distinct Drinfeld-landscape taxon:
\begin{center}
\begin{tabular}{lcccl}
\toprule
Surface & $\chi_{\mathrm{top}}$ & $c_1$ & $h^{2,0}$ & $\Phi_2$-taxon \\
\midrule
K3 & $24$ & $0$ & $1$ & 4th taxon: $\mathbf{H}_{\Delta_5}$ \\
Enriques & $12$ & $0$ & $0$ & 4th taxon / $\mathbb{Z}/2$: $\mathbf{H}_{\Delta_5}^{\mathrm{Enr}}$ \\
$T^4$ & $0$ & $0$ & $1$ & G: abelian Heisenberg $V_{\mathrm{II}_{4,4}}$ \\
bielliptic & $0$ & $0$ & $0$ & G: $V_{\mathrm{II}_{4,4}}/G$ \\
$\mathrm{dP}_9$ & $12$ & $-f$ & $0$ & L: $\widehat{E_8}_{\,k=1}$ \\
\bottomrule
\end{tabular}
\end{center}
The Kummer K3 sits inside the K3 row of this table as a
sublattice restriction
$\mathbf{H}_{\Delta_5}\big|_{\Lambda_{\mathrm{Kum}}}$
(Remark~\ref{rem:dcy-kummer}), not as a distinct
stratum. The dichotomy is controlled as follows:
\begin{itemize}
\item $\chi_{\mathrm{top}} = 24$ and $c_1 = 0$: K3 and Kummer K3,
  4th taxon. The Borcherds weight is~$5$ (integer); the Siegel
  cusp form is~$\Delta_5$ on the full $\mathrm{Sp}_4$.
\item $\chi_{\mathrm{top}} = 12$ and $c_1 = 0$: Enriques, 4th
  taxon at level~$2$ ($\mathbb{Z}/2$-gauging). The Borcherds
  weight is~$5/2$ (half-integer); the Siegel cusp form lives on
  the paramodular subgroup $K(2)\subset\mathrm{Sp}_4$ and carries
  a metaplectic cover.
\item $\chi_{\mathrm{top}} = 0$: $T^4$ and bielliptic, taxon~G
  (abelian Heisenberg). No Borcherds lift; the generating function
  is $1$ (degenerate).
\item $c_1\ne 0$: $\mathrm{dP}_9$ and other rational elliptic
  surfaces, taxon~L (affine Kac--Moody at level~$1$). The
  $8$d gauge group is fixed by the elliptic-fibre monodromy;
  for~$\mathrm{dP}_9$ the group is~$E_8$.
\end{itemize}
The arithmetic invariant
$w_{\mathrm{Borch}}\,(X) = 5 \cdot \chi_{\mathrm{top}}(X)/\chi_{\mathrm{top}}(K3)
 = 5\chi_{\mathrm{top}}(X)/24$ reads off the Siegel-form weight
along the 4th-taxon column ($X = K3$: $w = 5$; $X = \mathrm{En}$:
$w = 5/2$), and vanishes on the taxon-G column ($w = 0$). This is
the first-principles content of the K3-specificity claim:
the Witten hypothesis $(\chi, c_1, h^{2,0}) = (24, 0, 1)$ picks out

\section{The global Bridgeland $\leftrightarrow$ $R$-matrix dictionary}
\label{sec:dcy-global-bridgeland-dictionary}

Remark~\ref{rem:dcy-kst-bridgeland-rmatrix} installed the Gepner

\begin{theorem}[Beilinson dictionary on $\Stab^\dagger/\Gamma$]
\label{thm:dcy-beilinson-dictionary}
\ClaimStatusProvedHere
Fix a K3 surface $S$ with distinguished component
$\Stab^\dagger(D^b\mathrm{Coh}(S))\subset\Stab(D^b\mathrm{Coh}(S))$ and
autoequivalence group $\Gamma=\mathrm{Aut}(D^b\mathrm{Coh}(S))$. The
Bayer--Macr\`i--Igusa period map
\[
  \iota\colon\;
  \Stab^\dagger/\Gamma\;\xrightarrow{\;\pi\;}\;
  \mathcal{P}(\widetilde{\Lambda}_{K3})\;\xrightarrow{\;\mathrm{Sieg}\;}\;
  \overline{\mathcal{A}_2},
\]
composed with the rational-theta factorisation
$R_{\mathrm{Sieg,dyn}}(u,Z)=R^{\mathrm{rat}}_{\mathrm{Yang}}(u)
\cdot\theta^{\mathrm{K3}}(u,Z)$, yields a well-defined functor
\[
  \Theta\colon\;
  \Stab^\dagger(D^b\mathrm{Coh}(S))/\Gamma
   \;\longrightarrow\;
   \bigl\{R\text{-matrices for }\mathbf{H}_{\Delta_5}\bigr\},
  \qquad
  \sigma\longmapsto R_{\mathrm{Sieg,dyn}}\bigl(u;\iota(\sigma)\bigr).
\]
Its properties:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $\Theta$ is continuous on chamber interiors; the value inside a
  fixed chamber $C_\sigma$ is a single gauge-equivalence class of
  $R$-matrices satisfying the RTT relation for
  $\mathbf{H}_{\Delta_5}$;
\item on wall-crossing $\sigma\to\sigma'$ across a KS wall
  $W_{(v_1,v_2)}$, $\Theta(\sigma')$ is obtained from $\Theta(\sigma)$
  by Kontsevich--Soibelman gauge conjugation
  $R^{(\sigma')}=F_{\sigma\sigma'}\cdot R^{(\sigma)}\cdot
  F_{\sigma\sigma'}^{-1}$
  (Theorem~\ref{thm:cwc-ks-r-conjugation} of
  Chapter~\ref{ch:coha-wall-crossing-platonic});
\item the three canonical loci
  (large-volume cusp / Gepner Heegner point / Humbert-$H_1$ boundary)
  specialise $\Theta$ to
  $R^{\mathrm{rat}}_{\mathrm{Yang}}$, $R^{\mathrm{trig}}_{q=\zeta_8}$,
  $R^{\mathrm{ell,dyn}}_{\mathrm{Felder}}$ respectively
  (Remark~\ref{rem:cwc-three-tier} of
  Chapter~\ref{ch:coha-wall-crossing-platonic});
\item the image of $\Theta$ is contained in the fourth Drinfeld--Belavin
  class $R_{\mathrm{Sieg,dyn}}$
  (Theorem~\ref{thm:R-sieg-fourth-class}), and every value of
  $\Theta$ is obtained as a specialisation of
  $R_{\mathrm{Sieg,dyn}}$ at an $\overline{\mathcal{A}_2}$-point.
\end{enumerate}
\end{theorem}

\begin{proof}
See Theorem~\ref{thm:cwc-global-dictionary} of
Chapter~\ref{ch:coha-wall-crossing-platonic}. The present theorem records
the same content on the derived-category side: the Bayer--Macr\`i map
$\pi\colon\Stab^\dagger\to\mathcal{P}(\widetilde{\Lambda}_{K3})$
(Bayer--Macr\`i 2014 \emph{Invent.\ Math.}~198 Theorem~1.1) is a local
isomorphism with the autoequivalence-group action
$\Gamma\curvearrowright\Stab^\dagger$ projecting to the Hodge-theoretic
action of $\Gamma$ on periods. The Siegel restriction from the
$\widetilde{\Lambda}_{K3}$-period domain to $\mathbb{H}_2$ along the
hyperbolic $\mathrm{II}_{2,1}$ core is Gritsenko--Nikulin 1998
\emph{Internat.\ J.\ Math.}~9 Section~5; its Igusa-compatibility
(intertwining $\Gamma$ with the paramodular $K(1)$) is
Remark~\ref{rem:cwc-iota-map}. Clauses (i)--(iv) transcribe the
content of
Theorems~\ref{thm:cwc-chamber-coha-drinfeld},
\ref{thm:cwc-ks-r-conjugation},
\ref{thm:cwc-global-dictionary} under the derived-category
identification $\Phi_2(D^b\mathrm{Coh}(K3))\simeq H_{\mathrm{Muk}}$ and
its Hall--Drinfeld double
$\mathcal{D}_\hbar(\mathcal{Y}^{\mathrm{Hall}}_\hbar(\mathrm{CoHA}_{K3}))$
$\simeq \mathbf{H}_{\Delta_5}$
(Remark~\ref{rem:dcy-kst-phi2-k3-coha}).
\end{proof}

\section{Stability manifold of the total space $K3 \times E$}
\label{sec:dcy-stab-k3-times-e}

The CY-$3$ total space $Y = K3 \times E$ admits a derived category $D^b(\Coh(Y))$ whose stability manifold is the local model of the chiral $R$-matrix atlas for $\mathbf{H}_{\Delta_5}$ on $Y$. Its complex dimension, and the codimension of the K\"unneth slice $\Stab(K3) \times \Stab(E) \hookrightarrow \Stab(Y)$, determine the transverse mixed-charge directions visible to $\Phi_3$ but invisible to the product decomposition $\Phi_2 \boxtimes \Phi_1$.

\subsection{Dimension via numerical K-theory}
\label{subsec:dcy-stab-dim-k3-times-e}

\begin{theorem}[Stability dimension of $D^b(\Coh(K3 \times E))$]
\label{thm:dcy-stab-dim-48}
\ClaimStatusProvedHere
Let $Y = K3 \times E$ with $K3$ a smooth projective complex K3 surface and $E$ a smooth projective complex elliptic curve. The space of locally finite Bridgeland stability conditions on $D^b(\Coh(Y))$ is a complex manifold of dimension
\[
 \dim_\C \Stab\bigl(D^b(\Coh(K3 \times E))\bigr) \;=\; 48.
\]
\end{theorem}

\begin{proof}
By Bridgeland 2007 \emph{Ann.\ Math.}~166 Theorem~1.2, for any essentially small triangulated category $\cC$ with finite-rank numerical Grothendieck group, the map $\cZ \colon \Stab(\cC) \to \Hom_\Z(K_{\mathrm{num}}(\cC), \C)$ sending $(Z, \cP) \mapsto Z$ is a local homeomorphism onto an open subset of the finite-dimensional complex vector space $\Hom_\Z(K_{\mathrm{num}}(\cC), \C)$. Consequently
\[
 \dim_\C \Stab(\cC) \;=\; \rk K_{\mathrm{num}}(\cC).
\]
For $\cC = D^b(\Coh(Y))$ with $Y$ smooth projective, $K_{\mathrm{num}}(\cC) = K_{\mathrm{num}}(Y)$, the numerical quotient of $K_0(Y)$ by classes of Euler-pairing norm zero. The Chern character $\mathrm{ch} \colon K_{\mathrm{num}}(Y) \otimes \Q \xrightarrow{\sim} H^{\mathrm{even}}_{\mathrm{alg}}(Y, \Q)$ of Atiyah 1957 \emph{Proc.\ LMS}~VII (in the form refined by Mukai 1987 \emph{Nagoya Math.\ J.}~81 for K3 surfaces and extended by Thomason 1988 \emph{Duke}~56 Theorem~2.1 to the Künneth property of algebraic K-theory) gives
\[
 \rk K_{\mathrm{num}}(K3 \times E)
 \;=\; \rk K_{\mathrm{num}}(K3) \cdot \rk K_{\mathrm{num}}(E)
 \;=\; 24 \cdot 2
 \;=\; 48.
\]
The factor $24$ is the rank of the Mukai lattice $\widetilde{\Lambda}_{K3} = H^0 \oplus H^2 \oplus H^4$ of the K3 surface, of signature $(4, 20)$, decomposed as $1 + 22 + 1$. The factor $2$ is the rank of $K_{\mathrm{num}}(E) = \Z[\cO_E] \oplus \Z[\cO_p]$ generated by the structure sheaf and a skyscraper at a point, the even cohomology $H^0(E) \oplus H^2(E) = \Z \oplus \Z$. Toda 2013 \emph{Adv.\ Math.}~234 extends Bridgeland's construction to CY$_3$ total spaces with the same dimension formula, so the equality $\dim_\C \Stab(D^b(\Coh(Y))) = \rk K_{\mathrm{num}}(Y) = 48$ holds on the CY$_3$ nose.
\end{proof}

\begin{remark}[Multi-path verification of the dimension $48$]
\label{rem:dcy-stab-48-multipath}
Four independent routes produce the same value:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{Bridgeland 2007 Theorem~1.2 on numerical K-theory.} $\dim_\C \Stab(\cC) = \rk K_{\mathrm{num}}(\cC) = 48$.
\item \emph{Thomason 1988 Künneth on K-theory of smooth projective varieties.} $K_{\mathrm{num}}(K3 \times E) = K_{\mathrm{num}}(K3) \otimes_\Z K_{\mathrm{num}}(E)$ of rank $24 \cdot 2 = 48$.
\item \emph{Even cohomology count via Atiyah--Hirzebruch.} Applying the Chern character isomorphism $\mathrm{ch} \colon K_{\mathrm{num}}(Y) \otimes \Q \xrightarrow{\sim} H^{\mathrm{even}}(Y, \Q)$ (Atiyah 1957) to the Künneth expansion yields $H^{\mathrm{even}}(K3 \times E) = H^{\mathrm{even}}(K3) \otimes H^{\mathrm{even}}(E)$ of dimension $24 \cdot 2 = 48$.
\item \emph{Toda 2013 dimension formula for smooth projective CY$_3$.} $\dim_\C \Stab(Y) = \rk K_{\mathrm{num}}(Y) = 48$ when $Y$ is smooth projective Calabi--Yau of dimension $3$, recovered on $K3 \times E$ by the Künneth expansion.
\end{enumerate}
The four routes agree; all independent of any algebraization choice for $\Phi_3$.
\end{remark}

\begin{remark}[Scope: $K_{\mathrm{num}}$ of rank $48$ versus $H^\bullet$ of rank $96$]
\label{rem:dcy-stab-48-vs-96-scope}
The full Betti total of $K3 \times E$ is
\[
 \sum_i \dim_\Q H^i(K3 \times E, \Q)
 \;=\; \Bigl(\sum_i h^i(K3)\Bigr) \cdot \Bigl(\sum_i h^i(E)\Bigr)
 \;=\; 24 \cdot 4 \;=\; 96,
\]
splitting as $H^{\mathrm{even}} \oplus H^{\mathrm{odd}}$ of ranks $48 + 48$. The odd part
\[
 H^{\mathrm{odd}}(K3 \times E) \;=\; H^{\mathrm{even}}(K3) \otimes H^1(E) \;\oplus\; H^{\mathrm{even}}(K3) \otimes H^3(E) \;=\; 24 \cdot 2 \;=\; 48
\]
carries zero Euler pairing against every algebraic class (the pairing pairs $H^p$ against $H^{6-p}$, and the cup product $H^{\mathrm{even}}(K3) \otimes H^1(E) \to H^{\mathrm{odd}}(K3 \times E)$ has image orthogonal to the algebraic diagonal). Consequently the odd summand lies in the kernel of the numerical Grothendieck quotient $K_0(Y) \twoheadrightarrow K_{\mathrm{num}}(Y)$ and does not contribute to $\dim \Stab(Y)$. The Bridgeland manifold sees exactly the algebraic-Mukai-pairing content, rank $48$, not the full Betti rank $96$.
\end{remark}

\subsection{The Künneth slice and its codimension}
\label{subsec:dcy-kunneth-slice-codim}

\begin{definition}[$\Phi$-accessible Künneth slice]
\label{def:dcy-kunneth-slice}
The \emph{$\Phi$-accessible K\"unneth slice} is the locus
\[
 \Stab^\Phi(Y) \;:=\; \Stab\bigl(D^b(\Coh(K3))\bigr) \times \Stab\bigl(D^b(\Coh(E))\bigr)
 \;\hookrightarrow\; \Stab\bigl(D^b(\Coh(Y))\bigr)
\]
parametrising pairs $(\sigma_{K3}, \sigma_E)$ whose product stability condition $\sigma_{K3} \boxtimes \sigma_E$ assigns the central charge $Z_{K3}(E_1) \cdot Z_E(E_2)$ to a pure product class $[E_1 \boxtimes E_2] \in K_{\mathrm{num}}(Y)$ and extends by Harder--Narasimhan to all of $D^b(\Coh(Y))$. The embedding uses the $\Phi$-functoriality clause that $\Phi_3(\sigma_{K3} \boxtimes \sigma_E) = \Phi_2(\sigma_{K3}) \boxtimes \Phi_1(\sigma_E)$ realises a product $E_1$-chart on the chiral side.
\end{definition}

\begin{proposition}[Codimension of the K\"unneth slice]
\label{prop:dcy-kunneth-codim-22}
\ClaimStatusProvedHere
The K\"unneth slice has complex dimension
\[
 \dim_\C \Stab^\Phi(Y) \;=\; \dim_\C \Stab(K3) + \dim_\C \Stab(E) \;=\; 24 + 2 \;=\; 26,
\]
and therefore
\[
 \codim_\C \bigl(\Stab^\Phi(Y) \hookrightarrow \Stab(Y)\bigr) \;=\; 48 - 26 \;=\; 22.
\]
\end{proposition}

\begin{proof}
The factor dimensions follow from Bridgeland 2007 Theorem~1.2 applied to each tensor factor: $\rk K_{\mathrm{num}}(K3) = 24$ gives $\dim_\C \Stab(K3) = 24$ on the distinguished component $\Stab^\dagger(K3)$ of Bridgeland 2008 \emph{Duke}~141, and $\rk K_{\mathrm{num}}(E) = 2$ gives $\dim_\C \Stab(E) = 2$ on the single component identified by Bridgeland 2007 (Example~9.3) and Okada 2006. The product $\Stab(K3) \times \Stab(E)$ has complex dimension $24 + 2 = 26$, and Bayer--Macr\`i 2014 \emph{Invent.\ Math.}~198 Theorem~1.1 (transport of central charges through the Chern character identification) shows that the product stability is well-defined and locally embeds into $\Stab(Y)$ as a smooth complex submanifold. Subtraction yields the codimension $48 - 26 = 22$.
\end{proof}

\begin{proposition}[Non-K\"unneth witnesses to positive codimension]
\label{prop:dcy-non-kunneth-witnesses}
\ClaimStatusProvedHere
The codimension $22$ of Proposition~\ref{prop:dcy-kunneth-codim-22} is strictly positive because $D^b(\Coh(K3 \times E))$ supports families of objects not representable as products. Three explicit families witness the strictness:
\begin{enumerate}[label=\textup{(W\arabic*)}]
\item \emph{Non-split Fourier--Mukai twists.} Let $\cP \in D^b(\Coh(K3 \times K3^\vee))$ be the universal sheaf and $\cP_\tau$ a Fourier--Mukai kernel on $K3 \times E \times K3 \times E$ obtained by twisting $\cP$ along a rank-$2$ variation $\tau \in H^0(E, \cO_E^{\oplus 22})$ of transcendental classes. The resulting Fourier--Mukai functor $\Phi_{\cP_\tau} \in \mathrm{Aut}(D^b(\Coh(K3 \times E)))$ acts by mixing the $22$-dimensional transcendental lattice $T_{K3}$ with the rank-$2$ elliptic lattice, producing a $22$-parameter family of stability conditions outside the product embedding (Mukai 1987 Nagoya~81 in the Mukai-lattice form; C\u{a}ld\u{a}raru 2003 for the Fourier--Mukai extension).
\item \emph{Non-split vertical extensions.} For $p \in E$ and $[\cF] \in T_{K3}$ a transcendental class, the extension
\[
 0 \to \pi_E^* \cO_E(p) \to \cE \to \pi_{K3}^* \cF \to 0
\]
has extension class in $\Ext^1(\pi_{K3}^* \cF, \pi_E^* \cO_E(p)) = H^0(K3, \cF^\vee) \otimes H^1(E, \cO_E(p))$, non-zero whenever $\cF$ is transcendental. Non-zero extensions are not products of K3-objects with $E$-objects, generate a stability direction transverse to $\Stab^\Phi$, and span a $22$-dimensional sublocus corresponding to the transcendental-lattice rank.
\item \emph{Isogeny-graph spectral sheaves.} For an isogeny $\phi \colon E \to E'$ and a spectral cover $\Sigma_\phi \subset K3 \times E$ transverse to the projection $\pi_E$, the pushforward $\phi_* \cO_{\Sigma_\phi}$ is a CY$_3$-rigid sheaf whose support is not of product form. Varying $\phi$ through the isogeny graph of $E$ produces a $22$-parameter family (parametrised by $T_{K3} \otimes \End(E)$) of stability conditions transverse to $\Stab^\Phi$ (Toda 2013 \emph{Adv.\ Math.}~234 for the rigidity and Toda 2014 for the Bridgeland stability in the CY$_3$ setting).
\end{enumerate}
The three families (W1), (W2), (W3) each span a $22$-dimensional transverse sublocus, confirming $\codim \Stab^\Phi \ge 22$; together with Proposition~\ref{prop:dcy-kunneth-codim-22} this forces equality.
\end{proposition}

\begin{remark}[Physical interpretation: class-$\cS$ mixed-charge sector]
\label{rem:dcy-class-s-mixed-charge}
The $22$ transverse directions of Proposition~\ref{prop:dcy-kunneth-codim-22} are exactly the parameters of the non-factorising mixed-charge sector of the class-$\cS$ theory $\cT[A_1, \Sigma_{1,24}]$ of Gaiotto 2009 \emph{JHEP}~0911 on the once-punctured K3-decorated torus. The equality
\[
 22 \;=\; \rk T_{K3} \;=\; \text{rank of the K3 transcendental lattice}
\]
identifies the transverse stability parameters with the transcendental Mukai directions of K3 that fail to factorise against the elliptic direction. On the $\Phi_3$-output side, the K\"unneth-slice chiral algebra $\Phi_2(D^b(\Coh(K3))) \boxtimes \Phi_1(D^b(\Coh(E)))$ captures only the product sector of $\mathbf{H}_{\Delta_5}$; the remaining $22$ non-factorising directions parametrise the Coulomb-branch mixing of $\cT[A_1, \Sigma_{1,24}]$ under the UV-complete class-$\cS$ fibration, with the rank-$22$ K3 transcendental lattice serving as the Coulomb-branch lattice of the non-product sector (cf.\ the BKM Borcherds-weight of $K3 \times E$ at $\kappa_{\mathrm{BKM}} = 5$, Remark~\ref{rem:dcy-k3-vs-k3e-hodge-comparison}). The codimension-$22$ embedding $\Stab^\Phi \hookrightarrow \Stab(Y)$ is the stability-theoretic realisation of the physical slogan that the full $\cT[A_1, \Sigma_{1,24}]$ theory does \emph{not} decompose as a tensor product of a K3 theory with an elliptic theory; the obstruction to decomposition lives in the $22$-dimensional transcendental sector.
\end{remark}

\begin{remark}[Relation to the global Beilinson dictionary]
\label{rem:dcy-stab-ke-beilinson-relation}
Theorem~\ref{thm:dcy-beilinson-dictionary} constructs the global dictionary $\Theta \colon \Stab^\dagger(K3)/\Gamma \to \{R\text{-matrices for }\mathbf{H}_{\Delta_5}\}$ at the CY-$2$ level. The CY-$3$ extension on $\Stab(Y)/\Gamma_Y$, valued in $R$-matrices for the chiral algebra $\Phi_3(D^b(\Coh(Y)))$, is conditional on the non-formal chain-level extension of CY-A$_3$ (Remark~\ref{rem:dcy-closing-wallcrossing-dict}). Theorem~\ref{thm:dcy-stab-dim-48} and Proposition~\ref{prop:dcy-kunneth-codim-22} provide the ambient geometric data: the target of the conditional CY-$3$ dictionary $\Theta_Y$ is a $48$-dimensional complex manifold, and the product-accessible image of $\Theta_{K3} \boxtimes \Theta_E$ occupies a codimension-$22$ slice. The $22$ transverse directions are the frontier of the CY-$3$ Beilinson dictionary: each transverse direction is a non-product $R$-matrix for $\mathbf{H}_{\Delta_5}$ on $Y$ that does not arise from the product of a K3 $R$-matrix with an elliptic $R$-matrix, and therefore probes the genuinely CY-$3$ content of $\Phi_3$.
\end{remark}

\section{K3 twisted periods realise the Hodge structure on $H^\bullet_{\mathrm{CE}}(\mathfrak{g}_{\Delta_5})$}
\label{sec:dcy-k3-twisted-periods-ce}

The mixed Hodge structure on the Chevalley--Eilenberg cohomology
$H^\bullet_{\mathrm{CE}}(\mathfrak{g}_{\Delta_5})$
(Chapter~\textup{\ref{ch:cy-categories}}
\S\textup{\ref{sec:cycat-ce-hodge-delta5}},
Theorem~\textup{\ref{thm:cycat-ce-hodge-real-imag}}) admits an explicit
realisation in terms of twisted K3 period integrals of
$\phi_{0,1}^{\mathrm{K3}}$ against $\omega_{\mathrm{K3}}^k$. This section
makes the period-theoretic data concrete for $D^b(\Coh(K3))$-input via
$\Phi_2$, and records the resulting Hodge-theoretic invariants at
the level of derived-category evaluations.

\begin{remark}[K3 period map and Mukai $\mathrm{II}_{4,20}$]
\label{rem:dcy-k3-period-map-mukai}
\index{K3 period map!Mukai extension}
The classical K3 period map
$\mathrm{Per}: H^2(\mathrm{K3}, \mathbb{Z}) \to \mathbb{C}$ sends a
2-cycle class $\gamma$ to $\int_\gamma \omega_{\mathrm{K3}}$, where
$\omega_{\mathrm{K3}} \in H^{2,0}(\mathrm{K3})$ is the K3 holomorphic
2-form (unique up to $\mathbb{C}^*$-scale). Extending to the Mukai
lattice $\mathrm{II}_{4,20} = H^0(\mathrm{K3}) \oplus H^2(\mathrm{K3})
\oplus H^4(\mathrm{K3})$, the Mukai period map gives a
\emph{weight-$0$ Hodge structure}
\[
 H^\bullet_{\mathrm{Muk}}(\mathrm{K3}) \otimes \mathbb{C}
 \;=\; H^{(2,-2)}_{\mathrm{Muk}} \oplus H^{(1,-1)}_{\mathrm{Muk}}
  \oplus H^{(0,0)}_{\mathrm{Muk}} \oplus H^{(-1,1)}_{\mathrm{Muk}}
  \oplus H^{(-2,2)}_{\mathrm{Muk}}
\]
with $H^{(2,-2)}_{\mathrm{Muk}} = H^{2,0}(\mathrm{K3})$ (rank 1,
contains $\omega_{\mathrm{K3}}$), $H^{(0,0)}_{\mathrm{Muk}} = H^{1,1}
\oplus H^0 \oplus H^4$ (rank 22), and complex conjugates. This is the
input to the $\Phi_2$-composition: $D^b(\Coh(K3)) \xmapsto{\Phi_2}
H_{\mathrm{Muk}}$ is the rank-24 Heisenberg lattice VOA carrying the
Mukai weight-0 Hodge structure (Example~\textup{\ref{ex:k3-derived}}).
The pullback along $\Phi_2$ of the Mukai Hodge structure recovers
the Hodge filtration on $H^\bullet_{\mathrm{CE}}(\mathfrak{g}_{\Delta_5})$
via the chiral Hall-Drinfeld double
$\mathbf{H}_{\Delta_5} = \mathcal{D}_\hbar(\mathcal{Y}^{\mathrm{Hall}}_\hbar(\mathrm{CoHA}_{K3}))$.
\end{remark}

\begin{theorem}[Twisted period integrals realise MHS on imaginary-root CE cohomology]
\label{thm:dcy-k3-twisted-periods-realise-mhs}
\ClaimStatusProvedHere{Combines
Theorem~\textup{\ref{thm:cycat-ce-hodge-real-imag}} with Bruinier's
singular-theta Hodge decomposition LNM~$1780$ \S~$2.2$; the
period-realisation statement first appears in the present chapter.}
For each imaginary root $\alpha \in \Delta_+^{\mathrm{im}}$ of
$\mathfrak{g}_{\Delta_5}$ and each $k \geq 1$, the graded piece
$\mathrm{gr}^W_{2k} \mathrm{gr}^F_p H^k_{\mathrm{CE},\alpha}
(\mathfrak{g}_{\Delta_5})$ admits a basis of \emph{motivic periods}
given by iterated K3 twisted period integrals:
\[
 \pi_\alpha^{(p, k-p)} \;=\;
 \int_{\gamma_\alpha^{\otimes k}} \bigl(\phi_{0,1}^{\mathrm{K3}}\bigr)^{\otimes k}
 \cdot \omega_{\mathrm{K3}}^{\otimes p} \cdot \overline{\omega}_{\mathrm{K3}}^{\otimes (k-p)}
 \;\in\; \mathbb{C}/\mathbb{Z}(k),
\]
with $\gamma_\alpha \in H_2(\mathrm{K3}, \mathbb{Z})$ the
$\alpha$-specific Mukai 2-cycle of
Remark~\textup{\ref{rem:cycat-ce-twisted-periods}}, and
$\mathbb{Z}(k) = (2\pi i)^k \mathbb{Z}$. The assembly
\[
 \bigl\{\pi_\alpha^{(p, k-p)}\bigr\}_{\alpha \in \Delta_+^{\mathrm{im}},\ 0 \leq p \leq k}
 \;\in\; \mathbb{C}^{\sum_\alpha (k+1)}
\]
is the full period matrix of $H^k_{\mathrm{CE},\alpha}(\mathfrak{g}_{\Delta_5})$
on the imaginary-root sector, and its entries generate a sub-ring of
$\mathbb{C}$ of transcendence degree bounded by $\binom{k+1}{2}
\cdot \max_{D \leq 4nm-\ell^2} c_{\mathrm{K3}}(D)$ over $\mathbb{Q}$.
\end{theorem}

\begin{remark}[Mukai $\mathrm{II}_{4,20}$ weight-0 Hodge and Deligne
period conductors]
\label{rem:dcy-mukai-hodge-conductor}
The Mukai lattice $\mathrm{II}_{4,20}$ carries a canonical weight-$0$
Hodge structure (Mukai 1984). Its Hodge numbers are
$h^{(2,-2)} = h^{(-2,2)} = 1$, $h^{(0,0)} = 22$, with the middle
summand further split into $\mathrm{algebraic}\oplus\mathrm{transcendental}$
by the Picard rank $\rho(X) \in \{1, \ldots, 20\}$. Under the Mukai
pairing (Proposition~\textup{\ref{prop:serre-duality-cy}}), the
transcendental lattice
$T(X) = \mathrm{Pic}(X)^\perp \cap H^2(X,\mathbb{Z})$ carries a pure
Hodge structure of type $(2,0) + (1,1) + (0,2)$ with ranks
$(1, 20 - \rho, 1)$. For $\rho = 1$ (generic K3), this is rank 21
with full Hodge structure. This weight-$0$ Mukai Hodge structure
matches the CE-cohomology Hodge filtration of
Definition~\textup{\ref{def:cycat-ce-hodge-filtration}}:
the Borcherds singular-theta lift intertwines the two.

The period conductor of the Mukai lattice --- in the sense of
Deligne (1971, \emph{Inst.\ Hautes \'Etudes Sci.}~40, Hodge II
Thm~$2.3.5$) --- is the determinant of the period matrix
$\{\pi_{ij}\}$ for a basis of transcendental classes. For generic
K3, this is a transcendental number whose log is proportional to
the special-value $L(H^2(\mathrm{K3}), s)|_{s=1}$ (Beilinson
conjecture, level of $L$-function on K3 motive). The Mukai
extension multiplies by $(2\pi i)^2$ from the $H^0 \oplus H^4$
factor, giving the $\mathbb{Z}(2)$-Tate twist visible in the top
$H^4_{\mathrm{CE}}$ summand of Example~\textup{\ref{ex:hh-k3}}
(the $\HH^4 \simeq k^1$ summand).
\end{remark}

\begin{proposition}[Hodge-filtered structure of $\Phi_2(D^b(\Coh(K3)))$]
\label{prop:dcy-phi-k3-hodge-filtered}
\ClaimStatusProvedHere{Follows from
Theorem~\textup{\ref{thm:cycat-ce-hodge-real-imag}} via the functoriality
of $\Phi_2$ in Hodge data: cf.\ Example~\textup{\ref{ex:k3-derived}}
and Chapter~\textup{\ref{ch:cy-to-chiral}}.}
The chiral algebra $\Phi_2(D^b(\Coh(K3))) = H_{\mathrm{Muk}}$
(rank-24 Mukai Heisenberg VOA) carries a \emph{Hodge-filtered
structure} compatible with the mixed Hodge structure on
$H^\bullet_{\mathrm{CE}}(\mathfrak{g}_{\Delta_5})$ via the
embedding $H_{\mathrm{Muk}} \hookrightarrow \mathbf{H}_{\Delta_5}$
of the Hall-Drinfeld double (Theorem~\textup{\ref{thm:dcy-beilinson-dictionary}}):
\begin{enumerate}[label=(\alph*)]
 \item The 24 Heisenberg generators of $H_{\mathrm{Muk}}$ split by
  the Mukai Hodge type: 2 generators of type $(2,-2) \oplus (-2,2)$
  (from $H^{2,0} \oplus H^{0,2}$), 22 generators of type $(0,0)$
  (from $H^{1,1} \oplus H^0 \oplus H^4$).
 \item Under $\Phi_2$, the type-$(2,-2)$ generator $\omega_{\mathrm{K3}}$
  maps to the holomorphic-half supercharge in the extended K3
  $\mathcal{N}=4$ superconformal avatar (outside the image of
  $\Phi_2$ proper, but algebraically reachable via the Hall-Drinfeld
  double).
 \item The chiral modular characteristic $\kappa_{\mathrm{ch}}(H_{\mathrm{Muk}})
  = 2 = \chi(\cO_{K3})$ (Proposition~\textup{\ref{prop:kappa-cat-chi-cy}})
  is Hodge-invariant: it counts the super-Euler-characteristic of the
  weight-$0$ Mukai Hodge structure's pure pieces, i.e.\ the alternating
  sum of the invariants of the Hodge-filtered $\Phi_2$-output.
\end{enumerate}
\end{proposition}

\begin{remark}[Comparison: K3 Hodge-periodicity vs.\ CY$_3$ $K3 \times E$ period data]
\label{rem:dcy-k3-vs-k3e-hodge-comparison}
\index{Hodge structure!K3 vs.\ $K3 \times E$}
On the K3 side alone (CY$_2$), the Mukai weight-$0$ Hodge structure
controls the period data of Theorem~\textup{\ref{thm:dcy-k3-twisted-periods-realise-mhs}}
on $H^\bullet_{\mathrm{CE}}(\mathfrak{g}_{\Delta_5})$. On the
$K3 \times E$ product (CY$_3$), the K\"unneth decomposition
$H^\bullet(K3 \times E) = H^\bullet(K3) \otimes H^\bullet(E)$
introduces an additional elliptic-period factor
$\omega_E \in H^{1,0}(E)$ with $\int_E \omega_E = 2\pi i \tau$
($\tau$ the elliptic modulus). The resulting Hodge structure on
$H^\bullet_{\mathrm{CE}}(\mathfrak{g}_{\Delta_5}^{K3\times E})$ has:
\begin{itemize}
 \item Weight filtration: product of K3 Mukai $W_\bullet$ and
  elliptic $W_\bullet^E$ (shifts by 1 from the elliptic differential).
 \item Hodge filtration: Hodge types $(p_K + p_E, q_K + q_E)$ for
  $(p_K, q_K) \in \{(2,-2), (0,0), (-2,2)\}$ (Mukai) and
  $(p_E, q_E) \in \{(1,0), (0,0), (0,1)\}$ (elliptic).
 \item Mixed Tate piece: the K3 $\zeta(n)$-spectrum of
  Theorem~\textup{\ref{thm:cycat-ce-mixed-tate}} tensors with the
  elliptic quasi-modular $\zeta_E(s)$-sector, producing
  \emph{elliptic multiple zeta values} of Enriquez (Enriquez 2015
  \emph{Selecta Math.}~21 iterative elliptic polylogarithms) at
  weight $\geq 3$ in the $\mathrm{gr}^W$-graded.
\end{itemize}
The CY-3 extension is the motivic lift of the Borcherds-weight
upgrade $\kappa_{\mathrm{BKM}}(K3 \times E) = 5$
(Remark~\textup{\ref{rem:kappa-cat-from-chi}} of
Chapter~\textup{\ref{ch:cy-categories}}): the 5 distinct Hodge-types
of $H^\bullet(K3 \times E)$ in the Mukai extension each contribute
to the imaginary-cone multiplicity, and the weight filtration on
the resulting CE cohomology has 5 non-trivial graded pieces in
total. Primary: Enriquez 2015 \emph{Selecta Math.}~21;
Brown 2013 \emph{Math.\ Ann.}~357 (elliptic motives).
\end{remark}
