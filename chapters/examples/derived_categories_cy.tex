\chapter{Derived Categories of CY Manifolds}
\label{ch:derived-cy}

Homological mirror symmetry identifies $\Fuk(X) \simeq D^b(\Coh(X^\vee))$. Under the correspondence $\Phi_d$, this becomes an equivalence $A_{\Fuk(X)} \simeq A_{D^b(\Coh(X^\vee))}$ of quantum chiral algebras. The algebraic side requires the CY structure on $D^b(\Coh(X))$, the chiral algebra extracted by $\Phi_d$, exceptional collections, and the CoHA route for local CY threefolds.

\section{$D^b(\Coh(X))$ as a CY$_d$ category}
\label{sec:db-coh-cy}

Let $X$ be a smooth projective Calabi--Yau manifold of complex dimension $d$, so that the canonical bundle $\omega_X$ is trivial. The bounded derived category $D^b(\Coh(X))$ admits a canonical dg-enhancement, smooth and proper over the ground field. The Serre functor is
\[
 S_X \;=\; (-) \otimes \omega_X [d] \;=\; (-)[d],
\]
the shift by $d$.

\begin{definition}[CY$_d$ category]
\label{def:cy-d-category}
A smooth proper dg-category $\cC$ over $k$ is \emph{Calabi--Yau of dimension $d$} if there exists a quasi-isomorphism of $\cC$-bimodules
\[
 \cC^\vee \;\simeq\; \cC[d],
\]
where $\cC^\vee = \RHom_{\cC\text{-}\cC}(\cC, \cC \otimes \cC)$ is the inverse dualizing bimodule (Kontsevich 2006; Ginzburg 2006). Equivalently, the Serre functor satisfies $S_\cC \simeq [d]$ as an autoequivalence of the homotopy category $H^0(\cC)$. The CY structure is additional data (the bimodule quasi-isomorphism), not merely a property: different choices of trivialization yield different CY structures. For $\cC = D^b(\Coh(X))$ with $X$ a smooth projective variety, $S_X \simeq (-) \otimes \omega_X[d]$, and the CY condition becomes $\omega_X \simeq \cO_X$.
\end{definition}

The identification $S_X \simeq [d]$ for $D^b(\Coh(X))$ carries two consequences for the Vol~III programme. First, it provides the $S^d$-framing data required by the CY-to-chiral correspondence $\Phi_d$ (the CY structure determines a class in $\HC^-_d(\cC)$, the negative cyclic homology that refines the Hochschild trace;). Second, it constrains the Serre duality pairing that governs the $R$-matrix on the B-side.

\begin{proposition}[Serre duality on $D^b(\Coh(X))$ for CY $X$]
\label{prop:serre-duality-cy}
\ClaimStatusProvedElsewhere
Let $X$ be a smooth projective Calabi--Yau $d$-fold. For $\cE, \cF \in D^b(\Coh(X))$, Serre duality takes the form
\[
 \Ext^i(\cE, \cF) \;\simeq\; \Ext^{d-i}(\cF, \cE)^\vee,
\]
a non-degenerate pairing without any twist by $\omega_X$ (since $\omega_X \simeq \cO_X$). At the level of Hochschild cohomology, this induces a degree $(-d)$ symplectic pairing on $\HH^\bullet(\cC)$, the Mukai pairing (Caldararu 2003; Shklyarov 2013 for the dg-enhancement). The Mukai pairing is the B-side origin of the Ext-residue $R$-matrix: the classical $r$-matrix of the chiral algebra $\Phi(\cC)$ arises from the composition of Serre duality with the Grothendieck residue on $X$.
\end{proposition}

\begin{theorem}[Hochschild--Kostant--Rosenberg for smooth projective CY]
\label{thm:hkr-cy}
\ClaimStatusProvedElsewhere
For $X$ smooth projective of dimension $d$, there is an isomorphism of graded vector spaces
\[
 \HH^\bullet(D^b(\Coh(X))) \;\simeq\; \bigoplus_{p+q = \bullet} H^q\bigl(X, \Lambda^p T_X\bigr),
\]
intertwining the cup product on the left with the wedge product on polyvector fields on the right (Kontsevich 2003; Caldararu 2005 for the compatibility with the Hochschild--Kostant--Rosenberg isomorphism on chains). The Hochschild cohomology carries a canonical shifted Lie bracket (the Gerstenhaber bracket) which HKR sends to the Schouten--Nijenhuis bracket on polyvector fields.
\end{theorem}

For CY $X$, triviality of $\omega_X$ lets us identify polyvector fields with differential forms via the holomorphic volume form $\mathrm{vol}_X$, yielding
\[
 \HH^\bullet(D^b(\Coh(X))) \;\simeq\; \bigoplus_{p+q = \bullet} H^q(X, \Omega^{d-p}_X).
\]
The Schouten--Nijenhuis bracket transports to a degree $(-1)$ Lie bracket on Hodge cohomology. This is the input data for the Vol~III correspondence $\Phi_d$: its output is a chiral algebra whose generating fields are in bijection with $\HH^{\bullet+1}(\cC)$ (Theorem~\ref{thm:cy-to-chiral}), and whose modular characteristic equals the CY Euler characteristic at $d = 2$ (Proposition~\ref{prop:cy-kappa-d2}; conjectured at $d \geq 3$, Conjecture~\ref{conj:cy-kappa-identification}).

\begin{example}[K3 surfaces, $d = 2$]
\label{ex:k3-derived}
For a K3 surface $X$, the Kontsevich HKR sum $\HH^p = \bigoplus_{q+r=p} H^q(X, \wedge^r T_X)$, with $\wedge^r T_X \simeq \Omega^{2-r}_X$ on a CY$_2$, gives (Example~\ref{ex:hh-k3} of Chapter~\ref{ch:cy-categories})
\[
 \HH^0 \simeq H^0(\cO_X) \simeq k, \qquad
 \HH^1 \simeq 0, \qquad
 \HH^2 \simeq H^0(\wedge^2 T_X) \oplus H^1(T_X) \oplus H^2(\cO_X) \simeq k^{22},
\]
\[
 \HH^3 \simeq 0, \qquad \HH^4 \simeq H^2(\wedge^2 T_X) \simeq k,
\]
with $\dim_k \HH^\bullet = 1 + 0 + 22 + 0 + 1 = 24$ recovering the Mukai lattice rank / topological Euler characteristic. The CY$_2$ structure $S_X = [2]$ supplies the cyclic trace $\HH_2(D^b(\Coh(K3))) \to k$ dual to this decomposition. Under $\Phi$, Theorem~CY-A$_2$ produces the Mukai Heisenberg chiral algebra
\[
 \Phi\bigl(D^b(\Coh(K3))\bigr) \;=\; H_{\mathrm{Muk}}
\]
(Theorem~\ref{thm:phi-k3-explicit}): the rank-$24$ free-boson lattice VOA built from the Mukai lattice of signature $(4,20)$. The $\cN=4$ superconformal algebra at $c = 6$ is a distinct chiral algebra — the K3 sigma-model worldsheet algebra — that happens to share $\kappa_{\mathrm{ch}} = 2$ but is not in the image of $\Phi$. The categorical modular characteristic is
\[
 \kappa_{\mathrm{cat}}\bigl(D^b(\Coh(K3))\bigr) \;=\; \chi(\cO_{K3}) \;=\; 2,
\]
a manifold invariant determined by the Hodge data of K3 alone, independent of any algebraization. At $d = 2$ with $h^{1,0}(K3) = 0$, the algebraisation invariant $\kappa_{\mathrm{ch}}$ coincides: $\kappa_{\mathrm{ch}}(H_{\mathrm{Muk}}) = 2$ (Proposition~\ref{prop:kappa-cat-chi-cy}).
\end{example}

\begin{example}[Quintic threefold, $d = 3$]
\label{ex:quintic-derived}
The smooth quintic $X_5 \subset \P^4$ is a CY$_3$ with $h^{1,1} = 1$ and $h^{2,1} = 101$. The Kontsevich HKR sum $\HH^p = \bigoplus_{q+r=p} H^q(X_5, \wedge^r T_{X_5})$, with $\wedge^r T \cong \Omega^{3-r}$ on a CY$_3$, gives
\[
 \HH^2(X_5) \simeq H^1(X_5, T_{X_5}) \;\text{of dimension } 101 \;\text{(Kodaira--Spencer)}, \qquad
 \HH^3(X_5) \simeq k^4 \;\text{(top center, containing the Yukawa generator)}.
\]
$\HH^1(X_5) = H^0(T_{X_5}) \oplus H^1(\cO_{X_5}) = 0$ vanishes by simple-connectedness.
The CY$_3$ structure $S_{X_5} = [3]$ is the Serre functor used by Bridgeland in his construction of stability conditions. The chiral algebra $A_{X_5} = \Phi_3(D^b(\Coh(X_5)))$ is constructed by Theorem~CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}).
\end{example}

\begin{example}[CY$_4$ and SYZ fibrations]
\label{ex:cy4-syz}
For a CY $4$-fold $X$ admitting a Strominger--Yau--Zaslow special Lagrangian $T^4$-fibration $\pi \colon X \to B$, the dual fibration $\pi^\vee \colon X^\vee \to B$ is again a CY $4$-fold and mirror to $X$ (Strominger--Yau--Zaslow 1996; conjectural in general, known for hyperK\"ahler families). On the derived side, $D^b(\Coh(X))$ is CY$_4$ with $S_X = [4]$. Theorem~CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}) covers $d = 3$; the Vol~III programme at $d = 4$ requires a separate CY-A$_4$ construction.
\end{example}

\section{Homological mirror symmetry as chiral equivalence}
\label{sec:hms-chiral-derived}

Kontsevich's homological mirror symmetry conjecture (ICM 1994) predicts that mirror CY $d$-folds have equivalent derived categories, swapping the A-side and the B-side.

\begin{conjecture}[Homological mirror symmetry; Kontsevich 1994]
\label{conj:hms}
\ClaimStatusConjectured{}
For a mirror pair $(X, X^\vee)$ of smooth projective Calabi--Yau $d$-folds, there is an equivalence of CY$_d$ categories
\[
 D^b(\Coh(X)) \;\simeq\; D^\pi\Fuk(X^\vee),
\]
where $D^\pi\Fuk(X^\vee)$ denotes the split-closed derived Fukaya category (Kontsevich 1994). The equivalence intertwines the Serre functors, identifying the shifted polyvector fields on $X$ with the Floer cohomology of Lagrangians in $X^\vee$.
\end{conjecture}

The conjecture is known in the following cases.

\begin{theorem}[HMS for elliptic curves; Polishchuk--Zaslow 1998]
\label{thm:hms-d1}
\ClaimStatusProvedElsewhere
For $X = E_\tau$ an elliptic curve and $X^\vee = E_{-1/\tau}$ the mirror, there is an equivalence $D^b(\Coh(E_\tau)) \simeq D^\pi\Fuk(E_{-1/\tau})$ as CY$_1$ categories. The equivalence matches line bundles of degree $d$ with straight-line Lagrangians of slope $d$.
\end{theorem}

\begin{theorem}[HMS for K3 surfaces; Bridgeland, Sheridan--Smith]
\label{thm:hms-d2}
\ClaimStatusProvedElsewhere
For algebraic K3 surfaces in mirror families, homological mirror symmetry holds at the level of split-closed derived categories (Bridgeland 2008 for autoequivalence groups; Sheridan--Smith 2021 for mirror pairs inside hyperK\"ahler families). In particular, the equivalence of CY$_2$ categories is proved on all mirror families of polarized K3 surfaces with Picard rank at least $1$.
\end{theorem}

\begin{theorem}[HMS for the quintic threefold; Sheridan 2015]
\label{thm:hms-quintic}
\ClaimStatusProvedElsewhere
For $X_5$ the smooth quintic threefold and $X_5^\vee$ its Greene--Plesser mirror, there is a quasi-equivalence $D^\pi\Fuk(X_5) \simeq D^b(\Coh(X_5^\vee))$ of CY$_3$ categories (Sheridan 2015, building on Seidel's approach for Fano hypersurfaces).
\end{theorem}

The Vol~III reading is that HMS is a statement about the source of the correspondence $\Phi_d$. Applying $\Phi_d$ to both sides yields the chiral restatement.

\begin{conjecture}[HMS as chiral equivalence]
\label{conj:hms-chiral}
\ClaimStatusConjectured{}
Let $(X, X^\vee)$ be a mirror pair of CY $d$-folds for which Conjecture~\ref{conj:hms} holds. The correspondence $\Phi_d$ is defined for $d = 2$ (Theorem~CY-A$_2$) and $d = 3$ (Theorem~CY-A$_3$, Theorem~\ref{thm:cy-to-chiral-d3}). Then
\[
 \Phi\bigl(D^b(\Coh(X))\bigr) \;\simeq\; \Phi\bigl(D^\pi\Fuk(X^\vee)\bigr)
\]
as $\Etwo$-chiral algebras (for $d = 2$) or $\Eone$-chiral algebras (for $d \geq 3$). In particular, the chiral modular characteristics agree:
\[
 \kappa_{\mathrm{ch}}\bigl(\Phi(D^b(\Coh(X)))\bigr) \;=\; \kappa_{\mathrm{ch}}\bigl(\Phi(D^\pi\Fuk(X^\vee))\bigr).
\]
\end{conjecture}

\noindent
The manifold invariant $\kappa_{\mathrm{cat}} = \chi(\cO_X)$ is topological and automatically preserved by any derived equivalence; the content of Conjecture~\ref{conj:hms-chiral} is the agreement of the \emph{algebraisation invariant} $\kappa_{\mathrm{ch}}$, whose construction depends on the CY-to-chiral correspondence programme $\{\Phi_d\}$.

The $\kappa_{\mathrm{ch}}$-equality is verified unconditionally at $d = 1$: for mirror elliptic curves $E_\tau, E_{-1/\tau}$, both sides produce the Heisenberg vertex algebra and $\kappa_{\mathrm{ch}} = 1$. At $d = 2$, the equality reduces via Theorem~CY-A$_2$ to the identification $\kappa_{\mathrm{ch}} = \chi(\cO_X)$ (Proposition~\ref{prop:kappa-cat-chi-cy}), and mirror K3 surfaces share the same underlying smooth manifold.  At $d = 3$, $\Phi_3$ is supplied by Theorem~CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}); the remaining content is the comparison of $\kappa_{\mathrm{ch}}$ across the HMS equivalence. The chiral restatement transports $R$-matrices between the A-side and the B-side. On the A-side, the $R$-matrix comes from Floer-theoretic intersection pairings; on the B-side, from Ext-pairings and the Grothendieck residue. The mirror map intertwines them.


\subsection{K3 mirror symmetry through chiral Koszul duality}
\label{subsec:k3-mirror-koszul}

A K3 surface is self-mirror in the derived sense: for a lattice-polarized K3 surface $(S, M)$ with $M \hookrightarrow \Lambda_{K3}$, the Dolgachev--Nikulin mirror is the K3 surface $S^\vee$ polarized by the orthogonal complement $M^\perp \hookrightarrow \Lambda_{K3}$ inside the K3 lattice $\Lambda_{K3} = U^3 \oplus E_8(-1)^2$ of signature $(3,19)$. At the level of derived categories, the HMS equivalence $D^b(\Coh(S)) \simeq D^\pi\Fuk(S^\vee)$ is a theorem (Theorem~\ref{thm:hms-d2}), and the CY-to-chiral correspondence $\Phi_2$ is unconditional at $d = 2$ (Theorem~CY-A$_2$). Applying $\Phi_2$ to both sides yields chiral algebras that must be equivalent by functoriality, but the mirror involution acts nontrivially on their internal structure in ways that illuminate the relationship between mirror symmetry and chiral Koszul duality.

\begin{proposition}[Mukai lattice decomposition under mirror involution]
\label{prop:mukai-mirror-decomposition}
\ClaimStatusProvedHere
The Mukai lattice $\widetilde{\Lambda}_{K3} = U \oplus \Lambda_{K3} \simeq U^4 \oplus E_8(-1)^2$ has signature $(4,20)$. For a lattice-polarized K3 surface $(S, M)$ with Picard lattice $\Pic(S) \simeq M$ of rank $\rho$, the Mukai lattice decomposes as
\[
 \widetilde{\Lambda}_{K3} \;=\; \widetilde{M} \oplus \widetilde{M}^\perp, \qquad
 \widetilde{M} = U \oplus M, \quad
 \widetilde{M}^\perp = U \oplus M^\perp,
\]
where $\widetilde{M}$ has signature $(2, \rho)$ and $\widetilde{M}^\perp$ has signature $(2, 20-\rho)$. The Dolgachev--Nikulin mirror involution acts by
\[
 \iota_{\mathrm{DN}} \colon \widetilde{\Lambda}_{K3} \to \widetilde{\Lambda}_{K3}, \qquad
 \iota_{\mathrm{DN}}\big|_{\widetilde{M}} = -\id, \quad
 \iota_{\mathrm{DN}}\big|_{\widetilde{M}^\perp} = +\id.
\]
In particular, $\iota_{\mathrm{DN}}$ preserves the full signature $(4,20)$ but exchanges the roles of the B-model sublattice $\widetilde{M}$ (algebraic cycles, Ext-pairings) and the A-model sublattice $\widetilde{M}^\perp$ (Lagrangian cycles, Floer pairings).
\end{proposition}

\begin{proof}
The Mukai lattice $\widetilde{\Lambda}_{K3} = H^0(S,\Z) \oplus H^2(S,\Z) \oplus H^4(S,\Z)$ with the Mukai pairing $\langle (r_1, c_1, s_1), (r_2, c_2, s_2) \rangle = c_1 \cdot c_2 - r_1 s_2 - r_2 s_1$ has signature $(4,20)$. The hyperbolic summand $U = H^0 \oplus H^4$ pairs the rank and degree; the remaining $H^2(S,\Z) \simeq \Lambda_{K3}$ has signature $(3,19)$. For $(S,M)$ with $\Pic(S) = M$ of rank $\rho$, the algebraic part is $\widetilde{M} = U \oplus M$ of signature $(2,\rho)$, and the transcendental part is $\widetilde{M}^\perp = U \oplus M^\perp$ of signature $(2, 20-\rho)$. The Dolgachev--Nikulin construction defines the mirror by $\Pic(S^\vee) = M^\perp$; the action $\iota_{\mathrm{DN}}|_{\widetilde{M}} = -\id$ reflects the sign change in the Mukai vector under the mirror functor $\Phi_{\mathrm{HMS}} \colon D^b(\Coh(S)) \xrightarrow{\sim} D^\pi\Fuk(S^\vee)$, which acts on K-theory as a Hodge-theoretic rotation exchanging algebraic and transcendental periods.
\end{proof}

\begin{theorem}[K3 chiral Koszul self-duality]
\label{thm:k3-chiral-koszul-selfdual}
\ClaimStatusProvedHere
Let $S$ be a K3 surface with chiral algebra $\cA_{K3} = \Phi(D^b(\Coh(S)))$ (Theorem~CY-A$_2$). Then:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item The Koszul dual $\cA_{K3}^!$ has $\kappa_{\mathrm{ch}}(\cA_{K3}^!) = -2$ and central charge $c(\cA_{K3}^!) = -6$, with the Koszul conductor $K = \kappa_{\mathrm{ch}}(\cA_{K3}) + \kappa_{\mathrm{ch}}(\cA_{K3}^!) = 2 + (-2) = 0$ on the free-field/KM branch \textup{(}Theorem~\textup{\ref{thm:cy-complementarity-d2}}, Proposition~\textup{\ref{prop:koszul-dual-k3})}.
 \item The mirror K3 surface $S^\vee$ satisfies $\kappa_{\mathrm{ch}}(\Phi(D^b(\Coh(S^\vee)))) = \kappa_{\mathrm{ch}}(\Phi(D^b(\Coh(S)))) = 2$, since $\chi(\cO_{S^\vee}) = \chi(\cO_S) = 2$.
 \item Consequently, the HMS equivalence $D^b(\Coh(S)) \simeq D^\pi\Fuk(S^\vee)$ does \textbf{not} correspond to chiral Koszul duality \textup{(}which would require $\kappa_{\mathrm{ch}}^! = -2 \neq 2 = \kappa_{\mathrm{ch}}(S^\vee)$\textup{)}. HMS is an equivalence of CY$_2$ \emph{categories} that maps to an isomorphism of chiral algebras; Koszul duality is a duality of \emph{chiral algebras} that maps $\kappa_{\mathrm{ch}} \mapsto -\kappa_{\mathrm{ch}}$ on the free-field branch.
 \item The Dolgachev--Nikulin involution $\iota_{\mathrm{DN}}$ acts on the Heisenberg chiral algebra $H_{\mathrm{Muk}}$ of rank~$24$ \textup{(}generated by the Mukai lattice\textup{)} as a lattice automorphism that negates the algebraic sublattice $\widetilde{M}$ and preserves the transcendental sublattice $\widetilde{M}^\perp$. This does \emph{not} change $\kappa_{\mathrm{ch}}$: the modular characteristic $\kappa_{\mathrm{ch}} = 2 = \chi(\cO_S)$ depends only on the rank and CY dimension, not on the lattice polarization.
\end{enumerate}
\end{theorem}

\begin{proof}
Item (i): the K3 chiral algebra lies on the free-field/KM branch (the generating fields come from the Heisenberg algebra of the Mukai lattice), so the Koszul conductor vanishes: $K = 0$ (Theorem~\ref{thm:cy-complementarity-d2}, C2$^{\mathrm{CY}}$). With $\kappa_{\mathrm{ch}}(\cA_{K3}) = 2$ (Proposition~\ref{prop:cy-kappa-d2}), this forces $\kappa_{\mathrm{ch}}(\cA_{K3}^!) = -2$. The central charge follows from $c = 6k_R$ with $k_R' = -k_R = -1$, giving $c' = -6$.

Item (ii): every K3 surface has $h^{0,0} = 1$, $h^{1,0} = 0$, $h^{2,0} = 1$, so $\chi(\cO_S) = 2$ independently of the complex structure and lattice polarization. The mirror $S^\vee$ is again a K3 surface with the same $\chi(\cO)$, so $\kappa_{\mathrm{ch}}(\Phi(S^\vee)) = 2$ by Proposition~\ref{prop:cy-kappa-d2}.

Item (iii): suppose HMS corresponded to Koszul duality, i.e.\ $\Phi(\Fuk(S^\vee)) \simeq \Phi(\Coh(S))^!$. Then $\kappa_{\mathrm{ch}}(\Phi(\Fuk(S^\vee))) = \kappa_{\mathrm{ch}}(\cA_{K3}^!) = -2$. But HMS gives $\Fuk(S^\vee) \simeq D^b(\Coh(S))$ as CY$_2$ categories, so $\Phi(\Fuk(S^\vee)) \simeq \Phi(\Coh(S))$ with $\kappa_{\mathrm{ch}} = 2$. The values $2 \neq -2$ are contradictory: HMS is an equivalence (preserving $\kappa_{\mathrm{ch}}$), not Koszul duality (negating $\kappa_{\mathrm{ch}}$ on the free-field branch).

Item (iv): the Dolgachev--Nikulin involution acts on $\widetilde{\Lambda}_{K3} \otimes \C$ as $\iota_{\mathrm{DN}} = \mathrm{diag}(-1_{\widetilde{M}}, +1_{\widetilde{M}^\perp})$. On the Heisenberg algebra $H_{\mathrm{Muk}} = \bigoplus_{i=1}^{24} H_{k_i}$ with levels $k_i = \pm 1$ determined by the Mukai pairing signature, the involution permutes generators but preserves the total level $\sum k_i$. The modular characteristic $\kappa_{\mathrm{ch}} = \mathrm{str}(\mathrm{id}_V) = \chi(\cO_S) = 2$ is invariant under all lattice automorphisms of $\widetilde{\Lambda}_{K3}$, since it depends only on $\dim V = 24$ and the conformal weights, not on the lattice embedding.
\end{proof}

\begin{remark}[HMS $\neq$ Koszul: the K3 diagnostic]
\label{rem:hms-neq-koszul-k3}
The K3 case is the sharpest diagnostic for distinguishing HMS from chiral Koszul duality. Both are involutions on the space of CY$_2$ categories; they differ in their action on the modular characteristic:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{lcc}
 \toprule
 \textbf{Operation} & \textbf{Action on $\kappa_{\mathrm{ch}}$} & \textbf{K3 value} \\
 \midrule
 HMS ($S \leftrightarrow S^\vee$) & preserves $\kappa_{\mathrm{ch}}$ & $2 \mapsto 2$ \\
 Koszul duality ($A \leftrightarrow A^!$) & $\kappa_{\mathrm{ch}} \mapsto -\kappa_{\mathrm{ch}}$ (free-field) & $2 \mapsto -2$ \\
 Birational flop ($X \leftrightarrow X^+$) & preserves $\kappa_{\mathrm{ch}}$ & $2 \mapsto 2$ \\
 \bottomrule
\end{tabular}
\end{center}
At $d = 3$, Conjecture~\ref{conj:cy-koszul-mirror} predicts $\Phi(\cC^\vee) \simeq \Phi(\cC)^!$, i.e.\ that mirror symmetry \emph{is} chiral Koszul duality. The mechanism is different: for CY$_3$ mirror pairs $(X, X^\vee)$, the Hodge numbers swap ($h^{1,1} \leftrightarrow h^{2,1}$) and the topological Euler characteristic changes sign ($\chi_{\mathrm{top}}(X^\vee) = 2(h^{2,1} - h^{1,1}) = -\chi_{\mathrm{top}}(X)$), so the conjectural $d = 3$ chiral invariants derived from $\chi_{\mathrm{top}}$ also flip sign, which is consistent with Koszul duality. For CY$_2$, the mirror K3 has the \emph{same} Hodge numbers ($h^{1,1} = 20$ for all K3), so no sign change occurs: HMS at $d = 2$ is an equivalence, not a duality. The distinction is dimensional: the CY$_2$ Serre functor $[2]$ is even and $\chi(\cO_S) = 1 - 0 + 1 = 2$ is mirror-symmetric. At $d = 3$, Serre duality forces $\chi(\cO_X) = \sum_q (-1)^q h^{0,q}(X) = 1 - 0 + 0 - 1 = 0$ \emph{identically} for every compact CY$_3$ (AP-CY34a); the sign-flipping invariant is the \emph{topological} Euler characteristic $\chi_{\mathrm{top}}$, not the holomorphic $\chi(\cO)$.
\end{remark}

\begin{remark}[What distinguishes $\Phi(D^b(\Coh(S)))$ from $\Phi(\Fuk(S^\vee))$]
\label{rem:ab-side-distinction}
If both $D^b(\Coh(S))$ and $\Fuk(S^\vee)$ are CY$_2$-equivalent (Theorem~\ref{thm:hms-d2}), and $\Phi$ sends equivalent categories to equivalent chiral algebras (Conjecture~\ref{conj:hms-chiral}), then $\Phi(D^b(\Coh(S))) \simeq \Phi(\Fuk(S^\vee))$ as $\Etwo$-chiral algebras. The distinction is not in the chiral algebra but in the \emph{choice of generators}: the B-model generators come from Ext-groups between coherent sheaves (polyvector fields via HKR), while the A-model generators come from Floer complexes between Lagrangian submanifolds. The mirror map is a change of basis on $\HH^\bullet(\cC) \simeq \C^{24}$ (identifying the $(4,20)$-decomposition of the Mukai lattice), not a change of algebra. For a lattice-polarized K3 $(S, M)$ of Picard rank $\rho$:
\begin{itemize}
 \item The B-model presentation has $\rho + 2$ ``algebraic'' generators (from $\widetilde{M}$) and $22 - \rho$ ``transcendental'' generators (from $\widetilde{M}^\perp$);
 \item The A-model presentation of the mirror $(S^\vee, M^\perp)$ has $22 - \rho$ ``algebraic'' generators and $\rho + 2$ ``transcendental'' generators;
 \item The total is $24$ generators in both cases, producing the same Heisenberg algebra $H_{\mathrm{Muk}}$ of rank~$24$.
\end{itemize}
The mirror map permutes the generators according to the Dolgachev--Nikulin involution $\iota_{\mathrm{DN}}$ (Proposition~\ref{prop:mukai-mirror-decomposition}), but the $\Etwo$-chiral algebra structure is invariant. This is an automorphism of $H_{\mathrm{Muk}}$, not a Koszul duality.
\end{remark}

\begin{remark}[The CY structure on $\Fuk(S)$ versus $D^b(\Coh(S))$]
\label{rem:cy-structure-ab-sides}
Both $\Fuk(S)$ and $D^b(\Coh(S))$ are CY$_2$ categories, but their CY structures have different geometric origins. On the B-side, the CY structure comes from the triviality of $\omega_S$: the Serre functor $S_X = (-) \otimes \omega_S[2] = [2]$ and the CY trace is the Grothendieck--Serre residue $\HH_2(D^b(\Coh(S))) \to \C$. On the A-side, the CY structure comes from the holomorphic symplectic form $\sigma \in H^{2,0}(S)$: the cyclic pairing on Floer complexes $\mathrm{CF}^\bullet(L_0, L_1) \otimes \mathrm{CF}^\bullet(L_1, L_0) \to \Lambda_{\mathrm{nov}}[-2]$ is the Poincar\'e duality on Lagrangian intersections (Proposition~\ref{prop:fukaya-cy}). The HMS equivalence identifies these two CY structures: the holomorphic volume form $\sigma$ on the A-side maps to the Serre functor trace on the B-side. The $S^2$-framing (required for Step~3 of the CY-to-chiral passage, \S\ref{sec:cyclic-to-chiral}) is unconditional on both sides at $d = 2$, so $\Phi$ applies to both. The output chiral algebras are isomorphic as $\Etwo$-chiral algebras; they are \emph{not} Koszul dual.
\end{remark}

\begin{remark}[K3 $\times$ E and the d=3 mirror problem]
\label{rem:k3e-mirror-problem}
\ClaimStatusConjectured
The CY$_3$ threefold $K3 \times E$ is \emph{not} self-mirror. The mirror of an elliptic curve $E_\tau$ is $E_{-1/\tau}$ (Polishchuk--Zaslow), which is isomorphic to $E_\tau$ only at the self-dual points $\tau = i$ and $\tau = e^{2\pi i/3}$. For a generic $\tau$, the mirror of $K3 \times E_\tau$ is $K3 \times E_{-1/\tau}$, another product CY$_3$ but with a \emph{different} elliptic factor. The Hodge numbers are symmetric: $h^{1,1}(K3 \times E) = 21 = h^{2,1}(K3 \times E)$ (since $h^{1,1}(K3) = 20$, $h^{1,0}(E) = 1$, and the K\"unneth formula gives $h^{1,1}(K3 \times E) = h^{1,1}(K3) + h^{1,0}(K3) \cdot h^{0,1}(E) + h^{0,0}(K3) \cdot h^{1,1}(E) = 20 + 0 + 1 = 21$, with $h^{2,1}$ equal by the same count). Consequently, $\chi(K3 \times E) = 2(h^{1,1} - h^{2,1}) = 0$, so the conjectural $d = 3$ kappa-sum $\kappa_{\mathrm{ch}}(X) + \kappa_{\mathrm{ch}}(X^\vee) = 0$ is trivially satisfied. The Heisenberg-level chiral modular characteristic $\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(K3 \times E) = 3$ by additivity (\S\ref{sec:k3e-euler-kappa}; Remark~\ref{rem:beauville-kappa-formula-subscript-split}), and $\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(K3 \times E^\vee) = 3$ by the same computation, so the Koszul sum $3 + 3 = 6 \neq 0$. This places $K3 \times E$ \emph{outside} the free-field Koszul class at $d = 3$, consistently with the nonzero Koszul conductor $K = \kappa_{\mathrm{BKM}} = 5$ identified in the holographic modular Koszul datum (Remark~\ref{rem:holo-koszul-k3xe}).
\end{remark}


\section{Exceptional collections, quivers, and local CY}
\label{sec:exceptional-quivers}

When $D^b(\Coh(X))$ admits a full exceptional collection, the category collapses to modules over a finite-dimensional quiver algebra. This is the tightest combinatorial description of $D^b(\Coh(X))$ and provides the simplest non-trivial inputs to the CY-to-chiral correspondence (restricted to \emph{local} CY geometries, since a compact variety with a full exceptional collection is Fano, hence NOT Calabi--Yau).

\begin{theorem}[Beilinson--Bondal for Fano varieties with full exceptional collections]
\label{thm:beilinson-bondal}
\ClaimStatusProvedElsewhere
Let $X$ be a smooth projective variety admitting a full exceptional collection $\langle E_1, \ldots, E_n \rangle$ in $D^b(\Coh(X))$. Let $A = \End\bigl(\bigoplus_{i=1}^n E_i\bigr)$, the endomorphism algebra of the tilting object. Then
\[
 \bR\Hom(\oplus E_i, -) \colon D^b(\Coh(X)) \;\simeq\; D^b(\mathrm{mod}\text{-}A)
\]
is a triangulated equivalence (Beilinson 1978 for $\P^n$; Bondal 1989 in general). The algebra $A$ is the path algebra of a quiver with relations, the \emph{Beilinson quiver} of $(X, \langle E_i \rangle)$.
\end{theorem}

Passing from a Fano $X$ to a local CY geometry is standard: the total space $K_X = \mathrm{Tot}(\omega_X)$ of the canonical bundle is a non-compact Calabi--Yau of dimension $d + 1$, where $d = \dim X$. The derived category $D^b(\Coh(K_X))$ is described by the Beilinson quiver of $X$ together with a superpotential determined by the CY structure.

\begin{proposition}[Local CY quiver with potential from Fano exceptional collection]
\label{prop:local-cy-quiver}
\ClaimStatusProvedElsewhere
For $X$ a smooth projective Fano variety of dimension $d$ with a full strong exceptional collection, the derived category of compactly supported coherent sheaves on $K_X$ is equivalent to the derived category of finite-dimensional modules over the Ginzburg dg-algebra of a quiver with potential $(Q_X, W_X)$:
\[
 D^b_{cs}(\Coh(K_X)) \;\simeq\; D^b\bigl(\mathrm{mod}\text{-}\cG(Q_X, W_X)\bigr)
\]
(Bridgeland--King--Reid 2001 for McKay, Bocklandt 2008 for toric; Ginzburg 2006 for the dg-algebra formulation). The total space $K_X$ is a local CY$_{d+1}$, and $(Q_X, W_X)$ is its Beilinson quiver decorated with the superpotential.
\end{proposition}

\begin{example}[Local $\P^2$ and $K_{\P^2}$]
\label{ex:local-p2}
For $X = \P^2$, Beilinson's exceptional collection is $\langle \cO, \cO(1), \cO(2) \rangle$, with quiver
\[
 \bullet \rightrightarrows\!\!\rightarrow \bullet \rightrightarrows\!\!\rightarrow \bullet,
\]
three arrows between consecutive vertices. The local CY$_3$ total space $K_{\P^2}$ inherits a cubic superpotential. The critical CoHA $\cH(Q_{\P^2}, W_{\P^2})$ is an associative algebra isomorphic to a subalgebra of the affine Yangian $Y(\widehat{\fgl}_3)$. The chiral algebra $\Phi_3(D^b(\Coh(K_{\P^2})))$ produced by the CY-to-chiral correspondence is class~M (infinite shadow depth), not class~L: the $m_3$ generators do not close at finite degree.
\end{example}

\begin{example}[Conifold and the Klebanov--Witten quiver]
\label{ex:conifold-derived}
The resolved conifold $X = \cO_{\P^1}(-1) \oplus \cO_{\P^1}(-1)$ is a local CY$_3$ with toric diagram two triangles sharing an edge. Its Beilinson quiver is the Klebanov--Witten quiver (two vertices, two arrows in each direction) with quartic superpotential $W = \mathrm{tr}(A_1 B_1 A_2 B_2 - A_1 B_2 A_2 B_1)$. The critical CoHA is the positive half of the affine super Yangian $Y(\widehat{\mathfrak{gl}}(1|1))$; see \S\ref{sec:c3-yangian} for the parallel story on $\C^3$.
\end{example}

\begin{example}[$\C^3$ and the Jordan quiver]
\label{ex:c3-derived}
Affine space $X = \C^3$, viewed as the local CY$_3$ total space over a point, has Beilinson quiver the Jordan quiver with three loops and cubic superpotential $W = \mathrm{tr}(XYZ - XZY)$. The critical CoHA is $Y^+(\widehat{\fgl}_1)$ (Schiffmann--Vasserot 2013). The full affine Yangian is $\cW_{1+\infty}$ at the self-dual level (Proch\'azka--Rap\v{c}\'ak 2018), verifying the five-step functor chain of Chapter~\ref{ch:toric-coha}.
\end{example}

Across all three examples the pattern is the same: Beilinson quiver $\to$ superpotential $\to$ critical CoHA $\to$ positive half of an affine (super) Yangian $\to$ $\Eone$-sector of the Vol~III chiral algebra, via the CY-to-chiral correspondence $\Phi_3$ for toric CY$_3$ without compact $4$-cycles (Theorem~\ref{thm:rsyz}). The passage from $\Eone$ to $\Etwo$ requires the Drinfeld center, and is developed in Chapter~\ref{ch:drinfeld-center}. In every case the algebraization invariant $\kappa_{\mathrm{ch}}$ (computed intrinsically from the resulting chiral algebra) must be distinguished from the manifold invariant $\kappa_{\mathrm{cat}} = \chi(\cO_X)$ (holomorphic Euler characteristic, fixed by the geometry and independent of algebraization); agreement $\kappa_{\mathrm{ch}} = \kappa_{\mathrm{cat}}$ is a prediction of the functor, verified at $d = 2$ and conjectural at $d = 3$.

\section{Bridgeland stability conditions and wall-crossing}
\label{sec:bridgeland-stab}

The space of Bridgeland stability conditions $\Stab(\cC)$ on a CY$_d$ category $\cC$ is a complex manifold (Bridgeland 2007). Its chambers correspond to $t$-structures on $\cC$, and wall-crossing between chambers governs mutations of exceptional collections, DT invariant jumps, and (in the Vol~III reading) transitions between different $\Eone$-chart presentations of the chiral algebra.

\begin{definition}[Bridgeland stability condition]
\label{def:bridgeland-stability}
A \emph{Bridgeland stability condition} on a triangulated category $\cC$ is a pair $\sigma = (Z, \cP)$, where $Z \colon K_0(\cC) \to \C$ is a group homomorphism (the \emph{central charge}) and $\cP = \{\cP(\phi)\}_{\phi \in \bR}$ is a slicing of $\cC$ into full additive subcategories (the \emph{semistable objects of phase $\phi$}), subject to: (i) for $0 \neq E \in \cP(\phi)$, $Z(E) = m(E) \exp(i \pi \phi)$ with $m(E) > 0$; (ii) $\cP(\phi + 1) = \cP(\phi)[1]$; (iii) Harder--Narasimhan filtrations exist (Bridgeland 2007).
\end{definition}

For CY$_3$ categories the stability manifold can have many wall-crossing chambers. On K3 surfaces, Bridgeland (2008) proved that $\Stab(D^b(\Coh(X)))$ is a connected, simply connected complex manifold containing the complexified K\"ahler cone as an open subset. The autoequivalence group $\mathrm{Aut}(D^b(\Coh(X)))$ acts on $\Stab(X)$ by deck transformations, and the quotient is a period domain.

\begin{example}[Stability conditions on K3]
\label{ex:stab-k3}
For a K3 surface $X$ with Picard rank $\rho$, the distinguished connected component $\Stab^\dagger(X) \subset \Stab(X)$ is a $(\rho + 2)$-dimensional complex manifold. The central charge takes the form $Z_{\omega, B}(E) = -\int_X e^{-(B + i\omega)} \cdot \mathrm{ch}(E)$ for $\omega$ an ample class and $B \in H^{1,1}(X, \bR)$. Walls are real-codimension-one loci where a semistable object acquires a destabilizing subobject; crossing a wall exchanges the two hearts related by a tilt. Under the correspondence $\Phi_2$, different stability chambers yield different $\Eone$-chart presentations of $\cA_{K3}$, and the wall-crossing transformations correspond to $R$-matrix gauge equivalences between charts.
\end{example}

\begin{example}[Stability conditions on the resolved conifold]
\label{ex:stab-conifold}
For the resolved conifold $X = \cO_{\P^1}(-1)^{\oplus 2}$, the stability manifold $\Stab(D^b_{cs}(\Coh(X)))$ has a single wall at $\mathrm{Im}(Z(\cO_{\P^1}(-1))) = 0$, separating two chambers: the large-volume chamber (where $\cO_{\P^1}(-1)$ is stable) and the orbifold chamber (where the fractional branes are stable). The wall-crossing formula is the Kontsevich--Soibelman identity for the conifold DT invariants (Kontsevich--Soibelman 2008; Nagao--Nakajima 2011). On the chiral side, this single wall-crossing corresponds to the mutation relating the two Klebanov--Witten quiver presentations of Example~\ref{ex:conifold-derived}.
\end{example}

The general principle is that the stability manifold $\Stab(\cC)$ provides an atlas of $\Eone$-chart descriptions of the chiral algebra $\Phi(\cC)$; the transition functions are $R$-matrix gauge equivalences encoded by wall-crossing automorphisms. This perspective is developed in Chapter~\ref{ch:toric-coha} for toric CY$_3$ categories and is conjectural in general (conditional on CY-A$_3$).

\subsection{Wall-crossing as $R$-matrix gauge transformation}
\label{subsec:wallcrossing-rmatrix}

The dictionary between Bridgeland wall-crossing and $R$-matrix gauge transformations is the second functoriality clause U2 of the CY-to-chiral correspondence programme $\{\Phi_d\}$ (the first, U1, being the action on Morita equivalences; see Remark~\ref{rem:phi-functoriality-summary} below for the U1--U4 summary). Under $\Phi$, a Bridgeland slicing yields a $t$-structure on $\cC$, a $t$-structure yields a heart $\cC^\heartsuit$, the heart yields a tilting object (when one exists), the tilting object yields a quiver $Q_\sigma$, and the quiver gives an $\Eone$-chart of the chiral algebra with a specific generator basis. Two charts $\sigma_1, \sigma_2$ in adjacent chambers of $\Stab(\cC)$ produce two presentations $(Q_{\sigma_1}, R_{\sigma_1})$ and $(Q_{\sigma_2}, R_{\sigma_2})$ of the same chiral algebra; the wall-crossing automorphism is the $R$-matrix gauge transformation relating them.

\begin{conjecture}[Wall-crossing dictionary]
\label{conj:wallcrossing-dictionary}
\ClaimStatusConjectured
Let $\cC$ be a smooth proper CY$_d$ category for which $\Phi$ is defined ($d \in \{1, 2, 3\}$, the latter via Theorem~CY-A$_3$). Let $\sigma_1, \sigma_2 \in \Stab(\cC)$ lie in adjacent chambers separated by a single wall $W$, and let $\mathrm{KS}_W \in \mathrm{Aut}(\cC)$ be the Kontsevich--Soibelman wall-crossing automorphism. Then
\[
 \Phi(\mathrm{KS}_W) \;=\; g_W \in \mathrm{GL}_n\bigl(\Phi(\cC)\bigr),
\]
a gauge transformation of the chiral algebra realising the change of basis between the two $\Eone$-presentations. The classical $r$-matrix transforms by
\[
 r_{\sigma_2}(z) \;=\; g_W(z) \cdot r_{\sigma_1}(z) \cdot g_W(z)^{-1} \;-\; (\partial_z g_W)(z) \cdot g_W(z)^{-1},
\]
the standard gauge action on a meromorphic connection on the configuration space.
\end{conjecture}

\begin{remark}[Categorical content of the gauge action]
\label{rem:categorical-gauge}
The transformation rule in Conjecture~\ref{conj:wallcrossing-dictionary} is the classical gauge action of the loop group $G(\!(z)\!)$ on the space of $\mathfrak{g}$-valued meromorphic 1-forms on the punctured disk. The categorical lift comes from the fact that $\mathrm{KS}_W$ is a Fourier--Mukai-type autoequivalence of $\cC$ whose kernel changes the basis of $\HH^\bullet(\cC)$ by the wall-crossing matrix; transporting this through Theorem~\ref{thm:hkr-cy} gives a change of generator basis on the Hochschild side, and thence a change of $r$-matrix on the chiral side. The gauge transformation distinguishes the algebraisation invariant $\kappa_{\mathrm{ch}}$ (preserved under $\Phi(\mathrm{KS}_W)$, since gauge transformation acts trivially on the trace) from the manifold invariant $\kappa_{\mathrm{cat}}$ (preserved a fortiori).
\end{remark}

\begin{example}[Single-wall conifold wall-crossing as R-matrix gauge]
\label{ex:conifold-rmatrix-gauge}
For the resolved conifold of Example~\ref{ex:stab-conifold}, the single wall separates the large-volume chamber (geometric DT invariants) from the orbifold chamber (noncommutative DT invariants). The Kontsevich--Soibelman wall-crossing automorphism is
\[
 \mathrm{KS}_W \;=\; \mathrm{Ad}_{T_E}^{n_E} \quad\text{where } T_E = \prod_{m \in \Z} (1 - q^m e_E)^{(-1)^m},
\]
with $n_E$ the DT count of the wall-stable object $E = \cO_{\P^1}(-1)$ and $e_E$ the corresponding root vector in the affine super Yangian $Y(\widehat{\mathfrak{gl}}(1|1))$ (Example~\ref{ex:conifold-derived}). Under $\Phi_3$, this lifts to the gauge transformation $g_W(z) = T_E(z)$ of the chiral $r$-matrix; the two presentations of $\Phi_3(D^b_{cs}(\Coh(X)))$ are related by conjugation, and the modular characteristic is preserved on both sides ($\kappa_{\mathrm{ch}} = 0$ since the conifold has $\chi(\cO) = 0$).
\end{example}

The categorical content of wall-crossing is that the $R$-matrix is a connection on a flat bundle over $\Stab(\cC)$, and the holonomy of this connection around walls reproduces the Kontsevich--Soibelman wall-crossing formulae. This is the geometric counterpart of the fact that on the algebraic side, the $\Eone$-chiral algebra has a single isomorphism class but many quasi-equivalent presentations indexed by stability conditions.

\section{Birational geometry: flops and derived equivalences}
\label{sec:flops-derived}

A second source of derived autoequivalences is birational geometry. The Bondal--Orlov reconstruction theorem identifies smooth projective varieties with ample canonical or anti-canonical bundle from their derived categories, but for Calabi--Yau varieties the canonical bundle is trivial and birational CY varieties typically have equivalent derived categories. Flops are the prototypical example.

\begin{theorem}[Bridgeland's flop equivalence]
\label{thm:bridgeland-flop}
\ClaimStatusProvedElsewhere
Let $X \to Y \leftarrow X^+$ be a standard threefold flop between smooth projective Calabi--Yau threefolds (i.e.\ $X$ and $X^+$ are connected by contracting a $\P^1$ and re-resolving). Then there is a derived equivalence
\[
 D^b(\Coh(X)) \;\simeq\; D^b(\Coh(X^+))
\]
realised by a Fourier--Mukai functor with kernel the structure sheaf of the fibre product $X \times_Y X^+$ (Bridgeland 2002).
\end{theorem}

Under the correspondence $\Phi_3$, Theorem~\ref{thm:bridgeland-flop} produces a chiral equivalence: $\Phi_3(D^b(\Coh(X))) \simeq \Phi_3(D^b(\Coh(X^+)))$. The two presentations differ by an $R$-matrix gauge transformation analogous to wall-crossing (the flop corresponds to crossing a wall in $\Stab$).

\begin{example}[Atiyah flop]
\label{ex:atiyah-flop}
The Atiyah flop is the simplest non-trivial threefold flop: $X = \cO_{\P^1}(-1) \oplus \cO_{\P^1}(-1)$ (the resolved conifold of Examples~\ref{ex:conifold-derived} and~\ref{ex:stab-conifold}) flops to $X^+$, an isomorphic CY threefold with the orientation of the $\P^1$ reversed. The two are related by the Bridgeland equivalence, and on the chiral side by the gauge transformation of Example~\ref{ex:conifold-rmatrix-gauge}. In particular, $\kappa_{\mathrm{ch}}(\Phi_3(X)) = \kappa_{\mathrm{ch}}(\Phi_3(X^+)) = 0$ (both have $\chi(\cO) = 0$, per the resolved-conifold compactly-supported computation), confirming Conjecture~\ref{conj:wallcrossing-dictionary} in this birational instance.
\end{example}

\begin{remark}[McKay correspondence as derived equivalence]
\label{rem:mckay-derived}
\ClaimStatusProvedElsewhere
The McKay correspondence supplies a third source of derived equivalences. Bridgeland--King--Reid \cite{BridgelandKingReid2001} prove that for $\Gamma \subset \mathrm{SL}(d, \C)$ a finite subgroup with $d \leq 3$, there is an equivalence
\[
 D^b(\Coh^\Gamma(\C^d)) \;\simeq\; D^b(\Coh(Y)),
\]
where $Y \to \C^d/\Gamma$ is the canonical (crepant) resolution. The left side is the equivariant derived category, equivalent to the derived category of the orbifold $[\C^d/\Gamma]$. Under $\Phi$, the McKay equivalence becomes a chiral equivalence between the orbifold chiral algebra (constructed from the $\Gamma$-equivariant CoHA) and the chiral algebra of the resolution; this is the chiral analogue of the celebrated DHVW orbifold-resolution equivalence at the level of partition functions.
\end{remark}

\section{Functoriality and the Vol III programme}

The three sources of derived equivalences in this chapter -- HMS (mirror symmetry), Bridgeland wall-crossing, and birational geometry (flops, McKay) -- all factor through the CY-to-chiral correspondence $\Phi_d$ and produce chiral equivalences on the output side. This is the U1--U4 functoriality structure of $\Phi_d$ in operational form.

\begin{remark}[$\Phi$ functoriality summary]
\label{rem:phi-functoriality-summary}
The four universal properties of $\Phi$ (Theorem~\ref{thm:phi-platonic}) read in the derived-category setting:
\begin{itemize}
 \item \textbf{U1} (Morita equivalences): $\Phi$ sends Morita-equivalent CY$_d$ categories to isomorphic chiral algebras. The HMS equivalences (Theorems~\ref{thm:hms-d1}, \ref{thm:hms-d2}, \ref{thm:hms-quintic}) are instances.
 \item \textbf{U2} (gauge transformations): $\Phi$ sends Bridgeland wall-crossing to $R$-matrix gauge transformations (Conjecture~\ref{conj:wallcrossing-dictionary}). The Atiyah flop (Example~\ref{ex:atiyah-flop}) and McKay equivalence (Remark~\ref{rem:mckay-derived}) are derived consequences.
 \item \textbf{U3} (deformations): $\Phi$ converts deformations of the CY category (variations of complex structure, K\"ahler moduli) into deformations of the chiral algebra, the latter parametrising a flat bundle on $\Stab(\cC)$.
 \item \textbf{U4} (degenerations): $\Phi$ sends degenerations of CY$_d$ to degenerations of chiral algebras, including the orbifold/large-volume limits and the LG/CY correspondence (Chapter~\ref{ch:matrix-factorizations}, Section~\ref{sec:lg-cy-correspondence}).
\end{itemize}
The four clauses are the operational content of the slogan ``$\Phi$ is functorial and continuous on the moduli stack of CY$_d$ categories''. The Bridgeland stability manifold $\Stab(\cC)$ is the local model for U2 and U3; the Kontsevich--Soibelman wall-crossing formulae govern the holonomy of $\Phi$ along walls.
\end{remark}
