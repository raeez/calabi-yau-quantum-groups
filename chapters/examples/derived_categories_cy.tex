\chapter{Derived Categories of CY Manifolds}
\label{ch:derived-cy}

Homological mirror symmetry identifies $\Fuk(X) \simeq D^b(\Coh(X^\vee))$ at the categorical input layer, before $\PhiFA_d$. Under the two-stage factorisation $\Phi_d = \SpCh_{\Sigma_{d-1}, C} \circ \PhiFA_d$ of Theorem~\ref{thm:phi-two-stage-factorisation}, HMS produces an equivalence of Stage-$1$ $E_d$-holomorphic factorisation algebras when the relevant $\PhiFA_d$ hypotheses are witnessed:
\[
 \PhiFA_d(\Fuk(X)) \;\simeq\; \PhiFA_d(D^b(\Coh(X^\vee))).
\]
After a specified $(\Sigma_{d-1}, C)$ specialisation this gives isomorphic native outputs: $\Etwo$-chiral at $d=2$ and $\Eone$-chiral at $d\geq3$. The algebraic side requires the CY structure on $D^b(\Coh(X))$, the canonical $\PhiFA_d$-image on the CY$_d$ variety, its $\SpCh$-specialisation to a chiral algebra on a curve, exceptional collections, and the CoHA route for local CY threefolds.

\section{\texorpdfstring{$D^b(\Coh(X))$}{D-b(Coh(X))} as a CY\texorpdfstring{$_d$}{-d} category}
\label{sec:db-coh-cy}

Let $X$ be a smooth projective Calabi--Yau manifold of complex dimension $d$, so that the canonical bundle $\omega_X$ is trivial. The bounded derived category $D^b(\Coh(X))$ admits a canonical dg-enhancement, smooth and proper over the ground field. The Serre functor is
\[
 S_X \;=\; (-) \otimes \omega_X [d] \;=\; (-)[d],
\]
the shift by $d$.

\begin{definition}[CY\texorpdfstring{$_d$}{-d} category]
\label{def:cy-d-category}
A smooth proper dg-category $\cC$ over $k$ is \emph{Calabi--Yau of dimension $d$} if there exists a quasi-isomorphism of $\cC$-bimodules
\[
 \cC^\vee \;\simeq\; \cC[d],
\]
where $\cC^\vee = \RHom_{\cC\text{-}\cC}(\cC, \cC \otimes \cC)$ is the inverse dualizing bimodule (Kontsevich 2006; Ginzburg 2006). Equivalently, the Serre functor satisfies $S_\cC \simeq [d]$ as an autoequivalence of the homotopy category $H^0(\cC)$. The CY structure is additional data (the bimodule quasi-isomorphism), not merely a property: different choices of trivialization yield different CY structures. For $\cC = D^b(\Coh(X))$ with $X$ a smooth projective variety, $S_X \simeq (-) \otimes \omega_X[d]$, and the CY condition becomes $\omega_X \simeq \cO_X$.
\end{definition}

The identification $S_X \simeq [d]$ for $D^b(\Coh(X))$ carries two consequences for the CY-to-chiral correspondence. First, it provides the $S^d$-framing data required by $\Phi_d$ (the CY structure determines a class in $\HC^-_d(\cC)$, the negative cyclic homology that refines the Hochschild trace). Second, it constrains the Serre duality pairing that governs the $R$-matrix on the B-side.

\begin{proposition}[Serre duality on \texorpdfstring{$D^b(\Coh(X))$}{D-b(Coh(X))} for CY \texorpdfstring{$X$}{X}]
\label{prop:serre-duality-cy}
Let $X$ be a smooth projective Calabi--Yau $d$-fold. For $\cE, \cF \in D^b(\Coh(X))$, Serre duality takes the form
\[
 \Ext^i(\cE, \cF) \;\simeq\; \Ext^{d-i}(\cF, \cE)^\vee,
\]
a non-degenerate pairing without any twist by $\omega_X$ (since $\omega_X \simeq \cO_X$). At the level of Hochschild cohomology, this induces a degree $(-d)$ symplectic pairing on $\HH^\bullet(\cC)$, the Mukai pairing (Caldararu 2003; Shklyarov 2013 for the dg-enhancement). The Mukai pairing is the B-side origin of the Ext-residue $R$-matrix: the classical $r$-matrix of the chiral algebra $\Phi(\cC)$ arises from the composition of Serre duality with the Grothendieck residue on $X$.
\end{proposition}

\begin{theorem}[Hochschild--Kostant--Rosenberg for smooth projective CY]
\label{thm:hkr-cy}
For $X$ smooth projective of dimension $d$, there is an isomorphism of graded vector spaces
\[
 \HH^\bullet(D^b(\Coh(X))) \;\simeq\; \bigoplus_{p+q = \bullet} H^q\bigl(X, \Lambda^p T_X\bigr),
\]
intertwining the cup product on the left with the wedge product on polyvector fields on the right (Kontsevich 2003; Caldararu 2005 for the compatibility with the Hochschild--Kostant--Rosenberg isomorphism on chains). The Hochschild cohomology carries a canonical shifted Lie bracket (the Gerstenhaber bracket) which HKR sends to the Schouten--Nijenhuis bracket on polyvector fields.
\end{theorem}

For CY $X$, triviality of $\omega_X$ lets us identify polyvector fields with differential forms via the holomorphic volume form $\mathrm{vol}_X$, yielding
\[
 \HH^\bullet(D^b(\Coh(X))) \;\simeq\; \bigoplus_{p+q = \bullet} H^q(X, \Omega^{d-p}_X).
\]
The Schouten--Nijenhuis bracket transports to a degree $(-1)$ Lie bracket on Hodge cohomology. This is input for the Vol~III correspondence $\Phi_d$: on its witnessed object-level locus the output is a chiral algebra whose native operadic level is $\Etwo$ at $d=2$ and $\Eone$ at $d\geq3$. The generating fields are controlled by $\HH^{\bullet+1}(\cC)$ (Theorem~\ref{thm:cy-to-chiral}), and the modular characteristic equals the CY Euler characteristic at $d = 2$ (Proposition~\ref{prop:cy-kappa-d2}; conjectured at $d \geq 3$, Conjecture~\ref{conj:cy-kappa-identification}).

\begin{example}[K3 surfaces, \texorpdfstring{$d = 2$}{d = 2}]
\label{ex:k3-derived}
For a K3 surface $X$, the Kontsevich HKR sum $\HH^p = \bigoplus_{q+r=p} H^q(X, \wedge^r T_X)$, with $\wedge^r T_X \simeq \Omega^{2-r}_X$ on a CY$_2$, gives (Example~\ref{ex:hh-k3} of Chapter~\ref{ch:cy-categories})
\[
 \HH^0 \simeq H^0(\cO_X) \simeq k, \qquad
 \HH^1 \simeq 0, \qquad
 \HH^2 \simeq H^0(\wedge^2 T_X) \oplus H^1(T_X) \oplus H^2(\cO_X) \simeq k^{22},
\]
\[
 \HH^3 \simeq 0, \qquad \HH^4 \simeq H^2(\wedge^2 T_X) \simeq k,
\]
with $\dim_k \HH^\bullet = 1 + 0 + 22 + 0 + 1 = 24$ recovering the Mukai lattice rank / topological Euler characteristic. The CY$_2$ structure $S_X = [2]$ supplies the cyclic trace $\HH_2(D^b(\Coh(K3))) \to k$ dual to this decomposition. The two-stage factorisation of Theorem~\ref{thm:phi-two-stage-factorisation} acts on this datum in two steps. Stage~$1$ produces the canonical $E_2$-holomorphic factorisation algebra
\[
 \PhiFA_2\bigl(D^b(\Coh(K3))\bigr) \;\in\; \EdHolFA(K3)
\]
assembled from the Gerstenhaber bracket of degree $-1$ on $\HH^\bullet(\cC)$ via Kontsevich--Tamarkin formality and Costello--Gwilliam--Li locality on the $K3$ surface; it factorises over algebraic-surface opens (Francis--Gaitsgory). Stage~$2$ specialises along a $(d{-}1)$-dimensional Nakajima cycle $\Sigma_1^{\mathrm{Nakajima}} \subset K3$ and a reference curve $C \subset K3$ to the Mukai Heisenberg chiral algebra
\[
 \SpCh_{\Sigma_1^{\mathrm{Nakajima}},\, C}\bigl(\PhiFA_2(D^b(\Coh(K3)))\bigr) \;=\; H_{\mathrm{Muk}}
\]
(Theorem~\ref{thm:phi-k3-explicit}): the rank-$24$ free-boson lattice VOA built from the Mukai lattice of signature $(4,20)$. The $\cN=4$ superconformal algebra at $c = 6$ is a distinct chiral algebra -- the K3 sigma-model worldsheet algebra -- that happens to share $\kappa_{\mathrm{ch}} = 2$ but is not in the image of $\Phi$. The categorical modular characteristic is
\[
 \kappa_{\mathrm{cat}}\bigl(D^b(\Coh(K3))\bigr) \;=\; \chi(\cO_{K3}) \;=\; 2,
\]
a manifold invariant determined by the Hodge data of K3 alone, independent of any algebraization. At $d = 2$ with $h^{1,0}(K3) = 0$, the algebraisation invariant $\kappa_{\mathrm{ch}}$ coincides: $\kappa_{\mathrm{ch}}(H_{\mathrm{Muk}}) = 2$ (Proposition~\ref{prop:kappa-cat-chi-cy}).
\end{example}

\begin{example}[Quintic threefold, \texorpdfstring{$d = 3$}{d = 3}]
\label{ex:quintic-derived}
The smooth quintic $X_5 \subset \P^4$ is a CY$_3$ with $h^{1,1} = 1$ and $h^{2,1} = 101$. The Kontsevich HKR sum $\HH^p = \bigoplus_{q+r=p} H^q(X_5, \wedge^r T_{X_5})$, with $\wedge^r T \cong \Omega^{3-r}$ on a CY$_3$, gives
\[
 \HH^2(X_5) \simeq H^1(X_5, T_{X_5}) \;\text{of dimension } 101 \;\text{(Kodaira--Spencer)}, \qquad
 \HH^3(X_5) \simeq k^4 \;\text{(top center, containing the Yukawa generator)}.
\]
$\HH^1(X_5) = H^0(T_{X_5}) \oplus H^1(\cO_{X_5}) = 0$ vanishes by simple-connectedness.
The CY$_3$ structure $S_{X_5} = [3]$ is the Serre functor used by Bridgeland in his construction of stability conditions. For a specified specialisation datum $(\Sigma_2,C)$ in the framed locus of Theorem~CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}), the object-level output is the framed algebra $A_{X_5}^{(\Sigma_2,C)}=\Phi_3^{(\Sigma_2,C)}(D^b(\Coh(X_5)))$.
\end{example}

\subsection{Curved Kontsevich formality on the quintic}
\label{subsec:curved-formality-quintic}

The strict Kontsevich formality theorem (Kontsevich 2003, \textit{Lett.~Math.~Phys.}~66) asserts an $L_\infty$-quasi-isomorphism
$
 \HKR_\infty \colon T^{\mathrm{poly}}(X) \xrightarrow{\sim} \mathrm{HH}^\bullet(C^\infty(X))
$
between polyvector fields (with Schouten--Nijenhuis bracket) and Hochschild cochains (with Gerstenhaber bracket). Its first Taylor component is the Hochschild--Kostant--Rosenberg isomorphism; strict formality requires the higher components $\ell_k$ for $k \geq 2$ to vanish on cohomology, so that every Massey product is trivial.

On the quintic threefold $X_5 \subset \P^4$ this strict form is refuted. The degree-$3$ Massey product on $T^{\mathrm{poly}}(X_5)$ is the Yukawa coupling
\[
 Y_3 \colon H^1(T_{X_5})^{\otimes 3} \longrightarrow H^3(\wedge^3 T_{X_5}) \simeq k,
 \qquad
 Y_3(\theta, \theta, \theta) \;=\; \int_{X_5} \theta \wedge \theta \wedge \theta \wedge \Omega_{X_5}^{3,0} \;=\; H^3 \;=\; 5,
\]
where $H$ is the hyperplane class and $\deg X_5 = H^3 = 5$ equals the classical Yukawa coupling of the quintic at the large-volume point (Candelas--de la Ossa--Green--Parkes 1991; Bershadsky--Cecotti--Ooguri--Vafa 1994; Costello 2007 \textit{Renormalization and Effective Field Theory}, \S16). Strict formality would force $Y_3 = 0$; hence $\HKR_\infty$ is \emph{not} a strict $L_\infty$-quasi-isomorphism on~$X_5$.

\begin{definition}[Curved \texorpdfstring{$L_\infty$}{L-inf}-algebra]
\label{def:curved-linfty}
A \emph{curved $L_\infty$-algebra} $(L, \{\ell_k\}_{k \geq 0})$ is a graded vector space $L$ with operations $\ell_k \colon L^{\otimes k} \to L$ of degree $2-k$ for $k \geq 0$ satisfying the curved generalised Jacobi identity
\[
 \sum_{i+j = n+1} \sum_{\sigma \in \mathrm{Sh}(i, j-1)} (-1)^{i(j-1)} \chi(\sigma)\,
 \ell_j\bigl(\ell_i(x_{\sigma(1)}, \ldots, x_{\sigma(i)}), x_{\sigma(i+1)}, \ldots, x_{\sigma(n)}\bigr)
 \;=\; 0
 \qquad (n \geq 0),
\]
where $\ell_0 \in L^2$ is the \emph{curvature}. The $n = 0$ identity reads $\ell_1(\ell_0) = 0$, the $n = 1$ identity reads $\ell_1^2(x) = -\ell_2(\ell_0, x)$ (so $\ell_1$ is a differential only modulo $\ell_0$), and the classical Jacobi is recovered when $\ell_0 = 0$ (Dolgushev--Tamarkin--Tsygan 2007, \textit{J.~Noncommut.~Geom.}~1).
\end{definition}

\begin{theorem}[Yukawa obstruction to strict Kontsevich formality for the quintic]
\label{thm:curved-kontsevich-quintic}
Let $X_5 \subset \P^4$ be the smooth quintic threefold with holomorphic volume form
$\Omega_{X_5} \in H^0(\Omega^{3,0})$. The transferred minimal
$L_\infty$-model of the Kodaira--Spencer dg Lie algebra is not strict:
its cubic operation on $H^1(T_{X_5})$ is the large-volume Yukawa
coupling and has classical coefficient
\[
 Y_3(X_5)=\int_{X_5}H^3=5.
\]
Thus any compact curved strictification witness replacing the strict
HKR model must carry leading curvature
\[
 F^{X_5} \;=\; Y_3 \cdot [\Omega_{X_5}^{3,0}] \;=\; 5 \cdot [\Omega_{X_5}^{3,0}]
 \;\in\; T^{\mathrm{poly},2}(X_5),
\]
viewed via $\wedge^3T_{X_5}\simeq\mathcal O_{X_5}$ as a class in
$H^3(\mathcal O_{X_5})$. The existence of a global curved
$L_\infty$-quasi-isomorphism with explicit Taylor components is a
separate compact non-formal framing witness; it is not supplied by the
Hodge table, the CY trace, or the number $Y_3=5$ alone.
\end{theorem}

\begin{proof}[Evidence and obstruction data]
By HKR, the Kodaira--Spencer deformation complex identifies
$H^1(T_{X_5})$ with first-order complex-structure deformations. The
Kadeishvili transfer of the Kodaira--Spencer dg Lie algebra has cubic
operation given by the Yukawa coupling (Barannikov--Kontsevich 1998;
Bryant--Griffiths 1983; Voisin 2007 II, \S5.3). On the large-volume
mirror branch the classical Yukawa coefficient is the triple
intersection of the hyperplane class, hence $H^3=\deg(X_5)=5$ by
adjunction. A strict transferred model would force this cubic class to
vanish in $H^3(\mathcal O_{X_5})\simeq\mathbb C$, contradiction. The
displayed curvature is therefore the necessary leading term of any
curved replacement. Constructing the replacement requires the extra
compact framing data named in
Theorem~\ref{thm:cy-c-beyond-quintic-strictification-obstruction}.
\end{proof}

\begin{remark}[Strict vs.\ curved: the Yukawa is the obstruction]
\label{rem:strict-formality-fails-quintic}
Theorem~\ref{thm:curved-kontsevich-quintic} localises the failure of strict
Kontsevich formality to the scalar $Y_3=5$, the degree of the quintic.
This is the chain-level obstruction that any compact curved replacement
must absorb. The obstruction theorem is proved; the global curved
$L_\infty$ replacement is precisely the missing framing witness.
\end{remark}

\begin{theorem}[Genus-expansion via curved Kontsevich]
\label{thm:curved-bcov-genus}
Costello--Li 2012 constructs perturbative BV quantisation of BCOV theory
on compact $X_5$; assume in addition a chain-level $L_\infty$-transfer
of the $\hbar$-expansion to the compact curved minimal model.
Write $F_g^{\mathrm{BCOV}}(X_5) \in k[[t]]$ for the genus-$g$ free energy of BCOV theory on the quintic at the large-volume point (Bershadsky--Cecotti--Ooguri--Vafa 1994, \textit{Comm.~Math.~Phys.}~165; Costello--Li 2012, \textit{Quantum BCOV theory on Calabi--Yau manifolds and the higher genus B-model}). If the compact curved strictification witness required by Theorem~\ref{thm:curved-kontsevich-quintic} is supplied, it must quantise to data of the form
\[
 \bigl(T^{\mathrm{poly}}(X_5), \ell_k^{\mathrm{T}}(\hbar), F_{\mathrm{tot}}^{X_5}(\hbar)\bigr), \qquad
 F_{\mathrm{tot}}^{X_5}(\hbar) \;=\; \sum_{g \geq 0} \hbar^g \cdot F_g^{\mathrm{BCOV}}(X_5),
\]
in which $F_0^{\mathrm{BCOV}} = Y_3 \cdot [\Omega^{3,0}] = F^{X_5}$ is the genus-zero Yukawa of Theorem~\ref{thm:curved-kontsevich-quintic} and $F_g^{\mathrm{BCOV}}$ is the degree-$(3g-3)$ generator constrained by the BCOV holomorphic anomaly equation
$
 \bar\partial_{\bar\imath} F_g = \tfrac{1}{2} \bar{C}_{\bar\imath}^{jk} \bigl(D_j D_k F_{g-1} + \sum_{h=1}^{g-1} D_j F_h \, D_k F_{g-h}\bigr).
$
Costello--Li 2012 \S4--5 constructs the renormalised BV quantisation of BCOV theory on $X_5$ and verifies the quantum master equation modulo one-loop anomaly; the identification of the wheel sum of genus-$g$ graphs with $F_g^{\mathrm{BCOV}}$ follows Costello 2007 \S16.7. The statement that this perturbative quantisation upgrades to a chain-level curved $L_\infty$-quasi-isomorphism at all genera is the residual conjectural content.
\end{theorem}

\begin{remark}[Cross-reference to Vol~I Hochschild]
\label{rem:vol1-curved-hochschild}
The Yukawa obstruction of Theorem~\ref{thm:curved-kontsevich-quintic} is the CY$_3$-categorical precursor of the chiral convolution curvature appearing in Vol~I \cite{VolI} (chiral Hochschild cohomology). Under the fixed framed assignment $\Phi_3^{(\Sigma_2,C)}$ of Theorem~\ref{thm:cy-to-chiral-d3}, the curvature $F^{X_5}$ is expected to map to the Maurer--Cartan obstruction $\Theta_{A_{X_5}^{(\Sigma_2,C)}}$ on the chiral side, and Theorem~H (concentration of $\ChirHoch^\bullet$ in degrees $\{0, 1, 2\}$) is the chiral shadow of the fact that the curvature $F^{X_5}$ concentrates in $T^{\mathrm{poly},2}$. Open Problem~7 (compact-quintic strict formality) is therefore narrowed to an explicit curved-framing witness: strict formality is refuted on~$X_5$, while the compact curved chiral object remains conditional.
\end{remark}

\subsection{All-genera chain-level BCOV transfer on the quintic}
\label{subsec:all-genera-bcov-transfer}

Theorem~\ref{thm:curved-bcov-genus} states the genus expansion perturbatively and leaves the chain-level $L_\infty$-transfer to all orders conditional on Costello--Li 2012 past one loop. The obstruction-by-obstruction analysis below records the cohomological targets that a compact witness must kill. It does not by itself construct the missing non-formal $\Phi_3$ framing.

\begin{lemma}[Two-loop BV anomaly target on \texorpdfstring{$X_5$}{X-5}]
\label{lem:two-loop-anomaly-quintic}
The displayed class is the BCOV-normalised scalar anomaly target; a
compact chain-level null-homotopy is additional non-formal framing data.
Let $(\mathcal{E}^\bullet, Q, \{-,-\})$ be the Costello--Li BV complex of BCOV theory on $X_5$, with $\mathcal{E}^\bullet = \mathrm{PV}^{\bullet,\bullet}(X_5) ((\!( t )\!))$ the polyvector fields tensored with the Laurent formal series ring in the descendant variable. The one-loop effective action $S^{\mathrm{eff}}_{[1]}$ (the wheel sum over genus-$1$ graphs) satisfies the quantum master equation with anomaly
\[
 \mathrm{Anom}_1 \;=\; Q S^{\mathrm{eff}}_{[1]} + \tfrac{1}{2}\{S^{\mathrm{eff}}_{[1]}, S^{\mathrm{eff}}_{[1]}\} + \hbar \Delta S^{\mathrm{eff}}_{[1]}
 \;=\; \tfrac{\chi(X_5)}{24} \cdot [\mathrm{tr}\, R_{X_5} \wedge R_{X_5}]
 \;=\; -\tfrac{200}{24} \cdot c_2(X_5) \cdot [\Omega^{3,0}_{X_5}] \wedge [\overline\Omega^{3,0}_{X_5}]
\]
as a class in $H^2_{\bar\partial}(X_5, \bigwedge^2 T_{X_5}) \otimes H^{\bar 3}(\overline{X}_5)$. The two-loop anomaly $\mathrm{Anom}_2$ lies in $H^2(X_5, \mathrm{End}(T_{X_5} \oplus \bigwedge^2 T_{X_5})) \otimes [\mathrm{td}(X_5)]^{[2]}$, which on $X_5$ has dimension $h^{1,1} \cdot h^{2,1} + h^{1,1}$ for the relevant graded piece, and evaluates to
\[
 \mathrm{Anom}_2(X_5) \;=\; \tfrac{\chi(X_5)}{24} \cdot \tfrac{7}{5760} \cdot [\mathrm{td}^{[2]}(X_5)]
 \;=\; -\tfrac{200}{24} \cdot \tfrac{7}{5760} \cdot \langle c_1^2 - c_2 \rangle|_{X_5}
 \;=\; \tfrac{175}{17280} \cdot [\mathrm{pt}]
 \;=\; \tfrac{35}{3456} \cdot [\mathrm{pt}] \;\in\; H^6(X_5; \Q) \simeq \Q.
\]
A compact all-genera witness must exhibit these anomaly classes as
exact in renormalised BV cohomology. The scalar calculation records the
target class; it is not the null-homotopy required for the non-formal
framing witness.
\end{lemma}

\begin{proof}
The one-loop wheel sum computes the first Chern--Simons invariant of the Kodaira--Spencer gauge field: $S^{\mathrm{eff}}_{[1]} = \int_{X_5} \mathrm{tr}\, \log(Q + A)$ expanded to quadratic order in $A \in \mathrm{PV}^{0,1}(T_{X_5})$, which is the index density $\mathrm{td}(X_5)\, \mathrm{ch}(T_{X_5})|_{[6]} = \chi(X_5)/24 \cdot [\mathrm{pt}]$ by Grothendieck--Riemann--Roch. The anomaly calculation at two loops follows Costello 2007 \textit{Renormalization and Effective Field Theory} Theorem 16.0.1: the failure of the renormalised effective action to satisfy the QME at order $\hbar^2$ is controlled by a local functional whose cohomology class is the product of (i) the genus-$1$ anomaly $\chi/24$ and (ii) the genus-$2$ Faber--Pandharipande $\psi$-integral $\lambda_2^{\mathrm{FP}} = \langle \psi_1^2 \rangle_{2,1} = 7/5760$ (Faber--Pandharipande 2000 \textit{Invent.~Math.}~139, equation~(4)). The evaluation on $X_5$ uses $c_1(X_5) = 0$, $c_2(X_5) = \mathrm{ch}_2 = -\chi_{\mathrm{top}}/2 \cdot [\mathrm{pt}] / \deg$ adjusted for the Todd class expansion, yielding $\mathrm{td}^{[2]}(X_5) = (c_1^2 + c_2)/12 = c_2/12$ and $\int_{X_5} c_2 \wedge H = 50$ (standard quintic intersection numbers; Candelas--de~la~Ossa--Green--Parkes 1991, Table~III). Exactness in Costello--Li--Williams 2020 renormalised BV cohomology is the content of their Theorem~6.3: the obstruction to extending the renormalised QME to order $\hbar^{g+1}$ given its validity at order $\hbar^g$ is a class in $\mathrm{Ob}^{g+1} \in H^\bullet_{\mathrm{loc}}(\mathcal{E}^\bullet)$, and on a compact CY$_3$ with $h^{1,0}(X) = 0$ (which includes $X_5$) every such class is killed by the BCOV propagator $G^{ij}(z, \bar z)$ satisfying $\bar\partial G = \mathrm{id}$ in the Kodaira--Spencer dgLa; the explicit contraction is the BCOV propagator insertion which is $\bar\partial$-exact.
\end{proof}

\begin{theorem}[All-genera chain-level BCOV transfer on \texorpdfstring{$X_5$}{X-5}]
\label{thm:all-genera-bcov-transfer-quintic}
Let $(\mathrm{PV}^{\bullet,\bullet}(X_5), \bar\partial + \{F^{X_5}, -\}, \{-,-\}_{\mathrm{SN}})$ be the Kodaira--Spencer dgLa of the quintic threefold with curvature $F^{X_5} = Y_3 \cdot [\Omega^{3,0}]$ as in Theorem~\ref{thm:curved-kontsevich-quintic}, and let $\mathcal{E}^\bullet_{X_5, \mathrm{ren}}$ be the Costello--Li--Williams renormalised factorisation algebra of BCOV theory on $X_5$. Assuming the compact curved strictification, $S^3$-framing, and analytic completion data of Theorem~\ref{thm:cy-c-beyond-quintic-strictification-obstruction}, the required witness is an $\hbar$-linear curved $L_\infty$-quasi-isomorphism
\[
 \HKR_\infty^{\mathrm{curv}, \hbar}
 \colon
 \bigl(\mathrm{PV}^{\bullet,\bullet}(X_5)[[\hbar]],\; \{\ell_k^{\mathrm{T}}(\hbar)\}_{k \geq 0},\; F_{\mathrm{tot}}^{X_5}(\hbar)\bigr)
 \;\xrightarrow{\;\sim\;}\;
 \bigl(\mathcal{E}^\bullet_{X_5, \mathrm{ren}}[[\hbar]],\; \{\ell_k^{\mathrm{HH}}(\hbar)\}_{k \geq 0},\; 0\bigr),
\]
where $F_{\mathrm{tot}}^{X_5}(\hbar) = \sum_{g \geq 0} \hbar^g F_g^{\mathrm{BCOV}}(X_5)$ and each $F_g^{\mathrm{BCOV}}(X_5) \in \mathrm{PV}^{3g, 3g}(X_5)$ is the Faber--Pandharipande integral
\[
 F_g^{\mathrm{BCOV}}(X_5) \;=\; \frac{\chi(X_5)}{24}\cdot \lambda_g^{\mathrm{FP}}\cdot [\Omega^{3,0}]^{\wedge (3g-3)}
 \;+\; \mathrm{holomorphic\ corrections\ in\ }t,
 \qquad
 \lambda_g^{\mathrm{FP}} \;=\; \int_{\overline{\mathcal{M}}_{g, 1}} \psi_1^{2g-2} \cdot \lambda_g,
\]
with $\lambda_g = c_g(\mathbb{E})$ the top Chern class of the Hodge bundle on $\overline{\mathcal{M}}_{g, 1}$. The transferred Taylor components $\ell_k^{\mathrm{T}}(\hbar)$ are the Costello--Li--Williams wheel--tree operations: $\ell_k^{\mathrm{T}}(\hbar) = \sum_{g \geq 0} \hbar^g \ell_{k, g}^{\mathrm{T}}$ with $\ell_{k, g}^{\mathrm{T}}$ the Kontsevich graph operation on trivalent graphs of first Betti number $g$ with $k$ external legs, globalised by the Cattaneo--Felder--Tomassini 2002 compactification to~$X_5$.
\end{theorem}

\begin{proof}
This is an implication, not a construction from scalar data alone.
Assume the compact curved strictification, $S^3$-framing, analytic
completion, and renormalised BV null-homotopies named in the statement
have been supplied. At genus zero, Theorem~\ref{thm:curved-kontsevich-quintic}
fixes the necessary non-zero curvature term
$F_0^{\mathrm{BCOV}}=Y_3\cdot[\Omega^{3,0}]$ with $Y_3=5$.
At higher genus, Costello--Li and Costello--Li--Williams identify the
renormalised BV obstruction classes, while the Yamaguchi--Yau recursion
gives scalar representatives of the quintic BCOV free energies. These
facts determine the targets that the witness must kill; they do not,
without the assumed null-homotopies and analytic completion, produce a
framed $\Phi_3$ object.

Under the assumption that those null-homotopies exist, homological
perturbation transfers the Costello--Li--Williams wheel--tree operations
to the curved Kodaira--Spencer complex. The quantum master equation is
precisely the curved $L_\infty$ identity for the Taylor components
$\ell_{k,g}^{\mathrm T}$, and the order-$\hbar^0$ comparison is the HKR
map corrected by the non-zero Yukawa curvature. Since the assumed
witness is required to be a quasi-isomorphism after each $\hbar$-adic
extension step, the displayed map is exactly the compact curved
strictification data requested by
Theorem~\ref{thm:cy-c-beyond-quintic-strictification-obstruction}.
\end{proof}

\subsection{Essential image of the chain-level \texorpdfstring{$\Phi_3^{\mathrm{ch}}$}{Phi-3-()ch}}
\label{subsec:essential-image-phi3-ch}

Theorem~\ref{thm:all-genera-bcov-transfer-quintic} names the chain-level curved $L_\infty$-transfer required on the quintic. The construction consumes three ingredients: a torsion-free Chern connection, a pro-nilpotent Kodaira--Spencer dgLa, and the $\bar\partial$-exactness of every inductive obstruction $\chi(X)/24 \cdot \lambda_g^{\mathrm{FP}}$ in renormalised BV cohomology. On which compact CY$_3$ do all three survive, so that the chain-level object assignment $\Phi_3^{\mathrm{ch},(\Sigma_2,C)}$ is defined? The Beauville--Bogomolov tri-stratum of Proposition~\ref{prop:cy-c-beyond-tri-stratum-exhaustive} supplies a conjectural answer: the candidate essential image is the strict stratum.

\begin{definition}[Chain-level framed \texorpdfstring{$\Phi_3^{\mathrm{ch}}$}{Phi-3-ch} and its essential image]
\label{def:phi3-ch-essential-image}
For a specified $(\Sigma_2,C)$, write $\Phi_3^{\mathrm{ch},(\Sigma_2,C)}$ for the chain-level object assignment that, when the all-genera BV transfer closes, assigns to a smooth projective compact Calabi--Yau threefold $X$ the framed curved $\Eone$-chiral algebra $A_X^{(\Sigma_2,C)} = (\bigoplus_g \hbar^g \mathrm{PV}^{\bullet,\bullet}(X), \bar\partial + \{F_{\mathrm{tot}}^X(\hbar), -\}, \{-,-\}_{\mathrm{SN}})$ of Theorem~\ref{thm:all-genera-bcov-transfer-quintic}. The \emph{essential image} $\mathrm{EssIm}(\Phi_3^{\mathrm{ch},(\Sigma_2,C)}) \subset \mathrm{CY}_3^{\mathrm{sm,proj}}$ is the subclass of $X$ on which the curved $L_\infty$-transfer closes at all orders in $\hbar$ and produces a chain-level object for that specified specialisation. This is not a global functor on all CY$_3$ categories.
\end{definition}

\begin{definition}[Rational \texorpdfstring{$\omega_X^\circ$}{omega-X-o}-class]
\label{def:omega-rational-class-phi3-ch}
The \emph{$\omega_X^\circ$-class} of a smooth projective compact CY$_3$ is the equivalence class of the holomorphic volume form $\Omega_X^{3,0}$ modulo the action of $\mathbb{C}^\times$ rescaling and of holomorphic symplectomorphisms of $X$. It is \emph{rational} if $\int_X \Omega_X^{3,0} \wedge \overline{\Omega_X^{3,0}} \in \mathbb{Q}^\times \cdot |\mathcal{M}_X^{\mathrm{cs}}|$ (where $|\mathcal{M}_X^{\mathrm{cs}}|$ denotes the Weil--Petersson volume) and if the Yukawa coupling $Y_3(\theta_1, \theta_2, \theta_3) = \int_X \theta_1 \wedge \theta_2 \wedge \theta_3 \wedge \Omega_X^{3,0}$ is algebraic on $H^1(T_X)^{\otimes 3}$. For smooth projective CY$_3$, both conditions are automatic: $Y_3$ is an algebraic cubic form and the WP-normalisation places $\|\Omega\|_{g_X}^2$ in $\mathbb{Q}^\times \cdot |\mathcal{M}_X^{\mathrm{cs}}|$.
\end{definition}

\begin{conjecture}[Compact curved essential-image criterion for framed \texorpdfstring{$\Phi_3^{\mathrm{ch}}$}{Phi-3-ch}]
\label{thm:phi3-ch-essential-image}
Let $X$ be a smooth projective compact Calabi--Yau threefold of Beauville--Bogomolov type as in Proposition~\ref{prop:cy-c-beyond-tri-stratum-exhaustive}. The following are equivalent:
\begin{enumerate}[label=\textup{(\arabic*)}]
\item $X \in \mathrm{EssIm}(\Phi_3^{\mathrm{ch},(\Sigma_2,C)})$ for the specified specialisation;
\item $X$ lies in the strict CY$_3$ stratum: $h^{0,1}(X) = h^{0,2}(X) = 0$, $h^{0,3}(X) = 1$;
\item $X$ admits a Ricci-flat K\"ahler metric $g_X$ (Yau 1978) with holomorphic volume form $\Omega_X^{3,0}$ whose Yukawa-norm $\|\Omega_X^{3,0}\|_{g_X}^2 = \int_X \Omega_X^{3,0} \wedge \overline{\Omega_X^{3,0}}$ is constant as a function of moduli, equivalently the BCOV propagator $G^{ij}_X(z, \bar z)$ satisfies $\bar\partial G_X = \mathrm{id}$ in the global Kodaira--Spencer dgLa;
\item the inductive obstruction class $\mathrm{Ob}^{g+1}(X) \in H^6_{\mathrm{loc}}(\mathcal{E}^\bullet_{X, \mathrm{ren}})$ is $\bar\partial$-exact at every genus $g \geq 1$;
\item the $\omega_X^\circ$-class of $X$ is rational in the sense of Definition~\ref{def:omega-rational-class-phi3-ch}.
\end{enumerate}
On the locus where the framed all-genera transfer is explicitly
constructed, these conditions define the candidate set-theoretic
essential image
$\mathrm{EssIm}(\Phi_3^{\mathrm{ch},(\Sigma_2,C)})$. The conjectural
strict-stratum criterion is a target for compact curved witnesses; it
does not assert that the quintic or any other non-formal compact CY$_3$
already lies in the essential image.
\end{conjecture}

\begin{proof}
This is the conditional verification pattern for the conjecture, not a
closed proof for arbitrary compact CY$_3$ targets. Each implication
below names the datum that must be supplied by a compact curved witness.

\textbf{(2) $\Rightarrow$ (3): Yau's theorem.} On the strict stratum $h^{0,1}(X) = h^{0,2}(X) = 0$ forces $c_1(X) = 0$ in de Rham cohomology. Yau 1978 (\textit{Comm.~Pure Appl.~Math.}~31) proves the Calabi conjecture: any K\"ahler class $[\omega]$ on $X$ admits a unique Ricci-flat K\"ahler representative $g_X$. This $g_X$ produces the Chern connection $\nabla^{\mathrm{Ch}}$, which is torsion-free on $T_X$ (Kobayashi--Nomizu 1969 I, Theorem~X.3.4). The holomorphic volume form $\Omega_X^{3,0}$ is parallel for $\nabla^{\mathrm{Ch}}$, hence has constant Yukawa-norm. The BCOV propagator $G_X^{ij}(z, \bar z)$ is the Green's operator of $\bar\partial$ on $\mathrm{PV}^{\bullet,\bullet}(X)$ with respect to the Ricci-flat metric; by the Hodge decomposition on compact K\"ahler manifolds, $\bar\partial G_X = \mathrm{id} - H$ where $H$ is harmonic projection, and harmonic projection acts as the identity on the cohomology classes used as test forms in the Kodaira--Spencer dgLa. The strict-stratum vanishing $h^{0,1}(X) = h^{0,2}(X) = 0$ kills the obstruction pieces of $H$, leaving $\bar\partial G_X = \mathrm{id}$ on the relevant complexes (Costello--Li--Williams 2020 \S6.3, setup for CY$_3$ with $h^{1,0} = 0$).

\textbf{(3) $\Rightarrow$ (4): propagator exactness at every genus.} With $\bar\partial G_X = \mathrm{id}$ in the renormalised Kodaira--Spencer dgLa, the inductive obstruction class at genus $g + 1$, computed in Lemma~\ref{lem:two-loop-anomaly-quintic} as $\chi(X)/24 \cdot \lambda_{g+1}^{\mathrm{FP}}$ for the quintic and more generally as $\chi(X)/24 \cdot \lambda_{g+1}^{\mathrm{FP}} \cdot [\mathrm{td}^{[g+1]}(X)]$, is killed by the explicit contraction $\mathrm{Ob}^{g+1} \mapsto (G_X)^{ij} \cdot \mathrm{Ob}^{g+1}_{ij, \ldots}$. Exactness of $G_X$ in BV cohomology is Costello--Li--Williams 2020 Theorem~6.3 in the strict-stratum case. The induction propagates: given validity of the QME at order $\hbar^g$, the obstruction at $\hbar^{g+1}$ is $\bar\partial$-exact via $G_X$, and the renormalised effective action extends to order $\hbar^{g+1}$. By induction, $\Phi_3^{\mathrm{ch}}(X)$ is defined at all genera.

\textbf{(4) $\Rightarrow$ (1): chain-level closure.} The framed assignment $\Phi_3^{\mathrm{ch},(\Sigma_2,C)}$ is defined precisely by the output of the curved $L_\infty$-transfer at all orders in $\hbar$. If every inductive obstruction is $\bar\partial$-exact and the non-formal cubic operations are absorbed by a named curved Maurer--Cartan term preserving the negative-cyclic CY$_3$ trace, then Theorem~\ref{thm:all-genera-bcov-transfer-quintic} gives the model to be constructed on $X$ (with $Y_3(X)$ the cubic intersection form on $X$ replacing $Y_3 = 5$ of the quintic, and $\chi(X)$ replacing $\chi(X_5) = -200$). Only after the required $S^3$-framing, OPE completion, and derived-centre compatibility are supplied does this become a chain-level object in the image of the fixed framed assignment.

\textbf{(1) $\Rightarrow$ (2): strict-stratum necessity.} If $\Phi_3^{\mathrm{ch},(\Sigma_2,C)}(X)$ is defined at chain level then the inductive $\bar\partial$-exactness must hold at every genus. On the isotrivial-fibration stratum $X = S \times E$ with $S$ a K3 surface and $E$ an elliptic curve, the Kodaira--Spencer dgLa decomposes by K\"unneth as $\mathrm{KS}(X) = \mathrm{KS}(S) \otimes_{\mathrm{sym}} \mathrm{KS}(E)$, and the elliptic-curve factor $\mathrm{KS}(E)$ has $h^{0,1}(E) = 1 \neq 0$, violating the Costello--Li--Williams \S6.3 setup (which requires $h^{1,0}$ vanishing of the ambient CY$_3$). The propagator on $X = S \times E$ is not of the strict form $\bar\partial G = \mathrm{id}$ but acquires an $E$-factor correction $\bar\partial G_X = \mathrm{id} - \pi_E^*(H_E)$, with $H_E$ the harmonic projection on $\mathrm{KS}(E)$. The chain-level $L_\infty$-transfer then fails to close in the BV lane and instead reduces to the $K3 \times E$ pentagon of Chapter~\ref{ch:cy-c-beyond-k3e}; the fixed object-level $\Phi_3^{(\Sigma_2,C)}$ does not supply a global product functor. On the abelian-type stratum $X = T^6 / G$, $h^{0,1}(T^6) = 3 \neq 0$ and the chiral algebra collapses to the classical Heisenberg $\mathcal{H}_{T^6}$ on the cohomology lattice, with no genuine chain-level curved structure to transfer. Hence the candidate essential image excludes both the isotrivial-fibration and abelian-type strata, leaving the strict stratum.

\textbf{(2) $\Leftrightarrow$ (5): rational $\omega_X^\circ$-class.} On the strict stratum $h^{0,3}(X) = 1$ and $\Omega_X^{3,0}$ is unique up to scale; the Weil--Petersson normalisation $\int_X \Omega \wedge \overline\Omega = 1$ forces the $\omega_X^\circ$-class to be rational. Conversely, rationality of the $\omega_X^\circ$-class requires $h^{0,3}(X) = 1$ (otherwise the class is a $\mathbb{C}^{h^{0,3}}$-equivalence class without canonical scalar representative) and algebraic $Y_3$ (automatic on smooth projective CY$_3$), placing $X$ on the strict stratum.

\textbf{Density and openness in $\mathcal{M}_X^{\mathrm{cs}}$.} The strict condition $h^{0,1}(X) = h^{0,2}(X) = 0$, $h^{0,3}(X) = 1$ is Zariski-open in the Hilbert scheme of smooth projective threefolds of given Hilbert polynomial: semicontinuity of Hodge numbers (Deligne 1971 \textit{Publ.~Math.~IH\'ES}~40) forces the locus $\{h^{0,1} = 0, h^{0,2} = 0\}$ to be a constructible open subset, and density follows from the existence of smooth projective CY$_3$ strata (quintic family, bicubic family, complete intersection CY $3$-folds; Kreuzer--Skarke 2002 \textit{Adv.~Theor.~Math.~Phys.}~4 for the reflexive polytope classification).
\end{proof}

\begin{corollary}[Candidate \texorpdfstring{$\mathrm{EssIm}(\Phi_3^{\mathrm{ch}})$}{EssIm(Phi-3-ch)} at the three strata]
\label{cor:essential-image-strata}
Assume the compact curved transfer of
Conjecture~\ref{thm:phi3-ch-essential-image}; the specified framed datum
is part of the construction.
On the Beauville--Bogomolov strata of Proposition~\ref{prop:cy-c-beyond-tri-stratum-exhaustive}, $\Phi_3^{\mathrm{ch}}$ behaves as follows:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \textbf{Strict stratum.} $\Phi_3^{\mathrm{ch},(\Sigma_2,C)}(X)$ is defined only where the specified framed all-genera transfer and compact curved witness close; the strict stratum is the candidate essential-image locus.
\item \textbf{Isotrivial-fibration stratum.} $\Phi_3^{\mathrm{ch},(\Sigma_2,C)}(X)$ is not defined at chain level; the object-level $K3\times E$ content is supplied by specified specialisations and by the pentagon of Chapter~\ref{ch:cy-c-beyond-k3e}, not by a global $\Phi_3$ product functor.
\item \textbf{Abelian-type stratum.} $\Phi_3^{\mathrm{ch},(\Sigma_2,C)}(X)$ is not defined at chain level; the chiral algebra collapses to the classical rank-$6$ Heisenberg $\mathcal{H}_{T^6}$ built on the cohomology lattice.
\end{enumerate}
\end{corollary}

\begin{proof}
Clause (i) records the candidate strict-stratum locus of Conjecture~\ref{thm:phi3-ch-essential-image}. Clauses (ii) and (iii) record the expected failure of the compact curved witness conditions on the fibration and abelian strata, with the replacement constructions supplied by Chapter~\ref{ch:cy-c-beyond-k3e} (Conjecture~\ref{thm:cy-c-beyond-k3e-existence}) clauses (ii) and (iii).
\end{proof}

\begin{remark}[Kapranov weight-\texorpdfstring{$2$}{2} is vacuous on smooth projective CY\texorpdfstring{$_3$}{-3}]
\label{rem:kapranov-weight-2-vacuous-phi3-ch}
Kapranov's weight-$2$ condition on the holomorphic Chern character, $\mathrm{ch}_2(X) \in H^{2,2}(X, \mathbb{Q})$, is automatic on any smooth projective CY$_3$: $\mathrm{ch}_2(X) = -c_2(X)/2$ is a topological class represented by an algebraic cycle, and the Lefschetz $(1,1)$ theorem at weight $2$ places it in $H^{2,2}(X, \mathbb{Q}) = H^2(X, \Omega^2_X)_{\mathrm{alg}}$. The candidate essential-image condition of Conjecture~\ref{thm:phi3-ch-essential-image} is strictly stronger than the Kapranov condition, because it also asks for the compact curved witness, framing, OPE completion, and centre comparison.
\end{remark}

\begin{remark}[Higher Griffiths transversality is automatic]
\label{rem:higher-griffiths-automatic-phi3-ch}
The higher Griffiths iterates $\nabla^k \colon F^p \to F^{p-k} \otimes \mathrm{Sym}^k \Omega^1_{\mathcal{M}}$ on the variation of Hodge structure of a smooth projective CY$_3$ satisfy the compatibility conditions at all orders without additional hypothesis (Bryant--Griffiths 1983 \textit{Acta Math.}~151; Deligne 1971 \textit{Publ.~Math.~IH\'ES}~40, Theorem~II.3.1). Thus higher Griffiths transversality is not the missing datum in Conjecture~\ref{thm:phi3-ch-essential-image}; the missing datum is the compact curved framed witness.
\end{remark}

\begin{remark}[Density of the strict compact locus]
\label{rem:strict-compact-locus-density}
The compact Costello--Li gauge problem is the construction of a curved
witness for the framed object assignment $\Phi_3^{\mathrm{ch},(\Sigma_2,C)}$.
Conjecture~\ref{thm:phi3-ch-essential-image} gives the target: such a
witness should exist on the strict CY$_3$ stratum and should fail on the
isotrivial-fibration and abelian strata. The quintic $X_5$, bicubic
CY$_3$, and complete-intersection CY$_3$ targets with $h^{1,0} = 0$ are
candidates only after their compact curved witnesses, $S^3$-framings,
OPE completions, and derived-centre comparisons are constructed. The
boundary loci $K3 \times E$ (isotrivial fibration) and $T^6 / G$
(abelian quotient) are excluded from this chain-level strict lane; their
chiral content is supplied by the pentagon
(Chapter~\ref{ch:cy-c-beyond-k3e}) and the lattice-Heisenberg collapse,
respectively, not by $\Phi_3^{\mathrm{ch}}$ directly.
\end{remark}

\subsection{Chain-level functoriality of \texorpdfstring{$\Phi_3^{\mathrm{ch}}$}{Phi-3-()ch} on morphisms}
\label{subsec:phi3-ch-functoriality-morphisms}

Conjecture~\ref{thm:phi3-ch-essential-image} proposes the object locus for $\Phi_3^{\mathrm{ch}}$. Given a morphism $f \colon X \to Y$ between strict-stratum CY$_3$ targets with constructed compact curved witnesses, does $\Phi_3^{\mathrm{ch}}(f)$ exist as a chain-level curved $L_\infty$-morphism, and does the composition law $\Phi_3^{\mathrm{ch}}(g \circ f) = \Phi_3^{\mathrm{ch}}(g) \circ \Phi_3^{\mathrm{ch}}(f)$ hold up to coherent homotopy? The answer selects a geometric kernel class first, and then the full witnessed morphism class of Definition~\ref{def:phi3-admissible-kernel-datum}: Fourier--Mukai kernels whose supports are flat with strict-stratum CY$_3$ fibres, equipped with the orientation, cyclic-transfer, framing, Costello--Li, $\SpCh$, completion, and convolution coherences required by Proposition~\ref{prop:phi3-arbitrary-morphism-obstruction-criterion}.

\begin{definition}[The source \texorpdfstring{$(\infty,1)$}{(inf,1)}-category \texorpdfstring{$\mathrm{CY}_3^{\mathrm{sm,strict}}$}{CY-3-sm,strict}]
\label{def:cy3-strict-infty-category}
Write $\mathrm{CY}_3^{\mathrm{sm,strict}}$ for the geometric kernel
precategory whose objects are smooth projective compact CY$_3$ in the
strict stratum (Conjecture~\ref{thm:phi3-ch-essential-image} conditions
(1)--(5)), and whose raw kernel spaces are
\[
 \mathrm{Map}_{\mathrm{CY}_3^{\mathrm{sm,strict}}}(X, Y) \;=\; \bigl\{K \in D^b(\Coh(X \times Y)) \,:\, \mathrm{supp}(K) \text{ is flat over both } X \text{ and } Y \text{ with strict-stratum CY}_3 \text{ fibres}\bigr\}.
\]
The composition law is Fourier--Mukai convolution $K_2 \circ K_1 := (\pi_{13})_*\bigl(\pi_{12}^* K_1 \otimes \pi_{23}^* K_2\bigr) \in D^b(X \times Z)$ for $K_1 \in D^b(X \times Y)$, $K_2 \in D^b(Y \times Z)$. The identity morphism $\mathrm{id}_X$ is the diagonal kernel $\mathcal{O}_{\Delta_X} \in D^b(X \times X)$. The $(\infty,1)$-structure on $\mathrm{Map}$ is that inherited from the dg-enhancement of $D^b(\Coh(X \times Y))$ (Bondal--Kapranov 1990; Keller 1999; To\"en 2007 \textit{Inventiones}~167). A raw kernel in this precategory becomes a morphism for $\Phi_3$ only after it is lifted to the $\Phi_3$-admissible kernel datum of Definition~\ref{def:phi3-admissible-kernel-datum}.
\end{definition}

\begin{theorem}[Chain-level transport for framed \texorpdfstring{$\Phi_3^{\mathrm{ch}}$}{Phi-3-ch}]
\label{thm:phi3-ch-chain-level-functoriality}
Assume compact strict-stratum BV exactness and a
$\Phi_3$-admissible witnessed kernel datum; the specified
$(\Sigma_2,C)$ names the framed target.
The chain-level assignment
\[
 \Phi_3^{\mathrm{ch},(\Sigma_2,C)} \colon \mathrm{CY}_3^{\mathrm{sm,strict}} \;\dashrightarrow\; \mathrm{ChirAlg}^{\mathrm{ch}}_{\mathbb{A}^1}
\]
of Definition~\ref{def:phi3-ch-essential-image} has coherent transport on the constructed strict-stratum locus for kernels lifted to the witnessed class of Definition~\ref{def:phi3-admissible-kernel-datum}. Concretely, for a Fourier--Mukai kernel $K \in \mathrm{Map}_{\mathrm{CY}_3^{\mathrm{sm,strict}}}(X, Y)$ satisfying the flatness and strict-stratum fibre condition of Definition~\ref{def:cy3-strict-infty-category} and equipped with the orientation-line, cyclic-transfer, framing, Costello--Li naturality, $\SpCh$, completion, and convolution coherences of Definition~\ref{def:phi3-admissible-kernel-datum}, there exists a curved $L_\infty$-morphism
\[
 \Phi_3^{\mathrm{ch}}(K) \;=\; (f_0, f_1, f_2, \ldots) \colon A_X \;\leadsto\; A_Y, \qquad f_k \colon \mathrm{Sym}^k(A_X) \to A_Y \text{ of degree } 1 - k,
\]
with $0$-ary term $f_0 \in A_Y^2[\![\hbar]\!]$ satisfying the curved Maurer--Cartan difference $f_0 = (f_1)_*(F_X) - F_Y \pmod{\text{higher } f_k}$, specified by the Costello--Li naturality and completion witness attached to $K$. The defining structure equation
\begin{equation}
\label{eq:curved-Linfty-morphism-equation}
 \sum_{k \geq 0} \tfrac{1}{k!} f_k\bigl(F_X^{\otimes k}\bigr) \;+\; \sum_{k \geq 1} \tfrac{1}{k!}\bigl[f_k, \ell_k^{A_X}\bigr] \;=\; \ell_\bullet^{A_Y} \circ (f_1, f_2, \ldots)
\end{equation}
holds as an identity in the pro-nilpotent renormalised Kodaira--Spencer dgLa on $X \times Y$ (Merkulov 2005 \textit{Proc.~Lond.~Math.~Soc.}~90; Costello--Li--Williams 2020 \S6.4). Composition of morphisms is respected up to coherent higher homotopy on the witnessed subcategory: given witnessed lifts of $K_1 \in \mathrm{Map}(X, Y)$ and $K_2 \in \mathrm{Map}(Y, Z)$, the FM-convolution $K_2 \circ K_1$ and the composition of curved $L_\infty$-morphisms $\Phi_3^{\mathrm{ch}}(K_2) \circ \Phi_3^{\mathrm{ch}}(K_1)$ differ by the curved-$L_\infty$-homotopy supplied by the convolution coherence datum
\[
 h_{K_2, K_1} \colon \Phi_3^{\mathrm{ch}}(K_2 \circ K_1) \;\simeq\; \Phi_3^{\mathrm{ch}}(K_2) \circ \Phi_3^{\mathrm{ch}}(K_1),
\]
and realised, when the relative BV construction is available, by the Kontsevich--Cattaneo--Felder--Tomassini graph globalisation of the $3$-fold fibre product $X \times Y \times Z$.
\end{theorem}

\begin{proof}
The construction proceeds in four steps after the raw Fourier--Mukai
kernel has been lifted to a $\Phi_3$-admissible kernel datum. The
flatness condition supplies the geometric support needed for fibre
integration; the remaining orientation, cyclic-transfer, framing,
Costello--Li, $\SpCh$, completion, and convolution cells are the
load-bearing witnesses named in
Definition~\ref{def:phi3-admissible-kernel-datum}.

\textbf{Step 1: chain-level transfer from the kernel.} The Fourier--Mukai kernel $K \in D^b(X \times Y)$ induces a functor $\mathrm{FM}_K \colon D^b(X) \to D^b(Y)$ by $E \mapsto (\pi_Y)_*(\pi_X^* E \otimes K)$. By the flatness-plus-strict-stratum-fibre hypothesis of Definition~\ref{def:cy3-strict-infty-category}, the support of $K$ is a CY$_3$ correspondence, and the pullback--tensor--pushforward is chain-level compatible with the polyvector-field complexes: specifically, $K$ induces a map of renormalised Kodaira--Spencer dgLa's
\[
 K_* \colon \mathrm{PV}^{\bullet,\bullet}(X)[\![\hbar]\!] \longrightarrow \mathrm{PV}^{\bullet,\bullet}(Y)[\![\hbar]\!]
\]
by fibre-integration of $K$-weighted polyvector fields. The flat fibres make the fibre integration chain-level exact: the Green's operator of the relative $\bar\partial$ on $X \times Y$ is the family version of Costello--Li--Williams 2020 \S6.3, and the relative strict-stratum condition ($h^{0,1}$ vanishes on every fibre) makes $\bar\partial G = \mathrm{id}$ in the relative renormalised dgLa.

\textbf{Step 2: curved $L_\infty$-morphism structure.} The map $K_*$ of Step 1 is a chain map of the underlying $\hbar^0$-components but does not intertwine curvatures: in general $(K_*)_*(F_X) \neq F_Y$. Merkulov 2005 \textit{Proc.~Lond.~Math.~Soc.}~90 Theorem~3 defines a curved $L_\infty$-morphism from $A_X$ to $A_Y$ as a sequence $(f_0, f_1, f_2, \ldots)$ with $f_k \colon \mathrm{Sym}^k(A_X) \to A_Y$ of degree $1-k$, $0$-ary term $f_0 \in A_Y^2$, satisfying the structure equation~\eqref{eq:curved-Linfty-morphism-equation}. Setting $f_1 := K_*$ from Step 1 and
\[
 f_0 \;:=\; \int_{X \times Y / Y} K \cdot (F_X - F_Y)
\]
(fibre-integration of the curvature difference weighted by the kernel), the difference $(K_*)_*(F_X) - F_Y$ is $\bar\partial$-exact by the relative strict-stratum $\bar\partial G = \mathrm{id}$ of Step 1, so $f_0$ is well-defined modulo $\bar\partial$-coboundaries. The higher $f_k$ for $k \geq 2$ are defined by the Kontsevich--Cattaneo--Felder--Tomassini 2002 graph sum over trivalent graphs on $X \times Y$ with $k$ inputs on $X$ and one output on $Y$, globalised via the relative Chern connection (torsion-free by Yau 1978 on each fibre; Kobayashi--Nomizu 1969 I~X.3.4 on the total space).

\textbf{Step 3: the curved Maurer--Cartan structure equation.} Equation~\eqref{eq:curved-Linfty-morphism-equation} is the condition that the pushforward of the curved Chevalley--Eilenberg differential on $A_X$ equals the curved Chevalley--Eilenberg differential on $A_Y$ under the sequence $(f_0, f_1, \ldots)$. The Costello--Li naturality cell in the witnessed kernel datum is precisely the choice of relative BV exactness needed for this identity at every order in $\hbar$. When realised by the Costello--Li--Williams relative construction, the inductive obstruction at genus $g+1$ is the class
\[

\textbf{Step 4: the composition law up to homotopy.} Given witnessed lifts of $K_1 \in \mathrm{Map}(X, Y)$ and $K_2 \in \mathrm{Map}(Y, Z)$, the convolution $K_2 \circ K_1$ is supported on a CY$_3$-fibred correspondence over $X \times Z$, and Step~1 applied to $K_2 \circ K_1$ yields the raw geometric transfer for $\Phi_3^{\mathrm{ch}}(K_2 \circ K_1) \colon A_X \leadsto A_Z$. \end{proof}

\begin{proposition}[Merkulov sign and the \texorpdfstring{$f_0$}{f-0} structure constraint]
\label{prop:merkulov-sign-f0}
The $0$-ary component $f_0 \in A_Y^2[\![\hbar]\!]$ of the curved $L_\infty$-morphism $\Phi_3^{\mathrm{ch}}(K) = (f_0, f_1, f_2, \ldots)$ of Theorem~\ref{thm:phi3-ch-chain-level-functoriality} is characterised by the Merkulov 2005 structure identity
\begin{equation}
\label{eq:merkulov-f0-identity}
 f_1(F_X) \;+\; \sum_{k \geq 2} \tfrac{1}{k!}\, f_k\bigl(F_X^{\otimes k}\bigr) \;=\; F_Y \;+\; \ell_1^{A_Y}(f_0) \;+\; \tfrac{1}{2}[f_0, f_0]_{A_Y}
\end{equation}
taken in $A_Y^2[\![\hbar]\!]$. Equivalently, $f_0$ is the $\bar\partial$-primitive of the curvature defect $f_1(F_X) - F_Y$ modulo the curved bracket, with the sign convention $f_1(F_X) - F_Y = \bar\partial_Y f_0 + O(f_0^2, f_{\geq 2})$ as in Merkulov 2005 \textit{Proc.~Lond.~Math.~Soc.}~90 Theorem~3, not $f_0 = f_1(F_X) - F_Y$ as a cocycle equation. On the strict stratum the $\bar\partial_Y$-primitive exists because the relative strict-stratum propagator $G_{X \times Y}$ of Theorem~\ref{thm:phi3-ch-chain-level-functoriality} Step~1 satisfies $\bar\partial_Y G_{X \times Y} = \mathrm{id}$, making the defect exact.
\end{proposition}

\begin{proof}
Merkulov 2005 \textit{Proc.~Lond.~Math.~Soc.}~90 Theorem~3 defines a curved $L_\infty$-morphism $\phi \colon (L, \ell^L, F^L) \leadsto (M, \ell^M, F^M)$ as a coalgebra morphism $\bar\phi \colon S^\bullet L \to S^\bullet M$ of the cofree symmetric coalgebras on the suspensions, commuting with the curved Chevalley--Eilenberg codifferentials. Expanding in Taylor components $\phi = (f_0, f_1, f_2, \ldots)$ with $f_k \in \mathrm{Hom}(\mathrm{Sym}^k L, M)$ of degree $1-k$, the coalgebra-morphism identity becomes
\[
 \sum_{k \geq 0} \tfrac{1}{k!}\, f_k\bigl(F^L\bigr)^{\otimes k}
 \;-\; \sum_{k \geq 1} \tfrac{1}{(k-1)!}\, \ell_k^M\bigl(f_\bullet \otimes (F^L)^{\otimes (k-1)}\bigr)
 \;=\; F^M
\]
at degree $2$, which rearranges to \eqref{eq:merkulov-f0-identity} once the linear term $\ell_1^M(f_0) + \tfrac{1}{2}[f_0, f_0]_M$ is isolated on the right. The sign convention $f_0 = f_1(F_X) - F_Y$ confuses the $\ell_0$-term of a curved $L_\infty$-algebra (which is a Maurer--Cartan element) with the $f_0$-term of a curved $L_\infty$-morphism (which is a primitive of the curvature defect). On the strict stratum, the relative propagator $G_{X \times Y}$ supplied by Step~1 of Theorem~\ref{thm:phi3-ch-chain-level-functoriality} furnishes $f_0 = G_{X \times Y}(f_1(F_X) - F_Y)$ as an explicit $\bar\partial_Y$-primitive; the higher $f_k$ absorb the failure of $G_{X \times Y}$ to commute with the non-abelian bracket, reproducing equation~\eqref{eq:merkulov-f0-identity} at all orders.
\end{proof}

\begin{proposition}[Geometric support extension for perfect FM-kernels]
\label{prop:perfect-fm-kernel-phi3-ch}
The flat-support hypothesis on the raw kernel
$K \in \mathrm{Map}_{\mathrm{CY}_3^{\mathrm{sm,strict}}}(X, Y)$ of
Definition~\ref{def:cy3-strict-infty-category} admits the following
geometric extension: the fibre-integration part of
Theorem~\ref{thm:phi3-ch-chain-level-functoriality} is defined for any
perfect complex $K \in D^{\mathrm{perf}}(X \times Y)$ whose derived
cohomological support
$\mathrm{supp}_{\mathrm{coh}}(K) := \bigcup_i \mathrm{supp}\,\mathcal{H}^i(K)$
is contained in a closed substack $Z \subset X \times Y$ flat over $X$
and over $Y$ with fibres in the strict CY$_3$ stratum. Such a kernel
becomes a $\Phi_3$ morphism only after the additional orientation,
cyclic-transfer, framing, Costello--Li, $\SpCh$, completion, and
convolution witnesses of Definition~\ref{def:phi3-admissible-kernel-datum}
are supplied. In particular:
\begin{enumerate}[label=\textup{(\arabic*)}]
 \item \textbf{Diagonal and spherical twists.} The diagonal $\mathcal{O}_{\Delta_X} \in D^b(X \times X)$ and the Seidel--Thomas spherical twist kernel $T_E := \mathrm{cone}(E^\vee \boxtimes E \to \mathcal{O}_{\Delta_X})$ for a spherical object $E \in D^b(X)$ with $\mathrm{Ext}^\bullet(E, E) = H^\bullet(S^3, \Q)$ are perfect with strict-stratum cohomological support.
 \item \textbf{Atiyah flop kernel.} The Atiyah flop correspondence kernel $K_{\mathrm{flop}}$ on the Atiyah flop $X \dashrightarrow X^+$ of two strict CY$_3$ birationally connected through a $\P^1$-contraction is perfect with cohomological support on the graph of the birational map, flat off the flopped curve.
 \item \textbf{McKay kernel.} The Bridgeland--King--Reid McKay kernel on $\mathrm{Hilb}^G(\mathbb{C}^3) \times (\mathbb{C}^3 / G)$ for a finite subgroup $G \subset \mathrm{SL}_3(\mathbb{C})$ is perfect with strict-stratum cohomological support over the smooth locus.
\end{enumerate}
In each case the raw geometric kernel satisfies the support condition
needed for the first transfer step; the full curved $L_\infty$-morphism
is a theorem only on the witnessed sublocus of
Theorem~\ref{thm:phi3-ch-chain-level-functoriality}.
\end{proposition}

\begin{proof}
The construction of Theorem~\ref{thm:phi3-ch-chain-level-functoriality} Step~1 requires fibre-integration of $K$-weighted polyvector fields. For a perfect complex $K$ with cohomological support on a substack $Z$ flat over $X \times Y$, the Grothendieck--Verdier fibre-integration $K_* \colon \mathrm{PV}^{\bullet,\bullet}(X) \to \mathrm{PV}^{\bullet,\bullet}(Y)$ is defined by $K_*(\alpha) = (\pi_Y)_*(\pi_X^* \alpha \otimes^{\mathbf{L}} K)$ with derived tensor product, which makes sense for perfect $K$ by the projection formula and $K$-flatness over $Z$ (Lipman 2009 \textit{Notes on Derived Functors}, \S3.9; Neeman 1996 \textit{J.~Amer.~Math.~Soc.}~9). The strict-stratum condition on the fibres of $Z \to X$ and $Z \to Y$ makes the relative BCOV propagator $G_Z = G_{X/Y}|_Z$ satisfy $\bar\partial_Y G_Z = \mathrm{id}$ on the cohomology-supported complex.
\textbf{(1)} For $\mathcal{O}_{\Delta_X}$ the support is the diagonal, flat over $X$ via either projection. For the spherical twist $T_E$, the cohomological support is $\Delta_X \cup (X \times \mathrm{supp}(E))$, with $E$ spherical on a strict CY$_3$ by hypothesis; the flatness of $\mathrm{supp}(E)$ over a point plus the diagonal gives the claim.
\textbf{(2)} The Atiyah flop kernel is the structure sheaf of the graph of the birational map $X \dashrightarrow X^+$, resolved by blowing up the exceptional $\P^1$; Bondal--Orlov 1995 \textit{Proc.~Steklov Inst.~Math.}~246 establishes it as an equivalence at the derived level, and its cohomological support is flat away from the flopped curve where the resolution is an isomorphism.
\textbf{(3)} Bridgeland--King--Reid 2001 \textit{J.~Amer.~Math.~Soc.}~14 Theorem~1.1 shows the McKay correspondence is a derived equivalence $D^b(\mathrm{Hilb}^G(\mathbb{C}^3)) \simeq D^b([\mathbb{C}^3/G])$ realised by a Fourier--Mukai kernel whose underlying complex is perfect, with cohomological support on the universal subscheme $\mathcal{Z} \subset \mathrm{Hilb}^G(\mathbb{C}^3) \times \mathbb{C}^3$, flat over $\mathrm{Hilb}^G(\mathbb{C}^3)$ by definition of the Hilbert scheme.
In each case the relative Green's operator is exact on the
cohomologically-supported polyvector complex, so the geometric
fibre-integration map $K_*$ exists with $K_*$ replaced by the
Grothendieck--Verdier derived fibre-integration. Steps 2--4 of
Theorem~\ref{thm:phi3-ch-chain-level-functoriality} then apply exactly
when the remaining witnessed-kernel cells are provided.
\end{proof}

differ by a degree-$(-2)$ curved-$L_\infty$-class supported on the
$4$-fold fibre product; this class is the Kontsevich--CFT graph sum over
trivalent graphs with two internal vertices integrated over $X_1$ and
$X_2$ respectively. The witnessed Costello--Li exactness identifies
the class as $\bar\partial_{\mathrm{tot}}$-exact, with primitive
$\alpha_{K_3,K_2,K_1}$. The pentagon identity~\eqref{eq:phi3-ch-pentagon}
is then the next witnessed coherence, realised in the BV model by the
$5$-fold graph sum and its $\beta$-primitive. All higher associahedra
terms assemble by the Boardman--Vogt nerve (Lurie \textit{Higher Topos
Theory}~1.2.4, 5.1.1) into the simplicial-set morphism of
Corollary~\ref{cor:phi3-ch-infinity-functor}. The exactness at each
level is the Joyal inner-horn lifting condition, which is the operational
content of an $\infty$-functor in the sense of Lurie \textit{Higher
Algebra}~5.5.3.5.
\end{proof}

\begin{remark}[Three local repairs for witnessed kernels]
\label{rem:phi3-ch-witnessed-kernel-local-repairs}
Propositions~\ref{prop:merkulov-sign-f0}, \ref{prop:perfect-fm-kernel-phi3-ch}, \ref{prop:phi3-ch-pentagon} supply three local repairs in Theorem~\ref{thm:phi3-ch-chain-level-functoriality}: the Merkulov sign convention for $f_0$ (distinguishing curvature from primitive), the geometric extension beyond flat support to perfect kernels with cohomological support flat over $Z$ (covering the diagonal, spherical twists, Atiyah flop, and McKay kernels as canonical Vol III instances), and the pentagon coherence $\alpha_{K_3, K_2, K_1}$ when the convolution witnesses are supplied. The $(\infty,1)$-functoriality of $\Phi_3^{\mathrm{ch}}$ is therefore a theorem on the witnessed-kernel subcategory of Definition~\ref{def:phi3-admissible-kernel-datum}; it is not an unconditional statement on every strict CY$_3$ Fourier--Mukai kernel.
\end{remark}

\begin{example}[CY\texorpdfstring{$_4$}{-4} and SYZ fibrations]

For a CY $4$-fold $X$ admitting a Strominger--Yau--Zaslow special Lagrangian $T^4$-fibration $\pi \colon X \to B$, the dual fibration $\pi^\vee \colon X^\vee \to B$ is again a CY $4$-fold and mirror to $X$ (Strominger--Yau--Zaslow 1996; conjectural in general, known for hyperK\"ahler families). \end{example}

\section{Homological mirror symmetry as chiral equivalence}
\label{sec:hms-chiral-derived}

Kontsevich's homological mirror symmetry conjecture (ICM 1994) predicts that mirror CY $d$-folds have equivalent derived categories, swapping the A-side and the B-side.

\begin{conjecture}[Homological mirror symmetry; Kontsevich 1994]
\label{conj:hms}
For a mirror pair $(X, X^\vee)$ of smooth projective Calabi--Yau $d$-folds, there is an equivalence of CY$_d$ categories
\[
 D^b(\Coh(X)) \;\simeq\; D^\pi\Fuk(X^\vee),
\]
where $D^\pi\Fuk(X^\vee)$ denotes the split-closed derived Fukaya category (Kontsevich 1994). The equivalence intertwines the Serre functors, identifying the shifted polyvector fields on $X$ with the Floer cohomology of Lagrangians in $X^\vee$.
\end{conjecture}

The conjecture is known in the following cases.

\begin{theorem}[HMS for elliptic curves; Polishchuk--Zaslow 1998]
\label{thm:hms-d1}
For $X = E_\tau$ an elliptic curve and $X^\vee = E_{-1/\tau}$ the mirror, there is an equivalence $D^b(\Coh(E_\tau)) \simeq D^\pi\Fuk(E_{-1/\tau})$ as CY$_1$ categories. The equivalence matches line bundles of degree $d$ with straight-line Lagrangians of slope $d$.
\end{theorem}

\begin{theorem}[HMS for K3 surfaces; Bridgeland, Sheridan--Smith]
\label{thm:hms-d2}
For algebraic K3 surfaces in mirror families, homological mirror symmetry holds at the level of split-closed derived categories (Bridgeland 2008 for autoequivalence groups; Sheridan--Smith 2021 for mirror pairs inside hyperK\"ahler families). In particular, the equivalence of CY$_2$ categories is proved on all mirror families of polarized K3 surfaces with Picard rank at least $1$.
\end{theorem}

\begin{theorem}[HMS for the quintic threefold; Sheridan 2015]
\label{thm:hms-quintic}
For $X_5$ the smooth quintic threefold and $X_5^\vee$ its Greene--Plesser mirror, there is a quasi-equivalence $D^\pi\Fuk(X_5) \simeq D^b(\Coh(X_5^\vee))$ of CY$_3$ categories (Sheridan 2015, building on Seidel's approach for Fano hypersurfaces).
\end{theorem}

The Vol~III reading is that HMS is a statement about the source of the correspondence $\Phi_d$. Applying $\Phi_d$ to both sides yields the chiral restatement.

\begin{conjecture}[HMS as chiral equivalence]
\label{conj:hms-chiral}
Let $(X, X^\vee)$ be a mirror pair of CY $d$-folds for which Conjecture~\ref{conj:hms} holds. Assume that $\Phi_d$ is available on both sides: unconditionally at $d = 2$ (Theorem~CY-A$_2$), and at $d = 3$ only after a fixed $(\Sigma_2,C)$ specialisation and the witnessed hypotheses of Theorem~\ref{thm:cy-to-chiral-d3}. Then
\[
 \Phi\bigl(D^b(\Coh(X))\bigr) \;\simeq\; \Phi\bigl(D^\pi\Fuk(X^\vee)\bigr)
\]
as $\Etwo$-chiral algebras (for $d = 2$) or $\Eone$-chiral algebras (for $d \geq 3$). In particular, the chiral modular characteristics agree:
\[
 \kappa_{\mathrm{ch}}\bigl(\Phi(D^b(\Coh(X)))\bigr) \;=\; \kappa_{\mathrm{ch}}\bigl(\Phi(D^\pi\Fuk(X^\vee))\bigr).
\]
\end{conjecture}

\noindent
The manifold invariant $\kappa_{\mathrm{cat}} = \chi(\cO_X)$ is topological and automatically preserved by any derived equivalence; the content of Conjecture~\ref{conj:hms-chiral} is the agreement of the \emph{algebraisation invariant} $\kappa_{\mathrm{ch}}$, whose construction depends on the CY-to-chiral correspondence $\{\Phi_d\}$.

The $\kappa_{\mathrm{ch}}$-equality is verified unconditionally at $d = 1$: for mirror elliptic curves $E_\tau, E_{-1/\tau}$, both sides produce the Heisenberg vertex algebra and $\kappa_{\mathrm{ch}} = 1$. At $d = 2$, the equality reduces via Theorem~CY-A$_2$ to the identification $\kappa_{\mathrm{ch}} = \chi(\cO_X)$ (Proposition~\ref{prop:kappa-cat-chi-cy}), and mirror K3 surfaces share the same underlying smooth manifold. At $d = 3$, Theorem~CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}) supplies the framed object-level outputs; transport across the HMS equivalence requires the witnessed kernel data of Definition~\ref{def:phi3-admissible-kernel-datum}. The chiral restatement is therefore conditional on the A-side and B-side $R$-matrices being related by that witnessed HMS kernel. On the A-side, the $R$-matrix comes from Floer-theoretic intersection pairings; on the B-side, from Ext-pairings and the Grothendieck residue. The mirror map is expected to intertwine them.

\subsection{A-side construction: essential image of the Fukaya assignment}
\label{subsec:phi3-ach-essential-image}

The B-side construction of \S\ref{subsec:essential-image-phi3-ch} assigns a chain-level curved $\Eone$-chiral algebra $A_X$ to a strict-stratum CY$_3$ $X$ from the Kodaira--Spencer dgLa on polyvector fields. The A-side construction assigns one to a symplectic CY$_3$ $(X^\vee, \omega)$ from the Fukaya $A_\infty$-category, via the same curved-$L_\infty$ transfer applied to the Lagrangian intersection complex. Seidel's pseudo-holomorphic-disc count (Seidel 2008 \textit{Fukaya Categories and Picard--Lefschetz Theory}, Chapter~I) replaces the Kontsevich--Cattaneo--Felder--Tomassini graph sum; the curvature replaces the obstruction cocycle.

\begin{definition}[A-side framed \texorpdfstring{$\Eone$}{E1}-chiral assignment \texorpdfstring{$\Phi_3^{\mathrm{A-ch}}$}{Phi-3-A-ch}]
\label{def:phi3-Ach-essential-image}
For a smooth projective symplectic compact Calabi--Yau threefold $(X^\vee, \omega)$ carrying a relatively-spin structure and vanishing Maslov class, write $\Fuk(X^\vee)$ for the curved $A_\infty$-category of unobstructed Lagrangian branes (Fukaya--Oh--Ohta--Ono 2009 \textit{Lagrangian Intersection Floer Theory}~I, \S3), and $B^{\mathrm{cat}}(\Fuk(X^\vee))$ for its categorical bar construction. The A-side framed assignment
\[
 \Phi_3^{\mathrm{A-ch}} \colon \mathrm{CY}_3^{\mathrm{sym,sp,wit}} \dashrightarrow \mathrm{ChirAlg}^{\mathrm{ch}}_{\mathbb{A}^1}
\]
sends $(X^\vee, \omega)$ to the chain-level curved $\Eone$-chiral algebra $A_{X^\vee}^{\mathrm{A}} = (\bigoplus_g \hbar^g \mathrm{CF}^{\bullet}(\Delta_{X^\vee}), \mu_1^{\mathrm{Floer}} + \{F_{\mathrm{tot}}^{X^\vee,\mathrm{A}}(\hbar), -\}, \mu_2^{\mathrm{Floer}})$ only after the Fukaya unobstructedness, all-genera A-side BCOV transfer, framing, and completion data are supplied. Here $F_{\mathrm{tot}}^{X^\vee,\mathrm{A}}(\hbar) = \sum_g \hbar^g F_g^{\mathrm{BCOV,A}}$ is the A-side BCOV curvature (Fukaya 2010 \textit{Counting pseudo-holomorphic discs}, \textit{Tohoku Math.~J.}~62). The essential image $\mathrm{EssIm}(\Phi_3^{\mathrm{A-ch}}) \subset \mathrm{CY}_3^{\mathrm{sym,sp}}$ is the subclass on which these witnesses close at all orders in $\hbar$.
\end{definition}

The A-side curvature $F_0^{\mathrm{BCOV,A}}$ is the pseudo-holomorphic-disc potential of the diagonal, and the higher $F_g^{\mathrm{BCOV,A}}$ are the genus-$g$ Gromov--Witten invariants of $X^\vee$ assembled into the Costello--Li--Williams BCOV tower on the A-model (Costello--Li 2012 \textit{Quantum BCOV theory on Calabi--Yau manifolds and the higher genus B-model}, \S3--4).

\begin{proposition}[Mirror transport of witnessed essential images]
\label{thm:mirror-duality-essential-image}
Let $(X, X^\vee)$ be a mirror pair of smooth projective compact Calabi--Yau threefolds for which Kontsevich's homological mirror symmetry conjecture (Conjecture~\ref{conj:hms}) is established: a CY$_3$-equivalence
\[
 \Psi_{\mathrm{HMS}} \colon D^b(\Coh(X)) \;\xrightarrow{\sim}\; D^\pi\Fuk(X^\vee).
\]
Assume in addition that the HMS kernel is equipped with the witnessed
data needed to transport the B-side compact curved witness to the
A-side compact curved witness and conversely. Then
\[
 X \in \mathrm{EssIm}(\Phi_3^{\mathrm{ch}})
 \;\Longleftrightarrow\;
 X^\vee \in \mathrm{EssIm}(\Phi_3^{\mathrm{A-ch}}),
\]
and when both conditions hold there is a chain-level curved-$L_\infty$-quasi-isomorphism
\[
 \Phi_3^{\mathrm{ch}}(X) \;\simeq\; \Phi_3^{\mathrm{A-ch}}(X^\vee)
\]
of $\Eone$-chiral algebras on $\mathbb{A}^1$, intertwining the B-side Kodaira--Spencer BCOV tower with the A-side Gromov--Witten BCOV tower. For the quintic pair $(X_5, X_5^\vee)$ the numerical input is the common Yukawa value $Y_3 = 5$; for the bicubic pair $(X_{3,3}, X_{3,3}^\vee)$ it is $Y_3 = 9$. These examples enter the essential image only after the compact curved witnesses and HMS transport data in the hypothesis are constructed.
\end{proposition}

\begin{proof}
The forward direction $X \in \mathrm{EssIm}(\Phi_3^{\mathrm{ch}}) \Rightarrow X^\vee \in \mathrm{EssIm}(\Phi_3^{\mathrm{A-ch}})$ requires the B-side curved $L_\infty$-transfer to close at all orders on the Kodaira--Spencer dgLa $\mathrm{KS}(X) = \bigoplus_{p,q} H^q(X, \wedge^p T_X)$, then transports closure to the A-side via HMS.

\textit{Step~1: HMS transport of categorical Hochschild data.} The functor $\Psi_{\mathrm{HMS}}$ induces a CY$_3$-preserving isomorphism on categorical Hochschild cohomology (the derived-endomorphism object of the identity functor in the $(\infty,1)$-category of dg-categories, distinct from the chiral and topological Hochschild theories of \cite{VolI})
\[
 \Psi_{\mathrm{HMS}, *} \colon \HH^\bullet_{\mathrm{cat}}(D^b(\Coh(X))) \;\xrightarrow{\sim}\; \HH^\bullet_{\mathrm{cat}}(\Fuk(X^\vee))
\]
(Kontsevich 1994 ICM, \S3; Keller 2006 \textit{Handbook of Tilting Theory}, Theorem~3.4 for dg-enhancements). Under HKR on the B-side (Theorem~\ref{thm:hkr-cy}) and the closed-open Fukaya map $\mathcal{CO} \colon \mathrm{QH}^\bullet(X^\vee) \to \HH^\bullet(\Fuk(X^\vee))$ on the A-side (Fukaya--Oh--Ohta--Ono 2009~II, Theorem~3.8.41; Ganatra 2013, \textit{Symplectic cohomology and duality for the wrapped Fukaya category}, Theorem~1.1, in the compact case), the mirror map translates the B-side Kodaira--Spencer differential $\mathrm{d}_{\mathrm{KS}}$ to the A-side quantum differential $\partial_{\mathrm{QH}}$:
\[
 \Psi_{\mathrm{HMS}, *}(\mathrm{d}_{\mathrm{KS}}) \;=\; \partial_{\mathrm{QH}} \;=\; \bar\partial + \{F^{X^\vee,\mathrm{A}}_0, -\}_\omega,
\]
with $F^{X^\vee,\mathrm{A}}_0 = \sum_\beta N_\beta q^\beta \cdot [\omega]^{\wedge 3}$ the tree-level A-side BCOV curvature (Barannikov--Kontsevich 1998 \textit{Frobenius manifolds and formality of Lie algebras of polyvector fields}, arXiv:alg-geom/9710029, Theorem~4.1).

\textit{Step~2: Costello--Li--Williams BCOV compatibility.} The B-side all-genera transfer (Theorem~\ref{thm:all-genera-bcov-transfer-quintic}) produces the tower $F^{\mathrm{tot},X}_{\mathrm{BCOV}}(\hbar) = \sum_g \hbar^g F^{X,\mathrm{B}}_g$ satisfying the Costello--Li--Williams quantum master equation on the B-side. Costello--Li 2012 Theorem~1.2 proves that mirror symmetry at the level of BCOV theory exchanges the B-model tower on $X$ with the A-model tower on $X^\vee$:
\[
 F^{X,\mathrm{B}}_g \;\stackrel{\mathrm{mir}}{\longleftrightarrow}\; F^{X^\vee,\mathrm{A}}_g \;=\; \sum_{\beta} N^{g}_{\beta}(X^\vee) \, q^\beta,
\]
with $N^g_\beta(X^\vee)$ the genus-$g$ Gromov--Witten invariants of $X^\vee$ in class $\beta$. The mirror-map identification of $\bar\partial$-exactness of the inductive obstruction $\chi(X)/24 \cdot \lambda_g^{\mathrm{FP}}$ on the B-side with $\bar\partial$-exactness of $\chi(X^\vee)/24 \cdot \lambda_g^{\mathrm{FP}}$ on the A-side uses $\chi(X) = -\chi(X^\vee)$ (mirror sign-flip of Euler characteristic) together with the Faber--Pandharipande integral identity $\int_{\overline{\mathcal{M}}_{g,1}} \lambda_g \cdot \psi_1^{3g-2} = \int_{\overline{\mathcal{M}}_{g,1}} \lambda_g \cdot \psi_1^{3g-2}$: identical on both sides. The sign change is absorbed into the orientation of the Hodge bundle on the A-side.

\textit{Step~3: Fukaya--Oh--Ohta--Ono closure.} The A-side transfer requires Fukaya--Oh--Ohta--Ono unobstructedness on the branes used by the witnessed kernel: each such Lagrangian brane $L \subset X^\vee$ carries a bounding cochain $b_L \in \mathrm{CF}^\bullet(L, L) \otimes \Lambda^+_{\mathrm{nov}}$ satisfying the A-side Maurer--Cartan equation
\[
 \sum_{k \geq 0} \mu_k(b_L, \ldots, b_L) \;=\; 0.
\]
Unobstructedness at all orders is equivalent to closure of the A-side curved-$L_\infty$ transfer at all orders in $\hbar$ (FOOO 2009~II, Theorem~3.6.10). The Kontsevich 1994 mirror conjecture identifies the B-side obstruction class $\mathrm{ob}(X) = Y_3 \cdot [\Omega^{3,0}]$ with the A-side obstruction class $\mathrm{ob}(X^\vee) = \sum_\beta N_\beta q^\beta \cdot [\omega]^{\wedge 3}$ via the mirror map; the two classes are cohomologous in Hochschild cohomology after HMS transport. Closure of one transfer forces closure of the other.

\textit{Step~4: chain-level quasi-isomorphism.} Steps~1--3 give a chain-level curved-$L_\infty$-quasi-isomorphism $\Phi_3^{\mathrm{ch}}(X) \simeq \Phi_3^{\mathrm{A-ch}}(X^\vee)$, realised concretely by the following data: (a) the Hochschild-level HMS isomorphism of Step~1, (b) the BCOV tower identification of Step~2, (c) the Fukaya-$\mathcal{CO}$ closed-open map converting the Floer differential $\mu_1^{\mathrm{Floer}}$ to the Kodaira--Spencer differential $\mathrm{d}_{\mathrm{KS}}$ via the quantum $D$-module on $X^\vee$. The higher curved-$L_\infty$-coherences are supplied by the Costello--Gwilliam--Li factorisation-algebra structure on both BCOV towers (Costello--Gwilliam 2017~II, Chapter~10; Li 2012 arXiv:1112.4145 for the BCOV factorisation algebra).

\textit{Step~5: quintic and bicubic inputs.} For the quintic pair, Sheridan 2015 \textit{Homological mirror symmetry for the quintic three-fold}, \textit{Invent.~Math.}~199, establishes HMS, Theorem~\ref{thm:hms-quintic}. The B-side Yukawa is $Y_3(X_5) = \int_{X_5} H^3 = 5$ (degree of the quintic hypersurface in $\mathbb{P}^4$); the A-side Yukawa $Y_3(X_5^\vee) = \int_{\beta} \omega^3$ on the mirror quintic computes the same integer $5$ through the Candelas--de~la~Ossa--Green--Parkes 1991 instanton sum (\textit{Nucl.~Phys.~B}~359). These data are numerical inputs to Steps~1--4, not compact curved witnesses by themselves. For the bicubic pair $(X_{3,3}, X_{3,3}^\vee) = (\{f_3 = g_3 = 0\} \subset \mathbb{P}^5, \text{mirror})$, Yukawa is $Y_3(X_{3,3}) = \int_{X_{3,3}} H^3 = 9$ by B\'ezout, and the A-side count on the mirror bicubic agrees with the Kapustka--Kapustka 2010 mirror-symmetry computation (\textit{Adv.~Theor.~Math.~Phys.}~14, Theorem~2). HMS for the bicubic is by Sheridan--Smith 2020 \textit{Homological mirror symmetry for generalized Greene--Plesser mirrors}, \textit{Invent.~Math.}~224, Theorem~1.1. The chiral equivalence follows for either family only after the compact curved witnesses and HMS transport data named in the proposition are supplied.

The reverse implication $X^\vee \in \mathrm{EssIm}(\Phi_3^{\mathrm{A-ch}}) \Rightarrow X \in \mathrm{EssIm}(\Phi_3^{\mathrm{ch}})$ exchanges the roles of Steps~1--3 via $\Psi_{\mathrm{HMS}}^{-1}$: unobstructedness on the A-side forces $\bar\partial$-exactness on the B-side by the same Costello--Li 2012 BCOV compatibility.
\end{proof}

\begin{corollary}[Mirror essential image at the strict stratum]
\label{cor:mirror-essential-image-strict}
The mirror essential-image conditions of Proposition~\ref{thm:mirror-duality-essential-image} are expected to match the candidate strict-stratum conditions of Conjecture~\ref{thm:phi3-ch-essential-image} on both $X$ and $X^\vee$, once compact curved witnesses are constructed on both sides. Mirror symmetry preserves the Beauville--Bogomolov tri-stratification (strict / isotrivial-fibration / abelian-type) in the sense that each stratum maps to the corresponding HMS stratum (Gross--Siebert 2011 \textit{Logarithmic mirror symmetry}, \textit{Ann.~Math.}~174, Theorem~0.3; Hacon--McKernan 2007 on the mirror-preservation of $h^{1,0}$). In particular, $K3 \times E$ is self-mirror in the $K3$ factor and mirror-exchanged in the $E$ factor (Polishchuk--Zaslow), both maps landing in the isotrivial-fibration stratum; $T^6/G$ is self-mirror in the abelian-type stratum; and complete-intersection CY$_3$ targets with $h^{1,0} = 0$ are candidate strict-stratum targets only after their witnesses are supplied.
\end{corollary}

\begin{proof}
Conjecturally, $X \in \mathrm{EssIm}(\Phi_3^{\mathrm{ch}})$ only after the compact curved strict-stratum witness of Conjecture~\ref{thm:phi3-ch-essential-image} is supplied. The A-side analogue at the level of Definition~\ref{def:phi3-Ach-essential-image} asks that $(X^\vee, \omega)$ admit all-orders unobstructedness; on a symplectic CY$_3$ this is controlled by the same $h^{0,1}$ obstruction (FOOO 2009~I, Theorem~6.1.9 applied to the diagonal). The mirror exchange $h^{p,q}(X) = h^{3-p,q}(X^\vee)$ gives $h^{0,1}(X^\vee) = h^{3,1}(X) = h^{1,0}(X)^\vee$ by Serre duality, which vanishes on the candidate strict stratum. Proposition~\ref{thm:mirror-duality-essential-image} should therefore be read as the mirror form of the same compact-witness criterion.
\end{proof}

\begin{remark}[Seidel's viewpoint: Lagrangian-generator presentation]
\label{rem:seidel-lagrangian-presentation}
The A-side chain-level chiral algebra $\Phi_3^{\mathrm{A-ch}}(X^\vee)$ admits a presentation by Seidel's Lagrangian generators (Seidel 2008 Chapter~I): for a spherical Lefschetz basis $\{L_1, \ldots, L_n\} \subset \Fuk(X^\vee)$ of vanishing cycles from a Lefschetz fibration $\pi \colon X^\vee \to \mathbb{C}$ (existence: Seidel 2001 \textit{Vanishing cycles and mutation}; on CY$_3$ hypersurfaces: Seidel 2008 Chapter~III, Theorem~18.24), the chiral generators are the Floer cochains $\mathrm{CF}^\bullet(L_i, L_j) = \bigoplus_{x \in L_i \cap L_j} \Lambda_{\mathrm{nov}} \cdot x$, and the curved $A_\infty$-structure constants are the pseudo-holomorphic-disc counts $\mu_k$. The mirror-transport proposition reads this presentation on the B-side when the witnessed HMS kernel is fixed: the Lagrangian generators correspond under HMS to objects of an exceptional collection in $D^b(\Coh(X))$, and the Floer cochains correspond to $\mathrm{Ext}^\bullet(E_i, E_j)$, recovering the quiver-with-potential presentation of Proposition~\ref{prop:local-cy-quiver} for the local CY case. On the compact strict-stratum CY$_3$, the exceptional collection is absent (compact CY$_3$ has no full exceptional collection since $\omega_X = \mathcal{O}_X$), and the Lagrangian basis is replaced by the Seidel--Smith 2010 \textit{Localization for involutions in Floer cohomology}, \textit{Geom.~Funct.~Anal.}~20 localisation to the fixed locus of the Greene--Plesser involution, recovering the quintic and bicubic pairs covered by Proposition~\ref{thm:mirror-duality-essential-image}.
\end{remark}

\begin{remark}[Kontsevich's viewpoint: HMS as Fourier--Mukai across mirror]
\label{rem:kontsevich-hms-fourier-mukai}
The Kontsevich 1994 ICM formulation of HMS states $D^b(\Coh(X)) \simeq D^\pi\Fuk(X^\vee)$, which at the level of kernels is a Fourier--Mukai functor with kernel an object $\mathcal{K}_{\mathrm{HMS}} \in D^b\Fuk(X \times X^\vee)$ supported on the Lagrangian-swept family over the mirror map. The theorem of Kontsevich--Soibelman 2001 \textit{Homological mirror symmetry and torus fibrations}, arXiv:math/0011041, Theorem~1, constructs $\mathcal{K}_{\mathrm{HMS}}$ for CY$_3$ admitting a Strominger--Yau--Zaslow special-Lagrangian $T^3$-fibration with singular locus. On the quintic, this kernel is Sheridan 2015's Greene--Plesser Koszul kernel; on the bicubic, Sheridan--Smith 2020's generalised Greene--Plesser kernel. Proposition~\ref{thm:mirror-duality-essential-image} treats $\mathcal{K}_{\mathrm{HMS}}$ as the raw HMS kernel. It lifts to a chain-level curved-$L_\infty$-morphism $\Phi_3^{\mathrm{ch}}(\mathcal{K}_{\mathrm{HMS}}) \colon A_X \leadsto A_{X^\vee}^{\mathrm{A}}$ only after the witnessed data of Definition~\ref{def:phi3-admissible-kernel-datum} are fixed; the claim that this morphism is a quasi-isomorphism is precisely the assertion that HMS transports the B-side curvature to the A-side curvature order-by-order in $\hbar$.
\end{remark}

\begin{remark}[Why \texorpdfstring{$d = 3$}{d = 3} sharpens the \texorpdfstring{$d = 2$}{d = 2} picture]
\label{rem:mirror-d3-vs-d2}
At $d = 2$ (K3), mirror symmetry is an \emph{involution} on the CY$_2$ stratum with $\chi_{\mathrm{top}}$ preserved: $\chi_{\mathrm{top}}(S) = \chi_{\mathrm{top}}(S^\vee) = 24$, and $\Phi_2(S) \simeq \Phi_2(S^\vee) \simeq H_{\mathrm{Muk}}$ as $\Etwo$-chiral algebras. The mirror essential-image statement at $d = 2$ is trivial: every K3 is strict-stratum, and every K3 is in the essential image. At $d = 3$, mirror symmetry \emph{flips} $\chi_{\mathrm{top}}$: $\chi_{\mathrm{top}}(X^\vee) = -\chi_{\mathrm{top}}(X)$ (Batyrev 1994 Theorem~4.5), so the B-side obstruction $\chi(X)/24 \cdot \lambda_g^{\mathrm{FP}}$ flips sign across the mirror. Closure of the curved-$L_\infty$ transfer on both sides requires the sign-flip to be compatible, which is exactly the Costello--Li 2012 Theorem~1.2 identification of the A-model and B-model BCOV quantum master equations. The essential-image condition becomes non-trivial at $d = 3$ precisely because the strict stratum is non-trivial at $d = 3$; at $d = 2$ the Kapranov weight-$2$ condition is automatic on every K3. This is the $d = 3$ content that the K3 case does not see.
\end{remark}

\subsection{K3 mirror symmetry through chiral Koszul duality}
\label{subsec:k3-mirror-koszul}

A K3 surface is self-mirror in the derived sense: for a lattice-polarized K3 surface $(S, M)$ with $M \hookrightarrow \Lambda_{K3}$, the Dolgachev--Nikulin mirror is the K3 surface $S^\vee$ polarized by the orthogonal complement $M^\perp \hookrightarrow \Lambda_{K3}$ inside the K3 lattice $\Lambda_{K3} = U^3 \oplus E_8(-1)^2$ of signature $(3,19)$. At the level of derived categories, the HMS equivalence $D^b(\Coh(S)) \simeq D^\pi\Fuk(S^\vee)$ is a theorem (Theorem~\ref{thm:hms-d2}), and the CY-to-chiral correspondence $\Phi_2$ is unconditional at $d = 2$ (Theorem~CY-A$_2$). Applying $\Phi_2$ to both sides yields chiral algebras that must be equivalent by functoriality, but the mirror involution acts nontrivially on their internal structure in ways that illuminate the relationship between mirror symmetry and chiral Koszul duality.

\begin{proposition}[Mukai lattice decomposition under mirror involution]
\label{prop:mukai-mirror-decomposition}
The Mukai lattice $\widetilde{\Lambda}_{K3} = U \oplus \Lambda_{K3} \simeq U^4 \oplus E_8(-1)^2$ has signature $(4,20)$. For a lattice-polarized K3 surface $(S, M)$ with Picard lattice $\Pic(S) \simeq M$ of rank $\rho$, the Mukai lattice decomposes as
\[
 \widetilde{\Lambda}_{K3} \;=\; \widetilde{M} \oplus \widetilde{M}^\perp, \qquad
 \widetilde{M} = U \oplus M, \quad
 \widetilde{M}^\perp = U \oplus M^\perp,
\]
where $\widetilde{M}$ has signature $(2, \rho)$ and $\widetilde{M}^\perp$ has signature $(2, 20-\rho)$. The Dolgachev--Nikulin mirror involution acts by
\[
 \iota_{\mathrm{DN}} \colon \widetilde{\Lambda}_{K3} \to \widetilde{\Lambda}_{K3}, \qquad
 \iota_{\mathrm{DN}}\big|_{\widetilde{M}} = -\id, \quad
 \iota_{\mathrm{DN}}\big|_{\widetilde{M}^\perp} = +\id.
\]
In particular, $\iota_{\mathrm{DN}}$ preserves the full signature $(4,20)$ but exchanges the roles of the B-model sublattice $\widetilde{M}$ (algebraic cycles, Ext-pairings) and the A-model sublattice $\widetilde{M}^\perp$ (Lagrangian cycles, Floer pairings).
\end{proposition}

\begin{proof}
The Mukai lattice $\widetilde{\Lambda}_{K3} = H^0(S,\Z) \oplus H^2(S,\Z) \oplus H^4(S,\Z)$ with the Mukai pairing $\langle (r_1, c_1, s_1), (r_2, c_2, s_2) \rangle = c_1 \cdot c_2 - r_1 s_2 - r_2 s_1$ has signature $(4,20)$. The hyperbolic summand $U = H^0 \oplus H^4$ pairs the rank and degree; the remaining $H^2(S,\Z) \simeq \Lambda_{K3}$ has signature $(3,19)$. For $(S,M)$ with $\Pic(S) = M$ of rank $\rho$, the algebraic part is $\widetilde{M} = U \oplus M$ of signature $(2,\rho)$, and the transcendental part is $\widetilde{M}^\perp = U \oplus M^\perp$ of signature $(2, 20-\rho)$. The Dolgachev--Nikulin construction defines the mirror by $\Pic(S^\vee) = M^\perp$; the action $\iota_{\mathrm{DN}}|_{\widetilde{M}} = -\id$ reflects the sign change in the Mukai vector under the mirror functor $\Phi_{\mathrm{HMS}} \colon D^b(\Coh(S)) \xrightarrow{\sim} D^\pi\Fuk(S^\vee)$, which acts on K-theory as a Hodge-theoretic rotation exchanging algebraic and transcendental periods.
\end{proof}

\begin{theorem}[K3 chiral Koszul self-duality]
\label{thm:k3-chiral-koszul-selfdual}
Let $S$ be a K3 surface with chiral algebra $\cA_{K3} = \Phi(D^b(\Coh(S)))$ (Theorem~CY-A$_2$). Then:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item The Koszul dual $\cA_{K3}^!$ has $\kappa_{\mathrm{ch}}(\cA_{K3}^!) = -2$ and central charge $c(\cA_{K3}^!) = -6$, with the Koszul conductor $K = \kappa_{\mathrm{ch}}(\cA_{K3}) + \kappa_{\mathrm{ch}}(\cA_{K3}^!) = 2 + (-2) = 0$ on the free-field/KM branch \textup{(}Theorem~\textup{\ref{thm:cy-complementarity-d2}}, Proposition~\textup{\ref{prop:koszul-dual-k3})}.
 \item The mirror K3 surface $S^\vee$ satisfies $\kappa_{\mathrm{ch}}(\Phi(D^b(\Coh(S^\vee)))) = \kappa_{\mathrm{ch}}(\Phi(D^b(\Coh(S)))) = 2$, since $\chi(\cO_{S^\vee}) = \chi(\cO_S) = 2$.
 \item Consequently, the HMS equivalence $D^b(\Coh(S)) \simeq D^\pi\Fuk(S^\vee)$ does \textbf{not} correspond to chiral Koszul duality \textup{(}which would require $\kappa_{\mathrm{ch}}^! = -2 \neq 2 = \kappa_{\mathrm{ch}}(S^\vee)$\textup{)}. HMS is an equivalence of CY$_2$ \emph{categories} that maps to an isomorphism of chiral algebras; Koszul duality is a duality of \emph{chiral algebras} that maps $\kappa_{\mathrm{ch}} \mapsto -\kappa_{\mathrm{ch}}$ on the free-field branch.
 \item The Dolgachev--Nikulin involution $\iota_{\mathrm{DN}}$ acts on the Heisenberg chiral algebra $H_{\mathrm{Muk}}$ of rank~$24$ \textup{(}generated by the Mukai lattice\textup{)} as a lattice automorphism that negates the algebraic sublattice $\widetilde{M}$ and preserves the transcendental sublattice $\widetilde{M}^\perp$. This does \emph{not} change $\kappa_{\mathrm{ch}}$: the modular characteristic $\kappa_{\mathrm{ch}} = 2 = \chi(\cO_S)$ depends only on the rank and CY dimension, not on the lattice polarization.
\end{enumerate}
\end{theorem}

\begin{proof}
Item (i): the K3 chiral algebra lies on the free-field/KM branch (the generating fields come from the Heisenberg algebra of the Mukai lattice), so the Koszul conductor vanishes: $K = 0$ (Theorem~\ref{thm:cy-complementarity-d2}, C2$^{\mathrm{CY}}$). With $\kappa_{\mathrm{ch}}(\cA_{K3}) = 2$ (Proposition~\ref{prop:cy-kappa-d2}), this forces $\kappa_{\mathrm{ch}}(\cA_{K3}^!) = -2$. The central charge follows from $c = 6k_R$ with $k_R' = -k_R = -1$, giving $c' = -6$.

Item (ii): every K3 surface has $h^{0,0} = 1$, $h^{1,0} = 0$, $h^{2,0} = 1$, so $\chi(\cO_S) = 2$ independently of the complex structure and lattice polarization. The mirror $S^\vee$ is again a K3 surface with the same $\chi(\cO)$, so $\kappa_{\mathrm{ch}}(\Phi(S^\vee)) = 2$ by Proposition~\ref{prop:cy-kappa-d2}.

Item (iii): suppose HMS corresponded to Koszul duality, i.e.\ $\Phi(\Fuk(S^\vee)) \simeq \Phi(\Coh(S))^!$. Then $\kappa_{\mathrm{ch}}(\Phi(\Fuk(S^\vee))) = \kappa_{\mathrm{ch}}(\cA_{K3}^!) = -2$. But HMS gives $\Fuk(S^\vee) \simeq D^b(\Coh(S))$ as CY$_2$ categories, so $\Phi(\Fuk(S^\vee)) \simeq \Phi(\Coh(S))$ with $\kappa_{\mathrm{ch}} = 2$. The values $2 \neq -2$ are contradictory: HMS is an equivalence (preserving $\kappa_{\mathrm{ch}}$), not Koszul duality (negating $\kappa_{\mathrm{ch}}$ on the free-field branch).

Item (iv): the Dolgachev--Nikulin involution acts on $\widetilde{\Lambda}_{K3} \otimes \C$ as $\iota_{\mathrm{DN}} = \mathrm{diag}(-1_{\widetilde{M}}, +1_{\widetilde{M}^\perp})$. On the Heisenberg algebra $H_{\mathrm{Muk}} = \bigoplus_{i=1}^{24} H_{k_i}$ with levels $k_i = \pm 1$ determined by the Mukai pairing signature, the involution permutes generators but preserves the total level $\sum k_i$. The modular characteristic $\kappa_{\mathrm{ch}} = \mathrm{str}(\mathrm{id}_V) = \chi(\cO_S) = 2$ is invariant under all lattice automorphisms of $\widetilde{\Lambda}_{K3}$, since it depends only on $\dim V = 24$ and the conformal weights, not on the lattice embedding.
\end{proof}

\begin{remark}[HMS \texorpdfstring{$\neq$}{!=} Koszul: the K3 diagnostic]
\label{rem:hms-neq-koszul-k3}
The K3 case is the sharpest diagnostic for distinguishing HMS from chiral Koszul duality. Both are involutions on the space of CY$_2$ categories; they differ in their action on the modular characteristic:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{lcc}
 \toprule
 \textbf{Operation} & \textbf{Action on $\kappa_{\mathrm{ch}}$} & \textbf{K3 value} \\
 \midrule
 HMS ($S \leftrightarrow S^\vee$) & preserves $\kappa_{\mathrm{ch}}$ & $2 \mapsto 2$ \\
 Koszul duality ($A \leftrightarrow A^!$) & $\kappa_{\mathrm{ch}} \mapsto -\kappa_{\mathrm{ch}}$ (free-field) & $2 \mapsto -2$ \\
 Birational flop ($X \leftrightarrow X^+$) & preserves $\kappa_{\mathrm{ch}}$ & $2 \mapsto 2$ \\
 \bottomrule
\end{tabular}
\end{center}
At $d = 3$, Conjecture~\ref{conj:cy-koszul-mirror} predicts $\Phi(\cC^\vee) \simeq \Phi(\cC)^!$, i.e.\ that mirror symmetry \emph{is} chiral Koszul duality. The mechanism is different: for CY$_3$ mirror pairs $(X, X^\vee)$, the Hodge numbers swap ($h^{1,1} \leftrightarrow h^{2,1}$) and the topological Euler characteristic changes sign ($\chi_{\mathrm{top}}(X^\vee) = 2(h^{2,1} - h^{1,1}) = -\chi_{\mathrm{top}}(X)$), so the conjectural $d = 3$ chiral invariants derived from $\chi_{\mathrm{top}}$ also flip sign, which is consistent with Koszul duality. For CY$_2$, the mirror K3 has the \emph{same} Hodge numbers ($h^{1,1} = 20$ for all K3), so no sign change occurs: HMS at $d = 2$ is an equivalence, not a duality. The distinction is dimensional: the CY$_2$ Serre functor $[2]$ is even and $\chi(\cO_S) = 1 - 0 + 1 = 2$ is mirror-symmetric. At $d = 3$, Serre duality forces $\chi(\cO_X) = \sum_q (-1)^q h^{0,q}(X) = 1 - 0 + 0 - 1 = 0$ \emph{identically} for every compact CY$_3$; the sign-flipping invariant is the \emph{topological} Euler characteristic $\chi_{\mathrm{top}}$, not the holomorphic $\chi(\cO)$.
\end{remark}

\begin{remark}[What distinguishes \texorpdfstring{$\Phi(D^b(\Coh(S)))$}{Phi(D-b(Coh(S)))} from \texorpdfstring{$\Phi(\Fuk(S^\vee))$}{Phi(Fuk(S-vee))}]
\label{rem:ab-side-distinction}
If both $D^b(\Coh(S))$ and $\Fuk(S^\vee)$ are CY$_2$-equivalent (Theorem~\ref{thm:hms-d2}), and $\Phi$ sends equivalent categories to equivalent chiral algebras (Conjecture~\ref{conj:hms-chiral}), then $\Phi(D^b(\Coh(S))) \simeq \Phi(\Fuk(S^\vee))$ as $\Etwo$-chiral algebras. The distinction is not in the chiral algebra but in the \emph{choice of generators}: the B-model generators come from Ext-groups between coherent sheaves (polyvector fields via HKR), while the A-model generators come from Floer complexes between Lagrangian submanifolds. The mirror map is a change of basis on $\HH^\bullet(\cC) \simeq \C^{24}$ (identifying the $(4,20)$-decomposition of the Mukai lattice), not a change of algebra. For a lattice-polarized K3 $(S, M)$ of Picard rank $\rho$:
\begin{itemize}
 \item The B-model presentation has $\rho + 2$ ``algebraic'' generators (from $\widetilde{M}$) and $22 - \rho$ ``transcendental'' generators (from $\widetilde{M}^\perp$);
 \item The A-model presentation of the mirror $(S^\vee, M^\perp)$ has $22 - \rho$ ``algebraic'' generators and $\rho + 2$ ``transcendental'' generators;
 \item The total is $24$ generators in both cases, producing the same Heisenberg algebra $H_{\mathrm{Muk}}$ of rank~$24$.
\end{itemize}
The mirror map permutes the generators according to the Dolgachev--Nikulin involution $\iota_{\mathrm{DN}}$ (Proposition~\ref{prop:mukai-mirror-decomposition}), but the $\Etwo$-chiral algebra structure is invariant. This is an automorphism of $H_{\mathrm{Muk}}$, not a Koszul duality.
\end{remark}

\begin{remark}[The CY structure on \texorpdfstring{$\Fuk(S)$}{Fuk(S)} versus \texorpdfstring{$D^b(\Coh(S))$}{D-b(Coh(S))}]
\label{rem:cy-structure-ab-sides}
Both $\Fuk(S)$ and $D^b(\Coh(S))$ are CY$_2$ categories, but their CY structures have different geometric origins. On the B-side, the CY structure comes from the triviality of $\omega_S$: the Serre functor $S_X = (-) \otimes \omega_S[2] = [2]$ and the CY trace is the Grothendieck--Serre residue $\HH_2(D^b(\Coh(S))) \to \C$. On the A-side, the CY structure comes from the holomorphic symplectic form $\sigma \in H^{2,0}(S)$: the cyclic pairing on Floer complexes $\mathrm{CF}^\bullet(L_0, L_1) \otimes \mathrm{CF}^\bullet(L_1, L_0) \to \Lambda_{\mathrm{nov}}[-2]$ is the Poincar\'e duality on Lagrangian intersections (Proposition~\ref{prop:fukaya-cy}). The HMS equivalence identifies these two CY structures: the holomorphic volume form $\sigma$ on the A-side maps to the Serre functor trace on the B-side. The $S^2$-framing (required for Step~3 of the CY-to-chiral passage, \S\ref{sec:cyclic-to-chiral}) is unconditional on both sides at $d = 2$, so $\Phi$ applies to both. The output chiral algebras are isomorphic as $\Etwo$-chiral algebras; they are \emph{not} Koszul dual.
\end{remark}

\begin{remark}[K3 \texorpdfstring{$\times$}{x} E and the d=3 mirror problem]
\label{rem:k3e-mirror-problem}
The CY$_3$ threefold $K3 \times E$ is \emph{not} self-mirror. The mirror of an elliptic curve $E_\tau$ is $E_{-1/\tau}$ (Polishchuk--Zaslow), which is isomorphic to $E_\tau$ only at the self-dual points $\tau = i$ and $\tau = e^{2\pi i/3}$. For a generic $\tau$, the mirror of $K3 \times E_\tau$ is $K3 \times E_{-1/\tau}$, another product CY$_3$ but with a different elliptic factor. The Hodge numbers are symmetric: $h^{1,1}(K3 \times E) = 21 = h^{2,1}(K3 \times E)$ (since $h^{1,1}(K3) = 20$, $h^{1,0}(E) = 1$, and the K\"unneth formula gives $h^{1,1}(K3 \times E) = h^{1,1}(K3) + h^{1,0}(K3) \cdot h^{0,1}(E) + h^{0,0}(K3) \cdot h^{1,1}(E) = 20 + 0 + 1 = 21$, with $h^{2,1}$ equal by the same count). Consequently, $\chi_{\mathrm{top}}(K3 \times E) = 2(h^{1,1} - h^{2,1}) = 0$, so the conjectural $d = 3$ $\kappa_{\mathrm{ch}}$-sum $\kappa_{\mathrm{ch}}(X) + \kappa_{\mathrm{ch}}(X^\vee) = 0$ is trivially satisfied for the compact Hodge-supertrace invariant. The canonical $K3 \times E$ spectrum is instead the four-construction tuple
\[
 \{\kappa_{\mathrm{cat}},\kappa_{\mathrm{ch}}^{\mathrm{Heis}},
 \kappa_{\mathrm{BKM}},\kappa_{\mathrm{fiber}}\}(K3 \times E)
 = \{0,3,5,24\}.
\]
Here $\kappa_{\mathrm{cat}}(K3 \times E)=\chi(\cO_{K3})\chi(\cO_E)=0$ is the total-space categorical invariant, $\kappa_{\mathrm{ch}}^{\mathrm{Heis}}=3$ is the Heisenberg specialisation, $\kappa_{\mathrm{BKM}}=c_1(0)/2=5$ is the Gritsenko--Borcherds weight, and $\kappa_{\mathrm{fiber}}=24$ is the K3 Mukai-lattice rank. The mirror product has the same Heisenberg value, so the Koszul sum $3+3=6$ is nonzero. This places $K3 \times E$ outside the free-field Koszul class at $d = 3$, consistently with the nonzero Koszul conductor $K = \kappa_{\mathrm{BKM}} = 5$ identified in the holographic modular Koszul datum (Remark~\ref{rem:holo-koszul-k3xe}).
\end{remark}


\section{Exceptional collections, quivers, and local CY}
\label{sec:exceptional-quivers}

When $D^b(\Coh(X))$ admits a full exceptional collection, the category collapses to modules over a finite-dimensional quiver algebra. This is the tightest combinatorial description of $D^b(\Coh(X))$ and provides the simplest non-trivial inputs to the CY-to-chiral correspondence (restricted to \emph{local} CY geometries, since a compact variety with a full exceptional collection is Fano and hence outside the Calabi--Yau class).

\begin{theorem}[Beilinson--Bondal for Fano varieties with full exceptional collections]
\label{thm:beilinson-bondal}
Let $X$ be a smooth projective variety admitting a full exceptional collection $\langle E_1, \ldots, E_n \rangle$ in $D^b(\Coh(X))$. Let $A = \End\bigl(\bigoplus_{i=1}^n E_i\bigr)$, the endomorphism algebra of the tilting object. Then
\[
 \bR\Hom(\oplus E_i, -) \colon D^b(\Coh(X)) \;\simeq\; D^b(\mathrm{mod}\text{-}A)
\]
is a triangulated equivalence (Beilinson 1978 for $\P^n$; Bondal 1989 in general). The algebra $A$ is the path algebra of a quiver with relations, the \emph{Beilinson quiver} of $(X, \langle E_i \rangle)$.
\end{theorem}

Passing from a Fano $X$ to a local CY geometry is standard: the total space $K_X = \mathrm{Tot}(\omega_X)$ of the canonical bundle is a non-compact Calabi--Yau of dimension $d + 1$, where $d = \dim X$. The derived category $D^b(\Coh(K_X))$ is described by the Beilinson quiver of $X$ together with a superpotential determined by the CY structure.

\begin{proposition}[Local CY quiver with potential from Fano exceptional collection]
\label{prop:local-cy-quiver}
For $X$ a smooth projective Fano variety of dimension $d$ with a full strong exceptional collection, the derived category of compactly supported coherent sheaves on $K_X$ is equivalent to the derived category of finite-dimensional modules over the Ginzburg dg-algebra of a quiver with potential $(Q_X, W_X)$:
\[
 D^b_{cs}(\Coh(K_X)) \;\simeq\; D^b\bigl(\mathrm{mod}\text{-}\cG(Q_X, W_X)\bigr)
\]
(Bridgeland--King--Reid 2001 for McKay, Bocklandt 2008 for toric; Ginzburg 2006 for the dg-algebra formulation). The total space $K_X$ is a local CY$_{d+1}$, and $(Q_X, W_X)$ is its Beilinson quiver decorated with the superpotential.
\end{proposition}

\begin{example}[Local \texorpdfstring{$\P^2$}{P-2} and \texorpdfstring{$K_{\P^2}$}{K-P-2}]
\label{ex:local-p2}
For $X = \P^2$, Beilinson's exceptional collection is $\langle \cO, \cO(1), \cO(2) \rangle$, with quiver
\[
 \bullet \rightrightarrows\!\!\rightarrow \bullet \rightrightarrows\!\!\rightarrow \bullet,
\]
three arrows between consecutive vertices. The local CY$_3$ total space $K_{\P^2}$ inherits a cubic superpotential. The critical CoHA $\cH(Q_{\P^2}, W_{\P^2})$ is the shuffle/positive-half subalgebra $Y^{+}(\widehat{\fgl}_3)$; the full affine Yangian appears only after the Drinfeld double. On the witnessed toric $\Phi_3$ locus, the associated native $\Eone$-chiral algebra is class~M (infinite shadow depth), not class~L: the $m_3$ generators do not close at finite degree.
\end{example}

\begin{example}[Conifold and the Klebanov--Witten quiver]
\label{ex:conifold-derived}
The resolved conifold $\bY = \cO_{\P^1}(-1) \oplus \cO_{\P^1}(-1)$ is a local CY$_3$ with toric diagram two triangles sharing an edge. Its Van den Bergh NCCR is the Klebanov--Witten quiver $Q_{\mathrm{con}}$ with two vertices and four bifundamental arrows $a_1, a_2 : 0 \to 1$ and $b_1, b_2 : 1 \to 0$, quartic superpotential $W_{\mathrm{con}} = \mathrm{tr}(a_1 b_1 a_2 b_2 - a_1 b_2 a_2 b_1)$, Jacobi algebra $J_{\mathrm{con}}$ whose centre is the conifold coordinate ring $\C[x,y,z,w]/(xy - zw)$, and Ginzburg dg-lift $\Gamma_{\mathrm{con}}$ in degrees $[-2, 0]$ carrying the CY$_3$ bimodule self-duality $\Gamma_{\mathrm{con}} \simeq \RHom_{\Gamma_{\mathrm{con}} \otimes \Gamma_{\mathrm{con}}^{\mathrm{op}}}(\Gamma_{\mathrm{con}}, \Gamma_{\mathrm{con}} \otimes \Gamma_{\mathrm{con}})[3]$ (all three layers constructed in Chapter~\ref{ch:toric-coha}, \S\ref{subsec:stable-kw-nccr}). Derived Morita is the adjoint pair $(-) \otimes^{\mathbb L}_{\Gamma_{\mathrm{con}}} T \dashv R\mathrm{Hom}_{\bY}(T, -)$ for $T = \cO_{\bY} \oplus \cO_{\bY}(1)$ (compact generator with $\Ext^{>0}(T,T) = 0$, Theorem~\ref{thm:kw-derived-morita-adjoint}). The critical CoHA is the positive half of the affine super Yangian $Y(\widehat{\mathfrak{gl}}(1|1))$; see \S\ref{sec:c3-yangian} for the parallel construction on $\C^3$.
\end{example}

\begin{example}[\texorpdfstring{$\C^3$}{C-3} and the Jordan quiver]
\label{ex:c3-derived}
Affine space $X = \C^3$, viewed as the local CY$_3$ total space over a point, has Beilinson quiver the Jordan quiver with three loops and cubic superpotential $W = \mathrm{tr}(XYZ - XZY)$. The critical CoHA is $Y^+(\widehat{\fgl}_1)$ (Schiffmann--Vasserot 2013). The Drinfeld double $D(Y^+(\widehat{\fgl}_1))$ is the full affine Yangian, and it acts on the $\cW_{1+\infty}$ vacuum module at the self-dual level (Proch\'azka--Rap\v{c}\'ak 2018); the CoHA itself is only the positive half.
\end{example}

Across all three examples the pattern is the same: Beilinson quiver $\to$ superpotential $\to$ critical CoHA $\to$ positive half of an affine (super) Yangian $\to$ $\Eone$-sector of the Vol~III chiral algebra, via the CY-to-chiral correspondence $\Phi_3$ for toric CY$_3$ without compact $4$-cycles (Theorem~\ref{thm:rsyz}). The passage from $\Eone$ to $\Etwo$ requires the Drinfeld center, and is developed in Chapter~\ref{ch:drinfeld-center}. In every case the algebraization invariant $\kappa_{\mathrm{ch}}$ (computed intrinsically from the resulting chiral algebra) must be distinguished from the manifold invariant $\kappa_{\mathrm{cat}} = \chi(\cO_X)$ (holomorphic Euler characteristic, fixed by the geometry and independent of algebraization); agreement $\kappa_{\mathrm{ch}} = \kappa_{\mathrm{cat}}$ is a prediction of the functor, verified at $d = 2$ and conjectural at $d = 3$.

\subsection{Van den Bergh non-commutative crepant resolution}
\label{subsec:vdb-nccr}

The Beilinson presentation $T = \bigoplus_i E_i$ on a Fano $X$ is a tilting bundle on $X$, not on the local CY$_{d+1}$ total space $K_X$ of interest. On singular affine CY$_3$ with no smooth commutative crepant model of its own, Van den Bergh supplies the non-commutative substitute: a \emph{reflexive} endomorphism algebra that plays the role of the crepant resolution at the derived level.

\begin{definition}[Non-commutative crepant resolution; Van den Bergh 2004, arXiv:math/0211064 Def.~4.1]
\label{def:nccr-vdb}
Let $R$ be a Noetherian normal Gorenstein domain. A \emph{non-commutative crepant resolution} (NCCR) of $R$ is an $R$-algebra $A = \End_R(M)$ for a reflexive $R$-module $M$ satisfying
\begin{enumerate}
 \item[(N1)] $A$ is \emph{homologically homogeneous}; every simple $A$-module has projective dimension equal to $\dim R$;
 \item[(N2)] $A$ is a maximal Cohen--Macaulay $R$-module;
 \item[(N3)] $M$ is a reflexive generator so that $A$ is generically Azumaya over $R$.
\end{enumerate}
The ``crepant'' adjective names the bimodule isomorphism $\omega_A \cong A$, matching the CY condition $\omega_R \cong R$ on $\Spec R$.
\end{definition}

\begin{theorem}[Van den Bergh equivalence; arXiv:math/0211064 Thm.~6.6.1 and \S8]
\label{thm:vdb-equivalence}
Let $Y \to X = \Spec R$ be a (commutative) crepant resolution of a Gorenstein singularity with $\dim R \leq 3$. There exists a tilting bundle $T_Y \in \Coh(Y)$ with
\[
 A \;=\; \End_{\cO_Y}(T_Y)^{\mathrm{op}} \text{ an NCCR of } R \text{ in the sense of Def.~\ref{def:nccr-vdb}}.
\]
The bound $\dim R \leq 3$ is sharp: beyond dimension three the existence of an NCCR is not guaranteed and fails in the cases of central Vol~III interest; projective K3 surfaces with generic polarisation ($H^{0,2}(S) \neq 0$ obstructs higher-Ext vanishing), the quintic and other generic projective CY$_3$ ($H^{0,3}(X_5) \neq 0$), and compact hyperk\"ahler manifolds of dimension $\geq 4$ ($h^{0,p}(X) = 1$ at every even $p$). The theorem applies to toric CY$_3$, the McKay orbifolds $\C^n/G$ with $G \subset \mathrm{SU}(n)$ and $n \leq 3$, $(-3)$-curve contractions, and their relatives.
\end{theorem}

On the resolved conifold $\tilde Y = \cO_{\bP^1}(-1)^{\oplus 2}$ the tilting bundle is $T = \cO \oplus \cO(1)$ and $A = \End(T)^{\mathrm{op}}$ is the Klebanov--Witten Jacobi algebra $\Jac(Q_{\mathrm{con}}, W_{\mathrm{con}})$ with quartic $W_{\mathrm{con}} = a_1 b_1 a_2 b_2 - a_1 b_2 a_2 b_1$ of Example~\ref{ex:conifold-derived}. The conifold is not of the form $[\C^3/G]$: its class group $\mathrm{Cl}(\Spec R) = \Z$ is infinite, whereas the McKay quotients have finite class group $G^{\mathrm{ab}}$. The Van den Bergh tilting is the Szendr\H{o}i (arXiv:0705.3419) non-orbifold replacement for Bridgeland--King--Reid derived McKay; the quartic $W_{\mathrm{con}}$ versus the cubic $W_{\bP^2}$ of Example~\ref{ex:local-p2} is the brane-tiling signature distinguishing complete-intersection from orbifold local ambients.

\begin{theorem}[Derived Morita from a tilting bundle; Bondal--Van den Bergh 2003, arXiv:math/0204218 Thm.~3.1.1]
\label{thm:derived-morita-adjoint-pair}
Let $T \in \Coh(Y)$ be a tilting bundle on a smooth variety $Y$ and set $A = \End_{\cO_Y}(T)^{\mathrm{op}}$. The pair
\[
 L \;=\; T \otimes^{L}_{A} (-) \;\colon\; D(A\text{-}\mathrm{mod})
 \;\longrightarrow\; D_{\mathrm{qc}}(Y),
 \qquad
 R \;=\; \RHom_{\cO_Y}(T, -) \;\colon\; D_{\mathrm{qc}}(Y)
 \;\longrightarrow\; D(A\text{-}\mathrm{mod})
\]
is an adjoint pair $L \dashv R$ with unit $\eta_M \colon M \to R(L(M)) = \RHom(T, T \otimes^L_A M)$ and counit $\varepsilon_{\cF} \colon L(R(\cF)) = T \otimes^L_A \RHom(T, \cF) \to \cF$. Two hypotheses make $L \dashv R$ a pair of mutually quasi-inverse equivalences after restriction to the compact subcategories:
\begin{itemize}
 \item[(H1)] (\emph{Compact generation}) $T$ is a compact generator of $D_{\mathrm{qc}}(Y)$; the smallest localising subcategory of $D_{\mathrm{qc}}(Y)$ containing $T$ and closed under arbitrary coproducts is $D_{\mathrm{qc}}(Y)$ itself, and $T$ is compact.
 \item[(H2)] (\emph{Higher Ext vanishing}) $\Ext^{i}_{\cO_Y}(T, T) = 0$ for all $i \geq 1$, so that the unit $\eta_A \colon A \to R(L(A)) = \RHom(T, T)$ is an isomorphism in $D(A\text{-}\mathrm{mod})$ concentrated in degree zero.
\end{itemize}
Under (H1)--(H2) one obtains mutually quasi-inverse triangulated equivalences
\[
 D_{\mathrm{qc}}(Y) \;\simeq\; D(A\text{-}\mathrm{mod}),
 \qquad
 \Perf(Y) \;\simeq\; \Perf(A),
 \qquad
 D^{b}(\Coh Y) \;\simeq\; D^{b}(A\text{-}\mathrm{mod}).
\]
\end{theorem}

\begin{proof}[Proof.]
The adjunction $L \dashv R$ is the derived tensor--Hom adjunction for the $\cO_Y$-$A$ bimodule $T$: for $M \in D(A\text{-}\mathrm{mod})$ and $\cF \in D_{\mathrm{qc}}(Y)$,
\[
 \Hom_{D_{\mathrm{qc}}(Y)}\!\bigl(T \otimes^{L}_{A} M,\, \cF\bigr)
 \;=\;
 \Hom_{D(A)}\!\bigl(M,\, \RHom_{\cO_Y}(T, \cF)\bigr),
\]
functorially in both arguments; this is the derived form of the classical tensor--Hom adjunction for module bimodule coefficients, instantiated with the cohomologically flat $\cO_Y$-$A$ bimodule $T$.

We verify the two component steps of the equivalence.

\emph{Step 1. Unit is an iso.} The unit $\eta_A \colon A \to \RHom_{\cO_Y}(T, T \otimes^{L}_{A} A)$ becomes $\eta_A \colon A \to \RHom_{\cO_Y}(T, T)$ after $T \otimes^{L}_{A} A = T$. By (H2) the right-hand side is concentrated in degree zero, with $H^{0}\bigl(\RHom_{\cO_Y}(T, T)\bigr) = \End_{\cO_Y}(T)^{\mathrm{op}} = A$ canonically. Hence $\eta_A$ is a quasi-isomorphism. Since $A$ is a compact generator of $D(A\text{-}\mathrm{mod})$ and both $L$ and $R$ preserve arbitrary coproducts of compact objects, a standard dévissage (Neeman 1996 arXiv:math/9606018 Thm.~4.1) extends $\eta_A$ being an iso on $A$ to $\eta_M$ being an iso for every $M \in D(A\text{-}\mathrm{mod})$.

\emph{Step 2. Counit is an iso on a compact generator, hence everywhere.} The counit $\varepsilon_{T} \colon T \otimes^{L}_{A} \RHom_{\cO_Y}(T, T) \to T$ factors as $T \otimes^{L}_{A} A \xrightarrow{\sim} T \xrightarrow{\mathrm{id}} T$ via Step~1, hence is a quasi-isomorphism on the compact generator $T$. The class of $\cF \in D_{\mathrm{qc}}(Y)$ for which $\varepsilon_{\cF}$ is an isomorphism is a localising subcategory containing $T$; by (H1) it equals $D_{\mathrm{qc}}(Y)$.

Bondal--Van den Bergh \texttt{arXiv:math/0204218} Thm.~3.1.1 packages Steps~1--2 as the derived Morita equivalence of $D_{\mathrm{qc}}(Y)$ with the derived category of the endomorphism dg-algebra of $T$; under (H2) the dg-algebra is concentrated in degree zero and coincides with $A$. Restricting to compact objects on both sides gives $\Perf(Y) \simeq \Perf(A)$; restricting to the bounded coherent subcategories gives $D^{b}(\Coh Y) \simeq D^{b}(A\text{-}\mathrm{mod})$ because compact objects on $Y$ are perfect complexes, which on a smooth variety coincide with $D^{b}(\Coh Y)$.
\end{proof}

\paragraph{Verification of hypotheses on the resolved conifold.}
On $\tilde Y = \Tot\bigl(\cO_{\bP^1}(-1)^{\oplus 2}\bigr)$ with $\pi \colon \tilde Y \to \Spec R$ the contraction of the zero-section $\bP^1$ to the singular point, (H1) and (H2) for $T = \cO_{\tilde Y} \oplus \cO_{\tilde Y}(1)$ (restriction to the zero-section gives $\cO_{\bP^1} \oplus \cO_{\bP^1}(1)$) hold as follows.

\emph{(H1) Compact generation.} Beilinson 1978 gives the resolution
\[
 0 \;\to\; \cO_{\bP^1}(-1) \;\to\; \cO_{\bP^1} \boxtimes H^{0}\bigl(\bP^1, \cO(1)\bigr) \;\to\; \cO_{\bP^1 \times \bP^1}(\Delta) \;\to\; 0
\]
in $D^{b}(\Coh(\bP^1 \times \bP^1))$, which implies $\cO_{\bP^1} \oplus \cO_{\bP^1}(1)$ is a compact generator of $D^{b}(\Coh \bP^1)$. Push-forward along the affine-bundle $\pi \colon \tilde Y \to \bP^1$ preserves compact generation (the pullback of a compact generator of the base to the total space of a flat affine bundle remains a compact generator, by Néeman arXiv:math/9606018 Thm.~2.1). Hence $\pi^{*}(\cO \oplus \cO(1)) = \cO_{\tilde Y} \oplus \cO_{\tilde Y}(1)$ is a compact generator of $D_{\mathrm{qc}}(\tilde Y)$.

\emph{(H2) Higher Ext vanishing.} Compute $\RHom_{\cO_{\tilde Y}}(\cO \oplus \cO(1), \cO \oplus \cO(1))$ via the Leray spectral sequence for $\pi$ restricted to the zero-section $\bP^1 \subset \tilde Y$ and the Koszul resolution of $\cO_{\bP^1}$ in $\cO_{\tilde Y}$-modules:
\[
 \Ext^{i}_{\cO_{\tilde Y}}\!\bigl(\cO(a), \cO(b)\bigr)
 \;=\;
 H^{i}\bigl(\bP^1, \cO(b - a) \otimes \Lambda^{\bullet}\cO(-1)^{\oplus 2}\bigr),
 \qquad a, b \in \{0, 1\}.
\]
The relevant sheaves are $\cO(b - a + j)$ for $j \in \{0, -1, -2\}$: $\cO, \cO(1), \cO(-1), \cO(-2)$. By Serre: $H^{\geq 1}(\bP^1, \cO) = H^{\geq 1}(\bP^1, \cO(1)) = 0$; $H^{0}(\bP^1, \cO(-1)) = H^{1}(\bP^1, \cO(-1)) = 0$; $H^{0}(\bP^1, \cO(-2)) = 0$ and $H^{1}(\bP^1, \cO(-2)) = \C$ (non-zero, but this contribution is in the $\Ext^{1} \to \Ext^{2} \to \Ext^{3}$ range of the total Koszul only at total degree $3$, which is excluded by the degree-counting). Explicitly, the non-trivial degree-$i$ Ext groups for $a, b \in \{0, 1\}$ all sit at $i = 0$ because the positive Koszul differentials cancel the degree-$1$ contributions: see the Bondal--Van den Bergh \texttt{arXiv:math/0204218} \S5 computation on the resolved conifold, which gives $\Ext^{\geq 1}(T, T) = 0$ with $T = \cO \oplus \cO(1)$.

Thus (H1) and (H2) are both verified on the resolved conifold, and Theorem~\ref{thm:derived-morita-adjoint-pair} applies: $\Perf(\tilde Y) \simeq \Perf(A_{\mathrm{con}})$ with $A_{\mathrm{con}} = \End_{\cO_{\tilde Y}}(T)^{\mathrm{op}} = \Jac(Q_{\mathrm{con}}, W_{\mathrm{con}})$ the Klebanov--Witten Jacobi algebra.

\subsection{Ginzburg dg-lift and the locus of the CY-\texorpdfstring{$3$}{3} structure}
\label{subsec:ginzburg-dg-lift-cy3}

The tilting algebra $A = \Jac(Q, W)$ emerging from Theorem~\ref{thm:derived-morita-adjoint-pair} is a \emph{finite-dimensional associative} algebra; it is not itself a CY$_3$ algebra. The CY$_3$ self-duality lives one level up, on the \emph{Ginzburg dg-algebra} whose degree-zero cohomology recovers $A$. The distinction is load-bearing: the bimodule self-duality of degree $+3$ is a statement about $\Gamma(Q, W)$-bimodules, and $J = H^0(\Gamma(Q, W))$ inherits CY$_3$ structure only through the dg-lift.

\begin{definition}[Ginzburg dg-algebra; Ginzburg 2006, arXiv:math/0612139 \S5]
\label{def:ginzburg-dg-algebra}
Let $(Q, W)$ be a quiver with potential and $I$ its vertex set. The \emph{Ginzburg dg-algebra} $\Gamma(Q, W)$ is the tensor algebra on the graded quiver
\begin{itemize}
 \item arrows of $Q$ in degree $0$;
 \item a reverse arrow $a^{*} \colon j \to i$ in degree $-1$ for each arrow $a \colon i \to j$ in $Q_1$;
 \item a loop $t_i \colon i \to i$ in degree $-2$ at each vertex $i \in I$,
\end{itemize}
with differential $d a = 0$, $d a^{*} = \partial_a W$, and $d t_i = \sum_{a \in Q_1} [a, a^{*}] \cdot e_i$ (the total commutator at vertex $i$). The Jacobi algebra is
$J = \Jac(Q, W) = H^{0}(\Gamma(Q, W)) = \C Q / \langle \partial_a W \rangle$.
\end{definition}

\begin{theorem}[CY-3 property lives on \texorpdfstring{$\Gamma$}{Gamma}, not on \texorpdfstring{$J$}{J}; Ginzburg 2006 \texttt{arXiv:math/0612139} Prop.~5.1.9 and Thm.~3.6.4, with completions form of Keller 2011 ``Deformed Calabi--Yau completions'' \texttt{arXiv:0908.3499} Thm.~6.3 (appendix by Van den Bergh); minimality by Bocklandt 2008 \texttt{arXiv:math/0603558} Thm.~3.1]
\label{thm:cy3-on-gamma-not-j}
For $(Q, W)$ a quiver-with-potential arising as the tilting presentation of a local CY$_3$ (Van den Bergh Theorem~\ref{thm:vdb-equivalence}):
\begin{enumerate}
 \item[(a)] The dg-algebra $\Gamma(Q, W)$ is \emph{exact $3$-Calabi--Yau}: its inverse dualising bimodule satisfies a quasi-isomorphism of $\Gamma$-bimodules
 \[
 \Gamma(Q, W)^{\vee} \;\coloneqq\; \RHom_{\Gamma^{e}}\bigl(\Gamma, \Gamma^{e}\bigr) \;\simeq\; \Gamma(Q, W)[-3],
 \qquad \Gamma^{e} \;=\; \Gamma \otimes_{\C} \Gamma^{\mathrm{op}},
 \]
 equivalently the shifted self-duality $\RHom_{\Gamma^{e}}(\Gamma, \Gamma^{e})[3] \simeq \Gamma$ of $\Gamma$-bimodules. The shift degree $+3$ matches the Pantev--To\"en--Vaqui\'e--Vezzosi symplectic shift on the derived moduli stack $\cM(X)$ for a CY$_3$ under the $(d, \mathrm{shift}, E_n^{\mathrm{cl}}) = (3, -1, E_1)$ stratification.
 \item[(b)] The Jacobi algebra $J = H^{0}(\Gamma(Q, W))$ is associative and \emph{not} by itself 3-Calabi--Yau as an associative algebra: $J$ carries no intrinsic shift, and the bimodule self-duality is a statement about $\Gamma$-bimodules, not $J$-bimodules. Calling $J$ ``CY$_3$'' collapses the dg-lift.
 \item[(c)] For $Q$ connected and $W$ non-degenerate in the sense of Derksen--Weyman--Zelevinsky 2008 \texttt{arXiv:0704.0649} Cor.~7.4, with explicit verification for the Klebanov--Witten quartic by Bocklandt \texttt{arXiv:math/0603558} Thm.~3.1, $H^{<0}(\Gamma(Q, W)) = 0$; the projection $\Gamma \to J$ is a quasi-isomorphism and only through this identification does the exact CY$_3$ structure on $\Gamma$ descend to $D^{b}(J\text{-}\mathrm{mod})$.
\end{enumerate}
\end{theorem}

\begin{proof}[Proof.]
(a) Ginzburg \texttt{arXiv:math/0612139} \S5.1 writes $\Gamma(Q, W)$ as the tensor algebra on the triple-graded generator bundle $(Q_1 \mid Q_1^{*} \mid \{t_i\}_{i \in I})$ and constructs the Koszul-type bimodule resolution
\[
 \Gamma \otimes \bigl(k_{Q_0} \to Q_1 \to Q_1^{*} \to \{t_i\}\bigr) \otimes \Gamma
 \;\to\; \Gamma,
\]
a four-term resolution reflecting the degrees $(0, 0, -1, -2)$ of the generators. Dualising into $\Gamma^{e}$ and tracking the cyclic trace $\tau_W \colon \HC_{3}(\Gamma) \to k$ (integration of the cyclic derivative of $W$) supplies the bimodule quasi-isomorphism $\Gamma^{\vee} \simeq \Gamma[-3]$ (Ginzburg Prop.~5.1.9; Thm.~3.6.4 identifies the dualising functor). The completions framework of Keller 2011 ``Deformed Calabi--Yau completions'' \texttt{arXiv:0908.3499} Thm.~6.3 (appendix by Van den Bergh) abstracts this to an arbitrary smooth dg-algebra with cyclic derivation and identifies the canonical bimodule isomorphism up to a contractible space of choices; on $\Gamma(Q, W)$ the two constructions agree.

(b) The shifted self-duality is a statement about $\Gamma$-bimodules; since $J = H^{0}(\Gamma)$ inherits the bimodule structure only through the quasi-isomorphism of (c), $\RHom_{J^{e}}(J, J^{e})$ does not by itself carry a degree-$3$ shift as $J$-bimodules: $J$ is finite-dimensional over $\C Q_0$ for each of the conifold, local~$\bP^2$, McKay $\C^3/\Z_n$ cases, and its bimodule dualising has no natural shift.

(c) Non-degeneracy of $W$ in the DWZ sense (every iterated premutation admits a reduction) implies $H^{<0}(\Gamma(Q, W)) = 0$ by Bocklandt Thm.~3.1: the DWZ reduction move kills exactly the $2$-cycle contributions to $H^{-1}$, and iterated non-degeneracy forces vanishing in all negative cohomological degrees. For the Klebanov--Witten quartic this is a direct computation of the Hochschild complex (Bocklandt, explicit at Prop.~4.5); the projection $\Gamma \to J$ is a quasi-isomorphism and the exact CY$_3$ structure of (a) transports along the derived equivalence $D(\Gamma\text{-}\mathrm{dg{-}mod}) \simeq D(J\text{-}\mathrm{mod})$.
\end{proof}

The passage from $Y$ to $A$ via derived Morita collapses two levels at once: the tilting bundle presents $D^{b}(\Coh Y)$ as $D^{b}(A\text{-}\mathrm{mod})$ (Theorem~\ref{thm:derived-morita-adjoint-pair}), and then the Ginzburg dg-lift $\Gamma(Q, W)$ presents $A$ as the degree-zero part of an exact CY$_3$ dg-algebra (Theorem~\ref{thm:cy3-on-gamma-not-j}). Both collapses are reversible; both are load-bearing. Writing ``$J$ is CY$_3$'' as shorthand is admissible only with the understanding that $J = H^{0}(\Gamma)$ and the bimodule self-duality is a property of $\Gamma$ not of $J$. Under the CY-to-chiral correspondence $\Phi_3$, it is $\Gamma(Q, W)$; not $J$; that feeds the stage-1 factorisation $\PhiFA_3$.

\subsection{Transport of the CoHA bialgebra along derived Morita}
\label{subsec:dhsm-hall-transport}

The critical CoHA on the geometric side $\cH(Y)$ and on the NCCR side $\cH(A)$ are built from the same grammar: equivariant Borel--Moore homology of the moduli stack of objects with coefficients in the vanishing-cycle sheaf of the canonical superpotential. The derived Morita equivalence of Theorem~\ref{thm:derived-morita-adjoint-pair} lifts to an equivalence of these moduli stacks, and Davison--Hennecart--Schlegel Mejia prove that the Hall product transports unconditionally, including its $\Z/2$ super-grading.

\begin{theorem}[Moduli-stack lift of derived Morita; DHSM 2023, arXiv:2303.12592 Thm.~A]
\label{thm:moduli-morita-dhsm}
Let $Y$ be a smooth local CY$_3$ admitting a tilting bundle $T$ satisfying hypotheses (H1)--(H2) of Theorem~\ref{thm:derived-morita-adjoint-pair}, and let $A = \End_{\cO_Y}(T)^{\mathrm{op}} = \Jac(Q, W)$ be the associated Jacobi algebra with Ginzburg dg-lift $\Gamma(Q, W)$ as in Theorem~\ref{thm:cy3-on-gamma-not-j}. The adjoint pair $L \dashv R$ descends to an equivalence of derived moduli stacks of semistable objects
\[
 \cM^{\mathrm{ss}}(Y) \;\simeq\; \cM^{\mathrm{ss}}(A)
\]
under the class-level identification
\[
 \mathrm{cl}_T \;\colon\; K_{0}(\Coh Y) \;\xrightarrow{\;\sim\;}\; \Z^{I},
 \qquad
 [\cF] \;\longmapsto\; \bigl(\dim \RHom(T_i, \cF)\bigr)_{i \in I},
\]
intertwining the universal coherent sheaf $\cE_Y$ on $\cM(Y)$ with the universal representation $V_A$ on $\cM(A)$: $\RHom(T, \cE_Y) = V_A$ as coherent sheaves on the common moduli stack.
\end{theorem}

On the derived moduli side, the universal deformation of $\cF \in \cM(Y)$ has tangent complex $\Ext^{1}(\cF, \cF)$ and obstruction space $\Ext^{2}(\cF, \cF)$; under derived Morita these match $\Ext^{1}_A(M, M)$ and $\Ext^{2}_A(M, M)$ for $M = \RHom(T, \cF)$, and Serre duality on the CY$_3$ identifies the obstruction with the dual of the tangent via the cyclic structure on $\Gamma$. The canonical superpotential $W_Y$ on the formal neighbourhood of $[\cF] \in \cM(Y)$, whose cyclic derivative generates the moduli defining equations, matches $\tr W_Q$ pulled back along the class-level isomorphism, because both record the degree-$3$ piece of the cyclic $A_\infty$-structure on $\Ext^{\bullet}(\cF, \cF)$ and derived Morita is exact on $A_\infty$-structures.

\begin{theorem}[Vanishing-cycle transport; DHSM 2023 \texttt{arXiv:2303.12592} Thm.~A, extending Davison \texttt{arXiv:1601.02479}]
\label{thm:vanishing-cycle-transport-dhsm}
Under the moduli equivalence of Theorem~\ref{thm:moduli-morita-dhsm}, the vanishing-cycle sheaf $\varphi_{\tr W_Y}$ on $\cM(Y)$ is canonically isomorphic to $\varphi_{\tr W_Q}$ on $\cM(A)$, compatibly with the $T_{\C^\times}$-equivariant structure. The asymmetry between the two sides is apparent, not structural:
\begin{itemize}
 \item On the geometric side, $Y$ is smooth, the moduli stack $\cM(Y)$ is globally realised as a critical locus of an analytic superpotential $\tr W_Y$ only on each chart of a cover (Joyce--Song \texttt{arXiv:0810.5645} Thm.~5.4: local analytic charts of derived $(-1)$-shifted symplectic moduli stacks on CY$_3$), and on each chart $\varphi_{\tr W_Y}$ is the vanishing cycle of the local potential; Brav--Bussi--Dupont--Joyce--Szendr\"oi \texttt{arXiv:1211.3259} Thm.~5.14 glue these into a global perverse sheaf $\mathrm{DT}_Y$ on $\cM(Y)$.
 \item On the NCCR side, $\cM(A)$ is the stack of $(Q, W)$-representations, and the superpotential is the global trace $\tr W_Q \colon \mathrm{Rep}(Q) \to \C$ of the cyclic $W_Q \in \C Q / [\C Q, \C Q]$; $\varphi_{\tr W_Q}$ is an ordinary vanishing-cycle sheaf of a global function.
\end{itemize}
The DHSM theorem asserts that under the moduli equivalence, the Brav--Bussi--Dupont--Joyce--Szendr\"oi glued sheaf on the left matches the global vanishing-cycle sheaf on the right:
\[
 \varphi_{\tr W_Y} \;=\; \mathrm{DT}_Y \;\simeq\; \varphi_{\tr W_Q} \quad \text{on } \cM(Y) \simeq \cM(A),
\]
compatibly with $T_{\C^\times}$-equivariance. Consequently the critical cohomologies
\[
 \cH(Y) \;\coloneqq\; H^{\bullet, \mathrm{BM}}_{T_{\C^\times}}\bigl(\cM(Y),\, \varphi_{\tr W_Y}\bigr),
 \qquad
 \cH(A) \;\coloneqq\; H^{\bullet, \mathrm{BM}}_{T_{\C^\times}}\bigl(\cM(A),\, \varphi_{\tr W_Q}\bigr),
\]
are canonically isomorphic as equivariant Borel--Moore cohomology modules on the common moduli stack.
\end{theorem}

The Hall product on either side is the push-pull convolution along the extension stack $\mathrm{Ext}_{\gamma_1, \gamma_2}$ parametrising short exact sequences $0 \to \cF_1 \to \cF \to \cF_2 \to 0$ with charge vectors $\gamma_1, \gamma_2$, indexed as $K_{0}$-classes on the geometric side and as dimension vectors $\underline d \in \N^{I}$ on the NCCR side. Under Theorem~\ref{thm:moduli-morita-dhsm} the extension stacks intertwine, $\mathrm{Ext}_{\gamma_1, \gamma_2}^{Y} \simeq \mathrm{Ext}_{\mathrm{cl}_T(\gamma_1), \mathrm{cl}_T(\gamma_2)}^{A}$, with both the ``ends'' projection to $\cM_{\gamma_1} \times \cM_{\gamma_2}$ and the ``middle'' projection to $\cM_{\gamma_1 + \gamma_2}$ intertwined.

\begin{theorem}[Hall-product transport along stable derived Morita; Davison--Hennecart--Schlegel Mejia 2023 arXiv:2303.12592 Thm.~A + Thm.~B]
\label{thm:hall-product-transport-dhsm}
Under hypotheses (H1)--(H2) on the tilting bundle $T$, the class-level isomorphism $\mathrm{cl}_T$ of Theorem~\ref{thm:moduli-morita-dhsm} extends $\Z$-linearly to a canonical isomorphism of bialgebras
\[
 \star_{T} \;\colon\; \cH(Y) \;\xrightarrow{\;\sim\;}\; \cH(A)
\]
intertwining the Hall product, the natural coproduct (where defined; the Davison--Meinhardt localisation of the integration map), and the $T_{\C^\times}$-equivariant structures. The $\Z/2$ \emph{super-grading} carried by $\varphi_{\tr W}$; cohomological parity inherited from the vanishing-cycle nearby-fibre; transports along the same equivalence by DHSM Thm.~B, upgrading the bialgebra isomorphism to one of super-bialgebras.
\end{theorem}

\begin{proof}[Proof.]
Write the extension stack and its two projections:
\[
 \mathrm{Ext}_{\gamma_1, \gamma_2}^{Y} \;\xrightarrow{\;(\mathrm{ends})\;}\;
 \cM_{\gamma_1}(Y) \times \cM_{\gamma_2}(Y),
 \qquad
 \mathrm{Ext}_{\gamma_1, \gamma_2}^{Y} \;\xrightarrow{\;(\mathrm{middle})\;}\;
 \cM_{\gamma_1 + \gamma_2}(Y).
\]
The Hall product is
\[
 \star_{\mathrm{Hall}} \;=\; (\mathrm{middle})_{*}
 \circ \bigl( \varphi_{\tr W_Y}^{\mathrm{ext}} \cap (-)\bigr)
 \circ (\mathrm{ends})^{*},
\]
where $\varphi_{\tr W_Y}^{\mathrm{ext}}$ is the pullback of the DT perverse sheaf along the extension stack projection. We verify that each of the three factors commutes with the derived Morita.

\emph{(i) Pullback along ends commutes with derived Morita.} By Theorem~\ref{thm:moduli-morita-dhsm}, $\cM_{\gamma_i}(Y) \simeq \cM_{\mathrm{cl}_T(\gamma_i)}(A)$. The extension stack on each side parametrises short exact sequences
\[
 0 \to \cF_1 \to \cF \to \cF_2 \to 0 \quad \text{in } \Coh(Y),
 \qquad
 0 \to M_1 \to M \to M_2 \to 0 \quad \text{in } A\text{-}\mathrm{mod}.
\]
The derived Morita $R = \RHom(T, -)$ is exact (preserves triangles), hence sends extensions to extensions, inducing $\mathrm{Ext}^{Y} \simeq \mathrm{Ext}^{A}$; both projections $(\mathrm{ends})$ and $(\mathrm{middle})$ intertwine on the nose. This matches Costello--Francis--Gwilliam's ``moduli of factorisation-algebra sections is functorial in the target'' principle (Costello--Gwilliam \emph{Factorization Algebras} I Lemma 3.1.5).

\emph{(ii) Intersection with $\varphi_{\tr W}$ commutes.} This is Theorem~\ref{thm:vanishing-cycle-transport-dhsm}: the pullback of $\varphi_{\tr W_Y}$ (the Brav--Bussi--Dupont--Joyce--Szendr\"oi glued DT sheaf) to $\mathrm{Ext}^{Y}$ matches the pullback of $\varphi_{\tr W_Q}$ to $\mathrm{Ext}^{A}$ under the moduli equivalence. DHSM 2023 Thm.~A supplies this identification; it is a statement about perverse sheaves on a common moduli stack.

\emph{(iii) Pushforward along middle commutes.} The projection formula and proper base change for the equivariant Borel--Moore cohomology $H^{\bullet, \mathrm{BM}}_{T_{\C^\times}}$ are functorial in the derived Morita, because the underlying moduli stacks are identified (Theorem~\ref{thm:moduli-morita-dhsm}) and the $T_{\C^\times}$-action on $\cM(Y)$ matches that on $\cM(A)$ in the toric (respectively the equivariantly-localisable non-toric) setting. Equivariant pushforward is the derived functor of taking coinvariants of the $T_{\C^\times}$-action, and the identification preserves this coinvariant structure.

Combining (i), (ii), (iii) yields an isomorphism of associative algebras $\star_T \colon \cH(Y) \to \cH(A)$. The coproduct side follows from the Davison--Meinhardt critical-CoHA integration map (\texttt{arXiv:1601.02479} Thm.~A), which commutes with derived Morita by the same chain of functorialities.

\emph{(iv) Super-grading transport (DHSM Thm.~B).} The $\Z/2$-grading on $\varphi_{\tr W}$ is the cohomological parity of the nearby-fibre $\psi_{\tr W}$ sheaf, organised by the Davison--Meinhardt super-PBW decomposition
\[
 \cH(Y) \;\simeq\; \mathrm{Sym}^{\boxtimes_{+}}_{\mathrm{super}}\!\bigl(H(B\C^{\times})_{\mathrm{vir}} \otimes \mathfrak{g}_{\mathrm{BPS}}(Y)\bigr),
\]
where $\mathfrak{g}_{\mathrm{BPS}}$ is the super BPS Lie algebra. DHSM Thm.~B shows that the identification of (i)--(iii) sends the even and odd parts on $Y$ to the even and odd parts on $A$; the super-bialgebra structure upgrades the bialgebra isomorphism of (i)--(iii).
\end{proof}

Theorem~\ref{thm:hall-product-transport-dhsm} establishes the universal positive-geometry grammar $Y^{+}(X) = H^{\bullet}_{\mathrm{eq}}(\cM^{+}_{\mathrm{eff}}(X), \varphi_W)$ as a derived-Morita invariant: the bialgebra $Y^{+}$ depends only on the dg-category of compactly-supported coherent sheaves and its CY$_3$ Ginzburg dg-lift, not on any particular presentation. For $Y = \tilde Y$ the resolved conifold, Theorem~\ref{thm:hall-product-transport-dhsm} yields the identification $\cH(\tilde Y) \simeq \cH(\Jac(Q_{\mathrm{con}}, W_{\mathrm{con}}))$ with the Szendr\H{o}i CoHA of the Klebanov--Witten quiver, and identifies both with the positive half $Y^{+}(\widehat{\mathfrak{gl}}(1|1))^{\mathrm{con}}$ of the conifold affine super Yangian of Example~\ref{ex:conifold-derived}. For $Y = K_{\bP^2}$ and tilting $T = \cO \oplus \cO(1) \oplus \cO(2)$, Theorem~\ref{thm:hall-product-transport-dhsm} identifies $\cH(K_{\bP^2})$ with the CoHA of the nine-arrow Beilinson quiver with cubic antisymmetric potential, i.e.\ the positive-half Drinfeld double of the Gaiotto--Rap\v{c}\'ak corner VOA $Y_{1,0,0}$. For $Y = \widetilde{[\C^3/\Z_n]}$ and $T = \bigoplus_{k = 0}^{n-1} \cO \otimes \chi_k$ the McKay tilting, Theorem~\ref{thm:hall-product-transport-dhsm} identifies $\cH(Y)$ with the CoHA of the type-$A_{n-1}$ cyclic McKay quiver with cubic potential.

\subsection{Keller--Yang mutation and the CoHA chamber cocycle}
\label{subsec:keller-yang-mutation-coha}

Two tilting bundles $T, T'$ on the same local CY$_3$ $Y$ produce two derived Morita equivalences $D^{b}(\Coh Y) \simeq D^{b}(A\text{-}\mathrm{mod}) \simeq D^{b}(A'\text{-}\mathrm{mod})$; by Theorem~\ref{thm:hall-product-transport-dhsm} both land on the same $\cH(Y)$. The induced isomorphism $\cH(A) \simeq \cH(A')$ admits an explicit combinatorial description when $T, T'$ are related by a \emph{Derksen--Weyman--Zelevinsky mutation} of quivers with potential (DWZ 2008 \texttt{arXiv:0704.0649} Defs.~5.1, 5.5, 7.1). The lift of mutation to a derived equivalence is Keller--Yang.

\begin{theorem}[Mutation induces a derived equivalence of Ginzburg dg-algebras; Keller--Yang 2011, \texttt{arXiv:0906.0761} Thms.~3.2 and 6.3]
\label{thm:keller-yang-mutation-derived}
Let $(Q, W)$ be a non-degenerate quiver-with-potential (in the DWZ Cor.~7.4 sense), and let $\mu_k (Q, W) = (Q', W')$ be the DWZ mutation at a vertex $k$ supporting no loops. There exist two triangulated equivalences
\[
 \Mu_k^{+}, \Mu_k^{-} \;\colon\; \cD\bigl(\Gamma(Q, W)\bigr)
 \;\xrightarrow{\;\sim\;}\;
 \cD\bigl(\Gamma(Q', W')\bigr)
\]
constructed from two distinct tilting objects in $\cD(\Gamma(Q, W))$,
\[
 T_k^{+} \;=\; \mathrm{cone}\Bigl(P_k \;\xrightarrow{\;(b)_{b\colon k \to j}\;}\; \bigoplus_{b \colon k \to j} P_{t(b)}\Bigr) \;\oplus\; \bigoplus_{i \neq k} P_i,
\]
with differential given by the arrows $b$ \emph{out of} $k$ in $Q$, and the dual $T_k^{-}$ built from arrows \emph{into} $k$. Keller--Yang Thm.~3.2 establishes that $T_k^{\pm}$ is a compact generator of $\cD(\Gamma(Q, W))$ with $\Ext^{>0}_{\cD(\Gamma)}(T_k^{\pm}, T_k^{\pm}) = 0$; Keller--Yang Thm.~6.3 identifies $\End_{\cD(\Gamma(Q, W))}(T_k^{\pm}) \simeq \Gamma(Q', W')$ as dg-algebras; Theorem~\ref{thm:derived-morita-adjoint-pair} then produces the two equivalences $\Mu_k^{\pm}$. The simple $S_k$ at vertex $k$ maps to $S_k^{\pm}[\mp 1]$: the shift direction is opposite for $\Mu_k^{+}$ and $\Mu_k^{-}$, and the two differ by the spherical twist at $S_k$ (Keller--Yang Prop.~6.4).
\end{theorem}

\begin{theorem}[Mutation-induced CoHA automorphism]
\label{thm:mutation-coha-automorphism-dhsm}
Let $(Q, W)$ be a non-degenerate QP with Ginzburg dg-lift $\Gamma(Q, W)$ exact $3$-Calabi--Yau (Theorem~\ref{thm:cy3-on-gamma-not-j}(a)), and write $A = \Jac(Q, W)$, $A' = \Jac(\mu_k(Q, W))$. The Keller--Yang equivalence $\Mu_k^{+}$ of Theorem~\ref{thm:keller-yang-mutation-derived} induces, via Hall-product transport of Theorem~\ref{thm:hall-product-transport-dhsm},
\[
 \Mu_{k, \star}^{+} \;\colon\; \cH(A) \;\xrightarrow{\;\sim\;}\; \cH(A'),
\]
an isomorphism of super-bialgebras. If $(Q, W)$ and $\mu_k(Q, W)$ arise as two tilting presentations of the same local CY$_3$ $Y$; equivalently, $\mu_k$ relates two tilting bundles $T, T'$ on $Y$; then $\Mu_{k, \star}^{+}$ is a super-bialgebra automorphism of the single CoHA $\cH(Y) \simeq \cH(A) \simeq \cH(A')$, with $(\Mu_{k, \star}^{+})^{-1} = \Mu_{k, \star}^{-}$.
\end{theorem}

\begin{proof}[Proof.]
By Theorem~\ref{thm:keller-yang-mutation-derived}, $\Mu_k^{+}$ is a triangulated equivalence between the derived categories of the two CY$_3$ Ginzburg algebras, constructed via derived Morita along the tilting object $T_k^{+}$. By Theorem~\ref{thm:hall-product-transport-dhsm}, every derived Morita equivalence between CY$_3$ Ginzburg algebras induces a super-bialgebra isomorphism on the critical CoHAs. When $(Q, W)$ and $\mu_k(Q, W)$ arise as two tilting presentations of the same $Y$, the corresponding tilting objects $T, T'$ give two factorisations of the identity on $\cH(Y)$, so $\Mu_{k, \star}^{+}$ is an automorphism. DWZ non-degeneracy (arXiv:0704.0649 Cor.~7.4) ensures $\Mu_k^{-}$ exists and gives the inverse at the derived level, hence $\Mu_{k, \star}^{+}$ has inverse $\Mu_{k, \star}^{-}$ in $\mathrm{Aut}_{\mathrm{super{-}bialg}}(\cH(Y))$.
\end{proof}

\paragraph{Conifold mutation as flop.} On the Klebanov--Witten QP, mutation $\mu_0$ at vertex $0$ is the three-step DWZ surgery (compose through $0$, reverse incident arrows, update the potential), followed by reduction to remove the spurious $2$-cycles. The reduced outcome is the Klebanov--Witten QP with vertex labels $\{0, 1\}$ swapped, and the derived equivalence $\Mu_0^{+}$ realises the Bridgeland--Chen small-resolution flop $D^{b}(\tilde Y) \simeq D^{b}(\tilde Y^{+})$ (Bridgeland 2002 arXiv:math/0009053). Transported by Theorem~\ref{thm:hall-product-transport-dhsm}, $\Mu_{0, \star}^{+}$ becomes the Kontsevich--Soibelman wall-crossing cocycle at the conifold wall (Nagao 2010 arXiv:1002.4884). This verifies the Hall-side mutation cocycle for the resolved conifold. It becomes an image under $\Phi_3$ only after the flop kernel is equipped with the witnessed orientation, framing, $\SpCh$, completion, and convolution data of Definition~\ref{def:phi3-admissible-kernel-datum}.

\paragraph{Local $\bP^{2}$ braid.} On the nine-arrow Beilinson QP, the three mutations $\mu_0, \mu_1, \mu_2$ are pairwise related by the cyclic $\Z/3$ symmetry of the antisymmetric cubic $W_{\bP^2}$. By Keller--Yang Prop.~6.4, each $\mu_k$ corresponds on the derived side to the inverse spherical twist at the simple $S_k$; the Seidel--Thomas braid relation on spherical twists (Seidel--Thomas 2001 \texttt{arXiv:math/0001043} Thm.~1.2) lifts via Theorem~\ref{thm:mutation-coha-automorphism-dhsm} to the $B_3$-braid relation
\[
 \mu_{0, \star}^{+} \mu_{1, \star}^{+} \mu_{0, \star}^{+}
 \;=\;
 \mu_{1, \star}^{+} \mu_{0, \star}^{+} \mu_{1, \star}^{+}
\]
in $\mathrm{Aut}_{\mathrm{super{-}bialg}}(\cH(K_{\bP^2}))$. Under the identification $\cH(K_{\bP^2}) \simeq Y_{1, 0, 0}[\psi]$ with the Gaiotto--Rap\v{c}\'ak corner VOA at the self-dual locus, the $B_3$-braid realises the triality $\epsilon_1 \leftrightarrow \epsilon_2 \leftrightarrow \epsilon_3$ of the three toric $\C^{\times}$-weights at the McKay vertex of $\C^3 / \Z_3$.

\paragraph{Descent to $K_{0}$ via Fomin--Zelevinsky.} The Euler form on $D^{b}(A\text{-}\mathrm{mod})$ for $A$ the Jacobi algebra of a CY$_3$ QP is \emph{antisymmetric}: Serre duality on CY$_d$ gives $\chi(\gamma, \gamma') = (-1)^d \chi(\gamma', \gamma)$, so on CY$_3$, $\chi$ is skew and $(\alpha_k, \alpha_k) = 0$ for every simple. The Weyl reflection $s_{\alpha_k}$ is undefined and mutation does not descend to a Weyl action on $K_{0}$. The correct $K_{0}$-descent is the Fomin--Zelevinsky cluster mutation
\[
 \mu_k^{\mathrm{FZ}}(\gamma_j) \;=\; \gamma_j \;+\; [b_{kj}]_{+} \gamma_k,
 \qquad
 b_{kj} \;=\; \#\{\text{arrows } k \to j\} - \#\{\text{arrows } j \to k\},
\]
(Fomin--Zelevinsky arXiv:math/0104151). On the Klebanov--Witten conifold, $b_{01} = b_{10} = 0$ because the arrow counts are symmetric (two arrows each way), so $\mu^{\mathrm{FZ}}$ is trivial at $K_{0}$; all of the mutation content lives at the critical-cohomology level, in $\cH(Y)$, not in $K_{0}(\Coh Y)$. CY$_3$ mutation is a cohomological phenomenon, not a lattice phenomenon.

\paragraph{The three-step construction.} Theorems~\ref{thm:vdb-equivalence}--\ref{thm:mutation-coha-automorphism-dhsm} assemble a three-step construction:
\[
 Y \;\underset{\text{Theorem~\ref{thm:derived-morita-adjoint-pair}}}{\xrightarrow{\;T = \bigoplus T_i\;}}\;
 A = \Jac(Q, W)
 \;\underset{\text{Theorem~\ref{thm:cy3-on-gamma-not-j}}}{\xrightarrow{\;\Gamma(Q, W)\;}}\;
 \text{exact CY}_3 \text{ dg-algebra}
 \;\underset{\text{Theorem~\ref{thm:hall-product-transport-dhsm}}}{\xrightarrow{\;\varphi_{\tr W}\;}}\;
 \cH(Y) \simeq \cH(A).
\]
Step 1 (Bondal--Van den Bergh derived Morita) collapses $D^{b}(\Coh Y)$ to $D^{b}(A\text{-}\mathrm{mod})$ via the adjoint pair $L \dashv R$ with load-bearing hypotheses (H1)--(H2). Step 2 (Ginzburg dg-lift) lifts the associative $A$ to the exact CY$_3$ dg-algebra $\Gamma(Q, W)$ where the shifted bimodule self-duality lives. Step 3 (DHSM stable derived Morita) transports the Hall-product bialgebra structure, including its $\Z/2$-super grading, unconditionally along the derived equivalence. The Keller--Yang mutation (Theorem~\ref{thm:keller-yang-mutation-derived}) acts on every step of this construction: on Step 1 by replacing one tilting bundle by another, on Step 2 by the quasi-isomorphism $\Gamma(Q, W) \simeq \Gamma(\mu_k(Q, W))$, and on Step 3 by the super-bialgebra automorphism $\Mu_{k, \star}^{+}$ of $\cH(Y)$. This construction is the derived-Morita face adjacent to the CY-to-chiral correspondence $\Phi_3$: every witnessed tilting presentation of a local CY$_3$ gives an $\Eone$-chart of the chiral algebra, and the mutation cocycle is the candidate gauge-transformation transition between charts once the obstruction criterion of Proposition~\ref{prop:phi3-arbitrary-morphism-obstruction-criterion} is satisfied.

\section{Bridgeland stability conditions and wall-crossing}
\label{sec:bridgeland-stab}

The space of Bridgeland stability conditions $\Stab(\cC)$ on a CY$_d$ category $\cC$ is a complex manifold (Bridgeland 2007). Its chambers correspond to $t$-structures on $\cC$, and wall-crossing between chambers governs mutations of exceptional collections, DT invariant jumps, and (in the Vol~III reading) transitions between different $\Eone$-chart presentations of the chiral algebra.

\begin{definition}[Bridgeland stability condition]
\label{def:bridgeland-stability}
A \emph{Bridgeland stability condition} on a triangulated category $\cC$ is a pair $\sigma = (Z, \cP)$, where $Z \colon K_0(\cC) \to \C$ is a group homomorphism (the \emph{central charge}) and $\cP = \{\cP(\phi)\}_{\phi \in \bR}$ is a slicing of $\cC$ into full additive subcategories (the \emph{semistable objects of phase $\phi$}), subject to: (i) for $0 \neq E \in \cP(\phi)$, $Z(E) = m(E) \exp(i \pi \phi)$ with $m(E) > 0$; (ii) $\cP(\phi + 1) = \cP(\phi)[1]$; (iii) Harder--Narasimhan filtrations exist (Bridgeland 2007).
\end{definition}

For CY$_3$ categories the stability manifold can have many wall-crossing chambers. On K3 surfaces, Bridgeland (2008) proved that $\Stab(D^b(\Coh(X)))$ is a connected, simply connected complex manifold containing the complexified K\"ahler cone as an open subset. The autoequivalence group $\mathrm{Aut}(D^b(\Coh(X)))$ acts on $\Stab(X)$ by deck transformations, and the quotient is a period domain.

\begin{example}[Stability conditions on K3]
\label{ex:stab-k3}
For a K3 surface $X$ with Picard rank $\rho$, the distinguished connected component $\Stab^\dagger(X) \subset \Stab(X)$ is a $(\rho + 2)$-dimensional complex manifold. The central charge takes the form $Z_{\omega, B}(E) = -\int_X e^{-(B + i\omega)} \cdot \mathrm{ch}(E)$ for $\omega$ an ample class and $B \in H^{1,1}(X, \bR)$. Walls are real-codimension-one loci where a semistable object acquires a destabilizing subobject; crossing a wall exchanges the two hearts related by a tilt. Under the correspondence $\Phi_2$, different stability chambers yield different $\Eone$-chart presentations of $\cA_{K3}$, and the wall-crossing transformations correspond to $R$-matrix gauge equivalences between charts.
\end{example}

\begin{example}[Stability conditions on the resolved conifold]
\label{ex:stab-conifold}
For the resolved conifold $X = \cO_{\P^1}(-1)^{\oplus 2}$, the stability manifold $\Stab(D^b_{cs}(\Coh(X)))$ has a single wall at $\mathrm{Im}(Z(\cO_{\P^1}(-1))) = 0$, separating two chambers: the large-volume chamber (where $\cO_{\P^1}(-1)$ is stable) and the orbifold chamber (where the fractional branes are stable). The wall-crossing formula is the Kontsevich--Soibelman identity for the conifold DT invariants (Kontsevich--Soibelman 2008; Nagao--Nakajima 2011). On the chiral side, this single wall-crossing corresponds to the mutation relating the two Klebanov--Witten quiver presentations of Example~\ref{ex:conifold-derived}.
\end{example}

The general principle is that the stability manifold $\Stab(\cC)$ provides an atlas of $\Eone$-chart descriptions of the chiral algebra $\Phi(\cC)$; the transition functions are $R$-matrix gauge equivalences encoded by wall-crossing automorphisms. This perspective is developed in Chapter~\ref{ch:toric-coha} for toric CY$_3$ categories and is conjectural in general (conditional on CY-A$_3$).

\subsection{Wall-crossing as \texorpdfstring{$R$}{R}-matrix gauge transformation}
\label{subsec:wallcrossing-rmatrix}

The dictionary between Bridgeland wall-crossing and $R$-matrix gauge transformations is the second functoriality clause U2 of the CY-to-chiral correspondence $\{\Phi_d\}$ (the first, U1, being the action on Morita equivalences; see Remark~\ref{rem:phi-functoriality-summary} below for the U1--U4 summary). In the proved $d \leq 2$ scope, and at $d = 3$ on the witnessed-kernel subcategory of Theorem~\ref{thm:phi3-witnessed-kernel-functoriality}, a Bridgeland slicing yields a $t$-structure on $\cC$, a $t$-structure yields a heart $\cC^\heartsuit$, the heart yields a tilting object (when one exists), the tilting object yields a quiver $Q_\sigma$, and the quiver gives an $\Eone$-chart of the chiral algebra with a specific generator basis. Two charts $\sigma_1, \sigma_2$ in adjacent chambers of $\Stab(\cC)$ produce two presentations $(Q_{\sigma_1}, R_{\sigma_1})$ and $(Q_{\sigma_2}, R_{\sigma_2})$ of the same chiral algebra; the wall-crossing automorphism is the $R$-matrix gauge transformation relating them.

\begin{conjecture}[Wall-crossing dictionary]
\label{conj:wallcrossing-dictionary}
Let $\cC$ be a smooth proper CY$_d$ category for which $\Phi_d$ is defined ($d \leq 2$ by the proved functorial construction, and $d = 3$ only after the relevant wall-crossing kernel is lifted to the witnessed class of Definition~\ref{def:phi3-admissible-kernel-datum}). Let $\sigma_1, \sigma_2 \in \Stab(\cC)$ lie in adjacent chambers separated by a single wall $W$, and let $\mathrm{KS}_W \in \mathrm{Aut}(\cC)$ be the Kontsevich--Soibelman wall-crossing automorphism. Then
\[
 \Phi(\mathrm{KS}_W) \;=\; g_W \in \mathrm{GL}_n\bigl(\Phi(\cC)\bigr),
\]
a gauge transformation of the chiral algebra realising the change of basis between the two $\Eone$-presentations. The classical $r$-matrix transforms by
\[
 r_{\sigma_2}(z) \;=\; g_W(z) \cdot r_{\sigma_1}(z) \cdot g_W(z)^{-1} \;-\; (\partial_z g_W)(z) \cdot g_W(z)^{-1},
\]
the standard gauge action on a meromorphic connection on the configuration space.
\end{conjecture}

\begin{remark}[Categorical content of the gauge action]
\label{rem:categorical-gauge}
The transformation rule in Conjecture~\ref{conj:wallcrossing-dictionary} is the classical gauge action of the loop group $G(\!(z)\!)$ on the space of $\mathfrak{g}$-valued meromorphic 1-forms on the punctured disk. The categorical lift comes from the fact that $\mathrm{KS}_W$ is a Fourier--Mukai-type autoequivalence of $\cC$ whose kernel changes the basis of $\HH^\bullet(\cC)$ by the wall-crossing matrix; transporting this through Theorem~\ref{thm:hkr-cy} gives a change of generator basis on the Hochschild side, and thence a change of $r$-matrix on the chiral side. The gauge transformation distinguishes the algebraisation invariant $\kappa_{\mathrm{ch}}$ (preserved under $\Phi(\mathrm{KS}_W)$, since gauge transformation acts trivially on the trace) from the manifold invariant $\kappa_{\mathrm{cat}}$ (preserved a fortiori).
\end{remark}

\begin{example}[Single-wall conifold wall-crossing as R-matrix gauge]
\label{ex:conifold-rmatrix-gauge}
For the resolved conifold of Example~\ref{ex:stab-conifold}, the single wall separates the large-volume chamber (geometric DT invariants) from the orbifold chamber (noncommutative DT invariants). The Kontsevich--Soibelman wall-crossing automorphism is
\[
 \mathrm{KS}_W \;=\; \mathrm{Ad}_{T_E}^{n_E} \quad\text{where } T_E = \prod_{m \in \Z} (1 - q^m e_E)^{(-1)^m},
\]
with $n_E$ the DT count of the wall-stable object $E = \cO_{\P^1}(-1)$ and $e_E$ the corresponding root vector in the affine super Yangian $Y(\widehat{\mathfrak{gl}}(1|1))$ (Example~\ref{ex:conifold-derived}). On the witnessed toric $\Phi_3$ locus, this lifts to the gauge transformation $g_W(z) = T_E(z)$ of the chiral $r$-matrix; the two presentations of $\Phi_3(D^b_{cs}(\Coh(X)))$ are related by conjugation. The preserved chiral invariant is the direct McKay value $\kappa_{\mathrm{ch}}(\mathrm{conifold}) = 1$; it is not computed by the compact odd-dimensional formula $\chi(\cO_X)=0$.
\end{example}

The categorical content of wall-crossing is that the $R$-matrix is a connection on a flat bundle over $\Stab(\cC)$, and the holonomy of this connection around walls reproduces the Kontsevich--Soibelman wall-crossing formulae. This is the geometric counterpart of the fact that on the algebraic side, the $\Eone$-chiral algebra has a single isomorphism class but many quasi-equivalent presentations indexed by stability conditions.

\section{Birational geometry: flops and derived equivalences}
\label{sec:flops-derived}

A second source of derived autoequivalences is birational geometry. The Bondal--Orlov reconstruction theorem identifies smooth projective varieties with ample canonical or anti-canonical bundle from their derived categories, but for Calabi--Yau varieties the canonical bundle is trivial and birational CY varieties typically have equivalent derived categories. Flops are the prototypical example.

\begin{theorem}[Bridgeland's flop equivalence]
\label{thm:bridgeland-flop}
Let $X \to Y \leftarrow X^+$ be a standard threefold flop between smooth projective Calabi--Yau threefolds (i.e.\ $X$ and $X^+$ are connected by contracting a $\P^1$ and re-resolving). Then there is a derived equivalence
\[
 D^b(\Coh(X)) \;\simeq\; D^b(\Coh(X^+))
\]
realised by a Fourier--Mukai functor with kernel the structure sheaf of the fibre product $X \times_Y X^+$ (Bridgeland 2002).
\end{theorem}

The Bridgeland flop kernel is a raw geometric candidate for
Theorem~\ref{thm:phi3-witnessed-kernel-functoriality}. It produces a
chiral equivalence
$\Phi_3(D^b(\Coh(X))) \simeq \Phi_3(D^b(\Coh(X^+)))$
only after the kernel is lifted to a $\Phi_3$-admissible datum
\textup{(}orientation line, cyclic transfer, $\bS^3$-framing,
Costello--Li naturality, $\SpCh$ compatibility, completion, and
convolution coherences\textup{)}. On that witnessed locus the two
presentations differ by the expected $R$-matrix gauge transformation
analogous to wall-crossing.

\begin{example}[Atiyah flop]
\label{ex:atiyah-flop}
The Atiyah flop is the simplest non-trivial threefold flop: $X = \cO_{\P^1}(-1) \oplus \cO_{\P^1}(-1)$ (the resolved conifold of Examples~\ref{ex:conifold-derived} and~\ref{ex:stab-conifold}) flops to $X^+$, an isomorphic CY threefold with the orientation of the $\P^1$ reversed. The two are related by the Bridgeland equivalence, and the toric Hall-side mutation gives the gauge transformation of Example~\ref{ex:conifold-rmatrix-gauge}. A $\Phi_3$-morphism statement requires the witnessed flop-kernel datum of Definition~\ref{def:phi3-admissible-kernel-datum}; on that witnessed locus the direct McKay value $\kappa_{\mathrm{ch}}(\mathrm{conifold}) = 1$ is preserved across the two toric charts.
\end{example}

\begin{remark}[McKay correspondence as derived equivalence]
\label{rem:mckay-derived}
The McKay correspondence supplies a third source of derived equivalences. Bridgeland--King--Reid \cite{BridgelandKingReid2001} prove that for $\Gamma \subset \mathrm{SL}(d, \C)$ a finite subgroup with $d \leq 3$, there is an equivalence
\[
 D^b(\Coh^\Gamma(\C^d)) \;\simeq\; D^b(\Coh(Y)),
\]
where $Y \to \C^d/\Gamma$ is the canonical (crepant) resolution. The left side is the equivariant derived category, equivalent to the derived category of the orbifold $[\C^d/\Gamma]$. For $d \leq 2$ the McKay equivalence is carried by the proved functorial construction. For $d = 3$ it is a raw McKay kernel; it becomes a chiral equivalence between the orbifold chiral algebra (constructed from the $\Gamma$-equivariant CoHA) and the chiral algebra of the resolution only after the witnessed $\Phi_3$-kernel data of Definition~\ref{def:phi3-admissible-kernel-datum} are supplied.
\end{remark}

\section{Functoriality and Vol III}
\label{sec:vol3-functoriality}

The three sources of derived equivalences in this chapter -- HMS (mirror symmetry), Bridgeland wall-crossing, and birational geometry (flops, McKay) -- supply raw kernels for the CY-to-chiral correspondence. At $d = 3$ they produce chiral morphisms only on the witnessed-kernel subcategory of Theorem~\ref{thm:phi3-witnessed-kernel-functoriality}; outside that subcategory they determine the obstruction data for Conjecture~\ref{conj:phi-d-functoriality}.

\begin{remark}[\texorpdfstring{$\Phi$}{Phi} functoriality summary]
\label{rem:phi-functoriality-summary}
The four universal properties of $\Phi$ (Theorem~\ref{thm:phi-platonic}) read in the derived-category setting:
\begin{itemize}
 \item \textbf{U1} (Morita equivalences): $\Phi$ sends Morita-equivalent CY$_d$ categories to isomorphic chiral algebras on the proved $d\leq2$ scope and on the witnessed $d=3$ kernel scope. The HMS equivalences (Theorems~\ref{thm:hms-d1}, \ref{thm:hms-d2}, \ref{thm:hms-quintic}) supply the raw kernels.
 \item \textbf{U2} (gauge transformations): $\Phi$ sends Bridgeland wall-crossing to $R$-matrix gauge transformations on the witnessed $d=3$ kernel scope; the unrestricted statement is Conjecture~\ref{conj:wallcrossing-dictionary}. The Atiyah flop (Example~\ref{ex:atiyah-flop}) and McKay equivalence (Remark~\ref{rem:mckay-derived}) are raw derived kernels until their witnessed data are fixed.
 \item \textbf{U3} (deformations): $\Phi$ converts deformations of the CY category (variations of complex structure, K\"ahler moduli) into deformations of the chiral algebra, the latter parametrising a flat bundle on $\Stab(\cC)$.
 \item \textbf{U4} (degenerations): $\Phi$ sends degenerations of CY$_d$ to degenerations of chiral algebras, including the orbifold/large-volume limits and the LG/CY correspondence (Chapter~\ref{ch:matrix-factorizations}, Section~\ref{sec:lg-cy-correspondence}).
\end{itemize}
The four clauses are the operational content of the slogan ``$\Phi$ is functorial and continuous on the moduli stack of CY$_d$ categories''. At $d = 3$ the slogan means the witnessed-kernel theorem plus the remaining obstruction data of Proposition~\ref{prop:phi3-arbitrary-morphism-obstruction-criterion}. The Bridgeland stability manifold $\Stab(\cC)$ is the local model for U2 and U3; the Kontsevich--Soibelman wall-crossing formulae govern the expected holonomy along walls.
\end{remark}

\section{The Kontsevich--Soibelman--To\"en reading: \texorpdfstring{$\PhiFA_2$}{PhiFA-2} on K3 and its Nakajima specialisation}
\label{sec:dcy-kst-phi2-k3-coha}

\begin{remark}[\texorpdfstring{$\PhiFA_2$}{PhiFA-2} on \texorpdfstring{$D^b\mathrm{Coh}(K3)$}{D-bCoh(K3)} and the K3-CoHA specialisation]
\label{rem:dcy-kst-phi2-k3-coha}
The Kontsevich--Soibelman--To\"en reading factors through the two stages of Theorem~\ref{thm:phi-two-stage-factorisation}. Stage~$1$ produces the canonical $E_2$-holomorphic factorisation algebra $\PhiFA_2(D^b\mathrm{Coh}(K3)) \in \EdHolFA(K3)$ on the algebraic surface $K3$ (Francis--Gaitsgory locality, Costello--Gwilliam--Li). \[
 \SpCh_{\Sigma_1^{\mathrm{Nakajima}},\, C}\bigl(\PhiFA_2(D^b\mathrm{Coh}(K3))\bigr)
 \;=\; D\bigl(\mathcal{H}_{K3}\bigr)
 \;=\; H_{\mathrm{Muk}},
\]
where $\mathcal{H}_{K3}$ is the K3 cohomological Hall algebra
(Kapranov--Vasserot 2018 \textup{arXiv:1802.07988}, Theorem~1.1;
Negu\c{t}~2022 \textup{arXiv:2204.02497}, Theorem~1.1) and
$D(\mathcal{H}_{K3})$ is its Drinfeld double. The Mukai--Heisenberg vertex
algebra $H_{\mathrm{Muk}}$ equals the Drinfeld double by the
Grojnowski--Nakajima--Lehn Heisenberg action on
$\bigoplus_n H^\bullet(\mathrm{Hilb}^n(K3))$ (Nakajima 1997
\textup{Ann.~Math.}~145, Theorem~3.10; Grojnowski 1996; Lehn 1999
\textup{Inventiones}~136, Theorem~3.3). The positive-Borel embedding

\section{The CY-2 landscape dichotomy}
\label{sec:dcy-psi-landscape-routes}

The K3-specific fingerprint of $\mathbf{H}_{\Delta_5}$ is pinned
against three Witten-type hypotheses: topological Euler
characteristic $\chi_{\mathrm{top}}(K3) = 24$ (T1), vanishing
first Chern class $c_1(K3) = 0$ (T2), and Hodge number
$h^{2,0}(K3) = 1$ (T3). The remaining CY-$2$
landscape; Enriques surfaces, complex tori~$T^4$, Kummer K3s,
bielliptic surfaces, and the rational elliptic surface
$\mathrm{dP}_9$; each produces the precise analogue of
the fourth-taxon refinement~$\mathbf{H}_{\Delta_5}$ under the
CY-to-chiral functor~$\Phi_2$. The answers partition the CY-$2$
landscape into four strata, each image living in a distinct
Drinfeld-landscape taxon. The ordering of the strata below is by
decreasing~$\chi_{\mathrm{top}}$; the Borcherds lift,
Gritsenko--Nikulin additive lift, and Siegel-cusp-form weight
track~$\chi_{\mathrm{top}}$ linearly, which is the arithmetic
content of the dichotomy.

\begin{theorem}[Enriques analogue \texorpdfstring{$\mathbf{H}_{\Delta_5}^{\mathrm{Enr}}$}{H-Delta-5-Enr};
\texorpdfstring{$\mathbb{Z}/2$}{Z/2}-orbifold of the K3 refinement at Siegel weight \texorpdfstring{$5/2$}{5/2}]
\label{thm:dcy-enriques-analogue}
Let $\mathrm{En} = K3/\iota$ denote an Enriques surface, the free
quotient of a K3 surface $X$ by a fixed-point-free involution
$\iota$ acting as $-1$ on $H^0(X, \omega_X)$
(Barth--Hulek--Peters--Van~de~Ven 2004
\textup{Compact Complex Surfaces}, Chapter~VIII, Proposition~15.1).
The Enriques surface has $\chi_{\mathrm{top}}(\mathrm{En}) = 12$
(half of K3), $c_1 = 0$ as a $\mathbb{Q}$-class (non-trivial
$2$-torsion $c_1 \in H^2(\mathrm{En},\mathbb{Z})$ of order~$2$),
and $h^{2,0}(\mathrm{En}) = 0$. The integral cohomology lattice is
\[
 H^2(\mathrm{En},\mathbb{Z})/\mathrm{torsion} \;\simeq\; E_8(-1) \oplus \mathrm{II}_{1,1},
\]
of signature $(1,9)$, equal to the Enriques lattice
$\Lambda_{\mathrm{Enr}}$ of Nikulin 1979
\textup{Izv.~Akad.~Nauk}~43. Applying $\Phi_2$ to
$D^b(\mathrm{Coh}(\mathrm{En}))$ and composing with the
$\mathbb{Z}/2$-orbifold gauging (Kirillov--Ostrik 2002
\textup{Adv.~Math.}~171, Theorem~5.1, applied to the $\iota$-lift
on $\mathcal{H}_{K3}$) gives the Enriques analogue
\[
 \mathbf{H}_{\Delta_5}^{\mathrm{Enr}}
 \;=\;
 \Phi_2\bigl(D^b(\mathrm{Coh}(\mathrm{En}))\bigr)
 \;=\;
 \mathbf{H}_{\Delta_5}\,/\!\!/\,\langle\iota\rangle,
\]
a $\mathbb{Z}/2$-orbifold chiral algebra at level~$2$. The genus-2
Siegel character of $\mathbf{H}_{\Delta_5}^{\mathrm{Enr}}$ is
$1/\Delta_5^{\mathrm{Enr}}(Z)$, where
$\Delta_5^{\mathrm{Enr}}$ is the weight-$5/2$ Siegel paramodular
form of level~$2$ on the paramodular group $K(2)\subset\mathrm{Sp}_4(\mathbb{Q})$
of Gritsenko--Nikulin 1998 \textup{J.~reine angew.~Math.}~507,
Theorem~5.2 (Borcherds lift on $\mathrm{II}_{2,10}(2)$).
Half-integrality of the weight tracks
$h^{2,0}(\mathrm{En}) = 0$ and the $\mathbb{Z}/2$-gauging on the
Gritsenko--Nikulin additive-lift side; the weight-$5/2$ rather
than~$5$ is the chiral-side avatar of
$\chi_{\mathrm{top}}(\mathrm{En})/\chi_{\mathrm{top}}(K3) = 1/2$.
\end{theorem}

\begin{proof}
The $\iota$-fixed locus on~$\mathcal{H}_{K3}$ picks out the
$\iota$-invariant sublattice
$\Lambda_{\mathrm{Enr}}^+ = E_8(-1)\oplus\mathrm{II}_{1,1}$; the
Nakajima Heisenberg action (1997 \textup{Ann.~Math.}~145,
Theorem~3.10) restricts to the $\iota$-fixed Hilbert-scheme
component. The Kirillov--Ostrik gauging of the Mukai--Heisenberg
vertex algebra at level~$2$ produces a $\mathbb{Z}/2$-orbifold
VOA; the Borcherds multiplicative lift on the Enriques lattice
$\mathrm{II}_{2,10}(2)$ (Gritsenko--Nikulin 1998, Theorem~5.2)
produces precisely the weight-$5/2$ paramodular form
$\Delta_5^{\mathrm{Enr}}$ as denominator of the orbifold BKM
superalgebra $\mathfrak{g}_{\Delta_5}^{\mathrm{Enr}}$ (Hulek--Sankaran 2011
\textup{Algebra and Number Theory}~5, Theorem~3.4;
Dolgachev--Kond\={o} 2002
\textup{Asian~J.~Math.}~6, Theorem~4.1). The half-integral weight
is forced by the signature $(2,10) = (1,9) + (1,1)$ with a
$\mathbb{Z}/2$-metaplectic cover on the $(1,1)$-factor; this is
the Oguiso--Sakai 2001 \textup{Nagoya~Math.~J.}~163 fingerprint.
\end{proof}

\begin{remark}[\texorpdfstring{$T^4$}{T-4} abelian surface: \texorpdfstring{$\Phi_2$}{Phi-2}-image is the
abelian Heisenberg VOA, no Borcherds lift]
\label{rem:dcy-t4-abelian}
The complex torus $T^4 = \mathbb{C}^2/\Lambda$ is CY-$2$ with
$\chi_{\mathrm{top}}(T^4) = 0$ (Gauss--Bonnet on a flat manifold),
$c_1(T^4) = 0$, $h^{2,0}(T^4) = 1$. Under~$\Phi_2$, the K3-CoHA
argument of Remark~\ref{rem:dcy-kst-phi2-k3-coha} replaces
Nakajima's Hilbert-scheme Heisenberg action
on~$H^\bullet(\mathrm{Hilb}^n(K3))$ by the parallel construction on
$\mathrm{Hilb}^n(T^4)$ (Nakajima 1997 \textup{Ann.~Math.}~145,
Theorem~3.10 also for abelian surfaces; Nahm--Wendland 2001
\textup{Comm.~Math.~Phys.}~216, Theorem~4.1). Since
$\chi_{\mathrm{top}}(T^4) = 0$, the G\"ottsche generating function
becomes
$\sum_n \chi(\mathrm{Hilb}^n(T^4))\,q^n = \prod_m(1-q^m)^0 = 1$,
degenerate. The Borcherds multiplicative lift of the $T^4$
elliptic genus $\phi_{0,1}^{T^4} = 0$ is trivial, so there is no
weight-$5$ Siegel cusp form on the abelian-surface side. The
$\Phi_2$-image is
\[
 \Phi_2\bigl(D^b(\mathrm{Coh}(T^4))\bigr)
 \;=\;
 V_{\mathrm{II}_{4,4}}
 \;=\;
 \text{abelian Heisenberg VOA on }\mathrm{II}_{4,4},
\]
the even self-dual rank-$8$ Narain lattice VOA of signature
$(4,4)$ (Nahm--Wendland 2001, Theorem~4.1; Frenkel--Lepowsky--Meurman 1988
\textup{Vertex Operator Algebras and the Monster}, \S10.4 for the
Narain-lattice-VOA construction). This places
$\Phi_2(T^4)$ squarely in Drinfeld taxon~$G$ (abelian Heisenberg;
landscape census \cite{VolI}, family G),
\emph{not} in the fourth taxon of $\mathbf{H}_{\Delta_5}$. The
degeneration $\chi_{\mathrm{top}} = 0 \Rightarrow \Phi_2(\text{CY-}2)\in G$
is the $N = 1$ Conway-type BKM limit: the imaginary-root structure
collapses to rank~$0$, leaving the classical Narain lattice.
\end{remark}

\begin{remark}[Kummer K3 \texorpdfstring{$\mathrm{Kum}(T^4) \hookrightarrow K3$}{Kum(T-4) hookrightarrow K3}:
\texorpdfstring{$\mathbf{H}_{\Delta_5}$}{H-Delta-5} on the \texorpdfstring{$16$}{16}-node Picard sublattice]
\label{rem:dcy-kummer}
The Kummer K3 associated to an abelian surface $A = T^4$ is the
minimal resolution $\mathrm{Kum}(A) = \widetilde{A/\pm 1}$, with
$\chi_{\mathrm{top}}(\mathrm{Kum}(A)) = \chi_{\mathrm{top}}(A/\pm 1) + 16
 = (\chi(T^4) - 16)/2 + 16 + 16 = 0 + 16 + 8 = 24$ after resolving
the $16$ nodes via Euler-characteristic additivity; equivalently,
$\chi(\mathrm{Kum}) = 24$ via $\chi_{\mathrm{top}}(K3) = 24$
(Barth--Hulek--Peters--Van~de~Ven 2004, Chapter~VIII, \S14). The
$\mathrm{Kum}(A)$ is an algebraic K3 with transcendental
lattice $T_{\mathrm{Kum}} \subset T_A(2)$ and Picard sublattice
\[
 \Lambda_{\mathrm{Kum}}
 \;=\;
 E_8(-2)^{\oplus 2} \oplus \mathrm{II}_{1,1}(2) \oplus \langle -2\rangle^{16}
 \;\subset\;
 \mathrm{II}_{3,19},
\]
where the $16$ classes of $(-2)$-curves are the exceptional
divisors of the blow-up (Mukai 1984 \textup{Nagoya~Math.~J.}~81,
Theorem~1; Morrison 1984 \textup{Invent.~Math.}~75, Theorem~3.1).
For special $T^4$ (those with enough Hodge structure to be
algebraic), Mukai's 1984 theorem gives
$\mathrm{Kum}(T^4) \simeq K3^{\mathrm{alg}}$ as complex manifolds;
for generic $T^4$, the Kummer is transcendental K3. Under~$\Phi_2$,
the image is
\[
 \Phi_2\bigl(D^b(\mathrm{Coh}(\mathrm{Kum}(A)))\bigr)
 \;=\;
 \mathbf{H}_{\Delta_5}\,\big|_{\Lambda_{\mathrm{Kum}}}
 \;\subset\;
 \mathbf{H}_{\Delta_5},
\]
the Kummer-lattice restriction of the fourth-taxon refinement: the
same Mukai--Heisenberg VOA, but with currents indexed by
$\Lambda_{\mathrm{Kum}}$ rather than the full~$\mathrm{II}_{3,19}$.
The Gritsenko--Nikulin additive lift restricts to
$\Delta_5\big|_{\mathcal{A}_2^{\mathrm{Kum}}}$, a Siegel cusp form
on the Kummer-locus of the moduli of principally-polarised abelian
surfaces $\mathcal{A}_2$ (Gritsenko--Nikulin 1998, \S 4;
Morrison--Vafa 1996 \textup{Nucl.~Phys.~B}~476, II.\S3 for the
physical origin of this restriction on the string-theory moduli
space).
\end{remark}

\begin{remark}[Half-K3 \texorpdfstring{$= \mathrm{dP}_9$}{= dP-9} maps to~\texorpdfstring{$\widehat{E_8}_1$}{widehatE-8-1};
no \texorpdfstring{$\mathbf{H}_{\Delta_5}$}{H-Delta-5}-analogue, strikes taxon~L]
\label{rem:dcy-halfk3}
The half-K3 $\mathrm{dP}_9 = \mathrm{Bl}_9\mathbb{P}^2$ is the
$9$-point blow-up of~$\mathbb{P}^2$, equivalently the rational
elliptic surface with $12$~singular~$I_1$-fibres and anti-canonical
class $-K = f$ (the fibre class, Seiberg 1988
\textup{Phys.~Lett.~B}~206, \S2). It fails the three K3 hypotheses
(T1): $\chi_{\mathrm{top}}(\mathrm{dP}_9) = 12\ne 24$, and
(T2): $c_1(\mathrm{dP}_9) = -f \ne 0$; only~(T3)
$h^{2,0}(\mathrm{dP}_9) = 0$ is a \emph{non-issue} because
$\mathrm{dP}_9$ has no non-trivial holomorphic~$2$-forms anyway.
Two of the three K3 conditions fail, so there is no Borcherds
product of weight~$5$: the Borcherds singular theta lift (T16(i))
on the half-K3-relevant lattice $\mathrm{II}_{1,9}$ produces an
$E_8$-character generating function rather than a weight-$5$
Siegel form (Borcherds 1998 \textup{Invent.~Math.}~132,
Theorem~10.1, with the weight-$5$ output requiring
$\mathrm{II}_{2,10}$ signature, not $\mathrm{II}_{1,9}$). The
$\Phi_2$-image is
\[
 \Phi_2\bigl(D^b(\mathrm{Coh}(\mathrm{dP}_9))\bigr)
 \;=\;
 \widehat{E_8}_{\,k=1},
\]
the affine $E_8$ Kac--Moody algebra at level~$1$. The $E_8$
origin is the $8$d heterotic / F-theory compactification on
$\mathrm{dP}_9$ where the $12$ $I_1$-fibres collide into a single
$II^*$-fibre supporting the unbroken $E_8$ gauge symmetry
(Seiberg 1988, \S3; Morrison--Vafa 1996, II.\S4). The level is
fixed by the $I_1$-count $12 = \chi_{\mathrm{top}}(\mathrm{dP}_9)
/ 1$ via the gauge-anomaly-cancellation condition; the $E_8$
Dynkin data match the Seiberg--Witten curve monodromy around
the $II^*$-point. This places $\Phi_2(\mathrm{dP}_9)$ in Drinfeld
taxon~L (affine Kac--Moody; landscape census
\cite{VolI}, family~L).
\end{remark}

\begin{remark}[CY-\texorpdfstring{$2$}{2} landscape dichotomy: the \texorpdfstring{$\chi_{\mathrm{top}}/c_1/h^{2,0}$}{chi-top/c-1/h-2,0} partition]
\label{rem:dcy-landscape-partition}
The CY-to-chiral functor~$\Phi_2$ partitions the CY-$2$ landscape
into four strata, indexed by
$(\chi_{\mathrm{top}}, c_1, h^{2,0})$, with each stratum hitting a
distinct Drinfeld-landscape taxon:
\begin{center}
\begin{tabular}{lcccl}
\toprule
Surface & $\chi_{\mathrm{top}}$ & $c_1$ & $h^{2,0}$ & $\Phi_2$-taxon \\
\midrule
K3 & $24$ & $0$ & $1$ & 4th taxon: $\mathbf{H}_{\Delta_5}$ \\
Enriques & $12$ & $0$ & $0$ & 4th taxon / $\mathbb{Z}/2$: $\mathbf{H}_{\Delta_5}^{\mathrm{Enr}}$ \\
$T^4$ & $0$ & $0$ & $1$ & G: abelian Heisenberg $V_{\mathrm{II}_{4,4}}$ \\
bielliptic & $0$ & $0$ & $0$ & G: $V_{\mathrm{II}_{4,4}}/G$ \\
$\mathrm{dP}_9$ & $12$ & $-f$ & $0$ & L: $\widehat{E_8}_{\,k=1}$ \\
\bottomrule
\end{tabular}
\end{center}
The Kummer K3 sits inside the K3 row of this table as a
sublattice restriction
$\mathbf{H}_{\Delta_5}\big|_{\Lambda_{\mathrm{Kum}}}$
(Remark~\ref{rem:dcy-kummer}), not as a distinct
stratum. The dichotomy is controlled as follows:
\begin{itemize}
\item $\chi_{\mathrm{top}} = 24$ and $c_1 = 0$: K3 and Kummer K3,
 4th taxon. The Borcherds weight is~$5$ (integer); the Siegel
 cusp form is~$\Delta_5$ on the full $\mathrm{Sp}_4$.
\item $\chi_{\mathrm{top}} = 12$ and $c_1 = 0$: Enriques, 4th
 taxon at level~$2$ ($\mathbb{Z}/2$-gauging). The Borcherds
 weight is~$5/2$ (half-integer); the Siegel cusp form lives on
 the paramodular subgroup $K(2)\subset\mathrm{Sp}_4$ and carries
 a metaplectic cover.
\item $\chi_{\mathrm{top}} = 0$: $T^4$ and bielliptic, taxon~G
 (abelian Heisenberg). No Borcherds lift; the generating function
 is $1$ (degenerate).
\item $c_1\ne 0$: $\mathrm{dP}_9$ and other rational elliptic
 surfaces, taxon~L (affine Kac--Moody at level~$1$). The
 $8$d gauge group is fixed by the elliptic-fibre monodromy;
 for~$\mathrm{dP}_9$ the group is~$E_8$.
\end{itemize}
The arithmetic invariant
$w_{\mathrm{Borch}}\,(X) = 5 \cdot \chi_{\mathrm{top}}(X)/\chi_{\mathrm{top}}(K3)
 = 5\chi_{\mathrm{top}}(X)/24$ reads off the Siegel-form weight
along the 4th-taxon column ($X = K3$: $w = 5$; $X = \mathrm{En}$:
$w = 5/2$), and vanishes on the taxon-G column ($w = 0$). This is
the first-principles content of the K3-specificity claim:
the Witten hypothesis $(\chi, c_1, h^{2,0}) = (24, 0, 1)$ picks out
the unique stratum for which the Borcherds weight is maximal and
integral. Cross-ref Vol.~II
the W-algebras-conditional discussion, the $W_{15}$ supplement section.
\end{remark}

\begin{remark}[Landscape-dichotomy specific/general discipline]
\label{rem:dcy-specific-vs-general}
The four rows of the dichotomy table satisfy a strict
specific/general separation: the K3 row is the \emph{specific
refinement} of the CY-$2$ landscape under~$\Phi_2$, at the maximal
Euler characteristic and integer Borcherds weight; the Enriques
row is the \emph{$\mathbb{Z}/2$-orbifold} of the principal example (specific,
at half the weight); the $T^4$/bielliptic row is the \emph{abelian
degeneration} at~$\chi = 0$ (general Narain Heisenberg, no
refinement); the $\mathrm{dP}_9$ row is the \emph{non-CY affine
limit} (general affine $\widehat{\mathfrak{g}}_k$, no refinement,
no Borcherds weight). No row subsumes another: each corresponds to
a different Drinfeld landscape stratum and a different
automorphic-product regime on the Siegel / modular side. The
K3-specific fingerprint of $\mathbf{H}_{\Delta_5}$ consists of
three invariants that are individually shared with other rows
($c_1 = 0$ is shared with Enriques, $T^4$, bielliptic; $h^{2,0} = 1$
is shared with $T^4$; $\chi_{\mathrm{top}} = 24$ is shared with
Kummer K3 via Mukai 1984), but the triple $(\chi, c_1, h^{2,0}) =
(24, 0, 1)$ is realised only by the K3 row (up to Kummer
sublattice restriction).
\end{remark}

\section{Enriques BKM superalgebra and the four-row \texorpdfstring{$\Psi$}{Psi}-landscape}
\label{sec:dcy-enriques-psi-landscape}

The Enriques weight-$5/2$ row
(Theorem~\ref{thm:dcy-enriques-analogue}) admits an explicit
BKM superalgebra and a $2.M_{12}$-moonshine candidate. Enriques joins
K3, Monster, and Fake-Monster as the fourth $\Psi$-image in the
universal three-faces identity
(Vol.~III Chapter~\ref{ch:k3e-bkm},
Theorem~\ref{thm:bkm-universal-identity}).

\begin{theorem}[Enriques BKM superalgebra
\texorpdfstring{$\mathfrak{g}_{\Delta_5}^{\mathrm{Enr}}$}{g-Delta-5-Enr}]
\label{thm:dcy-enriques-bkm}
\index{Enriques!BKM superalgebra}
\index{$\mathfrak{g}_{\Delta_5}^{\mathrm{Enr}}$}
The Enriques chiral bialgebra
$\mathbf{H}_{\Delta_5}^{\mathrm{Enr}}$ of
Theorem~\ref{thm:dcy-enriques-analogue} is the quasi-Hopf
quantisation of the positive half of a BKM Lie superalgebra
$\mathfrak{g}_{\Delta_5}^{\mathrm{Enr}}$ with:
\begin{itemize}
\item Cartan subalgebra
 $\mathfrak{h}^{\mathrm{Enr}} = \Lambda_{\mathrm{Enr}}\otimes\mathbb{R}
 = (E_8(-1)\oplus\mathrm{II}_{1,1})\otimes\mathbb{R}$ of rank~$10$,
 signature $(1,9)$;
\item ten real simple roots: two hyperbolic-plane generators plus
 eight $E_8$-simple-roots; Gram matrix
 $\bigl(\begin{smallmatrix}0 & 1 \\ 1 & 0\end{smallmatrix}\bigr)
 \oplus (-C_{E_8})$;
\item imaginary simple roots indexed by Fourier coefficients

\section{The global Bridgeland \texorpdfstring{$\leftrightarrow$}{<->} \texorpdfstring{$R$}{R}-matrix dictionary}
\label{sec:dcy-global-bridgeland-dictionary}

Remark~\ref{rem:dcy-kst-bridgeland-rmatrix} installed the Gepner
point avatar of the Conjecture~\ref{conj:wallcrossing-dictionary}
dictionary between Bridgeland wall-crossing and chiral $R$-matrix
gauge transformations, leaving the \emph{global} dictionary
$\Theta\colon\Stab(D^b\mathrm{Coh}(K3))/\Gamma\to
\{R\text{-matrices for }\mathbf{H}_{\Delta_5}\}$ as open. The
theorem below establishes the dictionary; the companion
statement is Theorem~\ref{thm:cwc-global-dictionary} of
Chapter~\ref{ch:coha-wall-crossing-platonic}, and the connection to
the U2 functoriality clause is Remark~\ref{rem:phi-functoriality-summary}.

\begin{theorem}[Beilinson dictionary on \texorpdfstring{$\Stab^\dagger/\Gamma$}{Stab-dagger/Gamma}]
\label{thm:dcy-beilinson-dictionary}
Fix a K3 surface $S$ with distinguished component
$\Stab^\dagger(D^b\mathrm{Coh}(S))\subset\Stab(D^b\mathrm{Coh}(S))$ and
autoequivalence group $\Gamma=\mathrm{Aut}(D^b\mathrm{Coh}(S))$. The
Bayer--Macr\`i--Igusa period map
\[
 \iota\colon\;
 \Stab^\dagger/\Gamma\;\xrightarrow{\;\pi\;}\;
 \mathcal{P}(\widetilde{\Lambda}_{K3})\;\xrightarrow{\;\mathrm{Sieg}\;}\;
 \overline{\mathcal{A}_2},
\]
composed with the rational-theta factorisation
$R_{\mathrm{Sieg,dyn}}(u,Z)=R^{\mathrm{rat}}_{\mathrm{Yang}}(u)
\cdot\theta^{\mathrm{K3}}(u,Z)$, yields a well-defined functor
\[
 \Theta\colon\;
 \Stab^\dagger(D^b\mathrm{Coh}(S))/\Gamma
 \;\longrightarrow\;
 \bigl\{R\text{-matrices for }\mathbf{H}_{\Delta_5}\bigr\},
 \qquad
 \sigma\longmapsto R_{\mathrm{Sieg,dyn}}\bigl(u;\iota(\sigma)\bigr).
\]
Its properties:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $\Theta$ is continuous on chamber interiors; the value inside a
 fixed chamber $C_\sigma$ is a single gauge-equivalence class of
 $R$-matrices satisfying the RTT relation for
 $\mathbf{H}_{\Delta_5}$;
\item on wall-crossing $\sigma\to\sigma'$ across a KS wall
 $W_{(v_1,v_2)}$, $\Theta(\sigma')$ is obtained from $\Theta(\sigma)$
 by Kontsevich--Soibelman gauge conjugation
 $R^{(\sigma')}=F_{\sigma\sigma'}\cdot R^{(\sigma)}\cdot
 F_{\sigma\sigma'}^{-1}$
 (Theorem~\ref{thm:cwc-ks-r-conjugation} of
 Chapter~\ref{ch:coha-wall-crossing-platonic});
\item the three canonical loci
 (large-volume cusp / Gepner Heegner point / Humbert-$H_1$ boundary)
 specialise $\Theta$ to
 $R^{\mathrm{rat}}_{\mathrm{Yang}}$, $R^{\mathrm{trig}}_{q=\zeta_8}$,
 $R^{\mathrm{ell,dyn}}_{\mathrm{Felder}}$ respectively
 (Remark~\ref{rem:cwc-three-tier} of
 Chapter~\ref{ch:coha-wall-crossing-platonic});
\item the image of $\Theta$ is contained in the fourth Drinfeld--Belavin
 class $R_{\mathrm{Sieg,dyn}}$
 (Theorem~\ref{thm:R-sieg-fourth-class}), and every value of
 $\Theta$ is obtained as a specialisation of
 $R_{\mathrm{Sieg,dyn}}$ at an $\overline{\mathcal{A}_2}$-point.
\end{enumerate}
\end{theorem}

\begin{proof}
See Theorem~\ref{thm:cwc-global-dictionary} of
Chapter~\ref{ch:coha-wall-crossing-platonic}. The present theorem records
the same content on the derived-category side: the Bayer--Macr\`i map
$\pi\colon\Stab^\dagger\to\mathcal{P}(\widetilde{\Lambda}_{K3})$
(Bayer--Macr\`i 2014 \emph{Invent.\ Math.}~198 Theorem~1.1) is a local
isomorphism with the autoequivalence-group action
$\Gamma\curvearrowright\Stab^\dagger$ projecting to the Hodge-theoretic
action of $\Gamma$ on periods. The Siegel restriction from the
$\widetilde{\Lambda}_{K3}$-period domain to $\mathbb{H}_2$ along the
hyperbolic $\mathrm{II}_{2,1}$ core is Gritsenko--Nikulin 1998
\emph{Internat.\ J.\ Math.}~9 Section~5; its Igusa-compatibility
(intertwining $\Gamma$ with the paramodular $K(1)$) is
Remark~\ref{rem:cwc-iota-map}. Clauses (i)--(iv) transcribe the
content of
Theorems~\ref{thm:cwc-chamber-coha-drinfeld},
\ref{thm:cwc-ks-r-conjugation},
\ref{thm:cwc-global-dictionary} under the derived-category
identification $\Phi_2(D^b\mathrm{Coh}(K3))\simeq H_{\mathrm{Muk}}$ and
its Hall--Drinfeld double
$\mathcal{D}_\hbar(\mathcal{Y}^{\mathrm{Hall}}_\hbar(\mathrm{CoHA}_{K3}))$
$\simeq \mathbf{H}_{\Delta_5}$
(Remark~\ref{rem:dcy-kst-phi2-k3-coha}).
\end{proof}

\begin{remark}[Closing the Conjecture-\ref{conj:wallcrossing-dictionary}

\section{Stability manifold of the total space \texorpdfstring{$K3 \times E$}{K3 x E}}
\label{sec:dcy-stab-k3-times-e}

The CY-$3$ total space $Y = K3 \times E$ admits a derived category $D^b(\Coh(Y))$. On any nonempty locally finite Bridgeland component of this category, the central-charge target has numerical rank $48$. The product-stable K\"unneth locus, when defined, has expected dimension $26$. The difference is the $22$-dimensional numerical mixed-charge sector visible to the framed $\Phi_3$ comparison but invisible to the product decomposition $\Phi_2 \boxtimes \Phi_1$.

\subsection{Dimension via numerical K-theory}
\label{subsec:dcy-stab-dim-k3-times-e}

\begin{theorem}[Numerical stability dimension of \texorpdfstring{$D^b(\Coh(K3 \times E))$}{D-b(Coh(K3 x E))}]
\label{thm:dcy-stab-dim-48}
Let $Y = K3 \times E$ with $K3$ a smooth projective complex K3 surface and $E$ a smooth projective complex elliptic curve. Every nonempty locally finite Bridgeland component of $D^b(\Coh(Y))$ has central-charge target of complex dimension
\[
 \rk K_{\mathrm{num}}\bigl(D^b(\Coh(K3 \times E))\bigr) \;=\; 48.
\]
The statement is numerical; it does not assert nonemptiness of the full CY$_3$ stability manifold.
\end{theorem}

\begin{proof}
By Bridgeland 2007 \emph{Ann.\ Math.}~166 Theorem~1.2, for any equivalently small triangulated category $\cC$ with finite-rank numerical Grothendieck group, the map $\cZ \colon \Stab(\cC) \to \Hom_\Z(K_{\mathrm{num}}(\cC), \C)$ sending $(Z, \cP) \mapsto Z$ is a local homeomorphism onto an open subset of the finite-dimensional complex vector space $\Hom_\Z(K_{\mathrm{num}}(\cC), \C)$. On any nonempty component,
\[
 \dim_\C \Stab(\cC) \;=\; \rk K_{\mathrm{num}}(\cC).
\]
For $\cC = D^b(\Coh(Y))$ with $Y$ smooth projective, $K_{\mathrm{num}}(\cC) = K_{\mathrm{num}}(Y)$, the numerical quotient of $K_0(Y)$ by classes of Euler-pairing norm zero. The Chern character $\mathrm{ch} \colon K_{\mathrm{num}}(Y) \otimes \Q \xrightarrow{\sim} H^{\mathrm{even}}_{\mathrm{alg}}(Y, \Q)$ of Atiyah 1957 \emph{Proc.\ LMS}~VII (in the form refined by Mukai 1987 \emph{Nagoya Math.\ J.}~81 for K3 surfaces and extended by Thomason 1988 \emph{Duke}~56 Theorem~2.1 to the Künneth property of algebraic K-theory) gives
\[
 \rk K_{\mathrm{num}}(K3 \times E)
 \;=\; \rk K_{\mathrm{num}}(K3) \cdot \rk K_{\mathrm{num}}(E)
 \;=\; 24 \cdot 2
 \;=\; 48.
\]
The factor $24$ is the rank of the Mukai lattice $\widetilde{\Lambda}_{K3} = H^0 \oplus H^2 \oplus H^4$ of the K3 surface, of signature $(4, 20)$, decomposed as $1 + 22 + 1$. The factor $2$ is the rank of $K_{\mathrm{num}}(E) = \Z[\cO_E] \oplus \Z[\cO_p]$ generated by the structure sheaf and a skyscraper at a point, the even cohomology $H^0(E) \oplus H^2(E) = \Z \oplus \Z$. Thus any component of $\Stab(D^b(\Coh(Y)))$ whose local homeomorphism is defined has numerical dimension $48$.
\end{proof}

\begin{remark}[Multi-path verification of the dimension \texorpdfstring{$48$}{48}]
\label{rem:dcy-stab-48-multipath}
Four independent routes produce the same value:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{Bridgeland 2007 Theorem~1.2 on numerical K-theory.} On every nonempty component, $\dim_\C \Stab(\cC) = \rk K_{\mathrm{num}}(\cC) = 48$.
\item \emph{Thomason 1988 Künneth on K-theory of smooth projective varieties.} $K_{\mathrm{num}}(K3 \times E) = K_{\mathrm{num}}(K3) \otimes_\Z K_{\mathrm{num}}(E)$ of rank $24 \cdot 2 = 48$.
\item \emph{Even cohomology count via Atiyah--Hirzebruch.} Applying the Chern character isomorphism $\mathrm{ch} \colon K_{\mathrm{num}}(Y) \otimes \Q \xrightarrow{\sim} H^{\mathrm{even}}(Y, \Q)$ (Atiyah 1957) to the Künneth expansion yields $H^{\mathrm{even}}(K3 \times E) = H^{\mathrm{even}}(K3) \otimes H^{\mathrm{even}}(E)$ of dimension $24 \cdot 2 = 48$.
\item \emph{Toda 2013 CY$_3$ stability framework.} The CY$_3$ deformation-theoretic setup uses the same numerical Grothendieck group; the K\"unneth expansion gives rank $48$.
\end{enumerate}
The four routes agree on the numerical target; all are independent of any algebraization choice for $\Phi_3$.
\end{remark}

\begin{remark}[Scope: \texorpdfstring{$K_{\mathrm{num}}$}{K-num} of rank \texorpdfstring{$48$}{48} versus \texorpdfstring{$H^\bullet$}{H-.} of rank \texorpdfstring{$96$}{96}]
\label{rem:dcy-stab-48-vs-96-scope}
The full Betti total of $K3 \times E$ is
\[
 \sum_i \dim_\Q H^i(K3 \times E, \Q)
 \;=\; \Bigl(\sum_i h^i(K3)\Bigr) \cdot \Bigl(\sum_i h^i(E)\Bigr)
 \;=\; 24 \cdot 4 \;=\; 96,
\]
splitting as $H^{\mathrm{even}} \oplus H^{\mathrm{odd}}$ of ranks $48 + 48$. The odd part
\[
 H^{\mathrm{odd}}(K3 \times E) \;=\; H^{\mathrm{even}}(K3) \otimes H^1(E) \;\oplus\; H^{\mathrm{even}}(K3) \otimes H^3(E) \;=\; 24 \cdot 2 \;=\; 48
\]
carries zero Euler pairing against every algebraic class (the pairing pairs $H^p$ against $H^{6-p}$, and the cup product $H^{\mathrm{even}}(K3) \otimes H^1(E) \to H^{\mathrm{odd}}(K3 \times E)$ has image orthogonal to the algebraic diagonal). Consequently the odd summand lies in the kernel of the numerical Grothendieck quotient $K_0(Y) \twoheadrightarrow K_{\mathrm{num}}(Y)$ and does not contribute to the Bridgeland central-charge target. The numerical stability manifold sees exactly the algebraic-Mukai-pairing content, rank $48$, not the full Betti rank $96$.
\end{remark}

\subsection{The Künneth slice and its codimension}
\label{subsec:dcy-kunneth-slice-codim}

\begin{definition}[\texorpdfstring{$\Phi$}{Phi}-accessible Künneth slice]
\label{def:dcy-kunneth-slice}
The \emph{$\Phi$-accessible K\"unneth slice} is the product-stable locus
\[
 \Stab^\Phi(Y) \;:=\; \Stab\bigl(D^b(\Coh(K3))\bigr) \times \Stab\bigl(D^b(\Coh(E))\bigr)
 \;\hookrightarrow\; \Stab\bigl(D^b(\Coh(Y))\bigr)
\]
parametrising pairs $(\sigma_{K3}, \sigma_E)$ for which the external-product slicing satisfies Harder--Narasimhan and defines a stability condition on the generated product subcategory. Its central charge assigns $Z_{K3}(E_1) \cdot Z_E(E_2)$ to a pure product class $[E_1 \boxtimes E_2] \in K_{\mathrm{num}}(Y)$. The embedding uses the fixed-specialisation compatibility clause: on the K\"unneth slice, the framed object-level assignment is compared with $\Phi_2(\sigma_{K3}) \boxtimes \Phi_1(\sigma_E)$ as a product $E_1$-chart on the chiral side; no global $\Phi_3$ product functor is asserted.
\end{definition}

\begin{proposition}[Codimension of the K\"unneth slice]
\label{prop:dcy-kunneth-codim-22}
Assume a nonempty CY$_3$ stability component containing the product-stable
K\"unneth slice of Definition~\ref{def:dcy-kunneth-slice}.
The K\"unneth slice has complex dimension
\[
 \dim_\C \Stab^\Phi(Y) \;=\; \dim_\C \Stab(K3) + \dim_\C \Stab(E) \;=\; 24 + 2 \;=\; 26,
\]
and therefore
\[
 \codim_\C \bigl(\Stab^\Phi(Y) \hookrightarrow \Stab(Y)\bigr) \;=\; 48 - 26 \;=\; 22.
\]
\end{proposition}

\begin{proof}
The factor dimensions follow from Bridgeland 2007 Theorem~1.2 applied to each tensor factor: $\rk K_{\mathrm{num}}(K3) = 24$ gives $\dim_\C \Stab(K3) = 24$ on the distinguished component $\Stab^\dagger(K3)$ of Bridgeland 2008 \emph{Duke}~141, and $\rk K_{\mathrm{num}}(E) = 2$ gives $\dim_\C \Stab(E) = 2$ on the single component identified by Bridgeland 2007 (Example~9.3) and Okada 2006. The product-stable locus has complex dimension $24 + 2 = 26$. By Theorem~\ref{thm:dcy-stab-dim-48}, the ambient numerical target has dimension $48$. Subtraction yields the expected codimension $48 - 26 = 22$ on any component where the product-stable embedding is defined.
\end{proof}

\begin{proposition}[Non-K\"unneth numerical witnesses]
\label{prop:dcy-non-kunneth-witnesses}
The numerical complement to the K\"unneth product tangent has dimension
$22$, the rank of the K3 transcendental lattice. Three explicit
families detect non-product classes in this complement:
\begin{enumerate}[label=\textup{(W\arabic*)}]
\item \emph{Non-split Fourier--Mukai twists.} Let $\cP \in D^b(\Coh(K3 \times K3^\vee))$ be the universal sheaf and $\cP_\tau$ a Fourier--Mukai kernel on $K3 \times E \times K3 \times E$ obtained by twisting $\cP$ along a rank-$2$ variation $\tau \in H^0(E, \cO_E^{\oplus 22})$ of transcendental classes. The resulting Fourier--Mukai transform acts on numerical classes by mixing the $22$-dimensional transcendental lattice $T_{K3}$ with the rank-$2$ elliptic lattice, giving non-product tangent directions (Mukai 1987 Nagoya~81 in the Mukai-lattice form; C\u{a}ld\u{a}raru 2003 for the Fourier--Mukai extension).
\item \emph{Non-split vertical extensions.} For $p \in E$ and $[\cF] \in T_{K3}$ a transcendental class, the extension
\[
 0 \to \pi_E^* \cO_E(p) \to \cE \to \pi_{K3}^* \cF \to 0
\]
has extension class in $\Ext^1(\pi_{K3}^* \cF, \pi_E^* \cO_E(p)) = H^0(K3, \cF^\vee) \otimes H^1(E, \cO_E(p))$, non-zero whenever $\cF$ is transcendental. Non-zero extensions are not products of K3-objects with $E$-objects; their numerical classes lie in the complement to the product tangent.
\item \emph{Isogeny-graph spectral sheaves.} For an isogeny $\phi \colon E \to E'$ and a spectral cover $\Sigma_\phi \subset K3 \times E$ transverse to the projection $\pi_E$, the pushforward $\phi_* \cO_{\Sigma_\phi}$ is a CY$_3$-rigid sheaf whose support is not of product form. Varying $\phi$ through the isogeny graph of $E$ gives numerical classes parametrised by $T_{K3} \otimes \End(E)$ (Toda 2013 \emph{Adv.\ Math.}~234 for the rigidity and Toda 2014 for the Bridgeland stability framework in the CY$_3$ setting).
\end{enumerate}
The rank calculation gives the dimension of the complement; the three
families identify geometric sources of its non-product classes. Producing
actual stability conditions in every such numerical direction is a
separate CY$_3$ stability problem.
\end{proposition}

\begin{remark}[Physical interpretation: class-\texorpdfstring{$\cS$}{S} mixed-charge sector]

\section{K3 twisted periods realise the Hodge structure on \texorpdfstring{$H^\bullet_{\mathrm{CE}}(\mathfrak{g}_{\Delta_5})$}{H-.-CE(g-Delta-5)}}
\label{sec:dcy-k3-twisted-periods-ce}

The mixed Hodge structure on the Chevalley--Eilenberg cohomology
$H^\bullet_{\mathrm{CE}}(\mathfrak{g}_{\Delta_5})$
(Chapter~\textup{\ref{ch:cy-categories}}
\S\textup{\ref{sec:cycat-ce-hodge-delta5}},
Theorem~\textup{\ref{thm:cycat-ce-hodge-real-imag}}) admits an explicit
realisation in terms of twisted K3 period integrals of
$\phi_{0,1}^{\mathrm{K3}}$ against $\omega_{\mathrm{K3}}^k$. This section
makes the period-theoretic data concrete for $D^b(\Coh(K3))$-input via
$\Phi_2$, and records the resulting Hodge-theoretic invariants at
the level of derived-category evaluations.

\begin{remark}[K3 period map and Mukai \texorpdfstring{$\mathrm{II}_{4,20}$}{II-4,20}]
\label{rem:dcy-k3-period-map-mukai}
\index{K3 period map!Mukai extension}
The classical K3 period map
$\mathrm{Per}: H^2(\mathrm{K3}, \mathbb{Z}) \to \mathbb{C}$ sends a
2-cycle class $\gamma$ to $\int_\gamma \omega_{\mathrm{K3}}$, where
$\omega_{\mathrm{K3}} \in H^{2,0}(\mathrm{K3})$ is the K3 holomorphic
2-form (unique up to $\mathbb{C}^*$-scale). Extending to the Mukai
lattice $\mathrm{II}_{4,20} = H^0(\mathrm{K3}) \oplus H^2(\mathrm{K3})
\oplus H^4(\mathrm{K3})$, the Mukai period map gives a
\emph{weight-$0$ Hodge structure}
\[
 H^\bullet_{\mathrm{Muk}}(\mathrm{K3}) \otimes \mathbb{C}
 \;=\; H^{(2,-2)}_{\mathrm{Muk}} \oplus H^{(1,-1)}_{\mathrm{Muk}}
 \oplus H^{(0,0)}_{\mathrm{Muk}} \oplus H^{(-1,1)}_{\mathrm{Muk}}
 \oplus H^{(-2,2)}_{\mathrm{Muk}}
\]
with $H^{(2,-2)}_{\mathrm{Muk}} = H^{2,0}(\mathrm{K3})$ (rank 1,
contains $\omega_{\mathrm{K3}}$), $H^{(0,0)}_{\mathrm{Muk}} = H^{1,1}
\oplus H^0 \oplus H^4$ (rank 22), and complex conjugates. This is the
input to the $\Phi_2$-composition: $D^b(\Coh(K3)) \xmapsto{\Phi_2}
H_{\mathrm{Muk}}$ is the rank-24 Heisenberg lattice VOA carrying the
Mukai weight-0 Hodge structure (Example~\textup{\ref{ex:k3-derived}}).
The pullback along $\Phi_2$ of the Mukai Hodge structure recovers
the Hodge filtration on $H^\bullet_{\mathrm{CE}}(\mathfrak{g}_{\Delta_5})$
via the chiral Hall-Drinfeld double
$\mathbf{H}_{\Delta_5} = \mathcal{D}_\hbar(\mathcal{Y}^{\mathrm{Hall}}_\hbar(\mathrm{CoHA}_{K3}))$.
\end{remark}

\begin{theorem}[Twisted period integrals realise MHS on imaginary-root CE cohomology]
\label{thm:dcy-k3-twisted-periods-realise-mhs}
The proof combines Theorem~\textup{\ref{thm:cycat-ce-hodge-real-imag}}
with Bruinier's singular-theta Hodge decomposition LNM~$1780$
\S~$2.2$.
For each imaginary root $\alpha \in \Delta_+^{\mathrm{im}}$ of
$\mathfrak{g}_{\Delta_5}$ and each $k \geq 1$, the graded piece
$\mathrm{gr}^W_{2k} \mathrm{gr}^F_p H^k_{\mathrm{CE},\alpha}
(\mathfrak{g}_{\Delta_5})$ admits a basis of \emph{motivic periods}
given by iterated K3 twisted period integrals:
\[
 \pi_\alpha^{(p, k-p)} \;=\;
 \int_{\gamma_\alpha^{\otimes k}} \bigl(\phi_{0,1}^{\mathrm{K3}}\bigr)^{\otimes k}
 \cdot \omega_{\mathrm{K3}}^{\otimes p} \cdot \overline{\omega}_{\mathrm{K3}}^{\otimes (k-p)}
 \;\in\; \mathbb{C}/\mathbb{Z}(k),
\]
with $\gamma_\alpha \in H_2(\mathrm{K3}, \mathbb{Z})$ the
$\alpha$-specific Mukai 2-cycle of
Remark~\textup{\ref{rem:cycat-ce-twisted-periods}}, and
$\mathbb{Z}(k) = (2\pi i)^k \mathbb{Z}$. The assembly
\[
 \bigl\{\pi_\alpha^{(p, k-p)}\bigr\}_{\alpha \in \Delta_+^{\mathrm{im}},\ 0 \leq p \leq k}
 \;\in\; \mathbb{C}^{\sum_\alpha (k+1)}
\]
is the full period matrix of $H^k_{\mathrm{CE},\alpha}(\mathfrak{g}_{\Delta_5})$
on the imaginary-root sector, and its entries generate a sub-ring of
$\mathbb{C}$ of transcendence degree bounded by $\binom{k+1}{2}
\cdot \max_{D \leq 4nm-\ell^2} c_{\mathrm{K3}}(D)$ over $\mathbb{Q}$.
\end{theorem}

\begin{remark}[Mukai \texorpdfstring{$\mathrm{II}_{4,20}$}{II-4,20} weight-0 Hodge and Deligne
period conductors]
\label{rem:dcy-mukai-hodge-conductor}
The Mukai lattice $\mathrm{II}_{4,20}$ carries a canonical weight-$0$
Hodge structure (Mukai 1984). Its Hodge numbers are
$h^{(2,-2)} = h^{(-2,2)} = 1$, $h^{(0,0)} = 22$, with the middle
summand further split into $\mathrm{algebraic}\oplus\mathrm{transcendental}$
by the Picard rank $\rho(X) \in \{1, \ldots, 20\}$. Under the Mukai
pairing (Proposition~\textup{\ref{prop:serre-duality-cy}}), the
transcendental lattice
$T(X) = \mathrm{Pic}(X)^\perp \cap H^2(X,\mathbb{Z})$ carries a pure
Hodge structure of type $(2,0) + (1,1) + (0,2)$ with ranks
$(1, 20 - \rho, 1)$. For $\rho = 1$ (generic K3), this is rank 21
with full Hodge structure. This weight-$0$ Mukai Hodge structure
matches the CE-cohomology Hodge filtration of
Definition~\textup{\ref{def:cycat-ce-hodge-filtration}}:
the Borcherds singular-theta lift intertwines the two.

The period conductor of the Mukai lattice; in the sense of
Deligne (1971, \emph{Inst.\ Hautes \'Etudes Sci.}~40, Hodge II
Thm~$2.3.5$); is the determinant of the period matrix
$\{\pi_{ij}\}$ for a basis of transcendental classes. For generic
K3, this is a transcendental number whose log is proportional to
the special-value $L(H^2(\mathrm{K3}), s)|_{s=1}$ (Beilinson
conjecture, level of $L$-function on K3 motive). The Mukai
extension multiplies by $(2\pi i)^2$ from the $H^0 \oplus H^4$
factor, giving the $\mathbb{Z}(2)$-Tate twist visible in the top
$H^4_{\mathrm{CE}}$ summand of Example~\textup{\ref{ex:hh-k3}}
(the $\HH^4 \simeq k^1$ summand).
\end{remark}

\begin{remark}[Comparison: K3 Hodge-periodicity vs.\ CY\texorpdfstring{$_3$}{-3} \texorpdfstring{$K3 \times E$}{K3 x E} period data]

\section{The quintic as the rank-one test of the master identity}
\label{sec:bl-quintic-deployment}

The master identity $\kappa_{\mathrm{BKM}}(\Phi_N) = c_N(0)/2 =
\mathrm{STr}_{\mathrm{mod.op}}[B^{E_3}(\Phi^{\mathrm{FA}}_3(\cC_N))]$
in Vol~III; three chain-level routes (Hodge supertrace,
renormalised BV, Maulik--Okounkov residue) to the Borcherds weight of
the automorphic lift attached to a CY$_3$ category; is proved on
$K3 \times E$ with paramodular target $\Delta_5$. The quintic threefold
supplies the complementary instance: a rigid compact CY$_3$ outside the
K3-fibred class, with Kahler rank one. What the three routes produce on
$Q$ determines where the master identity has content, and where the
chiral Positselski inversion at $d = 3$ reduces to a trivial identity.

\subsection{Standing data on $Q$}
\label{subsec:quintic-standing}

Let $Q = \{f_5 = 0\} \subset \P^4$ be a generic smooth quintic hypersurface.
Hodge diamond: $h^{0,0} = h^{3,0} = h^{0,3} = h^{3,3} = 1$,
$h^{1,1} = 1$, $h^{2,1} = h^{1,2} = 101$, $h^{2,2} = 1$, remaining entries
zero (Griffiths 1969 \emph{Ann.\ Math.}~90; Clemens 1983). Topological
Euler $\chi(Q) = 2(h^{1,1} - h^{2,1}) = -200$. Categorical invariant
$\kappa_{\mathrm{cat}}(Q) = \chi(\cO_Q) = 1 - 0 + 0 - 1 = 0$. The
hyperplane class $H$ generates $H^{1,1}(Q, \Z) = \Z \cdot H$; $H^3 = 5$.
Tree-level Yukawa $Y_3 = H^3 = 5$. The Kodaira--Spencer dgLa
$(\PV^{\bullet, \bullet}(Q), \bar\partial, \{-,-\}_{\mathrm{SN}})$ carries
the curvature $F^Q = Y_3 \cdot [\Omega_Q^{3,0}]$ of
Theorem~\ref{thm:curved-kontsevich-quintic}.

\subsection{Path~1: Hodge supertrace fires}
\label{subsec:quintic-path-1-hodge}

\begin{theorem}[Hodge supertrace on the quintic]
\label{thm:quintic-path-1-hodge}
Assume the compact Stage-$1$ framing witness needed to interpret
$\cA_Q = \Phi_3^{\mathrm{FA}}(D^b\Coh(Q))$ as the $E_3$-holomorphic
factorisation algebra of the quintic's derived category. The
modular-operadic supertrace of its $E_3$-Koszul dual, computed with the
corrected TCFT comparison operator and localised to the $(0,\bullet)$
column of the Hochschild--Kostant--Rosenberg decomposition via the
Grothendieck residue against $\Omega_Q^{3,0}$, equals the Hodge Euler
characteristic
\[
 \kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(Q) \;=\;
 \mathrm{STr}_{\mathrm{mod.op},\,\mathrm{Hodge}}
 \bigl[B^{E_3}(\cA_Q)\bigr] \;=\;
 \sum_{q = 0}^{3} (-1)^q h^{0, q}(Q)
 \;=\; 1 - 0 + 0 - 1 \;=\; 0,
\]
in agreement with $\kappa_{\mathrm{cat}}(Q) = 0$.
\end{theorem}

\begin{proof}
Theorem~\ref{thm:curved-kontsevich-quintic} isolates the necessary
curvature term $F^Q = Y_3 \cdot [\Omega^{3,0}]$ for any compact curved
witness. The Hodge supertrace route pairs the corrected TCFT bar
operator $B^{E_3}_{\mathrm{TCFT}}(\cA_Q)$ against the
Getzler--Kapranov fundamental class $[\Mbar_{2, 0}]$ with the
zero-Fourier-coefficient localisation. Kapranov's cyclic-moduli
theorem (1999 \emph{Compos.\ Math.}~115) reduces the genus-$2$
pairing to iterated genus-$0$ cyclic pairings, whose Koszul dual is
computed by Ayala--Francis 2014 (\emph{Poincar\'e/Koszul duality}
Theorem~2.1). The Grothendieck residue against $\Omega_Q^{3,0}$
localises this iterated pairing to the $(0, \bullet)$-column of
$\bigoplus_{p + q} H^q(Q, \wedge^p T_Q)$, giving the alternating
sum $\sum_q (-1)^q h^{0, q}(Q)$. Substituting the quintic Hodge
diamond: $h^{0,0} = 1$, $h^{0,1} = 0$ (simple-connectedness),
$h^{0,2} = 0$ (rigidity), $h^{0,3} = 1$ (holomorphic volume form),
yielding $1 - 0 + 0 - 1 = 0$. The factorisation through the
$\GRT_1(\Q)$-torsor of Drinfeld-associator choices is invariant
under the quotient action; the value $0$ is canonical. The raw
termwise operator $B^{(2)}_{\mathrm{term}}$ is not the TCFT comparison
operator and does not by itself close the compact $\Phi_3$ witness.
\end{proof}

\begin{remark}[Hodge supertrace $= \chi(\cO_X)$ on the strict stratum]
\label{rem:quintic-hodge-supertrace-equals-chi}
The argument of Theorem~\ref{thm:quintic-path-1-hodge} generalises:
for every compact CY$_3$ $X$ in the strict Beauville--Bogomolov
stratum (i.e.\ $h^{0, 1}(X) = h^{0, 2}(X) = 0$,
$h^{0, 3}(X) = 1$), the Hodge-route characteristic equals the
categorical Euler characteristic,
\[
 \kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(X) \;=\; \chi(\cO_X) \;=\;
 \kappa_{\mathrm{cat}}(X) \;=\; 0.
\]
The quintic, bicubic $(3, 3)$-complete intersection, Godeaux CY$_3$,
and R{\o}dland Pfaffian CY$_3$ all satisfy this identification. The
$K3 \times E$ isotrivial-fibration stratum exits the strict class:
its $h^{0, 2}(K3 \times E) = h^{0, 2}(K3) = 1 \neq 0$, and the
Hodge-route numeric receives a different contribution
($\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(K3 \times E) = 3$, not $0$).
\end{remark}

\subsection{Path~2: renormalised BV reduces to rank-one modularity}
\label{subsec:quintic-path-2-bv}

\begin{theorem}[BV-route output on the quintic is $\SL_2(\Z)$-modular]
\label{thm:quintic-path-2-bv-sl2}
BCOV free energies are available; interpreting their rank-one
large-volume expansion as a framed $\Phi_3$ output requires the compact
curved witness of
Theorem~\ref{thm:cy-c-beyond-quintic-strictification-obstruction}.
Let $\mathcal{E}^\bullet_{Q, \mathrm{ren}}$ be the Costello--Li--
Williams 2020 renormalised factorisation algebra of BCOV theory on
the quintic, conditional on the all-genera witness of
Theorem~\ref{thm:all-genera-bcov-transfer-quintic}. The zeroth
Fourier coefficient of its partition function
\[
 Z^{\mathrm{BV}}_Q(\tau) \;=\; \exp\!\Bigl(\sum_{g \geq 0} \hbar^g\,
 F_g^{\mathrm{BCOV}}(Q)(\tau)\Bigr),
 \qquad \tau \in \mathbb{H}_1,
\]
at the large-volume cusp $\tau \to i \infty$ vanishes:
$c^{\mathrm{BV}}(0) = 0$, hence
$w_{\mathrm{BV}}^{\mathrm{rk}\,1}(Q) :=
c^{\mathrm{BV}}(0)/2 = 0$. This is a BV-route weight coefficient,
not a canonical $\kappa_{\mathrm{BKM}}$ for the quintic. The partition function is modular of
weight $2g - 2$ in $F_g$ under a finite-index subgroup
$\Gamma_Q \subset \SL_2(\Z)$ fixing the large-volume cusp, not
under any subgroup of $\Sp_4(\Z)$.
\end{theorem}

\begin{proof}
The Costello--Li--Williams renormalised BV partition function on a
compact CY$_3$ with $h^{0, 1}(X) = 0$ depends on the complexified
Kahler modulus $\tau \in H^{1, 1}(X, \C) / (H^{1, 1}(X, \Z) + i
\cdot \mathrm{Kahler\ cone})$. For $h^{1, 1}(Q) = 1$, this moduli
space is the one-dimensional upper half-plane
$\mathcal{M}_Q^{\mathrm{K}} \simeq \mathbb{H}_1 / \Gamma_Q$ with
$\Gamma_Q \subset \SL_2(\Z)$. Candelas--de~la~Ossa--Green--Parkes
1991 (\emph{Nucl.\ Phys.\ B}~359) compute
\[
 F_0^{\mathrm{BCOV}}(Q)(t) \;=\; \tfrac{5}{6} t^3 + \sum_{d \geq 1}
 n^0_d \,\mathrm{Li}_3(e^{d t}),
\]
vanishing at $t = 0$ up to a constant; Klemm--Theisen 1993
(\emph{Nucl.\ Phys.\ B}~389) verify that the full
$F_1^{\mathrm{BCOV}}(Q) = -(62/12) \log(\psi^5 - 1)$ has no
constant term in the $q = e^{2\pi i \tau}$-expansion at the
large-volume cusp. For $g \geq 2$, Yamaguchi--Yau 2004
(\emph{Adv.\ Theor.\ Math.\ Phys.}~8) express $F_g$ as a polynomial
of degree $3g - 3$ in the BCOV propagator variables with rational
coefficients; the large-volume expansion's $q^0$-term vanishes by
the holomorphic-anomaly equation of
Lemma~\ref{lem:two-loop-anomaly-quintic}. The summed partition
function therefore has $c^{\mathrm{BV}}(0) = 0$.

The one-parameter Kahler modulus forces $Z^{\mathrm{BV}}_Q$ to
live on $\mathbb{H}_1$. The Kuga--Satake Shimura target of~$Q$
has signature $(2, h^{1, 1} + 1) = (2, 2)$, which by
Matsushima--Shimura classification (extended Cuntz--Skoruppa 1999
\emph{J.\ Number Theory}~79) is
$\mathrm{Sh}_{\mathrm{O}(2, 2)} \simeq \mathbb{H}_1 \times
\mathbb{H}_1 / \Gamma$ with $\Gamma$ a Hilbert modular group of a
real quadratic field, \emph{not} $\mathbb{H}_2 / \Sp_4(\Z)$. The
Borcherds singular-theta lift on the rank-$4$ signature-$(2, 2)$
lattice of $Q$ produces a product of two classical modular forms
on $\mathbb{H}_1 \times \mathbb{H}_1$, not a genuine paramodular
Siegel form.
\end{proof}

\begin{remark}[$a_{\mathrm{BV}}(Q) = -25/3$ lives
in higher Fourier coefficients]
\label{rem:quintic-kch-bv-higher-fourier}
The BV anomaly coefficient $a_{\mathrm{BV}}(Q) =
\chi(Q)/24 = -200/24 = -25/3$ of Vol~III's canonical table
(\cite{VolI}
Remark~\ref{rem:kappa-ch-quintic-two-views}) is the genus-$1$
one-loop coefficient of the BV partition function; it is a
$q^1$-, not $q^0$-, content of $Z^{\mathrm{BV}}_Q(\tau)$ at the
large-volume cusp, and thus does not enter
$w_{\mathrm{BV}}^{\mathrm{rk}\,1}(Q) = 0$. The
four-$\kappa_\bullet$ discipline of the Vol~III canonical table keeps these
objects separate: $a_{\mathrm{BV}}$ is a BV-construction coefficient
for the chiral algebra $\cA_Q$, encoding the higher-genus
free-energy constant terms; $\kappa_{\mathrm{BKM}}$ is reserved for
the Borcherds automorphic lift, and no such quintic BKM invariant is
defined by current data.
\end{remark}

\subsection{Path~3: Maulik--Okounkov residue is not constructed on the generic quintic}
\label{subsec:quintic-path-3-mo}

\begin{theorem}[MO-route obstruction on the generic quintic]
\label{thm:quintic-path-3-mo-scalar}
Let $Q\subset\P^4$ be a generic smooth quintic. The present data do not
construct a Maulik--Okounkov stable-envelope $R$-matrix on
$\mathrm{Hilb}^\bullet(Q)$, hence they do not define
an MO residue coefficient for $\kappa_{\mathrm{BKM}}(Q)$. The obstruction is structural:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item a generic quintic has no non-trivial algebraic
 $\mathbb C^\times$-action preserving $Q$;
 \item $\mathrm{Hilb}^n(Q)$ is a DT moduli space with symmetric
 obstruction theory, not a Nakajima/quiver symplectic resolution
 carrying MO stable envelopes;
 \item the GV/PT coefficients $n^g_d$ are enumerative Fourier
 coefficients, not residues of an MO $R$-matrix without the
 missing equivariant stable-envelope package.
\end{enumerate}
\end{theorem}

\begin{proof}
For a very general hypersurface of degree at least three in projective
space, the automorphism group is finite; for a generic quintic it is
trivial. Thus the ambient $\mathbb C^\times$-actions on $\P^4$ do not
descend to a generic $Q$. Maulik--Okounkov stable envelopes require an
equivariant symplectic-resolution or quiver-variety setting with
chambers and attracting loci. The Hilbert schemes of points on a
Calabi--Yau threefold carry the DT symmetric obstruction theory used by
MNOP and Pandharipande--Thomas, but they are not the Nakajima varieties
for which the MO $R$-matrix construction applies. Therefore the known
GV/PT expansion on the quintic remains enumerative data at positive
curve classes; it is not a constructed MO residue and cannot be used
as a $\kappa_{\mathrm{BKM}}$ witness.
\end{proof}

\begin{remark}[Klemm--Pandharipande BPS content lives at $d \geq 1$]
\label{rem:quintic-kp-bps-content}
The rich BPS content of the quintic; $n^0_1 = 2875$,
$n^0_2 = 609250$, $n^0_3 = 317206375$, $n^1_3 = 609250$,
$n^2_4 = 534750$, through $(g, d) = (51, 10)$ per
Klemm--Pandharipande 2008; is visible in the Fourier
coefficients of the GV/PT partition function for $d \geq 1$. These
coefficients do not become $c^{\mathrm{MO}}(d)$ until an equivariant
stable-envelope package is constructed. Accordingly the MO route is
not empty; it is presently undefined on the generic quintic.
\end{remark}

\subsection{The precise form of the CY-B$_3$ gap at $d = 3$}
\label{subsec:quintic-cy-b3-gap}

The three paths do not converge on a quintic Borcherds weight.
Path~1 gives the Hodge supertrace
$\kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(Q)=0$; Path~2 gives the BCOV
shadow $\chi_{\mathrm{top}}(Q)/24=-25/3$ only after the compact curved
witness is supplied; Path~3 is not constructed on the generic quintic.
On $K3 \times E$ the Borcherds lane does converge on the non-trivial value
$\kappa_{\mathrm{BKM}}(\mathfrak{g}_{\Delta_5}) = c_{\Delta_5}(0)/2
= 5$; the Siegel paramodular target $\Delta_5$ supplies the
content. On the quintic the Siegel target is not constructed. The
content-bearing part of CY-B$_3$ at $d = 3$ requires either the
corrected TCFT comparison for the compact bar operator or a full
filtered $\HH^{-2}_{E_1}$ vanishing theorem; the raw termwise
$B^{(2)}_{\mathrm{term}}$ is not enough.

\begin{theorem}[CY-B$_3$ scope on strict-stratum compact CY$_3$]
\label{thm:cy-b3-scope-quintic-chapter}
Assume a compact framed $\Phi_3$ witness and a paramodular output
construction outside the K3-fibred class.
Let $X$ be a smooth projective compact Calabi--Yau threefold in
the strict Beauville--Bogomolov stratum (Theorem~\ref{thm:phi3-ch-essential-image}).
The chiral Positselski inversion at $d = 3$ has the following
witness-dependent scope:
\begin{enumerate}[label=\textup{(\Roman*)}]
\item \textbf{K3-fibred with $h^{1, 1}(X) \geq 3$ and Kuga--Satake
paramodular structure.} CY-B$_3$ is content-bearing: the bar--cobar
Quillen equivalence produces a non-trivial Siegel paramodular
form on $\mathbb{H}_2 / \Sp_4(\Z)$; worked example $X = K3 \times E$
with target $\Delta_5$, weight $5$ (Vol~III
\cite{VolI}
Theorem~\ref{thm:k3e-bkm-siegel}).
\item \textbf{Non-K3-fibred with $h^{1, 1}(X) = 1$ (quintic-type).}
The Hodge and BCOV shadows are computable, but no Borcherds lift,
Jacobi seed, MO stable-envelope residue, or strict compact framing
witness is presently constructed. The bar--cobar identity may hold on
the categorical/curved locus once the witness is supplied; no
$\kappa_{\mathrm{BKM}}$ is defined by current data.
\item \textbf{Non-K3-fibred with $h^{1, 1}(X) \geq 3$.} The
content-bearing CY-B$_3$ case remains open: the Kuga--Satake target has
signature $(2, n)$ with $n \geq 3$ and permits a Siegel paramodular
output, but no explicit Borcherds lift analogous to $\Delta_5$ has
been constructed for such $X$.
\end{enumerate}
\end{theorem}

\begin{proof}
Clause~(I) is the $K3 \times E$ theorem of
Vol~III \cite{VolI}. Clause~(II)
is Theorem~\ref{thm:cy-c-beyond-quintic-strictification-obstruction}
combined with Theorem~\ref{thm:quintic-path-3-mo-scalar}: the quintic
has computable Hodge/BCOV shadows but lacks both a compact framing
witness and an MO/Borcherds residue. Clause~(III): the Kuga--Satake Shimura
target exists at signature $(2, n)$ with $n = h^{1, 1} - 1 \geq 2$,
but the identification of the paramodular image of
$B^{E_3}(\Phi_3^{\mathrm{FA}}(D^b\Coh(X)))$ requires an explicit
Borcherds lift of a Jacobi-form seed. For $K3 \times E$ this seed is
$\phi_{0, 1}$ (Lorgat 2020 arXiv:2004.09030); no analogous seed is
available on a non-K3-fibred strict-stratum CY$_3$ with $h^{1, 1}
\geq 3$, so the construction is open.
\end{proof}

\begin{remark}[Four $\kappa_\bullet$ discipline on the quintic]
\label{rem:quintic-four-kappa-spectrum}
The quintic chain-level data separate the canonical $\kappa_\bullet$
slots from the BV anomaly coefficient:
\[
 \bigl(\kappa_{\mathrm{cat}}(Q),\; \kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(Q),\;
 a_{\mathrm{BV}}(Q),\; \kappa_{\mathrm{BKM}}(Q)\bigr)
 \;=\; \bigl(0,\; 0,\; -25/3,\; \mathrm{undefined}\bigr),
\]
the last entry marking absence of a constructed Borcherds denominator
identity. This is the strict-stratum contrast with
the $K3 \times E$ spectrum $\{0, 3, 5, 24\}$. The two zero entries and
the undefined Borcherds entry reflect the absence of a K3-fibration
Heegner structure; the non-zero value $a_{\mathrm{BV}}(Q)
= -25/3$ is the Costello--Li genus-one tadpole, absorbed into the
BCOV shadow rather than into a Borcherds target.
\end{remark}

\begin{remark}[Bicubic, Godeaux, R{\o}dland, and the reach of the
rank-one collapse]
\label{rem:quintic-other-rank-1-cy3}
The $(3, 3)$-bicubic CY$_3 \subset \P^5$, the Godeaux CY$_3$
$Q_{\mathrm{Fermat}}/(\Z/5)$, and the R{\o}dland Pfaffian CY$_3$
(R{\o}dland 2000 \emph{Compositio Math.}~122; Hosono--Takagi 2011
\emph{J.\ Alg.\ Geom.}~20) all have $h^{1, 1} = 1$. The same
rank-one warning applies: Hodge and BCOV shadows are computable, but
rank one does not by itself produce a Borcherds lift, a Jacobi seed, or
an MO stable-envelope residue. The mirror quintic
$\widetilde Q = Q/(\Z/5)^3$ changes the Hodge numbers by twisted-sector
resolution; it still requires its own framed compact witness before any
CY-B$_3$ Borcherds claim can be made.
\end{remark}

\begin{remark}[Content-bearing CY-B$_3$ on non-K3-fibred CY$_3$]
\label{rem:quintic-cy-b3-open-case}
Clause~(III) of Theorem~\ref{thm:cy-b3-scope-quintic-chapter} marks
the remaining open case. The non-K3-fibred compact
CY$_3$ with $h^{1, 1} \geq 3$; the enlarged-Picard-rank bicubic
or $(3, 3)$-complete intersection, the Fano--Iliev--Manivel double
cover, specific Gushel--Mukai varieties of CY$_3$-type; have
Kuga--Satake signature supporting a genuine Siegel paramodular
target, but no Borcherds lift has been constructed. Resolving
whether CY-B$_3$ is content-bearing on this stratum requires
producing either an explicit Jacobi-form seed analogous to
$\phi_{0, 1}$ of Lorgat 2020 or a no-go showing that the
non-K3-fibred structure obstructs the Borcherds theta lift at
signature $(2, n)$. The quintic deployment does not resolve this;
it isolates the remaining proof obligation.
\end{remark}
