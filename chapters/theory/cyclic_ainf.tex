\chapter{Cyclic $\Ainf$-Structures}
\label{ch:cyclic-ainf}

\section{$\Ainf$-categories and the bar complex}
\label{sec:ainf-bar}

The cyclic bar complex is the primary invariant of a CY category.  We review the $\Ainf$-formalism and its cyclic refinement, preparing for the bridge to chiral algebras in Part~III.

\begin{definition}[$\Ainf$-category]
\label{def:ainf-category}
An $\Ainf$-category $\cC$ over $k$ consists of:
\begin{enumerate}[label=(\roman*)]
  \item A set of objects $\mathrm{Ob}(\cC)$;
  \item For each pair of objects $X, Y$, a graded $k$-module $\Hom_\cC(X,Y)$;
  \item Higher composition maps $\mu_n \colon \Hom(X_0, X_1) \otimes \cdots \otimes \Hom(X_{n-1}, X_n) \to \Hom(X_0, X_n)[2-n]$ for $n \geq 1$, satisfying the $\Ainf$-relations.
\end{enumerate}
\end{definition}

\section{Cyclic structures}
\label{sec:cyclic-structures}

\begin{definition}[Cyclic $\Ainf$-algebra]
\label{def:cyclic-ainf}
A \emph{cyclic $\Ainf$-algebra of dimension $d$} is an $\Ainf$-algebra $(A, \{\mu_n\}_{n \geq 1})$ equipped with a non-degenerate pairing $\langle \cdot, \cdot \rangle \colon A \otimes A \to k[-d]$ satisfying the cyclic invariance condition
\[
  \langle \mu_n(a_1, \ldots, a_n), a_0 \rangle = \pm \langle \mu_n(a_0, a_1, \ldots, a_{n-1}), a_n \rangle
\]
for all $n \geq 1$, where the sign is determined by the Koszul rule.
\end{definition}

\begin{remark}
\label{rem:cyclic-bar-chiral-bridge}
The cyclic invariance condition is the chain-level precursor of the chiral algebra trace.  The passage from cyclic $\Ainf$ to chiral algebra requires promoting the cyclic pairing to a factorization structure on a curve --- this is the content of Part~III.
\end{remark}

\section{The cyclic bar complex}
\label{sec:cyclic-bar-complex}

\begin{construction}[Cyclic bar complex]
\label{constr:cyclic-bar}
Given a cyclic $\Ainf$-algebra $(A, \langle \cdot, \cdot \rangle)$ of dimension $d$, the \emph{cyclic bar complex} is
\[
  \mathrm{CC}_\bullet(A) = \bigoplus_{n \geq 0} (A[1])^{\otimes n} / \text{(cyclic)}
\]
with differential induced by the $\Ainf$-operations and the cyclic quotient enforced by the pairing.  The $S^1$-action by cyclic rotation on $\mathrm{CC}_\bullet(A)$ endows the cyclic homology $\HC_\bullet(A)$ with its defining structure.
\end{construction}

\section{The $\mathbb{S}^d$-framing}
\label{sec:sd-framing}

A $d$-dimensional CY structure enhances the $S^1$-action to an $\mathbb{S}^d$-action on the Hochschild complex.  This is the key datum: for $d = 2$, the $\mathbb{S}^2$-framing provides the $E_2$-algebra structure that will underpin the braided monoidal/quantum group side of the story.
