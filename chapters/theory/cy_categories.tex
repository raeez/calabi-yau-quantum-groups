\chapter{Calabi--Yau Categories}
\label{ch:cy-categories}

\section{Smooth and proper dg categories}
\label{sec:smooth-proper-dg}

\begin{definition}[Smooth dg category]
\label{def:smooth-dg-cat}
A dg category $\cC$ over $k$ is \emph{smooth} if the diagonal bimodule $\cC \in \cC^{\op} \otimes \cC\text{-}\Mod$ is a perfect bimodule (compact in the derived category of bimodules).
\end{definition}

\begin{definition}[Proper dg category]
\label{def:proper-dg-cat}
A dg category $\cC$ is \emph{proper} if for all objects $X, Y \in \cC$, the Hom-complex $\Hom_\cC(X,Y)$ has finite-dimensional total cohomology.
\end{definition}

\section{The Calabi--Yau condition}
\label{sec:cy-condition}

\begin{definition}[Calabi--Yau category, after Kontsevich]
\label{def:cy-category}
A smooth dg category $\cC$ is \emph{Calabi--Yau of dimension $d$} if there is a quasi-isomorphism of $\cC$-bimodules
\[
  \cC \xrightarrow{\;\sim\;} \cC^\vee[d]
\]
where $\cC^\vee = \RHom_{\cC^{\op} \otimes \cC}(\cC, \cC^{\op} \otimes \cC)$ is the inverse dualizing bimodule.  Equivalently, there exists a non-degenerate cyclic pairing
\[
  \langle \cdot, \cdot \rangle \colon \HH_\bullet(\cC) \otimes \HH_\bullet(\cC) \longrightarrow k[-d].
\]
\end{definition}

\begin{remark}
\label{rem:cy-vs-smooth-cy}
The CY condition combines two ingredients: smoothness (the bimodule $\cC$ is compact) and the Serre functor being a shift ($S_\cC \simeq [d]$).  A smooth and proper CY category is \emph{saturated} in the sense of Kontsevich--Soibelman.
\end{remark}

\section{The CY trace}
\label{sec:cy-trace}

\begin{definition}[Non-degenerate trace]
\label{def:cy-trace}
A \emph{$d$-dimensional CY structure} on a smooth dg category $\cC$ is a class
\[
  [\sigma] \in \HC^-_{d}(\cC)
\]
in the negative cyclic homology, whose image under the canonical map $\HC^- \to \HH$ is a non-degenerate trace $\Tr \colon \HH_d(\cC) \to k$.
\end{definition}

\section{Standard examples}
\label{sec:cy-standard-examples}

\begin{example}[Fukaya category]
\label{ex:fukaya-cy}
For a compact symplectic manifold $(M, \omega)$ with $c_1(M) = 0$, the Fukaya category $\Fuk(M)$ is CY of dimension $\dim_\mathbb{R}(M)/2 = n$.  The CY trace arises from the cyclic structure on the $\Ainf$-algebra of Lagrangian Floer cochains.
\end{example}

\begin{example}[Derived category of a CY manifold]
\label{ex:derived-cy}
For a smooth projective CY manifold $X$ of dimension $n$ (i.e., $\omega_X \simeq \cO_X$, $H^i(X, \cO_X) = 0$ for $0 < i < n$), the bounded derived category $D^b(\Coh(X))$ is CY of dimension $n$.  The CY trace is induced by the Serre duality pairing.
\end{example}

\begin{example}[Matrix factorizations]
\label{ex:mf-cy}
For $W \colon \mathbb{A}^n \to \mathbb{A}^1$ a polynomial with isolated singularity, the category $\MF(W)$ of matrix factorizations is CY of dimension $n - 2$ (Dyckerhoff; Orlov).  For the ADE singularities in two variables ($n = 2$), $\MF(W)$ is $\CY_0$ (semisimple, with the structure of a Frobenius algebra).  For threefold singularities ($n = 4$, e.g.\ $W = x_1^2 + \cdots + x_4^2$), $\MF(W)$ is $\CY_2$.
