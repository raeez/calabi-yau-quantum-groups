\section{Normal form}
\label{sec:cy3-chain-level-bridge-normal-form}

The CY$_3$ chain-level bridge resolves the passage from holomorphic
Chern--Simons to the categorical chiral side of $\PhiFA_3$ into four
typed objects: the classical hCS BV complex
\textup{(}Definition~\ref{def:cy3-hcs-bv-complex}\textup{);} its
Costello--Gwilliam quantisation
\textup{(}Definition~\ref{def:cy3-hcs-quantum-observables}\textup{);}
the Dolbeault--chiral Chevalley--Eilenberg envelope
\textup{(}Definition~\ref{def:cy3-many-variable-chiral-ce}\textup{);}
and the framed Hochschild target on a DWR-good cover
\textup{(}Definition~\ref{def:framed-hcs-hochschild-target}\textup{).}
Proposition~\ref{prop:cy3-continuous-e3-bar-ce} identifies the third with
the strong continuous dual of the Stage-$1$ $E_3$ bar of the
factorisation envelope; Theorem~\ref{thm:cy3-quantum-bv-bar-conditional}
extends this to a quantum BV/bar comparison under enumerated hypotheses;
Theorem~\ref{thm:framed-hcs-hochschild-descent-criterion} gives the
framed descent comparison to the Hochschild target via the seven-class
obstruction tuple $\mathfrak o_{\hCS,\HH}$. The Hall comparison
$\Theta_{\hCS\to\Hall}$ that closes onto $\CoHA_{\mathrm{crit}}(X)$ is a
separate primitive,
Problem~\ref{op:cy3-hcs-hall-comparison}.

\begin{definition}[The hCS BV complex on a Calabi--Yau threefold]
\label{def:cy3-hcs-bv-complex}
Let $X$ be a smooth Calabi--Yau threefold with holomorphic volume form
$\Omega_X$, and let $\mathfrak g$ be a finite-dimensional metric Lie
algebra with invariant pairing $\langle-,-\rangle$. The classical
holomorphic Chern--Simons BV field complex is
\[
 \cE_{\hCS}(X,\mathfrak g)
 :=
 \Omega^{0,\bullet}(X,\mathfrak g)[1],
 \qquad
 Q_{\hCS}=\dbar .
\]
On an open chart $U$ the local BV theory uses
$\cE_{\hCS,c}(U,\mathfrak g)=\Omega_c^{0,\bullet}(U,\mathfrak g)[1]$
paired with $\cE_{\hCS}(U,\mathfrak g)$ by Serre duality; when $X$ is
compact the support subscript may be suppressed.
The BV pairing is the degree-$(-1)$ local pairing
\[
 (\alpha,\beta)_{\hCS}
 =
 \int_X \Omega_X\wedge \langle \alpha,\beta\rangle ,
 \qquad
 \alpha,\beta\in\Omega^{0,\bullet}_c(X,\mathfrak g)[1],
\]
and the induced odd Poisson bracket on local functionals is the BV
bracket used below.
Its local classical functional is
\[
 I_{\hCS}(\alpha)
 =
 \int_X \Omega_X \wedge
 \left(
 \frac{1}{2}\langle \alpha,\dbar\alpha\rangle
 +
 \frac{1}{6}\langle \alpha,[\alpha,\alpha]\rangle
 \right),
\]
where the displayed superfield expression means: multiply in
$\Omega^{0,\bullet}(X)$, take the $\Omega^{0,3}$ component, and integrate
the resulting $(3,3)$-form. With
$[\alpha,\beta]=\alpha\wedge\beta\otimes[\,,\,]_{\mathfrak g}$, its
variation is
\[
 \delta I_{\hCS}(\alpha)
 =
 \int_X\Omega_X\wedge
 \left\langle
 \delta\alpha,\,
 \dbar\alpha+\frac{1}{2}[\alpha,\alpha]
 \right\rangle .
\]
The classical master equation
$\{I_{\hCS},I_{\hCS}\}_{\BV}=0$ follows from $\dbar^2=0$, invariance of
$\langle-,-\rangle$, and Jacobi. The factorisation algebra of quantum
observables, when the local anomaly is cancelled, is denoted
\[
 \Obs_{\hCS}^{q}(X,\mathfrak g).
\]
\end{definition}

\begin{definition}[Quantum hCS observables]
\label{def:cy3-hcs-quantum-observables}
On an admissible open $U\subset X$ with compact-support convention fixed
above, the quantum observable factorisation algebra is the
Costello--Gwilliam renormalised BV complex
\[
 \Obs_{\hCS}^{q}(U,\mathfrak g)
 =
 \left(
 \mathcal O_{\mathrm{ren,loc/multiloc}}
 (\cE_{\hCS,c}(U,\mathfrak g))[[\hbar]],
 Q_{\hCS}+\{I[L],-\}_{\BV}+\hbar\Delta_L
 \right).
\]
Here $\mathcal O_{\mathrm{ren,loc/multiloc}}$ denotes the
Costello-renormalised space of local and multilocal functionals on which
the heat-kernel BV operator is defined; it is not the full algebra of
all continuous polynomial functions on the compact-support LF space.
The effective interaction $I[L]$ is the scale-$L$ interaction,
$\Delta_L$ is the renormalised BV Laplacian determined by the
heat-kernel propagator, and the quantum master equation
\[
 Q_{\hCS} I[L]
 +\frac{1}{2}\{I[L],I[L]\}_{\BV}
 +\hbar\Delta_L I[L]
 =0
\]
is part of the anomaly-cancellation hypothesis. Factorisation products
are induced by disjoint support. This is the source object in
Problem~\ref{op:cy3-hcs-hall-comparison}.
\end{definition}

\begin{definition}[The many-variable chiral CE model]
\label{def:cy3-many-variable-chiral-ce}
Let $P=D_1\times D_2\times D_3\subset X$ be a holomorphic polydisc with
coordinates $z_1,z_2,z_3$. Let $\mathfrak l_{\cC}$ be the local dg
Lie algebra controlling the chosen CY$_3$ chart of $\cC$; in a pure hCS
gauge chart this is the constant dg Lie algebra $\mathfrak g$ with
zero internal differential. Topologise
$J^\infty_{\mathrm{hol}}\mathfrak l_{\cC}$ as the completed inverse
limit of finite holomorphic-jet bundles with its nuclear pro-Fr\'echet
topology, and put
\[
 \mathfrak L_{\cC}(P)
 :=
 \Omega^{0,\bullet}_c
 \bigl(P,J^\infty_{\mathrm{hol},z_1,z_2,z_3}\mathfrak l_{\cC}\bigr)[1],
 \qquad
 d_{\mathfrak L}=\dbar+d_{\mathfrak l},
 \qquad
 [\alpha,\beta]_{\mathfrak L}
 =
 \alpha\wedge\beta\otimes[\,,\,]_{\mathfrak l_{\cC}}.
\]
The space $\mathfrak L_{\cC}(P)$ carries the strict nuclear LF topology
obtained as the filtered colimit over compact subsets $K\Subset P$.
All tensor products below are completed projective tensor products
$\widehat\otimes_\pi$, and $(-)^\vee_b$ denotes the strong continuous
dual. The continuity of $\dbar$, $d_{\mathfrak l}$, and the bracket is
part of the local dg-Lie datum.
The local classical observable object retained by the CY$_3$
bridge is the Dolbeault--chiral Chevalley--Eilenberg factorisation
algebra
\[
 \Obs_{\cC}^{\mathrm{cl}}(P)
 =
 C^\bullet_{\mathrm{Lie,cont}}\bigl(\mathfrak L_{\cC}(P),\mathbb C\bigr)
 =
 \prod_{n\geq0}
 \mathrm{Hom}_{\mathrm{cont}}
 \!\left(
 \widehat{\mathrm{Sym}}^n_\pi(\mathfrak L_{\cC}(P)[1]),
 \mathbb C
 \right)
 =
 \widehat{\mathrm{Sym}}
 \bigl(\mathfrak L_{\cC}(P)^\vee_b[-1]\bigr),
\]
with differential $d_{\mathrm{CE}}+\dbar^\vee+d_{\mathfrak l}^\vee$.
The chain object behind it is the stage-one local model
\[
 \PhiFA_3(\cC)|_P
 \simeq
 U^{\mathrm{fact},E_3}_{P}
 \bigl(\mathfrak L_{\cC}\bigr),
 \qquad
 B_{E_3}\bigl(\PhiFA_3(\cC)|_P\bigr)
 \simeq
 \mathrm{CE}^{\mathrm{ch},E_3}_*
 \bigl(\mathfrak L_{\cC}\bigr),
\]
on the loci where the hCS realisation of $\PhiFA_3$ is available. Thus
observables are the continuous dual of the bar-chain coalgebra, not the
bar-chain coalgebra itself. This is the retained Stage-$1$ $\Ethree$
bar/CE object; after Stage-$2$ specialisation to a curve and forgetting
the Stage-$1$ datum, the output is only an $\Eone$-chiral algebra. The
factorisation products are products over disjoint holomorphic polydiscs;
their singularities lie only on the union of partial diagonals
$\Delta_\pi\subset P^n$. A collision operation is the local-cohomology
Grothendieck residue along a refinement of partitions; binary OPE is the
special case of one pairwise diagonal, and associativity is equality of
iterated residues for nested refinements with the Fulton--MacPherson
boundary orientation. In a meromorphically trivialised local-field chart
the binary diagonal singularity may be written as
\[
 a(z)b(w)
 \sim
 \sum_{\alpha\in\mathbb N^3}
 \frac{(a_{(\alpha)}b)(w)}
 {(z_1-w_1)^{\alpha_1+1}
 (z_2-w_2)^{\alpha_2+1}
 (z_3-w_3)^{\alpha_3+1}},
\]
where the coefficient is the three-normal-direction residue
\[
 a_{(\alpha)}b(w)
 :=
 \frac{1}{(2\pi i)^3}
 \int_{|z_i-w_i|=\varepsilon_i}
 (z-w)^\alpha\,a(z)b(w)\,
 dz_1\wedge dz_2\wedge dz_3 .
\]
The product torus is oriented in the order $(z_1,z_2,z_3)$; the
Calabi--Yau volume form $\Omega_X$ absorbs the Jacobian under holomorphic
coordinate change; and the residue is independent of small radii as long
as the torus remains in the complement of the other polar strata. The
factorisation algebra is defined by the preceding partial-diagonal
local-cohomology residue condition; the displayed Laurent expansion is a
coordinate normal form, not an additional structure. The ordinary
Chevalley--Eilenberg algebra
$C^\bullet(\mathfrak g)$ is obtained only after applying the locally
constant shadow
\[
 \Omega^{0,\bullet}(P)\simeq \mathbb C,
 \qquad
 J^\infty_{\mathrm{hol}}\mathfrak l_{\cC}\rightsquigarrow
 H^\bullet(\mathfrak l_{\cC,x}),
 \qquad
 \Obs_{\cC}^{\mathrm{cl}}(P)\rightsquigarrow
 C^\bullet\bigl(H^\bullet(\mathfrak l_{\cC,x})\bigr).
\]
In the constant hCS chart this last object is $C^\bullet(\mathfrak g)$.
The CY$_3$ chiral object retained by the bridge is
$\Obs_{\cC}^{\mathrm{cl}}(P)$ itself; its locally constant shadow under
the substitutions above is $C^\bullet(\mathfrak g)$.
\end{definition}

\begin{proposition}[Continuous dual \texorpdfstring{$E_3$}{E3} bar of the many-variable model]
\label{prop:cy3-continuous-e3-bar-ce}
Let $P=D_1\times D_2\times D_3$ be a holomorphic polydisc and assume
that the local dg Lie algebra $\mathfrak l_{\cC}$ is finite type in each
cohomological degree. Equip
$\Omega^{0,\bullet}_c(P,J^\infty_{\mathrm{hol}}\mathfrak l_{\cC})$ with
its standard LF topology and its continuous dual with the corresponding
Fr\'echet topology. Then the factorisation enveloping algebra in
Definition~\ref{def:cy3-many-variable-chiral-ce} satisfies a natural
quasi-isomorphism of conilpotent $E_3$-coalgebras
\[
 B_{E_3}\!\left(
 U^{\mathrm{fact},E_3}_{P}(\mathfrak L_{\cC})
 \right)
 \simeq
 \mathrm{CE}^{\mathrm{ch},E_3}_*(\mathfrak L_{\cC}),
\]
where the Chevalley--Eilenberg complex is formed with completed
projective tensor products and continuous alternating coinvariants.
Consequently the classical observable factorisation algebra is the
strong continuous dual
\[
 \Obs_{\cC}^{\mathrm{cl}}(P)
 \simeq
 \left(
 B_{E_3}\!\left(
 U^{\mathrm{fact},E_3}_{P}(\mathfrak L_{\cC})
 \right)
 \right)^\vee_b .
\]
Both identifications are compatible with restriction to smaller
polydiscs, extension by zero for disjoint compact supports, and hence
with left-end Dolbeault/Weiss/Ran descent. The Hall comparison
$\Theta_{\hCS\to\Hall}$ to $\CoHA_{\mathrm{crit}}(X)$ is the separate
primitive of
Problem~\ref{op:cy3-hcs-hall-comparison}.
\end{proposition}

\begin{proof}
The LF space
$\Omega^{0,\bullet}_c(P,J^\infty_{\mathrm{hol}}\mathfrak l_{\cC})$ is
strictly nuclear because compactly supported Dolbeault forms on a
polydisc are a filtered colimit of nuclear Fr\'echet spaces and
$J^\infty_{\mathrm{hol}}\mathfrak l_{\cC}$ is finite type in each
cohomological degree. On nuclear spaces the completed projective tensor
product is exact and commutes with finite coinvariants; hence the
continuous Chevalley--Eilenberg coalgebra is computed by
\[
 \widehat{\Sym}^{\,c}
 \bigl(\mathfrak L_{\cC}[1]\bigr),
 \qquad
 d=d_{\dbar}+d_{\mathfrak l}+d_{\mathrm{Lie}},
\]
with the Lie differential determined by the bracket in
Definition~\ref{def:cy3-many-variable-chiral-ce}. The
$E_3$-factorisation envelope
$U^{\mathrm{fact},E_3}_{P}(\mathfrak L_{\cC})$ is the left adjoint from
local dg Lie algebras to locally constant-in-the-normal-directions
$E_3$-factorisation algebras on the polydisc. The $E_3$ bar functor is
the corresponding right Koszul dual coalgebra functor. Applying the
bar-envelope unit to the free/enveloping object gives exactly the
continuous Chevalley--Eilenberg chains above. Naturality follows
because extension by zero for compactly supported forms and restriction
of holomorphic jets are continuous maps of nuclear complexes, and Weiss
descent is checked on the same completed tensor products. The
many-variable residues commute with this identification since both sides
compute collision operations by local cohomology along the same partial
diagonals. Taking the strong continuous dual is exact on the nuclear
subclass fixed in Definition~\ref{def:cy3-many-variable-chiral-ce}; it
converts completed direct sums of compact-support fields into products
of continuous functionals and sends the CE-chain differential to the
continuous CE-cochain differential. This gives the displayed observable
identification and preserves factorisation products for disjoint
polydiscs.
\end{proof}

\begin{theorem}[Quantum BV/bar comparison on a CY$_3$ polydisc]
\label{thm:cy3-quantum-bv-bar-conditional}
Let $P\subset X$ be a holomorphic polydisc in a smooth CY$_3$ with
holomorphic volume form $\Omega_X$, and let
$\mathfrak L_{\cC}(P)$ be the compact-support Dolbeault--jet dg Lie
algebra of Definition~\ref{def:cy3-many-variable-chiral-ce}. Assume:
\begin{enumerate}[label=\textup{(\roman*)}]
\item the Stage-$1$ $E_3$ formality point and Costello--Li holomorphic
witness used in Proposition~\ref{prop:cy3-continuous-e3-bar-ce};
\item a Costello--Gwilliam/Costello--Li renormalisation datum
$(I[L],\Delta_L,W_{\Gamma,L})$ on $P$ satisfying RG flow and the quantum
master equation;
\item vanishing, or chosen counterterm cancellation, of the local
Costello--Li quartic anomaly class
$[P_4]\in H^1_{\mathrm{loc}}
(\mathfrak L_{\cC},\mathcal O_{\mathrm{loc}})$;
\item a strict nuclear LF/DFS completion in which every renormalised
Feynman operation $W_{\Gamma,L}C_\Gamma$ is continuous and compatible
with extension by zero for disjoint polydiscs;
\item a continuous BV-to-bar transfer
\[
 T_L\colon
 B_{E_3}U^{\mathrm{fact},E_3}_P(\mathfrak L_{\cC})[[\hbar]]
 \longrightarrow
 \bigl(\Obs_{\hCS}^{q}(P,\mathfrak g)\bigr)^\vee_b
\]
whose classical limit is the strong-duality comparison of
Proposition~\ref{prop:cy3-continuous-e3-bar-ce};
\item continuous coderivations $D_r$ on the completed $E_3$ bar
coalgebra such that
\[
 D_B^\hbar=d_B+\sum_{r\geq 1}\hbar^rD_r,
 \qquad
 (D_B^\hbar)^2=0,
\]
and such that conjugation by $T_L^\vee$ identifies $D_B^\hbar$ with the
renormalised BV differential
$Q_{\hCS}+\{I[L],-\}_{\BV}+\hbar\Delta_L$.
\end{enumerate}
Then there is a natural quasi-isomorphism of $\hbar$-adically complete
factorisation algebras
\[
 \Obs_{\hCS}^{q}(P,\mathfrak g)
 \simeq
 \left(
 B_{E_3}^{\hbar}
 U^{\mathrm{fact},E_3}_P(\mathfrak L_{\cC})
 \right)^\vee_b ,
 \qquad
 B_{E_3}^{\hbar}:=
 \left(B_{E_3},D_B^\hbar\right).
\]
Its associated graded at $\hbar=0$ is
Proposition~\ref{prop:cy3-continuous-e3-bar-ce}. The Hall map
$\Theta_{\hCS\to\Hall}^{\mathrm{or}}$ to $\CoHA_{\mathrm{crit}}(X)$ is
the separate primitive of
Problem~\ref{op:cy3-hcs-hall-comparison}; the present theorem closes the
quantum BV/bar comparison on the source side, which is its input.
\end{theorem}

\begin{proof}
The assumptions place the Costello--Li BV complex and the completed
$E_3$ bar coalgebra in the same strict completed category. The quantum
master equation is equivalent to the square-zero identity for the
renormalised BV differential. The transfer $T_L$ transports this
differential to the completed bar coalgebra, and assumption~(vi) states
exactly that the transported differential is the coderivation
$D_B^\hbar$. Strong continuous duality is exact on the strict
nuclear/DFS subcategory fixed in assumption~(iv), so dualising gives
the displayed quasi-isomorphism. Factorisation compatibility follows
from the extension-by-zero compatibility of the renormalised Feynman
operations. Setting $\hbar=0$ removes the loop coderivations, the
quantum BV Laplacian, and the higher effective interactions, recovering
the classical continuous-duality theorem.
\end{proof}

\begin{definition}[Framed Hochschild target on the DWR/Ran nerve]
\label{def:framed-hcs-hochschild-target}
Let $X$ be a smooth CY$_3$ with holomorphic volume form $\Omega_X$ and
let $\mathfrak U$ be a DWR-good cover. Fix
$N\in \mathbb N\cup\{\infty\}$ and write
$\mathfrak g_N=\mathfrak{gl}_N$ for finite $N$. On a polydisc
$P\subset X$ let $\mathfrak l_{\Perf(X)}(P)$ denote the local
Hochschild dg Lie algebra of $\Perf(X)$ in the HKR model, with
Gerstenhaber bracket of degree $-2$ and CY$_3$ trace induced by
$\Omega_X$. Its rank-$N$ framed form is
\[
 \mathfrak L^{\HH}_{X,N}(P)
 :=
 \Omega^{0,\bullet}_c
 \bigl(P,J^\infty_{\mathrm{hol}}\mathfrak l_{\Perf(X)}\bigr)[1]
 \widehat\otimes \mathfrak g_N ,
\]
with completed projective tensor product and block-sum transition
$\mathfrak g_N\hookrightarrow\mathfrak g_{N+1}$. For a DWR/Ran simplex
$\sigma=(S,\{P_s\}_{s\in S})$ set
\[
 \PhiFA^{\HH,\mathrm{fr}}_{3,N}(\Perf(X))(\sigma)
 :=
 \widehat{\bigotimes}_{s\in S}
 \left(
 B^\hbar_{E_3}
 U^{\mathrm{fact},E_3}_{P_s}
 (\mathfrak L^{\HH}_{X,N})
 \right)^\vee_b .
\]
The stable framed target is the homotopy colimit
\[
 \PhiFA^{\HH,\mathrm{fr}}_{3,\infty}(\Perf(X))
 :=
 \operatorname*{hocolim}_{N}
 \PhiFA^{\HH,\mathrm{fr}}_{3,N}(\Perf(X)),
\]
formed only when the formality point, counterterm scheme, and
completion are compatible with block sum. Thus a comparison with
$\Obs^q_{\hCS}(X,\mathfrak g)$ is typed only after a framing
$\mathfrak g\to\mathfrak g_N$ or a stable $\mathfrak{gl}_\infty$
presentation is fixed. The direct target of a finite-rank hCS
comparison is the framed object
$\PhiFA^{\HH,\mathrm{fr}}_{3,N}(\Perf(X))$; the unframed symbol
$\PhiFA_3(\Perf(X))$ is the image after forgetting the rank/framing
coordinate.
\end{definition}

\begin{definition}[Framed hCS--Hochschild obstruction complex]
\label{def:hcs-hochschild-obstruction-complex}
Fix $X$, $\mathfrak U$, $N$, an anomaly-cancelled hCS quantisation
$\Obs_{\hCS}^{q}(-,\mathfrak g_N)$ in the sense of
Definition~\ref{def:cy3-hcs-quantum-observables}, and the framed
Hochschild target of Definition~\ref{def:framed-hcs-hochschild-target}.
For every finite intersection $U_I$ in $\mathfrak U$ put
\[
 \mathcal M^q_{\hCS,\HH}(U_I)
 :=
 \Hom^q_{\mathrm{cont},E_3}
 \left(
 \Obs_{\hCS}^{q}(U_I,\mathfrak g_N),
 \PhiFA^{\HH,\mathrm{fr}}_{3,N}(\Perf(X))(U_I)
 \right),
\]
where the subscript $E_3$ means continuous maps equipped with homotopies
for the little-$3$-disks operations on every polydisc refinement. The
\v{C}ech differential, the two internal differentials, the BV bracket on
hCS observables, and the commutator bracket of the Hochschild
factorisation envelope make
\[
 \mathfrak M_{\hCS,\HH}(\mathfrak U)
 :=
 \Tot\,\Cech^\bullet(\mathfrak U,\mathcal M^\bullet_{\hCS,\HH})
\]
a complete filtered dg Lie algebra. A chartwise family
$\theta^{(0)}=\{\theta_i\}$ has obstruction tuple
\[
 \mathfrak o_{\hCS,\HH}(\theta^{(0)})
 =
 \bigl(
 o_{\mathrm{MC}},\,
 o_{\mathrm{fr}},\,
 o_{\mathrm{Fed}},\,
 o_{\mathrm{zero}},\,
 o_{\mathrm{Weiss}},\,
 o_{\mathrm{comp}},\,
 o_{E_3}
 \bigr).
\]
Here $o_{\mathrm{MC}}$ is the Maurer--Cartan curvature in
$H^1(\mathfrak M_{\hCS,\HH})$; $o_{\mathrm{fr}}$ is the failure of the
gauge framing to agree with the rank-$N$ or stable
$\mathfrak{gl}_\infty$ Hochschild framing; $o_{\mathrm{Fed}}$ is the
Fedosov/HKR descent defect for the Atiyah-class transition functions;
$o_{\mathrm{zero}}$ is the defect of the harmonic zero-mode splitting
against the Hochschild trace; $o_{\mathrm{Weiss}}$ is the failure of
the prefactorisation maps to satisfy Weiss descent on the DWR/Ran
nerve; $o_{\mathrm{comp}}$ is the mismatch between the Costello
renormalised multilocal topology and the strong-dual Hochschild
completion; and $o_{E_3}$ is the operadic defect measuring failure to
preserve the $E_3$ collision operations, not merely cohomology.
\end{definition}

\begin{definition}[Primitive data for the hCS--Hochschild arrow]
\label{def:hcs-hochschild-primitive-package}
A primitive datum for a framed hCS--Hochschild comparison on
$(X,\mathfrak U,N)$ is the typed datum
\[
 \mathfrak P_{\hCS,\HH}
 =
 \bigl(
 \theta^{(0)},\,
 \eta_{\mathrm{MC}},\,
 \rho_{\mathrm{fr}},\,
 H_{\mathrm{Fed}},\,
 H_{\mathrm{zero}},\,
 H_{\mathrm{Weiss}},\,
 Q_{\mathrm{comp}},\,
 H_{E_3},\,
 Q
 \bigr)
\]
with the following entries:
\begin{enumerate}
\item chartwise continuous degree-zero maps
$\theta_i^{(0)}\colon
\Obs_{\hCS}^{q}(U_i,\mathfrak g_N)\to
\PhiFA^{\HH,\mathrm{fr}}_{3,N}(\Perf(X))(U_i)$;
\item a positive \v{C}ech/Ran filtration correction
$\eta_{\mathrm{MC}}\in F^1\mathfrak M_{\hCS,\HH}(\mathfrak U)^0$
such that
\[
 \theta=\theta^{(0)}+\eta_{\mathrm{MC}},
 \qquad
 d\theta+\frac12[\theta,\theta]=0;
\]
\item a rank/framing transport $\rho_{\mathrm{fr}}$ identifying the
hCS gauge framing with the chosen finite-rank or stable
Hochschild framing;
\item a Fedosov homotopy $H_{\mathrm{Fed}}$ trivialising the
HKR transition defect against the Atiyah-class cocycle;
\item a zero-mode contraction $H_{\mathrm{zero}}$ for the harmonic
summand of $\Omega^{0,\bullet}(X,\mathfrak g_N)[1]$, compatible with
the finite-dimensional residual BV integral;
\item a Weiss descent homotopy $H_{\mathrm{Weiss}}$ on all higher
DWR/Ran simplices;
\item a completion homotopy datum $Q_{\mathrm{comp}}$ identifying the
$\hbar$-adic Costello-renormalised topology with the strong-dual
Hochschild completion on the image of $\theta$;
\item operadic homotopies $H_{E_3}$ comparing all binary and higher
collision operations after the chosen Kontsevich--Tamarkin formality
point;
\item a quasi-isomorphism datum $Q$ giving continuous inverses up
to homotopy on vertices and on all finite DWR/Ran refinements.
\end{enumerate}
Each entry is finite in type; on an infinite refinement system it is a
compatible pro-cochain, not a finite list of opens.
\end{definition}

\begin{theorem}[Framed hCS--Hochschild descent criterion]
\label{thm:framed-hcs-hochschild-descent-criterion}
For fixed $X$, $\mathfrak U$, $N$, anomaly-cancelled hCS source, and
framed Hochschild target, a chartwise family
\[
 \theta_i^{(0)}\colon
 \Obs_{\hCS}^{q}(U_i,\mathfrak g_N)
 \longrightarrow
 \PhiFA^{\HH,\mathrm{fr}}_{3,N}(\Perf(X))(U_i)
\]
extends to a quasi-isomorphism of completed $E_3$ holomorphic
factorisation algebras
\[
 \Theta_{\hCS\to\HH}^{\mathrm{fr}}\colon
 \Obs_{\hCS}^{q}(-,\mathfrak g_N)
 \xrightarrow{\;\simeq\;}
 \PhiFA^{\HH,\mathrm{fr}}_{3,N}(\Perf(X))
\]
on the DWR/Ran nerve if and only if
\[
 \mathfrak o_{\hCS,\HH}(\theta^{(0)})=0
\]
and the resulting degree-zero Maurer--Cartan element is vertexwise
invertible in $H^0$. Equivalently, the comparison exists if and only if
a primitive datum $\mathfrak P_{\hCS,\HH}$ of
Definition~\ref{def:hcs-hochschild-primitive-package} exists. In the
stable case $N=\infty$, the statement is read after passing to the
block-sum homotopy colimit over $N$.
\end{theorem}

\begin{proof}
Forgetting the framing, Fedosov, zero-mode, completion, and operadic
structures, a comparison is a continuous natural transformation of
complexes on the \v{C}ech/Ran nerve. Descent for such transformations is
controlled by the total convolution dg Lie algebra
$\mathfrak M_{\hCS,\HH}(\mathfrak U)$; extending the vertex maps is
equivalent to solving
$d\theta+\frac12[\theta,\theta]=0$. This is the class
$o_{\mathrm{MC}}$ and the gauge action on its solutions.

Restoring the forgotten structures gives the remaining six classes.
The rank coordinate is discrete and is killed exactly by
$\rho_{\mathrm{fr}}$. The HKR local maps are glued through a formal
geometry connection; the obstruction to changing charts is the
Fedosov/Atiyah \v{C}ech class, killed exactly by $H_{\mathrm{Fed}}$. Compact
CY$_3$ zero modes are harmonic summands of the Dolbeault BV complex;
compatibility with the categorical trace is precisely the class
$o_{\mathrm{zero}}$, killed by $H_{\mathrm{zero}}$. Weiss descent on
disjoint polydiscs is the class $o_{\mathrm{Weiss}}$. Continuity in the
renormalised and strong-dual completions is the class
$o_{\mathrm{comp}}$. Preservation of all $E_3$ collision operations is
$o_{E_3}$. If these classes vanish, the corrected cochain
$\theta=\theta^{(0)}+\eta_{\mathrm{MC}}$ is a morphism of completed
$E_3$ holomorphic factorisation algebras. Conversely, any such morphism
restricts to vertex maps and supplies the listed transports and
homotopies; its defining naturality, continuity, and operadic
compatibility force every component of
$\mathfrak o_{\hCS,\HH}$ to vanish. The quasi-isomorphism datum
$Q$ is exactly vertexwise and refinementwise invertibility in $H^0$.
\end{proof}

\begin{corollary}[\texorpdfstring{$K3\times E$}{K3xE} Stage-\texorpdfstring{$1$}{1} comparison gate]
\label{cor:k3e-stage1-hcs-hochschild-gate}
Let $X=K3\times E$ with $\Omega_X=\Omega_{K3}\wedge dz_E$. The
comparison
\[
 \Obs_{\hCS}^{q}(X,\mathfrak g)
 \simeq
 \PhiFA_3(\Perf(X))
\]
as $E_3$ holomorphic factorisation algebras is a theorem on the
rank-stable framed branch precisely under the following typed
replacement:
\[
 \Obs_{\hCS}^{q}(-,\mathfrak g_N)
 \xrightarrow{\;\Theta_{\hCS\to\HH}^{\mathrm{fr}}\;}
 \PhiFA^{\HH,\mathrm{fr}}_{3,N}(\Perf(K3\times E)),
\]
with $N<\infty$ or $N=\infty$, and with
$\mathfrak P_{\hCS,\HH}$ supplied on a DWR-good cover. The harmonic
zero-mode part is the K\"unneth summand
\[
 H^{0,\bullet}(K3\times E,\mathfrak g_N)[1]
 =
 (H^{0,\bullet}(K3)\otimes H^{0,\bullet}(E)\otimes\mathfrak g_N)[1],
\]
so $H^{0,1}$ and $H^{0,2}$ must be retained by
$H_{\mathrm{zero}}$ or integrated by a specified residual BV measure.
An anomaly-free exceptional gauge algebra, an oriented hCS--Hall datum,
or the Gritsenko--Nikulin denominator formula supplies at most a proper
subpart of this gate; none supplies the rank-stable
hCS--Hochschild primitive datum.
\end{corollary}

\begin{proof}
The K\"unneth formula gives
$H^{0,\bullet}(K3\times E)=H^{0,\bullet}(K3)\otimes H^{0,\bullet}(E)$,
with nonzero classes in bidegrees $0,1,2,3$. These harmonic classes are
global BV zero modes and therefore are invisible to a purely local
polydisc comparison unless a residual finite-dimensional BV treatment
is fixed. After this zero-mode datum, the CY volume form, the product
Ricci-flat metric, and the chosen formality point place $X$ in the
scope of Definition~\ref{def:framed-hcs-hochschild-target}. The
existence criterion is then exactly
Theorem~\ref{thm:framed-hcs-hochschild-descent-criterion}. The final
sentence separates independent data: anomaly cancellation constructs
the source differential, the hCS--Hall datum constructs a map to a Hall
target, and the Borcherds denominator fixes a Stage-$2$ automorphic
trace. None is a Fedosov/HKR, zero-mode, completion, and $E_3$ descent
package between hCS and categorical Hochschild.
\end{proof}

\begin{construction}[The typed CY$_3$ bridge]
\label{con:cy3-typed-bridge}
For a CY$_3$ input $\cC$ the bridge has four different kinds of objects:

\begin{tikzcd}[column sep=large]
 \PhiFA_3(\cC)
 \arrow[r, dashed, "{\Theta_{\hCS\to\Hall}}"]
 & \CoHA_{\mathrm{crit}}(X)
 \arrow[r, "{\mathrm{SV/KS\ local\ model}}"]

 \arrow[r, "{\mathrm{Drinfeld\ double}}"]
 & \mathcal{D}(Y^+)
 \arrow[r, "{\mathrm{Fock/evaluation}}"]

\end{tikzcd}
\]
The first object is a holomorphic factorisation algebra. The second is

\begin{definition}[Hall-valued factorisation-cosheaf target]
\label{def:hall-valued-factorisation-cosheaf-target}
Let $X$ be a CY$_3$ equipped with a Dolbeault/Weiss/Ran site
$\mathsf{DW\!R}(X)$ and a Weiss-cofinal holomorphic-polydisc basis
$\mathcal B_{\mathrm{poly}}$ whose finite intersections are refined by
objects of the same basis. The DWR/Ran nerve attached to a chosen cover
$\mathfrak U=\{U_i\}_{i\in I}$ is the simplicial category
$\mathsf{N}_{\mathsf{DW\!R}}(\mathfrak U)$ whose $p$-simplices are
finite pairwise disjoint families
\[
 \sigma=(S,\{P_s\Subset U_{i_{s,0}}\cap\cdots\cap U_{i_{s,p}}\}_{s\in S})
\]
of holomorphic polydiscs, with face maps omitting a \v{C}ech index,
degeneracies repeating one, and Ran multiplication given by disjoint
union of finite sets. Refinement morphisms are inclusions
$P'_t\Subset P_s$ over maps of finite sets $t\mapsto s$. The associated
open is $|\sigma|=\bigsqcup_{s\in S}P_s$. A DWR-good cover is one for
which these intersections and refinements remain in
$\mathcal B_{\mathrm{poly}}$, compact-support extension is continuous,
and Weiss descent computes the same object after passing to the
homotopy colimit over refinements.

An object of
\[
 \mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}(X)
\]
is the following data:
\begin{enumerate}
\item for each admissible open $U$, a completed chain complex
$\mathcal H(U)$ whose local critical-chart presentation is a
vanishing-cycle Borel--Moore complex;
\item for disjoint opens, continuous factorisation products
$\mathcal H(U_1)\widehat\otimes\mathcal H(U_2)\to
\mathcal H(U_1\sqcup U_2)$ whose one-open restriction is the Hall
extension product;
\item for each critical chart, a Kontsevich--Soibelman/Joyce
orientation datum: a square root of the virtual determinant line,
together with its orientation local system and coherences under direct
sums, extensions, and restriction to overlaps;
\item a specified completion: charge/HN-adic and equivariant-localised
on the Hall side, $\hbar$-adic and continuous-dual on the hCS side;
\item coherent Thom--Sebastiani isomorphisms for iterated extension
correspondences, including the signs, Tate twists, and orientation
local systems needed for associativity.
\end{enumerate}
A morphism in this category is a continuous natural transformation on
the whole \v{C}ech/Ran nerve, not merely on $0$-simplices. Thus for
every finite string $U_0,\ldots,U_p$ in the chosen DWR cover the map on
$U_0\cap\cdots\cap U_p$ is specified, continuous, compatible with
coface maps, and multiplicative for disjoint holomorphic polydiscs.
The phrase ``hCS-to-Hall quasi-isomorphism'' is meaningful only after
both sides have first been placed in this category, or after an
explicit forgetful functor from this category has been named.
\end{definition}

\begin{definition}[Local critical-CoHA normalisation]
\label{def:local-critical-coha-normalisation}
On a quiver chart $(Q_U,W_U)$ with dimension vector $\mathbf d$, write
\[
 \mathfrak M_{U,\mathbf d}
 =
 [\mathrm{Rep}_{\mathbf d}(Q_U)/G_{\mathbf d}],
 \qquad
 f_{U,\mathbf d}=\mathrm{Tr}(W_U)_{\mathbf d}.
\]
An oriented critical-CoHA summand in the convention of
Definition~\ref{def:hall-valued-factorisation-cosheaf-target} is
\[
 H^{\mathrm{BM}}_{G_{\mathbf d}}
 \!\left(
 \mathrm{Crit}(f_{U,\mathbf d}),
 \phi_{f_{U,\mathbf d}}\otimes\mathscr L_{o_U}
 \right)
 [s(U,\mathbf d)](t(U,\mathbf d)),
\]
where $\phi$ is the chosen vanishing-cycle functor, $[s]$ is the
perverse/cohomological shift, $(t)$ is the Tate twist, and
$\mathscr L_{o_U}$ is the local system determined by the chosen
orientation datum $o_U$. The symbols $s(U,\mathbf d)$ and
$t(U,\mathbf d)$ are part of the convention; a comparison with hCS
observables is ungraded until they are fixed.
\end{definition}

\begin{definition}[Rees critical Hall interlayer]
\label{def:rees-critical-hall-interlayer}
Let $(Q_U,W_U)$ be a quiver-with-potential chart on an admissible
polydisc $U$, let $\mathbf d$ be a dimension vector, and put
\[
 V_{U,\mathbf d}:=\mathrm{Rep}_{\mathbf d}(Q_U),
 \qquad
 G_{\mathbf d}:=\prod_{i\in Q_0}\GL_{\mathbf d_i},
 \qquad
 f_{U,\mathbf d}:=\mathrm{Tr}(W_U)_{\mathbf d}.
\]
Assume a finite cyclic BV polarisation has fixed a
$G_{\mathbf d}$-equivariant algebraic volume form
$\omega_{U,\mathbf d}$ on the finite model. The Rees critical complex of
the chart is
\[
 \Krit^{\hbar}_{G_{\mathbf d}}(U,\mathbf d)
 :=
 \left(
 C^\bullet\!\left(
 \mathfrak g_{\mathbf d},
 \Gamma(V_{U,\mathbf d},\wedge^\bullet T_{V_{U,\mathbf d}})[[\hbar]]
 \right),
 d_{\CE}+\iota_{df_{U,\mathbf d}}+\hbar\Delta_{\omega_{U,\mathbf d}}
 \right),
\]
where $\Delta_{\omega_{U,\mathbf d}}$ is the BV divergence determined by
the volume form. Equivalently, the Fourier--BV transform
\[
 \mathfrak F_{\omega_{U,\mathbf d}}\colon
 \Gamma(V_{U,\mathbf d},\wedge^kT_{V_{U,\mathbf d}})[[\hbar]]
 \longrightarrow
 \Omega^{\dim V_{U,\mathbf d}-k}(V_{U,\mathbf d})[[\hbar]],
 \qquad
 \xi\longmapsto
 (-1)^{k(k-1)/2}\iota_\xi\omega_{U,\mathbf d},
\]
identifies this complex with the twisted de Rham Rees model
\[
 \DR_{f_{U,\mathbf d},\hbar}(V_{U,\mathbf d})
 :=
 \left(
 \Omega^\bullet(V_{U,\mathbf d})[[\hbar]],
 \hbar d+df_{U,\mathbf d}\wedge
 \right)
\]
after passing to equivariant cochains. The oriented Rees critical Hall
summand is
\[
 \Hall^{\mathrm{Rees},\mathrm{or}}_{U,\mathbf d}
 :=
 \Krit^{\hbar}_{G_{\mathbf d}}(U,\mathbf d)
 \otimes o_{U,\mathbf d}[s(U,\mathbf d)](t(U,\mathbf d)),
\]
with $o_{U,\mathbf d}$ the determinant-line square root induced by the
finite BV polarisation, and with the same shift and Tate convention as
Definition~\ref{def:local-critical-coha-normalisation}. The completed
Rees critical Hall object is
\[
 \Hall_{\mathrm{crit}}^{\mathrm{Rees},\mathrm{or}}(U)
 :=
 \widehat{\bigoplus}_{\mathbf d}
 \Hall^{\mathrm{Rees},\mathrm{or}}_{U,\mathbf d}.
\]
This is a chain-level algebraic target. It is not the
vanishing-cycle critical CoHA until a realization functor
\[
 \mathsf R_{\mathrm{vc}}\colon
 \Hall_{\mathrm{crit}}^{\mathrm{Rees},\mathrm{or}}(U)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(U)
\]
has been supplied, compatible with equivariance, compact support,
orientation local systems, shifts, Tate twists, Thom--Sebastiani, Hall
proper pushforward, lci pullback, and the chosen completions.
\end{definition}

\begin{lemma}[Relative Fourier--BV Rees identification]
\label{lem:relative-fourier-bv-rees-identification}
\ClaimStatusProvedHere
Let $V$ be a smooth affine complex vector space of dimension $n$ with a
$G$-invariant algebraic volume form $\omega$, and let
$W\in\Gamma(V,\mathcal O_V)^G\otimes\Omega^\bullet(\Delta^p)$. Put
$\mathfrak g=\mathrm{Lie}(G)$, and write $d_VW$ for the de Rham
differential in the $V$-direction. Extend
\[
 \mathfrak F_{\omega}\colon
 \Gamma(V,\wedge^kT_V)[[\hbar]]\otimes\Omega^\bullet(\Delta^p)
 \longrightarrow
 \Omega^{n-k}(V)[[\hbar]]\otimes\Omega^\bullet(\Delta^p),
 \qquad
 \xi\longmapsto (-1)^{k(k-1)/2}\iota_\xi\omega
\]
$\Omega^\bullet(\Delta^p)[[\hbar]]$-linearly. With the divergence
convention
\[
 \mathfrak F_\omega(\Delta_\omega\xi)=d_V\mathfrak F_\omega(\xi),
\]
the transform is an isomorphism of equivariant Rees complexes
\[
\begin{aligned}
&\left(
 C^\bullet\!\left(
 \mathfrak g,
 \Gamma(V,\wedge^\bullet T_V)[[\hbar]]
 \otimes\Omega^\bullet(\Delta^p)
 \right),
 d_{\CE}+\iota_{d_VW}+\hbar\Delta_\omega+d_{\Delta^p}
 \right)
\\
&\hspace{6em}\cong
 \left(
 C^\bullet\!\left(
 \mathfrak g,
 \Omega^{n-\bullet}(V)[[\hbar]]
 \otimes\Omega^\bullet(\Delta^p)
 \right),
 d_{\CE}+d_VW\wedge+\hbar d_V+d_{\Delta^p}
 \right).
\end{aligned}
\]
It commutes with pullback to every face of $\Delta^p$. For two such
models $(V_i,W_i,\omega_i)$, $i=1,2$, it carries the external tensor
product differential to the Rees differential for
$(V_1\times V_2,W_1\boxplus W_2,\omega_1\wedge\omega_2)$, with the
standard product orientation.
\end{lemma}

\begin{proof}
The assertion is local on $V$. Choose linear coordinates
$x_1,\ldots,x_n$ with
$\omega=dx_1\wedge\cdots\wedge dx_n$. For a homogeneous $k$-vector
$\xi$ and a one-form $\alpha$, the contraction identity
\[
 \alpha\wedge\iota_\xi\omega
 =
 (-1)^{k-1}\iota_{\iota_\alpha\xi}\omega
\]
gives, after multiplying by the signs
$(-1)^{k(k-1)/2}$ and $(-1)^{(k-1)(k-2)/2}$,
\[
 \mathfrak F_\omega(\iota_\alpha\xi)
 =
 \alpha\wedge\mathfrak F_\omega(\xi).
\]
Taking $\alpha=d_VW$ identifies contraction by $d_VW$ on polyvectors
with wedge multiplication by $d_VW$ on forms. The chosen divergence
convention identifies $\Delta_\omega$ with $d_V$. The map
$\mathfrak F_\omega$ is $G$-equivariant because $\omega$ and $W$ are
$G$-invariant, hence it commutes with $d_{\CE}$. It is
$\Omega^\bullet(\Delta^p)[[\hbar]]$-linear, hence commutes with
$d_{\Delta^p}$ and with restriction to faces. These identities give the
displayed chain isomorphism.

For products, the volume form
$\omega_1\wedge\omega_2$ identifies polyvector contraction on
$V_1\times V_2$ with the completed tensor product of the two individual
contractions with the Koszul sign of the product orientation. The
operators split as
\[
 d_{V_1\times V_2}=d_{V_1}+d_{V_2},\qquad
 d(W_1\boxplus W_2)=d_{V_1}W_1+d_{V_2}W_2,\qquad
 \Delta_{\omega_1\wedge\omega_2}
 =\Delta_{\omega_1}+\Delta_{\omega_2}.
\]
Thus the tensor product differential is carried to the Rees
differential for the Thom--Sebastiani sum.
\end{proof}

\begin{proposition}[The Rees split of the hCS--Hall target]
\label{prop:rees-split-hcs-hall-target}
\noindent\emph{Hypotheses.} Assume finite cyclic HPT contractions and
monoidal vanishing-cycle realisation.
The direct arrow
\[
 \Obs_{\hCS}^{q}(-,\mathfrak g)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(-)
\]
is a theorem only after it has been factored as
\[
 \Obs_{\hCS}^{q}(-,\mathfrak g)
 \xrightarrow{\;\Theta_{\hCS\to\Hall}^{\mathrm{Rees},\mathrm{or}}\;}
 \Hall_{\mathrm{crit}}^{\mathrm{Rees},\mathrm{or}}(-)
 \xrightarrow{\;\mathsf R_{\mathrm{vc}}\;}
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(-).
\]
The first arrow is the finite algebraic comparison produced, on each
DWR/Ran simplex, by cyclic homological perturbation from the hCS BV
complex to a finite critical model and then by the Fourier--BV
identification of Definition~\ref{def:rees-critical-hall-interlayer}.
The second arrow is a separate microlocal/vanishing-cycle realization
theorem. Without $\mathsf R_{\mathrm{vc}}$ one has a comparison to
derived critical Hall chains, not to the usual critical CoHA. Without
$\Theta_{\hCS\to\Hall}^{\mathrm{Rees},\mathrm{or}}$ one has Hall normal
forms, not an hCS--Hall comparison.
\end{proposition}

\begin{proof}
The finite cyclic contraction transfers the hCS $L_\infty$ operations
to a cyclic finite model. The transferred Hamiltonian is the function
$f_{U,\mathbf d}$, and the quantum BV differential becomes
$\iota_{df_{U,\mathbf d}}+\hbar\Delta_{\omega_{U,\mathbf d}}$ on
polyvectors. Lemma~\ref{lem:relative-fourier-bv-rees-identification}
carries this differential to
$\hbar d+df_{U,\mathbf d}\wedge$, so the algebraic target of the finite
comparison is precisely the Rees critical complex of
Definition~\ref{def:rees-critical-hall-interlayer}. Hall multiplication
there is the Rees/Koszul form of the extension correspondence; its
degree, orientation, and Tate normalisation are the same data already
required in Definition~\ref{def:local-critical-coha-normalisation}.

The passage from the twisted Rees de Rham model to
Borel--Moore cohomology with vanishing cycles is not formal from the
cyclic contraction. It is a realization statement: it must commute with
Thom--Sebastiani, proper pushforward along Hall extension stacks,
pullback along local complete-intersection maps, the determinant-line
orientation system, and the completions. Therefore the displayed
factorisation is exactly the non-circular form of the hCS--Hall target.
It is not a derived-center theorem, a logarithmic Swiss-cheese action, a
boundary-category construction, a slab bimodule, or a Hall--Drinfeld
double. Those are separate center/open--closed/double problems, each with
its own Morita, operadic, pairing, and completion data.
\end{proof}

\begin{definition}[Oriented hCS--Hall comparison datum on the DWR/Ran nerve]
\label{def:cy3-oriented-hcs-hall-comparison-datum}
Fix a DWR-good cover $\mathfrak U$, a metric gauge Lie algebra
$\mathfrak g$, an anomaly-cancelled hCS quantisation in the sense of
Definition~\ref{def:cy3-hcs-quantum-observables}, a charge monoid
$\Gamma$, a stability condition, and equivariant parameters for the
critical Hall charts. An oriented hCS--Hall comparison datum is the
following typed object on $\mathsf{N}_{\mathsf{DW\!R}}(\mathfrak U)$.

For every simplex
\[
 \sigma=(S,\{P_s\}_{s\in S})
\]
the hCS side is the completed factorisation product
\[
 \Obs_{\hCS}^{q}(\sigma,\mathfrak g)
 :=
 \widehat{\bigotimes}_{s\in S}
 \Obs_{\hCS}^{q}(P_s,\mathfrak g),
\]
with differential
$Q_{\hCS}+\{I[L],-\}_{\BV}+\hbar\Delta_L$ on each factor and with
compact-support extension maps for refinements. The Hall side is the
charge/HN-completed and equivariant-localised direct sum
\[
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(\sigma)
 :=
 \widehat{\bigoplus}_{\gamma\in\Gamma^S}
 \widehat{\bigotimes}_{s\in S}
 H^{\mathrm{BM}}_{G_{\gamma_s}}
 \!\left(
 \mathrm{Crit}(\mathrm{Tr}W_{P_s,\gamma_s}),
 \phi_{\mathrm{Tr}W_{P_s,\gamma_s}}\otimes
 \mathscr L_{o_{P_s,\gamma_s}}
 \right)
 [s(P_s,\gamma_s)](t(P_s,\gamma_s)).
\]
Here $\mathrm{Tr}W_{P_s,\gamma_s}$ is the trace function on the local
representation stack, $[s]$ is the fixed cohomological shift, $(t)$ is
the fixed Tate twist, and the completion is part of the datum.
This is the realised Hall target. Equivalently, a comparison datum may be
given in the Rees-split form of
Definition~\ref{def:rees-critical-hall-interlayer}: one first maps to
$\Hall_{\mathrm{crit}}^{\mathrm{Rees},\mathrm{or}}(\sigma)$ and then
applies a monoidal vanishing-cycle realisation
$\mathsf R_{\mathrm{vc}}$. In that presentation the displayed target
below is shorthand for
$\mathsf R_{\mathrm{vc}}\circ
\Theta_{\sigma}^{\mathrm{Rees},\mathrm{or}}$; the five coherence
conditions are required both before realisation and after realisation.

The orientation component is a Joyce--Kontsevich--Soibelman system of
square roots
\[
 \ell_{P,\gamma}^{\otimes 2}
 \xrightarrow{\;\sim\;}
 K^{\mathrm{vir}}_{P,\gamma}
\]
of the virtual determinant line of each critical chart, together with
the induced orientation local system $\mathscr L_{o_{P,\gamma}}$ and
transport isomorphisms on all refinements and overlaps. These
transport maps satisfy the \v{C}ech cocycle condition on triple
overlaps and are compatible with direct sums and extensions. The Hall
product for charges $\gamma_1,\gamma_2$ is the pull--push along the
short-exact-sequence correspondence
\[
\begin{tikzcd}
& \mathfrak E_{\gamma_1,\gamma_2}(P)
 \arrow[dl, "p_1\times p_2"']
 \arrow[dr, "p_3"] & \\
\mathfrak M_{\gamma_1}(P)\times\mathfrak M_{\gamma_2}(P)
&&
\mathfrak M_{\gamma_1+\gamma_2}(P),
\end{tikzcd}
\]
composed with the Thom--Sebastiani isomorphism for vanishing cycles,
the orientation-line transport, and the shift/Tate-twist correction
making the product degree zero.

The comparison maps are continuous chain maps
\[
 \Theta_\sigma\colon
 \Obs_{\hCS}^{q}(\sigma,\mathfrak g)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(\sigma)
\]
for all simplices $\sigma$, satisfying five coherence conditions:
\begin{enumerate}
\item for every face, degeneracy, or refinement
$\rho\colon\sigma'\to\sigma$, the naturality square
\[
\begin{tikzcd}
\Obs_{\hCS}^{q}(\sigma',\mathfrak g)
 \arrow[r, "\Theta_{\sigma'}"]
 \arrow[d, "\rho_*"']
&
\CoHA_{\mathrm{crit}}^{\mathrm{or}}(\sigma')
 \arrow[d, "\rho_*"]\\
\Obs_{\hCS}^{q}(\sigma,\mathfrak g)
 \arrow[r, "\Theta_\sigma"']
&
\CoHA_{\mathrm{crit}}^{\mathrm{or}}(\sigma)
\end{tikzcd}
\]
commutes in completed complexes;
\item for disjoint DWR/Ran families $\sigma_1,\sigma_2$, the product
square
\[
\begin{tikzcd}
\Obs_{\hCS}^{q}(\sigma_1,\mathfrak g)
\widehat\otimes
\Obs_{\hCS}^{q}(\sigma_2,\mathfrak g)
 \arrow[r, "\Theta_{\sigma_1}\widehat\otimes\Theta_{\sigma_2}"]
 \arrow[d, "\mu_{\BV}^{\mathrm{fact}}"']
&
\CoHA_{\mathrm{crit}}^{\mathrm{or}}(\sigma_1)
\widehat\otimes
\CoHA_{\mathrm{crit}}^{\mathrm{or}}(\sigma_2)
 \arrow[d, "\mu_{\Hall}^{\mathrm{TS},o}"]\\
\Obs_{\hCS}^{q}(\sigma_1\sqcup\sigma_2,\mathfrak g)
 \arrow[r, "\Theta_{\sigma_1\sqcup\sigma_2}"']
&
\CoHA_{\mathrm{crit}}^{\mathrm{or}}(\sigma_1\sqcup\sigma_2)
\end{tikzcd}
\]
commutes;
\item the BV bracket on the source is sent to the commutator of the
Hall correspondence product after the CY$_3$ shift fixed by the
normalisation;
\item the two parenthesisations of a triple Hall extension, with their
Thom--Sebastiani and orientation transports, agree with the two
iterated hCS factorisation products;
\item every map is continuous for the $\hbar$-adic, strong-dual,
charge/HN-adic, and equivariant-localised topologies, and preserves the
chosen trace functions, shifts, and Tate twists.
\end{enumerate}
The datum is a quasi-isomorphic comparison datum when the maps on
$0$-simplices are quasi-isomorphisms after these completions and
normalisations.
\end{definition}

\begin{theorem}[Oriented comparison from a DWR/Ran datum]
\label{thm:oriented-hcs-hall-comparison-from-dwr-datum}
Every quasi-isomorphic oriented hCS--Hall comparison datum of
Definition~\ref{def:cy3-oriented-hcs-hall-comparison-datum} determines
a morphism
\[
 \Theta_{\hCS\to\Hall}^{\mathrm{or}}\colon
 \Obs_{\hCS}^{q}(-,\mathfrak g)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(-)
\]
in
$\mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}(X)$. If both sides
satisfy DWR/Weiss descent on $\mathfrak U$, the induced map on the
homotopy colimit over $\mathsf{N}_{\mathsf{DW\!R}}(\mathfrak U)$ is a
quasi-isomorphism. The theorem constructs no such datum; it proves that
once the orientation, trace, twist, completion, product, and coherence
data exist, they are exactly sufficient for the hCS-to-Hall comparison
on that DWR/Ran nerve.
\end{theorem}

\begin{proof}
The first coherence square says that the maps $\Theta_\sigma$ form a
natural transformation on the entire \v{C}ech/Ran simplicial category,
including its vertices and higher simplices. The second square identifies the disjoint
support product of BV observables with the Thom--Sebastiani Hall
product. The third condition identifies the induced Lie brackets after
the CY$_3$ shift. The fourth condition is precisely the associativity
coherence for the Hall correspondence product, including the two
orientation transports. The fifth condition states that the maps live
in the completed target category and have degree zero after the named
shift and Tate twist. Thus the data define a morphism in
$\mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}(X)$ by
Definition~\ref{def:hall-valued-factorisation-cosheaf-target}.

Assume now that both sides satisfy DWR/Weiss descent. The cover is
refinement-closed by definition, so the polydisc attached to any higher
\v{C}ech simplex is also a vertex after passing to the chosen
refinement basis. The quasi-isomorphism condition on vertices therefore
applies to the complexes computing each finite intersection, and the
naturality squares identify the higher-simplex maps with these
refinement maps. The standard descent comparison for homotopy colimits
of complexes then identifies the induced map on total DWR/Ran descent
with a homotopy colimit of quasi-isomorphisms, hence with a
quasi-isomorphism. No existence claim for
$\Theta_{\hCS\to\Hall}^{\mathrm{or}}$ is hidden in the argument: the
theorem begins only after the full typed datum has been supplied.
\end{proof}

\begin{definition}[Finite DWR/Ran cyclic critical atlas]
\label{def:finite-dwr-ran-cyclic-critical-atlas}
Fix finite bounds
\[
 N\quad\text{(holomorphic jets)},\qquad
 r\quad\text{(renormalisation order)},\qquad
 L\quad\text{(charge/HN bound)},\qquad
 m\quad\text{(Ran arity)}.
\]
A finite DWR/Ran cyclic critical atlas on a DWR-good cover
$\mathfrak U$ consists of the following data for every
$p$-simplex
\[
 \sigma=(S,\{P_s\Subset U_{i_{s,0}}\cap\cdots\cap
 U_{i_{s,p}}\}_{s\in S})
 \in \mathsf N^{\Ran}_{\leq m}(\mathfrak U;\Gamma_{\leq L}).
\]

\begin{enumerate}
\item A finite hCS quotient
$\Obs_{\hCS}^{q,\leq r,\leq N}(\sigma)_{\leq L}$, obtained by
holomorphic jet order $\leq N$, renormalisation weight $\leq r$, and
charge bound $\Gamma_{\leq L}$, with bar coalgebra
$B\Obs_{\hCS}^{q,\leq r,\leq N}(\sigma)_{\leq L}$.
\item A $G_\sigma$-equivariant finite cyclic model
$(V_\sigma,W_\sigma,\omega_\sigma)$ over the polynomial-form cdga
$\Omega^\bullet(\Delta^p)$, where
\[
 W_\sigma\in
 \Gamma(V_\sigma,\mathcal O_{V_\sigma})^{G_\sigma}
 \otimes \Omega^\bullet(\Delta^p),
\]
and a face-compatible cyclic contraction
\[
\left(
 H_\sigma
 \underset{p_\sigma}{\stackrel{i_\sigma}{\rightleftarrows}}
 \Obs_{\hCS}^{q,\leq r,\leq N}(\sigma)_{\leq L}
 \otimes\Omega^\bullet(\Delta^p),
 \ h_\sigma
\right)
\]
with
\[
 d_{\mathrm{tot}}h_\sigma+h_\sigma d_{\mathrm{tot}}
 =1-i_\sigma p_\sigma,\qquad
 h_\sigma^2=0,\qquad
 h_\sigma i_\sigma=0,\qquad
 p_\sigma h_\sigma=0,
\]
where $d_{\mathrm{tot}}$ includes the hCS differential and
$d_{\Delta^p}$, and with
$\partial_j^*(i_\sigma,p_\sigma,h_\sigma)=
(i_{\partial_j\sigma},p_{\partial_j\sigma},h_{\partial_j\sigma})$.
\item The relative Rees critical Hall complex
\[
 \Hall_{\sigma}^{\mathrm{Rees},\mathrm{or}}(\Delta^p)
 :=
 \left(
 C^\bullet\!\left(
 \mathfrak g_\sigma,
 \Gamma(V_\sigma,\wedge^\bullet T_{V_\sigma})[[\hbar]]
 \otimes\Omega^\bullet(\Delta^p)
 \right)^{G_\sigma},
 d_{\CE}+\iota_{d_VW_\sigma}
 +\hbar\Delta_{\omega_\sigma}+d_{\Delta^p}
 \right)
 \otimes o_\sigma[s_\sigma](t_\sigma),
\]
with determinant-line square root $o_\sigma$, shift $s_\sigma$, and
Tate twist $t_\sigma$ compatible with faces and disjoint union. Here
$d_V$ denotes the algebraic de Rham differential in the finite critical
model direction; $d_{\Delta^p}$ is the simplex differential.
\end{enumerate}

Cyclic homological perturbation transfers the finite hCS structure to
the cyclic $L_\infty$ structure encoded by $W_\sigma$. Composing this
transfer with the Fourier--BV identification of
Definition~\ref{def:rees-critical-hall-interlayer} gives a relative
degree-zero chain map
\[
 \theta_\sigma^{\mathrm{rel}}\colon
 B\Obs_{\hCS}^{q,\leq r,\leq N}(\sigma)_{\leq L}
 \longrightarrow
 \Hall_{\sigma}^{\mathrm{Rees},\mathrm{or}}(\Delta^p).
\]
The atlas is multiplicative when, for disjoint DWR/Ran families,
$W_{\sigma_1\sqcup\sigma_2}=W_{\sigma_1}\boxplus W_{\sigma_2}$ and the
Rees twisted de Rham identification carries external tensor product to
Thom--Sebastiani product.
\end{definition}

\begin{definition}[The finite hCS--Rees-Hall convolution Lie algebra]
\label{def:finite-hcs-rees-hall-convolution-lie-algebra}
Given a finite DWR/Ran cyclic critical atlas, put
\[
 \mathfrak M_{\hCS,\Hall}^{N,r,L,m}
 :=
 \Tot_{\sigma\in\mathsf N^{\Ran}_{\leq m}(\mathfrak U;\Gamma_{\leq L})}
 \Hom_{\mathrm{cont}}
 \left(
 B\Obs_{\hCS}^{q,\leq r,\leq N}(\sigma)_{\leq L},
 \Hall_{\sigma}^{\mathrm{Rees},\mathrm{or}}
 \right).
\]
Its differential is
\[
 D_{\mathrm{tot}}
 =
 d_{\Hall}\circ(-)
 -(-1)^{|-|}(-)\circ d_{B\Obs}
 +\delta_{\Cech/\Ran},
\]
and its bracket is the convolution bracket built from the bar
coproduct on $B\Obs_{\hCS}^{q}$ and the Rees Hall product on
$\Hall^{\mathrm{Rees},\mathrm{or}}$. Thus a degree-zero element
$\Theta$ is a multiplicative natural transformation on the finite
DWR/Ran nerve exactly when
\[
 D_{\mathrm{tot}}\Theta+\frac{1}{2}[\Theta,\Theta]=0.
\]
\end{definition}

\begin{theorem}[Finite DWR/Ran hCS-to-Rees-Hall construction]
\label{thm:finite-dwr-ran-hcs-rees-hall-construction}
Let $X$ be a smooth Calabi--Yau threefold with holomorphic volume form,
let $\mathfrak U$ be a DWR-good cover, and fix finite bounds
$(N,r,L,m)$. Suppose a multiplicative finite DWR/Ran cyclic critical
atlas in the sense of
Definition~\ref{def:finite-dwr-ran-cyclic-critical-atlas} is given.
For each $p$-simplex $\sigma$ define
\[
 \Theta_\sigma^{(p)}
 :=
 \int_{\Delta^p}\theta_\sigma^{\mathrm{rel}}
 \in
 \Hom_{\mathrm{cont}}^{-p}
 \left(
 B\Obs_{\hCS}^{q,\leq r,\leq N}(\sigma)_{\leq L},
 \Hall_{\sigma}^{\mathrm{Rees},\mathrm{or}}
 \right),
\]
where integration is the chain map
$\int_{\Delta^p}\colon C\otimes\Omega^\bullet(\Delta^p)\to C[-p]$.
Then
\[
 \Theta_{\hCS\to\Hall}^{\mathrm{Rees},\mathrm{or};N,r,L,m}
 :=
 \sum_{0\leq p\leq m}
 \sum_{\sigma\in
 \mathsf N^{\Ran}_{p,\leq m}(\mathfrak U;\Gamma_{\leq L})}
 \Theta_\sigma^{(p)}
 \in
 \MC\!\left(\mathfrak M_{\hCS,\Hall}^{N,r,L,m}\right)
\]
is a continuous orientation-preserving multiplicative natural
transformation
\[
 \Theta_{\hCS\to\Hall}^{\mathrm{Rees},\mathrm{or};N,r,L,m}\colon
 \Obs_{\hCS}^{q,\leq r,\leq N}(-)_{\leq L}
 \longrightarrow
 \Hall_{\mathrm{crit}}^{\mathrm{Rees},\mathrm{or},\leq N,\leq L}(-)
\]
on the finite DWR/Ran nerve. If the vertex maps
$\theta_\sigma^{\mathrm{rel}}|_{\Delta^0}$ are quasi-isomorphisms, then
the induced map on finite DWR/Ran descent is a quasi-isomorphism.
\end{theorem}

\begin{proof}
The cyclic homological perturbation theorem applied to the contraction
in Definition~\ref{def:finite-dwr-ran-cyclic-critical-atlas} produces,
over every simplex, a relative cyclic $L_\infty$ structure and hence a
relative Hamiltonian $W_\sigma$ satisfying the BV master equation over
$\Omega^\bullet(\Delta^p)$. After the relative Fourier--BV
identification of
Lemma~\ref{lem:relative-fourier-bv-rees-identification}, this is
equivalent to saying that
\[
 (d_{\Hall}+d_{\Delta^p})\theta_\sigma^{\mathrm{rel}}
 -\theta_\sigma^{\mathrm{rel}}d_{B\Obs}
 +\frac{1}{2}
 [\theta_\sigma^{\mathrm{rel}},\theta_\sigma^{\mathrm{rel}}]_{\mathrm{conv}}
 =0
\]
inside the relative convolution Lie algebra over $\Omega^\bullet(\Delta^p)$.

Integrate this identity over $\Delta^p$. Stokes' formula identifies the
integral of the $d_{\Delta^p}$ term with the alternating sum of the
restrictions to the faces:
\[
 \int_{\Delta^p}d_{\Delta^p}\eta
 =
 \sum_{j=0}^{p}(-1)^j
 \int_{\Delta^{p-1}}\partial_j^*\eta
 \quad
 \text{up to the standard cochain sign convention.}
\]
Because the contractions and relative chart maps are face-compatible,
these boundary terms are exactly the
$\delta_{\Cech/\Ran}$ component in
Definition~\ref{def:finite-hcs-rees-hall-convolution-lie-algebra}.
The remaining terms are the two internal differentials and the
bar--Hall convolution bracket. Summing over all simplices gives
\[
 D_{\mathrm{tot}}
 \Theta_{\hCS\to\Hall}^{\mathrm{Rees},\mathrm{or};N,r,L,m}
 +\frac{1}{2}
 \left[
 \Theta_{\hCS\to\Hall}^{\mathrm{Rees},\mathrm{or};N,r,L,m},
 \Theta_{\hCS\to\Hall}^{\mathrm{Rees},\mathrm{or};N,r,L,m}
 \right]
 =0.
\]
This is precisely the Maurer--Cartan equation, hence the element
defines a multiplicative natural transformation on the finite DWR/Ran
nerve.

Orientation, shift, and Tate compatibility are part of the coefficient
system $o_\sigma[s_\sigma](t_\sigma)$ and of its face-compatible
transport. Multiplicativity under disjoint union follows from the
assumption
$W_{\sigma_1\sqcup\sigma_2}=W_{\sigma_1}\boxplus W_{\sigma_2}$ and from
the strict Rees Thom--Sebastiani identity
\[
 \DR_{W_{\sigma_1},\hbar}\widehat\otimes
 \DR_{W_{\sigma_2},\hbar}
 \cong
 \DR_{W_{\sigma_1}\boxplus W_{\sigma_2},\hbar}.
\]
Thus the Maurer--Cartan element is a morphism in the finite oriented
Hall-valued factorisation-cosheaf category.

Finally, the finite DWR/Ran nerve is a finite diagram of complexes in
the stated bounds. Objectwise quasi-isomorphisms induce a
quasi-isomorphism on finite totalisations, equivalently on finite
homotopy colimits of these diagrams. The vertex quasi-isomorphism
hypothesis, together with the face-compatible contraction data, gives
objectwise quasi-isomorphisms on every simplex after refinement, so the
map on finite DWR/Ran descent is a quasi-isomorphism.
\end{proof}

\begin{corollary}[Completed and realised comparison as two further arrows]
\label{cor:completed-realised-hcs-hall-two-arrows}
Assume the finite constructions of
Theorem~\ref{thm:finite-dwr-ran-hcs-rees-hall-construction} form a
strict inverse system of complete filtered convolution dg Lie algebras
under the four transition systems
\[
 N+1\to N,\qquad r+1\to r,\qquad
 \Gamma_{\leq L+1}\to\Gamma_{\leq L},\qquad
 \mathsf N^{\Ran}_{\leq m+1}\to\mathsf N^{\Ran}_{\leq m},
\]
with complete, exhaustive, separated filtrations for which the
Maurer--Cartan series converges strongly. Assume moreover that the
finite Maurer--Cartan elements, orientation transports, grading
transports, Thom--Sebastiani homotopies, factorisation homotopies, and
quasi-isomorphism data are transition-compatible, and that the inverse
systems of cohomology groups in the obstruction degrees satisfy the
Mittag--Leffler condition, killing the associated $\varprojlim^1$ term.
Then the inverse limit of the finite Maurer--Cartan elements defines a
completed Rees comparison
\[
 \Theta_{\hCS\to\Hall}^{\mathrm{Rees},\mathrm{or},\wedge}\colon
 \Obs_{\hCS}^{q,\wedge}(-)
 \longrightarrow
 \Hall_{\mathrm{crit}}^{\mathrm{Rees},\mathrm{or},\wedge}(-).
\]
If, in addition, an orientation-preserving realization functor
\[
 \mathsf R_{\mathrm{vc}}\colon
 \Hall_{\mathrm{crit}}^{\mathrm{Rees},\mathrm{or},\wedge}(-)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or},\wedge}(-)
\]
is supplied and is monoidal for Thom--Sebastiani, compact-support
pushforward, lci pullback, equivariance, determinant-line orientations,
shifts, Tate twists, and the chosen completions, then
\[
 \Theta_{\hCS\to\Hall}^{\mathrm{or},\wedge}
 :=
 \mathsf R_{\mathrm{vc}}\circ
 \Theta_{\hCS\to\Hall}^{\mathrm{Rees},\mathrm{or},\wedge}
\]
is the oriented hCS-to-critical-CoHA comparison on the completed
DWR/Ran nerve.
\end{corollary}

\begin{proof}
The finite construction gives Maurer--Cartan elements in the finite
convolution Lie algebras. The strict transition hypotheses make them an
inverse-system element together with the five primitive systems.
Completeness and strong convergence make the infinite
Maurer--Cartan expression meaningful, while separatedness prevents
phantom limit classes. The Mittag--Leffler hypothesis removes the
$\varprojlim^1$ obstruction in the named obstruction degrees, so the
inverse limit again satisfies the Maurer--Cartan equation and gives the
completed Rees map. Applying a monoidal realization
functor preserves the differential, Hall product, Thom--Sebastiani
product, orientation system, shifts, Tate twists, and completions by
hypothesis; hence its composite with the completed Rees map is a
morphism to the usual critical CoHA.
\end{proof}

\begin{theorem}[Finite positive half and compact-promotion obstruction]
\label{thm:finite-positive-half-obstruction-map}
Let the finite Rees comparison of
Theorem~\ref{thm:finite-dwr-ran-hcs-rees-hall-construction} be given on
a DWR/Ran sub-nerve whose affine charts are toric $\C^3$ charts with
abelian gauge $\ghat$. Suppose in addition that each such chart is
projected to the fixed abelian finite-mode shuffle sector of
Definition~\ref{def:c3-fixed-abelian-hcs-hall-chart}; equivalently, in
bounded charge the finite chart cohomology is identified with the
Schiffmann--Vasserot positive half
\[
 H^\bullet_{\mathrm{fp},+}\!\left(
 \Hall_{\mathrm{crit}}^{\mathrm{Rees},\mathrm{or},\leq N,\leq L}(\C^3)
 \right)
 \cong
 Y_T^+(\ghat)_{\leq L}.
\]
Then the projected composite is the finite positive-half comparison
\[
 \Theta_{+}^{N,r,L,m}\colon
 \Obs_{\hCS}^{q,\leq r,\leq N}(-)_{\leq L}
 \longrightarrow
 Y_T^+(\ghat)_{\leq L}
\]
on the finite DWR/Ran nerve. Its promotion to compact critical CoHA,
to a Hall--Drinfeld double, and to a PBW/Serre-kernel theorem is
measured by the obstruction vector
\[
 \mathfrak o_{\mathrm{cpt/dbl/PBW}}(\Theta_{+}^{N,r,L,m})
 =
 \bigl(
 o_{\mathrm{vc}},\,
 o_{\mathrm{or}},\,
 o_{\mathrm{dbl}},\,
 o_{\mathrm{PBW}}
 \bigr),
\]
where
\[
\begin{array}{ll}
o_{\mathrm{vc}} &
\textup{is the obstruction to a monoidal vanishing-cycle realisation }
\mathsf R_{\mathrm{vc}},\\
o_{\mathrm{or}} &
\textup{is the determinant-line square-root transport class on the
DWR/Ran nerve},\\
o_{\mathrm{dbl}} &
\textup{is the completed Manin-pair obstruction: negative half,
non-degenerate pairing, radical quotient},\\
o_{\mathrm{PBW}} &
\textup{is the associated-graded and Borcherds--Serre kernel
obstruction for the completed presentation}.
\end{array}
\]
The finite positive-half comparison gives none of these four structures
by itself.
\end{theorem}

\begin{proof}
The finite Rees comparison is a Maurer--Cartan element in
$\mathfrak M_{\hCS,\Hall}^{N,r,L,m}$ with target the Rees critical Hall
interlayer. On a toric $\C^3$ chart, the Kontsevich--Soibelman
critical-CoHA model and the Schiffmann--Vasserot theorem identify the
fixed abelian finite-mode shuffle sector with
$Y_T^+(\ghat)$ in bounded charge. Composing the finite Rees map with
this chartwise projection gives $\Theta_{+}^{N,r,L,m}$.

The target of $\Theta_{+}^{N,r,L,m}$ is only a positive half. A compact
critical CoHA requires the realisation functor
$\mathsf R_{\mathrm{vc}}$ with proper pushforward, lci pullback,
Thom--Sebastiani, orientation, and completion compatibilities. A
Hall--Drinfeld double requires a negative half and a non-degenerate
completed Hopf pairing after radical quotient. A PBW/Serre-kernel
statement requires a separated filtration, an associated-graded
identification, and the Borcherds--Serre kernel computation. These are
precisely the four displayed obstruction components. Thus the finite
positive-half map is a constructed finite algebraic comparison, while
compact CoHA, doubling, and PBW closure are exactly the additional
vanishing conditions named above.
\end{proof}

\section{The anomaly gate}
\label{sec:cy3-chain-level-bridge-anomaly-gate}

\begin{proposition}[Quartic anomaly slot for hCS on a complex threefold]
\label{prop:cy3-hcs-quartic-anomaly-slot}
For holomorphic Chern--Simons theory on a complex threefold, the local
one-loop obstruction belongs to the invariant-polynomial slot of degree
$4$ on $\mathfrak g$: the quartic Casimir datum in the Costello--Li
local Lie-algebra cohomology
description~\cite{CostelloLi2016, CostelloLi2020}. The cubic slot
governs the complex-surface and five-dimensional hCS descendants on a
different dimensional component of the family.
\end{proposition}

\begin{proof}
Costello--Li identify the one-loop obstruction for a holomorphic theory
on a complex $d$-fold with the local Lie-algebra cohomology class
attached to an invariant polynomial
$P_{d+1}\in\Sym^{d+1}(\mathfrak g^\vee)^{\mathfrak g}$. In local
coordinates the representative has schematic form
\[
 \Theta_{P_{d+1}}(\alpha_0,\ldots,\alpha_d)
 =
 \int_X\Omega_X\wedge
 P_{d+1}(\alpha_0,\partial\alpha_1,\ldots,\partial\alpha_d),
\]
with the Dolbeault heat-kernel regularisation supplying the local
functional. Thus the invariant-polynomial slot has degree $d+1$.
Substituting $d = 3$ gives
$P_4 \in \Sym^4(\mathfrak g^\vee)^{\mathfrak g}$. The cubic class
specialises to the $d = 2$ slot; the CY$_3$ hCS bridge runs through
the $d + 1 = 4$ slot, with the dimension shift carried by the
$\Omega_X \wedge P_{d+1}(\alpha_0, \partial\alpha_1, \ldots, \partial\alpha_d)$
representative above.
\end{proof}

\begin{remark}[Two quartic data: the gauge-anomaly class and the bar shadow coefficient]
\label{rem:cy3-quartic-anomaly-vs-shadow}
The Costello--Li anomaly polynomial
$P_4 \in \Sym^4(\mathfrak g^\vee)^{\mathfrak g}$ is the gauge-anomaly
class for quantising hCS observables. The shadow coefficient $S_4$ in
the $E_3$-bar spectral sequence is the bar differential coefficient of
a chiral algebra. The two quartic data live on different sides of the
hCS-to-chiral comparison; their identification proceeds through an
explicit BV-to-bar transfer map.
\end{remark}

\begin{remark}[CFG and the CY$_3$ hCS avatar as separate constructions]
\label{rem:cy3-no-cfg-shortcut}
Costello--Francis--Gwilliam's Chern--Simons/factorisation-homology
machine~\cite{CFG2026} constructs a filtered $E_3$-algebra by BV
quantisation of ordinary $3$d Chern--Simons theory and recovers
Reshetikhin--Turaev link invariants by factorisation-homology traces.
Its associated classical local model is $C^\bullet(\mathfrak g)$,
coming from the formal neighbourhood of the trivial flat bundle on a
real $3$-ball. The CY$_3$ hCS avatar is the many-variable
Dolbeault--chiral CE object of
Definition~\ref{def:cy3-many-variable-chiral-ce}; it carries
holomorphic jets in $z_1, z_2, z_3$, polydisc factorisation, and
multidirectional residues. The comparison from six-real-dimensional
hCS on a CY$_3$ to the CY$_3$ Hall algebra is the map
$\Theta_{\hCS \to \Hall}$ in Construction~\ref{con:cy3-typed-bridge},
a separate mathematical datum constructed on top of both the CFG
filtered-$E_3$ and Costello--Li holomorphic frameworks.
\end{remark}

\begin{definition}[Holomorphic perfect defects for a CY$_3$ Stage-$1$ algebra]
\label{def:cy3-holomorphic-perfect-defect}
Let $F_X=\PhiFA_3(\cC)_F$ be a constructed Stage-$1$
$E_3$-holomorphic factorisation algebra on the framed CY$_3$ locus, and
let $i\colon C\hookrightarrow X$ be a holomorphic curve with finite
marked endpoint set $S\subset C$. A holomorphic defect over $(X,C,S)$ is
a constructible $E_{1\subset 3}$-module coefficient system
\[
 M\in \mathrm{HolMod}_{F_X}(X,C,S)
\]
consisting of bulk coefficients $F_X(P)$ on polydiscs $P\subset X$,
curve coefficients $M_C(D)$ on discs $D\subset C$, endpoint objects
$M_p,M_p^\vee$ for $p\in S$, and descent data on the stratified
Dolbeault/Weiss/Ran nerve of $(X,C,S)$. In a normal chart
$P=D_z\times D_u\times D_v$ with
$C\cap P=D_z\times\{0,0\}$, the curve action is by the
normal-completed dg Lie algebra
\[
 \mathfrak L_{X,C}^{\wedge}(D_z)
 =
 \Omega^{0,\bullet}_c
 \!\left(
 D_z,
 J^\infty_{\mathrm{hol}}(\mathfrak l_{\cC}|_C)
 \widehat\otimes
 \widehat{\Sym}(N_{C/X}^\vee)
 \right)[1].
\]
The defect is perfect if it is compact and dualisable in
$\mathrm{HolMod}_{F_X}(X,C,S)$, finite over the normal-completed local
algebra, of finite Tor-amplitude on each DWR chart, compatible with the
CY orientation datum, and trace-class for the continuous Dolbeault
complexes. Endpoint data means evaluation and coevaluation maps
$M_p^\vee\otimes M_p\to \mathbb C$ and
\(\mathbb C\to M_p\otimes M_p^\vee\) satisfying the zigzag identities in
the one-sided local module category.
\end{definition}

\begin{proposition}[Formal holomorphic defect trace]
\label{prop:cy3-holomorphic-defect-trace}
Let $M$ be a perfect holomorphic defect for $F_X$ supported on
$(C,S)$. Then factorisation homology of the stratified pair gives a
trace class
\[
 \mathrm{Tr}^{\mathrm{hol}}_{F_X}(M)
 \in
 \int_{(X,C,S)} (F_X,M),
\]
and after a witnessed Stage-$2$ specialisation datum
$(\Sigma_2,C)$ it gives a trace class for the $E_1$ chiral algebra
\[
 A_C=\SpCh_{\Sigma_2,C}(F_X).
\]
This is the holomorphic analogue of the Costello--Francis--Gwilliam
ordinary Chern--Simons trace grammar. It does not identify the trace
with a Hall, DT, or Borcherds character unless a module-level extension
of $\Theta_{\hCS\to\Hall}^{\mathrm{or}}$ and the Hall--Borcherds
denominator comparison are supplied.
\end{proposition}

\begin{proof}
Perfectness gives compactness and dualisability in the holomorphic
factorisation-module category, so the evaluation and coevaluation maps
at the marked endpoints define the usual categorical trace in
stratified factorisation homology. The normal-completed action in
Definition~\ref{def:cy3-holomorphic-perfect-defect} is retained during
collision of curve discs; hence the trace is a class in the
holomorphic, not locally constant, stratified homology of $(X,C,S)$.
Applying the exact witnessed specialisation functor
$\SpCh_{\Sigma_2,C}$ transports the class to the curve algebra $A_C$.
No Hall statement follows from this formal trace construction: Hall
traces require vanishing cycles, orientation-line square roots,
charge/HN completions, and Thom--Sebastiani compatibility, precisely
the data isolated in Problem~\ref{op:cy3-hcs-hall-comparison} and
Definition~\ref{def:hcs-hall-descent-obstruction}.
\end{proof}

\begin{definition}[Module hCS--Hall obstruction complex]
\label{def:hcs-hall-module-obstruction}
Assume a base chartwise algebra comparison
\[
 \theta_i\colon
 \Obs_{\hCS}^{q}(U_i,\mathfrak g)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(U_i)
\]
has been fixed on a DWR-good cover, and let $M$ be a holomorphic
perfect defect in the sense of
Definition~\ref{def:cy3-holomorphic-perfect-defect}. For a candidate
Hall module $N$ over $\CoHA_{\mathrm{crit}}^{\mathrm{or}}$, put on each
intersection $U_I$
\[
 \mathcal N_{\mathrm{mod}}^q(U_I)
 :=
 \Hom^q_{\mathrm{cont},\,\theta_I\text{-}\mathrm{mod}}
 \bigl(M_{\mathrm{hol}}(U_I),N_{\Hall}(U_I)\bigr).
\]
The total module comparison complex is
\[
 \mathfrak M_{\hCS,\Hall}^{\mathrm{mod}}(\mathfrak U)
 :=
 \Tot\,\Cech^\bullet(\mathfrak U,\mathcal N_{\mathrm{mod}}^\bullet).
\]
It is a filtered dg module over
$\mathfrak M_{\hCS,\Hall}(\mathfrak U)$, and the pair
$(\theta,\eta)$ is governed by the semidirect convolution dg Lie algebra
\[
 \mathfrak M_{\hCS,\Hall}^{\mathrm{pair}}
 =
 \mathfrak M_{\hCS,\Hall}
 \ltimes
 \mathfrak M_{\hCS,\Hall}^{\mathrm{mod}}.
\]
A degree-zero pair is a module comparison datum when
\[
 d\theta+\frac12[\theta,\theta]=0,
 \qquad
 d_\theta\eta=0,
\]
and each local $\eta_i$ is a quasi-isomorphism after the chosen
completion, orientation, shifts, Tate twists, charge convention, and
endpoint/puncture datum. Its obstruction tuple is
\[
 \mathfrak o_{\mathrm{mod}}(\theta,\eta)
 =
 \bigl(
 o_{\mathrm{MC}}^{\mathrm{mod}},
 o_{\mathrm{or}}^{\mathrm{mod}},
 o_{\mathrm{end}},
 o_{\mathrm{punc}},
 o_{\mathrm{TS}}^{\mathrm{mod}},
 o_{\mathrm{tr}},
 o_{\mathrm{comp}}
 \bigr).
\]
Here $o_{\mathrm{MC}}^{\mathrm{mod}}$ is the class of
$d_\theta\eta$; $o_{\mathrm{or}}^{\mathrm{mod}}$ is the determinant-line
square-root mismatch for the module virtual normal complex;
$o_{\mathrm{end}}$ measures failure of endpoint evaluation and
coevaluation to match; $o_{\mathrm{punc}}$ measures nearby-cycle
monodromy, normal-residue, and puncture-flavour mismatch;
$o_{\mathrm{TS}}^{\mathrm{mod}}$ is the Thom--Sebastiani associator
defect for module convolution; $o_{\mathrm{tr}}$ is the failure of
holomorphic and Hall traces to commute with the endpoint trace maps; and
$o_{\mathrm{comp}}$ is the mismatch of charge/HN completions and
stability-chamber filtrations.
\end{definition}

\begin{theorem}[Module-level hCS--Hall comparison criterion]
\label{thm:hcs-hall-module-comparison-criterion}
Assume the base obstruction
$\mathfrak o(\{\theta_i\})$ of
Definition~\ref{def:hcs-hall-descent-obstruction} vanishes, so that
$\Theta_{\hCS\to\Hall}^{\mathrm{or}}$ exists on the DWR nerve. A
chartwise family of module quasi-isomorphisms $\eta_i$ extends to a
global module-level comparison
\[
 \Theta_{\hCS\to\Hall}^{\mathrm{mod}}(M)\colon
 M_{\mathrm{hol}}\longrightarrow N_{\Hall}
\]
if and only if
\[
 \mathfrak o_{\mathrm{mod}}(\theta,\eta)=0
\]
and the resulting class is invertible in
$H^0(\mathfrak M_{\hCS,\Hall}^{\mathrm{mod}})$ on every DWR/Ran
simplex. When this holds, the holomorphic trace maps to the Hall trace:
\[
 \Theta_{\hCS\to\Hall}^{\mathrm{tr}}
 \bigl(\mathrm{Tr}_{F_X}^{\mathrm{hol}}(M)\bigr)
 =
 \mathrm{Tr}_{\CoHA_{\mathrm{crit}}^{\mathrm{or}}}^{\Hall}
 \bigl(\Theta_{\hCS\to\Hall}^{\mathrm{mod}}(M)\bigr).
\]
\end{theorem}

\begin{proof}
With $\theta$ fixed, descent for module maps is governed by the twisted
differential $d_\theta$ on
$\mathfrak M_{\hCS,\Hall}^{\mathrm{mod}}$. Solving
$d_\theta\eta=0$ and imposing local quasi-isomorphism is exactly the
module analogue of the Maurer--Cartan descent condition in
Theorem~\ref{thm:hcs-hall-descent-criterion}. Restoring the Hall
module structures adds the six coherent/discrete conditions listed in
Definition~\ref{def:hcs-hall-module-obstruction}: orientation,
endpoint, puncture, Thom--Sebastiani, trace, and completion
compatibilities. Vanishing of all seven components is therefore
equivalent to a global morphism of oriented completed module-valued
factorisation cosheaves. Trace compatibility is the component
$o_{\mathrm{tr}}=0$ together with the endpoint zigzag identities in the
perfect defect definition.
\end{proof}

\section{The hCS--Hall obstruction}
\label{sec:cy3-chain-level-bridge-missing-lemma}

\begin{openproblem}[hCS-to-Hall comparison]
\label{op:cy3-hcs-hall-comparison}
Construct, for a smooth CY$_3$ $X$ with holomorphic volume form
$\Omega_X$ and a Dolbeault/Weiss/Ran-good Stein-polydisc cover
$\mathfrak U$, a quasi-isomorphic oriented comparison datum in the
sense of Definition~\ref{def:cy3-oriented-hcs-hall-comparison-datum},
equivalently a continuous natural transformation on the whole DWR
\v{C}ech/Ran nerve in
$\mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}(X)$
\[
 \Theta_{\hCS\to\Hall}^{\mathrm{or}}\colon
 \Obs_{\hCS}^{q}(-,\mathfrak g)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(-)
\]
where the target is the critical CoHA with the normalisation of
Definition~\ref{def:local-critical-coha-normalisation}. In local quiver
coordinates this means
\[
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(U)
 =
 \bigoplus_{\mathbf d}
 H^{\mathrm{BM}}_{G_{\mathbf d}}
 \!\left(
	 \mathrm{Crit}(\mathrm{Tr}W_{\mathbf d}),
	 \phi_{\mathrm{Tr}W_{\mathbf d}}\otimes\mathscr L_{o_U}
	 \right)
 [s(U,\mathbf d)](t(U,\mathbf d))
\]
with the perverse shift $s(U,\mathbf d)$ and Tate twist
$t(U,\mathbf d)$ fixed as part of the data. The orientation datum $o_U$
is a KS/Joyce square root of the virtual determinant line with extension
transport, orientation local systems, and overlap coherences; it is not
merely a formal notation for $K_{\mathfrak M_U}^{1/2}$. The comparison
must state the gauge/rank datum, equivariant parameters, stability
condition, anomaly cancellation, completion, and compact-support
convention, and it must satisfy all seven conditions:
\begin{enumerate}
\item it is local for the Dolbeault topology and respects Weiss descent;
\item it intertwines the BV bracket and the Hall correspondence product
after the CY$_3$ shift;
\item on the $\C^3$ chart it reduces to the
Kontsevich--Soibelman/Schiffmann--Vasserot positive-half model
$\CoHA(\C^3)=Y^+$;
\item after Drinfeld doubling it is compatible with the
$\cW_{1+\infty}$ Fock/evaluation representation;
\item it transports determinant-line square roots on overlaps and kills
the residual $\Z/2$ orientation cocycle on triple overlaps;
\item it preserves the specified shifts, Tate twists, and equivariant
weights, so the comparison is degree-zero only after those conventions
are fixed;
\item it respects Thom--Sebastiani for every iterated short exact
sequence correspondence, including both parenthesisations of Hall
convolution and the transported orientation local systems.
\end{enumerate}
This is the first lemma whose proof would turn the present local
bridge into a non-formal compact CY$_3$ theorem.
\end{openproblem}

\begin{definition}[The hCS--Hall descent obstruction]
\label{def:hcs-hall-descent-obstruction}
Fix the cover $\mathfrak U$ and the two objects
$\Obs_{\hCS}^{q}$ and $\CoHA_{\mathrm{crit}}^{\mathrm{or}}$ in
$\mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}(X)$. For every
intersection $U_I=U_{i_0}\cap\cdots\cap U_{i_p}$ set
\[
 \mathcal M^q(U_I)
 :=
 \Hom_{\mathrm{cont}}^q
 \bigl(\Obs_{\hCS}^{q}(U_I,\mathfrak g),
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(U_I)\bigr).
\]
The Cech differential, the two internal differentials, the BV bracket
on the source, and the Hall convolution bracket on the target make
\[
 \mathfrak M_{\hCS,\Hall}(\mathfrak U)
 :=
 \Tot\,\Cech^\bullet(\mathfrak U,\mathcal M^\bullet)
\]
a complete filtered dg Lie algebra. A degree-zero element
$\theta\in\mathfrak M_{\hCS,\Hall}(\mathfrak U)^0$ is a comparison
datum precisely when it satisfies the Maurer--Cartan equation
\[
 d\theta+\frac{1}{2}[\theta,\theta]=0
\]
and its $0$-simplex components are quasi-isomorphisms after the fixed
shift, Tate twist, completion, and orientation convention of
Definition~\ref{def:local-critical-coha-normalisation}. The obstruction
class of a chartwise family $\{\theta_i\}$ is the tuple
\[
 \mathfrak o(\{\theta_i\})
 =
 \bigl(
 o_{\mathrm{MC}},\,
 o_{\mathrm{or}},\,
 o_{\mathrm{gr}},\,
 o_{\mathrm{TS}},\,
 o_{\mathrm{fact}}
 \bigr),
\]
where $o_{\mathrm{MC}}$ is the class of
$d\theta+\frac{1}{2}[\theta,\theta]$ in
$H^1(\mathfrak M_{\hCS,\Hall})$, $o_{\mathrm{or}}$ is the Cech
$\mathbb Z/2$-class of the determinant-line square-root mismatch,
$o_{\mathrm{gr}}$ is the integer/Tate grading mismatch of the functions
$s(U,\mathbf d)$ and $t(U,\mathbf d)$, $o_{\mathrm{TS}}$ is the
associator defect for the two Thom--Sebastiani parenthesisations, and
$o_{\mathrm{fact}}$ is the defect of multiplicativity under disjoint
polydiscs.
\end{definition}

\begin{theorem}[Descent criterion for \texorpdfstring{$\Theta_{\hCS\to\Hall}^{\mathrm{or}}$}{Theta hCS to Hall}]
\label{thm:hcs-hall-descent-criterion}
For a fixed cover $\mathfrak U$, fixed local critical-CoHA
normalisation, and fixed compact-support convention, a chartwise family
of quasi-isomorphisms
\[
 \theta_i\colon
 \Obs_{\hCS}^{q}(U_i,\mathfrak g)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(U_i)
\]
extends to a global orientation-preserving morphism
\[
 \Theta_{\hCS\to\Hall}^{\mathrm{or}}\colon
 \Obs_{\hCS}^{q}(-,\mathfrak g)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(-)
\]
in
$\mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}(X)$ if and only if
\[
 \mathfrak o(\{\theta_i\})=0
\]
and the resulting degree-zero Maurer--Cartan element is invertible in
$H^0$ on every object of the DWR nerve. When this holds, the space of
extensions is a torsor under
$H^0(\mathfrak M_{\hCS,\Hall}(\mathfrak U))$ modulo the gauge action of
$\exp(F^1\mathfrak M_{\hCS,\Hall}^0)$.
\end{theorem}

\begin{proof}
A morphism of the Hall-valued factorisation-cosheaf category is, by
Definition~\ref{def:hall-valued-factorisation-cosheaf-target}, a
continuous natural transformation on the whole Cech/Ran nerve preserving
orientation, completion, grading, Thom--Sebastiani, and disjoint-union
products. Forget these five extra structures. Descent for continuous
maps of complexes says that extending the $0$-simplex maps
$\theta_i$ to all intersections is equivalent to solving the
Maurer--Cartan equation in the total Cech convolution dg Lie algebra
$\mathfrak M_{\hCS,\Hall}(\mathfrak U)$; the differential records Cech
failure and chain-map failure, while the bracket records failure to
intertwine the two products. This gives the obstruction
$o_{\mathrm{MC}}$ and the stated gauge torsor.

Restoring the forgotten structures gives the remaining four components.
The square roots of determinant lines form a $\mathbb Z/2$-torsor, so
their mismatch on double and triple overlaps is exactly
$o_{\mathrm{or}}$. The shifts and Tate twists are discrete grading data;
their mismatch is the class $o_{\mathrm{gr}}$. The
Thom--Sebastiani isomorphism has two parenthesisations on a triple
extension correspondence; their ratio is $o_{\mathrm{TS}}$. The
factorisation product for disjoint polydiscs is preserved precisely when
$o_{\mathrm{fact}}$ vanishes. If all five classes vanish, the local
quasi-isomorphisms glue to a morphism in
$\mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}(X)$. Conversely, any
such morphism restricts to a Maurer--Cartan solution and preserves each
of the four discrete/coherent structures, so every component of
$\mathfrak o$ is zero.
\end{proof}

\begin{definition}[Primitive data for the oriented hCS--Hall arrow]
\label{def:hcs-hall-primitive-witness-package}
Fix a DWR-good cover $\mathfrak U$, the local critical-CoHA
normalisation of Definition~\ref{def:local-critical-coha-normalisation},
and the compact-support and completion conventions on both sides. Let
\[
 \theta^{(0)}=\{\theta_i\}_{i\in I},
 \qquad
 \theta_i\colon
 \Obs_{\hCS}^{q}(U_i,\mathfrak g)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(U_i)
\]
be chartwise degree-zero continuous chain maps. Primitive data for
$\Theta_{\hCS\to\Hall}^{\mathrm{or}}$ are the finite typed datum
\[
 \mathfrak P_{\Theta}^{\mathrm{or}}
 =
 \bigl(
 \theta^{(0)},\,
 \eta_{\mathrm{MC}},\,
 \lambda_{\mathrm{or}},\,
 \eta_{\mathrm{gr}},\,
 H_{\mathrm{TS}},\,
 H_{\mathrm{fact}},\,
 Q
 \bigr)
\]
with the following entries. It is finite in type: on an infinite
Ran-refinement the entries are compatible pro-cochains, not a finite
enumeration of opens.
\begin{enumerate}
\item $\eta_{\mathrm{MC}}\in F^1
\mathfrak M_{\hCS,\Hall}(\mathfrak U)^0$ is a positive Cech/Ran
filtration correction such that
\[
 \theta:=\theta^{(0)}+\eta_{\mathrm{MC}}
 \quad\text{satisfies}\quad
 d\theta+\frac12[\theta,\theta]=0.
\]
Thus $\eta_{\mathrm{MC}}$ is the higher-simplex primitive for the
Maurer--Cartan curvature of the vertex maps.
\item $\lambda_{\mathrm{or}}\in
\Cech^1(\mathfrak U,\mathbb Z/2)$ is a square-root transport cochain for
the orientation local systems satisfying
\[
 \delta\lambda_{\mathrm{or}}=o_{\mathrm{or}}.
\]
\item $\eta_{\mathrm{gr}}=(\eta_s,\eta_t)$ is an integral/Tate grading
transport cochain satisfying
\[
 \delta\eta_s=o_{\mathrm{gr}}^{s},
 \qquad
 \delta\eta_t=o_{\mathrm{gr}}^{t},
\]
where $o_{\mathrm{gr}}=(o_{\mathrm{gr}}^s,o_{\mathrm{gr}}^t)$ records
the mismatch of $s(U,\mathbf d)$ and $t(U,\mathbf d)$ on overlaps.
\item $H_{\mathrm{TS}}$ is a chain homotopy on every triple
short-exact-sequence correspondence satisfying
\[
 dH_{\mathrm{TS}}=o_{\mathrm{TS}},
\]
with the two orientation transports inserted before the comparison.
\item $H_{\mathrm{fact}}$ is a chain homotopy for every disjoint
DWR/Ran product square satisfying
\[
 dH_{\mathrm{fact}}=o_{\mathrm{fact}}.
\]
\item $Q$ is a quasi-isomorphism datum: on every $0$-simplex it
gives a continuous inverse up to homotopy for $\theta_i$, and these
homotopies preserve the $\hbar$-adic, charge/HN-adic,
equivariant-localised, shift, Tate-twist, and orientation conventions.
\end{enumerate}
\end{definition}

\begin{theorem}[Primitive criterion for
\texorpdfstring{$\Theta_{\hCS\to\Hall}^{\mathrm{or}}$}{Theta hCS to Hall}]
\label{thm:hcs-hall-primitive-criterion}
For fixed $\mathfrak U$, local critical-CoHA normalisation, compact-support
convention, and completions, a chartwise family
$\theta^{(0)}=\{\theta_i\}$ extends to a global orientation-preserving
morphism
\[
 \Theta_{\hCS\to\Hall}^{\mathrm{or}}\colon
 \Obs_{\hCS}^{q}(-,\mathfrak g)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(-)
\]
in $\mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}(X)$ with
quasi-isomorphic vertex maps if and only if primitive data
$\mathfrak P_{\Theta}^{\mathrm{or}}$ exists. In that case the underlying
Maurer--Cartan element is
$\theta=\theta^{(0)}+\eta_{\mathrm{MC}}$, and the four discrete/coherent
structures are transported by
$\lambda_{\mathrm{or}},\eta_{\mathrm{gr}},H_{\mathrm{TS}},H_{\mathrm{fact}}$.

A Hall-side orientation trivialisation, a local critical-CoHA
normalisation, a Schiffmann--Vasserot positive-half identification, or a
finite torus-fixed $\C^3$ chart map supplies only a proper subpackage.
None implies the compact oriented comparison unless the chartwise maps,
all five primitive entries, and the quasi-isomorphism datum $Q$ are
present.
\end{theorem}

\begin{proof}
Assume first that the primitive data are fixed. The equation for
$\eta_{\mathrm{MC}}$ makes $\theta$ a Maurer--Cartan element of the total
Cech/Ran convolution dg Lie algebra
$\mathfrak M_{\hCS,\Hall}(\mathfrak U)$, hence it gives a continuous
natural transformation on the whole nerve after forgetting the Hall
orientation, grading, Thom--Sebastiani, and factorisation structures.
The cochain $\lambda_{\mathrm{or}}$ kills the determinant-line
square-root mismatch; $\eta_{\mathrm{gr}}$ kills the shift and Tate-twist
mismatch; $H_{\mathrm{TS}}$ identifies the two Hall parenthesisations
with the hCS iterated product; and $H_{\mathrm{fact}}$ identifies
disjoint-union factorisation products. The datum $Q$ gives the
vertexwise quasi-isomorphism. Thus all hypotheses of
Theorem~\ref{thm:hcs-hall-descent-criterion} hold, and the global
oriented morphism exists.

Conversely, a global morphism in
$\mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}(X)$ restricts to vertex
maps $\theta_i$ and to a degree-zero total Cech/Ran cochain $\theta$
satisfying the Maurer--Cartan equation. Put
$\eta_{\mathrm{MC}}=\theta-\theta^{(0)}$. Preservation of the
orientation local systems, shifts, Tate twists, Thom--Sebastiani
isomorphisms, and disjoint-union products supplies the four primitives
with the displayed coboundary equations. The vertexwise
quasi-isomorphism supplies $Q$. These seven entries form
$\mathfrak P_{\Theta}^{\mathrm{or}}$.

The final assertion is the contrapositive of the same equivalence.
Each listed partial datum omits at least one obstruction component in
Definition~\ref{def:hcs-hall-descent-obstruction}; the omitted component
can be non-zero on a nontrivial DWR/Ran nerve, so no compact comparison
is obtained from the partial datum alone.
\end{proof}

\begin{remark}[One-chart normalisation is not the global comparison]
\label{rem:hcs-hall-one-chart-normalisation}
On a single contractible critical chart, after an orientation square
root, the functions $s(U,\mathbf d)$ and $t(U,\mathbf d)$, the Hall
Thom--Sebastiani associator, and the completion convention have been
chosen as part of the local datum, the four coherent obstructions
$o_{\mathrm{or}}$, $o_{\mathrm{gr}}$, $o_{\mathrm{TS}}$, and
$o_{\mathrm{fact}}$ vanish tautologically for that normalised chart.
This does not construct
$\Theta_{\hCS\to\Hall}^{\mathrm{or}}$. The Maurer--Cartan component
$o_{\mathrm{MC}}$ is defined only after a degree-zero chartwise chain map
$\theta_U$ is supplied, and it vanishes only when that map is a
chain-level multiplicative comparison. On a genuine DWR cover, the
orientation, grading/Tate, Thom--Sebastiani, and factorisation classes
are overlap data and must be killed on the whole nerve.
\end{remark}

\begin{remark}[Comparison objects and arrows]
\label{rem:cy3-chain-level-comparison-diagram}
The bridge contains five typed passages. Costello--Gwilliam and
Costello--Li supply $\Obs_{\hCS}^{q}$ after the quartic anomaly is
cancelled. Kontsevich--Soibelman and Schiffmann--Vasserot identify the
local Hall algebra of $\C^3$ with $Y^+$. The passage from $Y^+$ to
$\cW_{1+\infty}$ goes through the Drinfeld double and then a
Fock/evaluation representation. The compact arrow
$\PhiFA_3(\cC)\dashrightarrow\CoHA_{\mathrm{crit}}(X)$ is precisely
Problem~\ref{op:cy3-hcs-hall-comparison}; a supplied
$\Theta_{\hCS\to\Hall}^{\mathrm{or}}$ is a datum in
$\mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}$ by
Theorem~\ref{thm:oriented-hcs-hall-comparison-from-dwr-datum}.
Toric chart gluing is the separate hypothesis of
Proposition~\ref{prop:cy3-local-to-toric-descent-package}. The finite
chain data for the compact comparison are the primitive data of
Theorem~\ref{thm:hcs-hall-primitive-criterion}.
\end{remark}

\section{The seven rigidifications}
\label{sec:cy3-seven-rigidifications}

The bridge becomes a theorem only after seven independent
rigidifications are supplied. Four are formal once their data are
named: ordered chiral Hochschild chains, scale-compatible quantum hCS,
DWR descent, and bialgebra transport. Three are additional geometric data:
the global oriented hCS--Hall comparison, non-formal CY$_3$
strictification outside the verified locus, and the physical comparison
functor from BPS states to the chiral/BKM side.

\begin{definition}[The complete CY$_3$ bridge datum]
\label{def:complete-cy3-bridge-datum}
A complete CY$_3$ bridge datum on a smooth CY$_3$ $X$ with holomorphic
volume form $\Omega_X$ is a septuple
\[
 \mathfrak D_X
 =
 \bigl(
 \mathcal Q_{\hCS},\,
 \mathcal B^{\mathrm{ord,ch}}_{E_3},\,
 \mathcal S_3,\,
 \Theta_{\hCS\to\Hall}^{\mathrm{or}},\,
 \mathcal B_{\Delta},\,
 \mathcal P_{\mathrm{BPS}},\,
 \mathcal C_{\mathrm{coh}}
 \bigr)
\]
with the following entries.
\begin{enumerate}
\item $\mathcal Q_{\hCS}$ is a scale-dependent Costello--Gwilliam
quantisation package for hCS: propagator, effective interactions
$I[L]$, RG flow, QME, compact-support convention, and anomaly
trivialisation.
\item $\mathcal B^{\mathrm{ord,ch}}_{E_3}$ is the completed ordered
vertex-bar model of Definition~\ref{def:ordered-chiral-e3-bar}; it
retains repeated inputs and the full partial-diagonal residue calculus.
\item $\mathcal S_3$ is a non-formal CY$_3$ strictification datum:
negative-cyclic CY class, $E_3$ formality point, chain-level
$S^3$-framing witness, and Costello--Li holomorphic witness.
\item $\Theta_{\hCS\to\Hall}^{\mathrm{or}}$ is a degree-zero
Maurer--Cartan element of
$\mathfrak M_{\hCS,\Hall}(\mathfrak U)$ whose $0$-simplex maps are
quasi-isomorphisms and whose five obstruction components in
Definition~\ref{def:hcs-hall-descent-obstruction} vanish; equivalently,
it is carried by the primitive data of
Definition~\ref{def:hcs-hall-primitive-witness-package}.
\item $\mathcal B_{\Delta}$ is a Hall--Borcherds bialgebra comparison
datum: coproduct preservation, associator transport, dynamical
$R$-matrix transport, CHL equivariance, and denominator normalisation.
\item $\mathcal P_{\mathrm{BPS}}$ is a protected physical comparison
datum from the BPS Hilbert space to the chiral/BKM trace: index
preservation, wall-crossing compatibility, Cardy/black-hole asymptotic
normalisation, and holographic boundary-condition matching.
\item $\mathcal C_{\mathrm{coh}}$ is the coherence datum tying the
previous six pieces together: Beck--Chevalley, Fubini, orientation,
completion, and equivariant-parameter compatibilities on every object
of the DWR nerve.
\end{enumerate}
The datum is called \emph{exact} when all listed coherences commute on
the nose in the completed homotopy category and with respect to the
specified coherent gauge torsor before passage to that category.
\end{definition}

\begin{definition}[Completed ordered chiral \texorpdfstring{$E_3$}{E3} bar]
\label{def:ordered-chiral-e3-bar}
Let $\mathfrak L_{\cC}(P)$ be the Dolbeault--jet dg Lie algebra of
Definition~\ref{def:cy3-many-variable-chiral-ce}. Its completed ordered
chiral $E_3$ bar complex is
\[
 B^{\mathrm{ord,ch}}_{E_3}(\mathfrak L_{\cC})(P)
 :=
 \prod_{n\geq0}
 C_\bullet(\FM_3(n))
 \widehat\otimes_{\Sigma_n}
 \mathfrak L_{\cC}(P)^{\widehat\otimes_\pi n}[n],
\]
where $\FM_3(n)$ is the Fulton--MacPherson compactification of
configurations of $n$ ordered points in a real $3$-ball and
$\widehat\otimes_\pi$ is the completed projective tensor product. The
differential is
\[
 d_{\mathrm{ord}}
 =
 d_{\FM}
 +d_{\dbar}
 +d_{\mathfrak l}
 +d_{\mathrm{coll}}
 +d_{\mathrm{br}},
\]
where $d_{\mathrm{coll}}$ is the boundary differential summing over
partial collisions in $\FM_3(n)$ and $d_{\mathrm{br}}$ applies the Lie
bracket of $\mathfrak L_{\cC}$ on the colliding ordered block. The
chiral cochains are the strong continuous dual
\[
 C^{\bullet}_{\mathrm{vHoch},E_3}(\cC;P)
 :=
 \bigl(B^{\mathrm{ord,ch}}_{E_3}(\mathfrak L_{\cC})(P)\bigr)^\vee_b .
\]
The adjective \emph{ordered} is load-bearing: tensors
$x\widehat\otimes x\widehat\otimes y$ and
$x\widehat\otimes y\widehat\otimes x$ are distinct before the
$\Sigma_n$-coinvariant operation is applied with its configuration-chain
coefficient. Repeated fields therefore survive; they are not killed by
exterior antisymmetrisation.
\end{definition}

\begin{proposition}[Exterior CE is the shadow, not the controller]
\label{prop:exterior-ce-shadow-not-controller}
There is a natural quotient map
\[
 q_{\wedge}\colon
 B^{\mathrm{ord,ch}}_{E_3}(\mathfrak L_{\cC})(P)
 \longrightarrow
 \mathrm{CE}^{\mathrm{ch},E_3}_*(\mathfrak L_{\cC})(P)
\]
which sends the ordered tensor coalgebra to its alternating
Chevalley--Eilenberg quotient. This map is a quasi-isomorphism only
after imposing the locally constant abelian shadow in which
configuration-chain boundary operations, repeated-input vertex
operations, and partial-diagonal residues have been discarded. In
particular, a finite exterior CE computation cannot prove a class
$\mathbf M$ repeated-input obstruction or a full chiral Hochschild
vanishing theorem.
\end{proposition}

\begin{proof}
The ordinary CE coalgebra is the cofree cocommutative coalgebra on
$\mathfrak L_{\cC}[1]$ with the differential induced by the dg Lie
bracket. It is obtained from the ordered tensor coalgebra by imposing
shuffle antisymmetry and then forgetting the configuration-chain
coefficient. This gives the displayed quotient. The kernel contains
every ordered diagonal tensor whose image in the exterior algebra has a
repeated input, for example $x\widehat\otimes x$ and
$x\widehat\otimes x\widehat\otimes y$ in the relevant signs. Those
classes are precisely the vertex-bar classes detected by ordered OPE
collisions and by the boundary strata of $\FM_3(n)$. Passing to the
locally constant abelian shadow makes $d_{\mathrm{coll}}$ and
$d_{\mathrm{br}}$ invisible and kills the repeated-input sector, so the
quotient becomes the familiar finite CE model. Before that quotient,
the repeated-input sector is part of the deformation complex, hence an
exterior CE calculation can be a consistency check but not the controller
of the chiral deformation problem. For the Virasoro generator $T$, the
ordered words $(T,T,T)$ and $(T,T,T,T)$ survive and carry the existing
$L_\infty$ coefficients $l_3(T,T,T)=-2T$ and
$l_4(T,T,T,T)=\frac{40}{27}T$, while their exterior CE projections
vanish.
\end{proof}

\begin{definition}[CY$_3$ strictification obstruction tower]
\label{def:cy3-strictification-obstruction-tower}
Let $(\cC,\mu_\bullet,\sigma)$ be a smooth proper cyclic $\Ainf$
category with a negative-cyclic CY$_3$ class
$[\sigma]\in HC^{-}_3(\cC)$ and let $F\in\mathrm{Form}_3(\Q)$ be an
$E_3$ formality point. The strictification tower
\[
 \mathfrak{Obs}_{\mathrm{str}}(\cC,F,\sigma)
 =
 \bigl(
 \omega_{\mathrm{cyc}},\,
 \omega_{S^3},\,
 \omega_{\Ainf},\,
 \omega_{\mathrm{CL}},\,
 \omega_{\mathrm{des}}
 \bigr)
\]
has components:
\begin{enumerate}
\item $\omega_{\mathrm{cyc}}$, the failure of the Hochschild trace
$HH_3(\cC)\to k$ to lift to the chosen negative-cyclic class;
\item $\omega_{S^3}$, the obstruction to extending the framed
$E_2$ / $\BV_\infty$ structure to the selected $E_3$ formality point;
\item $\omega_{\Ainf}$, the ordered-bar commutator of the non-formal
higher multiplications $\mu_{\geq3}$ with the second bar differential;
\item $\omega_{\mathrm{CL}}$, the Costello--Li holomorphic anomaly
class in the quartic invariant-polynomial slot;
\item $\omega_{\mathrm{des}}$, the DWR descent obstruction for the
resulting holomorphic factorisation algebra.
\end{enumerate}
The tower lives in the product of the corresponding cohomology groups
of the ordered chiral Hochschild complex, the graph/formality complex,
the Costello--Li local Lie-algebra cohomology complex, and the DWR
Cech complex.
\end{definition}

\begin{proposition}[Raw bar contraction and the two closure routes]
\label{prop:cy3-raw-bterm-two-closure-routes}
Let $B_{\mathrm{term}}^{(2)}$ denote the raw terminal-slot pair
contraction on the reduced bar complex of a cyclic $\Ainf$ model. In a
non-formal CY$_3$ cyclic model with
$m_3(a,a,a)=\alpha b$ and $\alpha\neq0$, one has
\[
 [m_3,B_{\mathrm{term}}^{(2)}]\,[a|a|a|a|b]
 =
 2\alpha [b]\neq0 .
\]
Thus raw cancellation of
$[m_k,B_{\mathrm{term}}^{(2)}]$ is false even on the basic strict
witness. The operator $B_{\mathrm{TCFT}}^{(2)}$ appearing in Costello's

\begin{proof}

\begin{proposition}[Two-loop YBE normalisation and the sunset obstruction]
\label{prop:k3-hcs-two-loop-yang-normalisation}
For the rational defect $R$-matrix of $6$d hCS on $K3\times E$, let
\[
 a=12+\frac{h^\vee}{2},\qquad
 A_2=\Bigl(12+\frac{h^\vee}{2}\Bigr)^2-\frac{(h^\vee)^2}{12}.

\[
 \frac{a}{uv(u-v)}(P_{12}P_{23}-P_{23}P_{12}),
\]
so no order-$\hbar^4$ two-loop counterterm can cancel the one-loop fish term
before the one-loop obstruction is removed. The one-loop normalisation is the
Yang-coupling normalisation
\[
 R_{\mathrm{tree}}(u;\hbar)+R_{\mathrm{fish}}(u;\hbar)+CT_1(u;\hbar)
 =
 R_{\mathrm{tree}}(u;\hbar+a\hbar^2).
\]
After this normalisation, the raw sunset tensor still has a non-zero
linearised order-$\hbar^5$ obstruction. The constructed algebraic
two-loop YBE counterterm is therefore the projection of the sunset
tensor to the tangent line of the rational Yang family:
\begin{equation}
\label{eq:k3-hcs-two-loop-yang-counterterm}
 CT_{2,ij}^{\mathrm{YBE}}
 =
 b\,\frac{P_{ij}-1}{u_{ij}}
 -
 Q^{\mathrm{sunset,raw}}_{2,ij},
 \qquad b=A_2\ \textup{by default}.
\end{equation}
Equivalently, the renormalised order-$\hbar^4$ contribution is
$b(P_{ij}-1)/u_{ij}$, which is an infinitesimal change of the Yang
coupling and hence has zero linearised Yang--Baxter obstruction. For
$\mathfrak{sl}_2$ the executable exact oracle gives $A_2=506/3$ and
verifies that the raw $\hbar^5$ obstruction is non-zero while the
renormalised obstruction is zero in the group algebra of $S_3$.
\end{proposition}

\begin{proof}
The rational Yang solution has the form
$R(u;\hbar)=1+\hbar(P-1)/u$. Its tangent direction at any order in the
formal coupling is therefore $(P-1)/u$. Expanding the Yang--Baxter
equation to the first order in an inserted one-loop correction $q_{ij}$
gives the linear map
\[
 (q_{12},q_{13},q_{23})
 \longmapsto
 q_{12}r_{13}+q_{12}r_{23}+r_{12}q_{13}+q_{13}r_{23}
 +r_{12}q_{23}+r_{13}q_{23}
 -(\textup{the reversed product}),
\]
where $r_{ij}=(P_{ij}-1)/u_{ij}$. Substituting
$q_{ij}=aP_{ij}/u_{ij}^2$ reduces exactly to the displayed commutator
coefficient, so the unrenormalised one-loop input is not tangent to the Yang
family. Substituting $q_{ij}=a(P_{ij}-1)/u_{ij}$ gives zero because it
is the derivative of the exact Yang solution in the coupling parameter.
The same linearised calculation at order $\hbar^5$ applies to the
two-loop slot: the raw sunset tensor
$Q^{\mathrm{sunset,raw}}_{2,ij}$ has non-zero image, while
$b(P_{ij}-1)/u_{ij}$ has zero image. Hence
\eqref{eq:k3-hcs-two-loop-yang-counterterm} is the unique minimal
YBE-normalising correction modulo the tangent coefficient $b$ and
central scalar terms. The exact arithmetic is the
\(\mathfrak{sl}_2\) coefficient calculation together with the vanishing
of the renormalised linearised obstruction.
\end{proof}

 \mathrm{Map}_{\mathrm{dgLie}}\bigl(\mathrm{Obs}^{\BV}_{\hCS}(U),
 \mathrm{Obs}_{\Hall}(U)\bigr).
\]
A chartwise formula is not descent. The obstruction tuple is
\[
 (\mathrm{ob}_{\MC},\mathrm{ob}_{\mathrm{or}},\mathrm{ob}_{\mathrm{gr}},
 \mathrm{ob}_{\mathrm{TS}},\mathrm{ob}_{\mathrm{fac}})
 \in
 \prod_{\alpha\in\{\MC,\mathrm{or},\mathrm{gr},\mathrm{TS},\mathrm{fac}\}}
 H^2(\mathfrak g_{\hCS,\Hall}^{\mathrm{Ran}})_\alpha .
\]
The bridge is global exactly when chartwise quasi-isomorphisms are
equipped with nullhomotopies for the Maurer--Cartan, orientation,
grading, Thom--Sebastiani, and factorisation components.

\item The quintic strictification is controlled by the compact analytic
curved transfer tower
\[
 \mathfrak g_{X_5}^{\mathrm{curv}}
 =
 \mathrm{Def}_{\mathrm{an}}\bigl(\Perf(X_5),Y_3,S^3,
 \mathrm{OPE},Z^{\mathrm{der}}_{\mathrm{ch}}\bigr).
\]
The first obstruction is the non-formal Yukawa component
$Y_3=5$ before a compact analytic curved $L_\infty$ primitive is chosen.
The finite normal form records how the component must be absorbed; the
global quintic theorem needs the analytic transfer, the actual
$S^3$-framing, OPE-completion, and derived-centre comparison.

\item The Hall--Drinfeld/Super-Yangian/BKM identification is controlled
by the completed double-comparison complex
\[
 \mathfrak g_{\mathrm{HDYB}}
 =
 \mathrm{Def}\bigl(\mathrm{PBW},\Delta,\mathcal I_{\mathrm{Borcherds}},
 Z,\widehat{\phantom A},\langle-,-\rangle_{\Hall}\bigr).
\]
A denominator identity or a candidate presentation does not determine a
completed Hopf algebra. The first obstruction is the associated-graded
PBW mismatch; the remaining components are coproduct, Borcherds--Serre
ideal, centre, completion, Hall-pairing, associator, and dynamical
$R$-matrix compatibility. The completed isomorphism exists exactly
when all listed classes vanish by specified primitives.

\item The pure mathematical holographic bridge is controlled by the
symmetric-monoidal protected-trace coherence complex
\[
 \mathfrak g_{\mathrm{hol}}
 =
 \mathrm{Def}_{\otimes}\bigl(\Tr_{\mathrm{BPS}}\Rightarrow
 \Tr_{\mathrm{ch}/\mathrm{BKM}}\bigr).
\]
The physical bridge and the finite protected-trace normal form prove no
pure functor by themselves. The first obstruction is the product,
coproduct, orientation, wall-crossing, and Drinfeld-centre
half-braiding coherence class for protected traces. The required datum
is a product-, coproduct-, and orientation-coherent functor from
oriented BPS factorisation homology to the completed chiral/BKM trace
category, preserving Hall products, Hall coproducts, wall-crossing, and
the centre half-braiding.

\item The two-loop 6d hCS counterterm is controlled by the local
functional complex
\[
 \mathfrak g_{\hCS}^{(2)}
 =
 \mathrm{Loc}_{\hCS}^{(2)} \xrightarrow{\;\rho_{\mathrm{YBE}}\;}
 T_{\mathrm{YBE}}^{\hbar^5}.
\]
The algebraic Yang-normalised oracle proves which $CT_2$ kills the
$\hbar^5$ Yang--Baxter tangent. It does not derive that same local
functional from the 6d hCS factorisation-RG integral. The first
obstruction is
\[
 [\,\rho_{\mathrm{YBE}}(CT_2^{\mathrm{Feyn/RG}})
 - CT_2^{\mathrm{Yang}}\,]\in H^1(\mathfrak g_{\hCS}^{(2)}).
\]
The sunset-only counterterm leaves this class non-zero. The global theorem
requires a Feynman/RG derivation of the same $CT_2^{\mathrm{Yang}}$
already forced by the algebraic oracle.
\end{enumerate}
\end{proposition}

\begin{proof}
Each line is the deformation complex of the structure named in the
corresponding theorem. A morphism theorem for $\Phi_3$ is a
Maurer--Cartan point in the witnessed-kernel deformation problem, so the
first non-trivial associativity failure lives in $H^2$ of
$\mathfrak g_{\Phi_3}$. A double theorem for $G(K3\times E)$ is a
completed Manin pair; degeneracy of the pairing and discontinuity of the
centre are precisely the first two ways a positive-half construction
fails to be a double. A DWR/Ran comparison is a map of factorisation
objects on every simplex; the five named compatibilities are exactly
the five cochains whose coboundaries must trivialise the mapping
Maurer--Cartan defect.

For $X_5$, the compact analytic curved transfer is the deformation
problem whose lowest non-formal input is the Yukawa component
$Y_3=5$; a finite algebraic witness gives the target normal form, while
analytic compactness and framing give the missing primitives. A
Hall--Drinfeld/Super-Yangian/BKM isomorphism is an isomorphism of
completed Hopf objects, hence PBW, coproduct, Serre, centre,
completion, pairing, associator, and dynamical $R$-matrix coherence are
not optional decorations but the components of the comparison complex.
A holographic theorem without physical hypothesis is a
symmetric-monoidal protected-trace functor into a Drinfeld-centre
boundary category, so product, coproduct, orientation, wall-crossing,
and half-braiding are the first coherence obstructions.
At two loops, the executable Yang oracle computes the target tangent in
$T_{\mathrm{YBE}}^{\hbar^5}$; deriving it from 6d hCS is the statement that the
Feynman/RG local functional maps to the same tangent. These
identifications are functorial deformation theory, so killing all
listed first classes by explicit primitives is exactly the passage from
normal form to global theorem in
Proposition~\ref{prop:cy3-seven-bridge-normal-form}.
\end{proof}

\begin{corollary}[Primitive criterion for compact CY\texorpdfstring{$_3$}{3} bridge promotions]
\label{cor:cy3-primitive-criterion}
Let a proposed compact CY$_3$ bridge theorem factor through one of the
seven promotions in Proposition~\ref{prop:cy3-seven-first-obstruction-complexes}.
Then the theorem is unconditional only after it supplies, in the named
deformation complex, explicit primitives for every first obstruction
class used by that factorisation and proves compatibility of those
primitives with the relevant completion and orientation data. A
chartwise quasi-isomorphism, a denominator identity, a positive-half
Hall presentation, a finite normal form, or a protected-index equality
is therefore sufficient only for the corresponding local or numerical
layer; it does not imply the compact bridge unless the primitive data
are also constructed.
\end{corollary}

\begin{proof}
Each promotion in Proposition~\ref{prop:cy3-seven-first-obstruction-complexes}
is the Maurer--Cartan extension problem for the structure being
promoted: functorial kernels, completed Manin pairs, DWR/Ran
hCS--Hall comparison, compact curved transfer, completed Hopf
comparison, protected traces, or the two-loop hCS counterterm. In a
complete filtered deformation complex, passing from a local solution to
a global solution is equivalent to killing the obstruction cocycles by
chosen nullhomotopies and then checking the discrete structures that do
not live in the linear complex: orientation, grading, topology, and
completion. The listed local data determine only the degree-zero or
associated-graded part of the Maurer--Cartan problem. They become a
compact bridge theorem precisely when the obstruction classes displayed
in the proposition vanish by specified primitives compatible with those
discrete structures.
\end{proof}

\begin{theorem}[Universal primitive envelope of the seven CY$_3$ bridge obstructions]
\label{thm:cy3-universal-primitive-envelope}
For each gate $j$ in
Proposition~\ref{prop:cy3-seven-first-obstruction-complexes}, let
$\mathfrak g_j$ be its controlling dg Lie complex and let
$o_{j,a}\in Z^{n_{j,a}}(\mathfrak g_j)$ be the first obstruction
cocycles indexed by the required primitive labels $a$. Define the
completed free primitive extension
\[
 \widetilde{\mathfrak g}_j
 :=
 \mathfrak g_j
 \widehat{\oplus}
 \prod_a \mathbb C\cdot h_{j,a}[\,n_{j,a}-1\,],
 \qquad
 d h_{j,a}=o_{j,a}.
\]
Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item every first obstruction class of gate $j$ vanishes in
$H^\bullet(\widetilde{\mathfrak g}_j)$;
\item the construction is universal: if
$\mathfrak g_j\to \mathfrak h$ is any completed dg extension in which
each $o_{j,a}$ is a boundary $d\eta_{j,a}$, then there is a continuous
dg map
\[
 \widetilde{\mathfrak g}_j\longrightarrow\mathfrak h,\qquad
 h_{j,a}\longmapsto \eta_{j,a},
\]
unique in the homotopy fibre after the primitives
$\{\eta_{j,a}\}_a$ are fixed; replacing
$\eta_{j,a}$ by $\eta_{j,a}+d\xi_{j,a}$ composes the lift with the
explicit gauge homotopy determined by $\xi_{j,a}$;
\item the product
\[
 \widetilde{\mathfrak G}_{CY_3}
 :=
 \prod_{j\in\{\Phi_3,CY\textup{-}C,\hCS\textup{-}\Hall,
 X_5,\mathrm{HDYB},\mathrm{hol},CT_2\}}
 \widetilde{\mathfrak g}_j
\]
is the first-obstruction primitive envelope of the seven requested bridge
problems. It becomes an unconditional compact CY$_3$ theorem only when
$\widetilde{\mathfrak G}_{CY_3}$ is realised by the named geometric or
analytic objects: global witnessed kernels; the completed compact Hall
double; DWR/Ran hCS--Hall quasi-isomorphisms and five nullhomotopies;
the compact quintic curved transfer; the completed
Hall--Drinfeld/Super-Yangian/BKM isomorphism with associator and
dynamical $R$-matrix coherence; the protected trace functor with
coproduct and Drinfeld-centre half-braiding coherence; and the 6d hCS
Feynman/RG derivation of
$CT_2^{\mathrm{Yang}}$, with the higher obstruction and compatibility
towers checked in the same completed topology.
\end{enumerate}
Thus the first obstruction layer is resolved formally by adjoining the
displayed primitives. The compact theorem is a further geometric
realisation statement inside the corresponding analytic/completed
categories; it is not a consequence of adjoining formal generators
alone.
\end{theorem}

\begin{proof}
Because $o_{j,a}$ is a cocycle, the assignment $d h_{j,a}=o_{j,a}$ and
the old differential on $\mathfrak g_j$ define a square-zero
differential on the extension:
\[
 d^2h_{j,a}=d o_{j,a}=0.
\]
The class of $o_{j,a}$ is therefore zero in the extended complex, since
it is the boundary of the explicitly adjoined generator $h_{j,a}$.
Completion is harmless because the product topology is used and the
differential is continuous on each factor.

For universality, let $\mathfrak h$ be a completed dg extension of
$\mathfrak g_j$ in which $o_{j,a}=d\eta_{j,a}$. The identity map on
$\mathfrak g_j$ and the assignment $h_{j,a}\mapsto\eta_{j,a}$ commute
with differentials, hence define a continuous dg map from the free
completed extension. If another primitive is chosen, it differs from
$\eta_{j,a}$ by a closed element; changing it by a boundary gives a
chain-homotopic map. The ambiguity is therefore controlled by the next
cohomology group of the deformation complex. It is contractible only
after the corresponding higher negative groups vanish, a condition not
supplied by the formal extension itself.

Taking the product over the seven gates preserves the same universal
property componentwise. Proposition~\ref{prop:cy3-seven-first-obstruction-complexes}
identifies these components with the seven requested bridge
promotions; Corollary~\ref{cor:cy3-primitive-criterion} says that an
actual compact theorem is obtained exactly when the formal primitives
come from the named orientation-, completion-, and analytically
continuous geometric data and their higher compatibilities. This proves
the first-obstruction envelope and the stated criterion for genuine
global closure.
\end{proof}

\section{Chain-level assembly of the left end: braces, BV, envelope}
\label{sec:cy3-chain-level-bridge-envelope-stage-one}

The comparison diagram isolates
$\PhiFA_3(\cC)\dashrightarrow\CoHA_{\mathrm{crit}}(X)$ as the open
arrow. Its left end, $\PhiFA_3(\cC)$ itself, is an explicit composite of
three chain-level steps.
Deligne--Tamarkin supplies a chain-level $\Etwo$-action on the
Hochschild cochain complex via braces and cup-products;
Tradler--Menichi--Ginzburg supply the framed $\Etwo$ / BV layer from the
cyclic pairing of degree $-3$; the $\Ethree$ input is the additional
Stage-$1$ lift fixed by a formality point and by the Costello--Li
holomorphic witness on the verified CY$_3$ locus. Costello--Gwilliam's
factorisation envelope sends that chosen $\Ethree$ lift to a
factorisation algebra, whose Costello--Li holomorphic refinement lands
in $\EdHolFA(X)$. The right end of the bridge, the CoHA, is not produced
by this construction: it is the image of $\Theta_{\hCS\to\Hall}$, a
separate piece of data whose existence is
Problem~\ref{op:cy3-hcs-hall-comparison}. The present section records
the left end alone, at the level of proof.

\begin{proposition}[Deligne--Tamarkin braces on $\HH^\bullet(\cC)$]
\label{prop:cy3-braces-action}
For a smooth proper cyclic $\Ainf$-category $\cC$ of any CY dimension,
the Hochschild cochain complex $\HH^\bullet(\cC)$ carries a chain-level
operadic action
\[
\mathrm{Brace}_\bullet\;:\;C_\bullet(\Conf_2;\Q)\;\longrightarrow\;\mathrm{End}(\HH^\bullet(\cC)),
\qquad \Conf_2:=\text{little $2$-discs},
\]
from the singular chains on the little $2$-discs operad. The primary
operations are the Getzler--Voronov braces
\[
f\{g_1,\ldots,g_r\}(a_1,\ldots,a_{m+\sum n_i - r})
\;=\;\sum_{i_1<\cdots<i_r}\pm
f\bigl(a_1,\ldots,g_1(a_{i_1},\ldots),\ldots,g_r(a_{i_r},\ldots),\ldots\bigr),
\]
the May--Steenrod $\cup_i$-products, the cup product $\cup$, and the
Gerstenhaber bracket $\{-,-\}_\Ger$. On cohomology the action induces
the classical Gerstenhaber structure. The chain-level action is
canonical up to a torsor under $\GRT_1(\Q)$ of choices of Drinfeld
associator.
\end{proposition}

\begin{proof}
Kontsevich--Soibelman~$2000$ (\emph{Algebra and Representation Theory}
$3$, \S~$5$ Theorem~$5.1$) present the Kontsevich--Soibelman operad
$\cK=\mathrm{Br}\otimes\mathrm{Sur}$ acting on $\HH^\bullet(\cC)$ and prove it is
quasi-isomorphic to $C_\bullet(\Conf_2;\Q)$. McClure--Smith~$2003$
(\emph{Topology} $42$, Theorem~$1.1$) prove the brace operad
$\mathrm{Br}\hookrightarrow C_\bullet(\Conf_2;\Q)$ is a quasi-isomorphism of
operads. Tamarkin~$1998$ (\texttt{arXiv:math/9803025}; reproved
Tamarkin~$2007$, \emph{Lett.\ Math.\ Phys.}~$81$) identifies the space
of such $\Etwo$-structures with a torsor under $\GRT_1(\Q)$.
Kontsevich~$1999$ (\emph{Lett.\ Math.\ Phys.}~$48$, \S~$4$) singles
out the Drinfeld-associator point of this torsor.
\end{proof}

\begin{proposition}[Tradler--Menichi--Ginzburg BV operator from cyclicity]
\label{prop:cy3-tmg-bv}
For a smooth proper cyclic $\Ainf$-category $\cC$ of CY dimension $d$,
the cyclic pairing $\langle-,-\rangle:\cC\otimes\cC\to k[-d]$ induces
a chain-level BV operator
\[
\Delta\;:\;\HH^\bullet(\cC)\;\longrightarrow\;\HH^{\bullet-1}(\cC),
\]
of cohomological degree $-1$, with $\Delta^2=0$ on cohomology,
and gives a chain-level $\BV_\infty$ structure whose homology satisfies
the BV identity
\[
\{a,b\}_\Ger\;=\;\Delta(a\cup b)\;-\;\Delta(a)\cup b
\;-\;(-1)^{|a|}a\cup\Delta(b)
\]
after passing to cohomology. Thus $(\cup,\{-,-\}_\Ger,\Delta)$ is the
homological shadow of a framed little-$2$-discs action; at chain level
the higher homotopies are part of the data and cannot be discarded.
\end{proposition}

\begin{proof}
Tradler--Zeinalian~$2006$ (\emph{J.\ Pure Appl.\ Algebra} $204$,
Theorem~$3.1$); Menichi~$2004$ (\emph{K-Theory} $32$,
Proposition~$3.4$); Ginzburg~$2006$ (\texttt{arXiv:math/0612139},
Theorem~$3.4.3$). The defining formula on a cochain
$f:(s\bar\cC)^{\otimes n}\to\cC$ is the cyclic-rotation average
\[
\Delta f(a_1,\ldots,a_{n-1})\;=\;\sum_{i=1}^{n}(-1)^{\epsilon_i}
\bigl\langle a_i,f(a_{i+1},\ldots,a_n,a_1,\ldots,a_{i-1},\mathbf 1_\cC)\bigr\rangle^\vee,
\]
with $\langle-,-\rangle^\vee$ the pairing composed with the duality
$\cC^\vee\simeq\cC[d]$; cyclic invariance of $\mu_n$ is exactly what
forces the cyclic Deligne homotopies whose homology is the displayed BV
operator and bracket identity. This $\Delta$ is the cochain BV operator
dual to cyclic rotation; it is not the Connes $B$ operator on chains and
not the hCS field-theoretic BV bracket $\{-,-\}_{\BV}$. The Getzler identification
$H_\bullet(\Conf_2^{\mathrm{fr}};\Q)\simeq\BV$ is
Getzler~$1994$ (\emph{Comm.\ Math.\ Phys.}~$159$); the chain-level
statement is the framed cyclic Deligne action, not a strict equality
with the homology operad.
\end{proof}

\begin{proposition}[The \texorpdfstring{$E_3$}{E3} lift is Stage-\texorpdfstring{$1$}{1} data]
\label{prop:cy3-e3-structure-hochschild}
\noindent\emph{Hypotheses.} Work on the constructed Stage-$1$ CY$_3$
locus, after fixing the formality point, CY$_3$ chain-level framing, and
Costello--Li holomorphic witness.
For a smooth proper cyclic $\Ainf$-category $\cC$ of CY dimension $3$,
Propositions~\ref{prop:cy3-braces-action} and~\ref{prop:cy3-tmg-bv}
give a framed $\Etwo$ / $\BV_\infty$ structure on
$\HH^\bullet(\cC)$. They do not by themselves produce an $\Ethree$
structure. An $\Ethree$-algebra structure on the same Hochschild

$F\in\mathrm{Form}_3(\Q)$ together with the CY$_3$ chain-level

\]
The ambiguity is the stated $\GRT_1(\Q)$-torsor; the BV circle action is
a compatibility constraint on the lift, not a construction of the third

\end{proposition}

\begin{proof}
Dunn--Lurie additivity (Lurie, \emph{Higher Algebra}, Theorem
$5.1.2.2$) says that an $\Ethree$-algebra is equivalently an $\Eone$
algebra object in $\Etwo$-algebras. This is a recognition theorem, not
a source of the missing $\Eone$-object. The cyclic Deligne theorem
supplies the framed $\Etwo$ layer of
Proposition~\ref{prop:cy3-tmg-bv}; the third direction is the
configuration-space/formality datum named in
Proposition~\ref{prop:phi-fa-three-arrow-assembly}. Willwacher's graph
complex computation identifies the resulting space of rational

Chapter~\ref{ch:cy-to-chiral}. The Costello--Li witness transports

\end{proof}

\begin{definition}[Topological factorisation envelope on $\R^3$]

\qquad A_\pi:=A,

\cU^\FA\;:\;\Ethree\text{-}\mathrm{Alg}(\mathrm{Ch}(k))\;\longrightarrow\;\mathrm{FAct}(\R^3;\mathrm{Ch}(k)).

\end{definition}

finite transition systems as a strict morphism of complete filtered
complexes.

This theorem constructs the compact CY$_3$ Costello--Li/open--closed
source bridge. It does not construct the oriented Hall-valued map
$\Theta_{\hCS\to\Hall}^{\mathrm{or}}$, the Joyce--Kontsevich
orientation square roots, or the DWR/Ran Thom--Sebastiani descent of
Problem~\ref{op:cy3-hcs-hall-comparison}.
\end{theorem}

\begin{proof}
The negative-cyclic class $\sigma$ makes the smooth proper dg category
$\cC$ into a cyclic $\Ainf$ category of CY dimension $3$. Costello's
open--closed TCFT construction sends this cyclic datum, after the
corrected carrier or filtered $\HH^{-2}$ theorem of
Proposition~\ref{prop:cy3-raw-bterm-two-closure-routes} has been fixed,
to a chain-level open--closed field theory whose closed state complex is
the Hochschild chain complex with Connes differential and whose open
state complex is the Hochschild cochain/endomorphism complex of the
boundary algebra. The TCFT carrier in this sentence is
$B_{\mathrm{TCFT}}^{(2)}$, not the raw
$B_{\mathrm{term}}^{(2)}$; the latter has the non-zero strict witness
displayed in Proposition~\ref{prop:cy3-raw-bterm-two-closure-routes}.
The map $\mathsf{OC}_X$ is the TCFT operation represented by the disc with one
closed input in the interior and one open boundary output; the relation
\[
 d_{\mathrm{TCFT}}\circ \mathsf{OC}_X
 =
 \mathsf{OC}_X\circ(b+uB)
\]
is the boundary identity in the moduli chain complex of bordered
Riemann surfaces. This is the source theorem: cyclic
$\Ainf$-categories yield open--closed TCFTs, and the closed-to-open
operation is a chain map before any Hall or DT theory is introduced.

The Stage-$1$ datum identifies the TCFT Hochschild source with the
holomorphic factorisation envelope of
$\HH^\bullet(\cC)_F$ by
Theorem~\ref{thm:cfg-factorization-envelope-stage-one}. Truncating
holomorphic jets at order $N$ replaces
$J^\infty_{\mathrm{hol}}\mathfrak l_{\cC}$ by the finite-rank bundle
$J^N_{\mathrm{hol}}\mathfrak l_{\cC}$; truncating Feynman graph order at
$r$ replaces Costello--Li observables by the finite RG stage
$\Obs_{\hCS}^{q,\leq r}$. The heat-kernel RG equations commute with
restriction of jets and with compact-support extension, so the TCFT
operation transports to the displayed continuous map
$\mathsf{OC}_X^{N,r}$ in the nuclear LF category.

The Stage-$2$ specialisation $\SpCh_{\Sigma_2,C}$ is a factorisation
homology/evaluation functor. Applied to the holomorphic BV
factorisation algebra it gives the finite-stage chiral algebra
$A_C^{N,r}$, and functoriality of factorisation modules gives the
boundary evaluation map $\partial^{N,r}_{X,C}$. The three differentials
match because they are images of the same TCFT boundary:
$(b+uB)$ on the closed Hochschild side, the Costello--Li BV differential
in the holomorphic field theory, and the chiral Hochschild differential
after specialisation. The finite-stage compatibility squares therefore
commute.

The transition maps $J^{N+1}\to J^N$ and RG projections $r+1\to r$ are
surjective on the finite jet/RG systems and continuous in the completed
projective topology. The strict compatibility hypothesis places the
finite-stage chain maps in a complete, exhaustive, separated, strongly
convergent inverse system. In the TCFT route the corrected carrier gives
the obstruction primitive before the limit is taken. In the filtered
$\HH^{-2}_{E_1}$ route the empty total-degree $-2$ first page prevents a
hidden $\Ainf$ obstruction class from entering the limit. Under the
nuclear completion fixed in
Definition~\ref{def:cy3-many-variable-chiral-ce}, these hypotheses make
the inverse limit of finite-stage maps a chain map. Cancelling the
Costello--Li anomaly class is exactly the condition that the completed
renormalised BV differential squares to zero. This gives
$\mathsf{BB}^\wedge_{X,C}$.

None of these constructions mention vanishing cycles, critical CoHA
correspondences, Joyce--Kontsevich orientation data, or
Thom--Sebastiani transport. Hence the theorem heals the compact
open--closed source residual, while leaving the Hall-valued comparison
as the separate obstruction complex of
Definition~\ref{def:hcs-hall-descent-obstruction}.
\end{proof}

\section{The five-way quasi-isomorphism on \texorpdfstring{$\C^3$}{C3}}
\label{sec:cy3-chain-level-bridge-five-way-c3}

On the affine toric chart $X=\C^3$ with its standard $T=(\C^\times)^3$
action and abelian gauge $\mathfrak g=\ghat=\C\cdot T$, the five
mathematical input traditions have a common normal form. They become
quasi-isomorphic chain models only after the local comparison map
$\theta_{\C^3}$ and the vanishing of
$\mathfrak o_{\C^3}^{\mathrm{br}}$ are supplied.
The five faces are:
\begin{itemize}
\item $(\mathrm E) \;=\; \cU^\FA\bigl(\HH^\bullet(\Perf(\C^3))_F\bigr)$,
the topological $\Ethree$-factorisation envelope of Hochschild cochains
of perfect coherent sheaves at a chosen formality point $F$
\textup{(}Definition~\ref{def:cy3-topological-envelope} and
Theorem~\ref{thm:cfg-factorization-envelope-stage-one}\textup{)};
\item $(\mathrm B) \;=\; \Obs_{\hCS}^{q}(\R_t\times\C^2;\ghat)$,
the Costello--Yagi $5$d holomorphic Chern--Simons BV observable
factorisation algebra on the topological--holomorphic presentation of
$\C^3=\R_t\times\C^2$
\textup{(}Definition~\ref{def:cy3-hcs-quantum-observables}\textup{)};
\item $(\mathrm H) \;=\; \CoHA_{\mathrm{crit}}^{\mathrm{or}}(\C^3)$,
the Kontsevich--Soibelman/Schiffmann--Vasserot critical cohomological
Hall algebra with KS/Joyce orientation datum
\textup{(}Proposition~\ref{prop:cy3-c3-bridge-core}\textup{)};
\item $(\mathrm Y) \;=\; Y^+(\ghat)$, the positive half of the affine
Yangian of $\ghat$
\textup{(}Schiffmann--Vasserot~$2013$,~\cite{SV13}\textup{)};
\item $(\mathrm F) \;=\; H^\bullet(\FM_3(-);\HH^\bullet(\Perf(\C^3))_F)$,
the Fulton--MacPherson labelled-configuration-space homology of the
$\Ethree$-algebra $\HH^\bullet(\Perf(\C^3))_F$.
\end{itemize}

\begin{theorem}[Five-way normal form on $\C^3$ under a chart map]
\label{thm:r6-quad-equivalence-c3}
Fix a formality point $F\in\mathrm{Form}_3(\Q)$, and assume a
degree-zero continuous multiplicative chart map
\[
 \theta_{\C^3}\colon
 \Obs_{\hCS}^{q}(\C^3;\ghat)
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(\C^3)
\]
whose Maurer--Cartan obstruction vanishes and whose cohomology reduction
is the Kontsevich--Soibelman/Schiffmann--Vasserot positive-half
identification. Then the Hall-side and envelope normal forms fit into
a pentagon of $T$-equivariant $\Ethree$-factorisation objects on $\C^3$
\[
\begin{tikzcd}[column sep=2.6em, row sep=1.8em]
& (\mathrm E)
 \arrow[dr, "\Theta_{\mathrm E\to\mathrm B}"{description}, "\sim"']
& \\
(\mathrm F)
 \arrow[ur, "\Theta_{\mathrm F\to\mathrm E}"{description}, "\sim"']
&&
(\mathrm B)
 \arrow[d, "\theta_{\C^3}"] \\
&
(\mathrm Y)
 \arrow[ul, "\Theta_{\mathrm Y\to\mathrm F}"{description}, "\sim"']
&
(\mathrm H)
 \arrow[l, "\mathrm{SV}"{description}, "\sim"']
\end{tikzcd}
\]
commuting modulo a single $\GRT_1(\Q)$-torsor action rigidified by a
choice of Drinfeld associator. In cohomology the Hall-side composite
reproduces the Schiffmann--Vasserot associative-algebra identification
$Y^+(\ghat)\cong\CoHA(\C^3)$. The theorem does not construct
$\theta_{\C^3}$; it records the normal form any such chart map must have.
\end{theorem}

\begin{proof}
The envelope, labelled-configuration, and Hall positive-half normal
forms are independent of the missing comparison map. The pentagon is
closed only after the chartwise map $\theta_{\C^3}$ is supplied.

\emph{HKR through the $\Ethree$-envelope.} On $V=\C^3$ the HKR graded
isomorphism
$\HH^\bullet(\Perf(V))\cong\PV^\bullet(V)=\C[z_1,z_2,z_3]\otimes
\Lambda^\bullet\langle\partial_1,\partial_2,\partial_3\rangle$
upgrades to a chain-level $\Ethree$-quasi-isomorphism by the combined
inputs of Kontsevich $L_\infty$-formality~\cite{Kontsevich1999},
Tamarkin $\Etwo$-upgrade (Tamarkin~$2007$,
Lett.~Math.~Phys.~$81$), and Tradler--Menichi--Ginzburg cyclic
BV-lift at the $(-3)$-cyclic pairing induced by
$\Omega_{\C^3}=dz_1\wedge dz_2\wedge dz_3$
\textup{(}Proposition~\ref{prop:cy3-tmg-bv}\textup{)}. The topological
envelope $\cU^\FA$ on $\R^3$ is a left Quillen functor
\textup{(}Costello--Gwilliam, vol.~I,~$\S 3.5$\textup{)} and so
preserves this quasi-isomorphism. The envelope's cohomology on $\R^3$
is the SV rational shuffle algebra with kernel
\[
\phi_{\mathrm{SV}}(u,v)\;=\;
\frac{(u-v+\epsilon_1)(u-v+\epsilon_2)(u-v+\epsilon_3)}
{(u-v)(u-v+\epsilon_1+\epsilon_2)(u-v+\epsilon_2+\epsilon_3)}
\]
on the CY locus $\epsilon_1+\epsilon_2+\epsilon_3=0$; in cohomology the
envelope of $\PV^\bullet(\C^3)_{\mathrm{SN}}$ computes the equivariant
Euler class of the tangent--obstruction complex at monomial-ideal fixed
points of $\Hilb^\bullet(\C^3)$. This gives
$(\mathrm E)\simeq(\mathrm Y)$.

\emph{Topological $\R_t$ reduction of $5$d hCS.} The
Costello--Gwilliam holomorphic--topological tensor decomposition
\textup{(}\emph{Factorisation Algebras in QFT},
vol.~$2$,~$\S 4.10$\textup{)} gives
\[
\Obs^{\mathrm{cl},5\hCS}(\R_t\times\C^2,\ghat)
\;\simeq\;
\Obs^{\mathrm{cl},1\text{-top}}(\R_t,\ghat[1])
\;\boxtimes\;
\Obs^{\mathrm{cl},4\text{-hol}}(\C^2,\ghat).
\]
The topological factor reduces by Francis~$2013$
\textup{(}Theorem~$2.29$\textup{)} to
$C^\bullet_{\mathrm{Lie}}(\ghat[1])$, the Chevalley--Eilenberg complex;
its classical $\hbar\to 0$ limit for abelian $\ghat$ is the completed
symmetric algebra on $\ghat[1]$. The $4$-holomorphic factor on $\C^2$
carries an $E_4$-algebra structure whose reduction to $\R^3$ via the
Costello--Li constant-sheaf shadow
$\Omega^{0,\bullet}(P)\simeq\C$ for Stein-contractible $P$
collapses to the Kontsevich $\Ethree$-formality target
$C^\bullet(\ghat)$. The composite yields a chain-level
$\Theta_{\mathrm E\to\mathrm B}\colon
\cU^\FA(\HH^\bullet(\Perf(\C^3))_F)\xrightarrow{\sim}
\Obs_{\hCS}^{q}(\R_t\times\C^2;\ghat)|_{\mathrm{top\text{-}reduced}}$.
This gives $(\mathrm E)\simeq(\mathrm B)$.

\emph{Candidate BV-equivariant localisation and Nakajima pullback.} The
$T$-equivariant BV partition function localises
\textup{(}Atiyah--Bott~$1984$\textup{)} to a sum over $T$-fixed points
of $\Hilb^\bullet(\C^3)$, indexed by plane partitions
$\pi\vdash_3 n$:
\[
Z^T_{5\hCS}(\C^3;\ghat)
\;=\;
\sum_{n\geq 0}\sum_{\pi\vdash_3 n}
\frac{\iota_\pi^* e^{S[\cA_\pi]/\hbar}}
{e_T\bigl(T_{I_\pi}\Hilb^n(\C^3)\bigr)},
\]
with denominator the MNOP equivariant Euler class computed via
arm--leg statistics
\textup{(}Maulik--Nekrasov--Okounkov--Pandharipande~$2006$,
arXiv:math/$0312059$, Theorem~$1$\textup{)}. The Nakajima
convolution coproduct
$\mu_{\CoHA}^\vee=p_{1,*}\circ p_2^!$ along the flag correspondence
$\operatorname{Flag}^{n,m}(\C^3)\subset\Hilb^n(\C^3)\times\Hilb^{n+m}(\C^3)$ packages
these coefficients into a CoHA class. The composite
\[
\Theta_{\hCS\to\Hall}^{\C^3}
\;:\;
\Obs_{\hCS}^{q}(\R_t\times\C^2;\ghat)
\xrightarrow{\mathrm{Loc}_T}
\bigoplus_{n,\pi}\C\cdot\iota_\pi^*[I_\pi]
\xrightarrow{\mu_{\CoHA}^\vee}
\bigoplus_n H^*_T(\Hilb^n(\C^3))
=\mathcal F_{\C^3}^{\mathrm{fr}}
\]
where \(\mathcal F_{\C^3}^{\mathrm{fr}}\) is the framed DT/Fock module
on which \(\CoHA(\C^3)=Y^+(\widehat{\mathfrak{gl}}_1)\) acts.  This is
the expected target shape of the missing chart map. The present
theorem does not prove that this formula is a continuous chain map,
multiplicative for BV/Hall products, compatible with the renormalised
differential, or a quasi-isomorphism. Those properties are precisely
the supplied hypothesis on $\theta_{\C^3}$.

Under that hypothesis, on a chain-level observable expanded in
Kontsevich-weighted Feynman
graphs $\mathcal O=\sum_\Gamma w_\Gamma(\mathcal O)\cdot\Gamma$,
$\theta_{\C^3}$ is expected to act by
\[
\theta_{\C^3}(\mathcal O)
\;=\;
\sum_{n\geq 0}\sum_{\pi\vdash_3 n}
\frac{\sum_\Gamma w_\Gamma(\mathcal O)\cdot\chi_\Gamma(\pi)}
{\prod_{s\in\pi}(\epsilon_1 l(s)-\epsilon_2(a(s)+1))
(-\epsilon_1(l(s)+1)+\epsilon_2 a(s))}
\cdot [I_\pi],
\]
where $\chi_\Gamma(\pi)$ is the graph evaluation on the $T$-character
of the tangent--obstruction complex at $I_\pi$ and
$l(s),a(s)$ are leg, arm statistics of the box $s\in\pi$.

The cohomology reduction of the supplied chart map recovers the
Schiffmann--Vasserot identification~$Y^+(\ghat)=\CoHA(\C^3)$
\textup{(}SV~$2013$, Theorem~$4.1$\textup{)}. The $\GRT_1(\Q)$-torsor
ambiguity arises once, in
the combined formality upgrade used for $(\mathrm E)\simeq(\mathrm Y)$
and $(\mathrm E)\simeq(\mathrm B)$; a single Drinfeld associator
rigidifies both edges. This gives $(\mathrm B)\simeq(\mathrm H)$.

\emph{Pentagon closure via labelled-config homology.} For any
$\Ethree$-algebra $A$, the Getzler--Jones identification
\textup{(}\cite{GetzlerJones1994}\textup{)} together with Lurie's
computation of factorisation homology
\textup{(}\cite{LurieHA}, Theorem~$5.5.4.10$\textup{)} gives a
chain-level equivalence
\[
H^\bullet(\FM_3(-);A)\;\simeq\;
H^\bullet(\cU^\FA(A)(\R^3)),
\]
so $(\mathrm F)\simeq(\mathrm E)$ via the $\Ethree$-labelled-config
homology computation. Transporting through the HKR envelope equivalence
already established,
$(\mathrm Y)\xrightarrow{\sim}(\mathrm E)\xrightarrow{\sim}(\mathrm F)$
gives $\Theta_{\mathrm Y\to\mathrm F}$.

Combining the envelope, labelled-configuration, Hall positive-half
normal form, and the supplied chart map gives the pentagon under the
stated hypothesis.
Commutativity modulo $\GRT_1(\Q)$ follows from naturality of the
constructed edges and is imposed on $\theta_{\C^3}$ as part of the
comparison datum: every edge passes through the $\Ethree$-formality
datum at some point, and a single associator choice rigidifies all five
simultaneously. In a Costello--Li local propagator model on $\R^6$, the
chosen $F$ is the Kontsevich point of the torsor.
\end{proof}

\begin{definition}[$\C^3$ local bridge obstruction vector]
\label{def:c3-local-bridge-obstruction-vector}
For the abelian chart $(X,\mathfrak g)=(\C^3,\ghat)$ fix the standard
holomorphic volume form, the torus action, and a formality point
$F\in\mathrm{Form}_3(\Q)$. The local bridge obstruction vector is
\[
 \mathfrak o_{\C^3}^{\mathrm{br}}
 =
 \bigl(
 o_{\mathrm{QME}},\,
 o_{\mathrm{ord}},\,
 o_{\mathrm{St1}},\,
 o_{\theta},\,
 o_{\mathrm{SV}},\,
 o_{\mathcal D}
 \bigr).
\]
Its components are:
\begin{enumerate}
\item $o_{\mathrm{QME}}$, the Costello--Li all-scale quantum master
equation/anomaly obstruction for $\Obs_{\hCS}^{q}(\C^3;\ghat)$;
\item $o_{\mathrm{ord}}$, the failure to use the completed ordered
chiral $E_3$ bar rather than the exterior CE shadow;
\item $o_{\mathrm{St1}}$, the Stage-$1$ formality and holomorphic
factorisation-envelope obstruction of
Theorem~\ref{thm:cfg-factorization-envelope-stage-one};
\item $o_{\theta}$, the Maurer--Cartan, continuity, multiplicativity,
and differential-compatibility obstruction for a chart map
$\theta_{\C^3}\colon\Obs_{\hCS}^{q}(\C^3;\ghat)\to
\CoHA_{\mathrm{crit}}^{\mathrm{or}}(\C^3)$;
\item $o_{\mathrm{SV}}$, the failure of the cohomology target to be the
Schiffmann--Vasserot positive half $\CoHA(\C^3)=Y^+(\ghat)$;
\item $o_{\mathcal D}$, the shortcut obstruction forbidding a direct
map $\CoHA(\C^3)\to\cW_{1+\infty}$ before passing through the Drinfeld
double and a Fock/evaluation representation.
\end{enumerate}
The vector is \emph{zero} precisely when every component is supplied and
the six compatibilities commute in the completed homotopy category.
\end{definition}

\begin{theorem}[$\C^3$ local closure criterion]
\label{thm:c3-local-bridge-closure-criterion}
For the abelian chart $(\C^3,\ghat)$ the following data are equivalent.
\begin{enumerate}
\item A closed local chain-level bridge
\[
 \Obs_{\hCS}^{q}(\C^3;\ghat)
 \xrightarrow{\;\theta_{\C^3}\;}
 \CoHA_{\mathrm{crit}}^{\mathrm{or}}(\C^3)
 \xrightarrow{\;\mathrm{SV}\;}
 Y^+(\ghat)
\]
compatible with the Stage-$1$ $\Ethree$ factorisation envelope and with
the typed passage to $\cW_{1+\infty}$ through the Drinfeld double.
\item Vanishing of the local obstruction vector
$\mathfrak o_{\C^3}^{\mathrm{br}}=\mathbf 0$ of
Definition~\ref{def:c3-local-bridge-obstruction-vector}.
\item A formality point $F$, an anomaly-free all-scale hCS package, the
completed ordered chiral $E_3$ bar, a continuous multiplicative
Maurer--Cartan chart map $\theta_{\C^3}$, the
Schiffmann--Vasserot positive-half identification, and the explicit
typing rule
\[
 \CoHA(\C^3)\;=\;Y^+(\ghat)
 \longrightarrow
 \mathcal D(Y^+(\ghat))
 \longrightarrow
 \cW_{1+\infty}.
\]
\end{enumerate}
If all components except $o_{\theta}$ vanish, the result is exactly the
normal form of Theorem~\ref{thm:r6-quad-equivalence-c3}, not a closed
bridge. Thus the missing local mathematical content is a single typed
object: the hCS--Hall chart map together with its Maurer--Cartan and
multiplicative coherences.
\end{theorem}

\begin{proof}
The implication \textup{(1)}$\Rightarrow$\textup{(2)} is obtained by
restricting the closed bridge to each of the six typed gates in
Definition~\ref{def:c3-local-bridge-obstruction-vector}: the QME
differential is square-zero, the controller is the ordered chiral
$E_3$ bar, the Stage-$1$ envelope exists, the chart map is a
Maurer--Cartan multiplicative chain map, the cohomology target is the
Schiffmann--Vasserot positive half, and $\cW_{1+\infty}$ is reached
only after doubling. Hence every obstruction component is zero.

For \textup{(2)}$\Rightarrow$\textup{(3)}, vanishing of the first three
components supplies the hCS observable factorisation algebra and the
Stage-$1$ envelope; vanishing of $o_{\theta}$ supplies the comparison
map with its chain-level coherences; vanishing of $o_{\mathrm{SV}}$
identifies the Hall cohomology with $Y^+(\ghat)$; and vanishing of
$o_{\mathcal D}$ fixes the only admissible route to
$\cW_{1+\infty}$. These are exactly the six pieces listed in
\textup{(3)}.

Finally \textup{(3)}$\Rightarrow$\textup{(1)} is composition: the first
three pieces give the left factorisation object, $\theta_{\C^3}$ gives
the hCS--Hall edge, SV gives the positive-half Yangian edge, and the
Drinfeld-double rule gives the typed $\cW_{1+\infty}$ passage. The
commuting coherences are part of the zero-obstruction hypothesis, so
the composite is a closed local bridge. If $o_{\theta}$ is not killed,
the other five pieces still identify the Hall-side target normal form,
but there is no map from hCS observables to that target; this is exactly
the obstruction recorded in
Problem~\ref{op:cy3-hcs-hall-comparison}.
\end{proof}

\begin{definition}[Fixed abelian $\C^3$ hCS--Hall chart]
\label{def:c3-fixed-abelian-hcs-hall-chart}
Let $T=(\C^\times)^3$ act on $\C^3$ with equivariant weights
$\varepsilon_1,\varepsilon_2,\varepsilon_3$ satisfying
$\varepsilon_1+\varepsilon_2+\varepsilon_3=0$. The positive
torus-fixed abelian hCS chart retained here is the
Schiffmann--Vasserot shuffle-normal-form completion of the free positive
hCS modes
\[
 \Obs_{\hCS}^{q,\mathrm{fp},+}(\C^3;\ghat)
 =
 \widehat{\mathrm T}_{\geq 0}\langle J_0,J_1,J_2,\ldots\rangle_{\mathrm{SV}},
 \qquad d_{\mathrm{fp}}J_r=0,
\]
with filtration by charge, total mode, and pair order. Equivalently,
it is the quotient of the completed ordered tensor algebra by the kernel
of the shuffle-normal-form map defined below. The
Schiffmann--Vasserot positive shuffle model
$Y_T^+(\ghat)$ has charge-$n$ component the symmetric rational
functions in variables $z_1,\ldots,z_n$ generated from charge $1$ by
the product
\[
 (f*g)(z_1,\ldots,z_{m+n})
 =
 \Sym_{m+n}
 \left[
 f(z_1,\ldots,z_m)\,
 g(z_{m+1},\ldots,z_{m+n})
 \prod_{\substack{1\leq i\leq m\\m<j\leq m+n}}
 \omega(z_j-z_i)
 \right],
\]
where
\[
 \omega(u)
 =
 \frac{(u+\varepsilon_1)(u+\varepsilon_2)(u+\varepsilon_3)}
 {u(u+\varepsilon_1+\varepsilon_2)(u+\varepsilon_2+\varepsilon_3)}.
\]
The fixed chart map is the filtered linear map
\[
 \theta_{\C^3}^{\mathrm{fp}}\colon
 \Obs_{\hCS}^{q,\mathrm{fp},+}(\C^3;\ghat)
 \longrightarrow
 Y_T^+(\ghat),
 \qquad
 J_r\longmapsto z_1^r\quad (r\geq 0),
\]
extended from words by the above shuffle product.
\end{definition}

\begin{theorem}[Fixed abelian $\C^3$ chart map]
\label{thm:c3-fixed-abelian-chart-map}
The map $\theta_{\C^3}^{\mathrm{fp}}$ of
Definition~\ref{def:c3-fixed-abelian-hcs-hall-chart} is a continuous
filtered chain algebra morphism from the positive torus-fixed abelian
hCS finite-mode sector to the Schiffmann--Vasserot positive half
$Y_T^+(\ghat)=\CoHA_T(\C^3)^{+}$. It is an isomorphism onto the
completed shuffle subalgebra generated by charge-$1$ monomials
$z_1^r$, $r\geq 0$. Consequently the obstruction component
$o_{\theta}$ of Definition~\ref{def:c3-local-bridge-obstruction-vector}
vanishes after projection to this fixed abelian finite-mode sector:
\[
 \operatorname{pr}_{\mathrm{fp},+}(o_\theta)=0.
\]
\end{theorem}

\begin{proof}
The source is the completed shuffle-normal-form algebra on the
generators $J_r$, and the target subalgebra is the completed
Schiffmann--Vasserot shuffle algebra generated by the charge-$1$
monomials $z_1^r$. The assignment $J_r\mapsto z_1^r$ therefore extends
uniquely to a continuous filtered algebra map because both sides
multiply words using the same cross-kernel $\omega$. If one begins
instead with the abstract completed free tensor algebra, the kernel of
this extension is exactly the shuffle-normal-form relation ideal
specified in Definition~\ref{def:c3-fixed-abelian-hcs-hall-chart}.

The chain statement is immediate on this sector: the abelian
torus-fixed hCS differential vanishes on the modes $J_r$, and the
shuffle target is placed in the corresponding cohomological degree with
zero differential. Hence
$d\,\theta_{\C^3}^{\mathrm{fp}}(J_r)=0=
\theta_{\C^3}^{\mathrm{fp}}(d_{\mathrm{fp}}J_r)$, and multiplicativity
extends the equality to all completed words.

The filtration is preserved because a word
$J_{r_1}\cdots J_{r_n}$ maps to charge $n$, total mode
$r_1+\cdots+r_n$, and pair-pole order at most
$\binom n2$: each pair of letters contributes exactly one
Schiffmann--Vasserot cross-kernel factor. This proves continuity for
the completed filtration. Surjectivity onto the stated subalgebra is
tautological from generation by the images of the $J_r$, and
injectivity follows because the source is the corresponding
shuffle-normal-form quotient, not the unquotiented abstract free tensor
algebra. The witness consists of the kernel, the two-point localization
formula, the zero differential, and the filtration bound.

Thus the Maurer--Cartan, differential, multiplicativity, and continuity
pieces of $o_\theta$ are killed on the projected fixed abelian
finite-mode chart. The theorem makes no claim about the analytic
extension from this chart to the full renormalised hCS factorisation
algebra, nor about descent over a non-affine cover.
\end{proof}

\begin{corollary}[Residual local obstruction after the fixed chart]
\label{cor:c3-residual-theta-after-fixed-chart}
After Theorem~\ref{thm:c3-fixed-abelian-chart-map}, the local
$\C^3$ comparison obstruction decomposes as
\[
 o_\theta
 =
 o_\theta^{\mathrm{fp},+}
 \oplus
 o_\theta^{\mathrm{ren}}
 \oplus
 o_\theta^{\mathrm{des}},
 \qquad
 o_\theta^{\mathrm{fp},+}=0,
\]
where $o_\theta^{\mathrm{ren}}$ is the extension problem from the
finite fixed shuffle chart to the full renormalised hCS factorisation
algebra and $o_\theta^{\mathrm{des}}$ is the descent/gluing problem for
chartwise maps.
\end{corollary}

\begin{proof}
The definition of $o_\theta$ contains exactly four conditions:
Maurer--Cartan, continuity, multiplicativity, and differential
compatibility for the hCS--Hall chart map. Projection to the positive
torus-fixed finite-mode sector is functorial for all four conditions,
and Theorem~\ref{thm:c3-fixed-abelian-chart-map} verifies them there.
The only remaining terms are therefore the two not touched by that
projection: analytic renormalised extension and descent.
\end{proof}

\begin{proposition}[Convolution vs. BV bracket under the exact local datum]
\label{prop:r6-convolution-vs-bv-bracket}
Assume the exact local $\C^3$ bridge datum of
Theorem~\ref{thm:c3-local-bridge-closure-criterion}; equivalently, fix
$\theta_{\C^3}$ with
$\mathfrak o_{\C^3}^{\mathrm{br}}=\mathbf 0$. Then
The Nakajima convolution action
$\mu_{\mathcal F}=p_{2,*}\circ p_1^*$ on the framed DT/Fock module
$\bigoplus H^*_T(\Hilb^n(\C^3))$
and the Feynman-expansion BV bracket
$\{-,-\}_{\BV}$ on $\Obs_{\hCS}^{q}(\R_t\times\C^2;\ghat)$ are
intertwined by $\Theta^{\C^3}_{\hCS\to\Hall}$ up to a chain homotopy
$h=O(\hbar)$ built from the Costello--Li propagator
$P^\beta_\varepsilon\propto\dbar^{-1}\delta$ on $\R_t\times\C^2$.
At the tree level $\hbar\to 0$ the homotopy vanishes and
$\mu_{\mathcal F}$ and $\{-,-\}_{\BV}$ coincide strictly; at higher loops
$h$ encodes the BV Laplacian counter-terms. The cohomology-level
identification is strict.
\end{proposition}

\begin{proof}
The exact local datum contains a Maurer--Cartan multiplicative chain map
$\theta_{\C^3}$ from hCS observables to the oriented critical CoHA. Its
multiplicativity component is a specified homotopy between the
BV/Feynman product on the source and the Nakajima push-pull product on
the target; the proposition identifies that homotopy in the local
Costello--Li model. Without this datum there is no chain-level
intertwiner to compare.

The BV bracket on interactions is the antisymmetrisation of the cubic
Feynman-vertex expansion
$\int_X \Omega_X\wedge\langle\alpha,[\alpha,\alpha]\rangle/6$; Nakajima
convolution is the push-pull of the flag correspondence at the moduli
level. The two agree in cohomology because both compute the SV shuffle
kernel $\phi_{\mathrm{SV}}(u,v)$ as the equivariant Euler class at the
monomial-ideal fixed point. At the chain level the BV Laplacian
contributes higher-loop insertions $\hbar^r\Delta^r_L$ that cancel
cohomologically but survive as a chain homotopy; the Costello--Li
propagator $P^\beta_\varepsilon$ provides the explicit $h$. At
$\hbar=0$ the BV differential reduces to the classical hCS
differential $\dbar+\{I,-\}_{\BV}$, whose bracket on Feynman trees
matches the tree-level Nakajima push-pull by direct evaluation at
fixed points.
\end{proof}

\section{Local charts of K3\texorpdfstring{$\times$E}{xE} under
\texorpdfstring{$\Theta_{\hCS\to\Hall}$}{Theta hCS to Hall}}
\label{sec:cy3-chain-level-bridge-k3e-local-charts}

The $\C^3$ five-way quasi-isomorphism of
Theorem~\ref{thm:r6-quad-equivalence-c3} propagates to
K3$\times$E chart-by-chart under the same comparison datum, with the
five-obstruction calculus of
Definition~\ref{def:hcs-hall-descent-obstruction} supplying the gluing
criterion.

\begin{definition}[Dolbeault--Weiss--Ran cover of K3$\times$E]
\label{def:r6-dwr-cover-k3e-inscribed}
Fix $X=K3\times E$ with K3 presented as a Kummer surface with toric
resolution and $E$ the elliptic curve. A \emph{DWR cover}
$\mathfrak U=\{U_i\}_{i\in I}$ is an analytic Stein cover by
holomorphic polydiscs such that each $U_i$ is Dolbeault-contractible,
carries a holomorphic volume form $\Omega_i$ agreeing with
$\Omega_X=\Omega_{K3}\wedge\Omega_E$ on overlaps, and is Weiss-cofinal
in the sense of Costello--Gwilliam~\textup{(}vol.~I,~$\S 6.1$\textup{)}.
On the Kummer toric strata the chosen polydiscs may be equipped with
toric $\C^3$ normal forms; for generic K3 with no toric structure the
same DWR cover uses Stein-contractible holomorphic polydiscs and the
explicit $\C^3$ fixed-point formula is replaced by the supplied local
chart map.
\end{definition}

\begin{proposition}[Chartwise anomaly vanishing for abelian $\ghat$]
\label{prop:r6-chartwise-anomaly-vanish-k3e-inscribed}
On each Stein polydisc chart $U_i$ of a DWR cover of $X=K3\times E$ with
abelian gauge $\ghat=\C\cdot T$, both components of the
Costello--Yagi~$2018$ one-loop $5$d hCS anomaly vanish identically:
the kinetic-field class
$A_{\mathrm{w.f.}}=-C_2/(2\pi)^3$ with quadratic Casimir $C_2=2h^\vee$
is zero because $h^\vee=0$ for abelian $\ghat$, and the cohomological
true anomaly $A_{\mathrm{anom}}=d^{abc}d_{abc}/(2\pi)^3$ is zero
because the structure constants $f^{abc}$ vanish for abelian
$\ghat$, forcing $d^{abc}=0$.
\end{proposition}

\begin{proof}
The Costello--Yagi~$2018$ anomaly-splitting theorem
\textup{(}Theorem~$4.1$\textup{)} attributes the $5$d hCS one-loop
anomaly to two invariants of the gauge Lie algebra:
$C_2$ and $d^{abc}d_{abc}$. Both are built from contractions of the
structure constants with the Killing form; for abelian $\ghat$, every
such contraction vanishes. The $A_{\mathrm{w.f.}}$ component is
scheme-dependent and absorbed by a BV-trivial counter-term even in
non-abelian cases; the $A_{\mathrm{anom}}$ component is cohomological
and is the genuine obstruction. For abelian $\ghat$ both are zero,
and the quantum master equation on each chart admits a solution
\textup{(}Costello--Li~$2020$, Theorem~$6.7$~\cite{CostelloLi2020}\textup{)}.
\end{proof}

\begin{proposition}[Hall orientation torsor on the Kummer DWR cover]
\label{prop:r6-orientation-torsor-k3e-inscribed}
On the DWR cover $\mathfrak U$ of $X=K3\times E$, the Hall-side
KS/Joyce orientation datum $o_X$ trivialises globally. This prepares
the target orientation line for a comparison map; it does not by itself
prove that the relative comparison obstruction
$o_{\mathrm{or}}$ of
Definition~\ref{def:hcs-hall-descent-obstruction} vanishes.
\end{proposition}

\begin{proof}
The orientation datum on $\CoHA_{\mathrm{crit}}^{\mathrm{or}}(X)$ is a
square root of the virtual determinant line
$\det T^{\mathrm{vir}}_{\mathfrak M}$ of the moduli stack. On each
Stein polydisc chart $U_i$, a holomorphic coordinate volume form
$\Omega_i=dz_1\wedge dz_2\wedge dz_3$ gives a canonical local
trivialisation of $\det T^{\mathrm{vir}}$, hence a canonical square
root. On overlaps $U_{ij}$, the ratio $\Omega_i/\Omega_j$ is a
holomorphic function (never zero, by existence of both
trivialisations); its square root is a section of the orientation
line bundle. The cocycle
$\{\Omega_i/\Omega_j\cdot\Omega_j/\Omega_k\cdot\Omega_k/\Omega_i\}=1$
identically as holomorphic functions, so the square-root cocycle lies
in the kernel of $\mathcal O_X^\times\to\mathcal O_X^\times/\pm 1$.
For $K3\times E$ the only global holomorphic units are constants
because K3 is simply-connected and $E$'s non-constant holomorphic
functions are unbounded, so the cocycle is $\pm 1$ valued and its
class lies in $H^1(X,\mathbb Z/2)=H^1(K3,\mathbb Z/2)\oplus
H^1(E,\mathbb Z/2)=0\oplus(\mathbb Z/2)^2$; the K\"unneth-product
orientation $\Omega_{K3}\wedge\Omega_E$ pins down the E-factor square
root by the compatibility $o_{K3\times E}=o_{K3}\boxtimes o_E$. The
resulting class is zero by Joyce--Song orientation
data~\cite{JoyceSong2012} applied to the K\"unneth factorisation. This
trivialises the Hall-side orientation torsor. The relative comparison
class $o_{\mathrm{or}}$ still belongs to the supplied hCS--Hall
comparison datum and is not killed by this Hall-side construction alone.
\end{proof}

\begin{proposition}[Hall grading, Thom--Sebastiani, and separate factorisation data]
\label{prop:r6-grading-thom-sebastiani-fact-k3e-inscribed}
On the DWR cover $\mathfrak U$ of $X=K3\times E$ with abelian
$\ghat$, the following Hall-side and source-side structures are
available:
\begin{enumerate}
\item the local normalisation's shift
$s(U,\mathbf d)=-\dim_\R\Hilb^{\mathbf d}(\C^3)=-6n$ and Tate twist
$t(U,\mathbf d)=-n$ are intrinsic to the moduli problem and
compatible on each polydisc chart with product coordinates on
$K3\times E$. Global compatibility of the grading/Tate conventions on
overlaps is part of the supplied comparison datum, not a Hilbert-scheme
product isomorphism.
\item the internal Hall Thom--Sebastiani associator for iterated Hall
convolution is coherent by the Kontsevich--Soibelman
wall-crossing formalism~\cite{KS2008}, which makes the CoHA
convolution strictly associative at chain level on the CY locus; for
abelian $\ghat$ there is no non-trivial associator correction.
\item on disjoint opens $U_1,U_2\subset X$, the BV Feynman-graph
expansion decomposes into disconnected graphs supported singly, and on
the CoHA side the disjoint-union product is the tensor product of
moduli supports.
\end{enumerate}
These are preparatory structures. They do not imply vanishing of the
relative comparison classes $o_{\mathrm{gr}}$, $o_{\mathrm{TS}}$, or
$o_{\mathrm{fact}}$ until a chartwise hCS--Hall comparison map is fixed.
\end{proposition}

\begin{proof}
\emph{(1)} The perverse shift and Tate twist depend only on the
virtual dimension of the local critical chart, which is $n$, and on the
real dimension $6n$ of the smooth local model. On a product polydisc
inside $K3\times E$ these values are the same as for $\C^3$. Passing
from this local normalisation to a global DWR cover requires overlap
compatibility of the grading/Tate transports, which is precisely one of
the relative comparison classes in
Definition~\ref{def:hcs-hall-descent-obstruction}.

\emph{(2)} Kontsevich--Soibelman~$2008$
\textup{(}arXiv:$0811.2435$, Theorem~$3$\textup{)} establishes strict
associativity of the CoHA convolution on the CY locus via the motivic
wall-crossing formula. The two parenthesisations of iterated
convolution
\[
(\mu_{\CoHA}\otimes 1)\circ\mu_{\CoHA}
\quad\text{vs.}\quad
(1\otimes\mu_{\CoHA})\circ\mu_{\CoHA}
\]
coincide strictly. For abelian $\ghat$ the Thom--Sebastiani
isomorphism carries only the sign permutation of plane-partition
dimension vectors, which is associative by construction.

\emph{(3)} Disjoint-union factorisation on both sides is a direct
computation. The BV partition function on $U_1\sqcup U_2$ is the
product of partition functions on $U_1$ and $U_2$ (no connected
Feynman graph bridges disjoint opens). The CoHA on
$\Hilb(U_1\sqcup U_2)=\Hilb(U_1)\times\Hilb(U_2)$ is the tensor
product of chartwise CoHAs. A future comparison map must be checked
against both decompositions; that is exactly the relative
factorisation obstruction.
\end{proof}

\begin{theorem}[Finite DWR/Ran comparison from supplied critical-CoHA charts]
\label{thm:r6-k3e-local-chart-qiso-inscribed}
\label{thm:finite-dwr-ran-oriented-hcs-hall-comparison}
\noindent\emph{Finite descent criterion.} Completed comparison requires
pro-compatible inverse-limit descent.
Fix $X=K3\times E$, a DWR cover $\mathfrak U$ as in
Definition~\ref{def:r6-dwr-cover-k3e-inscribed}, abelian
$\ghat=\C\cdot T$, finite holomorphic-jet order $N$, finite
renormalisation order $r$, finite charge bound $\Gamma_{\leq L}$, and
finite Ran arity bound $m$. Let
$\mathsf N_{\leq m}^{\Ran}(\mathfrak U;\Gamma_{\leq L})$ denote the
full finite sub-nerve of ordered DWR/Ran simplices in these bounds.
Assume that the finite Rees map of
Theorem~\ref{thm:finite-dwr-ran-hcs-rees-hall-construction} has been
realised on every simplex by a monoidal vanishing-cycle functor
$\mathsf R_{\mathrm{vc}}$ compatible with compact support, lci pullback,
Thom--Sebastiani, completion, and determinant-line local systems.
Equivalently, assume that every simplex $\sigma$ carries a full
renormalised
degree-zero chart map
\[
 \theta^{N,r,L}_{\sigma}\colon
 \Obs_{\hCS}^{q,\leq r}
 \bigl(|\sigma|,J^N_{\mathrm{hol}}\ghat\bigr)_{\Gamma_{\leq L}}
 \longrightarrow
 \CoHA_{\mathrm{crit}}^{\mathrm{or},\leq N}
 \bigl(|\sigma|\bigr)_{\Gamma_{\leq L}}
\]
and the following finite primitive data:
\[
 \eta_{\mathrm{MC}},\quad
 \lambda_{\mathrm{or}},\quad
 \eta_{\mathrm{gr}},\quad
 H_{\mathrm{TS}},\quad
 H_{\mathrm{fact}},\quad Q .
\]
Here $\eta_{\mathrm{MC}}$ solves the finite Cech/Ran Maurer--Cartan
equation, $\lambda_{\mathrm{or}}$ transports the
Joyce--Kontsevich determinant-line square roots, $\eta_{\mathrm{gr}}$
identifies the perverse shift and Tate twist, $H_{\mathrm{TS}}$ is the
oriented Thom--Sebastiani homotopy for every iterated Hall
correspondence in the finite nerve, $H_{\mathrm{fact}}$ is the
disjoint-Ran factorisation homotopy, and $Q$ is a vertexwise
quasi-isomorphism datum invertible on $H^0$ throughout the finite
nerve. Equivalently, the five finite obstruction components
\[
 (o_{\mathrm{MC}},o_{\mathrm{or}},o_{\mathrm{gr}},
 o_{\mathrm{TS}},o_{\mathrm{fact}})
\]
of Definition~\ref{def:hcs-hall-descent-obstruction} vanish in the
finite completed total Cech/Ran convolution dg Lie algebra.

Then the finite descended comparison map is
\[
\boxed{\;
 \Theta_{\hCS\to\Hall,\leq m}^{\mathrm{or},N,r,L}
 :=
 \Tot_{\sigma\in\mathsf N_{\leq m}^{\Ran}(\mathfrak U;\Gamma_{\leq L})}
 \bigl(
 \lambda_{\mathrm{or},\sigma}\,
 \eta_{\mathrm{gr},\sigma}\,
 \theta^{N,r,L}_{\sigma}
 \bigr)
 \;}
\]
together with the structure homotopies $H_{\mathrm{TS}}$ and
$H_{\mathrm{fact}}$. It is a morphism in the finite Hall-valued
oriented factorisation-cosheaf category
\[
 \mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}
 \bigl(\mathsf N_{\leq m}^{\Ran}(\mathfrak U;\Gamma_{\leq L})\bigr)
\]
and induces a quasi-isomorphism on finite DWR/Ran descent.

This statement constructs neither the realisation functor
$\mathsf R_{\mathrm{vc}}$ nor the determinant-line orientation system:
both have been assumed in order to define the critical-CoHA chart maps.
It also constructs no Hall--Drinfeld double and proves no
PBW/Serre-kernel theorem. Those promotions require vanishing of
the obstruction vector
\[
 \mathfrak o_{\mathrm{cpt/dbl/PBW}}(\Theta_{+}^{N,r,L,m})
\]
of Theorem~\ref{thm:finite-positive-half-obstruction-map}.

The source theorem is the compact finite-stage open--closed source
bridge, Theorem~\ref{thm:compact-cy3-finite-stage-open-closed-source}.
The target theorem is the descent criterion,
Theorem~\ref{thm:hcs-hall-descent-criterion}, applied to the finite
completed nerve. The Hall-side square-root trivialisation of
Proposition~\ref{prop:r6-orientation-torsor-k3e-inscribed} and the
Hall-side Thom--Sebastiani associativity of
Proposition~\ref{prop:r6-grading-thom-sebastiani-fact-k3e-inscribed}
prepare the target normalisation; they do not replace the relative
cochains $\lambda_{\mathrm{or}}$ and $H_{\mathrm{TS}}$.

The remaining global descent condition after this finite construction
is compatibility of the primitive data under the four transition
systems
\[
 N+1\to N,\qquad r+1\to r,\qquad
 \Gamma_{\leq L+1}\to\Gamma_{\leq L},\qquad
 \mathsf N_{\leq m+1}^{\Ran}\to\mathsf N_{\leq m}^{\Ran},
\]
equivalently vanishing of the pro-descent obstruction
\[
 \varprojlim\nolimits^1_{N,r,L,m}
 \prod_{\alpha\in\{\MC,\mathrm{or},\mathrm{gr},\mathrm{TS},\mathrm{fact}\}}
 H^0\bigl(
 \mathfrak M_{\hCS,\Hall}^{N,r,L,m}
 \bigr)_\alpha .
\]
\end{theorem}

\begin{proof}
Restrict
Definition~\ref{def:hcs-hall-descent-obstruction} to the finite
DWR/Ran sub-nerve and the finite jet, RG, and charge bounds. The result
is a finite completed Cech/Ran convolution dg Lie algebra
$\mathfrak M_{\hCS,\Hall}^{N,r,L,m}$. The cochain
\[
 \theta^{N,r,L}+\eta_{\mathrm{MC}}
\]
satisfies the Maurer--Cartan equation in this dg Lie algebra by
hypothesis. This is exactly the chain-map and overlap-naturality
condition for the displayed total map after forgetting the orientation,
grading, Thom--Sebastiani, and disjoint-factorisation structures.

The cochain $\lambda_{\mathrm{or}}$ kills the relative square-root
mismatch between the hCS BV determinant transport and the
Kontsevich--Soibelman/Joyce orientation local system on the Hall side.
The cochain $\eta_{\mathrm{gr}}$ makes the comparison degree-zero after
the shift and Tate normalisation. The homotopies $H_{\mathrm{TS}}$ and
$H_{\mathrm{fact}}$ identify the two transported Thom--Sebastiani
parenthesisations and the disjoint-Ran product squares. These are
precisely the four coherent structures forgotten in the first
paragraph. The datum $Q$ gives a homotopy inverse on every
$0$-simplex, and finite descent of complexes preserves such
quasi-isomorphisms. The displayed map is therefore a morphism in the
finite Hall-valued oriented factorisation-cosheaf category and a
finite-descent quasi-isomorphism.

Theorem~\ref{thm:compact-cy3-finite-stage-open-closed-source} supplies
the source-side finite Costello--Li/open--closed map; the present
criterion begins only after the monoidal realisation functor and full
renormalised hCS--Hall simplex maps have been supplied.
Proposition~\ref{prop:r6-orientation-torsor-k3e-inscribed}
trivialises the Hall orientation torsor, and
Proposition~\ref{prop:r6-grading-thom-sebastiani-fact-k3e-inscribed}
gives the Hall-side grading and Thom--Sebastiani normalisations, but a
relative comparison class can still survive until
$\lambda_{\mathrm{or}}$, $\eta_{\mathrm{gr}}$, $H_{\mathrm{TS}}$, and
$H_{\mathrm{fact}}$ are chosen.

Passing from the finite construction to the completed compact comparison
is an inverse-limit problem. The four transition systems must carry the
chosen primitives to one another inside complete, exhaustive, separated,
strongly convergent filtrations. Failure of such compatibility is the
displayed $\varprojlim^1$ class; its vanishing is the first global
descent obstruction. It still does not construct the compact double,
the completed Hall pairing, or a PBW/Serre-kernel theorem; those remain
the obstruction components of
Theorem~\ref{thm:finite-positive-half-obstruction-map}.
\end{proof}

\begin{corollary}[Igusa scalar line and compact Hall obstruction]
\label{cor:k3e-igusa-scalar-compact-hall-obstruction}
For the $K3\times E$ finite comparison the Borcherds/Igusa scalar convention is
\[
 D_5=64^{-1}\Delta_5,\qquad
 \operatorname{den}(\mathfrak g_{\Delta_5})=\Delta_5
 \quad
 \textup{(monic form }D_5\textup{ in OP variables),}
\]
and
\[
 \chi_{10}^{\mathrm{OP}}
 =
 D_5^2
 =
 4096^{-1}\Delta_5^2,
 \qquad
 Z^{\mathrm{OP/DT}}_{K3\times E}
 =
 -(\chi_{10}^{\mathrm{OP}})^{-1}
 =
 -D_5^{-2}
 =
 -4096\,\Delta_5^{-2}.
\]
These equations are character constraints. They do not construct the
compact critical CoHA, the orientation transport, the Hall--Drinfeld
double, the Hopf-pairing radical quotient, or the
PBW/Serre-kernel comparison.
\end{corollary}

\begin{proof}
The normalisation $D_5=64^{-1}\Delta_5$ gives
$D_5^2=4096^{-1}\Delta_5^2$. Oberdieck--Pandharipande's reduced
OP/DT scalar is the reciprocal square in OP normalisation, with the
Behrend sign giving the displayed minus sign. The primitive BKM
denominator is the unsquared Gritsenko--Nikulin factor $\Delta_5$,
equivalently the monic product $D_5$ after the OP change of scalar.
Squaring forgets the multiplier and every square-root sign; a scalar
reciprocal has no Hall correspondence product, coproduct, determinant
line, negative half, completed Hopf pairing, or associated-graded
presentation. Hence the Igusa identities constrain any compact Hall
realisation but do not supply the missing components of
$\mathfrak o_{\mathrm{cpt/dbl/PBW}}$.
\end{proof}

\begin{remark}[Scope of Theorem~\ref{thm:r6-k3e-local-chart-qiso-inscribed}]
\label{rem:r6-scope-k3e-inscribed}
Three scope conditions are recorded:
\begin{enumerate}
\item \emph{Abelian gauge is assumed}, with $\ghat=\C\cdot T$
throughout; the anomaly-vanishing argument of
Proposition~\ref{prop:r6-chartwise-anomaly-vanish-k3e-inscribed} uses
$f^{abc}=d^{abc}=0$. For a non-abelian simple $\frakg$, the same
conclusion requires the invariant-theoretic condition
$d^{abc}_\frakg=0$; it holds for $A_1, B_n, C_n, D_n, E_6, E_7, E_8,
F_4, G_2$ and fails for $A_n$, $n\geq2$. Even when it holds,
$A_{\mathrm{w.f.}}=-C_2/(2\pi)^3$ is non-zero; this is
scheme-dependent and absorbed by a BV-trivial counter-term, so the
quasi-isomorphism extends after that counter-term choice is fixed.
\item \emph{Toric DWR cover assumed} (K3 a Kummer surface). For
generic K3 without toric structure, the cover is by
Stein-contractible polydiscs; a comparison would require a
Morse-theoretic construction of the chartwise maps and a fresh check of
the five obstruction classes. The explicit
fixed-point formula of
Theorem~\ref{thm:r6-quad-equivalence-c3} is replaced by a
Morse-theoretic gradient-flow sum.
\item \emph{Kummer-K\"unneth compatibility} of the orientation datum
is used in
Proposition~\ref{prop:r6-orientation-torsor-k3e-inscribed}: the
global trivialisation $\Omega_{K3\times E}=\Omega_{K3}\wedge\Omega_E$
is required. For a twisted K\"ahler deformation of K3 this may fail
and an additional $\mathbb Z/2$ cocycle appears.
\end{enumerate}
For $A_n$, $n\geq2$, the cubic Casimir $d^{abc}$ does not vanish and
the anomaly gate of
Proposition~\ref{prop:cy3-hcs-quartic-anomaly-slot} activates;
Theorem~\ref{thm:r6-k3e-local-chart-qiso-inscribed} does not
apply. The correspondence scope in this chapter is abelian unless the
vanishing of $d^{abc}$ and the BV counter-term are supplied.
\end{remark}
