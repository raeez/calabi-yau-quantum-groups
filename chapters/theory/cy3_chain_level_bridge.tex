\chapter{The CY3 Chain-Level Bridge}
\label{ch:cy3-chain-level-bridge}

\section{Normal form}
\label{sec:cy3-chain-level-bridge-normal-form}

The useful CY$_3$ object is not the slogan ``six-dimensional holomorphic
Chern--Simons gives a chiral algebra''.  The useful object is the
following chain of typed constructions.

\begin{definition}[The hCS BV complex on a Calabi--Yau threefold]
\label{def:cy3-hcs-bv-complex}
\ClaimStatusDefinitional{}
Let $X$ be a smooth Calabi--Yau threefold with holomorphic volume form
$\Omega_X$, and let $\mathfrak g$ be a finite-dimensional metric Lie
algebra with invariant pairing $\langle-,-\rangle$.  The classical
holomorphic Chern--Simons BV field complex is
\[
  \cE_{\hCS}(X,\mathfrak g)
  :=
  \Omega^{0,\bullet}(X,\mathfrak g)[1],
  \qquad
  Q_{\hCS}=\dbar .
\]
Its local classical functional is
\[
  I_{\hCS}(\alpha)
  =
  \int_X \Omega_X \wedge
  \left(
    \frac{1}{2}\langle \alpha,\dbar\alpha\rangle
    +
    \frac{1}{6}\langle \alpha,[\alpha,\alpha]\rangle
  \right),
\]
where the displayed superfield expression means: multiply in
$\Omega^{0,\bullet}(X)$, take the $\Omega^{0,3}$ component, and integrate
the resulting $(3,3)$-form.  The factorisation algebra of quantum
observables, when the local anomaly is cancelled, is denoted
\[
  \Obs_{\hCS}^{q}(X,\mathfrak g).
\]
\end{definition}

\begin{definition}[The many-variable chiral CE model]
\label{def:cy3-many-variable-chiral-ce}
\ClaimStatusDefinitional{}
Let $P=D_1\times D_2\times D_3\subset X$ be a holomorphic polydisc with
coordinates $z_1,z_2,z_3$.  Let $\mathfrak l_{\cC}$ be the local dg
Lie algebra controlling the chosen CY$_3$ chart of $\cC$; in a pure hCS
gauge chart this is the constant dg Lie algebra $\mathfrak g$ with
zero internal differential.  Put
\[
  \mathfrak L_{\cC}(P)
  :=
  \Omega^{0,\bullet}_c
  \bigl(P,J^\infty_{\mathrm{hol},z_1,z_2,z_3}\mathfrak l_{\cC}\bigr)[1],
  \qquad
  d_{\mathfrak L}=\dbar+d_{\mathfrak l},
  \qquad
  [\alpha,\beta]_{\mathfrak L}
  =
  \alpha\wedge\beta\otimes[\,,\,]_{\mathfrak l_{\cC}}.
\]
The local classical observable object retained by the CY$_3$
bridge is the Dolbeault--chiral Chevalley--Eilenberg factorisation
algebra
\[
  \Obs_{\cC}^{\mathrm{cl}}(P)
  =
  C^\bullet_{\mathrm{Lie,cont}}\bigl(\mathfrak L_{\cC}(P),\mathbb C\bigr)
  =
  \widehat{\mathrm{Sym}}\bigl(\mathfrak L_{\cC}(P)^\vee[-1]\bigr),
\]
with differential $d_{\mathrm{CE}}+\dbar^\vee+d_{\mathfrak l}^\vee$.
Equivalently, the stage-one local model is
\[
  \PhiFA_3(\cC)|_P
  \simeq
  U^{\mathrm{fact},E_3}_{P}
  \bigl(\mathfrak L_{\cC}\bigr),
  \qquad
  B_{E_3}\bigl(\PhiFA_3(\cC)|_P\bigr)
  \simeq
  \mathrm{CE}^{\mathrm{ch},E_3}_*
  \bigl(\mathfrak L_{\cC}\bigr),
\]
on the loci where the hCS realisation of $\PhiFA_3$ is available.  The
factorisation products are products over disjoint holomorphic polydiscs;
their singularities lie only on partial diagonals.  In local fields this
means that an operator product is expanded as
\[
  a(z)b(w)
  \sim
  \sum_{\alpha\in\mathbb N^3}
  \frac{(a_{(\alpha)}b)(w)}
  {(z_1-w_1)^{\alpha_1+1}
   (z_2-w_2)^{\alpha_2+1}
   (z_3-w_3)^{\alpha_3+1}},
\]
where $a_{(\alpha)}b$ is computed by iterated residues in the three
holomorphic directions.  The ordinary Chevalley--Eilenberg algebra
$C^\bullet(\mathfrak g)$ is obtained only after applying the locally
constant shadow
\[
  \Omega^{0,\bullet}(P)\simeq \mathbb C,
  \qquad
  J^\infty_{\mathrm{hol}}\mathfrak l_{\cC}\rightsquigarrow
  H^\bullet(\mathfrak l_{\cC,x}),
  \qquad
  \Obs_{\cC}^{\mathrm{cl}}(P)\rightsquigarrow
  C^\bullet\bigl(H^\bullet(\mathfrak l_{\cC,x})\bigr).
\]
In the constant hCS chart this last object is $C^\bullet(\mathfrak g)$.
Thus $C^\bullet(\mathfrak g)$ is a topological associated model, not
the CY$_3$ chiral object.
\end{definition}

\begin{construction}[The typed CY$_3$ bridge]
\label{con:cy3-typed-bridge}
\ClaimStatusDefinitional{}
For a CY$_3$ input $\cC$ the bridge has four different kinds of objects:
\[
\begin{tikzcd}[column sep=large]
  \PhiFA_3(\cC)
    \arrow[r, dashed, "{\Theta_{\hCS\to\Hall}}"]
  & \CoHA_{\mathrm{crit}}(X)
    \arrow[r, "{\mathrm{SV/KS\ local\ model}}"]
  & Y^+
    \arrow[r, "{\mathrm{Drinfeld\ double}}"]
  & \mathcal{D}(Y^+)
    \arrow[r, "{\mathrm{Fock/evaluation}}"]
  & \cW_{1+\infty}.
\end{tikzcd}
\]
The first object is a holomorphic factorisation algebra.  The second is
an associative cohomological Hall algebra with vanishing-cycle input.
The third is a positive half.  The fourth is the double on which the
$\cW_{1+\infty}$ comparison lives.  Thus the bridge never identifies
$\CoHA(\C^3)$ with $\cW_{1+\infty}$ directly; the direct algebraic
identification is $\CoHA(\C^3)\cong Y^+$.
\end{construction}

\begin{definition}[Hall-valued factorisation-cosheaf target]
\label{def:hall-valued-factorisation-cosheaf-target}
\ClaimStatusDefinitional{}
Let $X$ be a CY$_3$ equipped with a Dolbeault/Weiss/Ran site
$\mathsf{DW\!R}(X)$.  An object of
\[
  \mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}(X)
\]
is the following data:
\begin{enumerate}
\item for each admissible open $U$, a completed chain complex
$\mathcal H(U)$ whose local critical-chart presentation is a
vanishing-cycle Borel--Moore complex;
\item for disjoint opens, continuous factorisation products
$\mathcal H(U_1)\widehat\otimes\mathcal H(U_2)\to
\mathcal H(U_1\sqcup U_2)$ whose one-open restriction is the Hall
extension product;
\item for each critical chart, a Kontsevich--Soibelman/Joyce
orientation datum: a square root of the virtual determinant line,
together with its orientation local system and coherences under direct
sums, extensions, and restriction to overlaps;
\item a specified completion: charge/HN-adic and equivariant-localised
on the Hall side, $\hbar$-adic and continuous-dual on the hCS side;
\item coherent Thom--Sebastiani isomorphisms for iterated extension
correspondences, including the signs, Tate twists, and orientation
local systems needed for associativity.
\end{enumerate}
A morphism in this category is a continuous natural transformation of
factorisation cosheaves preserving all five pieces of structure.  Thus
the phrase ``hCS-to-Hall quasi-isomorphism'' is meaningful only after
both sides have first been placed in this category, or after an
explicit forgetful functor from this category has been named.
\end{definition}

\begin{definition}[Local critical-CoHA normalisation]
\label{def:local-critical-coha-normalisation}
\ClaimStatusDefinitional{}
On a quiver chart $(Q_U,W_U)$ with dimension vector $\mathbf d$, write
\[
  \mathfrak M_{U,\mathbf d}
  =
  [\mathrm{Rep}_{\mathbf d}(Q_U)/G_{\mathbf d}],
  \qquad
  f_{U,\mathbf d}=\mathrm{Tr}(W_U)_{\mathbf d}.
\]
An oriented critical-CoHA summand in the convention of
Definition~\ref{def:hall-valued-factorisation-cosheaf-target} is
\[
  H^{\mathrm{BM}}_{G_{\mathbf d}}
  \!\left(
    \mathrm{Crit}(f_{U,\mathbf d}),
    \phi_{f_{U,\mathbf d}}\otimes\mathscr L_{o_U}
  \right)
  [s(U,\mathbf d)](t(U,\mathbf d)),
\]
where $\phi$ is the chosen vanishing-cycle functor, $[s]$ is the
perverse/cohomological shift, $(t)$ is the Tate twist, and
$\mathscr L_{o_U}$ is the local system determined by the chosen
orientation datum $o_U$.  The symbols $s(U,\mathbf d)$ and
$t(U,\mathbf d)$ are part of the convention; a comparison with hCS
observables is ungraded until they are fixed.
\end{definition}

\section{The proved local core}
\label{sec:cy3-chain-level-bridge-local-core}

\begin{proposition}[$\C^3$ core]
\label{prop:cy3-c3-bridge-core}
\ClaimStatusProvedElsewhere{}
On the affine toric chart $X=\C^3$, the algebraic Hall side of
Construction~\ref{con:cy3-typed-bridge} has the following proved core:
\[
  \CoHA(\C^3)\cong Y^+(\widehat{\mathfrak{gl}}_1),
  \qquad
  \mathcal{D}(\CoHA(\C^3))
  \simeq
  \mathcal{D}(Y^+(\widehat{\mathfrak{gl}}_1)).
\]
The $\cW_{1+\infty}$ object is reached from this doubled algebra by the
standard Fock/evaluation representation; it is not the positive-half
CoHA itself.
\end{proposition}

\begin{proof}
The first displayed isomorphism is the Schiffmann--Vasserot
identification recorded in Theorem~\ref{thm:sv-c3}, with the
Kontsevich--Soibelman critical-CoHA formalism supplying the Hall
construction of the local three-loop quiver model~\cite{KS2008,SV13}.
Taking the Drinfeld double is functorial for this presentation, so the
second displayed equivalence is a formal consequence of the first.
The final sentence is the evaluation-chain discipline already separated
in Remark~\ref{rem:evaluation-chain-c3} and in the internal Drinfeld
centre comparison Theorem~\ref{thm:c3-drinfeld-center}: the module
category and its centre see the doubled algebra, while the CoHA before
doubling is only the positive half.
\end{proof}

\begin{proposition}[Local-to-toric descent package]
\label{prop:cy3-local-to-toric-descent-package}
\ClaimStatusConditional{} \textup{(conditional on the comparison map
in Problem~\ref{op:cy3-hcs-hall-comparison})}
Assume that on every affine toric chart $U\simeq \C^3$ in a smooth toric
CY$_3$ there is a natural transformation in
$\mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}(X)$ whose value on $U$
is a quasi-isomorphism
\[
  \Theta_U\colon
  \Obs_{\hCS}^{q}(U,\mathfrak g)
  \longrightarrow
  \CoHA_{\mathrm{crit}}^{\mathrm{or}}(U)
\]
compatible with restriction to overlaps, orientation-torsor transport,
and the Hall correspondence product.  Then the chartwise identifications
$\CoHA(U)\cong Y^+$ glue through the toric descent diagram of
Theorem~\ref{thm:toric-chart-gluing} to give a chain-level witness for
the CY$_3$ stage-one bridge on that toric locus.
\end{proposition}

\begin{proof}
The hypotheses make $\{\Theta_U\}$ a morphism of factorisation cosheaves
of local observables into the oriented Hall cosheaf; this is stronger
than a list of chartwise quasi-isomorphisms.  Weiss descent for
factorisation algebras then identifies the global observable algebra
with the homotopy colimit of the chartwise observables.  The Hall
multiplication is associative and compatible with restriction by
assumption, hence the homotopy colimit carries the same $E_1$ product.
On each chart the Schiffmann--Vasserot theorem supplies the $Y^+$ model,
and Theorem~\ref{thm:toric-chart-gluing} is precisely the manuscript's
gluing theorem for these toric $Y^+$ charts.  No new global CY$_3$
existence theorem is used: the non-formal data are the oriented
comparison maps $\Theta_U$ on charts and overlaps.
\end{proof}

\section{The anomaly gate}
\label{sec:cy3-chain-level-bridge-anomaly-gate}

\begin{proposition}[Quartic, not cubic, anomaly slot]
\label{prop:cy3-hcs-quartic-anomaly-slot}
\ClaimStatusProvedElsewhere{}
For holomorphic Chern--Simons theory on a complex threefold, the local
one-loop obstruction belongs to the invariant-polynomial slot of degree
$4$ on $\mathfrak g$.  Equivalently, the obstruction is governed by the
quartic Casimir datum in the Costello--Li local Lie-algebra cohomology
description~\cite{CostelloLi2016,CostelloLi2020}.  It is not the cubic
slot governing the complex-surface or five-dimensional hCS descendants.
\end{proposition}

\begin{proof}
For a holomorphic theory on a complex $d$-fold, the wheel anomaly is a
local functional with $d+1$ Lie-algebra inputs: the Dolbeault heat-kernel
contraction supplies the holomorphic volume density, and the remaining
vertex count is fixed by locality and degree.  Thus the invariant
polynomial has degree $d+1$.  Substituting $d=3$ gives degree $4$.  The
cubic class is the $d=2$ specialization and therefore cannot be
transported unchanged into the CY$_3$ hCS bridge.
\end{proof}

\begin{warning}[No CFG shortcut]
\label{warn:cy3-no-cfg-shortcut}
\ClaimStatusDefinitional{}
Costello--Francis--Gwilliam's Chern--Simons/factorisation-homology
machine~\cite{CFG2026} constructs a filtered $E_3$-algebra by BV
quantisation of ordinary $3$d Chern--Simons theory and recovers
Reshetikhin--Turaev link invariants by factorisation-homology traces.
Its associated classical local model is $C^\bullet(\mathfrak g)$,
coming from the formal neighbourhood of the trivial flat bundle on a
real $3$-ball.  The CY$_3$ hCS avatar is instead the many-variable
Dolbeault--chiral CE object of
Definition~\ref{def:cy3-many-variable-chiral-ce}; it remembers
holomorphic jets in $z_1,z_2,z_3$, polydisc factorisation, and
multidirectional residues.  CFG is therefore not a proof that
six-real-dimensional hCS on a CY$_3$ is already the same object as the
CY$_3$ Hall algebra.  The latter comparison is the map
$\Theta_{\hCS\to\Hall}$ in Construction~\ref{con:cy3-typed-bridge}, and
it remains a separate mathematical datum.
\end{warning}

\section{The first missing lemma}
\label{sec:cy3-chain-level-bridge-missing-lemma}

\begin{openproblem}[hCS-to-Hall comparison]
\label{op:cy3-hcs-hall-comparison}
\ClaimStatusOpen{}
Construct, for the local CY$_3$ charts needed by $\PhiFA_3$, a natural
orientation-preserving quasi-isomorphism in
$\mathsf{FactCosh}_{\Hall}^{\mathrm{or},\wedge}(X)$
\[
  \Theta_{\hCS\to\Hall}^{\mathrm{or}}\colon
  \Obs_{\hCS}^{q}(-,\mathfrak g)
  \longrightarrow
  \CoHA_{\mathrm{crit}}^{\mathrm{or}}(-)
\]
where the target is the critical CoHA with the normalisation of
Definition~\ref{def:local-critical-coha-normalisation}.  In local quiver
coordinates this means
\[
  \CoHA_{\mathrm{crit}}^{\mathrm{or}}(U)
  =
  \bigoplus_{\mathbf d}
  H^{\mathrm{BM}}_{G_{\mathbf d}}
  \!\left(
    \mathrm{Crit}(\mathrm{Tr}W_{\mathbf d}),
    \phi_{\mathrm{Tr}W_{\mathbf d}}\otimes\mathscr L_{o_U}
  \right)
\]
with the shift and Tate twist fixed as part of the data.  The
orientation datum $o_U$ is a KS/Joyce square root of the virtual
determinant line with extension and overlap coherences; it is not merely
a formal notation for $K_{\mathfrak M_U}^{1/2}$.  The comparison must
state the gauge/rank datum, equivariant parameters, stability condition,
anomaly cancellation, completion, and compact-support convention, and it
must satisfy all six conditions:
\begin{enumerate}
\item it is local for the Dolbeault topology and respects Weiss descent;
\item it intertwines the BV bracket and the Hall correspondence product
after the CY$_3$ shift;
\item on the $\C^3$ chart it reduces to the
Kontsevich--Soibelman/Schiffmann--Vasserot positive-half model
$\CoHA(\C^3)=Y^+$;
\item after Drinfeld doubling it is compatible with the
$\cW_{1+\infty}$ Fock/evaluation representation;
\item it transports determinant-line square roots on overlaps and kills
the residual $\Z/2$ orientation cocycle on triple overlaps;
\item it respects Thom--Sebastiani for short exact sequences and the
Hall convolution product.
\end{enumerate}
This is the first lemma whose proof would turn the present local
bridge into a non-formal compact CY$_3$ theorem.
\end{openproblem}

\section{Chain-level assembly of the left end: braces, BV, envelope}
\label{sec:cy3-chain-level-bridge-envelope-stage-one}

The status ledger records $\PhiFA_3(\cC)\dashrightarrow\CoHA_{\mathrm{crit}}(X)$
as the open arrow; its left end, $\PhiFA_3(\cC)$ itself, is not a
slogan but an explicit composite of three chain-level steps.
Deligne--Tamarkin supplies a chain-level $\Etwo$-action on the
Hochschild cochain complex via braces and cup-products;
Tradler--Menichi--Ginzburg promote this to a BV algebra using the
cyclic pairing of degree $-3$; Costello--Gwilliam's factorisation
envelope sends the resulting $\Ethree$-algebra to a factorisation
algebra on $\R^3$, whose Costello--Li holomorphic twist lands in
$\EdHolFA(X)$. The right end of the bridge, the CoHA, is not produced
by this construction: it is the image of $\Theta_{\hCS\to\Hall}$, a
separate piece of data whose existence is
Problem~\ref{op:cy3-hcs-hall-comparison}. The present section records
the left end alone, at the level of proof.

\begin{proposition}[Deligne--Tamarkin braces on $\HH^\bullet(\cC)$]
\label{prop:cy3-braces-action}
\ClaimStatusProvedElsewhere{}
For a smooth proper cyclic $\Ainf$-category $\cC$ of any CY dimension,
the Hochschild cochain complex $\HH^\bullet(\cC)$ carries a chain-level
operadic action
\[
\mathrm{Brace}_\bullet\;:\;C_\bullet(\Conf_2;\Q)\;\longrightarrow\;\mathrm{End}(\HH^\bullet(\cC)),
\qquad \Conf_2:=\text{little $2$-discs},
\]
from the singular chains on the little $2$-discs operad. The primary
operations are the Getzler--Voronov braces
\[
f\{g_1,\ldots,g_r\}(a_1,\ldots,a_{m+\sum n_i - r})
\;=\;\sum_{i_1<\cdots<i_r}\pm
f\bigl(a_1,\ldots,g_1(a_{i_1},\ldots),\ldots,g_r(a_{i_r},\ldots),\ldots\bigr),
\]
the May--Steenrod $\cup_i$-products, the cup product $\cup$, and the
Gerstenhaber bracket $\{-,-\}_\Ger$. On cohomology the action induces
the classical Gerstenhaber structure. The chain-level action is
canonical up to a torsor under $\GRT_1(\Q)$ of choices of Drinfeld
associator.
\end{proposition}

\begin{proof}[Attribution and audit]
Kontsevich--Soibelman~$2000$ (\emph{Algebra and Representation Theory}
$3$, \S~$5$ Theorem~$5.1$) present the Kontsevich--Soibelman operad
$\cK=\mathrm{Br}\otimes\mathrm{Sur}$ acting on $\HH^\bullet(\cC)$ and prove it is
quasi-isomorphic to $C_\bullet(\Conf_2;\Q)$. McClure--Smith~$2003$
(\emph{Topology} $42$, Theorem~$1.1$) prove the brace operad
$\mathrm{Br}\hookrightarrow C_\bullet(\Conf_2;\Q)$ is a quasi-isomorphism of
operads. Tamarkin~$1998$ (\texttt{arXiv:math/9803025}; reproved
Tamarkin~$2007$, \emph{Lett.\ Math.\ Phys.}~$81$) identifies the space
of such $\Etwo$-structures with a torsor under $\GRT_1(\Q)$.
Kontsevich~$1999$ (\emph{Lett.\ Math.\ Phys.}~$48$, \S~$4$) singles
out the Drinfeld-associator point of this torsor.
\end{proof}

\begin{proposition}[Tradler--Menichi--Ginzburg BV operator from cyclicity]
\label{prop:cy3-tmg-bv}
\ClaimStatusProvedElsewhere{}
For a smooth proper cyclic $\Ainf$-category $\cC$ of CY dimension $d$,
the cyclic pairing $\langle-,-\rangle:\cC\otimes\cC\to k[-d]$ induces
a chain-level BV operator
\[
\Delta\;:\;\HH^\bullet(\cC)\;\longrightarrow\;\HH^{\bullet-1}(\cC),
\]
of cohomological degree $-1$, with $\Delta^2=0$ on cohomology,
satisfying the BV identity
\[
\{a,b\}_\Ger\;=\;\Delta(a\cup b)\;-\;\Delta(a)\cup b
\;-\;(-1)^{|a|}a\cup\Delta(b)
\]
on the nose. The triple $(\cup,\{-,-\}_\Ger,\Delta)$ upgrades
$\HH^\bullet(\cC)$ to a chain-level BV algebra; equivalently, to an
algebra over framed little $2$-discs via the Getzler equivalence
$\BV\simeq C_\bullet(\Conf_2^{\mathrm{fr}};\Q)$.
\end{proposition}

\begin{proof}[Attribution and audit]
Tradler--Zeinalian~$2006$ (\emph{J.\ Pure Appl.\ Algebra} $204$,
Theorem~$3.1$); Menichi~$2004$ (\emph{K-Theory} $32$,
Proposition~$3.4$); Ginzburg~$2006$ (\texttt{arXiv:math/0612139},
Theorem~$3.4.3$). The defining formula on a cochain
$f:(s\bar\cC)^{\otimes n}\to\cC$ is the cyclic-rotation average
\[
\Delta f(a_1,\ldots,a_{n-1})\;=\;\sum_{i=1}^{n}(-1)^{\epsilon_i}
\bigl\langle a_i,f(a_{i+1},\ldots,a_n,a_1,\ldots,a_{i-1},\mathbf 1_\cC)\bigr\rangle^\vee,
\]
with $\langle-,-\rangle^\vee$ the pairing composed with the duality
$\cC^\vee\simeq\cC[d]$; cyclic invariance of $\mu_n$ is exactly what
forces $\Delta^2=0$ and the BV identity. The Getzler equivalence
$\BV\simeq C_\bullet(\Conf_2^{\mathrm{fr}};\Q)$ is
Getzler~$1994$ (\emph{Comm.\ Math.\ Phys.}~$159$).
\end{proof}

\begin{proposition}[$\Ethree$-structure on $\HH^\bullet(\cC)$ at $d=3$ via Dunn]
\label{prop:cy3-e3-structure-hochschild}
\ClaimStatusProvedElsewhere{}
Under the Dunn--Lurie decomposition $\Ethree\simeq\Etwo\otimes_{\mathrm{Dunn}}\Eone$,
the framed $\Etwo$-structure of Proposition~\ref{prop:cy3-tmg-bv}
combines with the canonical $\Eone$-structure on the third real
direction of $\R^3$ to produce a chain-level $\Ethree$-algebra
structure on $\HH^\bullet(\cC)$ for a cyclic $\Ainf$-category of
CY dimension $3$. The canonicity of this $\Ethree$-structure is under a
single $\GRT_1(\Q)$-torsor, with the torsor acting on the $\Etwo$-factor
and the $\Eone$-factor rigid.
\end{proposition}

\begin{proof}
Dunn--Lurie additivity (Lurie, \emph{Higher Algebra}, Theorem
$5.1.2.2$) gives $\Ethree\simeq\Etwo\otimes_{\mathrm{Dunn}}\Eone$ as
$\infty$-operads. Proposition~\ref{prop:cy3-tmg-bv} realises
$\BV\simeq\Etwo^{\mathrm{fr}}$ on $\HH^\bullet(\cC)$. The Dunn-tensor
$\Eone$-factor acts by the canonical chain-level commutative
multiplication on the extra real direction; framed $\Etwo
\otimes_{\mathrm{Dunn}}\Eone\simeq\Ethree^{\mathrm{top}}$ (as topological operads) is the
unframed $\Ethree$ because $\pi_1(\Conf_2(\R^3))=0$. The torsor
computation is Willwacher~$2014$ (\emph{Invent.\ Math.}~$200$):
$\pi_0\mathrm{Aut}_\BV=\GRT_1(\Q)$, with the $\Eone$-factor contributing
no extra freedom (McClure--Smith~$2003$ Proposition~$2.7$).
\end{proof}

\begin{definition}[Topological factorisation envelope on $\R^3$]
\label{def:cy3-topological-envelope}
\ClaimStatusDefinitional{}
For $A\in\Ethree\text{-}\mathrm{Alg}(\mathrm{Ch}(k))$, the
\emph{topological factorisation envelope} is the pre-factorisation
algebra
\[
\cU^\FA(A)(U)\;=\;\bigoplus_{\pi\in\pi_0(U)}A_\pi,
\qquad A_\pi:=A,
\]
for $U\subset\R^3$ open, with structure maps for disjoint
$U_1\sqcup\cdots\sqcup U_k\hookrightarrow V$ built from the
$\Ethree$-multiplication
\[
\mu_k^\Ethree:A^{\otimes k}\longrightarrow A
\]
together with a chain in $C_\bullet(\Conf_k(V);\Q)$ representing the
embedding. Weiss descent (Costello--Gwilliam, \emph{Factorization
Algebras in QFT}, Vol.~I Theorem~$6.1.7$) promotes the
pre-factorisation algebra to a genuine factorisation algebra
$\cU^\FA(A)\in\mathrm{FAct}(\R^3;\mathrm{Ch}(k))$; the assignment is a
left Quillen functor
\[
\cU^\FA\;:\;\Ethree\text{-}\mathrm{Alg}(\mathrm{Ch}(k))\;\longrightarrow\;\mathrm{FAct}(\R^3;\mathrm{Ch}(k)).
\]
\end{definition}

\begin{theorem}[Chain-level Stage-$1$ envelope: $\PhiFA_3$ through $\HH^\bullet$ and $\cU^\FA$]
\label{thm:cfg-factorization-envelope-stage-one}
\ClaimStatusConditional{}\textup{(on the Stage-$1$ verified locus of
Theorem~\ref{thm:phi-two-stage-factorisation})}
Let $(\cC,\mu_n,\langle-,-\rangle)$ be a smooth proper cyclic
$\Ainf$-category of CY dimension $3$ satisfying \textup{(H1)--(H3)} of
Proposition~\ref{prop:phi-fa-three-step-assembly}. Let $X$ be a smooth
complex CY$_3$ variety with CY form $\Omega_X\in H^0(X,K_X)$. The
Stage-$1$ native factorisation algebra
$\PhiFA_3(\cC)\in\EdHolFA(X)$ of
Definition~\ref{def:phi-fa-and-sp}\textup{(i)} factors as
\[
\PhiFA_3(\cC)\;=\;\mathrm{Hol}_X\!\left(\cU^\FA\bigl(\HH^\bullet(\cC)\bigr)\right),
\]
whose CY$_3$ local normal form on a holomorphic polydisc is the
many-variable Dolbeault--chiral CE model of
Definition~\ref{def:cy3-many-variable-chiral-ce}.  Ordinary
$C^\bullet(\mathfrak g)$ appears only after the locally constant shadow
$\Omega^{0,\bullet}(P)\simeq\mathbb C$ has been applied.
The factors are:
\begin{enumerate}
\item $\HH^\bullet(\cC)\in\Ethree\text{-}\mathrm{Alg}(\mathrm{Ch}(k))$ by
Proposition~\ref{prop:cy3-e3-structure-hochschild};
\item $\cU^\FA$ the topological factorisation envelope on $\R^3$ of
Definition~\ref{def:cy3-topological-envelope};
\item $\mathrm{Hol}_X:\mathrm{FAct}(\R^6;\mathrm{Ch}(k))\to\EdHolFA(X)$
the Costello--Li holomorphic twist at the Dolbeault operator $\dbar_X$,
rigidified by the CY form $\Omega_X$
\textup{(Costello--Li~$2016$~\cite{CostelloLi2016},
Costello--Li~$2020$~\cite{CostelloLi2020}, Theorem~$6.7$)}.
\end{enumerate}
Canonicity is up to a single $\GRT_1(\Q)$-torsor of Drinfeld
associators, with the Costello--Li propagator picking the Kontsevich
point.
\end{theorem}

\begin{proof}[Audit against the three-step decomposition]
The three steps of Proposition~\ref{prop:phi-fa-three-step-assembly}
match item-by-item:
\begin{itemize}
\item[(a)] Kontsevich $E_d$-formality plus Tamarkin
deformation-quantisation is exactly
Propositions~\ref{prop:cy3-braces-action}--\ref{prop:cy3-e3-structure-hochschild}
at $d=3$: the $\Ethree$-structure on $\HH^\bullet(\cC)$ composed of
the unframed $\Etwo$ from braces and the framed $\Etwo\otimes_{\mathrm{Dunn}}
\Eone=\Ethree$ from the BV operator.
\item[(b)] Costello--Gwilliam locality on the topological space
underlying $X$ is Definition~\ref{def:cy3-topological-envelope}: the
factorisation envelope $\cU^\FA(\HH^\bullet(\cC))\in\mathrm{FAct}(\R^3;
\mathrm{Ch}(k))$ for the $\Ethree$-algebra input.
\item[(c)] Costello--Li holomorphic twist with CY form $\Omega_X$ is
$\mathrm{Hol}_X$: the topological factorisation algebra on $\R^6$ gets
refined to a holomorphic factorisation algebra on $X$ by replacing
$\Omega^\bullet(\R^6)$ with $\Omega^{0,\bullet}(X)$ and using
$\Omega_X$ to trivialise the BV propagator
$P^\beta_\varepsilon\propto\dbar_X^{-1}\delta$.
\end{itemize}
Stage-$1$ canonicity is up to $\GRT_1(\Q)$-torsor by
Proposition~\ref{prop:cy3-e3-structure-hochschild}; Costello--Li's
explicit propagator on $\R^6$ reduces to the Kontsevich integral of
multi-zeta values~\cite{CostelloLi2016}, picking the Drinfeld-associator
point of the torsor.
\end{proof}

\begin{remark}[What this section proves and what it does not]
\label{rem:cfg-envelope-scope}
\ClaimStatusDefinitional{}
Theorem~\ref{thm:cfg-factorization-envelope-stage-one} gives the
object-level Stage-$1$ envelope: $\PhiFA_3(\cC)$ is the
holomorphic-twist of the factorisation envelope of the
$\Ethree$-Hochschild cochain complex. Three scope restrictions are
recorded:
\begin{enumerate}
\item \emph{Object level, not morphism level.} Promoting a cyclic
$\Ainf$-functor $f:\cC\to\cC'$ to a morphism of factorisation algebras
$\PhiFA_3(f)$ requires a further chain-level check of compatibility
with braces, BV, and envelope; this is part of
Conjecture~\ref{conj:phi-d-functoriality}.
\item \emph{Local construction; global anomaly.} The envelope is
constructed on $\R^3$ and transported to $X$ by $\mathrm{Hol}_X$. On
compact non-affine $X$, the holomorphic twist is subject to the
quartic anomaly obstruction of
Proposition~\ref{prop:cy3-hcs-quartic-anomaly-slot}; the envelope
itself is insensitive to this, but the landing in $\EdHolFA(X)$ is.
\item \emph{Does not reach the CoHA.} The envelope lands in
$\EdHolFA(X)$, the left end of the typed bridge of
Construction~\ref{con:cy3-typed-bridge}. The right end, the
critical-CoHA, is reached only by $\Theta_{\hCS\to\Hall}$ of
Problem~\ref{op:cy3-hcs-hall-comparison}. The present section does not
supply that comparison; it is the chain-level content of the left
end alone.
\end{enumerate}
\end{remark}
