\chapter{The CY3 Chain-Level Bridge}
\label{ch:cy3-chain-level-bridge}

\section{Normal form}
\label{sec:cy3-chain-level-bridge-normal-form}

The useful CY$_3$ object is not the slogan ``six-dimensional holomorphic
Chern--Simons gives a chiral algebra''.  The useful object is the
following chain of typed constructions.

\begin{definition}[The hCS BV complex on a Calabi--Yau threefold]
\label{def:cy3-hcs-bv-complex}
\ClaimStatusDefinitional{}
Let $X$ be a smooth Calabi--Yau threefold with holomorphic volume form
$\Omega_X$, and let $\mathfrak g$ be a finite-dimensional metric Lie
algebra with invariant pairing $\langle-,-\rangle$.  The classical
holomorphic Chern--Simons BV field complex is
\[
  \cE_{\hCS}(X,\mathfrak g)
  :=
  \Omega^{0,\bullet}(X,\mathfrak g)[1],
  \qquad
  Q_{\hCS}=\dbar .
\]
Its local classical functional is
\[
  I_{\hCS}(\alpha)
  =
  \int_X \Omega_X \wedge
  \left(
    \frac{1}{2}\langle \alpha,\dbar\alpha\rangle
    +
    \frac{1}{6}\langle \alpha,[\alpha,\alpha]\rangle
  \right),
\]
where the displayed superfield expression means: multiply in
$\Omega^{0,\bullet}(X)$, take the $\Omega^{0,3}$ component, and integrate
the resulting $(3,3)$-form.  The factorisation algebra of quantum
observables, when the local anomaly is cancelled, is denoted
\[
  \Obs_{\hCS}^{q}(X,\mathfrak g).
\]
\end{definition}

\begin{construction}[The typed CY$_3$ bridge]
\label{con:cy3-typed-bridge}
\ClaimStatusDefinitional{}
For a CY$_3$ input $\cC$ the bridge has four different kinds of objects:
\[
\begin{tikzcd}[column sep=large]
  \PhiFA_3(\cC)
    \arrow[r, dashed, "{\Theta_{\hCS\to\Hall}}"]
  & \CoHA_{\mathrm{crit}}(X)
    \arrow[r, "{\mathrm{SV/KS\ local\ model}}"]
  & Y^+
    \arrow[r, "{\mathrm{Drinfeld\ double}}"]
  & \mathcal{D}(Y^+)
    \arrow[r, "{\mathrm{Fock/evaluation}}"]
  & \cW_{1+\infty}.
\end{tikzcd}
\]
The first object is a holomorphic factorisation algebra.  The second is
an associative cohomological Hall algebra with vanishing-cycle input.
The third is a positive half.  The fourth is the double on which the
$\cW_{1+\infty}$ comparison lives.  Thus the bridge never identifies
$\CoHA(\C^3)$ with $\cW_{1+\infty}$ directly; the direct algebraic
identification is $\CoHA(\C^3)\cong Y^+$.
\end{construction}

\section{The proved local core}
\label{sec:cy3-chain-level-bridge-local-core}

\begin{proposition}[$\C^3$ core]
\label{prop:cy3-c3-bridge-core}
\ClaimStatusProvedElsewhere{}
On the affine toric chart $X=\C^3$, the algebraic Hall side of
Construction~\ref{con:cy3-typed-bridge} has the following proved core:
\[
  \CoHA(\C^3)\cong Y^+(\widehat{\mathfrak{gl}}_1),
  \qquad
  \mathcal{D}(\CoHA(\C^3))
  \simeq
  \mathcal{D}(Y^+(\widehat{\mathfrak{gl}}_1)).
\]
The $\cW_{1+\infty}$ object is reached from this doubled algebra by the
standard Fock/evaluation representation; it is not the positive-half
CoHA itself.
\end{proposition}

\begin{proof}
The first displayed isomorphism is the Schiffmann--Vasserot
identification recorded in Theorem~\ref{thm:sv-c3}, with the
Kontsevich--Soibelman critical-CoHA formalism supplying the Hall
construction of the local three-loop quiver model~\cite{KS2008,SV13}.
Taking the Drinfeld double is functorial for this presentation, so the
second displayed equivalence is a formal consequence of the first.
The final sentence is the evaluation-chain discipline already separated
in Remark~\ref{rem:evaluation-chain-c3} and in the internal Drinfeld
centre comparison Theorem~\ref{thm:c3-drinfeld-center}: the module
category and its centre see the doubled algebra, while the CoHA before
doubling is only the positive half.
\end{proof}

\begin{proposition}[Local-to-toric descent package]
\label{prop:cy3-local-to-toric-descent-package}
\ClaimStatusConditional{} \textup{(conditional on the comparison map
in Problem~\ref{op:cy3-hcs-hall-comparison})}
Assume that on every affine toric chart $U\simeq \C^3$ in a smooth toric
CY$_3$ there is a quasi-isomorphism
\[
  \Theta_U\colon
  \Obs_{\hCS}^{q}(U,\mathfrak g)
  \longrightarrow
  \CoHA_{\mathrm{crit}}(U)
\]
compatible with restriction to overlaps and with the Hall
correspondence product.  Then the chartwise identifications
$\CoHA(U)\cong Y^+$ glue through the toric descent diagram of
Theorem~\ref{thm:toric-chart-gluing} to give a chain-level witness for
the CY$_3$ stage-one bridge on that toric locus.
\end{proposition}

\begin{proof}
The hypotheses make $\{\Theta_U\}$ a morphism of cosheaves of local
observables into the Hall cosheaf.  Weiss descent for factorisation
algebras identifies the global observable algebra with the homotopy
colimit of the chartwise observables.  The Hall multiplication is
associative and compatible with restriction by assumption, hence the
homotopy colimit carries the same $E_1$ product.  On each chart the
Schiffmann--Vasserot theorem supplies the $Y^+$ model, and
Theorem~\ref{thm:toric-chart-gluing} is precisely the manuscript's
gluing theorem for these toric $Y^+$ charts.  No new global CY$_3$
existence theorem is used: the only non-formal datum is the comparison
map $\Theta_U$ on charts and overlaps.
\end{proof}

\section{The anomaly gate}
\label{sec:cy3-chain-level-bridge-anomaly-gate}

\begin{proposition}[Quartic, not cubic, anomaly slot]
\label{prop:cy3-hcs-quartic-anomaly-slot}
\ClaimStatusProvedElsewhere{}
For holomorphic Chern--Simons theory on a complex threefold, the local
one-loop obstruction belongs to the invariant-polynomial slot of degree
$4$ on $\mathfrak g$.  Equivalently, the obstruction is governed by the
quartic Casimir datum in the Costello--Li local Lie-algebra cohomology
description~\cite{CostelloLi2016,CostelloLi2020}.  It is not the cubic
slot governing the complex-surface or five-dimensional hCS descendants.
\end{proposition}

\begin{proof}
For a holomorphic theory on a complex $d$-fold, the wheel anomaly is a
local functional with $d+1$ Lie-algebra inputs: the Dolbeault heat-kernel
contraction supplies the holomorphic volume density, and the remaining
vertex count is fixed by locality and degree.  Thus the invariant
polynomial has degree $d+1$.  Substituting $d=3$ gives degree $4$.  The
cubic class is the $d=2$ specialization and therefore cannot be
transported unchanged into the CY$_3$ hCS bridge.
\end{proof}

\begin{warning}[No CFG shortcut]
\label{warn:cy3-no-cfg-shortcut}
\ClaimStatusDefinitional{}
Costello--Francis--Gwilliam's Chern--Simons/factorisation-homology
machine is useful for organizing factorisation observables, but it is
not a proof that six-real-dimensional hCS on a CY$_3$ is already the
same object as the CY$_3$ Hall algebra.  The latter comparison is the
map $\Theta_{\hCS\to\Hall}$ in Construction~\ref{con:cy3-typed-bridge},
and it remains a separate mathematical datum.
\end{warning}

\section{The first missing lemma}
\label{sec:cy3-chain-level-bridge-missing-lemma}

\begin{openproblem}[hCS-to-Hall comparison]
\label{op:cy3-hcs-hall-comparison}
\ClaimStatusOpen{}
Construct, for the local CY$_3$ charts needed by $\PhiFA_3$, a natural
quasi-isomorphism of factorisation algebras
\[
  \Theta_{\hCS\to\Hall}\colon
  \Obs_{\hCS}^{q}(-,\mathfrak g)
  \longrightarrow
  \CoHA_{\mathrm{crit}}(-)
\]
which satisfies all four conditions:
\begin{enumerate}
\item it is local for the Dolbeault topology and respects Weiss descent;
\item it intertwines the BV bracket and the Hall correspondence product
after the CY$_3$ shift;
\item on the $\C^3$ chart it reduces to the
Kontsevich--Soibelman/Schiffmann--Vasserot positive-half model
$\CoHA(\C^3)=Y^+$;
\item after Drinfeld doubling it is compatible with the
$\cW_{1+\infty}$ Fock/evaluation representation.
\end{enumerate}
This is the first lemma whose proof would turn the present local
bridge into a non-formal compact CY$_3$ theorem.
\end{openproblem}

\begin{remark}[Status ledger]
\label{rem:cy3-chain-level-status-ledger}
\ClaimStatusDefinitional{}
\[
\begin{array}{c|c|c}
\textup{arrow} & \textup{status} & \textup{reason}\\ \hline
\Obs_{\hCS}^{q} & \textup{proved elsewhere, anomaly-gated}
  & \textup{Costello--Gwilliam/Costello--Li factorisation observables}\\
\CoHA(\C^3)\to Y^+ & \textup{proved elsewhere}
  & \textup{Kontsevich--Soibelman and Schiffmann--Vasserot}\\
Y^+\to \mathcal{D}(Y^+)\to \cW_{1+\infty}
  & \textup{proved elsewhere / represented}
  & \textup{double first, Fock/evaluation second}\\
\PhiFA_3(\cC)\dashrightarrow \CoHA_{\mathrm{crit}}(X)
  & \textup{open in general}
  & \textup{Problem~\ref{op:cy3-hcs-hall-comparison}}\\
\textup{toric chart gluing} & \textup{conditional}
  & \textup{Proposition~\ref{prop:cy3-local-to-toric-descent-package}}
\end{array}
\]
The ledger is deliberately typed: it records which arrows are theorems,
which arrows are representation-theoretic passages, and which arrow is
the actual frontier.
\end{remark}
