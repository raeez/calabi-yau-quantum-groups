\begin{theorem}[$E_2$-bar-cobar --- Theorem CY-B]
\label{thm:e2-bar-cobar}
For an $E_2$-chiral algebra $\cA$, the bar-cobar adjunction
\[
  B_{E_2} \colon E_2\text{-}\mathrm{ChirAlg} \rightleftarrows E_2\text{-}\mathrm{ChirCoalg} \cocolon \Omega_{E_2}
\]
satisfies:
\begin{enumerate}[label=(\roman*)]
  \item The $E_2$ bar complex $B_{E_2}(\cA)$ is a factorization $E_2$-coalgebra;
  \item Cobar-bar inversion: $\Omega_{E_2}(B_{E_2}(\cA)) \simeq \cA$ on the Koszul locus;
  \item The CY trace manifests as curvature: at genus $g \geq 1$, $d^2 = \kappa \cdot \omega_g$.
\end{enumerate}
\end{theorem}
