A braided monoidal category is a monoidal category equipped with a
coherent system of isomorphisms $V \otimes W \xrightarrow{\sim} W \otimes V$
satisfying the hexagon axioms. The bar-cobar adjunction of Volume~I,
extended to the $E_2$ setting, produces factorization coalgebras whose
degree-$(1,1)$ component is a categorical $R$-matrix. Chapter~\ref{ch:quantum-groups}
collects the classical Drinfeld--Jimbo presentation of $\Uq(\frakg)$, the
universal $R$-matrix, and the quantum Yang--Baxter equation; here we assume
that material and construct its $E_2$-chiral / factorization avatar. The
fundamental problem solved in this chapter: how does the $E_2$-chiral bar
complex $B_{E_2}(\cA)$ encode a categorical braiding, and what is the precise
relationship between the braided Koszul dual and the quantum group targets
of Conjecture~\ref{conj:qgf-cy-c}?

\section{The categorical \texorpdfstring{$R$}{R}-matrix and Yang--Baxter equation}
\label{sec:bf-categorical-r-matrix}

The Drinfeld--Jimbo universal $R$-matrix of
Proposition~\ref{prop:qgf-classical-limit-r} lives in $\Uq(\frakg)^{\hat\otimes 2}$
and is an \emph{algebraic} datum. The Vol~III programme produces $R$-matrices
of a different kind: \emph{categorical} $R$-matrices, extracted from the
degree-$(1,1)$ component of an $E_2$-chiral bar complex. The two pictures
are compatible through Conjecture~CY-C (Conjecture~\ref{conj:qgf-cy-c}):
when the CY-to-chiral correspondence $\Phi_2$ produces an affine Kac--Moody chiral algebra,
the categorical $R$-matrix coincides with the Drinfeld--Jimbo $R$-matrix
under the Kazhdan--Lusztig equivalence (Theorem~\ref{thm:qgf-kazhdan-lusztig}).

\begin{remark}[Drinfeld--Jimbo vs.\ FRT, from the bar-complex standpoint]
\label{rem:bf-dj-vs-frt}
The Drinfeld--Jimbo presentation constructs $\Uq(\frakg)$ from generators
and relations; the Faddeev--Reshetikhin--Takhtajan (FRT) construction
recovers $\Uq(\frakg)$ from the $R$-matrix via the RTT relation
$R_{12} T_1 T_2 = T_2 T_1 R_{12}$. The FRT construction is the
bar-complex-natural one: the $R$-matrix lives in degree $(1,1)$ of
$B^{\mathrm{ord}}(A)$, and the RTT relation is the
degree-$(1,1,1)$ bar differential. This is the standpoint
of the present chapter: the categorical $R$-matrix below is the
degree-$(1,1)$ component of $B_{E_2}(\Phi(\cC))$, and the parametric
Yang--Baxter equation is coassociativity of the deconcatenation coproduct.
\end{remark}

\begin{definition}[Categorical \texorpdfstring{$R$}{R}-matrix]
\label{def:categorical-r-matrix}
For a CY category $\cC$ of dimension $d = 2$, the \emph{categorical
$R$-matrix} $\cR_\cC(z)$ is the degree-$(1,1)$ component of the
$E_2$-bar complex $B_{E_2}(\Phi(\cC))$:
\[
 \cR_\cC(z) \in \End(V \otimes V)[[z, z^{-1}]]
\]
where $V = s^{-1}\bar{A}_\cC$ is the desuspended generating space and
$z$ is the spectral parameter. The assignment $\cC \mapsto \cR_\cC(z)$
is functorial in CY categories (for exact CY functors preserving the
$E_2$-structure).
\end{definition}

\begin{proposition}[Yang--Baxter from bar associativity]
\label{prop:ybe-from-bar}
\ClaimStatusProvedElsewhere
The categorical $R$-matrix $\cR_\cC(z)$ satisfies the Yang--Baxter
equation
\[
 \cR_{12}(z_1 - z_2) \, \cR_{13}(z_1 - z_3) \, \cR_{23}(z_2 - z_3)
 = \cR_{23}(z_2 - z_3) \, \cR_{13}(z_1 - z_3) \, \cR_{12}(z_1 - z_2).
\]
This is a formal consequence of the coassociativity of the bar coproduct
$\Delta_{\mathrm{decat}}$ applied to degree $3$ of $B^{\mathrm{ord}}(A)$.
\end{proposition}

\begin{proof}
The bar complex $B^{\mathrm{ord}}(A)$ is a coassociative coalgebra with
deconcatenation coproduct. The two compositions
$(\Delta \otimes \id) \circ \Delta$ and $(\id \otimes \Delta) \circ \Delta$
coincide on degree-$3$ elements; the $R$-matrix arises from the bar
differential restricted to degree $(1,1)$. Coassociativity at degree $3$
gives $\cR_{12}\cR_{13}\cR_{23} = \cR_{23}\cR_{13}\cR_{12}$, which is
the Yang--Baxter equation. The spectral parameters arise from the
$z$-dependence of the collision residue (so the $R$-matrix has one
fewer pole order than the OPE). See Volume~II for the
full derivation.
\end{proof}

\subsection{\texorpdfstring{$E_2$}{E-2}-chiral algebras at \texorpdfstring{$d = 2$}{d = 2}: the Mukai--Heisenberg lattice VOA on \texorpdfstring{$K3$}{K3}}
\label{subsec:mukai-heisenberg}

The simplest non-trivial native $\Etwo$-chiral algebra produced by the
CY-to-chiral correspondence at $d = 2$ is the \emph{Mukai--Heisenberg lattice
VOA} associated to a K3 surface. We collect its data here as the
running example for the rest of the chapter.

\begin{construction}[Mukai--Heisenberg lattice VOA]
\label{constr:mukai-heisenberg}
Let $S$ be a K3 surface and $\Lambda_{\mathrm{Muk}}(S) = H^*(S, \Z)$
the integral Mukai lattice equipped with the Mukai pairing
\[
 \langle (r_1, c_1, s_1), (r_2, c_2, s_2) \rangle_{\mathrm{Muk}}
 \;=\; c_1 \cdot c_2 - r_1 s_2 - r_2 s_1,
\]
which is even, unimodular, of signature $(4, 20)$, isomorphic to
$U^{\oplus 4} \oplus E_8(-1)^{\oplus 2}$. The CY-to-chiral correspondence
$\Phi_2$ at $d = 2$ outputs the lattice vertex algebra
\[
 V_{\Lambda_{\mathrm{Muk}}(S)}
 \;=\; H_{\mathrm{Muk}}(S) \otimes \C[\Lambda_{\mathrm{Muk}}(S)],
\]
where $H_{\mathrm{Muk}}(S)$ is the rank-$24$ Heisenberg algebra
generated by $\{a^\alpha_n\}$ for $\alpha \in \Lambda_{\mathrm{Muk}}(S)
\otimes \C$, $n \in \Z$, with relations
\[
 [a^\alpha_m, a^\beta_n]
 \;=\; m \, \langle \alpha, \beta \rangle_{\mathrm{Muk}} \, \delta_{m+n, 0},
\]
and $\C[\Lambda_{\mathrm{Muk}}(S)]$ is the group ring of lattice
translations. The vacuum has central charge $c = 24$ (one Heisenberg
boson per lattice direction).
\end{construction}

\begin{proposition}[Native \texorpdfstring{$\Etwo$}{E2}-structure of the Mukai--Heisenberg VOA]
\label{prop:mukai-heisenberg-e2}
\ClaimStatusProvedHere
The Mukai--Heisenberg lattice VOA $V_{\Lambda_{\mathrm{Muk}}(S)}$ is
natively $\Etwo$-chiral: the deconcatenation bar coalgebra
$B_{E_2}(V_{\Lambda_{\mathrm{Muk}}(S)})$ exists at the chain level and
its degree-$(1,1)$ component is the lattice $R$-matrix
\[
 \cR_{\Lambda_{\mathrm{Muk}}(S)}(z)
 \;=\; \exp\!\Bigl(\, \tfrac{1}{z} \sum_{i, j} G^{ij}_{\mathrm{Muk}} \, a^i_{(0)} \otimes a^j_{(0)} \Bigr) \cdot \epsilon_{\mathrm{Muk}},
\]
where $G^{ij}_{\mathrm{Muk}}$ is the Mukai Gram matrix and
$\epsilon_{\mathrm{Muk}} \colon \Lambda_{\mathrm{Muk}}(S) \times
\Lambda_{\mathrm{Muk}}(S) \to \{\pm 1\}$ is the Dotsenko--Fateev cocycle
fixing the lattice vertex operator signs. The braided shadow class
(Definition~\ref{def:braided-kappa}) is class~$\mathbf{G}$ (Gaussian).
\end{proposition}

\begin{proof}
$V_{\Lambda_{\mathrm{Muk}}(S)}$ is a free-field lattice VOA, hence
strongly $C_2$-cofinite and class~$\mathbf{G}$. The CY-A theorem
(Vol~III Theorem~\ref{thm:cy-a-d2}) supplies the native
$\Etwo$-chiral structure at $d = 2$. The degree-$(1,1)$ bar
component is the OPE residue of two vertex operators
$\Gamma_\alpha(z) \Gamma_\beta(w)$ with the
$d \log(z - w)$ propagator weight; the lattice
exponential structure (Frenkel--Lepowsky--Meurman, Borcherds)
makes this an exponential of the Mukai pairing acting on the
zero modes, with the Dotsenko--Fateev cocycle on the level of
$\C[\Lambda]$. Strong unitarity
$\cR_{12}(z) \cR_{21}(-z) = \id$
(Proposition~\ref{prop:e2-bar-degree-11}) follows from the
symmetry $\langle \alpha, \beta \rangle_{\mathrm{Muk}} = \langle
\beta, \alpha \rangle_{\mathrm{Muk}}$. Coassociativity
$\cR_{12}\cR_{13}\cR_{23} = \cR_{23}\cR_{13}\cR_{12}$ at
degree~$3$ holds for any abelian (Heisenberg) $R$-matrix
trivially, since the zero modes commute.
\end{proof}

\begin{remark}[Lattice VOA \texorpdfstring{$R$}{R}-matrix is bar data, not CoHA data]
\label{rem:bf-coha-not-bar}
The lattice $R$-matrix $\cR_{\Lambda_{\mathrm{Muk}}(S)}(z)$ of
Proposition~\ref{prop:mukai-heisenberg-e2} is the degree-$(1,1)$
component of the $\Etwo$-chiral bar complex $B_{E_2}(V_{\Lambda_{\mathrm{Muk}}(S)})$.
It is \emph{not} a Cohomological Hall Algebra (CoHA) construction.
The CoHA of $\C^3$ is the positive part $Y^+$ of the
Yangian (Schiffmann--Vasserot), which is an $\Eone$ object on the
ordered Ran space; the bar complex above is an $\Etwo$ object on
the curve $C \subset S$ along which the chiral algebra lives. Numerical
coincidences between CoHA generating functions and bar Euler
characteristics (e.g., MacMahon vs Heisenberg characters) reflect
the Schiffmann--Vasserot identification CoHA$(\C^3) = Y^+$ rather
than a chain-level identification CoHA $=$ bar.
\end{remark}


\section{Braided monoidal categories and \texorpdfstring{$E_2$}{E-2}-algebras}
\label{sec:braided-e2}

The connection between $E_2$-algebras and braided monoidal categories is
given by the Joyal--Street coherence theorem and its
$\infty$-categorical refinement.

\begin{proposition}[Dunn additivity for \texorpdfstring{$E_2$}{E-2}]
\label{prop:dunn-e2}
\ClaimStatusProvedElsewhere
The $E_2$ operad satisfies $\Etwo \simeq \Eone \otimes_{E_0} \Eone$.
An $\Etwo$-algebra is an $\Eone$-algebra in $\Eone$-algebras. In
particular, an $\Etwo$-algebra has two compatible associative products
whose interchange law is the braiding.
\end{proposition}

\begin{corollary}
\label{cor:e2-braided-monoidal}
The representation category $\Rep^{E_2}(\cA)$ of an $E_2$-algebra
$\cA$ is braided monoidal. The braiding on $V \otimes W$ is given by
the action of the $R$-matrix: $\sigma_{V,W} = \cR_{V,W} \circ \tau$
where $\tau \colon V \otimes W \to W \otimes V$ is the transposition
and $\cR_{V,W}$ is the $R$-matrix evaluated on the pair $(V,W)$.
\end{corollary}


\section{Factorization homology with \texorpdfstring{$E_2$}{E-2}-coefficients}
\label{sec:fh-e2}

Factorization homology $\int_M \cA$ of an $\En$-algebra $\cA$ over an
$n$-manifold $M$ (Ayala--Francis) generalizes Hochschild homology
($n = 1$, $M = S^1$) and chiral homology ($n = 2$, $M$ a surface).
The relevance to Vol~III is that CY$_2$ categories produce $\Etwo$-chiral
algebras (by CY-A at $d = 2$), and factorization homology of these
$\Etwo$-algebras over surfaces computes genus-$g$ conformal block spaces;
the shadow obstruction tower of Vol~I controls the moduli dependence.

\begin{definition}[Factorization homology]
\label{def:factorization-homology}
Let $\cA$ be an $\En$-algebra and $M$ an $n$-manifold. The
\emph{factorization homology} $\int_M \cA$ is the colimit
\[
 \int_M \cA \;=\; \colim_{\mathrm{Disk}_{n/M}} \cA
\]
over the $\infty$-category $\mathrm{Disk}_{n/M}$ of embeddings of
disjoint unions of disks into $M$. The colimit is computed in the
$\infty$-category of chain complexes (Ayala--Francis, 2015).
\end{definition}

\begin{proposition}[Excision for factorization homology]
\label{prop:fh-excision}
\ClaimStatusProvedElsewhere
Let $M = M_1 \cup_{N \times \R} M_2$ be a collar-gluing decomposition
of an $n$-manifold. For an $\En$-algebra $\cA$, there is a canonical
equivalence
\[
 \int_M \cA
 \;\simeq\;
 \int_{M_1} \cA \;\otimes_{\int_{N \times \R} \cA}\; \int_{M_2} \cA,
\]
where the tensor product is taken over the $\Eone$-algebra
$\int_{N \times \R} \cA$ (Ayala--Francis excision).
\end{proposition}

Excision is the engine that converts local data (disk-level OPE) into
global invariants (genus-$g$ amplitudes). Applied to the pair-of-pants
decomposition $\Sigma_g = \#_{2g-2}(\text{pair of pants})$, excision
expresses $\int_{\Sigma_g} \cA$ as an iterated relative tensor product
over Hochschild homology, recovering the sewing prescription of conformal
field theory.

\begin{proposition}[\texorpdfstring{$E_2$}{E-2} factorization homology on surfaces]
\label{prop:fh-e2-surface}
\ClaimStatusProvedElsewhere
For an $\Etwo$-algebra $\cA$ and a closed oriented surface
$\Sigma_g$ of genus $g$:
\begin{enumerate}[label=(\roman*)]
 \item $\int_{\Sigma_g} \cA$ is the genus-$g$ conformal block space;
 \item $\int_{S^1 \times [0,1]} \cA \simeq \HH_\bullet(\cA)$, the
 Hochschild homology;
 \item $\int_{T^2} \cA \simeq \HH_\bullet(\HH_\bullet(\cA))$
 (double Hochschild homology, computing the genus-$1$ amplitude).
\end{enumerate}
The shadow obstruction tower controls the dependence of
$\int_{\Sigma_g} \cA$ on the moduli of $\Sigma_g$.
\end{proposition}

\begin{proposition}[Genus tower from FH]
\label{prop:fh-genus-tower}
\ClaimStatusConditional
For a CY$_2$ category $\cC$ with $\Etwo$-chiral algebra
$\cA = \Phi(\cC)$, the factorization homology genus tower satisfies
\[
 \chi\!\left(\int_{\Sigma_g} \cA\right)
 \;=\;
 \kappa_{\mathrm{ch}}(\cA) \cdot \lambda_g
 \qquad\textup{(UNIFORM-WEIGHT)}
\]
at genus $g \geq 1$, where $\kappa_{\mathrm{ch}} = \chi^{\CY}(\cC)$
is the CY Euler characteristic (at $d = 2$, this coincides with
the manifold invariant $\kappa_{\mathrm{cat}} = \chi(\cO_X)$;
at $d \ge 3$ they may differ)
(Theorem~\textup{\ref{thm:cy-modular-characteristic}}). For
multi-weight algebras at $g \geq 2$, the scalar formula receives
the cross-channel correction $\delta F_g^{\mathrm{cross}}$ of Vol~I,
Theorem~D. The conditional dependence is on Theorem~CY-B(iii)
(curvature identification).
\end{proposition}

\begin{remark}[FH versus chiral homology]
\label{rem:fh-vs-chiral}
The Ayala--Francis factorization homology $\int_{\Sigma_g} \cA$ and the
Beilinson--Drinfeld chiral homology $H^{\mathrm{ch}}_\bullet(\Sigma_g, \cA)$
are equivalent for $\Etwo$-algebras that arise as factorization envelopes
of chiral Lie algebras (by the comparison theorem of Francis, 2013).
In the CY setting, $\cA = \Phi(\cC)$ is always of this type when CY-A
holds (at $d = 2$), so the two invariants agree. The FH formulation has
the advantage of applying to topological $\Etwo$-algebras without
holomorphic structure.
\end{remark}

\subsection{Weiss covers and the descent target dichotomy}
\label{subsec:weiss-covers-descent-target}

Factorisation locality in the Beilinson--Drinfeld and Costello--Gwilliam sense is \emph{not} the same descent condition as quasi-coherent locality on the underlying variety. The difference is carried by the Weiss cover axiom.

\begin{definition}[Weiss cover, after Costello--Gwilliam]
\label{def:weiss-cover}
Let $X$ be a topological space \textup{(}smooth manifold or analytic space\textup{).} A family $\mathfrak U = \{U_\alpha\}_{\alpha \in A}$ of opens in $X$ is a \emph{Weiss cover} of $X$ when for every integer $k \geq 1$ and every finite set of points $\{x_1, \ldots, x_k\} \subset X$ there exists $\alpha \in A$ with $\{x_1, \ldots, x_k\} \subset U_\alpha$ \textup{(}Costello--Gwilliam \emph{Factorization Algebras in Quantum Field Theory} Vol.~I, CUP~$2021$, Definition~$6.1.0.5$; the Weiss covers define the \emph{Weiss Grothendieck topology} on $\mathrm{Opens}(X)$\textup{).} Equivalently, for every $k \geq 1$ the family $\{U_\alpha^k\}_{\alpha \in A}$ of Cartesian powers covers the configuration space $\mathrm{Conf}_k(X) \subset X^k$. A Weiss cover is in particular an ordinary open cover \textup{(}take $k = 1$\textup{);} the converse fails, and this is the source of the descent dichotomy below.
\end{definition}

\begin{definition}[Weiss refinement of an arbitrary cover]
\label{def:weiss-refinement}
Let $\mathfrak U = \{U_\alpha\}_{\alpha \in A}$ be an open cover of $X$ \textup{(}not necessarily Weiss\textup{).} The \emph{Weiss refinement generated by} $\mathfrak U$ is the cover $\mathfrak U^\sqcup$ whose members are the finite disjoint unions $V_1 \sqcup \cdots \sqcup V_k \subset X$ where each $V_i$ is a contractible open contained in some $U_{\alpha(i)}$ and the $V_i$ are pairwise disjoint in $X$; in Costello--Gwilliam Vol.~I Definition~$6.1.0.6$ this is the condition that $\mathfrak U$ \emph{generates} $\mathfrak U^\sqcup$. The refinement is a Weiss cover by construction: any finite $\{x_1, \ldots, x_k\} \subset X$ with $x_i \in U_{\alpha(i)}$ embeds in a disjoint-union open of $\mathfrak U^\sqcup$ built from small contractible neighbourhoods $V_i \ni x_i$ pairwise disjoint.
\end{definition}

\begin{theorem}[Two descent targets through a single cover]
\label{thm:two-descent-targets}
\ClaimStatusProvedElsewhere
Let $X$ be a smooth complex variety and $\mathfrak U$ an open cover by affine (or Stein) opens. A prefactorisation algebra $\cF_\alpha$ on each $U_\alpha$ together with compatible transition data on overlaps descends to two parallel global objects:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \textbf{Quasi-coherent sheaf of algebras.} The homotopy colimit $\cF_X^{\mathrm{QC}} := \mathrm{hocolim}_{\mathfrak U} \cF_\alpha$ computed in the $(\infty, 1)$-category of sheaves of $\En$-algebras on~$X$ exists whenever $\mathfrak U$ satisfies Cartan--Serre Theorem~B (covering by Stein opens with vanishing higher cohomology of coherent sheaves). Kontsevich--Soibelman 2008 (arXiv:$1006.2706$, §$6.3$) uses this for the motivic cohomological-Hall algebra.
 \item \textbf{Factorisation algebra on~$X$.} The homotopy colimit $\cF_X^{\mathrm{FA}} := \mathrm{hocolim}_{\mathfrak U^\sqcup} \cF_\alpha$ computed on the Weiss refinement $\mathfrak U^\sqcup$ is an $\En$-prefactorisation algebra on~$X$; its value on a disjoint union $U' \sqcup V'$ is the factorisation product $\cF(U') \otimes \cF(V')$, and this datum is invisible on the unrefined $\mathfrak U$ when $\mathfrak U$ contains no disjoint-union open of that type.
\end{enumerate}
The comparison map $\cF_X^{\mathrm{QC}} \to \cF_X^{\mathrm{FA}}$ is the inclusion of single-chart configurations into disjoint configurations; it is an equivalence exactly when the original cover is already Weiss, and strict otherwise.
\end{theorem}

\begin{proof}[Citation]
Part~(i) is Kontsevich--Soibelman 2008 arXiv:$1006.2706$ §$6.3$ for the motivic cohomological-Hall case and the standard Čech--Alexander argument at general $X$ with Stein opens \textup{(}Cartan--Serre Theorem~B\textup{).} Part~(ii) is Costello--Gwilliam Vol.~I \S$6.1.1$--$6.1.2$: by Definition~$6.1.1.1$ \textup{(}lax\textup{)} and Definition~$6.1.2.1$ \textup{(}homotopy cosheaf\textup{)}, a prefactorisation algebra extends to a factorisation algebra iff its Čech complex on every Weiss cover computes the global sections; the refinement $\mathfrak U^\sqcup$ is a Weiss cover by Definition~$6.1.0.6$, so the hocolim on $\mathfrak U^\sqcup$ is the factorisation-algebra incarnation. Beilinson--Drinfeld \emph{Chiral Algebras} \textup{(}AMS~$51$, Chapter~$3$, \S$3.4.11$; the factorisation/chiral-Lie equivalence is Theorem~$3.4.9$\textup{)} treats the holomorphic variant on curves. Francis--Gaitsgory 2011 \textup{(}arXiv:$1103.5803$, Theorem~$1.2.4$\textup{)} establishes the chiral Koszul equivalence between $\mathrm{Lie}\text{-alg}^{\mathrm{ch}}(\Ran X)$ and the factorisation coalgebras $\mathrm{Com}\text{-coalg}^{\mathrm{ch,fact}}(\Ran X)$; Theorem~$5.2.1$ identifies the subcategory of factorisation coalgebras with chiral Lie algebras supported on~$X$ via the $C^{\mathrm{ch}} \dashv \mathrm{Prim}^{\mathrm{ch}}[1]$ adjunction. Francis 2013 \textup{(}arXiv:$1303.0305$\textup{)} establishes the excision / pushforward package for factorisation homology of framed manifolds. Lurie \emph{Higher Algebra} §$5.5.4$ provides the $(\infty, 1)$-category of factorisable cosheaves on $\Ran X$ and the Ran integration functor.
\end{proof}

\begin{example}[Two-chart $\{U_+, U_-\}$ atlas on resolved conifold is not Weiss]
\label{ex:two-chart-not-weiss}
The resolved conifold $\bY = \mathrm{Tot}(\cO_{\bP^1}(-1) \oplus \cO_{\bP^1}(-1))$ admits the two-chart cover $\{U_+, U_-\}$ with $U_\pm \cong \C^3$ (Section~\ref{sec:conifold-two-chart-cech} of Chapter~\ref{ch:toric-coha}). This cover is \emph{not} Weiss: a configuration $S = \{x_+, x_-\}$ with $x_\pm \in U_\pm \setminus U_\mp$ embeds in no single~$U_\pm$. The Weiss refinement $\{U_+, U_-\}^\sqcup$ contains the disjoint-union open $V_+ \sqcup V_-$ with $V_\pm \subset U_\pm$ contractible, and this is the open that houses $\{x_+, x_-\}$. The two descent targets of Theorem~\ref{thm:two-descent-targets} are both load-bearing on $\bY$: the QC descent produces $\mathrm{CoHA}(\bY)$ as a sheaf of associative algebras (Schiffmann--Vasserot 2013 on each chart, Fourier--Mukai glue on overlaps, Kontsevich--Soibelman 2008 descent); the Weiss descent produces the $E_3$-factorisation algebra whose disjoint-configuration product $m_{x_+, x_-}$ is genuinely new data. Both live on the same $\bY$ through the same chart atlas; they are two objects, not one.
\end{example}

\begin{remark}[Orientation: factorisation category versus quasi-coherent sheaf of categories]
\label{rem:fa-vs-qc-category}
The same dichotomy categorifies. For a CY category $\cC$ on $X$, the $\PhiFA_d$-stage produces a factorisation $\En$-algebra in the $(\infty, 1)$-category of dg categories; descent through a non-Weiss chart atlas gives a \emph{quasi-coherent sheaf of $\Eone$-categories} on~$X$, while descent through the Weiss refinement gives an $\En$-factorisation category on~$X$. In the context of Conjecture~CY-C, the Kazhdan--Lusztig and Heisenberg factorisation categories of §\ref{sec:factorization-categories} live at the Weiss level; the NCCR-descent versions of Van~den~Bergh and Bridgeland--King--Reid (arXiv:math/$9908027$, arXiv:math/$0009053$) live at the QC level. The Drinfeld-centre categorical $R$-matrix of §\ref{sec:bf-categorical-r-matrix} is a disjoint-configuration datum: it is visible only on the Weiss refinement.
\end{remark}

\begin{proposition}[Categorical $R$-matrix as the Weiss disjoint-configuration datum]
\label{prop:R-matrix-weiss-datum}
\ClaimStatusProvedElsewhere
For an $\Etwo$-chiral algebra $\cA$ on a curve $C$, the categorical $R$-matrix $\cR_\cA(z) \in \cA^{\hat\otimes 2}$ of Definition~\ref{def:categorical-r-matrix} is the value of the factorisation algebra on a two-point disjoint configuration: $\cR_\cA(z - w)$ encodes the factorisation product $m_{z, w} : \cA(D_z) \otimes \cA(D_w) \to \cA(D_z \sqcup D_w)$ composed with the half-monodromy in $\pi_1(\Conf_2(C)) = \Z$ that exchanges $(z, w) \leftrightarrow (w, z)$ \textup{(}Costello--Gwilliam Vol.~I Chapter~$3$ for the factorisation-product construction; Francis 2013 arXiv:$1303.0305$ §$3$ for the factorisation-homology / Dunn-additivity braiding\textup{).} On a non-Weiss cover, the disjoint open $D_z \sqcup D_w$ may not appear in the cover category, and the $R$-matrix datum is forgotten by the QC-descent hocolim. Weiss refinement restores the datum by construction.
\end{proposition}

\begin{proof}[Scope]
That the $R$-matrix is the holonomy of the factorisation product around $\mathbb S^1 = \Conf_2(\R^2) / \Sigma_2 \simeq \pi_1(\Conf_2(\R^2))$ is the topological content of Ayala--Francis and Costello--Gwilliam Vol.~I Chapter~$3$. Lurie \emph{Higher Algebra} Example~$5.1.2.4$ \textup{(}Braided Monoidal Categories\textup{)} formulates it as the Dunn-additivity braiding on an $\Etwo$-algebra: $\mathrm{Alg}_{\Etwo}(\Cat_\infty) \simeq \mathrm{Alg}_{\Eone}(\mathrm{Alg}_{\Eone}(\Cat_\infty))$ via Theorem~$5.1.2.2$ \textup{(}Dunn Additivity\textup{)}, combined with the Eckmann--Hilton isomorphism $X \otimes Y \simeq Y \otimes X$ coherently parametrised by a loop in $\mathrm{Conf}_2(\R^2)$. The Weiss-forgetting statement is a direct corollary of Theorem~\ref{thm:two-descent-targets}(ii): disjoint-union opens $D_z \sqcup D_w \subset \Conf_2(C)$ are the configuration-space opens that carry the $R$-matrix datum, and QC-descent does not see them.
\end{proof}


\section{The \texorpdfstring{$E_2$}{E-2}-equivariant bar complex}
\label{sec:e2-equivariant-bar}

%% FOUR OBJECTS:
%% 1. B(A) = bar coalgebra = T^c(s^{-1} A-bar) with deconcatenation + twist
%% 2. A^i = H^*(B(A)) = dual coalgebra (Koszul cohomology of bar)
%% 3. A^! = ((A^i)^v) = dual ALGEBRA (linear or Verdier dual)
%% 4. Z^der_ch(A) = derived chiral center = Hochschild cochains = bulk

The passage from the ordered bar complex to the braided bar complex is the central construction of this section.  The question it answers: how does the $R$-matrix, which lives algebraically in degree $(1,1)$ of the bar complex, extend to a coherent braided structure at all degrees?  The answer uses the Dunn decomposition $\Etwo \simeq \Eone \otimes_{E_0} \Eone$: the two $\Eone$-directions each contribute a bar differential, and the $R$-matrix is precisely the intertwiner that makes them commute.

The $E_1$-bar complex $B^{\mathrm{ord}}(\cA) = T^c(s^{-1}\bar{\cA})$
of Volume~I is an ordered coalgebra with deconcatenation coproduct and
a single differential from OPE residues along one real direction.
In the $\Etwo$ setting, the two $\Eone$-directions each contribute
a bar differential. The $\Etwo$-equivariant bar complex intertwines them
through the $R$-matrix.

\begin{definition}[\texorpdfstring{$E_2$}{E-2}-equivariant bar complex]
\label{def:e2-equivariant-bar}
For an $\Etwo$-chiral algebra $\cA$ with augmentation ideal
$\bar{\cA} = \ker(\varepsilon)$, the \emph{$E_2$-equivariant bar
complex} is the graded coalgebra
\[
 B_{E_2}(\cA)
 \;=\;
 T^c(s^{-1}\bar{\cA})
\]
equipped with:
\begin{enumerate}[label=(\roman*)]
 \item two commuting differentials $d_X, d_Y$ from OPE residues
 in each $\Eone$-direction, satisfying $d_X^2 = d_Y^2 = 0$ and
 $d_X \circ d_Y = d_Y \circ d_X$;
 \item two deconcatenation coproducts $\Delta_X, \Delta_Y$ giving
 $B_{E_2}(\cA)$ the structure of an $\Etwo$-coalgebra;
 \item a braid group action: the symmetric group $S_n$ action on
 $B^{\mathrm{ord}}_n(\cA) = (s^{-1}\bar{\cA})^{\otimes n}$ lifts
 to a braid group $B_n$ action via the $R$-matrix
 $\cR_\cA(z)$ (Definition~\ref{def:categorical-r-matrix}).
\end{enumerate}
The total differential is $d = d_X + d_Y$; it satisfies $d^2 = 0$ on
the nose (genus-$0$ bar complex). At genus $g \geq 1$, the fiberwise
differential satisfies $d_{\mathrm{fib}}^2 = \kappa_{\mathrm{ch}} \cdot \omega_g$
(Hodge curvature; see Theorem~\ref{thm:e2-bar-cobar}(iii)).
\end{definition}

\begin{remark}[\texorpdfstring{$B_{E_2}(\cA)$}{B-E-2(cA)} is not a Swiss-cheese coalgebra]
\label{rem:e2-bar-not-sc}
The $\Etwo$-bar complex $B_{E_2}(\cA)$ is an $\Etwo$-coalgebra, \emph{not}
a Swiss-cheese ($\SC^{\mathrm{ch,top}}$) coalgebra. The SC structure of
Vol~II is two-coloured (holomorphic + topological) and emerges on the
\emph{derived center} $\Zder(\cA) = C^\bullet_{\mathrm{ch}}(\cA, \cA)$
(the bulk algebra), not on the bar complex of $\cA$ itself. The bar
complex encodes the boundary; the SC structure governs the bulk-boundary
coupling. Concretely:
\begin{itemize}
 \item $B_{E_2}(\cA)$: $\Etwo$-coalgebra, boundary data,
 deconcatenation coproduct, braid group action;
 \item $\Zder(\cA)$: $\Etwo$-algebra (by the BZF theorem,
 Theorem~\ref{thm:bzfn}), bulk data, carries the SC structure
 when coupled with the boundary $\cA$.
\end{itemize}
The identification $\cZ(\Rep^{E_1}(\cA)) \simeq \Rep^{E_2}(\Zder(\cA))$
(Chapter~\ref{ch:drinfeld-center}) relates the two: the braided monoidal
structure on representations of the bulk is the Drinfeld center of the
monoidal structure on representations of the boundary.
\end{remark}

\begin{proposition}[Three bar complexes and their relation]
\label{prop:three-bars-relation}
\ClaimStatusProvedElsewhere
For an $\Etwo$-chiral algebra $\cA$, the three bar complexes of
Construction~\ref{constr:three-bar-complexes} are related by:
\[
 B_{E_\infty}^n(\cA) = B_{E_1}^n(\cA) / S_n, \qquad
 H^*(B_{E_\infty}^n(\cA)) = H^*(B_{E_2}^n(\cA))^{S_n}.
\]
The $\Eone$-bar is the primitive ordered object; the $\Etwo$-bar
refines the $S_n$-action to the braid group $B_n$ via the $R$-matrix;
the $\Einf$-bar is the $S_n$-coinvariant shadow. The averaging map
$\mathrm{av} \colon B_{E_1} \to B_{E_\infty}$ factors through $B_{E_2}$:
\[
 B_{E_1} \xrightarrow{B_n\text{-equivariantization}} B_{E_2}
 \xrightarrow{B_n \twoheadrightarrow S_n} B_{E_\infty}.
\]
\end{proposition}

\begin{remark}[Bi-based Ran-space and Torelli--Kuga-Satake averaging for \texorpdfstring{$K3 \times E$}{K3 x E}]
\label{rem:brf-k3xe-bi-based-ran-torelli-ks}
The $\Etwo$-bar complex $B_{E_2}(\cA)$ is, by Theorem~\ref{thm:e2-bar-cobar}(i), a factorisation
$\Etwo$-coalgebra on $\Ran(X) \times \Ran(Y)$. For the $K3 \times E$
chiral algebra $A_{K3 \times E}$ (Conjecture~\ref{conj:6d-k3xe}),
this bi-based Ran structure is not a formal placeholder: the two
Ran factors are explicitly
\[
 \mathrm{Ran}^{\mathrm{K3 \times E}}_{\mathrm{bi}}
 \;=\; \mathrm{Ran}\!\bigl(E^{\mathrm{nod,sm}}_{24}\bigr)
 \;\times\; \bar{\cA}_2,
\]
where $E^{\mathrm{nod,sm}}_{24}$ is the $24$-punctured elliptic
curve obtained by removing the $24$ $I_1$-nodal singular points of
a generic elliptic fibration $K3 \to \mathbb{P}^1$ (equivalently,
the smooth locus of the $24$-fibre degeneration forced by
$\chi(K3) = 24$ as in Remark~\ref{rem:brf-delta5-11d-sugra-k3-t2}), and
$\bar{\cA}_2 = \Sp_4(\Z) \backslash \cH_2^{\mathrm{Baily-Borel}}$
is the Baily--Borel compactification of the genus-$2$ Siegel
moduli space carrying the period matrix $\Omega \in \cH_2$ of
$\Sigma_2 \times S^1$. The two factors realise the two
$\Eone$-directions of the Dunn decomposition
$\Etwo \simeq \Eone \otimes_{E_0} \Eone$
(Proposition~\ref{prop:dunn-e2}): the first $\Eone$-direction
lives on the elliptic chiral curve $E^{\mathrm{nod,sm}}_{24}$
(supplying the 2d holomorphic CFT data of the $K3$ elliptic genus),
the second lives on $\bar{\cA}_2$ (supplying the genus-$2$ Siegel
modularity of the block partition function).

The averaging map $\mathrm{av} \colon B_{E_1} \to B_{E_\infty}$
factoring through $B_{E_2}$ (the composition
$B_n \twoheadrightarrow S_n$ above) admits, on this bi-based
Ran space, the explicit factorisation
\[
 \mathrm{av}_{K3 \times E}
 \;=\;
 \underbrace{\mathrm{Torelli}_{K3}}_{\Sigma_g \,\to\, \cA_g^{\mathrm{Muk}}}
 \;\circ\;
 \underbrace{\mathrm{KugaSatake}}_{\cA_g^{\mathrm{Muk}} \,\to\, \cA_2}
 \;\circ\;
 \underbrace{j_{\mathrm{Kodaira}}}_{E^{\mathrm{nod,sm}}_{24} \,\hookrightarrow\, \bar{\cA}_2},
\]
where $j_{\mathrm{Kodaira}}$ is the classical Kodaira $j$-map
($E$-modulus to $\cH_1 \hookrightarrow \cH_2$ by block diagonal
embedding, sending an $I_1$-nodal fibre to its monodromy fixed
point), $\mathrm{KugaSatake}$ is the Kuga--Satake construction
(Deligne~1972 \emph{Lect.~Notes Math.}~244; attaching to each
$K3$-type Hodge structure on $\Lambda_{\mathrm{Muk}}$ an abelian
variety of genus~$2$ so the Hodge structure becomes a Tate twist
of the abelian one), and $\mathrm{Torelli}_{K3}$ is the
$K3$ Torelli isomorphism
(Piatetski-Shapiro--Shafarevich 1971, Burns--Rapoport 1975)
identifying the $K3$ moduli space with its period domain
$\cD_{K3} / O(\Lambda_{\mathrm{Muk}})^+ \subset \cA_g^{\mathrm{Muk}}$.
This triple composition is the $\Sigma_2$-coinvariant map that
averages the $B_n$-equivariant bi-based Ran data to the scalar
shadow $\kappa_{\mathrm{ch}}(A_{K3 \times E})$, and its kernel is
precisely the $\Etwo$-level $R$-matrix information forgotten by
averaging. This is the $K3 \times E$ instantiation of
Proposition~\ref{prop:three-bars-relation}: the bi-based Ran
space supplies the $B_{E_2}$-content, and the triple Torelli /
Kuga-Satake / Kodaira map averages it to Vol~I's scalar shadow
(cf.\ Chapter~\ref{ch:k3-chiral-bialgebra-platonic} for the
chiral-bialgebra target and Chapter~\ref{ch:k3e-bkm} for the BKM
realisation).
\end{remark}


\section{The braided bar-cobar adjunction}
\label{sec:braided-bar-cobar}

The Vol~I bar-cobar adjunction (Theorems~A and~B) operates at the $\Eone$
level: $B^{\mathrm{ord}}$ is an $\Eone$-coalgebra and $\Omega$ recovers
the $\Eone$-algebra. The $\Etwo$-refinement lifts the adjunction to
braided coalgebras, with the $R$-matrix providing the additional datum
that distinguishes $\Etwo$ from $\Eone$.

\begin{theorem}[\texorpdfstring{$E_2$}{E-2}-bar-cobar adjunction: Theorem CY-B]
\label{thm:e2-bar-cobar}
\ClaimStatusConditional
For an $\Etwo$-chiral algebra $\cA$, the bar-cobar adjunction
\[
 B_{E_2} \colon E_2\text{-}\mathrm{ChirAlg}
 \rightleftarrows E_2\text{-}\mathrm{ChirCoalg}
 \cocolon \Omega_{E_2}
\]
satisfies:
\begin{enumerate}[label=(\roman*)]
 \item The $\Etwo$ bar complex $B_{E_2}(\cA)$ is a factorization
 $\Etwo$-coalgebra on $\Ran(X) \times \Ran(Y)$;
 \item Cobar-bar inversion:
 $\Omega_{E_2}(B_{E_2}(\cA)) \simeq \cA$ on the Koszul locus;
 \item The CY trace manifests as curvature: at genus $g \geq 1$,
 $d_{\mathrm{fib}}^2 = \kappa_{\mathrm{ch}} \cdot \omega_g$,
 where $\omega_g = c_1(\lambda)$ is the Hodge class on
 $\overline{\cM}_g$ and $\kappa_{\mathrm{ch}} = \chi^{\CY}(\cC)$.
\end{enumerate}
\end{theorem}

\begin{remark}[Status of Theorem CY-B]
\label{rem:cy-b-status}
Items~(i) and~(ii) follow from the Volume~I bar-cobar machine
(Theorems~A and~B) applied to each $\Eone$-direction separately, with
compatibility ensured by Dunn additivity
(Proposition~\ref{prop:dunn-e2}). The argument proceeds in three steps.

\emph{Step 1: $\Eone \times \Eone$ bar-cobar.} Apply the Vol~I
bar-cobar adjunction to each $\Eone$-factor independently. This produces
two commuting bar-cobar adjunctions $(B_X, \Omega_X)$ and
$(B_Y, \Omega_Y)$.

\emph{Step 2: $R$-matrix intertwining.} The $R$-matrix
$\cR_\cA(z) \in \End(V \otimes V)[[z, z^{-1}]]$ at degree $(1,1)$
intertwines the two bar differentials: $\cR \circ d_X = d_Y \circ \cR$
on the degree-$(1,1)$ component. This is the braided compatibility
condition.

\emph{Step 3: Higher coherences.} The full $\Etwo$-equivariant
refinement requires coherent intertwining at all degrees, governed by
the braid group $B_n$ action. Steps~1 and~2 are proved; the detailed
verification of higher coherences (degree $\geq 4$) is established as a
rigorous proof sketch. The result is unconditional at degree $\leq 3$
and conditional on the higher-degree coherences beyond that.

Item~(iii) uses Theorem~D of Volume~I for the chiral modular
characteristic $\kappa_{\mathrm{ch}}(\Phi(\cC))$; in the $d = 2$
case at hand, Theorem~\ref{thm:cy-modular-characteristic} identifies
this with the CY Euler characteristic
$\kappa_{\mathrm{cat}} = \chi^{\CY}(\cC)$. The curvature
$d_{\mathrm{fib}}^2 = \kappa_{\mathrm{ch}}(\Phi(\cC)) \cdot \omega_g$ is the
\emph{fiberwise} statement on $\overline{\cM}_g$; the bar differential
$d^2_{\mathrm{bar}} = 0$ always holds (the curvature is Hodge, not bar).
\end{remark}

\begin{proposition}[Degree-\texorpdfstring{$(1,1)$}{(1,1)} of \texorpdfstring{$B_{E_2}$}{B-E-2}]
\label{prop:e2-bar-degree-11}
\ClaimStatusProvedHere
The degree-$(1,1)$ component of $B_{E_2}(\cA)$ is the space
$\End(V \otimes V)$ equipped with the $R$-matrix
$\cR_\cA(z)$. The bar differential restricted to this component
recovers the unitarity condition $\cR_{12}(z) \cR_{21}(-z) = \id$
(strong unitarity), and the braiding constraint
$\cR_{12}(z) = \tau \circ \sigma_{12}(z)$.
\end{proposition}

\begin{proof}
The $E_2$-bar complex at degree $(1,1)$ consists of elements
$\alpha \in (s^{-1}\bar{\cA})^{\otimes 2}$ with one element
from each $E_1$-direction. The bar differential
$d(\alpha) = \mu(\alpha)$ is the OPE residue extracted via
$d\log(z_1 - z_2)$ (Volume~I, bar differential). The
compatibility of the two $E_1$-structures forces
$d_{X} \circ d_{Y} = d_{Y} \circ d_{X}$; evaluating on
$(s^{-1}v) \otimes (s^{-1}w)$ gives
$\cR_{12}(z)\cR_{21}(-z) = \id$ (strong unitarity).
The braiding $\sigma_{12}(z) = \cR_{12}(z) \circ \tau^{-1}$
follows from the identification of $\cR$ with the
half-braiding (Definition~\ref{def:categorical-r-matrix}).
\end{proof}


\section{Quantum group structure from braided Koszul duality}
\label{sec:qg-from-braided-koszul}

The Vol~I Koszul dual $A^! = (H^*(B(A))^{\vee})$ is a linear-dual
algebra. In the $\Etwo$ setting, the bar cohomology
$H^*(B_{E_2}(\cA))$ carries a quasi-triangular Hopf algebra structure
from the $R$-matrix, and the question becomes: does this Hopf algebra
coincide with the quantum group $\Uq(\frakg)$ when the input category
has $\frakg$-symmetry?

\begin{conjecture}[Braided bar cohomology recovers \texorpdfstring{$\Uq(\frakg)$}{Uq(frakg)}]
\label{conj:braided-bar-uq}
\ClaimStatusConjectured
For the $\Etwo$-chiral algebra $\cA = \Phi(\cC)$ associated to a
CY$_2$ category $\cC$ with $\frakg$-symmetry, the braided bar
cohomology
\[
 H^*(B_{E_2}(\cA)) \;\simeq\; \Uq(\frakg)
\]
as quasi-triangular Hopf algebras, where $q = e^{2\pi i / (k + h^\vee)}$
and $k$ is the level determined by the chiral modular characteristic
$\kappa_{\mathrm{ch}}(\cA)$.
\end{conjecture}

\begin{remark}[Evidence and scope]
\label{rem:braided-bar-evidence}
Conjecture~\ref{conj:braided-bar-uq} is verified in two cases:
\begin{enumerate}[label=(\roman*)]
 \item For $\cC = D^b(\mathrm{Coh}(\mathrm{pt}/G))$ with $G$ simple
 and simply connected, the CY-to-chiral correspondence $\Phi_2$ produces
 $V_k(\frakg)$ (by CY-A at $d = 2$). The $\Etwo$-bar cohomology
 should recover $\Uq(\frakg)$ via the Kazhdan--Lusztig equivalence
 (Theorem~\ref{thm:qgf-kazhdan-lusztig}). This is verified at the
 level of categories: $\Rep_q(\frakg) \simeq \cO_{k}(\hat{\frakg})$
 at roots of unity.
 \item For $\cC = D^b(\mathrm{Coh}(K3))$, the chiral algebra is a lattice
 VOA and the $\Etwo$-bar cohomology recovers a lattice quantum group
 (Example~\ref{ex:braided-k3}).
\end{enumerate}
The conjecture is open for non-geometric CY categories and for $d = 3$:
the functor $\Phi_3$ is constructed at the infinity-categorical level
(Theorem~\ref{thm:cy-to-chiral-d3}), but the quantum-group target CY-C
remains conjectural.
\end{remark}

\begin{remark}[Three dualities]
\label{rem:braided-koszul-langlands}
%: Three distinct operations must be kept separate.
The braided Koszul dual $\cA^!_{E_2}$ produces the quantum group
$\Uq(\frakg)$. This must be distinguished from:
\begin{enumerate}[label=(\alph*)]
 \item the Feigin--Frenkel involution $k \mapsto -k - 2h^\vee$, which
 acts within the same family of affine algebras;
 \item the Langlands dual $\frakg \mapsto \frakg^L$ (root-coroot
 exchange), which produces a genuinely different algebra.
\end{enumerate}
For $\fsl_2$ (self-Langlands-dual), all three operations coincide on
representation categories. For non-simply-laced types (e.g.,
$G_2, F_4$), they diverge. The braided Koszul dual exchanges the
braiding parameter $q \leftrightarrow q^{-1}$; the Langlands dual
exchanges the Cartan matrix $A \leftrightarrow A^T$; the Feigin--Frenkel
involution exchanges levels $k \leftrightarrow -k - 2h^\vee$.
\end{remark}


\section{The braided shadow obstruction tower}
\label{sec:braided-shadow-tower}

The Vol~I shadow obstruction tower $\Theta_A$ is the $\Sigma_n$-coinvariant
projection of a richer $E_1$ object. In the $\Etwo$ setting, the braided
shadow tower $\Theta^{E_2}_\cA$ sits between the ordered $E_1$ tower and
the symmetric $E_\infty$ shadow, with the braid group $B_n$ replacing
the symmetric group $S_n$.

\begin{definition}[Braided modular characteristic]
\label{def:braided-kappa}
For an $\Etwo$-chiral algebra $\cA$, the \emph{braided modular
characteristic} is the pair
\[
 (\kappa_{\mathrm{ch}}(\cA), \, \cR_\cA(z))
\]
consisting of the scalar modular characteristic
$\kappa_{\mathrm{ch}} = \mathrm{av}(\cR_\cA(z))$ (the
$\Sigma_2$-coinvariant of the $R$-matrix) and the full
$R$-matrix itself. The scalar $\kappa_{\mathrm{ch}}$ controls
the genus tower
$F_g = \kappa_{\mathrm{ch}} \cdot \lambda_g^{FP}$ (UNIFORM-WEIGHT);
the $R$-matrix controls the ordered/quantum-group sector.
\end{definition}

\begin{remark}[\texorpdfstring{$\kappa_{\mathrm{ch}}$}{kappa-ch} vs \texorpdfstring{$\kappa_{\mathrm{cat}}$}{kappa-cat} in the braided context]
\label{rem:braided-kappa-cat-vs-ch}
The scalar $\mathrm{av}(\cR_\cA(z))$ is an \emph{algebraization invariant}:
it depends on the constructed algebra $\cA$, and is properly denoted
$\kappa_{\mathrm{ch}}(\cA)$.  The manifold invariant
$\kappa_{\mathrm{cat}} = \chi(\cO_X)$ is topological and independent of
algebraization.  At $d = 2$, Proposition~\ref{prop:kappa-cat-chi-cy}
proves $\kappa_{\mathrm{ch}} = \kappa_{\mathrm{cat}}$; at $d \ge 3$ they
may differ.
\end{remark}

\begin{proposition}[Braided shadow structure]
\label{prop:braided-shadow}
\ClaimStatusProvedHere
The $\Etwo$-bar complex $B_{E_2}(\cA)$ carries a braided shadow
obstruction tower $\Theta^{E_2}_\cA$ whose degree-by-degree
projection under the averaging map recovers the Volume~I
shadow obstruction tower:
\[
 \mathrm{av}(\Theta^{E_2}_{\cA, n})
 = \Theta_{A_\cC, n}
 \quad \text{for all $n \geq 2$.}
\]
At degree $2$: $\mathrm{av}(\cR(z)) = \kappa_{\mathrm{ch}}$.
At degree $3$: $\mathrm{av}(\Phi_{KZ}) = C$ (the cubic shadow of
Volume~I).
\end{proposition}

\begin{proof}
The averaging map $\mathrm{av} \colon B_{E_2}(\cA) \to B_{E_\infty}(\cA)$
is the composition $B_{E_2} \xrightarrow{B_n \twoheadrightarrow S_n}
B_{E_\infty}$ of Proposition~\ref{prop:three-bars-relation}. The
shadow obstruction tower $\Theta^{E_2}_{\cA, n}$ is the degree-$n$
component of the $\Etwo$-bar complex with its $B_n$-action; the
Vol~I shadow tower $\Theta_{A_\cC, n}$ is the $S_n$-coinvariant of
the same underlying data. Since $\mathrm{av}$ is precisely the
$S_n$-coinvariant projection (taking the $B_n$-equivariant data and
quotienting by the surjection $B_n \twoheadrightarrow S_n$), the
identity $\mathrm{av}(\Theta^{E_2}_{\cA, n}) = \Theta_{A_\cC, n}$
holds by construction for all $n \geq 2$.

At degree $2$: the $\Etwo$-shadow is the $R$-matrix
$\cR(z) \in \End(V \otimes V)((z))$, and its $S_2$-coinvariant
$\mathrm{av}(\cR(z)) = \frac{1}{2}(\cR_{12}(z) + \cR_{21}(-z))$
reduces to the scalar $\kappa_{\mathrm{ch}}$ by strong unitarity
$\cR_{12}(z)\cR_{21}(-z) = \mathrm{id}$
(Proposition~\ref{prop:e2-bar-degree-11}).

At degree $3$: the $\Etwo$-shadow is the Knizhnik--Zamolodchikov
associator $\Phi_{KZ}$, and its $S_3$-coinvariant is the cubic
shadow invariant $C$ of Vol~I by the same projection mechanism.
\end{proof}

\begin{remark}[The three shadow tiers]
\label{rem:three-shadow-tiers}
The three bar complexes produce three tiers of the shadow tower:
\begin{center}
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{lll}
 \toprule
 Bar complex & Group action & Shadow data \\
 \midrule
 $B_{E_1}$ (ordered) & trivial ($1$) & full $R$-matrix $\cR(z)$, Yangian \\
 $B_{E_2}$ (braided) & braid group $B_n$ & $B_n$-equivariant tower, quantum group \\
 $B_{E_\infty}$ (symmetric) & symmetric $S_n$ & scalar $\kappa_{\mathrm{ch}}$, genus tower \\
 \bottomrule
\end{tabular}
\end{center}
Each tier retains strictly less information than the one above.
The averaging map $\mathrm{av}$ is the $B_n \twoheadrightarrow S_n$
projection applied to each degree. It is lossy: the $R$-matrix is
forgotten, and only its $S_2$-coinvariant $\kappa_{\mathrm{ch}}$
survives.
\end{remark}

\begin{example}[Braided structure for \texorpdfstring{$K3$}{K3}; Hodge supertrace identification]
\label{ex:braided-k3}
For a K3 surface $S$ with $\cC = D^b(\Coh(S))$, the
$\Etwo$-chiral algebra has $\kappa_{\mathrm{ch}}(\Phi(\cC)) = 2$
which at $d = 2$ coincides with the manifold invariant
$\kappa_{\mathrm{cat}} = \chi(\cO_S) = 2$. The
$R$-matrix is the Casimir
$\cR(z) = 1 + \kappa_{\mathrm{ch}}\,\Omega/z + O(z^{-2})$,
where $\Omega$ is the Mukai pairing on $H^*(S, \Z)$. The braided
bar cohomology $H^*(B_{E_2}(\Phi(\cC)))$ recovers the lattice
quantum group associated to the Mukai lattice
$\Lambda_{K3} = U^3 \oplus E_8(-1)^2$.
The shadow class is $\mathbf{G}$ (Gaussian, $r_{\max} = 2$):
the lattice VOA is a free-field algebra and the shadow tower
terminates at degree $2$ (both at the $E_1$ and $E_2$ levels).

The value $\kappa_{\mathrm{ch}}(K3) = 2$ admits a transparent
\emph{Hodge supertrace} expression. The identification
$\kappa_{\mathrm{ch}}(\Phi_2(\cC)) = \chi(\cO_S)$ is a theorem
(Proposition~\ref{prop:kappa-cat-chi-cy}; route-independence of
$\kappa_{\mathrm{ch}}$ via categorical invariance of $\Phi_2$,
specialized at $d = 2, h^{1,0}(S) = 0$), not a definition: the
Hochschild--Kostant--Rosenberg decomposition together with the
Mukai-pairing identification of the Calabi--Yau categorical trace
gives
\[
 \kappa_{\mathrm{ch}}(\Phi_2(\cC))
 \;=\; \chi(\cO_S)
 \;=\; \sum_{q \ge 0} (-1)^q \, h^{0,q}(S),
\]
and for a K3 surface the Hodge diamond yields
$h^{0,0}(K3) = 1$, $h^{0,1}(K3) = 0$, $h^{0,2}(K3) = 1$, hence
\[
 \kappa_{\mathrm{ch}}(K3)
 \;=\; 1 - 0 + 1
 \;=\; 2.
\]
This is the Calabi--Yau realisation of the $\Sigma_2$-coinvariant
$\mathrm{av}(\cR(z)) = \kappa_{\mathrm{ch}}$ of
Definition~\ref{def:braided-kappa}: the Hodge diamond of $S$ supplies
the $E_2$-bar shadow at degree $2$, with the holomorphic symplectic
form $\sigma \in H^{0,2}(K3)$ (generator of $h^{0,2} = 1$) as the
geometric source of the second supertrace contribution. Compare
Chapter~\ref{ch:cy-d-kappa-stratification} for the full
$d$-stratification.
\end{example}


\section{Braided complementarity}
\label{sec:braided-complementarity}

Vol~I Theorem~C establishes complementarity:
$\kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!) = K$
(the Koszul conductor, family-dependent). The $\Etwo$-refinement
upgrades this scalar relation to an $R$-matrix-level statement.

\begin{conjecture}[Braided complementarity]
\label{conj:braided-complementarity}
\ClaimStatusConjectured
For an $\Etwo$-chiral algebra $\cA$ on the Koszul locus, the braided
Koszul dual $\cA^!_{E_2}$ satisfies:
\begin{enumerate}[label=(\roman*)]
 \item \emph{Scalar complementarity}:
 $\kappa_{\mathrm{ch}}(\cA) + \kappa_{\mathrm{ch}}(\cA^!_{E_2}) = K$,
 the family-dependent Koszul conductor (Vol~I, Theorem~C);
 \item \emph{$R$-matrix inversion}:
 $\cR_{\cA^!_{E_2}}(z) = \cR_\cA(z)^{-1}$ as formal power series
 in $z^{-1}$, so the braided Koszul dual reverses the braiding;
 \item \emph{Conductor from $R$-matrix}:
 the Koszul conductor $K$ is recovered from the $R$-matrix
 as $K = \mathrm{av}(\cR_\cA(z)) + \mathrm{av}(\cR_\cA(z)^{-1})$.
\end{enumerate}
\end{conjecture}

\begin{remark}[Status of braided complementarity]
\label{rem:braided-complementarity-status}
Item~(i) is an immediate consequence of Vol~I Theorem~C applied
to the underlying $\Einf$-data. Item~(ii) is the genuine $\Etwo$
content: it asserts that Koszul duality acts on the $R$-matrix by
inversion, so $q \mapsto q^{-1}$. For the Kazhdan--Lusztig
equivalence, this corresponds to the duality
$\Rep_q(\frakg) \simeq \Rep_{q^{-1}}(\frakg)^{\mathrm{rev}}$,
which is the standard anti-braiding equivalence for quantum groups.
Item~(iii) follows formally from (i) and (ii), since
$\mathrm{av}(\cR^{-1}) = \kappa_{\mathrm{ch}}(\cA^!_{E_2})$.
The conjecture is conditional on Theorem~CY-B (the braided bar-cobar
adjunction at the $\Etwo$ level) and on
Conjecture~\ref{conj:braided-bar-uq} (braided bar cohomology
recovering the quantum group).
\end{remark}

\begin{remark}[The Siegel-elliptic dynamical \texorpdfstring{$R$}{R}-matrix as a fourth \texorpdfstring{$R$}{R}-matrix class]
\label{rem:brf-siegel-elliptic-dynamical-r-matrix}
Braided complementarity operates on $\cR_\cA(z)$ viewed as a
dynamical $R$-matrix on an elliptic curve. The $K3 \times E$ BKM
datum supplies a strictly richer object: the
\emph{Siegel-elliptic dynamical $R$-matrix}
$R_{\mathrm{Sieg,\,dyn}}(u, v \mid \Omega)$, built from the
genus-$2$ Riemann theta via the Pasol--Zagier~2013 extension of
the classical Kronecker--Eisenstein series. Written as a
generating series in $(u, v)$ with Siegel modulus
$\Omega = \bigl(\begin{smallmatrix} \tau_1 & z \\ z & \tau_2
\end{smallmatrix}\bigr) \in \cH_2$,
\[
 R_{\mathrm{Sieg,\,dyn}}(u, v \mid \Omega)
 \;=\; \sum_{n \geq 0} \frac{1}{n!}\,
 \frac{\theta^{(n)}\!\bigl[\begin{smallmatrix}a\\b\end{smallmatrix}\bigr]\!(u, v \mid \Omega)}{\theta\!\bigl[\begin{smallmatrix}a\\b\end{smallmatrix}\bigr]\!(0, 0 \mid \Omega)}
 \;\cdot\; (\text{spectral tensor}),
\]
where the genus-$2$ theta constants replace the genus-$1$ Jacobi
theta used in the Felder 1994 dynamical $R$-matrix on $E$, and the
Pasol--Zagier 2013 (\emph{Publ.~Math.~IHES}~118) extension supplies
the higher-genus Kronecker quasi-period. This is a \emph{fourth}
taxonomic $R$-matrix class distinct from the three classical
families:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \emph{rational} $r(z) = \Omega/z$ (Yangian, genus~$0$),
 \item \emph{trigonometric} $r(z) = \Omega\,\coth(z)$ ($U_q(\widehat{\frakg})$, genus~$0$ with loop),
 \item \emph{elliptic} $r(z \mid \tau) = \Omega\cdot \zeta(z \mid \tau)$ (Felder, Belavin 1981, genus~$1$),
 \item \emph{Siegel-elliptic dynamical} $R_{\mathrm{Sieg,dyn}}(u, v \mid \Omega)$ (this remark, genus~$2$).
\end{enumerate}
Class~(iv) satisfies a \emph{dynamical Yang--Baxter equation on the
chromatic layer $K(1)$} (first Morava $K$-theory at $p = 2$): on
the $K(1)$-local sector the BKM imaginary-root multiplicities

reduces to Felder's elliptic DYBE on $E$. Off the $K(1)$ layer,
the Siegel-elliptic $R$ picks up an anomaly proportional to the
Jacobi index $1/2$ of $\Delta_5$ (the paramodular cusp form of
weight~$5$ on $\Gamma_{\mathrm{para}}$):
\[
 \text{YBE}\!\left[R_{\mathrm{Sieg,\,dyn}}\right]
 \;=\; \bigl(1 - \chi_{K(1)}\bigr) \cdot
 \tfrac{1}{2} \, \partial_z \log \Delta_5(\Omega) \cdot (\text{obstruction tensor}),
\]
so the YBE holds exactly on $K(1)$ and is obstructed off~$K(1)$ by
the Gritsenko--Nikulin~1998 paramodular Siegel form.

The fourth-class claim is buttressed by three faces of the
conductor~$8$ in the $K3 \times E$ BKM landscape:
\[
 K^{\kappa_{\mathrm{ch}}}
 \;=\; 2\,c_+\!\bigl(\mathrm{Mukai}(K3)\bigr)
 \;=\; 8
 \;=\; \mathrm{ord}\!\bigl(H_1\text{-monodromy}\bigr)
 \;=\; \ell_{\mathrm{Lusztig}},
\]
where $c_+ = 4$ is the positive-signature part of the Mukai
lattice (rank-$24$ of signature $(4, 20)$), $\mathrm{ord}(H_1
\text{-mono}) = 8$ is the order of the Picard--Lefschetz monodromy

$\kappa_{\mathrm{BKM}} = \kappa_{\mathrm{ch}}^{\mathrm{Heis}}(A_{K3 \times E})
 + \chi(\cO_{\mathrm{fiber}}) = 3 + 2 = 5$
(Remark~\ref{rem:brf-delta5-11d-sugra-k3-t2}) feeds the universal-$\hbar$ relation

\end{remark}


\section{The categorical \texorpdfstring{$S$}{S}-matrix and \texorpdfstring{$E_3$}{E-3} scattering}
\label{sec:categorical-s-matrix}

At $\Etwo$, the categorical $R$-matrix $\cR_{V,W}(z) \colon V \otimes W
\to W \otimes V$ encodes line-operator braiding. The modular $S$-matrix
$S_{ij} = \tr_i(\cR_{ij})$ arises by taking the trace over one factor:
it is a \emph{number} (the Hopf link invariant), and when
$\Rep^{E_2}(\cA)$ is a modular tensor category, $S$ governs the
$\SL_2(\Z)$ action on conformal blocks. At $E_3$, the braiding is a
surface phenomenon; the relevant topology is $3$-dimensional, and
the analogue of the $S$-matrix is a \emph{categorical $2$-morphism},
not a scalar. This section formulates the $E_3$ categorical $S$-matrix
and connects it to Siegel modular forms.

\begin{construction}[From \texorpdfstring{$\Etwo$}{E2} \texorpdfstring{$S$}{S}-matrix to \texorpdfstring{$E_3$}{E-3} categorical
\texorpdfstring{$S$}{S}-matrix]
\label{constr:e3-s-matrix}
The construction proceeds by dimensional promotion. At $\Etwo$, the
$S$-matrix is the partial trace of the $R$-matrix:
\[
 S_{V,W} \;=\; \tr_W\!\bigl(\cR_{V,W}(z) \circ \cR_{W,V}(z)\bigr)
 \;\in\; \End(V).
\]
This is the Hopf link invariant: two circles linked in $\R^3$, each
carrying a representation. At $E_3$, the relevant braiding operates on
\emph{surface operators} (not lines), since $E_3$ factorization governs
$3$-dimensional topology where the codimension-$2$ objects are surfaces.
The $E_3$ categorical $S$-matrix is:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \emph{Input}: a pair of objects $\cV, \cW$ in the $2$-category
 $\Rep^{E_3}(\cA)$ of surface-operator representations of the
 $E_3$-chiral algebra.
 \item \emph{Operation}: the $E_3$ $R$-matrix
 $\cR^{E_3}_{\cV,\cW}(u,v)$ is a $2$-morphism (a natural
 transformation between functors), depending on two spectral
 parameters $(u,v)$ from the two extra $\C$-directions of the
 $E_3$ factorization on $\C^3$. The categorical $S$-matrix is
 the \emph{categorical trace}
 \[
  \cS_{\cV,\cW}(u,v)
  \;=\;
  \Tr_{\cW}\!\bigl(
   \cR^{E_3}_{\cV,\cW}(u,v) \circ \cR^{E_3}_{\cW,\cV}(u,v)
  \bigr)
  \;\in\; \End_{\mathrm{2\text{-}cat}}(\cV),
 \]
 where $\Tr_{\cW}$ is the categorical trace (Hochschild homology of
 the endomorphism category of $\cW$). The result is an
 \emph{endomorphism} of $\cV$ in the $2$-category, not a scalar.
 \item \emph{Topology}: this $2$-morphism is the invariant of two
 linked $2$-surfaces in the $5$-sphere $S^5$. The operadic
 analogue of the Hopf link $S^1 \hookrightarrow S^3$ at
 $\Etwo$ is a linked pair $S^2 \hookrightarrow S^5$ (boundaries
 of $3$-disks in $\R^5 \cup \{\infty\}$). For surface
 operators wrapping tori, the relevant embedding is
 $T^2 \hookrightarrow S^5$; such a linking is well-defined
 since $2 \cdot \dim(T^2) + 1 = 5 = \dim(S^5)$.
\end{enumerate}
\end{construction}

\begin{conjecture}[Double \texorpdfstring{$S$}{S}-matrix for the quantum toroidal algebra]
\label{conj:double-s-matrix}
\ClaimStatusConjectured
Let $\cA = U_{q,t}(\widehat{\widehat{\fgl}}_1)$ be the quantum
toroidal algebra with parameters $q = e^{2\pi i \tau_1}$,

 S(\tau_1, \tau_2)

  q^{L_0}\, t^{L_0'}\,
  \cR^{E_3}(u,v) \circ \cR^{E_3}(v,u)
 \right)

\]
where $L_0, L_0'$ are the two Virasoro zero-modes of the
quantum toroidal algebra corresponding to the two loop
directions. The product is the square of the MacMahon-type
generating function, reflecting the two spectral parameters

Conjecture~\textup{\ref{conj:chiral-qg-c3}(ii)}.
\end{conjecture}

\begin{remark}[The \texorpdfstring{$\Etwo \to E_3$}{E2 -> E-3} modular promotion mechanism]
\label{rem:e2-e3-modular-promotion}
The passage from $\SL_2(\Z)$ to $\Sp_4(\Z)$ is not an \emph{ad hoc}
enlargement: it is forced by the factorization homology of the
$E_3$-algebra over the mapping class group of $3$-manifolds. The
concrete mechanism:
\begin{enumerate}[label=\textup{(\alph*)}]
 \item At $\Etwo$: factorization homology $\int_{T^2} \cA$ computes
 the genus-$1$ amplitude. The moduli space of $T^2$ is
 $\SL_2(\Z) \backslash \cH_1$. The $S$-transformation $\tau \mapsto
 -1/\tau$ and $T$-transformation $\tau \mapsto \tau + 1$ generate
 $\SL_2(\Z)$. The $S$-matrix \emph{is} the $S$-transformation of
 conformal blocks.
 \item At $E_3$: factorization homology $\int_{\Sigma_2 \times S^1} \cA$
 computes the genus-$2$ Siegel amplitude. The moduli space of
 genus-$2$ Riemann surfaces is $\Sp_4(\Z) \backslash \cH_2$.
 The $E_3$ $S$-matrix is the $\Sp_4(\Z)$ transformation matrix
 of genus-$2$ conformal blocks. The two spectral parameters
 $(\tau_1, \tau_2)$ are the diagonal entries of the period matrix;
 the off-diagonal entry $z$ is the relative spectral parameter
 from the $E_3$ $R$-matrix $R_{\mathrm{ch}}(u,v)$.
 \item The Fourier--Jacobi expansion $\Phi_{10} = \sum_m \phi_m\, p^m$
 is the $\Etwo \hookrightarrow E_3$ restriction: each
 $\phi_m(\tau_1, z)$ is an $\SL_2(\Z)$ Jacobi form (the $\Etwo$
 datum), and the assembly into a Siegel form is the $E_3$ datum.
 The index $m$ is the Fourier--Jacobi charge, which in the
 categorical picture is the degree of the braided shadow tower
 $\Theta^{E_2}_{\cA, m}$.
\end{enumerate}
The factorization homology perspective explains \emph{why} the
genus-$2$ modular group appears: $E_3$ factorization on a
$3$-manifold $M^3$ produces invariants that depend on the moduli of
$M^3$; for $M^3 = \Sigma_g \times S^1$, these moduli include the
Siegel period matrix of $\Sigma_g$. At $g = 2$, the period matrix
lives in $\cH_2$ with $\Sp_4(\Z)$ symmetry.
\end{remark}

\begin{remark}[Scope and status]
\label{rem:e3-s-matrix-scope}
Construction~\ref{constr:e3-s-matrix} is a mathematical definition
(of the categorical $S$-matrix as a categorical trace of the
$E_3$ $R$-matrix). Conjectures~\ref{conj:double-s-matrix},
\ref{conj:sp4-modularity}, and~\ref{conj:s-matrix-phi10} are
conjectural: the first requires the existence of the $E_3$ $R$-matrix
on Fock modules (Conjecture~\ref{conj:chiral-qg-c3}); the second
requires the $\Sp_4(\Z)$ modularity of the resulting function;
the third requires the full $K3 \times E$ programme
(CY-A$_3$, Theorem~\ref{thm:cy-to-chiral-d3}, plus the BKM
identification). The three conjectures remain conditional on the
existence of the $E_3$ $R$-matrix on Fock modules through the
dependency chain $A_X \to R^{E_3} \to \cS \to \hat{S}$.
None of the conjectures above is a theorem. They are formulated
here because the structural parallels
($\Etwo/\SL_2(\Z)/S_{ij}$ vs.\ $E_3/\Sp_4(\Z)/\hat{S}$) are
sufficiently rigid to be falsifiable: compute the $E_3$ factorization
homology of the quantum toroidal Fock module over $\Sigma_2 \times S^1$
and check whether $\Sp_4(\Z)$ modularity holds.
\end{remark}

\begin{remark}[Computational evidence: the \texorpdfstring{$\Sp_4(\Z)$}{Sp-4(Z)} modularity pipeline]
\label{rem:sp4-pipeline-evidence}
The pipeline of Conjectures~\ref{conj:sp4-modularity}
and~\ref{conj:s-matrix-phi10} has been implemented numerically in
\texttt{sp4\_modularity\_pipeline.py} (53 tests).
Three results sharpen the conjectural picture:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \emph{ZTE trace vanishing.}
 The Zamolodchikov tetrahedron obstruction $\Delta = \mathrm{LHS} - \mathrm{RHS}$
 on the charge-$2$ sector is \emph{antisymmetric}
 ($\Delta + \Delta^T = 0$, Theorem~\ref{thm:zte-failure}).
 Therefore $\Tr(\Delta) = 0$: the \emph{scalar} $S$-matrix trace
 $\hat{S} = \Tr_{\cW}(\cS)$ receives \emph{no} correction at
 $O(\hbar_{\mathrm{Y}}^2)$ from the tetrahedron equation.
 The factored Zamolodchikov approximation
 $\cS^{\mathrm{fact}} = \cS(u)\cS(v)\cS(u{-}v)$
 is exact for the scalar trace through $O(\hbar_{\mathrm{Y}}^2)$;
 the first correction is $O(\hbar_{\mathrm{Y}}^4)$.
 This explains why the factored expression can exhibit approximate
 $\Sp_4(\Z)$ covariance.

 \item \emph{Fourier--Jacobi structure.}
 The $\Etwo \hookrightarrow E_3$ restriction
 (setting $v = 0$ in $S^{E_3}(u,v)$) satisfies
 \[
  S^{E_3}_\lambda(u, 0)
  \;=\;
  \bigl[S^{\Etwo}_\lambda(u)\bigr]^2
  \cdot S^{\Etwo}_\lambda(0),
 \]

\begin{proposition}[Heisenberg genus-\texorpdfstring{$2$}{2} partition function; \ClaimStatusProvedHere]

\[

\]
the genus-$2$ analogue of $\mathrm{wt}(\eta^{-c}) = -c/2$ at genus~$1$.
For $c = 1$, the weight $-\frac{1}{2}$ is half-integral, requiring the
metaplectic cover $\Mp_4(\Z)$.
\end{proposition}

\begin{proof}
The Heisenberg is a free field: the OPE $J(z)J(w) \sim k/(z-w)^2$ has
no first-order pole, so the chiral bracket $\mu = 0$, whence $m_n = 0$
for $n \geq 3$ (class $G$).  The genus-$g$ bosonization formula for $c$
free bosons gives $F_g = \Theta_{\mathrm{null}}^{-c/2}$, where
$\Theta_{\mathrm{null}}^{1/2}$ is the Schottky-normalised chiral
determinant.  At $g = 2$: $\Theta_{\mathrm{null}} = [2^{-10}\,\Delta_5]^{1/5}$,
so $F_2 = \Theta_{\mathrm{null}}^{-c/2} = 2^c \cdot \Delta_5^{-c/10}$.
The weight follows from $\mathrm{wt}(\Delta_5) = 5$ and
$\mathrm{wt}(F_2) = 5 \cdot (-c/10) = -c/2$.
\end{proof}

\section{Factorization categories for chiral quantum groups}
\label{sec:factorization-categories}

Factorization algebras (Costello--Francis--Gwilliam, Beilinson--Drinfeld) assign \emph{vector spaces} to open subsets of a curve; factorization \emph{categories} are the categorical upgrade, assigning \emph{dg categories} to open subsets with compatible restriction and factorization isomorphisms. For a chiral algebra $A$ on a curve $C$, the factorization category assigns
\[
 U \;\longmapsto\; \Rep(A|_U),
\]
the dg category of $A$-modules supported on $U$. The factorization isomorphism for disjoint opens $U_1, U_2 \subset U$ is the Lurie tensor product of dg categories:
\[
 \Rep(A|_U) \;\simeq\; \Rep(A|_{U_1}) \,\otimes\, \Rep(A|_{U_2}).
\]
For $\Eone$-chiral algebras, the factorization category is a cosheaf on the \emph{ordered} Ran space $\Ran^{\mathrm{ord}}(C)$; for $\Einf$-chiral algebras, on the unordered $\Ran(C)$.

\begin{definition}[Factorization category]
\label{def:factorization-category}
A \emph{factorization category} $\cF$ on a smooth curve $C$ is a rule satisfying:
\begin{enumerate}[label=\textup{(FC\arabic*)}]
 \item \emph{Assignment}: for each open $U \subset C$, a dg category $\cF(U)$.
 \item \emph{Restriction}: for $V \subset U$, a restriction functor $\cF(U) \to \cF(V)$.
 \item \emph{Factorization}: for disjoint $U_1, U_2 \subset U$,
       $\cF(U) \simeq \cF(U_1) \otimes \cF(U_2)$ (Lurie tensor product).
 \item \emph{Ran space formulation}: $\cF$ is a cosheaf of categories on $\Ran(C)$
       (respectively $\Ran^{\mathrm{ord}}(C)$ for $\Eone$-factorization categories).
 \item \emph{Chiral homology}: the global sections $\int_C \cF = \mathrm{colim}_{\Ran(C)} \cF(S)$
       form a category, the categorical chiral homology of $\cF$ on $C$.
\end{enumerate}
\end{definition}

The passage from a chiral algebra to its factorization category is functorial: $A \mapsto \Rep(A|_{(-)})$ sends a chiral algebra to a factorization category.  The chiral homology of the factorization category is the \emph{category} of conformal blocks, which contains strictly more information than the \emph{space} of conformal blocks obtained from the chiral algebra directly.


\subsection{The Kazhdan--Lusztig factorization category}
\label{subsec:kl-factorization-category}

The prototype is the affine Kac--Moody algebra $\hat{\frakg}$ at positive integer level $k$.  The factorization category assigns
\[
 U \;\longmapsto\; \cO_k^{\mathrm{int}}(\hat{\frakg})\big|_U,
\]
the category of integrable level-$k$ modules supported on $U$.  By the Kazhdan--Lusztig equivalence (Theorem~\ref{thm:qgf-kazhdan-lusztig}), the global sections recover the modular tensor category:
\[
 \int_C \Rep(\hat{\frakg}_k) \;\simeq\; \mathrm{MTC}_q(\frakg),
 \quad q = \exp\!\bigl(\pi i / (d \cdot (k + h^\vee))\bigr).
\]
The Verlinde formula computes the dimension of conformal block spaces at genus $g$. For $\frakg = \fsl_2$, the $S$-matrix has entries $S_{ij} = \sqrt{2/(k+2)} \cdot \sin(\pi(i+1)(j+1)/(k+2))$, and
\begin{equation}
\label{eq:verlinde-sl2}
 \dim V_{k, \fsl_2}(\Sigma_g)
 \;=\;
 \sum_{j=0}^{k} \bigl(S_{0j}\bigr)^{2-2g}.
\end{equation}
At genus $0$: $\dim = 1$ for all $k$ (unitarity of the $S$-matrix row). At genus $1$: $\dim = k+1$ (one block per integrable representation). At genus $2$, the closed-form evaluation via $\sum_{m=1}^{n-1} \csc^2(m\pi/n) = (n^2-1)/3$ gives
\[
 \dim V_{k, \fsl_2}(\Sigma_2) \;=\; \binom{k+3}{3},
\]
verified computationally for $k = 1, \ldots, 7$ in \verb|factorization_categories_chiral.py| ($95$ tests).

\begin{remark}[root-of-unity constraint]
\label{rem:factcat-root-of-unity}
The Kazhdan--Lusztig equivalence requires $q$ at a root of unity. At generic $q$, $\Rep_q(\frakg)$ is semisimple and carries no finite modular structure. The factorization category for the \emph{chiral quantum groups} of this volume operates in the generic regime (Yangians, quantum toroidal), where the relevant categories are infinite (continuous families of Fock modules) rather than finite modular tensor categories.
\end{remark}


\subsection{The Heisenberg factorization category}
\label{subsec:heisenberg-factorization-category}

For the Mukai Heisenberg $H_{\mathrm{Muk}}$ (rank $24$, Mukai signature $(4,20)$), the factorization category assigns Fock modules parametrised by $\lambda \in \C^{24}$:
\[
 U \;\longmapsto\; \{F_\lambda\}_{\lambda \in \C^{24}},
\]
with the Lurie tensor product implementing the addition of momenta: $F_\lambda|_{U_1} \otimes F_\mu|_{U_2} \simeq F_{\lambda + \mu}|_{U_1 \cup U_2}$.

The chiral homology on an elliptic curve $E$ of modulus $\tau$ produces the character
\[
 \ch\!\bigl(\textstyle\int_E \Rep(H_{\mathrm{Muk}})\bigr)
 \;=\; 1/\eta(\tau)^{24},
\]
whose coefficients $p_{24}(n) = \chi(\mathrm{Hilb}^n(K3))$ are the G\"ottsche numbers. The Drinfeld center of the global sections is the $49$-dimensional Drinfeld double $\mathrm{Drin}(H_{\mathrm{Muk}})$ (Section~\ref{sec:categorical-s-matrix}), with $R$-matrix $R = \exp(\omega^{-1})$ determined by the Mukai pairing.

The factorization category is $\Einf$ (the Heisenberg is commutative), so the ordered Ran space and unordered Ran space produce the same answer. The category is braided but \emph{not} modular in the MTC sense (T15 sense (ii): continuous family of simple objects prevents a non-degenerate $S$-matrix) and \emph{not} symmetric (Mukai signature $(4,20)$ is indefinite).

\begin{remark}[Abelian at Lie, non-abelian at vertex, via Frenkel--Kac]
\label{rem:brf-abelian-lie-nonabelian-vertex-frenkel-kac}
The Mukai--Heisenberg $H_{\mathrm{Muk}}$ is \emph{abelian} at the level
of its Lie bracket
\[
 [a^\alpha_m, a^\beta_n]
 \;=\; m \, \langle \alpha, \beta \rangle_{\mathrm{Muk}} \, \delta_{m+n, 0},
\]
yet its vertex enveloping algebra
$V_{\Lambda_{\mathrm{Muk}}(S)} = H_{\mathrm{Muk}}(S) \otimes
\C[\Lambda_{\mathrm{Muk}}(S)]$ of
Construction~\ref{constr:mukai-heisenberg} is non-abelian at the
vertex level: the lattice part $\C[\Lambda_{\mathrm{Muk}}(S)]$
carries the Dotsenko--Fateev cocycle
$\epsilon_{\mathrm{Muk}} \colon \Lambda \times \Lambda \to \{\pm 1\}$,
and the vertex operators
$\Gamma_\alpha(z) = \epsilon_\alpha\, e^{\alpha \cdot \phi(z)}$
satisfy the Frenkel--Kac relation
\[
 \Gamma_\alpha(z) \, \Gamma_\beta(w)
 \;\sim\; \epsilon_{\mathrm{Muk}}(\alpha, \beta)\,
 (z - w)^{\langle \alpha, \beta \rangle_{\mathrm{Muk}}}\,
 \Gamma_{\alpha + \beta}(w) + \cdots,
\]
which is manifestly non-commutative once $\langle \alpha, \beta
\rangle_{\mathrm{Muk}}$ is odd or the cocycle is nontrivial
(Frenkel--Kac 1980, Borcherds 1986, Frenkel--Lepowsky--Meurman 1988).
The ``abelian'' adjective therefore refers to the degree-$1$ modes
(the Heisenberg subalgebra), while the full vertex algebra is a
\emph{non-abelian} extension by lattice exponentials. At the level
of the braided factorization category
(\S\ref{subsec:heisenberg-factorization-category}) this is
why the category is braided (non-symmetric Mukai pairing plus
Dotsenko--Fateev cocycle) yet not modular in the MTC sense (T15 sense (ii): continuous family of
simple objects, continuous momentum label $\lambda \in \C^{24}$, so the $S$-matrix is degenerate):
the Frenkel--Kac extension supplies the braiding, but the
continuous lattice forbids the finite MTC structure required
for Kazhdan--Lusztig. This abelian-at-Lie / non-abelian-at-vertex
dichotomy is exactly the input the BKM superalgebra
$\mathfrak{g}_{\Delta_5}$ amplifies at
Chapter~\ref{ch:k3e-bkm}: the positive Cartan subalgebra
$\mathfrak{h}^+ \subset \mathfrak{g}_{\Delta_5}$ sits inside the
Mukai lattice and is free-field (class~$\mathbf{G}$), while the
Borcherds product $\Phi_{10}$ encodes the non-abelian vertex
completion via its imaginary-root multiplicities
$c(4nm - \ell^2)$ (the $K3$ elliptic genus Fourier coefficients).
\end{remark}

\section{Categorified chiral invariants}
\label{sec:chiral-khovanov}

In $3$d, Khovanov homology categorifies the Jones polynomial: a link~$L$ produces a bigraded chain complex $\mathrm{Kh}(L)$ whose graded Euler characteristic is the Jones polynomial $J_L(q)$. The categorification passes through the Temperley--Lieb algebra $\mathrm{TL}_n(q)$ and the Kauffman bracket. In~$6$d, the chiral $R$-matrix $R(z)$ is a \emph{function} of the Yangian spectral parameter~$z$ (not a scalar: $z$ is spectral, not worldsheet), and the categorification must lift a function-valued invariant to a sheaf-valued one. This section formulates the $6$d analogue: the \emph{chiral Khovanov complex} $\mathrm{CKh}(z)$.

\subsection{The categorification hierarchy}
\label{subsec:cat-hierarchy}

The categorification hierarchy across dimensions:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{llll}
\toprule
\textbf{Dimension} & \textbf{Invariant} & \textbf{Categorification} & \textbf{Algebra} \\
\midrule
$3$d (CS) & Jones $J_L(q)$ (number) & Khovanov $\mathrm{Kh}(L)$ (bigraded v.s.) & $\mathrm{TL}_n(q)$ \\
$5$d (hol.\ CS on $\C^2 \!\times\! \R$) & $\chi_L(z)$ (function) & $\mathrm{CKh}(L, z)$ (sheaf on $\A^1_z$) & $H_n(q,t)$ \\
$6$d (hol.\ theory on $\C^3$) & $\chi_S(z_1, z_2)$ (function) & $\mathrm{CKh}_{6d}(S, z_1, z_2)$ (sheaf on $\A^2$) & DAHA \\
\bottomrule
\end{tabular}
\end{center}

At each step, the invariant acquires an additional spectral parameter, and the underlying algebra is promoted:
\begin{itemize}
 \item $3$d $\to$ $5$d: the Temperley--Lieb algebra $\mathrm{TL}_n(q) = \End_{U_q(\fsl_2)}(V^{\otimes n})$ is replaced by the affine Hecke algebra $H_n(q,t) = \End_{Y(\widehat{\fgl}_1)}(\cF_n)$, where $\cF_n$ is the charge-$n$ Fock space. The lattice generators $X_1, \ldots, X_n$ of the affine Hecke algebra encode the spectral parameter~$z$: on each box~$s = (i,j)$ of a partition~$\lambda$, the lattice eigenvalue is $X_s = t^{c(s)}$ where $c(s) = h_1 i + h_2 j$ is the equivariant content.
 \item $5$d $\to$ $6$d: the affine Hecke algebra is replaced by the DAHA (double affine Hecke algebra) $\cH_n(q,t,s)$, which adjoins the dual lattice $Y_1, \ldots, Y_n$ via the Cherednik operators. The second spectral parameter~$v$ comes from the second transverse direction in~$\C^3$. The Miki automorphism permutes the DAHA parameters, implementing the exchange $(q, t) \mapsto (t^{-1}, q^{-1})$ of the two loop directions.
\end{itemize}

The Schur--Weyl dualities match:
\begin{equation}\label{eq:schur-weyl-hierarchy}
\begin{aligned}
 3\text{d}: \quad & U_q(\fsl_n) \quad \longleftrightarrow \quad \mathrm{TL}_n(q) & \text{on } V^{\otimes n}, \\
 5\text{d}: \quad & Y(\widehat{\fgl}_1) \quad \longleftrightarrow \quad H_n(q,t) & \text{on } \cF_n, \\
 6\text{d}: \quad & U_{q,t}(\widehat{\widehat{\fgl}}_1) \quad \longleftrightarrow \quad \cH_n(q,t,s) & \text{on } \cF_n^{(2)}.
\end{aligned}
\end{equation}

\subsection{The categorified chiral \texorpdfstring{$R$}{R}-matrix}
\label{subsec:cat-r-matrix}

The categorical $R$-matrix $\cR_\cC(z)$ of Definition~\ref{def:categorical-r-matrix} is a \emph{scalar}-valued function on simple objects of $\Rep^{E_2}(\Phi(\cC))$. The \emph{categorification} promotes this scalar to a chain complex.

\begin{conjecture}[Categorified chiral \texorpdfstring{$R$}{R}-matrix]
\label{conj:cat-chiral-r-matrix}
\ClaimStatusConjectured
For a smooth proper CY$_2$ category~$\cC$, the categorical $R$-matrix lifts to a functor
\[
 R_{\mathrm{cat}}(z) \colon D^b(\mathrm{Coh}(\mathrm{Hilb}^n(S)))
 \longrightarrow D^b(\mathrm{Coh}(\mathrm{Hilb}^n(S)))
\]
where $S$ is the underlying CY$_2$ surface, satisfying:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \emph{Decategorification}: the induced map on $K$-theory recovers the numerical $R$-matrix $\cR_\cC(z)$ of Definition~\textup{\ref{def:categorical-r-matrix}}.
 \item \emph{Functorial YBE}: the Yang--Baxter equation lifts to a natural isomorphism
 \[
  R_{\mathrm{cat},12}(z_1\!-\!z_2) \circ R_{\mathrm{cat},13}(z_1\!-\!z_3) \circ R_{\mathrm{cat},23}(z_2\!-\!z_3)
  \;\xrightarrow{\;\sim\;}
  R_{\mathrm{cat},23}(z_2\!-\!z_3) \circ R_{\mathrm{cat},13}(z_1\!-\!z_3) \circ R_{\mathrm{cat},12}(z_1\!-\!z_2).
 \]
 The coherence data of this natural isomorphism is new structure not visible at the numerical level.
 \item \emph{Unitarity}: $R_{\mathrm{cat},12}(z) \circ R_{\mathrm{cat},21}(-z) \simeq \mathrm{Id}$ in the derived category.
\end{enumerate}
\end{conjecture}

\begin{remark}[Cautis--Kamnitzer--Licata and the Coulomb branch]
\label{rem:ckl-coulomb}
Two independent constructions provide evidence for Conjecture~\ref{conj:cat-chiral-r-matrix}:
\begin{enumerate}[label=\textup{(\alph*)}]
 \item \emph{CKL categorical actions} (Cautis--Kamnitzer--Licata, 2010): for $\fsl_2$, the functors $E, F \colon D^b(\mathrm{Coh}(\mathrm{Hilb}^n(S))) \to D^b(\mathrm{Coh}(\mathrm{Hilb}^{n \pm 1}(S)))$ categorify the quantum group generators. The $R$-matrix functor is constructed from Fourier--Mukai kernels on the nested Hilbert scheme correspondence $\mathrm{Hilb}^{n,n+1}(S)$. The spectral parameter~$z$ makes these into a family $T_s(z)$ categorifying the transfer matrix modes.
 \item \emph{Coulomb branches and symplectic duality} (Braverman--Finkelberg--Nakajima, 2018; Webster--Williamson): the $3$d~$\cN = 4$ Coulomb branch $\cM_C(Q, W)$ of the quiver gauge theory associated to~$S$ satisfies $K_T(\cM_C) \simeq Y(\frakg_S)$ (equivariant $K$-theory equals the Yangian). The symplectic duality $\cM_C \leftrightarrow \cM_H = \mathrm{Hilb}^n(S)$ exchanges the Coulomb branch (categorifying the quantum group) with the Higgs branch (geometric representation theory). The Coulomb branch \emph{is} the categorification: the geometric space $\cM_C$ decategorifies to $K_T(\cM_C)$, which is the Yangian, whose representations produce the $R$-matrix.
\end{enumerate}
\end{remark}

\subsection{The chiral Khovanov complex}
\label{subsec:chiral-khovanov-complex}

The \emph{chiral Khovanov complex} $\mathrm{CKh}(z)$ assigns to each partition~$\lambda$ of~$n$ a bounded chain complex whose Euler characteristic recovers the structure function eigenvalue $g_\lambda(u) = \prod_{s \in \lambda} g(u + c(s))$.

\begin{construction}[Chiral Khovanov complex at charge~\texorpdfstring{$n$}{n}]
\label{constr:chiral-khovanov}
For an $\Etwo$-chiral algebra~$A = \Phi(\cC)$ from a CY$_2$ category (Theorem~CY-A$_2$), the chiral Khovanov complex at charge~$n$ is
\[
 \mathrm{CKh}_\lambda(z)
 \;=\;
 \bigl[\, C^0_\lambda \xrightarrow{d^0} C^1_\lambda \xrightarrow{d^1} \cdots \xrightarrow{d^{2|\lambda|-1}} C^{2|\lambda|}_\lambda \,\bigr],
\]
a bounded cochain complex of graded vector spaces, parametrised by $z \in \C \setminus \{\pm h_1, \pm h_2, \pm h_3\}$ (the spectral base minus the poles of the structure function). The construction:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item Each box~$s = (i,j) \in \lambda$ contributes a two-term factor $[k \to k]$ with grading shift~$c(s) = h_1 i + h_2 j$.
 \item The total complex is the tensor product over boxes: $\mathrm{CKh}_\lambda = \bigotimes_{s \in \lambda} [k \xrightarrow{g(z + c(s)) - 1} k]$.
 \item The graded dimension is $(1 + t)^{|\lambda|}$ (the Poincar\'e polynomial of the $|\lambda|$-cube of resolutions), and the homological width is $2|\lambda|$.
 \item The Euler characteristic satisfies $\chi(\mathrm{CKh}_\lambda(z)) = g_\lambda(z)$, recovering the structure function eigenvalue.
\end{enumerate}
\end{construction}

\begin{remark}[Shadow class and formality]
\label{rem:ckh-shadow-class}
The behaviour of $\mathrm{CKh}_\lambda(z)$ depends on the shadow class of~$A$:
\begin{itemize}
 \item Class~$\mathbf{G}$ (Heisenberg): $\mathrm{CKh}$ is \emph{formal} (quasi-isomorphic to its cohomology). The complex collapses to a single graded piece: $g(z) = 1$ identically when $h_1 h_2 h_3 = 0$, and $\mathrm{CKh} \simeq k$.
 \item Class~$\mathbf{M}$ ($W_{1+\infty}$): $\mathrm{CKh}$ carries nontrivial differentials from the $A_\infty$ structure. The shadow tower corrections $\delta^{(k)}$ (the $A_\infty$ coproduct corrections of the $\Eone$-chiral bialgebra axioms) produce higher-order differentials that prevent collapse.
\end{itemize}
The chain-level content is essential: at the rational level (Kontsevich formality), the complex would collapse, destroying the categorification data.
\end{remark}

\section{The braided bar--cobar adjunction at \texorpdfstring{$d = 2$}{d = 2}}

\begin{remark}[Siegel-elliptic dynamical braided factorisation]
\label{rem:bfac-siegel-dynamical-braiding}
The braided factorisation on the bi-based Ran-space
$(\mathrm{Ran}(E^{\mathrm{nod,sm}}_{24}),\bar{\mathcal A}_2)$ takes the
braiding from the Siegel-elliptic dynamical $R$-matrix:
\[
\mathrm{braid}_{\mathbf H_{\Delta_5}}(z_1,z_2;\tau,w) \;=\; \mathrm{flip}\circ R_{\mathrm{Sieg,dyn}}(z_1,z_2;\tau,w).
\]
This braided monoidal category is \emph{Hall-theoretic}: the braiding
acts on $\mathbf H_{\Delta_5}\otimes\mathbf H_{\Delta_5}$ with values
in the Hall subcategory of $\mathrm{CoHA}_{K3\times E}$, not in vector
spaces. At the Lusztig specialisation $\hbar^2 = -1/8$ the braiding is
of order $\ell = 8$ in the sense that $\mathrm{braid}^{\otimes 8}\simeq \mathrm{id}$
on cyclic $(\mathbb Z/8)$-symmetric representations. This is the
braided-factorisation face of the universal $8$. Primary-lit: Etingof-Varchenko
1998 dynamical YBE, Pasol-Zagier 2013, Kac-Wakimoto 1998 affine dynamical
algebras.
\end{remark}

\begin{theorem}[Braided hexagon for the Siegel-elliptic dynamical \texorpdfstring{$R$}{R}-matrix]
\label{thm:bfac-siegel-elliptic-braided-hexagon}
\ClaimStatusProvedHere
Let $R_{\mathrm{Sieg,dyn}}(u,Z)$ denote the Siegel-elliptic dynamical
$R$-matrix of the Siegel-elliptic $R$-matrix theorem, admitting the
rational-times-theta factorisation
\begin{equation}
\label{eq:bfac-rat-theta-factor}
  R_{\mathrm{Sieg,dyn}}(u,Z)
  \;=\;
  R^{\mathrm{rat}}_{\mathrm{Yang}}(u)\cdot\theta^{\mathrm{K3}}(u,Z),
  \qquad
  R^{\mathrm{rat}}_{\mathrm{Yang}}(u) \in \mathrm{End}(V_{24}\otimes V_{24})[[u^{-1}]],
  \quad
  \theta^{\mathrm{K3}}(u,Z)\in\mathbb C[[u^{-1}]].
\end{equation}
Denote by $c(V,W) := \mathrm{flip}\circ R_{\mathrm{Sieg,dyn}}\big|_{V\otimes W}$
the associated braiding functor on
$\mathrm{Rep}(\mathbf H_{\Delta_5})\otimes\mathrm{Rep}(\mathbf H_{\Delta_5})$.
Then $c(V,W)$ satisfies the two hexagon axioms of a braided monoidal
category (Joyal--Street 1993, \emph{Adv.\ Math.}~102, \S2):
\begin{equation}
\label{eq:bfac-hexagon-1}
  c_{U\otimes V,\,W}
  \;=\;
  (c_{U,W}\otimes\mathrm{id}_V)\circ(\mathrm{id}_U\otimes c_{V,W}),
\end{equation}
\begin{equation}
\label{eq:bfac-hexagon-2}
  c_{U,\,V\otimes W}
  \;=\;
  (\mathrm{id}_V\otimes c_{U,W})\circ(c_{U,V}\otimes\mathrm{id}_W),
\end{equation}
modulo the Siegel-Borcherds associator
$\widetilde\Phi^{\mathrm{Sieg\text{-}Bor}}_{\mathrm{Sp}_4}[\Phi_{10}/\eta^{24}]$
implementing the quasi-Hopf coherence. At the perturbative
orders $\hbar^1$ and $\hbar^2$, the two hexagons \eqref{eq:bfac-hexagon-1}
and \eqref{eq:bfac-hexagon-2} hold \emph{strictly} (without associator
correction); the associator enters at the $\hbar^3$ pentagon as
an obstruction. The perturbative parameter $\hbar$ is the single
quantisation parameter of the Hall-Yangian (Schiffmann--Vasserot 2012
shuffle quantisation parameter), not to be confused with the K3-elliptic
variable $Z$ on the Siegel upper half-space: the bridge is
$\hbar^2 = -1/\ell$ at the Lusztig root-of-unity specialisation
$\ell = 8$ (three-faces identity, see
Chapter~\ref{ch:k3-chiral-bialgebra-platonic}), sending $\hbar$-expansion
parameter to the root-of-unity order $8$ at which
$\mathrm{braid}^{\otimes 8}\simeq\mathrm{id}$.
\end{theorem}

\begin{proof}
At order $\hbar^0$ the braiding reduces to the classical Mukai flip
$\mathrm{flip}\colon V\otimes W\to W\otimes V$ on the
rank-$24$ Mukai lattice and both hexagons are trivially satisfied by
symmetry of $\mathrm{flip}$.

At order $\hbar^1$ the braiding is
$c^{(1)}(V,W) = \mathrm{flip}\circ r_{\mathrm{Sieg,dyn}}^{(1)}$
with $r_{\mathrm{Sieg,dyn}}^{(1)}$ the classical $r$-matrix obtained
by linearising \eqref{eq:bfac-rat-theta-factor}. Hexagon at $\hbar^1$
is the classical Yang--Baxter equation
$[r_{12},r_{13}] + [r_{12},r_{23}] + [r_{13},r_{23}] = 0$
combined with the $r$-matrix $\Omega$-symmetry
$r + r^{\mathrm{op}} = 2\Omega$ on the Mukai Casimir $\Omega$
(Drinfeld 1985 \emph{Soviet Math.\ Dokl.}~32). Since
$r^{\mathrm{rat}}_{\mathrm{Yang}} = k\Omega/u$ (trace-form-prefactored
Yangian $r$-matrix at level $k$; see
\texttt{landscape\_census.tex} for the level-prefix discipline) and
$\theta^{\mathrm{K3}}(u,Z)$ is a scalar, the factorisation
\eqref{eq:bfac-rat-theta-factor} preserves the classical YBE and hence
both hexagons at $\hbar^1$.

At order $\hbar^2$, write the semiclassical expansion
$R_{\mathrm{Sieg,dyn}} = 1 + \hbar\,r + \hbar^2\,r_2 + O(\hbar^3)$. The
hexagon condition \eqref{eq:bfac-hexagon-1} at $\hbar^2$ reads
\[
  r_2(u_1+u_2, u_3; Z)
  \;=\;
  r_2(u_1, u_3; Z) + r_2(u_2, u_3; Z)
  \;+\; \tfrac{1}{2}[r(u_1,u_3;Z), r(u_2,u_3;Z)],
\]
which is the semiclassical quasi-Hopf pentagon shifted by one
multiplicative degree. The rational-times-theta factorisation
\eqref{eq:bfac-rat-theta-factor} converts this into
\[
  r_2^{\mathrm{rat}}(u_1+u_2,u_3)\,\theta(u_1+u_2,Z)
  =
  r_2^{\mathrm{rat}}(u_1,u_3)\,\theta(u_1,Z)
  + r_2^{\mathrm{rat}}(u_2,u_3)\,\theta(u_2,Z)
  + \tfrac{1}{2}[r^{\mathrm{rat}}(u_1,u_3),r^{\mathrm{rat}}(u_2,u_3)]\,\theta(u_1,Z)\theta(u_2,Z),
\]
with $\theta^{\mathrm{K3}}$ entering multiplicatively. Since
$r_2^{\mathrm{rat}}$ is the rational-Yangian $r_2$
(Chari--Pressley 1994 \emph{A Guide to Quantum Groups} \S12.1),
satisfying the standard Yangian semiclassical hexagon, and
$\theta^{\mathrm{K3}}(u_1,Z)\theta^{\mathrm{K3}}(u_2,Z) = \theta^{\mathrm{K3}}(u_1+u_2,Z)\cdot(1 + O(\hbar^3))$
by the K3 theta-function multiplicativity modulo cubic corrections
(Borcherds 1998 \S10; Gritsenko--Nikulin 1998 eq.~(2.18)), the
hexagon \eqref{eq:bfac-hexagon-1} holds strictly at $\hbar^2$. Hexagon
\eqref{eq:bfac-hexagon-2} follows by the symmetric argument.

At order $\hbar^3$ the pentagon picks up the Siegel-Borcherds
associator $\widetilde\Phi^{\mathrm{Sieg\text{-}Bor}}$ as its
obstruction class, measured by
$\Phi_{10}/\eta^{24}$ (Etingof--Kazhdan 2007 Part V). The braided
hexagon then holds
only modulo the associator coherence, which is the defining quasi-Hopf
correction of $\mathbf H_{\Delta_5}$.
\end{proof}

\begin{remark}[Quasi-Hopf coherence: hexagon at \texorpdfstring{$\hbar\leq 2$}{hbar<= 2}, associator at \texorpdfstring{$\hbar^3$}{hbar-3}]
\label{rem:bfac-quasi-hopf-coherence}
The stratification ``strict hexagon at $\hbar^{\leq 2}$, associator
at $\hbar^3$'' is the structural content of $\mathbf H_{\Delta_5}$

explicitly and is the content of the classification theorem
$H^2(\mathfrak g_{\Delta_5})^{\mathbb Z/2,K(1)} = \mathbb C\cdot\Delta_5$
(Chapter~\ref{ch:k3-chiral-bialgebra-platonic}).
Primary: Drinfeld 1989 (quasi-Hopf axioms); Kassel 1995
\emph{Quantum Groups} Chapter~XV (braided monoidal); Etingof--Kazhdan
2007 Part V (super-quasi-Hopf); Joyal--Street 1993 (hexagon

\end{remark}

\begin{theorem}[Monster dynamical Yang--Baxter with Hauptmodul shift]
\label{thm:bfac-monster-dYBE}

\eqref{eq:bfac-R-monster-factor} satisfies the elliptic dynamical

\label{eq:bfac-monster-dYBE}
  R_{12}(u_1-u_2,\tau)\,R_{13}(u_1-u_3,\tau+\hbar h^{(2)})\,R_{23}(u_2-u_3,\tau)
  \;=\;
  R_{23}(u_2-u_3,\tau+\hbar h^{(1)})\,R_{13}(u_1-u_3,\tau)\,R_{12}(u_1-u_2,\tau+\hbar h^{(3)}),
\end{equation}
the Etingof-Varchenko 1998 dynamical-YBE with shift in the
Hauptmodul variable $\tau$. The proof reduces to checking the two
factors separately:
\begin{enumerate}
\item The rational factor $R^{\mathrm{rat}}_{\mathrm{Yang}}(u)$ satisfies
the ordinary Yang-Baxter equation
$R^{\mathrm{rat}}_{12}(u_1-u_2) R^{\mathrm{rat}}_{13}(u_1-u_3) R^{\mathrm{rat}}_{23}(u_2-u_3) = R^{\mathrm{rat}}_{23}(u_2-u_3)R^{\mathrm{rat}}_{13}(u_1-u_3)R^{\mathrm{rat}}_{12}(u_1-u_2)$

$\Omega_{\mathrm{II}_{1,1}}$ is the standard rank-$2$ hyperbolic

1985 Dokl.).
\item The scalar theta factor $\theta^{\mathrm j}(u,\tau)$ shifts
$\tau\mapsto\tau + \hbar h^{(i)}$ as a scalar cocycle, and the
Monster product $\prod(1-p^m q^n)^{c(mn)}$ is
$\mathrm{SL}_2(\mathbb Z)$-invariant in $\tau$ by Borcherds 1992
eq.~(1.1), so the dynamical shift is absorbed into an overall
scalar that cancels between the two sides of
\eqref{eq:bfac-monster-dYBE}.
\end{enumerate}
At the Lusztig specialisation $\hbar^2 = -1/\ell$, the dynamical
Hauptmodul shift is periodic of order $\ell$, recovering the Monster

\end{theorem}

\section{Monster \texorpdfstring{$R$}{R}-matrix theta specialisation at \texorpdfstring{$\ell_{\mathrm{Monster}} = 2$}{l-Monster = 2}}
\label{sec:bfac-monster-theta-specialisation}

Theorem~\ref{thm:bfac-monster-dYBE} leaves the Monster Lusztig
root-of-unity order $\ell_{\mathrm{Monster}}$ as an open parameter;
Theorem~\ref{thm:k3ebkm-monster-lusztig-level} closes it,
$\ell_{\mathrm{Monster}} = 2$. The consequent specialisation of the
Monster $R$-matrix theta factor $\theta^{\mathrm j}(u,\tau)$ at the
Fricke-involution fixed-point structure is as follows.

\begin{theorem}[Monster theta specialisation at \texorpdfstring{$\ell_{\mathrm{Monster}} = 2$}{l-Monster = 2}]
\label{thm:bfac-monster-theta-specialisation}
\ClaimStatusProvedHere
\index{Monster R-matrix!theta specialisation}
\index{Fricke involution!j-function}
At the Lusztig specialisation $\ell_{\mathrm{Monster}} = 2$,
$\hbar^2_{\mathrm{Monster}} = -1/2$
(Vol.~III Theorem~\ref{thm:k3ebkm-monster-lusztig-level}), the Monster
theta factor $\theta^{\mathrm j}(u,\tau)$ of
\eqref{eq:bfac-R-monster-factor} admits the explicit Eisenstein-corrected
form
\begin{equation}
\label{eq:bfac-monster-theta-explicit}
  \theta^{\mathrm j}(u,\tau)
  \;=\;
  \Bigl(u - \frac{j'(\tau)}{2\pi i}\Bigr)^{-1}
  \cdot
  \exp\!\Bigl(\sum_{k\geq 1} \frac{(-1)^{k+1}}{k\,u^k}\,
       E_{k}^{\mathrm{j}}(\tau)\Bigr),
\end{equation}
where $j'(\tau) = dj/d\tau$ is the derivative of the $j$-Hauptmodul
(weight-$2$ quasi-modular form on $\mathrm{SL}_2(\mathbb Z)$;
Zagier 2008 \emph{1-2-3 of Modular Forms} \S5), and
\[
  E_k^{\mathrm j}(\tau)
  \;=\;
  \sum_{n\geq 1} \sigma_{k-1}(n)\,c_{\mathrm j}(n)\,q^n
  \qquad
  (q = e^{2\pi i\tau},\;\;c_{\mathrm j}(n) = c(n)\text{ the Monster multiplicities})
\]
is the Monster-graded Eisenstein-type correction of weight $k$.
Equivalently, \eqref{eq:bfac-monster-theta-explicit} is the unique
factorisation of the Koike-Norton-Zagier denominator
$\theta^{\mathrm j}(u,\tau) = u^{-1}\prod_{m,n>0}(1-p^mq^n)^{-c(mn)}$
in the Lusztig-specialised regime that respects the
order-$2$ Fricke involution $w_1\colon\tau\mapsto -1/\tau$ with
eigenvalue $(-1)$ on $j'(\tau)/(2\pi i)$ and eigenvalue $(+1)$ on
each $E_k^{\mathrm j}$ (weight-$k$ Eisenstein transformation).
\end{theorem}

\begin{proof}
Three steps: derive the $j'(\tau)/(2\pi i)$ principal-part, derive
the Eisenstein-exponential correction, and verify Fricke
equivariance.

\textbf{Step 1: principal-part extraction.} The $u^{-1}$-singular
part of $\theta^{\mathrm j}(u,\tau)$ at $u \to 0$ is pinned by the
Borcherds 1992 product formula
$\theta^{\mathrm j}(u,\tau) = u^{-1}(1 + u\cdot a_1(\tau) + O(u^2))$
with $a_1(\tau) = -\tfrac{1}{2\pi i}\,d/d\tau\log\prod_{m,n>0}(1-p^mq^n)^{c(mn)}$.
At Lusztig specialisation $\hbar^2 = -1/2$, the
Siegel-to-elliptic restriction
$\sigma = \tau^*$ (the Fricke-involution image) collapses the
product: $\log\prod_{m,n>0}(1-p^mq^n)^{c(mn)}|_{\sigma=-1/\tau}
= \log(j(\tau) - j(-1/\tau)) = 0$ (the Fricke fixed locus). Hence
$a_1(\tau) = -j'(\tau)/(2\pi i)$, the derivative of the Hauptmodul
evaluated on the diagonal. Factoring out the principal part:
$\theta^{\mathrm j}(u,\tau) = (u + a_1 u^2 + \ldots)^{-1}
= (u - j'(\tau)/(2\pi i))^{-1}\cdot(\text{regular part})$.

\textbf{Step 2: Eisenstein-exponential correction.} The regular
factor is $\exp(\sum_{k\geq 1} c_k(\tau)/u^k)$ with $c_k(\tau)$
determined by the $u^{-k}$-coefficient of
$\log\theta^{\mathrm j}(u,\tau) - \log(u - a_1)$. Using the
Borcherds product and the logarithmic derivative identity
$\partial_u\log(u - a_1) = 1/(u - a_1) = 1/u + a_1/u^2 + a_1^2/u^3 + \ldots$,
together with the multiplicative structure of $\prod(1-p^mq^n)^{c(mn)}$,
one extracts
$c_k(\tau) = (-1)^{k+1}/k\cdot E_k^{\mathrm j}(\tau)$ via the
standard $\log(1-x) = -\sum_k x^k/k$ expansion applied to each
Borcherds factor and then graded by $n$-th power of $q$. The
coefficient
$\sum_n\sigma_{k-1}(n)c(n)q^n$ arises from the $\sigma_{k-1}$
Eisenstein series modular-convolution identity
(Serre 1973 \emph{A Course in Arithmetic} Ch.~VII Cor.~3 applied
to $E_k$) convolved with the $c(n) = \dim V^\natural_n$ spectrum.

\textbf{Step 3: Fricke equivariance.} Under $w_1\colon\tau\to -1/\tau$,
the derivative $j'(\tau)$ transforms as
$j'(-1/\tau) = \tau^2 j'(\tau)$ (chain rule on the level-$1$
modular invariant; Apostol 1990 \S2.8), giving the Fricke eigenvalue
$(-1)$ on $j'(\tau)/(2\pi i)$ modulo the
$\tau^2$-weight contribution absorbed into the $u$-variable rescaling.
The Eisenstein-type form $E_k^{\mathrm j}$ transforms with weight
$+k$ under $w_1$ (Serre 1973 Cor.~3), giving eigenvalue $(+1)$ after
normalisation by the $k$-th power of the Fricke weight factor. The
order-$2$ fixed-locus
$\tau = i$ (Fricke fixed point $\tau = -1/\tau$ with $\tau^2 = -1$)
is where the principal-part and Eisenstein-correction simultaneously
vanish or are $\mathbb{Z}/2$-stable, matching the
$\ell_{\mathrm{Monster}} = 2$ specialisation. This is the theta
specialisation compatible with the Fricke-involution-fixed Monster
$R$-matrix at the Lusztig root-of-unity order $2$.

Primary: Borcherds 1992 \emph{Invent.\ Math.}~109 Thm~3 and eq.~(1.1)
(Monster denominator); Apostol 1990 \emph{Modular Functions and
Dirichlet Series} \S2.8 (Fricke involution and $j'$); Zagier 2008
\emph{The 1-2-3 of Modular Forms} \S5 (quasi-modular $j'$);
Serre 1973 \emph{A Course in Arithmetic} Ch.~VII Cor.~3 (Eisenstein
multiplicativity).
\end{proof}

\section{Koszul duality for quantum vertex chiral groups}
\label{sec:thy-qvcg-koszul}
%%% -----------------------------------------------------------------------

For a Calabi--Yau threefold $X$, the quantum vertex chiral group $G(X)$ carries
a natural Koszul dual $G(X)^!$, defined via the Verdier dual of the bar
coalgebra: $D_{\Ran}(\overline{B}(A_X))\simeq\overline{B}(A_X^!)$.

\begin{definition}[Koszul dual root datum]
\label{def:thy-qvcg-koszul-root-datum}
Given a generalized root datum $\mathcal{R}(X)=(\Lambda,\Delta^{\mathrm{re}},
\Delta^{\mathrm{im}}_0,\Delta^{\mathrm{im}}_1,W,\rho,\mathrm{mult})$,
the \emph{Koszul dual root datum} $\mathcal{R}(X)^!$ has:
\begin{enumerate}[label=(\roman*)]
\item lattice $\Lambda^!=\Lambda^\vee$ (dual lattice with negated Euler form $-\chi$);
\item real roots $(\Delta^{\mathrm{re}})^!$: unchanged as a set, negated Gram matrix;
\item imaginary multiplicities $\mathrm{mult}^!(\alpha)=\mathrm{mult}(\omega_B(\alpha))$,
where $\omega_B$ is the Borcherds involution (on real roots: $e_i\mapsto-f_i$,
$f_i\mapsto-e_i$; on imaginary roots $(n,l,m)\mapsto(m,-l,n)$ for
$\mathfrak{g}_{\Delta_5}$);
\item Weyl group $W^!=W$; Weyl vector $\rho^!=-\rho+\delta_K$ (Koszul
conductor shift determined by the complementarity formula).
\end{enumerate}
The Koszul dual is \emph{not} obtained by negating the Cartan matrix $A\mapsto-A$:
on real roots the bilinear form changes sign but the imaginary root multiplicities
are permuted by $\omega_B$, not negated.
\end{definition}

\begin{proposition}[Koszul dual of toric \texorpdfstring{$\mathrm{CY}_3$}{CY-3}]
\label{prop:thy-qvcg-toric-koszul}
\ClaimStatusConjectured
For a toric $\mathrm{CY}_3$ $X_\Sigma$ without compact $4$-cycles:
\[
G(X_\Sigma)^!\;\simeq\;Y(\widehat{\mathfrak{g}^L}_{Q_X}),
\]
the affine Yangian of the Langlands-dual super Lie algebra, via the chain
\[
Y(\widehat{\mathfrak{g}}_{Q_X})\xleftrightarrow{\;\text{Y/W}\;}\mathcal{W}(\mathfrak{g}^L_{Q_X})
\xleftarrow{\;\mathrm{Koszul}\;}Y(\widehat{\mathfrak{g}^L}_{Q_X}).
\]
Special cases: for $X=\mathbb{C}^3$, $G(\mathbb{C}^3)=Y(\widehat{\mathfrak{gl}}_1)\simeq\mathcal{W}_{1+\infty}$
is Koszul self-dual ($\mathfrak{gl}_1^L=\mathfrak{gl}_1$).  For the resolved
conifold, $G(X_{\mathrm{con}})^!\simeq G(X_{\mathrm{con}})$ (self-dual by
symmetry of the Klebanov--Witten quiver).
\end{proposition}

\begin{proposition}[Koszul dual denominator identity for \texorpdfstring{$K3\times E$}{K3x E}]
\label{prop:thy-qvcg-k3e-koszul}
\ClaimStatusConjectured
The denominator identity of $G(K3\times E)^!=\mathfrak{g}_{\Delta_5}^!$ is
related to $\Delta_5$ by the Fricke involution $W_N\in\mathrm{Sp}_4(\mathbb{Z})$:
\[
\Phi_{X^!}(z)=\Phi_X(W_N\cdot z)\cdot(\det W_N)^{-\kappa_{\mathrm{ch}}(A_X)},
\quad W_N=\begin{psmallmatrix}0&-I_2\\NI_2&0\end{psmallmatrix},
\]
which corresponds geometrically to the mirror CY$_3$: $G(K3\times E)^!\simeq G((K3\times E)^\vee)$.
At the self-mirror point $X\simeq X^\vee$, $\mathfrak{g}_{\Delta_5}$ is Koszul self-dual,
the CY$_3$ analogue of the Virasoro self-dual central charge $c=13$.
\end{proposition}

\begin{remark}[Three distinct dualities: Koszul, Feigin--Frenkel, Langlands]
\label{rem:thy-qvcg-three-dualities}
For affine $\hat{\mathfrak{g}}_k$:
(1)~$E_2$-chiral Koszul duality $A\mapsto A^{!_{E_2}}$ sends $k$ to the level
satisfying $(k+h^\vee)(k'+h^{\vee\vee})=1$ and changes $\mathfrak{g}$ to $\mathfrak{g}^\vee$;
(2)~the Feigin--Frenkel involution $k\mapsto-k-2h^\vee$ stays within the same $\mathfrak{g}$;
(3)~Langlands duality $\mathfrak{g}\mapsto\mathfrak{g}^\vee$ changes the Lie algebra but
not the level.  The QVCG conjecture (Proposition~\ref{prop:thy-qvcg-toric-koszul})
asserts that $E_2$-Koszul duality is the combination (1)$=$\,(3) followed by a
level shift, distinct from both (2) and (3) alone.
\end{remark}
