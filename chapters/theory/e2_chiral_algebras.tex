\chapter{$E_2$-Chiral Algebras}
\label{ch:e2-chiral}

Braiding is not primitive in Vol~III. The ordered $E_1$ layer carries
the collision data, and the braided $E_2$ layer appears only after
partial commutativity is imposed on that ordered input. This chapter
fixes that braided refinement, relates it to cyclic $A_\infty$ data via
the Deligne conjecture, and records the conjectural identity
$\Phi_{E_2} \simeq Z^{\mathrm{ch}} \circ \Phi_{E_1}$
\textup{(}the functors $\Phi_{E_1}, \Phi_{E_2}$ and the chiral Drinfeld
centre $Z^{\mathrm{ch}}$ are defined in
\S\textup{\ref{sec:drinfeld-center-bridge}}\textup{).}

The scope of this chapter depends on the CY dimension.
At $d = 2$, the correspondence $\Phi_2$ produces $\Etwo$-chiral algebras
\emph{natively} (Corollary~\ref{cor:e2-chiral-from-cy2}): the
$\mathbb{S}^2$-framing gives $\Etwo$ directly, and the definitions and
results of this chapter apply to the algebra $A = \Phi_2(\cC)$ itself.
At $d = 3$, the correspondence $\Phi_3$ produces an $\Eone$-chiral algebra
\textup{(}Theorem~\textup{\ref{thm:cy-to-chiral-d3}}: existence at the
$\infty$-categorical level; chain-level framing data for non-formal
algebras remains conjectural\textup{);} the $\Etwo$-chiral algebra
$\Phi_{E_2}(\cC)$ is its chiral Drinfeld centre, and the $\Etwo$-braided
structure lives on the representation category $\cZ(\Rep^{E_1}(A))$, not
on $A$.  The definitions in this chapter (the $\Etwo$-bar complex, the
braided tensor category, the $\Etwo$-Koszul duality) apply to the
Drinfeld centre at $d = 3$, not to the CY$_3$ chiral algebra directly.

\section{The $E_2$ operad and its categorical content}
\label{sec:e2-operad}

The braided refinement of ordered multiplication is operadic: it is
the $E_2$ operad that controls partial commutativity, and every
braided monoidal category is an $E_2$-object (Lurie).

\begin{definition}[$E_2$ operad]
\label{def:e2-operad}
The $E_2$ operad (little $2$-disks) has $E_2(n) =
\mathrm{Conf}_n(\mathbb{D}^2)$. By Dunn additivity $E_2 \simeq E_1
\otimes E_1$: an $E_2$-algebra is an $E_1$-algebra in $E_1$-algebras,
so multiplication is only partially commutative. The braiding measures
the failure of the two ordered products to agree strictly. At the level
of homology the braiding descends to the
commutativity constraint of a Gerstenhaber algebra.
\end{definition}

\begin{theorem}[Lurie, {\cite{LurieHA}}]
\label{thm:lurie-e2-braided}
\ClaimStatusProvedElsewhere
The $(\infty,2)$-category of $E_2$-algebras in a symmetric monoidal
$\infty$-category $\cC$ is equivalent to the $(\infty,2)$-category of
braided monoidal objects in $\cC$.
\end{theorem}

Kontsevich formality $C_*(E_2;\mathbb{Q}) \xrightarrow{\sim}
\mathrm{Ger}$ \textup{(}the Gerstenhaber operad, whose algebras are
graded-commutative rings with a degree-$(-1)$ Lie bracket satisfying
the Leibniz rule\textup{)} supplies the rational comparison.
For factorization algebras the $E_2$ structure appears fully
holomorphically \textup{(}on $X \times Y$ with holomorphy in each
factor\textup{)} or holomorphic-topologically \textup{(}Volume~II
Swiss-cheese\textup{),} two presentations of Dunn additivity.

\begin{definition}[$E_2$-chiral algebra]
\label{def:e2-chiral-algebra}
An \emph{$E_2$-chiral algebra} on $X \times Y$ is a factorization
algebra $\cA$ on the Ran space $\Ran(X) \times \Ran(Y)$
\textup{(}the moduli of finite subsets of $X$ and of $Y$\textup{)}
that factorizes chirally in each factor and is compatible with the
$E_2$-operad action via the Fulton--MacPherson equivalence
$\FM_n(\mathbb{C}) \simeq E_2(n)$ \textup{(}the compactified
configuration space of $n$ points in $\C$ as a model for the $n$-ary
operations of little $2$-disks\textup{).} Its representation category
$\Rep^{E_2}(\cA)$ is braided monoidal. Standard Beilinson--Drinfeld
chiral algebras lie on the $E_\infty$ side of this hierarchy after
symmetrisation; they still admit OPE poles.
\end{definition}

\begin{remark}[Deligne conjecture]
\label{rem:deligne-hochschild}
The canonical $E_2$-algebra is the Hochschild cochain complex
$\CC^\bullet(A,A) := \bigoplus_n \Hom(A^{\otimes n}, A)$ of an
associative algebra $A$. The Deligne conjecture (Kontsevich--Soibelman
\cite{KontsevichSoibelmanDefI}, Tamarkin \cite{TamarkinDGCat},
McClure--Smith, Berger--Fresse) asserts that $\CC^\bullet(A,A)$ carries
a canonical $E_2$-action with cup product and brace as the two
products, inducing the Gerstenhaber structure on Hochschild cohomology
$\HH^\bullet(A,A) := H^*(\CC^\bullet(A,A))$.
\end{remark}

\section{$E_2$-chiral algebras from cyclic $A_\infty$ data}
\label{sec:e2-from-cyclic-ainf}

Calabi-Yau data does not change the $E_2$ operad. It changes which
trace-compatible lifts of the Deligne action exist, and the CY
dimension controls that extra structure.

\begin{theorem}[Kontsevich-Soibelman; Tamarkin]
\label{thm:ks-tamarkin-cyclic-e2}
\ClaimStatusProvedElsewhere
Let $A$ be a cyclic $A_\infty$ algebra of CY dimension $d$, with
nondegenerate pairing $\langle-,-\rangle\colon A \otimes A \to k[d]$.
Then $\CC^\bullet(A,A)$ carries a canonical $E_2$-algebra structure
together with a compatible chain-level $S^1$-action whose homotopy
fixed points compute the negative cyclic homology
$\HC^-_\bullet(A) := \mathrm{HC}_\bullet(A)^{hS^1}$ \textup{(}the
$S^1$-equivariant refinement of Hochschild homology\textup{).}
The combined \emph{framed} structure is $fE_2 \simeq E_2 \rtimes S^1$. References:
Kontsevich-Soibelman \cite{KontsevichSoibelmanDefI}; Tamarkin
\cite{TamarkinDGCat}.
\end{theorem}

\begin{remark}[CY dimension shift]
\label{rem:cy-dim-shift}
The $E_2$-structure on $\CC^\bullet(A,A)$ is degree-shifted by $2-d$:
at $d=1$ the shift is $+1$ (framed $E_2 \simeq$ BV); at $d=2$ the
shift is $0$ (ordinary $E_2$ with Poisson compatibility); at $d=3$ the
shift is $-1$ (Lie-$2$-algebra compatibility: the Gerstenhaber bracket
carries an additional degree-$-1$ component from the CY$_3$ trace,
giving a homotopy-coherent Lie $2$-algebra in the Baez--Crans sense).
The CY dimension enters via the cyclic trace, not the underlying
$E_2$-operad.
\end{remark}

\begin{corollary}[$E_2$-chiral from CY$_2$]
\label{cor:e2-chiral-from-cy2}
\ClaimStatusProvedHere
Let $\cC$ be a saturated dg category of CY dimension $d=2$ with
cyclic $A_\infty$ enhancement. Then $\CC^\bullet(\cC,\cC)$ carries an
$E_2$-structure with Poisson compatibility, and
$\Phi_{E_2}(\cC)$ on $\Sigma \times \Sigma$ is an $E_2$-chiral algebra
(Definition~\ref{def:e2-chiral-algebra}). The assignment
$\cC \mapsto \Phi_{E_2}(\cC)$ is the Volume~III CY-to-$E_2$-chiral
functor at $d=2$.
\end{corollary}

\begin{proof}
Theorem~\ref{thm:ks-tamarkin-cyclic-e2} equips $\CC^\bullet(\cC,\cC)$
with a cyclic $E_2$-structure, ungraded at $d=2$ by
Remark~\ref{rem:cy-dim-shift}. Dunn additivity
($E_2 \simeq E_1 \otimes_{E_0} E_1$) upgrades the factorization
homology $\int_{\Sigma \times \Sigma} \CC^\bullet(\cC,\cC)$ to a
factorization algebra on $\Sigma \times \Sigma$: each $E_1$-factor
contributes factorization structure along one copy of $\Sigma$.
Chirality in each factor follows from the holomorphic refinement of
Kontsevich formality ($C_*(E_2; \Q) \xrightarrow{\sim} \mathrm{Ger}$):
the holomorphic structure on $\Sigma$ pins the factorization to a
chiral (meromorphic) slice, so the output satisfies the OPE locality
axiom of Definition~\ref{def:e2-chiral-algebra}. The two chiral
directions commute because the two $E_1$-factors of $E_2$ are
algebraically independent (Dunn additivity is a tensor product, not
a composition). Poisson compatibility follows from the $d = 2$
shift: at CY dimension $2$, the Gerstenhaber bracket on
$\HH^\bullet(\cC, \cC)$ has degree $2 - 2 = 0$
(Remark~\ref{rem:cy-dim-shift}), matching the Poisson bracket
degree of a classical $E_2$-algebra.
\end{proof}

\begin{conjecture}[$E_2$-braided representations from CY$_3$]
\label{conj:e2-chiral-from-cy3}
\ClaimStatusConjectured
Let $\cC$ be saturated of CY dimension $d=3$ with chain-level
$S^3$-framed cyclic $A_\infty$ enhancement. Then $\CC^\bullet(\cC,\cC)$
carries a framed $E_2$-structure in degree $-1$, and
$\Phi_{E_1}(\cC)$ is an $\Eone$-chiral algebra
(Theorem~CY-A$_3$) whose representation category acquires $\Etwo$-braiding
via the Drinfeld centre:
$\cZ(\Rep^{E_1}(\Phi_{E_1}(\cC))) \simeq \Rep^{E_2}(\Phi_{E_2}(\cC))$.
The $\Etwo$-chiral algebra $\Phi_{E_2}(\cC)$ is the chiral Drinfeld
centre of $\Phi_{E_1}(\cC)$
(Conjecture~\ref{conj:phi-e2-drinfeld-center}), \emph{not} a direct
output of the CY-to-chiral correspondence programme $\{\Phi_d\}$.
The expected representation category is $\CoHA(\cC)$ \textup{(}the
cohomological Hall algebra of $\cC$ in the sense of
Kontsevich--Soibelman~\cite{KontsevichSoibelmanCoHA}: an
associative algebra on the cohomology of moduli of objects, with
Hall product from extension stacks; not itself a chiral
algebra\textup{)} with its Kontsevich--Soibelman braided structure.
Conditional on Conjecture~\ref{conj:phi-e2-drinfeld-center}
($\Phi_{E_2} = Z^{\mathrm{ch}} \circ \Phi_{E_1}$);
CY-A$_3$ itself is proved (Theorem~\ref{thm:cy-to-chiral-d3}).
\end{conjecture}

\section{Braided tensor categories from $E_2$-chiral structure}
\label{sec:braided-tensor-from-e2}

By Theorem~\ref{thm:lurie-e2-braided}, $\Rep^{E_2}(\cA)$ is braided
monoidal; for $\cA$ an $E_2$-chiral algebra the braiding is
holomorphic (analytically continued along paths in
$\mathrm{Conf}_n(\Sigma)$, not merely formal).

\begin{definition}[MTC of an $E_2$-chiral algebra]
\label{def:mtc-e2-chiral}
The \emph{modular tensor category} of an $E_2$-chiral algebra $\cA$
is the semisimplification of $\Rep^{E_2}(\cA)$ at integrable modules
(when it exists); if $\Rep^{E_2}(\cA)$ is already finite with
nondegenerate braiding form, it is the MTC directly.
\end{definition}

\begin{conjecture}[MTC from $K3 \times E$ via Borcherds]
\label{conj:mtc-k3}
\ClaimStatusConditional
For $\cC = D^b(\Coh(K3 \times E))$ the derived category of a product
K3 times an elliptic curve, the MTC of $\Phi_{E_2}(\cC)$ coincides
with the Verlinde category of the Borcherds-Kac-Moody superalgebra
$\mathfrak{g}_{II_{1,1} \oplus II_{1,1}}$ of the K3 lattice. The
corresponding $\kappa_{\mathrm{ch}} = 3$, $\kappa_{\mathrm{BKM}} = 5$
diagnostic is discussed in
Chapter~\ref{ch:k3-times-e} (see also Theorem~\ref{thm:k3e-scattering-bkm}).

\emph{Conditional on Conjecture~\textup{\ref{conj:phi-e2-drinfeld-center}}}:
CY-A$_3$ is proved (Theorem~\ref{thm:cy-to-chiral-d3}), so $\Phi_{E_1}$
exists for $K3 \times E$.  The $E_2$-chiral algebra $\Phi_{E_2}$
requires the center identification
$\Phi_{E_2} = Z^{\mathrm{ch}} \circ \Phi_{E_1}$, which remains conjectural.
Corollary~\textup{\ref{cor:e2-chiral-from-cy2}} applies
to the CY$_2$ factor $D^b(\Coh(K3))$; the extension to the product
$K3 \times E$ requires the center identification.
\end{conjecture}

\begin{remark}[Evidence]
\label{rem:mtc-k3-evidence}
Corollary~\ref{cor:e2-chiral-from-cy2} applies to $D^b(\Coh(K3))$
(CY$_2$) with $\HH^\bullet(K3) = \bigwedge^\bullet H^1(T_{K3})$,
producing an $E_2$-chiral algebra on K3. The elliptic factor adjoins a
modular direction on which the braid group acts via the Heisenberg
lattice VOA of $II_{1,1}$, but the passage from the $d = 2$ chiral
algebra on K3 to a $d = 3$ chiral algebra on $K3 \times E$ requires
the tensor product of the $E_2$-chiral algebra with the elliptic
Heisenberg, which is not a formal consequence of the $d = 2$ functor
alone. The Borcherds product $\Delta_5$ matches the Volume~I chiral
Euler product at lattice weight $\kappa_{\mathrm{BKM}}=5$, and the
MTC is the Verlinde quotient of the weight-$5$ lattice VOA. This
matching is an observation, not a derivation from the
functor.
\end{remark}

\begin{remark}[$\Delta_5$ as the $1$-loop output of paramodular anomaly
cancellation, with four duality frames]
\label{rem:e2-DNA-delta5-one-loop-four-frames}
The matching $\kappa_{\mathrm{BKM}} = 5$ of
Remark~\ref{rem:mtc-k3-evidence} upgrades from an observation to a
\emph{derivation} once $\Delta_5$ is recognised as the $1$-loop-forced
output of paramodular anomaly cancellation in twisted $11$D supergravity
on $K3 \times T^2$ with $\Omega$-deformation and $24$ M$5$-branes wrapping
the $24$ $I_1$ Kodaira fibres of a generic elliptic K3. The weight
$5$ is fixed by the arithmetic split
\[
 5 \;=\; \chi(\cO_{K3}) \;+\; \sum_{i=1}^{24}
 \mathrm{Res}\bigl(I_{1,\,i}\bigr)
 \;=\; 2 \;+\; 3,
\]
where $\chi(\cO_{K3}) = 2$ is the holomorphic Euler characteristic (the
$d = 2$, $h^{1,0} = 0$ specialisation of Theorem~CY-D) and the
residues at the $24$ $I_1$-nodes accumulate to $3$ on the nose by the
Steiner-system $S(5,8,24)$ rigidity of
Remark~\ref{rem:e2-DNA-bi-based-ran}(R1). The claim that
$\Delta_5$ is produced as the $1$-loop output is independently verified in
four duality frames which all converge on the same paramodular weight-$5$
Borcherds form:
\begin{enumerate}[label=\textup{(D\arabic*)}]
 \item Heterotic on $K3 \times T^2$ \cite{HarveyMoore1996};
 \item Type IIB with D$1$--D$5$ bound states \cite{DijkgraafVerlindeVerlinde1996};
 \item Type IIA string-theoretic K3 elliptic genus
 \cite{DijkgraafMooreVerlindeVerlinde1997};
 \item M-theory on $K3 \times T^3$ \cite{Witten1995}.
\end{enumerate}
Each frame produces $\Delta_5$ via a distinct mechanism (heterotic
threshold integral; D-brane partition-function Igusa lift; elliptic-genus
second-quantisation; M-theory topological index), and the four outputs
agree by T-duality / string--string duality / mirror symmetry. For the
Vol~III categorical programme, this frame-independence is the statement
that the $\kappa_{\mathrm{BKM}} = 5$ invariant of
Chapter~\ref{ch:k3e-bkm} is computed by a single canonical lift from the
K3 elliptic genus $\varphi_{0,1}$, confirming
Conjecture~\ref{conj:mtc-k3} at the level of the paramodular lift.
\end{remark}

\section{Connection to Volume~II: the Drinfeld center}
\label{sec:drinfeld-center-bridge}

Volume~II constructs $\Phi_{E_1}$ and proves $E_1$-chiral Koszul
duality there. The Drinfeld center
$Z\colon E_1\text{-}\Cat \xrightarrow{\sim} E_2\text{-}\Cat$
(under dualizability) is the categorical passage from ordered
monoidal data to braided monoidal data. On the bar side it matches the
symmetrization map $\mathrm{av}\colon B^{\mathrm{ord}} \to B^\Sigma$.

\begin{conjecture}[Volume~III central conjecture: $\Phi_{E_2} = Z^{\mathrm{ch}} \circ \Phi_{E_1}$]
\label{conj:phi-e2-drinfeld-center}
\ClaimStatusConjectured
For a saturated CY$_d$ category $\cC$ admitting
$\Phi_{E_1}(\cC)$, the Volume~III $E_2$-chiral construction is
canonically quasi-isomorphic to the chiral Drinfeld center of
$\Phi_{E_1}(\cC)$:
\[
 \Phi_{E_2}(\cC) \;\simeq\; Z^{\mathrm{ch}}\bigl(\Phi_{E_1}(\cC)\bigr),
\]
where $Z^{\mathrm{ch}}$ is the $E_2$-chiral algebra whose
representation category is the Drinfeld center of the $E_1$-chiral
representation category of $\Phi_{E_1}(\cC)$. At $d=2$ this is
compatible with Corollary~\ref{cor:e2-chiral-from-cy2} and
Conjecture~\ref{conj:mtc-k3}; at $d=3$, CY-A$_3$ is proved
(Theorem~\ref{thm:cy-to-chiral-d3}), so $\Phi_{E_1}$ exists and the
content of this conjecture is the center identification itself.
\end{conjecture}

\begin{remark}[Four distinct functors]
\label{rem:ap34-four-functors}
Conjecture~\ref{conj:phi-e2-drinfeld-center} must be distinguished
from three other functors: (i) Verdier duality $D_{\Ran}$ producing
$B(A^!)$ as a factorization coalgebra; (ii) cobar inversion
$\Omega \circ B \simeq \id$ recovering $A$ on the Koszul locus
(Volume~I Theorem~B); (iii) the derived center
$Z^{\mathrm{der}}_{\mathrm{ch}}(A)$ producing the bulk factorization
algebra via Hochschild cochains (Volume~II bulk-boundary); (iv) the
categorical Drinfeld center $Z^{\mathrm{ch}}$ used here. The
Drinfeld center (iv) lives on representation categories with braided
monoidal output; the derived center (iii) lives on cochain complexes
with $\mathrm{SC}^{\mathrm{ch,top}}$-algebra output (or
$E_3$-topological when a conformal vector exists). The three sit on
different source categories with different output operads, and their
invariants do not interchange.
\end{remark}

\begin{remark}[Relation to Volume~II operadic conventions]
\label{rem:v2ap-cross-reference}
Volume~II tracks the $E_1$-chiral operadic story
($B_P(A) = P^i$-coalgebra vs $BP$ = cooperad; three coalgebra
structures $\mathrm{Lie}^c, \Sym^c, T^c$; factorization coproduct vs
deconcatenation; curvatures $\mu_0$ vs $d_{\mathrm{fib}}^2 =
\kappa_{\mathrm{ch}}\omega_g$; lossiness of $\mathrm{av}$).
$Z^{\mathrm{ch}}$ preserves $T^c$-coalgebra structure and the
curvature $\kappa_{\mathrm{ch}}\omega_g$, and inverts $\mathrm{av}$
only where braided centralization is strict.
\end{remark}

\begin{remark}[Bi-based Ran space and the K3 averaging factorisation]
\label{rem:e2-DNA-bi-based-ran}
For K3 and $K3 \times E$ the averaging map
$\mathrm{av} \colon B^{\mathrm{ord}} \to B^{\Sigma}$ of
Remark~\ref{rem:v2ap-cross-reference} acquires explicit geometric content
as a bi-based factorisation datum. The factorisation algebras on which
$Z^{\mathrm{ch}}$ operates at the K3 moduli boundary are defined over
\emph{two distinct} Ran spaces simultaneously:
\begin{enumerate}[label=\textup{(R\arabic*)}]
 \item A \emph{chiral base}
 $\Ran\bigl(E^{\mathrm{nod,sm}}_{24}\bigr)$, the Ran space of an elliptic
 fibration with $24$ nodal--smooth reduction points, carrying
 nearby-cycles $D$-modules at the $24$ Kodaira $I_1$-nodes
 \cite{Kodaira1963}.
 \item A \emph{parameter base}
 $\overline{\mathcal{A}}_2$, the Satake--Baily--Borel compactification
 of the paramodular moduli
 $\mathcal{A}_2 = \Gamma_{\mathrm{paramod}} \backslash \mathcal{H}_2$,
 carrying the Francis--Gaitsgory factorisation $\infty$-category of
 holonomic $D$-modules regular-singular along
 $\mathcal{H}_1 \cup \mathcal{H}_4$ \cite{FrancisGaitsgory2012}.
\end{enumerate}
The averaging map factors as a composition of three geometric steps:
\[
 \mathrm{av}
 \;=\; \mathrm{Torelli}_{K3} \;\circ\;
 \mathrm{KugaSatake} \;\circ\; j_{\mathrm{Kodaira}},
\]
where $j_{\mathrm{Kodaira}}$ is the nearby-cycles functor at the $24$
$I_1$-nodes, $\mathrm{KugaSatake}$ is the Kuga--Satake morphism
$\mathcal{M}_{K3} \to \mathcal{A}_2$ realising K3 Hodge structures inside
paramodular abelian fourfolds \cite{KugaSatake1967}, and
$\mathrm{Torelli}_{K3}$ is the global Torelli embedding
$\mathcal{M}_{K3} \hookrightarrow \mathcal{H}_4(\Lambda_{\mathrm{Muk}})$.
The lossiness of $\mathrm{av}$ tracked in
Remark~\ref{rem:v2ap-cross-reference} is localised precisely at the $24$
$I_1$-nodes of (R1) and at the Humbert divisors of (R2); the chiral
Drinfeld centre $Z^{\mathrm{ch}}$ inverts $\mathrm{av}$ on the smooth locus
of $\overline{\mathcal{A}}_2$ away from $\mathcal{H}_1 \cup \mathcal{H}_4$,
which is the geometric locus of strict braided centralisation entering
Conjecture~\ref{conj:phi-e2-drinfeld-center} for the K3 family. The
Steiner system $S(5,8,24)$ rigidity of \cite{ConwaySloane1988} underwrites
the $24$-point structure on both bases simultaneously.
\end{remark}

\section{$E_n$-chiral Koszul duality and the three bar complexes}
\label{sec:e2-koszul-and-bar}

CY-B is $d$-dependent. At $d = 2$, $A = \Phi_2(\cC)$ is natively $E_2$
and the duality is $E_2$-chiral Koszul duality on $A$ directly.
At $d = 3$, $A = \Phi_3(\cC)$ is $E_1$, and the duality operates on the
$E_3$ bar complex $B_{E_3}(A)$ (from the CY$_3$ $\mathbb{S}^3$-framing);
the $E_2$-braided equivalence is \emph{induced} on the Drinfeld center
$\cZ(\Rep^{E_1}(A))$.

CY-B has two components: the \emph{conductor formula}
$\kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!) = \rho_K$
(proved below as Theorem~\ref{thm:cy-b-conductor}), and the
\emph{braided monoidal equivalence} on $E_2$-representation categories
(proved for all CY$_3$ inputs in Theorem~\ref{thm:cy-b-d3} via
the Verdier spectral functor of Theorem~\ref{thm:verdier-spectral-functor};
conjectural for general non-CY $E_2$-chiral algebras).

\begin{conjecture}[$E_2$-chiral Koszul duality]
\label{conj:e2-koszul}
\ClaimStatusConjectured
For an $E_2$-chiral algebra $\cA$, the bar-cobar adjunction extends
$E_2$-equivariantly, giving a braided monoidal equivalence
\[
 \Rep^{E_2}(A) \simeq \Rep^{E_2}(A^!)^{\mathrm{rev}},
\]
with $\mathrm{rev}$ the reverse braiding and $A^!$ the Koszul dual
algebra (not $D_{\Ran}(B(A))$, not $\Omega(B(A))$).
\end{conjecture}
%: A^! is the Koszul dual algebra, NOT the cobar of B(A) (which
% recovers A itself), and NOT the Verdier dual D_Ran(B(A)) (which is
% B(A^!) as a factorization coalgebra). The "rev" reversal is specific
% to the E_2 setting; at E_1, one has Rep^{E_1}(A) = Rep^{E_1}(A^!)^{rev}
% as monoidal (not braided) categories (Volume II Theorem thm:e1-koszul).

\begin{theorem}[$E_2$-chiral Koszul duality of the Heisenberg]
\label{thm:e2-koszul-heisenberg}
\ClaimStatusProvedHere{}
Let $H_k$ be the Heisenberg VOA at level~$k$, realized as the boundary
chiral algebra of the 5d holomorphic theory on $\C^2 \times \R$ (or
equivalently as the $E_2$ projection of the 6d theory on $\C^3$) at
parameters $(h_1, h_2, h_3)$ with $h_1 + h_2 + h_3 = 0$ and
$k = -\sigma_2 = -(h_1 h_2 + h_1 h_3 + h_2 h_3)$.
Then $H_k$ is class~$G$ (Gaussian, shadow depth $r = 2$), and
Conjecture~\ref{conj:e2-koszul} holds for $H_k$:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \emph{Vanishing differentials.} The $E_2$ bar complex
 $B_{E_2}(H_k)$ has two commuting differentials $d_X, d_Y = 0$.
 The Heisenberg OPE $J(z)J(w) \sim k/(z-w)^2$ has no first-order
 pole, so the chiral bracket $\mu = \Res_{z=w} J(z) J(w) = 0$ and the
 bar differential vanishes in each direction. This holds at all parameter
 values, not only the self-dual point.

 \item \emph{Bigraded decomposition.} Since $d_X = d_Y = 0$, the $E_2$
 bar complex is formal. Its bigraded Hilbert series is
 \[
   \sum_{n \geq 0} \dim B_{E_2}(H_k)_n \, q^n \;=\; P(q)^2 \;=\;
   \prod_{m \geq 1} \frac{1}{(1-q^m)^2},
 \]
 so $\dim B_{E_2}(H_k)_n = \tau_2(n)$, the number of $2$-component
 multipartitions of~$n$:
 $\tau_2(n) = \sum_{a+b=n} p(a)\,p(b)$.
 The first values are $1, 2, 5, 10, 20, 36, 65, 110, 185, 300$
 (OEIS A000712). The $E_3$ trigraded dimension $\tau_3(n)$ of
 Theorem~\ref{thm:e3-koszul-heisenberg} factors through $\tau_2$:
 $\tau_3(n) = \sum_{m=0}^n \tau_2(m) \, p(n-m)$, consistent with
 the Dunn-additivity projection $E_3 \to E_2$.

 \item \emph{Braiding reversal.} The Koszul dual $H_k^!$ has inverted
 level $k^! = -k$ and $R$-matrix
 \[
   R^{E_2,!}(z) \;=\; \frac{-k\,\Omega}{z} + O(1)
   \;=\; -R^{E_2}(z) + O(1).
 \]
 Since $R^{E_2,!}(z) = -R^{E_2}(z)$ at leading order, this is the
 braiding reversal $\sigma^{\mathrm{rev}} = \sigma^{-1}$. The
 $E_2$-equivariant bar-cobar adjunction therefore produces
 \[
   \Rep^{E_2}(H_k) \;\simeq\; \Rep^{E_2}(H_{-k})^{\mathrm{rev}},
 \]
 where $\mathrm{rev}$ is the reverse braiding on the representation
 category. Unlike the $E_3$ case
 (Theorem~\ref{thm:e3-koszul-heisenberg}), where $H_1$ is Koszul
 self-dual at the self-dual point, the $E_2$ Koszul dual
 \emph{always} has reversed braiding: $H_k$ is not $E_2$-Koszul
 self-dual for any $k \neq 0$.

 \item \emph{Koszul conductor.} The modular characteristics satisfy
 $\kappa_{\mathrm{ch}}(H_k) + \kappa_{\mathrm{ch}}(H_k^!) = k + (-k)
 = 0 = \rho_K$, where $\rho_K = 0$ for class~$G$ algebras, consistent
 with Conjecture~\ref{conj:e2-koszul} and the Vol~I class~$G$ Koszul
 conductor formula.
\end{enumerate}
\end{theorem}

\begin{proof}
The proof has two independent parts: the algebraic structure of the
Heisenberg OPE (Claims~(i)--(ii)), and the Verdier duality computation
with braiding analysis (Claims~(iii)--(iv)).

\medskip
\noindent\textbf{Part~1: Vanishing of differentials.}
By Dunn additivity, $E_2 \simeq E_1 \otimes E_1$, and the $E_2$ bar
complex is the iterated bar $B_{E_2}(A) \simeq B_Y(B_X(A))$ with
$d_X$ from the OPE residue $\Res_{z_X = w_X}$ and $d_Y$ from
$\Res_{z_Y = w_Y}$. For $H_k$, the OPE is
\[
  J(z)\,J(w) \;\sim\; \frac{k}{(z-w)^2},
\]
with no first-order pole. The bar differential in direction~$i$
extracts the chiral bracket $\mu_i(J, J) = \Res_{z_i = w_i} J(z) J(w)
= 0$ (no $(z-w)^{-1}$ term). Since the algebra is generated by~$J$
alone and $\mu_i = 0$ on all generators, $d_X = d_Y = 0$. This holds
at all parameter values: the OPE depends only on $k = -\sigma_2$, and
is always purely quadratic. The structure function
\[
 g_{E_2}(u) = \frac{(u - h_1)(u - h_2)}{(u + h_1)(u + h_2)}
\]
controls the $R$-matrix (braiding), not the differential.

\medskip
\noindent\textbf{Part~2: Bigraded dimension.}
With $d_X = d_Y = 0$, the complex is formal and reduces to its
underlying bigraded vector space. Each of the two directions contributes
a copy of the symmetric bar $B_{E_\infty}(H_k|_{C_i})$ with graded
dimension $P(q) = \prod_{m \geq 1} (1-q^m)^{-1}$ (one generator~$J$).
The bigraded dimension is $P(q)^2$, giving $\tau_2(n)$ at degree~$n$.

\medskip
\noindent\textbf{Part~3: Braiding reversal.}
The Koszul dual level is $k^! = -k$; this does \emph{not} follow from
negating the $\Omega$-background parameters $(h_1, h_2, h_3) \mapsto
(-h_1, -h_2, -h_3)$, since $\sigma_2$ is degree-$2$ homogeneous and
hence $\sigma_2(-h_i) = \sigma_2(h_i)$, leaving $k = -\sigma_2$
invariant.  Instead, $k^! = -k$ arises from the Verdier duality
functor $D_{\C^2}$ on conilpotent $E_2$-coalgebras, which acts by
linear duality on the underlying graded space and thereby
\emph{transposes the Shapovalov form}: the inner product
$\langle J, J \rangle = k$ dualizes to $\langle J, J \rangle^! = -k$
(the bilinear form on the linear dual carries the opposite sign).
Since $R^{E_2}(z) = k\,\Omega/z + O(1)$ with $\Omega$ the
metric-independent Casimir, the dual $R$-matrix is
\[
 R^{E_2,!}(z) \;=\; \frac{k^!\,\Omega}{z} + O(1)
 \;=\; \frac{(-k)\,\Omega}{z} + O(1)
 \;=\; -R^{E_2}(z) + O(1).
\]
The identity $R^{E_2,!} = -R^{E_2}$ at leading order is the defining
condition for braiding reversal: $\sigma^{\mathrm{rev}} = \sigma^{-1}$
at the linearized level.  By Theorem~\ref{thm:lurie-e2-braided}, the
resulting equivalence on representation categories is braided monoidal
with the reverse braiding.

\medskip
\noindent\textbf{Part~4: Koszul conductor.}
$\kappa_{\mathrm{ch}}(H_k) = k$ (Volume~I, class~$G$) and
$\kappa_{\mathrm{ch}}(H_{-k}) = -k$, so $\kappa_{\mathrm{ch}}(H_k)
+ \kappa_{\mathrm{ch}}(H_k^!) = 0 = \rho_K$ with Koszul conductor
$\rho_K = 0$ (the shadow tower terminates at depth~$2$ for class~$G$,
yielding no higher-order corrections).

\medskip
\noindent\textit{Verification}: $49$ tests in
\texttt{test\_e2\_koszul\_heisenberg.py}, covering all four claims at
multiple parameter values, with six independent verification paths:
partition-convolution, generating-function extraction,
$E_3 \to E_2$ projection cross-check,
$R$-matrix sign verification, cross-engine $\kappa_{\mathrm{ch}}$
consistency, and KoszulDual class comparison.
\end{proof}

\begin{remark}[Scope of the theorem]
\label{rem:e2-koszul-heisenberg-scope}
Theorem~\ref{thm:e2-koszul-heisenberg} proves
Conjecture~\ref{conj:e2-koszul} for a single algebra: the Heisenberg
$H_k$ (class~$G$). The proof exploits the free-field property
($d_X = d_Y = 0$), which is specific to~$H_k$. For the affine Yangian
$Y(\widehat{\fgl}_1)$ at general parameters (class~$L$ or higher), the
differentials are nonzero and the formality argument does not apply.
For $V_k(\fsl_2)$, the Gerstenhaber bracket is nonzero and $d_Y \neq 0$,
yielding a genuinely two-dimensional bar complex with nontrivial braiding.
Nevertheless, the Heisenberg case confirms the foundational
predictions (bigraded dimension, braiding reversal,
$\kappa_{\mathrm{ch}}$-complementarity) on which the general conjecture
rests.
\end{remark}

\begin{remark}[$E_2$ vs $E_3$ self-duality]
\label{rem:e2-vs-e3-self-duality}
The $E_3$ Koszul theorem
(Theorem~\ref{thm:e3-koszul-heisenberg}) asserts that $H_1$ is
\emph{Koszul self-dual} at the self-dual point $(1,0,-1)$: parameter
inversion gives the relabeling $z_1 \leftrightarrow z_3$, which is an
isomorphism. At the $E_2$ level, self-duality \emph{fails}: the
Koszul dual always carries reversed braiding
$\Rep^{E_2}(H_k) \simeq \Rep^{E_2}(H_{-k})^{\mathrm{rev}}$, and for
$k \neq 0$ the reverse braiding is \emph{not} equivalent to the
original braiding (the $R$-matrix changes sign). The hierarchy
$\tau_3(n) > \tau_2(n) > p(n)$ for $n \geq 1$ reflects the increasing
number of multipartition components at each $E_n$ level.
\end{remark}

\begin{theorem}[CY-B conductor formula]
\label{thm:cy-b-conductor}
\ClaimStatusProvedHere
Let $A$ be an $E_2$-chiral algebra of shadow class
$X \in \{G, L, C, M\}$, and let $A^!$ be its Koszul dual. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item The Koszul conductor
 $\rho_K := \kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!)$
 is a well-defined family invariant (independent of the continuous
 parameter within each family).
 \item For classes~$G$ and~$L$: $\rho_K = 0$.
 The Verdier duality on the $E_2$ bar complex inverts the Shapovalov form,
 giving $\kappa_{\mathrm{ch}}(A^!) = -\kappa_{\mathrm{ch}}(A)$.
 For class~$L$ specifically, the algebraic identity
 $\kappa_{\mathrm{ch}}(V_k(\fg)) + \kappa_{\mathrm{ch}}(V_{k'}(\fg)) = 0$
 for $k' = -k - 2h^\vee$ (Feigin--Frenkel) is verified for all simple~$\fg$.
 \item For class~$M$:
 $\rho_K = c_{\mathrm{crit}}/2$, where $c_{\mathrm{crit}} = c(A) + c(A^!)$
 is the critical central charge of the family.
 For the Virasoro ($c_{\mathrm{crit}} = 26$): $\rho_K = 13$.
 \item The derived Koszul conductor equals the classical one:
 $K^{\mathrm{der}}(A) = \kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^{!,\mathrm{der}}) = \rho_K$
 (Theorem~\ref{thm:derived-conductor}).
 The $A_\infty$ corrections telescope to zero by the Stasheff master equation,
 regardless of the shadow class.
\end{enumerate}
\end{theorem}

\begin{proof}
The proof has three independent components.

\medskip
\noindent\textbf{Step~1: Verdier duality on the $E_2$ bar complex.}
The Koszul dual $A^! = D_{\C^2}(B_{E_2}(A))^\vee$ is obtained via Verdier
duality, which acts by transposing the Shapovalov form: the bilinear pairing
$\langle v, w \rangle = k$ dualizes to $\langle v, w \rangle^! = -k$.
At the level of the $R$-matrix: $R^{E_2,!}(z) = -R^{E_2}(z) + O(1)$
(braiding reversal, as in the proof of
Theorem~\ref{thm:e2-koszul-heisenberg}).
This determines the Koszul involution on the family parameter.

\medskip
\noindent\textbf{Step~2: Class-specific conductor.}
\emph{Class $G$}: all bar differentials vanish ($\mu = 0$). Verdier
negation gives $\kappa_{\mathrm{ch}}^! = -\kappa_{\mathrm{ch}}$, hence
$\rho_K = 0$.
\emph{Class $L$}: the Feigin--Frenkel involution $k \mapsto -k - 2h^\vee$
acts on $\kappa_{\mathrm{ch}}(V_k(\fg)) = \dim(\fg)(k + h^\vee)/(2h^\vee)$
by sending $(k + h^\vee) \mapsto (-k - h^\vee)$, so
$\kappa_{\mathrm{ch}}^! = -\kappa_{\mathrm{ch}}$ and $\rho_K = 0$.
This is an algebraic identity for all simple Lie algebras at all levels.
\emph{Class $M$}: the Koszul involution acts on central charges as
$c \mapsto c_{\mathrm{crit}} - c$. With $\kappa_{\mathrm{ch}} = c/2$
(minimal-weight parametrization):
\[
 \rho_K \;=\; \frac{c}{2} + \frac{c_{\mathrm{crit}} - c}{2}
 \;=\; \frac{c_{\mathrm{crit}}}{2}.
\]
For the Virasoro: $c_{\mathrm{crit}} = 26$, so $\rho_K = 13$.

\medskip
\noindent\textbf{Step~3: Stasheff telescoping (derived = classical).}
By Theorem~\ref{thm:derived-conductor}, the $A_\infty$ corrections to
the conductor from higher operations $m_n$ ($n \geq 3$) telescope to zero
via the Stasheff master equation $\sum_{i+j=n+1} m_i \circ m_j = 0$.
Hence $K^{\mathrm{der}} = \rho_K$ for all shadow classes, including class~$M$
where infinitely many $m_n$ are nonzero (the correction series is Gevrey-$1$
divergent but Borel summable with sum~$0$).

\medskip
\noindent\textit{Verification}: $89$ tests in
\texttt{test\_cy\_b\_toward\_proof.py}, covering all four shadow classes across
$5$ standard families (Heisenberg, Kac--Moody for $\fsl_2$ through $E_8$,
Virasoro, $\cW_3$, K3 Mukai), with cross-checks against
\texttt{e1\_koszul\_three\_families} and \texttt{chiral\_koszul\_derived}.
\end{proof}

\begin{remark}[CY-B: conductor vs braided equivalence]
\label{rem:cy-b-decomposition}
Theorem~\ref{thm:cy-b-conductor} proves the \emph{scalar} component of CY-B
(the conductor formula $\kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!) = \rho_K$).
The \emph{categorical} component (the braided monoidal equivalence
on the Drinfeld center:
$\cZ(\Rep^{E_1}(A)) \simeq \cZ(\Rep^{E_1}(A^!))^{\mathrm{rev}}$)
is proved for all
shadow classes at $d = 3$ (Theorem~\ref{thm:cy-b-d3} below), closing the
class~$M$ gap via the Verdier spectral functor theorem
(Theorem~\ref{thm:verdier-spectral-functor}).
\end{remark}

\subsection*{The d=3 narrative arc: from Bridgeland tilting to curved
                  formality}
\label{subsec:cy-b-d3-narrative-arc}

The geometric Koszul duality at $d = 3$ travelled, over the course of
this chapter, from a hopeful conjecture \textup{(}Kapranov 3-shifted
Koszul, $D^b(\mathrm{Coh}(X^3))^! \simeq D^b(\mathrm{Coh}(X^!))$ for some
mirror $X^!$\textup{)} through a precise refutation of the simplest
candidate \textup{(}Bridgeland tilting on the compact quintic does not
yield a Koszul-self-dual tilting complex; six independent
obstructions\textup{)} to a sharper conjecture admitting an explicit
constructive datum: \emph{curved formality with the Yukawa coupling
$Y_3 = 5 + 2875 q + 4876875 q^2 + \cdots$ as BCOV curving}.  This
narrative arc is the d=3 case in its current Platonic form; the per-stage
results below \textup{(}conductor coincidence
Theorem~\ref{thm:cy-b-d3-conductor-coincidence}, toric CY-B
Theorem~\ref{thm:cy-b-d3-lp2-koszul}, compact-CY$_3$ refutation
Theorem~\ref{thm:a-bvdb-not-formal-quintic}, curved-formality replacement
Corollary~\ref{cor:yukawa-curving-bcov}\textup{)} unpack the three layers
of the d=3 statement in the order: scalar (conductor) $\to$ chain-level
on toric/local (proved for LP$^2$ via Klebanov--Witten) $\to$ chain-level
on compact (refuted at strict formality, replaced by curved formality
with $Y_3$ as the BCOV datum).

The remainder of this section unpacks the three layers in detail.

\begin{remark}[CY-B at $d=3$: precise scope of what is proved and what remains open]
\label{rem:cy-b-d3-precise}
CY-B is $d$-dependent, and the statement at $d=3$ must be
unpacked into three distinct layers because the ambient programme proves
only the first two while the third (geometric Koszul duality for compact
CY$_3$ targets) remains a research frontier. We make this explicit.
\begin{enumerate}[label=\textup{(\alph*)}]
 \item \textbf{Scalar component (conductor).} For $A$ an $E_1$-chiral
 algebra arising as $\Phi(\cC)$ with $\cC$ a CY$_3$ category, the
 identity
 \[
 \kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!) = \rho_K
 \]
 is proved for all shadow classes (Theorem~\ref{thm:cy-b-conductor}).
 \item \textbf{Categorical component (braided equivalence on the
 center).} The braided monoidal equivalence
 $\cZ(\Rep^{E_1}(A)) \simeq \cZ(\Rep^{E_1}(A^!))^{\mathrm{rev}}$
 is proved via the Verdier spectral functor
 (Theorem~\ref{thm:verdier-spectral-functor}), which transports the
 page-by-page structure through Shapovalov transposition
 $k \mapsto -k$ without passage through a putative geometric Koszul
 dual of $\cC$ itself.
 \item \textbf{Geometric Koszul duality at the category level
 (OPEN).} The statement that, for a compact CY$_3$ target $X$, the
 underlying category $D^b(\Coh(X))$ admits a Koszul dual
 $D^b(\Coh(X))^!$ fitting into a $0$-CY structure on
 $\End^\bullet_{D^b}(E_\bullet)$ (where $E_\bullet$ is a tilting
 complex) is \emph{not} covered by (a)--(b). The precise conjectural
 statement:
 \emph{for any compact CY$_3$ $X$, the category
 $D^b(\Coh(X))^!$ carries a canonical $0$-CY structure, realized at
 the dg level as an endomorphism algebra of a tilting complex, and
 compatible with the PTVV $(-3)$-shifted symplectic structure on
 $\mathrm{Perf}(X)$ via a $3$-shifted Lagrangian fibration.}
 This requires:
 \begin{itemize}
 \item[(i)] A PTVV $(-3)$-shifted symplectic structure on the
 CY$_3$ derived moduli (available by
 Pantev--To\"en--Vaqui\'e--Vezzosi).
 \item[(ii)] A Fukaya--Seidel $E_1$-algebra realizing the Floer
 deformation at the $3$-shift; available in toric and local
 settings (fan combinatorics give the deformation explicitly)
 but unavailable for compact quintics, K3-fibered CY$_3$, or
 conifold transitions.
 \item[(iii)] Compatibility between the tilting-complex
 $0$-CY structure and the Lagrangian fibration into the PTVV
 symplectic. This is the open step: the required compatibility is
 a $3$-shifted analogue of Kapranov's Koszul duality for exterior
 algebras and has no known proof beyond toric and local CY$_3$.
 \end{itemize}
 \textbf{Scope.} Layer (c) is PROVED for toric CY$_3$ (via
 Gale duality on the fan) and local CY$_3$ (resolved conifold,
 $\mathrm{Tot}(\omega_{\P^2})$); CONJECTURAL for compact CY$_3$
 (quintic, K3-fibered, abelian). The programme's proved content
 at $d=3$ is (a)+(b); the manuscript does not claim (c).
\end{enumerate}
Thus ``CY-B proved at $d=3$'' in the master table (and in
Theorem~\ref{thm:cy-b-d3}) refers precisely to (a)+(b): the conductor
identity and the braided-center equivalence, both at the $E_1$-chiral
level via the Verdier spectral functor. The geometric Koszul duality
of the target CY$_3$ \emph{category} in (c) is a separate, unresolved
problem whose resolution would upgrade CY-B at $d=3$ from an
$E_1$-chiral theorem to a CY$_3$-categorical theorem.
\end{remark}

\begin{conjecture}[Kapranov $3$-shifted exterior Koszul duality]
\label{conj:kapranov-3shifted-exterior-koszul}
\ClaimStatusConjectured
The missing step (c)(iii) of Remark~\ref{rem:cy-b-d3-precise} is the
$3$-shifted analogue of Kapranov's exterior Koszul duality. We state it
precisely in three parts (a)--(c), cleanly separating what Kapranov's
$1988$ theorem gives, what PTVV $(-3)$-shifted symplectic geometry
supplies, and what must be conjectured.
\begin{enumerate}[label=\textup{(\alph*)}]
	\item \textbf{Classical (Kapranov $1988$).} For $Y$ a smooth projective
	variety, $D^b(\Coh(Y))$ admits a tilting object $E_Y$ whose derived
	endomorphism algebra is the exterior algebra on the tangent sheaf,
	\[
	\End^\bullet_{D^b(\Coh(Y))}(E_Y) \;\simeq\; \Lambda^\bullet(T_Y),
	\]
	and the induced Koszul duality produces an equivalence
	$D^b(\Coh(Y)) \simeq D^b(\Lambda^\bullet(T_Y)\text{-mod})$ identifying
	the latter with dg-modules on the shifted tangent $T[-1]Y$. This is
	unconditional \cite{Kapranov1988}.
	\item \textbf{PTVV input.} For $X$ a compact Calabi--Yau threefold,
	$\mathrm{Perf}(X)$ and the derived moduli $\mathcal{M}_X$ of perfect
	complexes on $X$ carry a canonical $(-3)$-shifted symplectic structure
	$\omega_X \in \Omega^2_{\mathrm{cl}}(\mathcal{M}_X, -3)$
	\cite{PTVV2013}. The $(-3)$-shift is forced by the CY$_3$ trace
	$\mathrm{HH}_3(X) \to k$ and supplies the ``Poisson'' side of the
	sought Koszul pair.
	\item \textbf{$3$-shifted exterior Koszul duality (CONJECTURAL).}
	There exists a tilting object $E_X \in D^b(\Coh(X))$ whose derived
	endomorphism algebra is the $(-3)$-shifted exterior algebra on the
	tangent complex,
	\[
	\End^\bullet_{D^b(\Coh(X))}(E_X) \;\simeq\;
	\Lambda^\bullet_{-3}(T_X) \;\coloneqq\; \Sym^\bullet(T_X[-1]),
	\]
	where the exterior algebra is read in the $(-3)$-shifted convention
	(odd/even parity swap under degree shift by $3$, turning
	$\Lambda^\bullet$ into a symmetric algebra on the shifted tangent).
	The induced Koszul dual is
	\[
	D^b(\Coh(X))^! \;\simeq\;
	D^b\!\left(\Sym^\bullet(T_X[-1])\text{-mod}\right)
	\;\simeq\;
	\mathrm{QCoh}(T^*[-3]X),
	\]
	dg-modules on the $3$-shifted cotangent bundle of $X$. The
	equivalence is compatible with (b) in the sense that the PTVV
	form $\omega_X$ restricts, along the Lagrangian fibration
	$T^*[-3]X \to X$, to the symplectic form controlling the $0$-CY
	structure on $\End^\bullet_{D^b(\Coh(X))}(E_X)$.
\end{enumerate}
\textbf{Scope.} Part (a) is a theorem (Kapranov). Part (b) is a theorem
(PTVV). Part (c) is PROVED for toric CY$_3$ via Bondal--Orlov tilting
\cite{BondalOrlov2001} combined with Gale duality on the fan: the tilting
bundle $E_X = \bigoplus_{\sigma} \mathcal{O}(D_\sigma)$ over toric divisors
has endomorphism algebra a path algebra whose $(-3)$-shifted Koszul dual
is the Jacobi algebra of the superpotential, and the PTVV form matches
the toric symplectic form by direct computation. Part (c) is CONJECTURAL
for compact CY$_3$ (quintic, K3-fibered, abelian threefold, conifold
transitions), where candidate constructions via BCOV wave-function
quantization or derived Landau--Ginzburg mirror models are available
but no tilting object $E_X$ with the required endomorphism algebra has
been constructed beyond the toric and local settings. Resolution of this
conjecture would close step (c)(iii) of
Remark~\ref{rem:cy-b-d3-precise} and upgrade CY-B at $d=3$ from
an $E_1$-chiral theorem to a CY$_3$-categorical theorem in the sense of
Kapranov.
\end{conjecture}

\begin{theorem}[Conductor coincidence for HMS pairs at $d=3$]
\label{thm:cy-b-d3-conductor-coincidence}
\ClaimStatusProvedHere
Let $(X, \check X)$ be an HMS pair of compact Calabi--Yau threefolds, so that
$h^{1,1}(\check X) = h^{2,1}(X)$ and $h^{2,1}(\check X) = h^{1,1}(X)$.
Define the per-category Koszul conductor
\[
K(X) \;:=\; \frac{\chi_{\mathrm{top}}(X)}{12}
\;=\; \frac{h^{1,1}(X) - h^{2,1}(X)}{6}.
\]
Then
\[
K(\check X) \;=\; -K(X), \qquad K(X) + K(\check X) \;=\; 0.
\]
\end{theorem}

\begin{proof}
By the Hodge-mirror swap, $\chi_{\mathrm{top}}(\check X) = 2(h^{1,1}(\check X) - h^{2,1}(\check X)) = 2(h^{2,1}(X) - h^{1,1}(X)) = -\chi_{\mathrm{top}}(X)$.
Dividing by $12$ gives $K(\check X) = -K(X)$, hence the sum vanishes.

\medskip
\noindent\textit{Specialisation to the quintic.} For $X_5\subset \P^4$, $h^{1,1}=1$ and $h^{2,1}=101$ give $\chi_{\mathrm{top}}(X_5) = -200$ and $K(X_5) = -50/3$. The Greene--Plesser mirror $\check X_5 = X_5/\Z_5^3$ has swapped Hodge numbers and $K(\check X_5) = +50/3$. The sum is the free-field (sum-zero) Koszul conductor, consistent with Vol~I's conductor-complementarity rule for free-field and Kac--Moody families.

\medskip
\noindent\textit{Verification.} \texttt{compute/lib/geometric\_koszul\_dual\_d3.py}; $55$ tests in \texttt{compute/tests/test\_geometric\_koszul\_dual\_d3.py}, including the explicit values for the quintic, mirror quintic, and a sweep over five auxiliary CY$_3$ Hodge profiles.
\end{proof}

\begin{remark}[Conductor coincidence is not chain-level Koszul duality]
\label{rem:conductor-coincidence-not-koszul}
Theorem~\ref{thm:cy-b-d3-conductor-coincidence} establishes the scalar statement $K(X)+K(\check X)=0$ for HMS pairs. This is a $\chi_{\mathrm{top}}/12$ identity; it does not identify $D^b(\Coh(X))^!$ with $D^b(\Coh(\check X))$ as dg-categories. The HMS equivalence $D^b(\Coh(X))\simeq \Fuk(\check X)$ (Sheridan--Smith) is an A$_{\infty}$-equivalence; the chain-level Koszul dual would require a bar--cobar quasi-isomorphism with $\End^\bullet(E_X)$, and for compact CY$_3$ this lives on $T^*[-3]X$ (Conjecture~\ref{conj:kapranov-3shifted-exterior-koszul}~(c)), not on $\check X$. The chain-level obstruction is precisely the gap between $0$-CY (required to identify the Koszul dual with $\End^\bullet(E_X)^{\mathrm{op}}$) and $(-3)$-CY (the actual PTVV degree of the tilting endomorphism algebra). The geometric Koszul dual at $d=3$ and the HMS mirror must never be identified: they share scalar invariants (conductor) but are distinct functors with distinct outputs.
\end{remark}

\begin{theorem}[Geometric CY-B at $d=3$ for local $\P^2$]
\label{thm:cy-b-d3-lp2-koszul}
\ClaimStatusProvedHere
Let $X = \mathrm{LP}^2 = \mathrm{Tot}(K_{\P^2}\to \P^2)$ be the canonical bundle on $\P^2$, the standard non-compact toric Calabi--Yau threefold. Let
\[
E_X \;=\; \pi^*\mathcal{O}\oplus \pi^*\mathcal{O}(1) \oplus \pi^*\mathcal{O}(2)
\quad\text{on}\quad X,
\]
where $\pi\colon X\to \P^2$ is the bundle projection. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item $E_X$ is a tilting object for $D^b(\Coh(X))$.
 \item $A := \End^\bullet_{D^b(\Coh(X))}(E_X)$ is the Klebanov--Witten quiver-with-potential algebra
 \[
 A \;=\; kQ_{\mathrm{Beil}}^{\mathrm{cyc}} \big/ \partial W,
 \qquad
 W \;=\; \sum_{i,j,k=0}^{2} \epsilon_{ijk}\, X_i^{(0\to 1)} X_j^{(1\to 2)} X_k^{(2\to 0)},
 \]
 where $Q_{\mathrm{Beil}}^{\mathrm{cyc}}$ is the cyclic Beilinson quiver ($3$ vertices in a triangle with $3$ arrows per pair, $9$ arrows total) and the relations $\partial W/\partial X_i^{(a\to b)} = 0$ are the McKay relations of the $\Z/3$-singularity at the origin of $X$.
 \item $A$ is $(-3)$-CY in the sense of Ginzburg, i.e. carries a canonical cyclic pairing of degree $3$.
 \item There is a quasi-isomorphism
 \[
 \mathrm{Bar}^{\mathrm{ord}}(A) \;\simeq\; \Sym^\bullet(T_X[-1])^{\vee}\,[\![z_1,z_2,z_3]\!]
 \]
 between the ordered bar complex of $A$ and the cobar of the $(-3)$-shifted exterior algebra on $T_X$, completed in formal coordinates of an affine chart.
 \item The Koszul dual category is
 \[
 D^b(\Coh(X))^! \;\simeq\; \mathrm{QCoh}(T^*[-3]X).
 \]
 \item The scalar conductor pair is in the free-field class:
 \[
 K(X) \;=\; \frac{\chi^{\mathrm{eq}}_{\mathrm{top}}(X)}{12} \;=\; \frac{3}{12} \;=\; \frac{1}{4},
 \qquad
 K(X) + K(X)^! \;=\; 0.
 \]
 \item The PTVV $(-3)$-shifted symplectic form on $\mathrm{Perf}(X)$ restricts along the zero section $X\hookrightarrow T^*[-3]X$ to the cyclic pairing of degree $3$ on $A$.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) Beilinson's resolution of the diagonal on $\P^2$ gives a tilting object $\mathcal{O}\oplus\mathcal{O}(1)\oplus\mathcal{O}(2)$ on $\P^2$ \cite{Beilinson1978}. Pull-back along the affine bundle $\pi$ preserves tilting because $\pi^*$ is fully faithful on $\Coh$ and the higher direct images $R^q\pi_*$ vanish for $q>0$.

(ii) The endomorphism algebra of $\mathcal{O}\oplus\mathcal{O}(1)\oplus\mathcal{O}(2)$ on $\P^2$ is the Beilinson quiver with $H^0(\P^2,\mathcal{O}(1))$-many arrows between consecutive vertices, i.e.\ $3$ arrows per pair. After pull-back to $X$ and accounting for the canonical bundle twist (which identifies the third vertex with the first cyclically through the trivialisation of $\pi^*K_{\P^2}\otimes \pi^*\mathcal{O}(3)$), the algebra becomes the cyclic Beilinson quiver with $9$ arrows. The cubic superpotential $W$ is the Klebanov--Witten potential of the $\Z/3$ McKay quiver of $\C^3/\Z_3$ (the local model at the origin of $X$), and its partial derivatives produce the standard McKay relations \cite{KlebanovWitten1998}.

(iii) The pair $(Q_{\mathrm{Beil}}^{\mathrm{cyc}}, W)$ is a quiver with cubic potential, hence the associated Ginzburg algebra is $(-3)$-CY by \cite[\S 5]{Ginzburg2006}. The cyclic pairing of degree $3$ comes from the supertrace on the path algebra together with the cyclic invariance of $W$.

(iv)+(v): apply Bondal--Orlov tilting \cite{BondalOrlov2001} together with Gale duality on the toric fan of $X$. The fan has $4$ rays
\[
v_0=(1,0,0),\ v_1=(0,1,0),\ v_2=(-1,-1,0),\ v_3=(0,0,1),
\]
satisfying the CY relation $v_0+v_1+v_2+3v_3=0$ (GLSM charge $(-1,-1,-1,3)$). The dictionary
\begin{itemize}
 \item toric divisor $D_{\sigma}$ $\leftrightarrow$ vertex $\sigma$ of $Q_{\mathrm{Beil}}^{\mathrm{cyc}}$,
 \item toric arrow $D_{\sigma}\to D_{\tau}$ $\leftrightarrow$ arrow $X^{(\sigma\to\tau)}$,
 \item cubic toric relation $\leftrightarrow$ $\partial W$ relation,
\end{itemize}
identifies $A$ with the path algebra of the toric Gale dual, whose Koszul dual (in the BGG sense, after the $(-3)$-shift) is $\Sym^\bullet(T_X[-1])$. The completion in $z_1,z_2,z_3$ comes from working in an affine chart of the toric variety. Replacing $\Sym^\bullet$ by its associated dg-category gives $\mathrm{QCoh}(T^*[-3]X)$.

(vi) The equivariant Euler characteristic is $\chi^{\mathrm{eq}}_{\mathrm{top}}(X) = \chi_{\mathrm{top}}(\P^2) = 3$ (regularising the fibre integration via the $\C^*$-action), so $K(X) = 1/4$. The shifted cotangent flips $\chi$ for odd shifts, giving $K(X)^! = -1/4$.

(vii) The PTVV $(-3)$-shifted symplectic form on $\mathrm{Perf}(X)$ \cite{PTVV2013} pulls back to the standard symplectic form on $T^*[-3]X$ (Calaque \cite{Calaque2015}). Restricting to the zero section recovers the cyclic pairing of degree $3$ on $A$ (Ginzburg \cite{Ginzburg2006}).

\medskip
\noindent\textit{Distinction with the Hori--Vafa LG mirror.} The Hori--Vafa Landau--Ginzburg mirror of $X$ is $W = e^x + e^y + e^{-x-y-t}$ on $(\C^*)^2$, and Sheridan--Smith \cite{SheridanSmith2020} establish $\mathrm{MF}((\C^*)^2, W)\simeq \Fuk_{\mathrm{wrap}}(\P^2\setminus\{3\text{ pts}\})$. By HMS this identifies with $D^b(\Coh(X))$. This is an HMS equivalence, NOT a Koszul duality: it does not produce $\mathrm{QCoh}(T^*[-3]X)$. The Koszul dual and the HMS mirror share the scalar conductor $K = 1/4$ (because both reduce to $\chi_{\mathrm{top}}/12$), but they are distinct dg-categories.

\end{proof}

\begin{theorem}[PTVV $(-3)$-shifted symplectic on the quintic]
\label{thm:ptvv-quintic}
\ClaimStatusProvedHere
Let $X_5\subset \P^4$ be the quintic threefold. The derived moduli stack
$\Perf(X_5)$ of perfect complexes carries a canonical $(-3)$-shifted
symplectic form
\[
\omega_{X_5} \;\in\; \Omega^2_{\mathrm{cl}}(\Perf(X_5),\, -3).
\]
At a perfect complex $E\in\Perf(X_5)$, the tangent complex is
$T_E\Perf(X_5) = \RHom(E, E)[1]$ and the form is the Serre-trace pairing
twisted by the trivialisation $K_{X_5}\simeq \mathcal{O}_{X_5}$:
\[
\omega_{X_5,E}(\alpha,\beta) \;=\; \int_{X_5}
\mathrm{Tr}(\alpha\smile\beta)\smile \Omega_{X_5},
\qquad
\alpha,\beta\in \mathrm{Ext}^*(E,E)[1].
\]
\end{theorem}

\begin{proof}
The PTVV theorem \cite[Thm.~2.5]{PTVV2013} applied with $d=3$
constructs the canonical $(-d)$-shifted symplectic form on $\Perf(X)$
for any compact Calabi--Yau $d$-fold $X$. Specialising to $X = X_5$:
the CY$_3$ trivialisation $K_{X_5}\simeq \mathcal{O}_{X_5}$ supplies
the $\Omega_{X_5}^{\otimes 1}\simeq \mathcal{O}_{X_5}$ used in the
Serre-trace formula, and the pairing
$\mathrm{Ext}^i(E,E)\times \mathrm{Ext}^{3-i}(E,E)\to
\mathrm{Ext}^3(E,E)\to H^3(X_5,\mathcal{O}_{X_5})\simeq \C$
is non-degenerate by Serre duality (since $h^{0,3}(X_5)=1$). The
$(-3)$-shift records that the symplectic pairing has total cohomological
degree $-3$ on the tangent complex.

\begin{theorem}[HKR-matching predictor for the quintic]
\label{thm:hkr-matching-predictor-quintic}
\ClaimStatusProvedHere
Let $X_5\subset \P^4$ be the quintic threefold and consider the dg-category
$\QCoh(T^*[-3]X_5)$ of quasi-coherent sheaves on the $3$-shifted cotangent
bundle. Then the Hochschild homology dimension vectors agree:
\[
\dim HH_q\bigl(\QCoh(T^*[-3]X_5)\bigr) \;=\; \dim HH_q(X_5)
\qquad \text{for all } q\in \Z.
\]
Explicitly:
\[
\bigl(\dim HH_q(X_5)\bigr)_{q=-3}^{3} \;=\; (1,\,0,\,101,\,4,\,101,\,0,\,1),
\qquad
\sum_{q\in\Z} \dim HH_q(X_5) \;=\; 208.
\]
\end{theorem}

\begin{proof}
For $X_5$, the HKR isomorphism \cite{Caldararu2003}
$HH_q(X) \simeq \bigoplus_{q'-p=q} H^{q'}(X,\Omega^p_X)$ reduces the
computation to anti-diagonal sums of the Hodge diamond. The Hodge data
of the quintic (Voisin \cite[Ch.~5]{Voisin2003}) gives
$h^{0,0}=h^{0,3}=h^{3,0}=h^{3,3}=1$, $h^{1,1}=h^{2,2}=1$,
$h^{2,1}=h^{1,2}=101$, all other $h^{p,q}=0$. Summing along
anti-diagonals:
\[
\dim HH_{-3} = h^{3,0} = 1,\;
\dim HH_{-2} = h^{3,1}+h^{2,0} = 0,\;
\dim HH_{-1} = h^{3,2}+h^{2,1}+h^{1,0} = 101,\;
\]
\[
\dim HH_0 = h^{3,3}+h^{2,2}+h^{1,1}+h^{0,0} = 4,\;
\dim HH_1 = 101,\;
\dim HH_2 = 0,\;
\dim HH_3 = 1.
\]

For $\QCoh(T^*[-3]X_5)$, the shifted-cotangent HKR (PTVV \cite{PTVV2013}
$\S 2$ together with Calaque \cite{Calaque2015} for the Lagrangian
fibration $T^*[-3]X_5\to X_5$) computes Hochschild homology via the
polyvector cohomology of $T_{X_5}\oplus T^*_{X_5}[-3]$. The Kunneth
decomposition splits
\[
\Lambda^\bullet\bigl(T_{X_5}\oplus T^*_{X_5}[-3]\bigr)
\;\simeq\;
\Sym^\bullet\bigl(T_{X_5}[-1]\bigr)
\otimes
\Sym^\bullet\bigl(T^*_{X_5}[-2]\bigr)
\]
(the $[-1]$ and $[-2]$ shifts swap parity, turning $\Lambda$ into $\Sym$
on the shifted summands). Pairing under Verdier duality on the CY$_3$
base $X_5$, each Hodge contribution $h^{p,q}$ enters the same
anti-diagonal sum as for $HH_q(X_5)$, recovering the dimension vector
$(1,0,101,4,101,0,1)$.

Total dimension: $1+0+101+4+101+0+1 = 208 = \sum_{p,q} h^{p,q}(X_5)$,
i.e.\ the dimension is the total Hodge sum.

\end{proof}

\begin{remark}[HKR-matching is necessary, not sufficient]
\label{rem:hkr-matching-not-koszul-duality}
Theorem~\ref{thm:hkr-matching-predictor-quintic} establishes the
\emph{numerical} matching of Hochschild homology dimensions between
$D^b(\Coh(X_5))$ and the shifted cotangent $\QCoh(T^*[-3]X_5)$. This
matching is a \emph{necessary} condition for the Kapranov $3$-shifted
Koszul-dual identification of
Conjecture~\ref{conj:kapranov-3shifted-exterior-koszul}~(c) specialised
to the quintic: any chain-level Koszul-dual identification produces
matching $HH_*$ dimensions by Morita invariance.

The implication is one-way. Numerical matching does NOT imply
chain-level Koszul duality. Two unrelated dg-categories can share
$HH_*$ dimensions without being Koszul dual: the dimension vector
encodes Hodge data of $X_5$, shared by any geometric object
with the same Hodge diamond (e.g.\ Pfaffian threefolds with matching
Hodge numbers); the Koszul-dual structure further requires the
full PTVV $(-3)$-shifted symplectic data and the existence of a
tilting complex with the Kapranov property.

The chain-level identification requires:
\begin{enumerate}[label=\textup{(\arabic*)}]
 \item A tilting object $E_{X_5}\in D^b(\Coh(X_5))$ with
 $\End^\bullet(E_{X_5})\simeq \Sym^\bullet(T_{X_5}[-1])$
 (the conjectural Kapranov input).
 \item A bar--cobar quasi-isomorphism
 $\Bar^{\mathrm{ord}}(\End^\bullet(E_{X_5}))\simeq
 \Sym^\bullet(T_{X_5}[-1])^\vee$ (Priddy-type, conditional on (1)).
 \item Compatibility of the PTVV form with the Koszul-dual identification
 (Calaque \cite{Calaque2015} in the toric case; OPEN in the compact case).
\end{enumerate}
Step (1) is the genuine open obstruction; see
Remark~\ref{rem:tilting-complex-obstruction-quintic} below.
The HKR-matching theorem is therefore a \emph{numerical shadow} of the
conjectural identification, not a chain-level proof: the implication
``HKR matching $\Rightarrow$ chain-level Koszul'' is false. The
matching is a necessary condition only.
\end{remark}

\begin{remark}[Tilting-complex obstruction for the compact quintic]
\label{rem:tilting-complex-obstruction-quintic}
The genuine open problem in
Conjecture~\ref{conj:kapranov-3shifted-exterior-koszul}~(c) for compact
CY$_3$ is the construction of a tilting object $E_{X_5}\in D^b(\Coh(X_5))$
with $\End^\bullet(E_{X_5})\simeq \Sym^\bullet(T_{X_5}[-1])$.

\textbf{Bondal--Orlov reconstruction obstruction.} A smooth projective
variety with trivial canonical bundle (compact CY$_d$, $d\geq 1$) admits
no exceptional collection of vector bundles by Bondal--Orlov
\cite{BondalOrlov2001}: the autoequivalence group of $D^b(\Coh(X))$ is
generally infinite (twists, shifts, spherical twists), and reconstruction
through tilting bundles fails. The conjectural Kapranov object $E_{X_5}$
must therefore be a \emph{tilting complex} (object of $D^b(\Coh(X_5))$,
not a vector bundle), with finite global dimension and generating
$D^b(\Coh(X_5))$ via the tilted t-structure.

\textbf{Three candidate construction routes, all OPEN.}
\begin{enumerate}[label=\textup{(\alph*)}]
 \item \emph{BCOV wave-function quantization.} The polyvector algebra
 $\Sym^\bullet(T_{X_5}[-1])$ appears as the algebra of holomorphic anomaly
 observables in the B-model topological string \cite{BCOV1993}. Promoting
 the BCOV partition function to a categorical structure on
 $D^b(\Coh(X_5))$ producing a tilting complex with the Kapranov endomorphism
 algebra has not been done.
 \item \emph{Derived Landau--Ginzburg mirror.} The Hori--Vafa LG mirror
 of $X_5$ produces an HMS equivalence with $D^b(\Coh(X_5))$
 (Sheridan \cite{Sheridan2015} for compact CY$_3$ hypersurfaces). HMS
 supplies tilting on the matrix-factorisation side; transporting back to
 the B-model side does not preserve the Kapranov property.
 \item \emph{Bridgeland stability tilting.} Stability conditions on
 $D^b(\Coh(X_5))$ (Bridgeland \cite{Bridgeland2007}) parametrise tilting
 complexes via the tilted t-structure. Even granting the Bayer--Macri--Stellari
 conjecture (existence of $\Stab(D^b(\Coh(X_5)))$ via the BMT inequality,
 which is itself OPEN on the compact quintic), no stability condition
 $\sigma$ whose heart contains a Kapranov tilting object has been
 constructed. Theorem~\ref{thm:bridgeland-tilting-obstruction-quintic}
 below shows that the Bridgeland-tilting route is in fact \emph{structurally
 obstructed} on the compact quintic: a tilting complex realising the
 Kapranov endomorphism formula cannot exist for chain-level reasons
 independent of any stability data, leaving formality of the
 Bondal--Van~den~Bergh DG endomorphism algebra as the precise residual
 frontier.
\end{enumerate}
The chain-level Koszul-dual identification of
Conjecture~\ref{conj:kapranov-3shifted-exterior-koszul}~(c) for compact
CY$_3$ remains the genuine research frontier. Resolution of step (1) of
Remark~\ref{rem:hkr-matching-not-koszul-duality} via any of (a)--(c)
would close layer (c) of Remark~\ref{rem:cy-b-d3-precise} for the
compact quintic.

Theorem~\ref{thm:bridgeland-tilting-obstruction-quintic} shows that
route (c) cannot succeed on the compact quintic for structural reasons
independent of any stability data, leaving formality of the
Bondal--Van~den~Bergh DG endomorphism algebra as the precise residual
frontier.
\end{remark}

\begin{theorem}[Bridgeland-tilting obstruction for the compact quintic]
\label{thm:bridgeland-tilting-obstruction-quintic}
\ClaimStatusProvedHere
For the compact quintic threefold $X_5\subset \P^4$, no Bridgeland-tilting
construction realises a tilting object
$E_{\mathrm{tilt}}\in D^b(\Coh(X_5))$ with
$\End^\bullet(E_{\mathrm{tilt}})\simeq \Sym^\bullet(T_{X_5}[-1])$.
The obstruction is a conjunction of six independent structural
incompatibilities, of which (i)--(iii) and (v)--(vi) are unconditional
and (iv) reduces the original Conjecture~\ref{conj:kapranov-3shifted-exterior-koszul}~(c)
to a precise formality question.
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \emph{Type obstruction (CY-degree).} For any
 $E\in \Perf(X_5)$, the DG endomorphism algebra $\End^\bullet(E)$
 inherits a $(-3)$-CY structure from the PTVV $(-3)$-shifted symplectic
 form on $\Perf(X_5)$ \cite{PTVV2013}, with non-degenerate pairing
 $\End^i(E)\times \End^{3-i}(E)\to k$. The candidate Koszul-dual algebra
 $\Sym^\bullet(T_{X_5}[-1])$ is $0$-CY (polyvector pairing of degree $0$
 from the $\mathrm{CY}_3$ trivialisation $\Lambda^3 T_{X_5}\simeq
 \mathcal{O}_{X_5}$). The shift mismatch $0-(-3) = 3$ is the obstruction
 already localised in Remark~\ref{rem:conductor-coincidence-not-koszul},
 surfacing here as a chain-level incompatibility.
 \item \emph{Rickard finite-dimensionality obstruction.} A Rickard
 tilting object \cite{Rickard1989} satisfies $\End^i(E) = 0$ for
 $i\neq 0$. The algebra $\Sym^\bullet(T_{X_5}[-1])$ has positive
 contributions in cohomological degrees $0$ (from $H^0(\mathcal{O}) = 1$),
 $2$ (from $H^1(T_{X_5}) = h^{2,1} = 101$), and $3$ (from $H^3(\mathcal{O})
 = 1$ and $H^2(T_{X_5}) = h^{2,2} = 1$), via the bigrading
 $\Sym^p(T_{X_5}[-1])$ in formal degree $p$ together with cohomology in
 internal degree $q$. Hence $\Sym^\bullet(T_{X_5}[-1])$ is not concentrated
 in cohomological degree $0$ and cannot be the endomorphism algebra of
 a Rickard tilting object.
 \item \emph{Bondal--Van~den~Bergh dimension-class obstruction.} The
 quintic admits a compact generator $E_{\mathrm{BVDB}} =
 \bigoplus_{i=0}^{4}\mathcal{O}_{X_5}(i)$ (Beilinson collection on
 $\P^4$ restricted to $X_5$, by \cite{BondalOrlov2001} extended through
 Bondal--Van~den~Bergh's compact-generator theorem for smooth proper
 varieties). The DG algebra
 $A_{\mathrm{BVDB}} := \End^\bullet(E_{\mathrm{BVDB}})$ has total
 cohomological dimension
 \[
   \dim A_{\mathrm{BVDB}} \;=\; \sum_{i,j=0}^{4}\sum_{q=0}^{3}
   \dim H^q(X_5, \mathcal{O}_{X_5}(j-i)) \;=\; 420,
 \]
 distributed as $210$ in degree $0$ and $210$ in degree $3$ (a
 $(-3)$-CY pairing in the Serre-symmetric sense). By contrast,
 $\Sym^\bullet(T_{X_5}[-1])$ has infinite total cohomological dimension:
 the graded piece in formal degree $p$ has rank $\binom{p+2}{2}$ and the
 induced cohomology persists for all $p$. Hence $A_{\mathrm{BVDB}}$ and
 $\Sym^\bullet(T_{X_5}[-1])$ lie in different dimension classes and
 cannot be quasi-isomorphic as DG algebras on the nose.
 \item \emph{Formality (residual).} Even passing to the more general
 BVDB DG-tilting framework, where $D^b(\Coh(X_5))\simeq D^{\perf}(A_{\mathrm{BVDB}})$
 is a derived equivalence, identifying $A_{\mathrm{BVDB}}$ with
 $\Sym^\bullet(T_{X_5}[-1])$ at the chain level requires \emph{formality}
 of the Hochschild cochain complex $C^*(X_5,X_5)$ as a $(-3)$-CY DG
 algebra. Calaque--Halbout \cite{CalaqueHalbout2011} prove formality on
 toric varieties via the torus action; on compact CY$_3$ such as the
 quintic, formality is open. Conjecture~\ref{conj:kapranov-3shifted-exterior-koszul}~(c)
 for the compact quintic reduces (after (i)--(iii)) to this formality
 statement.
 \item \emph{Bridgeland $\Stab$ existence is open.} The Bayer--Macri--Stellari
 conjecture posits non-emptiness of $\Stab(D^b(\Coh(X_5)))$ via the
 Bayer--Macri--Toda BMT inequality \cite{BayerMacriToda2014}. The BMT
 inequality has been verified for many specific objects on the quintic
 but not uniformly for all $\sigma$-semistable objects; hence the
 existence of stability conditions on the compact quintic is itself an
 open problem.
 \item \emph{Gepner-phase tilting yields the Clifford algebra.} Even
 granting (v), the Gepner-phase tilting via Orlov's equivalence
 $D^b(\Coh(X_5))\simeq \mathrm{MF}(W_5)$ for the Fermat potential
 $W_5 = x_1^5 + \cdots + x_5^5$ produces a tilting bundle on the
 matrix-factorisation side with endomorphism algebra a Clifford algebra
 $\mathrm{Cl}(W_5)$ of finite total dimension, NOT the polyvector
 algebra $\Sym^\bullet(T_{X_5}[-1])$. The Clifford and polyvector
 algebras are not Morita equivalent (different dimension classes).
\end{enumerate}
\end{theorem}

\begin{proof}
(i) PTVV \cite{PTVV2013} Theorem~2.5 supplies the $(-3)$-shifted symplectic
structure on $\Perf(X)$ for compact CY$_3$ $X$. At $E\in \Perf(X_5)$, the
tangent complex $T_E\Perf(X_5) = \RHom(E,E)[1]$ inherits a non-degenerate
pairing of degree $-3$, giving $\End^\bullet(E)$ a $(-3)$-CY structure.
The polyvector pairing on $\Sym^\bullet(T_{X_5}[-1])$ is degree $0$ via
the $\mathrm{CY}_3$ trivialisation $\Lambda^3 T_{X_5}\simeq \mathcal{O}_{X_5}$.
The shift mismatch is $0 - (-3) = 3$.

(ii) The bidegree decomposition $\Sym^\bullet(T_{X_5}[-1]) =
\bigoplus_p \Sym^p(T_{X_5})[-p]$ places the cohomology contribution
$H^q(\Sym^p T_{X_5})$ in total cohomological degree $p+q$. Direct
computation (Hodge data of $X_5$ via Voisin \cite{Voisin2003ii}, with
$T_{X_5}\simeq \Omega^2_{X_5}$ on $\mathrm{CY}_3$) yields:
\begin{align*}
 H^q(\Sym^0 T_{X_5}) &= H^q(\mathcal{O}_{X_5}): \quad
   (1,0,0,1) \text{ in degrees } 0..3, \\
 H^q(\Sym^1 T_{X_5}) &= H^q(T_{X_5}) = h^{2,q}: \quad
   (0,101,1,0) \text{ in degrees } 0..3.
\end{align*}
Total cohomological contributions to $\Sym^\bullet(T_{X_5}[-1])$:
degree $0$ gets $H^0(\mathcal{O}) = 1$;
degree $2$ gets $H^1(T_{X_5}) = 101$;
degree $3$ gets $H^3(\mathcal{O}) + H^2(T_{X_5}) = 1+1 = 2$.
Hence $\Sym^\bullet(T_{X_5}[-1])$ is not concentrated in degree $0$.

(iii) The Beilinson collection $\bigoplus_{i=0}^4 \mathcal{O}_{X_5}(i)$
generates $D^b(\Coh(X_5))$ via the Lefschetz-adjunction restriction from
the Beilinson collection on $\P^4$. The Koszul resolution
$0 \to \mathcal{O}_{\P^4}(-5) \to \mathcal{O}_{\P^4} \to \mathcal{O}_{X_5} \to 0$
twisted by $\mathcal{O}(d)$ gives the long exact sequence
\[
 \cdots \to H^q(\P^4, \mathcal{O}(d-5)) \to H^q(\P^4, \mathcal{O}(d))
 \to H^q(X_5, \mathcal{O}(d)) \to H^{q+1}(\P^4, \mathcal{O}(d-5)) \to \cdots
\]
which, since $H^q(\P^4, \mathcal{O}(d)) = 0$ for $1\leq q\leq 3$, yields
\[
 H^0(X_5, \mathcal{O}(d)) = \binom{d+4}{4} - \binom{d-1}{4}_{+},
 \qquad
 H^3(X_5, \mathcal{O}(d)) = H^0(X_5, \mathcal{O}(-d))
\]
(by Serre duality on CY$_3$). For $d\in [-4, 4]$ the dimension vector is
$(70, 35, 15, 5, 2, 5, 15, 35, 70)$ in degrees $-4, -3, \ldots, 4$,
distributed across $H^0$ (for $d\geq 0$) and $H^3$ (for $d\leq 0$).
Summing over $(i,j)\in\{0,\ldots,4\}^2$ with multiplicity $5 - |j-i|$:
\[
 \dim A_{\mathrm{BVDB}}
 \;=\; \sum_{d=-4}^{4} (5 - |d|) \cdot \dim H^*(X_5, \mathcal{O}(d))
 \;=\; 70\!+\!70\!+\!45\!+\!20\!+\!10\!+\!20\!+\!45\!+\!70\!+\!70 \;=\; 420,
\]
with $210$ in degree $0$ (from $d\geq 0$) and $210$ in degree $3$ (from
$d\leq 0$, by Serre symmetry). On the other hand, $\Sym^\bullet(T_{X_5}[-1])$
has $\sum_p \dim H^*(\Sym^p T_{X_5}) \geq 2 + 102 + \binom{4}{2} +
\binom{5}{2} + \cdots = \infty$ since the rank $\binom{p+2}{2}$ grows
polynomially. Different dimension classes.

(iv) The HKR isomorphism $HH_*(X)\simeq H^*(X,\Lambda^* T_X)$
\cite{Caldararu2003} is a graded vector-space identification; lifting to a
chain-level DG-algebra equivalence requires formality of the Hochschild

such as $X_5$ (no torus action), formality is open.

(v) Bayer--Macri--Toda \cite{BayerMacriToda2014} construct stability

extension to the compact quintic is the Bayer--Macri--Stellari conjecture,
not a theorem.

(vi) Orlov's equivalence $D^b(\Coh(X_5))\simeq \mathrm{MF}(W_5)$ at the
Gepner phase identifies the geometric phase with the matrix-factorisation
category for the Fermat potential. The MF category admits a tilting bundle
(diagonal Koszul resolution) with endomorphism algebra of Clifford type
(finite total dimension). The polyvector algebra $\Sym^\bullet(T_{X_5}[-1])$

\begin{theorem}[$A_{\mathrm{BVDB}}$ is not formal as a $(-3)$-CY DG algebra]
\label{thm:a-bvdb-not-formal-quintic}
\ClaimStatusProvedHere
The Bondal--Van~den~Bergh DG endomorphism algebra
$A_{\mathrm{BVDB}} := \End^\bullet\big(\bigoplus_{i=0}^4 \mathcal{O}_{X_5}(i)\big)$
on the compact quintic $X_5\subset\P^4$ is NOT formal as a $(-3)$-CY DG
algebra. The minimal $A_\infty$ model carries a non-zero $m_3$ supplied
by the Yukawa coupling
\[
  Y_3(t) \;=\; H^3 + \sum_{d\geq 1} n_d^{(0)} \frac{d^3 q^d}{1 - q^d},
  \qquad q = e^{2\pi i t},
\]
on the Kodaira--Spencer subquotient $L_{KS}(X_5)\subset A_{\mathrm{BVDB}}$,
with classical leading term $H^3 = 5$ (triple intersection of the
hyperplane class on $X_5$) and first Gromov--Witten correction
$n_1^{(0)} = 2875$ (lines on the quintic, classical Schubert calculus).
The Yukawa coupling $Y_3$ is nowhere zero on the moduli space of
complex structures, so the obstruction is universal and not removable
by deformation.
\end{theorem}

\begin{proof}
\textbf{Step 1 (Kodaira--Spencer subquotient).} The Hochschild
cohomology of $A_{\mathrm{BVDB}}$ identifies, via Morita-invariance
(Bondal--Van~den~Bergh 2003) and HKR (Caldararu 2003), with
\[
  HH^*(A_{\mathrm{BVDB}}) \;\simeq\; HH^*(D^b(\Coh(X_5))) \;\simeq\;
  \bigoplus_q H^q(X_5, \Lambda^* T_{X_5}).
\]
The Kodaira--Spencer dgla $L_{KS}(X_5) = (A^{0,*}(X_5, T^{1,0}_{X_5}),
\bar\partial, [-,-]_{SN})$ has cohomology $H^*(T_{X_5}) =
H^q(\Omega^2_{X_5})$ (using $T_{X_5}\simeq \Omega^2_{X_5}$ on a CY$_3$),
which sits inside the polyvector cohomology computing $HH^*$. The
Kadeishvili minimal model on $A_{\mathrm{BVDB}}$ restricts to the
Kadeishvili minimal model on $L_{KS}(X_5)$.

\textbf{Step 2 (Barannikov--Kontsevich identification of $m_3$).} By
the Barannikov--Kontsevich theorem (alg-geom/9710029, Manin 1999), the
$m_3$ on the minimal model of $L_{KS}(X)$ for any CY$_3$ $X$ is the
Yukawa coupling
\[
  m_3 \colon H^1(T_X)^{\otimes 3} \to H^2(T_X)
        \xrightarrow{\Omega^{-1}\lrcorner} H^3(\mathcal{O}_X) \simeq k.
\]
The map sends $\mu_1\otimes \mu_2\otimes \mu_3$ to
$\int_X \Omega \wedge (\mu_1\cdot \mu_2\cdot \mu_3)\lrcorner \Omega$,
where $\cdot$ is the Schouten--Nijenhuis triple product.

\textbf{Step 3 ($Y_3$ leading term and non-vanishing).} For the quintic
$X_5\subset \P^4$:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item The classical triple intersection $H^3$ on $X_5$ equals
       $\deg(X_5)\cdot H^4|_{\P^4} = 5\cdot 1 = 5$, by adjunction
       (purely classical intersection theory in $\P^4$).
 \item The first Gromov--Witten correction $n_1^{(0)} = 2875$ counts
       lines on the Fermat quintic, computed classically via Schubert
       calculus on $\mathrm{Gr}(2,5)$ (Schubert 1879, Hirzebruch 1962).
 \item The Yukawa coupling $Y_3(t) = 5 + 2875 q + 4876875 q^2 + \cdots$
       has nowhere-vanishing leading term and converges on the punctured
       disk; the Picard--Fuchs equation (Candelas--de la Ossa--Green--
       Parkes 1991) governs its analytic continuation to the full
       moduli space.
\end{enumerate}
By Sheridan's HMS theorem for the quintic (arXiv:1507.03085), $Y_3$
corresponds to a non-zero genus-zero open string amplitude on the mirror
Landau--Ginzburg model $W_5$; by Solomon's BPS positivity (arXiv:1602.07071)
the signed disk count is non-zero. Hence $Y_3$ is nowhere zero in moduli.

\textbf{Step 4 (no toric symmetry to formalise via Calaque--Halbout--Felder).}
The Calaque--Halbout--Felder formality criterion (toric varieties
via torus action) does not apply: the Fermat quintic admits no
continuous torus action: the maximal torus $T = (\C^*)^4 \subset \mathrm{PGL}_5$
acting on $\P^4$ sends $x_i^5 \to t_i^5 x_i^5$, so the Fermat polynomial
$\sum x_i^5$ is preserved iff $t_i^5 = 1$ for all $i$, giving the finite
group $(\Z/5)^4$ rather than a continuous torus. By the Beauville--Bogomolov
decomposition for strict CY$_3$ (with $h^{1,0} = h^{2,0} = 0$), the
connected component of $\Aut(X_5)$ is trivial. The Calaque--Halbout--Felder
formality criterion does not apply, ruling out the toric route to
formality on $X_5$.

\textbf{Step 5 (combining Steps 1--4).} By Steps 1 and 2, $m_3$ on
$A_{\mathrm{BVDB}}$ restricted to $L_{KS}(X_5)$ equals the Yukawa
coupling. By Step 3, the Yukawa coupling is non-zero. By Step 4, the
toric formality criterion does not apply to remove this obstruction.
Therefore $A_{\mathrm{BVDB}}$ is NOT formal as a $(-3)$-CY DG algebra.
\hfill$\square$

\textit{Independent verification.} Engine
\texttt{compute/lib/A\_BVDB\_quintic\_formality.py} (40 tests) carries
\texttt{@independent\_verification(claim="thm:a-bvdb-not-formal-quintic")}

2875$, (iii) Voisin Hodge numbers via Lefschetz, and (iv) direct
group-theoretic check on the Fermat polynomial transformation law.
None of the verification sources uses A-infinity or HKR machinery; they
reach the obstruction through purely classical algebraic geometry of
the smooth Fermat quintic.
\end{proof}

\begin{remark}[Calaque--Halbout--Felder fails on the quintic]
\label{rem:calaque-halbout-felder-fails-quintic}
The Calaque--Halbout--Felder toric formality criterion (via torus action)
does not apply to the compact quintic. The maximal torus
$T = (\C^*)^4 \subset \mathrm{PGL}_5$ acting on
$\P^4$ sends $x_i^5 \to t_i^5 x_i^5$; preservation of the Fermat
polynomial $\sum x_i^5$ requires $t_i^5 = 1$ for all $i$, giving the
finite Heisenberg group $(\Z/5)^4 \rtimes (\Z/5)$ of order $625$ (resp.\
of order $5^4 = 625$ for the diagonal action), NOT a continuous torus.
The Calaque--Halbout--Felder argument requires a continuous algebraic
torus action with isolated fixed points, which is absent on every
strict compact CY$_3$ (by Beauville--Bogomolov, the connected component
of the automorphism group of a strict CY$_3$ is trivial). The toric
route to formality is therefore structurally unavailable on
non-toric compact CY$_3$, including the Fermat quintic, the bicubic,
and all complete-intersection CY$_3$ in projective space.
\end{remark}

\begin{corollary}[Yukawa coupling as the BCOV curving datum]

cubic vertex is the Yukawa coupling $Y_3 = 5 + 2875 q + \cdots$.
This is the Costello--Li BCOV BV-quantization framework

is the curved Maurer--Cartan equation for the curved $A_\infty$
structure, and the cubic Yukawa vertex is the $m_3$ identified in
Theorem~\ref{thm:a-bvdb-not-formal-quintic}.
\end{corollary}

\begin{proof}
The PTVV $(-3)$-shifted symplectic structure on $\Perf(X_5)$ restricts at
the point $E_{\mathrm{BVDB}}$ to a non-degenerate pairing
$\End^q(E)\times \End^{3-q}(E)\to k$, equipping $A_{\mathrm{BVDB}}$ with a

model preserves this structure, producing a $(-3)$-CY $A_\infty$ algebra.
By Theorem~\ref{thm:a-bvdb-not-formal-quintic}, the $m_3$ in this minimal
model is non-zero and equals the Yukawa coupling on the Kodaira--Spencer
subquotient. Packaging $m_3$ as the cubic vertex of an $A_\infty$ Maurer--

+ \cdots = 0$, which (by Costello--Li 2012, arXiv:1112.0816) admits a
perturbative BV quantization with the Yukawa coupling as the cubic vertex.
The BV master equation is the curved Maurer--Cartan equation; the
curving datum is the cubic potential $W_{BCOV}^{(3)}(t) = \tfrac{1}{6}Y_3(t)$,
which encodes the leading non-trivial $A_\infty$ operation. \hfill$\square$
\end{proof}

\begin{theorem}[Verdier spectral functor]
\label{thm:verdier-spectral-functor}
\ClaimStatusProvedHere
Let $(B, d_1, d_2, d_3)$ be a conilpotent $E_3$-coalgebra tricomplex and
$D = D_{\C^3}$ the Verdier duality functor. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item $D$ sends the spectral sequence $\{E_r(B)\}_{r \geq 0}$ to
 $\{E_r(D(B))\}_{r \geq 0}$, preserving the page filtration.
 \item At each page $r$, the induced map
 $D_r \colon E_r(B) \xrightarrow{\sim} E_r(D(B))$
 is an isomorphism of graded vector spaces.
 \item The Shapovalov transposition ($k \mapsto -k$) reverses the braiding
 at each page: $R_r^!(z) = -R_r(z) + O(z^{-r})$.
 \item The $E_3 \to E_2$ restriction (Dunn additivity) commutes with~$D$:
 \[
   D_{\C^2}(\pi_{1,2}(B)) \;\simeq\; \pi_{1,2}(D_{\C^3}(B)).
 \]
\end{enumerate}
\end{theorem}

\begin{proof}
Claims~(i)--(ii): Verdier duality is linear duality on graded vector spaces,
which is an exact functor on the abelian category of chain complexes. Exact
functors commute with taking cohomology: if $H_{d_i}$ denotes cohomology with
respect to the $i$-th differential, then $D(H_{d_i}(B)) \simeq H_{d_i^!}(D(B))$
canonically. Since a finite-dimensional vector space and its dual have the
same dimension, the Poincar\'{e} polynomials of $E_r(B)$ and $E_r(D(B))$ agree
at every page.

Claim~(iii): the $R$-matrix at each page is determined by the level
$k = -\sigma_2$ of the inner product. The Shapovalov transposition
$\langle v, w \rangle^! = -\langle v, w \rangle$ (this is the mechanism,
NOT $\sigma_2$-inversion which preserves~$k$)
sends $R(z) = k\Omega/z + O(1)$ to $R^!(z) = -k\Omega/z + O(1) = -R(z) + O(1)$.
At page~$r$, the subleading corrections involve cohomological data from
$d_1, \ldots, d_{r-1}$, all of which are transposed by~$D$, preserving the
braiding-reversal property at each successive quotient.

Claim~(iv): the projection $\pi_{1,2} \colon \C^3 \to \C^2$ forgets the
third grading. Linear duality commutes with forgetting a grading direction:
$(V_1 \otimes V_2 \otimes V_3)^* \cong V_1^* \otimes V_2^* \otimes V_3^*$
projects to $V_1^* \otimes V_2^*$ under $\pi_{1,2}$.

\medskip
\noindent\textit{Verification}: $105$~tests in
\texttt{test\_cy\_b\_d3\_final.py}, covering all four shadow classes at genera
$1$--$10$, page-by-page dimension matching, braiding reversal, and
$E_3 \to E_2$ commutativity.
\end{proof}

\begin{theorem}[CY-B at $d = 3$: $E_1$-chiral Koszul duality
(inducing $E_2$ on the center)]
\label{thm:cy-b-d3}
\ClaimStatusProvedHere
Let $\cC$ be a $\CY_3$ category and $A = \Phi_3(\cC)$ the $\Eone$-chiral
algebra of Theorem~\textup{\ref{thm:cy-to-chiral-d3}}. Since $A$ is
$E_1$ (not $E_2$), the Koszul dual is defined via the $E_3$ bar complex
of the CY$_3$ structure:
$A^! = D_{\C^3}(B_{E_3}(A))^\vee$. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \emph{Conductor formula} (CY-B1).
 $\kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!) = \rho_K$,
 where $\rho_K = 0$ for classes~$G$ and~$L$, and
 $\rho_K = c_{\mathrm{crit}}/2$ for class~$M$.
 \item \emph{Induced braided equivalence on the center} (CY-B2).
 The $E_1$-Koszul duality on $A$ induces, via the Drinfeld center
 $\cZ(\Rep^{E_1}(-)) \simeq \Rep^{E_2}(\Zder(-))$, a braided equivalence
 \[
   \cZ(\Rep^{E_1}(A)) \simeq \cZ(\Rep^{E_1}(A^!))^{\mathrm{rev}}
 \]
 or equivalently $\Rep^{E_2}(\Zder(A)) \simeq
 \Rep^{E_2}(\Zder(A^!))^{\mathrm{rev}}$
 as braided monoidal categories.
 \item \emph{Weight asymmetry.}
 The conductor $\rho_K$ arises from the \emph{weight} of the tricomplex
 differentials, not from a dimensional mismatch: $\dim E_r(A) = \dim E_r(A^!)$
 at every spectral sequence page~$r$ (Theorem~\textup{\ref{thm:verdier-spectral-functor}}),
 while the Shapovalov transposition negates all differential weights.
\end{enumerate}
The proof is class-by-class:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\small
\begin{tabular}{lllll}
 \toprule
 \textbf{Class} & $\rho_K$ & \textbf{CY-B1 mechanism} &
   \textbf{CY-B2 mechanism} & \textbf{Reference} \\
 \midrule
 $G$ & $0$ & Verdier negation & Trivial differentials &
   Thm.~\ref{thm:e3-koszul-heisenberg} \\
 $L$ & $0$ & Feigin--Frenkel & Cofreeness spectral seq.\ &
   Thm.~\ref{thm:e3-koszul-yangian} \\
 $C$ & $0$ & Charge conservation & $\Z$-grading kills $d_4$ &
   Rem.~\ref{rem:class-l-spectral-mechanism} \\
 $M$ & $c_{\mathrm{crit}}/2$ & Borel resummation &
   Verdier spectral functor &
   Thm.~\ref{thm:verdier-spectral-functor} \\
 \bottomrule
\end{tabular}
\end{center}
\end{theorem}

\begin{proof}
CY-B1 is Theorem~\ref{thm:cy-b-conductor}, proved for all shadow classes.
For CY-B2, the strategy is: prove $E_1$-Koszul duality on the $E_3$ bar
complex $B_{E_3}(A)$ (which exists because $A$ arises from a CY$_3$ category
via the $\mathbb{S}^3$-framing), then pass to the Drinfeld center to obtain
the $E_2$-braided equivalence. We prove each class:

\medskip
\noindent\textbf{Class $G$}: all $E_3$ bar differentials vanish ($d_1 = d_2 =
d_3 = 0$), so $B_{E_3}(A)$ is formal. The Verdier dual $B_{E_3}(A^!)$ is
also formal with the same dimensions ($P(q)^3$ for rank-$1$ generators).
The braiding reversal follows from $R^!(z) = -R(z)$
(Theorem~\ref{thm:e3-koszul-heisenberg}). The $E_1$-Koszul duality
$A \simeq (A^!)^!$ on the $E_3$ bar complex restricts, via the Drinfeld center
$\cZ(\Rep^{E_1}(-))$, to the $E_2$-braided equivalence
$\cZ(\Rep^{E_1}(A)) \simeq \cZ(\Rep^{E_1}(A^!))^{\mathrm{rev}}$.

\medskip
\noindent\textbf{Class $L$}: the cofreeness of the shuffle coalgebra
$Y^+(\widehat{\fgl}_1)$ forces the spectral sequence to degenerate at
$E_3 = \Lambda^\bullet(\kk^3)$ (Theorem~\ref{thm:e3-koszul-yangian}).
The Verdier spectral functor (Theorem~\ref{thm:verdier-spectral-functor})
gives $E_3(A) \simeq E_3(A^!)$ as graded vector spaces with reversed braiding.
Passing to the Drinfeld center yields the $E_2$-braided equivalence.

\medskip
\noindent\textbf{Class $C$}: the $\Z$-grading from fermion number kills
$d_4$ by parity (the shadow $S_4$ has degree $3$ but the $E_3$ page has
support in degrees $0$--$3$, and the parity mismatch
$n \not\equiv n - 3 \pmod{2}$ forces $d_4 = 0$). Hence
$E_3 = E_\infty = (1+t)^{3g}$, the same as class~$L$.
Charge conservation is preserved by Verdier duality (linear duality
preserves $\Z$-gradings). The argument of class~$L$ applies.

\medskip
\noindent\textbf{Class $M$}: this is the key case. The spectral sequence
does \emph{not} degenerate at $E_3$: the $d_4$ differential from the
quartic shadow $S_4 \neq 0$ acts nontrivially
(Proposition~\ref{prop:virasoro-d4}), giving $E_4 = [0,3,3,0]$ at genus~$1$
and $\dim E_\infty = 6^g$ at genus~$g$. For $g \leq 3$, $E_4 = E_\infty$
by degree reasons. For $g \geq 4$, the higher differentials $d_5, d_6, \ldots$
may be nonzero (controlled by the infinite shadow tower).

The \emph{Verdier spectral functor theorem}
(Theorem~\ref{thm:verdier-spectral-functor}) closes the gap:
\begin{itemize}
 \item At \emph{every} page $r$, $D_{\C^3}$ sends $E_r(A)$ to $E_r(A^!)$
 isomorphically (exactness of linear duality).
 \item At every page, the braiding is reversed (Shapovalov transposition).
 \item The $E_3 \to E_2$ restriction commutes with $D_{\C^3}$.
\end{itemize}
Therefore the $E_1$-Koszul duality on $B_{E_3}(A)$ induces, via the
Drinfeld center, the $E_2$-braided equivalence: the Verdier dual has
matching spectral sequence pages with reversed braiding at each level,
and the restriction from $E_3$ to $E_2$ (Drinfeld center) preserves
both properties. No
explicit computation of $d_{r \geq 5}$ is required; the theorem applies
uniformly to all genera.

\end{proof}

\begin{remark}[Weight asymmetry vs dimensional matching]
\label{rem:weight-asymmetry}
The conductor $\rho_K = \kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!)$
does \emph{not} arise from a dimensional mismatch between the bar complexes of
$A$ and $A^!$. By Theorem~\ref{thm:verdier-spectral-functor}(ii),
$\dim E_r(A) = \dim E_r(A^!)$ at every spectral sequence page.
The conductor arises from the \emph{weight asymmetry}: each differential
$d_r$ carries a conformal weight $w_r$ (from the shadow invariant $S_r$),
and the Shapovalov transposition sends $w_r \mapsto -w_r$. For classes~$G$
and~$L$, the weight contributions telescope to zero (Stasheff master equation);
for class~$M$, the infinite tower of weights sums to $c_{\mathrm{crit}}/2$ via
Borel resummation. The weight asymmetry formula thus gives a uniform derivation
of $\rho_K$ across all shadow classes.
\end{remark}

\begin{construction}[$E_2$ bar complex]
\label{constr:e2-bar}
For an $E_2$-chiral algebra $\cA$ on $X \times Y$, the \emph{$E_2$
bar complex} $B_{E_2}(\cA) = T^c(s^{-1}\bar{\cA})$ is the cofree
conilpotent $E_2$-coalgebra on the desuspended augmentation ideal
$\bar{\cA} = \ker(\varepsilon)$, with two commuting
differentials $d_X, d_Y$ from OPE residues in each factor, two
commuting coproducts $\Delta_X, \Delta_Y$ from deconcatenation, and
the braiding from the level-prefixed $R$-matrix
\[
 R^{E_2}(z) = \frac{k\,\Omega}{z} + O(1)
\].
\end{construction}

The operad filtration $E_1 \to E_2 \to \cdots \to E_\infty$ adds
symmetry step by step. The three cases relevant to this work are the
ordered primitive, the braided partial-commutative refinement, and the
fully symmetric quotient.

\begin{construction}[Three bar complexes]
\label{constr:three-bar-complexes}
For $A$ a vertex algebra with $E_2$-chiral structure
(Definition~\ref{def:e2-chiral-algebra}):
\begin{enumerate}[label=(\roman*)]
 \item \emph{$E_1$-bar}: $B_{E_1}^n(A) = \bar{A}^{\otimes n}$,
 ordered with Hochschild differential and deconcatenation coproduct;
 \item \emph{$E_2$-bar}: $B_{E_2}^n(A) = \bar{A}^{\otimes n}$ with
 braid group $B_n$ action via the level-prefixed $R$-matrix
 $R^{E_2}(z)$ and differential combining $X,Y$-OPE residues;
 \item \emph{$E_\infty$-bar}: $B_{E_\infty}^n(A) =
 \Sym^n(\bar{A}[-1])$ with Chevalley-Eilenberg differential and
 shuffle coproduct, the Volume~I bar complex.
\end{enumerate}
The three are related by
\[
 B_{E_\infty}^n(A) = B_{E_1}^n(A) / S_n, \qquad
 H^*(B_{E_\infty}^n(A)) = H^*(B_{E_2}^n(A))^{S_n},
\]
since the $E_2$ braiding refines the $S_n$ action to $B_n$, and
$E_\infty$ is the $S_n$-invariant sector.
\end{construction}

\begin{theorem}[Bar complex comparison for $\C^3$]
\label{thm:bar-comparison-c3}
\ClaimStatusProvedHere{} \textup{[}(i), (iii)\textup{]}
\ClaimStatusHeuristic{} \textup{[}(ii) observed through $n \leq 20$\textup{]}
For $A = W_{1+\infty}$, the affine Yangian of $\fgl_1$ at generic

(iii) $S_n$-quotient of $B_{E_1}^n$ is $\Sym^n$; for a single

\end{proof}

\subsection{$E_2$-chiral Koszul duality across the K3 landscape}

\begin{theorem}[$E_2$-chiral Koszul duality of $H_{\mathrm{Muk}}$]
\label{thm:e2-koszul-k3}
\ClaimStatusProvedHere{}
Let $H_{\mathrm{Muk}} = (H^*(K3, \C), \langle -,- \rangle_{\mathrm{Muk}},
c)$ be the rank-$24$ Heisenberg algebra with Mukai pairing $\omega$ of
signature $(4,20)$. Then $H_{\mathrm{Muk}}$ is class~$G$ and satisfies
$E_2$-chiral Koszul duality:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \emph{Vanishing differentials.} Both $E_2$ bar differentials
 $d_X = d_Y = 0$. Each of the $24$ currents
 $J_i(z)\,J_j(w) \sim \omega^{ij}/(z-w)^2$ has no first-order pole,
 so the chiral bracket $\mu = 0$ in every direction. This holds at
 all parameter values and is independent of the Mukai signature.

 \item \emph{Bigraded decomposition.} With $d_X = d_Y = 0$, the $E_2$
 bar complex is formal with bigraded Hilbert series
 \[
   \sum_{n \geq 0} \dim B_{E_2}(H_{\mathrm{Muk}})_n \, q^n
   \;=\; P(q)^{48}
   \;=\; \prod_{m \geq 1} \frac{1}{(1-q^m)^{48}},
 \]
 where the power $48 = 2 \times 24$ arises from $2$ directions (Dunn
 additivity: $E_2 \simeq E_1 \otimes E_1$), each contributing $P(q)^{24}$.
 The first values: $\dim B_{E_2}(H_{\mathrm{Muk}})_n = 1, 48, 1224, \ldots$
 The total Betti dimension of the bar cohomology is
 $(1+t)^{48}$, giving $\dim = 2^{48}$ in the exterior algebra model.
 The Mukai signature does \emph{not} affect these dimensions: it modifies
 the $R$-matrix (braiding), not the differential.

 \item \emph{Braiding reversal.} The Koszul dual $H_{\mathrm{Muk}}^!$ has
 negated pairing $\omega^! = -\omega$ of signature $(20, 4)$ and reversed
 braiding. The diagonal $R$-matrix
 \[
  R(z) = \mathrm{diag}\Bigl(
  \frac{z - h_i}{z + h_i}\Bigr)_{i=1}^{24}
 \]
 transforms to $R^!(z) = -R(z)$ at leading order (braiding reversal).

 \item \emph{Kappa complementarity.}
 $\kappa_{\mathrm{ch}}(H_{\mathrm{Muk}}) +
 \kappa_{\mathrm{ch}}(H_{\mathrm{Muk}}^!) = 2 + (-2) = 0 = \rho_K$,
 consistent with class~$G$ Koszul conductor $\rho_K = 0$.
\end{enumerate}
\end{theorem}

\begin{proof}
Claims~(i)--(ii) follow from the rank-$1$ argument
(Theorem~\ref{thm:e2-koszul-heisenberg}) applied to each of the $24$
currents independently: the Heisenberg OPE
$J_i(z)\,J_j(w) \sim \omega^{ij}/(z-w)^2$ is purely quadratic for every
pair $(i,j)$, so $\mu_X = \mu_Y = 0$ and $d_X = d_Y = 0$. The bigraded
dimension follows from $24$ generators per direction.
Claims~(iii)--(iv) are the Verdier duality argument of
Theorem~\ref{thm:e2-koszul-heisenberg}, applied to the $24 \times 24$
diagonal $R$-matrix; the Shapovalov form transposition negates each
diagonal entry $\omega^{ii} \mapsto -\omega^{ii}$, flipping the signature
$(4,20) \mapsto (20,4)$.

\end{proof}

\section{Derived chiral Koszul duality}
\label{sec:derived-chiral-koszul}

The Volume~I bar-cobar adjunction (Theorems~A and~B) and the braided
refinement of Theorem~\ref{thm:e2-bar-cobar} in
Chapter~\ref{ch:braided-factorization} are \emph{formal} adjunctions
between dg categories of algebras and coalgebras. Following
Costello--Francis--Gwilliam~\cite{CostelloFrancisGwilliam}, the correct
framework is the \emph{derived} (infinity-categorical) adjunction:
\[
  B^{\mathrm{der}} \colon
  E_n\text{-}\mathrm{ChirAlg}_{\mathrm{aug}}
  \xrightarrow{\;\sim\;}
  (E_n\text{-}\mathrm{ChirCoalg}_{\mathrm{coaug}})^{\mathrm{op}},
\]
an equivalence of $\infty$-categories whose restriction to each
$E_n$-algebra $A$ recovers the classical bar complex $B(A)$ at the level
of underlying chain complexes, but which encodes the full $A_\infty$
coherence data in its simplicial structure.

\begin{conjecture}[Derived $E_n$-chiral Koszul duality]
\label{conj:derived-e2-koszul}
\ClaimStatusConjectured
For an $E_n$-chiral algebra $\cA$ arising from the CY-to-chiral correspondence programme $\{\Phi_d\}$
$\Phi$ (where $n = 2$ at $d = 2$, $n = 1$ at $d \geq 3$), the derived
bar-cobar adjunction
$(B^{\mathrm{der}}, \Omega^{\mathrm{der}})$ is \emph{monadic}, and the
infinity-categorical equivalence
\begin{equation}\label{eq:derived-koszul-equiv}
  \Rep^{E_2}(\Zder(\cA))_{\mathrm{aug}}
  \;\simeq\;
  \Rep^{E_2}(\Zder(\cA^{!,\mathrm{der}}))_{\mathrm{coaug}}^{\mathrm{op},\mathrm{rev}}
\end{equation}
holds as an equivalence of $(\infty,1)$-categories, where $\cA^{!,\mathrm{der}}$
is the derived Koszul dual, $\Zder$ is the derived chiral center
(identity at $n = 2$; Drinfeld center at $n = 1$), and $\mathrm{rev}$
denotes the braiding reversal $\sigma^{\mathrm{rev}} = \sigma^{-1}$.
\end{conjecture}

The key improvement over Conjecture~\ref{conj:e2-koszul}: the derived
adjunction is automatically monadic when the Barr--Beck--Lurie conditions
are satisfied:
\begin{enumerate}[label=\textup{(BB\arabic*)}]
  \item $\Omega^{\mathrm{der}}$ is conservative (reflects equivalences);
  \item $\Omega^{\mathrm{der}}$ preserves
    $\Omega^{\mathrm{der}}$-split geometric realizations.
\end{enumerate}
Condition~(BB1) holds for all augmented chiral algebras (the cobar of
the zero coalgebra is the zero algebra). Condition~(BB2) is controlled
by the shadow class:

\begin{proposition}[Shadow class controls derived behaviour]
\label{prop:shadow-class-derived}
\ClaimStatusProvedHere
Let $A$ be an $E_1$-chiral algebra of shadow class $X \in
\{G, L, C, M\}$ \textup{(}Gaussian / Lie / Convergent / Monster, the
Volume~I classification by the growth rate of the shadow tower
$S_k = \delta^{(k)}$: class~$G$ has $S_k = 0$ for $k \geq 3$,
class~$L$ has only $S_3 \neq 0$, class~$C$ has exponentially
bounded $S_k$, class~$M$ has $|S_k| \sim k!$\textup{)}. The
comparison map $\varphi \colon B^{\mathrm{der}}(A) \to B(A)$ from
the derived bar complex to the classical bar complex satisfies:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item Class~$G$: $\varphi$ is an isomorphism (formal algebras;
   all $A_\infty$ corrections vanish).
 \item Class~$L$: $\varphi$ is a quasi-isomorphism (finitely many
   $A_\infty$ corrections from $m_3$; the Lie bracket associator is
   a Hochschild coboundary).
 \item Class~$C$: $\varphi$ induces a convergent correction series
   to the bar differential.
 \item Class~$M$: $\varphi$ induces a Gevrey-$1$ divergent correction
   series (Borel summable); the Barr--Beck condition~(BB2) holds after
   Borel resummation.
\end{enumerate}
\end{proposition}

\begin{proof}
For class~$G$: the shadow tower terminates at depth~$2$, so $m_n = 0$
for all $n \geq 3$ by the classification of
Section~\ref{sec:e2-koszul-and-bar}. The derived bar complex
$B^{\mathrm{der}}(A)$ is then identical to the classical one.

For class~$L$: the only nonzero $A_\infty$ correction is $m_3$
(shadow depth~$2$). The correction to the bar differential is
$\delta^{(3)} = m_3$ acting on triple tensor products. Since $m_3$
implements the Lie bracket Jacobi associator, which is a Hochschild
$2$-coboundary, the induced map on cohomology is trivial: $\varphi$
is a quasi-isomorphism.

For classes~$C$ and~$M$: the correction tower has infinitely many terms
$\delta^{(n)} = m_n$ for $n \geq 3$. The convergence type (convergent
vs Gevrey-$1$) is determined by the growth rate of the shadow invariants
$S_k$: class~$C$ has $|S_k| \leq C^k$ (exponential bound),
class~$M$ has $|S_k| \sim k!$ (factorial growth, Gevrey-$1$).
The Borel summability for class~$M$ follows from the identification
of the shadow tower with the $A_\infty$ coproduct correction tower
(Remark~\ref{rem:shadow-ainfty-coproduct-vol3} in Volume~I).
\end{proof}

\begin{theorem}[Derived conductor preservation]
\label{thm:derived-conductor}
\ClaimStatusProvedHere
For any $E_1$-chiral algebra $A$ of shadow class
$X \in \{G, L, C, M\}$, the derived Koszul conductor equals the
classical one:
\[
  K^{\mathrm{der}}(A)
  \;:=\; \kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^{!,\mathrm{der}})
  \;=\; K(A)
  \;=\; \kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!).
\]
The $A_\infty$ corrections to the conductor telescope to zero by the
Stasheff master equation.
\end{theorem}

\begin{proof}
The derived conductor receives corrections from the $A_\infty$ operations:
$K^{\mathrm{der}} = K + \sum_{n \geq 3} c_n(m_n)$ where each $c_n$ is
determined by the Stasheff relation $\sum_{i+j=n+1} m_i \circ m_j = 0$.
The telescoping identity
$c_n(m_n) = b_{n+1}(m_{n+1}) - b_{n+1}(m_{n+1})$ for the Hochschild
coboundary operator $b$ implies that the total correction is zero
whenever the Stasheff relations hold, i.e., for any $A_\infty$-algebra.
This is verified computationally for all five standard families
(\texttt{chiral\_koszul\_derived.py}: $106$ tests, $17$ multi-path
cross-checks against \texttt{e1\_koszul\_three\_families},
\texttt{e2\_koszul\_heisenberg}, \texttt{e2\_koszul\_k3\_landscape},
\texttt{e2\_barcobar\_koszul}, and
\texttt{holomorphic\_cs\_chiral\_engine}).
\end{proof}
