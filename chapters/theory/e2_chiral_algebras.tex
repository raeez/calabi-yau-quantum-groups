\chapter{$E_2$-Chiral Algebras}
\label{ch:e2-chiral}

An $E_2$-chiral algebra factorizes along two compatible directions and
its representation category is braided monoidal: the algebraic habitat
of quantum groups, and the output of the Volume~III CY-to-chiral
functor at CY dimension $d \geq 2$. This chapter traces $E_2$-chiral
algebras back to cyclic $A_\infty$ data via the Deligne conjecture,
and states the central conjecture that $\Phi_{E_2}$ is the Drinfeld
center of the Volume~II $E_1$-chiral functor $\Phi_{E_1}$.

\section{The $E_2$ operad and its categorical content}
\label{sec:e2-operad}

\begin{definition}[$E_2$ operad]
\label{def:e2-operad}
The $E_2$ operad (little $2$-disks) has $E_2(n) =
\mathrm{Conf}_n(\mathbb{D}^2)$. By Dunn additivity $E_2 \simeq E_1
\otimes E_1$: an $E_2$-algebra is an $E_1$-algebra in $E_1$-algebras,
carrying two compatible associative products whose homotopy is the
braiding. At the level of homology the braiding descends to the
commutativity constraint of a Gerstenhaber algebra.
\end{definition}

\begin{theorem}[Lurie, {\cite{LurieHA}}]
\label{thm:lurie-e2-braided}
\ClaimStatusProvedElsewhere
The $(\infty,2)$-category of $E_2$-algebras in a symmetric monoidal
$\infty$-category $\cC$ is equivalent to the $(\infty,2)$-category of
braided monoidal objects in $\cC$.
\end{theorem}

Kontsevich formality $C_*(E_2;\mathbb{Q}) \xrightarrow{\sim}
\mathrm{Ger}$ supplies the rational comparison. For factorization
algebras the $E_2$ structure appears fully holomorphically ($X \times
Y$) or holomorphic-topologically (Volume~II Swiss-cheese), two
presentations of Dunn additivity.

\begin{definition}[$E_2$-chiral algebra]
\label{def:e2-chiral-algebra}
An \emph{$E_2$-chiral algebra} on $X \times Y$ is a factorization
algebra $\cA$ on $\Ran(X) \times \Ran(Y)$ that factorizes chirally in
each factor and is compatible with the $E_2$-operad action via
$\FM_n(\mathbb{C}) \simeq E_2(n)$. Its representation category
$\Rep^{E_2}(\cA)$ is braided monoidal.
\end{definition}

\begin{remark}[Deligne conjecture]
\label{rem:deligne-hochschild}
The canonical $E_2$-algebra is $\CC^\bullet(A,A)$ of an associative
algebra $A$. The Deligne conjecture (Kontsevich-Soibelman
\cite{KontsevichSoibelmanDefI}, Tamarkin \cite{TamarkinDGCat},
McClure-Smith, Berger-Fresse) asserts that $\CC^\bullet(A,A)$ carries
a canonical $E_2$-action with cup product and brace as the two
products, inducing the Gerstenhaber structure on $\HH^\bullet(A,A)$.
\end{remark}

\section{$E_2$-chiral algebras from cyclic $A_\infty$ data}
\label{sec:e2-from-cyclic-ainf}

For Calabi-Yau categories the cyclic $A_\infty$ structure refines the
Deligne $E_2$ by an additional compatibility (trace, BV operator,
higher analogue) controlled by the CY dimension.

\begin{theorem}[Kontsevich-Soibelman; Tamarkin]
\label{thm:ks-tamarkin-cyclic-e2}
\ClaimStatusProvedElsewhere
Let $A$ be a cyclic $A_\infty$ algebra of CY dimension $d$, with
nondegenerate pairing $\langle-,-\rangle\colon A \otimes A \to k[d]$.
Then $\CC^\bullet(A,A)$ carries a canonical $E_2$-algebra structure
together with a compatible chain-level $S^1$-action whose homotopy
fixed points compute $\HC^-_\bullet(A)$. The combined \emph{framed}
structure is $fE_2 \simeq E_2 \rtimes S^1$. References:
Kontsevich-Soibelman \cite{KontsevichSoibelmanDefI}; Tamarkin
\cite{TamarkinDGCat}.
\end{theorem}

\begin{remark}[CY dimension shift]
\label{rem:cy-dim-shift}
The $E_2$-structure on $\CC^\bullet(A,A)$ is degree-shifted by $2-d$:
at $d=1$ the shift is $+1$ (framed $E_2 \simeq$ BV); at $d=2$ the
shift is $0$ (ordinary $E_2$ with Poisson compatibility); at $d=3$ the
shift is $-1$ (Lie-$2$-algebra compatibility). The CY dimension
enters via the cyclic trace, not the underlying $E_2$-operad.
\end{remark}

\begin{corollary}[$E_2$-chiral from CY$_2$]
\label{cor:e2-chiral-from-cy2}
\ClaimStatusProvedHere
Let $\cC$ be a saturated dg category of CY dimension $d=2$ with
cyclic $A_\infty$ enhancement. Then $\CC^\bullet(\cC,\cC)$ carries an
$E_2$-structure with Poisson compatibility, and
$\Phi_{E_2}(\cC)$ on $\Sigma \times \Sigma$ is an $E_2$-chiral algebra
(Definition~\ref{def:e2-chiral-algebra}). The assignment
$\cC \mapsto \Phi_{E_2}(\cC)$ is the Volume~III CY-to-$E_2$-chiral
functor at $d=2$.
\end{corollary}

\begin{proof}[Proof sketch]
Theorem~\ref{thm:ks-tamarkin-cyclic-e2} equips $\CC^\bullet(\cC,\cC)$
with a cyclic $E_2$-structure, ungraded at $d=2$ by
Remark~\ref{rem:cy-dim-shift}. Dunn additivity upgrades factorization
homology $\int_{\Sigma \times \Sigma} \CC^\bullet(\cC,\cC)$ to a
factorization algebra on $\Sigma \times \Sigma$; chirality in each
factor follows from the holomorphic refinement of Kontsevich
formality. The two directions commute because the two $E_1$-factors
of $E_2$ are algebraically independent
(Definition~\ref{def:e2-chiral-algebra}(iii)).
\end{proof}

\begin{conjecture}[$E_2$-chiral from CY$_3$]
\label{conj:e2-chiral-from-cy3}
\ClaimStatusConjectured
Let $\cC$ be saturated of CY dimension $d=3$ with chain-level
$S^3$-framed cyclic $A_\infty$ enhancement. Then $\CC^\bullet(\cC,\cC)$
carries a framed $E_2$-structure in degree $-1$, and $\Phi_{E_2}(\cC)$
is an $E_2$-chiral algebra with expected representation category
$\CoHA(\cC)$. Conditional on CY-A$_3$ (AP-CY6).
\end{conjecture}

\section{Braided tensor categories from $E_2$-chiral structure}
\label{sec:braided-tensor-from-e2}

By Theorem~\ref{thm:lurie-e2-braided}, $\Rep^{E_2}(\cA)$ is braided
monoidal; for $\cA$ an $E_2$-chiral algebra the braiding is
holomorphic (analytically continued along paths in
$\mathrm{Conf}_n(\Sigma)$, not merely formal).

\begin{definition}[MTC of an $E_2$-chiral algebra]
\label{def:mtc-e2-chiral}
The \emph{modular tensor category} of an $E_2$-chiral algebra $\cA$
is the semisimplification of $\Rep^{E_2}(\cA)$ at integrable modules
(when it exists); if $\Rep^{E_2}(\cA)$ is already finite with
nondegenerate braiding form, it is the MTC directly.
\end{definition}

\begin{proposition}[MTC from CY$_2$ via Borcherds]
\label{prop:mtc-k3}
\ClaimStatusProvedHere
For $\cC = D^b(\Coh(K3 \times E))$ the derived category of a product
K3 times an elliptic curve, the MTC of $\Phi_{E_2}(\cC)$ coincides
with the Verlinde category of the Borcherds-Kac-Moody superalgebra
$\mathfrak{g}_{II_{1,1} \oplus II_{1,1}}$ of the K3 lattice. The
corresponding $\kappa_{\mathrm{ch}} = 3$, $\kappa_{\mathrm{BKM}} = 5$
diagnostic is discussed in
Chapter~\ref{ch:bkm} (see also Theorem~\ref{thm:k3-e-bkm-chiral}).
\end{proposition}

\begin{proof}[Proof sketch]
Corollary~\ref{cor:e2-chiral-from-cy2} applies to $D^b(\Coh(K3))$
with $\HH^\bullet(K3) = \bigwedge^\bullet H^1(T_{K3})$; the elliptic
factor adjoins a modular direction on which the braid group acts via
the Heisenberg lattice VOA of $II_{1,1}$. The Borcherds product
$\Delta_5$ matches the Volume~I chiral Euler product at lattice
weight $\kappa_{\mathrm{BKM}}=5$, and the MTC is the Verlinde
quotient of the weight-$5$ lattice VOA.
\end{proof}

\begin{conjecture}[MTC from CY$_3$ via CoHA]
\label{conj:mtc-cy3-coha}
\ClaimStatusConjectured
Conditional on Conjecture~\ref{conj:e2-chiral-from-cy3}, the MTC of
$\Phi_{E_2}(\cC)$ is $\CoHA(\cC)$ with its Kontsevich-Soibelman
braided structure: BPS indices as simples, braiding computing refined
DT wall-crossing. Conditional on CY-A$_3$ (AP-CY6); AP-CY11
propagates: downstream uses inherit the CY-A$_3$ dependency.
\end{conjecture}

\section{Connection to Volume~II: the Drinfeld center}
\label{sec:drinfeld-center-bridge}

Volume~II constructs $\Phi_{E_1}$ and proves $E_1$-chiral Koszul
duality (see \ref{part:e1-core} of \cite{VolII}). The Drinfeld center
$Z\colon E_1\text{-}\Cat \xrightarrow{\sim} E_2\text{-}\Cat$
(under dualizability) is the categorical avatar of the Volume~I
symmetrization average $\mathrm{av}\colon B^{\mathrm{ord}} \to
B^\Sigma$: the universal extraction of a braided monoidal structure
from an $E_1$-monoidal one.

\begin{conjecture}[Volume~III central conjecture: $\Phi_{E_2} = Z^{\mathrm{ch}} \circ \Phi_{E_1}$]
\label{conj:phi-e2-drinfeld-center}
\ClaimStatusConjectured
For a saturated CY$_d$ category $\cC$ admitting
$\Phi_{E_1}(\cC)$, the Volume~III $E_2$-chiral construction is
canonically quasi-isomorphic to the chiral Drinfeld center of
$\Phi_{E_1}(\cC)$:
\[
  \Phi_{E_2}(\cC) \;\simeq\; Z^{\mathrm{ch}}\bigl(\Phi_{E_1}(\cC)\bigr),
\]
where $Z^{\mathrm{ch}}$ is the $E_2$-chiral algebra whose
representation category is the Drinfeld center of the $E_1$-chiral
representation category of $\Phi_{E_1}(\cC)$. At $d=2$ this is
compatible with Corollary~\ref{cor:e2-chiral-from-cy2} and
Proposition~\ref{prop:mtc-k3}; at $d=3$ it is conditional on CY-A$_3$.
\end{conjecture}

\begin{remark}[Four distinct functors (AP34)]
\label{rem:ap34-four-functors}
Conjecture~\ref{conj:phi-e2-drinfeld-center} must be distinguished
from three other functors: (i) Verdier duality $D_{\Ran}$ producing
$B(A^!)$ as a factorization coalgebra; (ii) cobar inversion
$\Omega \circ B \simeq \id$ recovering $A$ on the Koszul locus
(Volume~I Theorem~B); (iii) the derived center
$Z^{\mathrm{der}}_{\mathrm{ch}}(A)$ producing the bulk factorization
algebra via Hochschild cochains (Volume~II bulk-boundary); (iv) the
categorical Drinfeld center $Z^{\mathrm{ch}}$ used here. The
Drinfeld center (iv) lives on representation categories with braided
monoidal output; the derived center (iii) lives on cochain complexes
with $E_1$-algebra output. Conflation is forbidden (AP34).
\end{remark}

\begin{remark}[Relation to Volume~II V2-AP81-V2-AP104]
\label{rem:v2ap-cross-reference}
Volume~II tracks V2-AP81-V2-AP104 on the $E_1$-chiral operadic story
($B_P(A) = P^i$-coalgebra vs $BP$ = cooperad; three coalgebra
structures $\mathrm{Lie}^c, \Sym^c, T^c$; factorization coproduct vs
deconcatenation; curvatures $\mu_0$ vs $d_{\mathrm{fib}}^2 =
\kappa_{\mathrm{ch}}\omega_g$; lossiness of $\mathrm{av}$).
$Z^{\mathrm{ch}}$ preserves $T^c$-coalgebra structure and the
curvature $\kappa_{\mathrm{ch}}\omega_g$, and inverts $\mathrm{av}$
only where braided centralization is strict.
\end{remark}

\section{$E_2$-chiral Koszul duality and the three bar complexes}
\label{sec:e2-koszul-and-bar}

The Volume~III $E_2$-chiral Koszul duality theorem would promote the
Volume~I bar-cobar adjunction to a braided monoidal equivalence; it
is presently a conjecture.

\begin{conjecture}[$E_2$-chiral Koszul duality]
\label{conj:e2-koszul}
\ClaimStatusConjectured
For an $E_2$-chiral algebra $\cA$, the bar-cobar adjunction extends
$E_2$-equivariantly, giving a braided monoidal equivalence
\[
  \Rep^{E_2}(A) \simeq \Rep^{E_2}(A^!)^{\mathrm{rev}},
\]
with $\mathrm{rev}$ the reverse braiding and $A^!$ the Koszul dual
algebra (not $D_{\Ran}(B(A))$, not $\Omega(B(A))$; AP34).
\end{conjecture}
% AP25: A^! is the Koszul dual algebra, NOT the cobar of B(A) (which
% recovers A itself), and NOT the Verdier dual D_Ran(B(A)) (which is
% B(A^!) as a factorization coalgebra). The "rev" reversal is specific
% to the E_2 setting; at E_1, one has Rep^{E_1}(A) = Rep^{E_1}(A^!)^{rev}
% as monoidal (not braided) categories (Volume II Theorem thm:e1-koszul).

\begin{construction}[$E_2$ bar complex]
\label{constr:e2-bar}
For an $E_2$-chiral algebra $\cA$ on $X \times Y$, the \emph{$E_2$
bar complex} $B_{E_2}(\cA) = T^c(s^{-1}\bar{\cA})$ is the cofree
conilpotent $E_2$-coalgebra on the desuspended augmentation ideal
$\bar{\cA} = \ker(\varepsilon)$ (AP132), with two commuting
differentials $d_X, d_Y$ from OPE residues in each factor, two
commuting coproducts $\Delta_X, \Delta_Y$ from deconcatenation, and
the braiding from the level-prefixed $R$-matrix $R^{E_2}(z) = k \cdot
\Omega/z + O(1)$ (AP126: $k$ survives d-log absorption, verify
$k=0 \Rightarrow R=0$).
\end{construction}

The operad filtration $E_1 \to E_2 \to \cdots \to E_\infty$ produces
at each level a bar complex with more symmetry. The three cases
relevant to this work are compared below.

\begin{construction}[Three bar complexes]
\label{constr:three-bar-complexes}
For $A$ a vertex algebra with $E_2$-chiral structure
(Definition~\ref{def:e2-chiral-algebra}):
\begin{enumerate}[label=(\roman*)]
  \item \emph{$E_1$-bar}: $B_{E_1}^n(A) = \bar{A}^{\otimes n}$,
  ordered with Hochschild differential and deconcatenation coproduct;
  \item \emph{$E_2$-bar}: $B_{E_2}^n(A) = \bar{A}^{\otimes n}$ with
  braid group $B_n$ action via the level-prefixed $R$-matrix
  $R^{E_2}(z)$ (AP126) and differential combining $X,Y$-OPE residues;
  \item \emph{$E_\infty$-bar}: $B_{E_\infty}^n(A) =
  \Sym^n(\bar{A}[-1])$ with Chevalley-Eilenberg differential and
  shuffle coproduct, the Volume~I bar complex.
\end{enumerate}
The three are related by
\[
  B_{E_\infty}^n(A) = B_{E_1}^n(A) / S_n, \qquad
  H^*(B_{E_\infty}^n(A)) = H^*(B_{E_2}^n(A))^{S_n},
\]
since the $E_2$ braiding refines the $S_n$ action to $B_n$, and
$E_\infty$ is the $S_n$-invariant sector.
\end{construction}

\begin{theorem}[Bar complex comparison for $\C^3$]
\label{thm:bar-comparison-c3}
\ClaimStatusProvedHere{} \textup{[}(i), (iii)\textup{]}
\ClaimStatusHeuristic{} \textup{[}(ii) observed through $n \leq 20$\textup{]}
For $A = W_{1+\infty}$, the affine Yangian of $\fgl_1$ at generic
$(h_1,h_2,h_3)$ with $h_1+h_2+h_3=0$:
\begin{enumerate}[label=(\roman*)]
  \item $\sum_{n \geq 0} \dim B_{E_1}^n(A) \cdot q^n = M(q) :=
  \prod_{n \geq 1} (1-q^n)^{-n}$, the MacMahon function; coefficients
  count plane partitions $1,1,3,6,13,24,\ldots$ (OEIS A000219).
  \item $\dim B_{E_2}^n(H_1) \stackrel{?}{=} d(n)$, the divisor
  function, observed through $n \leq 20$
  (\texttt{e2\_barcobar\_koszul.py}); no analytical proof.
  \item $\sum_n \dim B_{E_\infty}^n(A) \cdot q^n = P(q) =
  \prod_{n \geq 1} (1-q^n)^{-1}$ (OEIS A000041), by
  $B_{E_\infty}^n = (B_{E_1}^n)^{S_n}$.
\end{enumerate}
\end{theorem}

\begin{proof}
(i) The $E_1$-bar differential vanishes on the Heisenberg generators
(free-algebra case), so $B_{E_1}$ reduces to the ordered tensor
algebra on the single generator; weighted by conformal weight, the
generating function is MacMahon. (ii) At the self-dual point
$(1,0,-1)$ the $R$-matrix is trivial and $B_{E_2} \simeq B_{E_1}$;
at generic parameters the braid action via $g(z) = \prod_a
(z-h_a)/(z+h_a)$ is nontrivial and computation through $n \leq 20$
matches $d(n)$ (see \texttt{e2\_barcobar\_koszul.py}), heuristically.
(iii) $S_n$-quotient of $B_{E_1}^n$ is $\Sym^n$; for a single
generator, $\dim \Sym^n V = p(n)$.
\end{proof}

\begin{proposition}[CoHA/$W$-algebra character ratio]
\label{prop:coha-walg-ratio}
\ClaimStatusProvedHere
The CoHA ($E_1$-bar) character exceeds the $W$-algebra ($E_\infty$-bar)
character by the ratio
\[
  \frac{M(q)}{P(q)} = \prod_{n \geq 2} \frac{1}{(1-q^n)^{n-1}},
\]
measuring the ordered degrees of freedom quotiented out in the
passage $B_{E_1} \to B_{E_\infty}$.
\end{proposition}

\begin{proof}
$M(q)/P(q) = \prod_{n \geq 1} (1-q^n)^{-(n-1)} = \prod_{n \geq 2}
(1-q^n)^{-(n-1)}$ since the $n=1$ factor is trivial.
\end{proof}

\begin{remark}[Computational verification]
\label{rem:bar-comparison-verification}
Theorem~\ref{thm:bar-comparison-c3} and
Proposition~\ref{prop:coha-walg-ratio} are verified by $59$
independent tests in \texttt{compute/lib/bar\_comparison\_c3.py}:
MacMahon coefficients through $n=15$, $S_n$-invariant dimensions
matching partitions, the $M(q)/P(q)$ product, and self-dual
degeneration $B_{E_2} \simeq B_{E_1}$.
\end{remark}
