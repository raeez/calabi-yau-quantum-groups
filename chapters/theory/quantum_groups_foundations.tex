\chapter{Quantum Groups: Foundations}
\label{ch:quantum-groups}

Conjecture~CY-C asks when the CY-to-chiral correspondence programme $\{\Phi_d\}$ produces a quantum group at the representation-theoretic level. To answer that question one needs the standard quantum-group package in a form compatible with the ordered bar and the Drinfeld center. The quantized enveloping algebra $U_q(\frakg)$ and its $R$-matrix are the targets of $\Phi_d$ on that side: Conjecture~CY-C predicts that the braided monoidal category $\Rep_q(\frakg)$ arises as $\Rep^{\Etwo}(A_\cC)$ for a CY category $\cC(\frakg, q)$. The Drinfeld--Jimbo presentation and the FRT presentation give two access points, corresponding respectively to the Koszul dual presentation by generators and relations and the ordered-bar RTT formalism.

Vol~III reads this material backwards. Instead of deforming $U(\frakg)$ first and discovering braiding later, it treats $U_q(\frakg)$ as an \emph{output} of the CY-to-chiral correspondence programme $\{\Phi_d\}$ applied to a CY category whose Drinfeld center recovers the modular tensor category of conformal blocks. Everything below is classical and due to Drinfeld, Jimbo, Lusztig, Reshetikhin--Turaev, and Kazhdan--Lusztig; the Vol~III content is the organization around $\{\Phi_d\}$.

\section{Quantized enveloping algebras}
\label{sec:qgf-quantized-enveloping}
% ============================================================

The Drinfeld--Jimbo presentation realises $U_q(\frakg)$ as a deformation of $U(\frakg)$ in the Hopf-algebra category: generators and Serre relations deform with $q$-corrections, and the coproduct acquires an asymmetry that at $q = 1$ reduces to the symmetric classical coproduct. The data entering the definition is the Cartan matrix and its symmetrisers; everything else is forced.

Let $\frakg$ be a finite-dimensional simple Lie algebra over $\C$ of rank $\ell$ with Cartan matrix $(a_{ij})_{i,j=1}^{\ell}$ and symmetrizers $d_1, \ldots, d_\ell \in \{1, 2, 3\}$ such that $d_i a_{ij} = d_j a_{ji}$. Let $q \in \C^*$ be a formal parameter (or complex number, not a root of unity) and write $q_i = q^{d_i}$, $[n]_q = (q^n - q^{-n})/(q - q^{-1})$, $[n]_q! = [1]_q [2]_q \cdots [n]_q$.

\begin{definition}[Drinfeld--Jimbo quantized enveloping algebra]
\label{def:qgf-quantum-group}
\label{def:qgf-drinfeld-jimbo}
The \emph{quantized enveloping algebra} $U_q(\frakg)$ is the unital associative $\C(q)$-algebra generated by $E_i, F_i, K_i^{\pm 1}$ for $i = 1, \ldots, \ell$ subject to the relations
\begin{align}
 K_i K_j &= K_j K_i, \qquad K_i K_i^{-1} = K_i^{-1} K_i = 1, \\
 K_i E_j K_i^{-1} &= q_i^{a_{ij}} E_j, \qquad K_i F_j K_i^{-1} = q_i^{-a_{ij}} F_j, \\
 [E_i, F_j] &= \delta_{ij} \, \frac{K_i - K_i^{-1}}{q_i - q_i^{-1}},
\end{align}
together with the quantum Serre relations, for $i \ne j$,
\begin{equation}
 \sum_{k=0}^{1 - a_{ij}} (-1)^k \binom{1 - a_{ij}}{k}_{q_i} E_i^{1 - a_{ij} - k} E_j E_i^k = 0,
 \label{eq:qgf-quantum-serre-E}
\end{equation}
and the analogous identity for the $F_i$, where $\binom{n}{k}_{q_i} = [n]_{q_i}!/([k]_{q_i}! [n-k]_{q_i}!)$.
\end{definition}

\begin{remark}[Classical limit]
\label{rem:qgf-classical-limit}
Writing $q = e^{\hbar/2}$ and $K_i = q_i^{H_i} = e^{d_i \hbar H_i / 2}$ and formally expanding in $\hbar$, the relation $[E_i, F_j] = \delta_{ij} (K_i - K_i^{-1})/(q_i - q_i^{-1})$ becomes $[E_i, F_j] = \delta_{ij} H_i + O(\hbar)$, and the quantum Serre relations reduce to the classical Serre relations at $\hbar = 0$. Thus $U_q(\frakg)/(q - 1) \cong U(\frakg)$ as Hopf algebras (Drinfeld 1986, Jimbo 1985).
\end{remark}

\begin{proposition}[Hopf algebra structure]
\label{prop:qgf-hopf}
\ClaimStatusProvedElsewhere
$U_q(\frakg)$ carries a Hopf algebra structure with coproduct $\Delta$, counit $\varepsilon$, and antipode $S$ given on generators by
\begin{align}
 \Delta(K_i) &= K_i \otimes K_i, \\
 \Delta(E_i) &= E_i \otimes 1 + K_i \otimes E_i, \\
 \Delta(F_i) &= F_i \otimes K_i^{-1} + 1 \otimes F_i, \\
 \varepsilon(K_i) &= 1, \qquad \varepsilon(E_i) = \varepsilon(F_i) = 0, \\
 S(K_i) &= K_i^{-1}, \qquad S(E_i) = -K_i^{-1} E_i, \qquad S(F_i) = -F_i K_i.
\end{align}
\end{proposition}

\begin{proof}[Attribution]
Drinfeld, ``Quantum groups'' (ICM Berkeley 1986); Jimbo, ``A $q$-difference analogue of $U(\frakg)$ and the Yang--Baxter equation'' (Lett.\ Math.\ Phys.\ 1985). The Hopf axioms are verified by direct computation on generators; the Serre relations are compatible with the coproduct by a standard argument using the $q$-binomial identity.
\end{proof}

The algebra $U_q(\frakg)$ admits a triangular decomposition $U_q(\frakg) \cong U_q^-(\frakg) \otimes U_q^0(\frakg) \otimes U_q^+(\frakg)$, with $U_q^{\pm}(\frakg)$ generated by the $E_i$ respectively $F_i$, and $U_q^0(\frakg) \cong \C(q)[K_1^{\pm 1}, \ldots, K_\ell^{\pm 1}]$ the Cartan subalgebra. On the Vol~III side, this decomposition matches the bar filtration of $B(A_\cC)$: weight zero is $U_q^0$, positive weights are $U_q^+$, negative weights are $U_q^-$.

\begin{remark}[Universal $R$-matrix, formal form]
\label{rem:qgf-universal-R-formal}
Drinfeld (1986) and independently Rosso, Kirillov--Reshetikhin show that $U_q(\frakg)$ is a \emph{quasi-triangular} Hopf algebra: there exists a formal element
\begin{equation}
 \cR = \sum_{\mu \in Q^+} \cR_\mu \in U_q(\frakg) \mathbin{\hat{\otimes}} U_q(\frakg),
 \label{eq:qgf-universal-R}
\end{equation}
where the sum ranges over the positive root lattice $Q^+$ and each $\cR_\mu \in U_q^+ \otimes U_q^-$ is a finite product of quantum exponentials. The element $\cR$ intertwines $\Delta$ with its opposite: $\cR \Delta(x) \cR^{-1} = \Delta^{\mathrm{op}}(x)$ for all $x \in U_q(\frakg)$. Closed expressions exist only for $\frakg = \mathfrak{sl}_2$ and low-rank cases.
\end{remark}

\section{The universal $R$-matrix and Yang--Baxter equation}
\label{sec:qgf-r-matrix-ybe}
% ============================================================

\begin{proposition}[Quantum Yang--Baxter equation]
\label{prop:qgf-qybe}
\ClaimStatusProvedElsewhere
The universal $R$-matrix $\cR \in U_q(\frakg)^{\hat{\otimes} 2}$ of Remark~\ref{rem:qgf-universal-R-formal} satisfies the quantum Yang--Baxter equation
\begin{equation}
 \cR_{12} \, \cR_{13} \, \cR_{23} = \cR_{23} \, \cR_{13} \, \cR_{12}
 \qquad \text{in } U_q(\frakg)^{\hat{\otimes} 3},
 \label{eq:qgf-qybe}
\end{equation}
where $\cR_{ij}$ denotes the embedding of $\cR$ into the $(i, j)$-tensor factors (with identity on the remaining factor).
\end{proposition}

\begin{proof}[Attribution]
Drinfeld, ``Quantum groups'' (ICM Berkeley 1986), Theorem~2; see also Chari--Pressley, \emph{A Guide to Quantum Groups}, \S4.2. The proof is a direct consequence of quasi-triangularity and the coassociativity of $\Delta$.
\end{proof}

\begin{proposition}[Classical limit and cross-volume compatibility]
\label{prop:qgf-classical-limit-r}
\ClaimStatusProvedElsewhere
Write $q = e^{\hbar/2}$ and expand the universal $R$-matrix as
\begin{equation}
 \cR = 1 + \hbar \, r + O(\hbar^2), \qquad r \in \frakg \otimes \frakg.
 \label{eq:qgf-R-expansion}
\end{equation}
Then $r$ is the \emph{classical $r$-matrix} of $\frakg$, satisfying the symmetry identity $r + r^{21} = \Omega_{\frakg}$ where $\Omega_{\frakg} \in (\frakg \otimes \frakg)^{\frakg}$ is the quadratic Casimir tensor (normalised by the invariant form). Explicitly, $r$ decomposes as $r = \tfrac{1}{2}\Omega_{\frakg} + r_{\mathrm{sk}}$ with $r_{\mathrm{sk}} = \sum_{\alpha > 0} E_\alpha \wedge F_\alpha$ the skew-symmetric Drinfeld--Sklyanin part; the skew part carries the Lie-bialgebra cobracket data and the symmetric part is forced by quasi-triangularity. The quantum Yang--Baxter equation~\eqref{eq:qgf-qybe} expands to the classical Yang--Baxter equation
\begin{equation}
 [r_{12}, r_{13}] + [r_{12}, r_{23}] + [r_{13}, r_{23}] = 0
 \label{eq:qgf-cybe}
\end{equation}
at order $\hbar^2$.
\end{proposition}

\begin{proof}[Attribution]
Drinfeld, ``Hamiltonian structures on Lie groups, Lie bialgebras and the geometric meaning of classical Yang--Baxter equations'' (Soviet Math.\ Dokl.\ 1983); Drinfeld (1986), Theorem~3.
\end{proof}

\begin{remark}[Level-stripped affine $r$-matrix and the two sanity points]
\label{rem:qgf-level-stripped-r}
Passing from $U_q(\frakg)$ (finite type) to the affine quantum group $U_q(\hat{\frakg})$ at level $k$, Proposition~\ref{prop:qgf-classical-limit-r} acquires a level prefix: the classical limit produces
\begin{equation}
 r(z) = k \cdot \frac{\Omega_{\frakg}}{z} + O(\hbar, z^0),
 \label{eq:qgf-r-level-k}
\end{equation}
matching the Vol~I and Vol~II convention. The affine formula has two distinguished values of $k$ at which the algebraic structure collapses for different reasons; both must be verified whenever an affine $r$-matrix is written.
\begin{enumerate}[label=(\alph*)]
 \item \emph{Level zero} ($k = 0$). The level prefix in~\eqref{eq:qgf-r-level-k} vanishes, so $r(z) = 0$ and $\cR(z) = 1$: the affine algebra collapses to the loop algebra, whose invariant form is identically zero. The Sugawara central charge $c_{\mathrm{Sug}} = k\dim(\frakg)/(k + h^\vee)$ vanishes as well at $k = 0$, while the Kac--Moody modular characteristic $\kappa_{\mathrm{ch}}^{\mathrm{KM}} = \dim(\frakg)(k + h^\vee)/(2 h^\vee)$ takes the value $\dim(\frakg)/2$ (the free-field value, governed by the $h^\vee$ shift of the Sugawara construction, not the level prefix of the $r$-matrix).
 \item \emph{Critical level} ($k = -h^\vee$). The prefix $(k + h^\vee)$ vanishes, so $\kappa_{\mathrm{ch}}^{\mathrm{KM}} = 0$; the $R$-matrix degenerates and the quantum group collapses to the classical enveloping algebra of the loop algebra. This is the Feigin--Frenkel regime.
\end{enumerate}
The two limits give independent consistency conditions: the $r$-matrix vanishes as $k \to 0$ by the level prefix, and degenerates at $k = -h^\vee$ by the critical-level prefactor.
\end{remark}

The spectral-parameter version extends Proposition~\ref{prop:qgf-qybe} to a family.

\begin{proposition}[Parametric Yang--Baxter equation for affine quantum groups]
\label{prop:qgf-parametric-ybe}
\ClaimStatusProvedElsewhere
For the affine quantum group $U_q(\hat{\frakg})$ at generic $q$, there is a universal $R$-matrix $\cR(z/w) \in U_q(\hat{\frakg})^{\hat{\otimes} 2}[[z^{\pm 1}, w^{\pm 1}]]$ depending on a multiplicative spectral parameter, which satisfies the parametric Yang--Baxter equation
\begin{equation}
 \cR_{12}(z_1/z_2) \, \cR_{13}(z_1/z_3) \, \cR_{23}(z_2/z_3)
 = \cR_{23}(z_2/z_3) \, \cR_{13}(z_1/z_3) \, \cR_{12}(z_1/z_2).
 \label{eq:qgf-parametric-ybe}
\end{equation}
In the classical limit, the trigonometric solution reduces to the Baxterised $r(z)$ of Vol~II, Chapter on the DK bridge.
\end{proposition}

\begin{proof}[Attribution]
Drinfeld, ``A new realization of Yangians and quantized affine algebras'' (Soviet Math.\ Dokl.\ 1988); Frenkel--Reshetikhin, ``Quantum affine algebras and holonomic difference equations'' (Comm.\ Math.\ Phys.\ 1992).
\end{proof}

\section{Representation categories}
\label{sec:qgf-rep-categories}
% ============================================================

Let $\Rep_q(\frakg) := \Rep^{\mathrm{fd}}(U_q(\frakg))$ denote the category of finite-dimensional type-$1$ representations of $U_q(\frakg)$, equipped with the tensor product provided by the coproduct. The universal $R$-matrix promotes $\Rep_q(\frakg)$ to a braided monoidal category: the braiding on $V \otimes W$ is given by $c_{V, W} = \tau \circ \cR|_{V \otimes W}$ where $\tau$ is the flip.

\begin{proposition}[Ribbon structure]
\label{prop:qgf-ribbon}
\ClaimStatusProvedElsewhere
For generic $q$, the category $\Rep_q(\frakg)$ is a ribbon category in the sense of Reshetikhin--Turaev, with twist $\theta_V$ acting on an irreducible $V$ of highest weight $\lambda$ by $q^{(\lambda, \lambda + 2\rho)}$, where $\rho$ is the half-sum of positive roots.
\end{proposition}

\begin{proof}[Attribution]
Reshetikhin--Turaev, ``Ribbon graphs and their invariants derived from quantum groups'' (Comm.\ Math.\ Phys.\ 1990); Turaev, \emph{Quantum Invariants of Knots and $3$-Manifolds}, Chapter~XI.
\end{proof}

\begin{proposition}[Drinfeld--Kohno, generic $q$]
\label{prop:qgf-drinfeld-kohno}
\ClaimStatusProvedElsewhere
At generic $q$, the category $\Rep_q(\frakg)$ is equivalent as a braided monoidal category to the category $\Rep(\frakg)$ of finite-dimensional $\frakg$-modules equipped with the Drinfeld associator (built from the KZ connection) and the braiding induced by the classical $r$-matrix of Proposition~\ref{prop:qgf-classical-limit-r}.
\end{proposition}

\begin{proof}[Attribution]
Drinfeld, ``On almost cocommutative Hopf algebras'' (Leningrad Math.\ J.\ 1990) and ``Quasi-Hopf algebras'' (Leningrad Math.\ J.\ 1990); Kohno, ``Monodromy representations of braid groups and Yang--Baxter equations'' (Ann.\ Inst.\ Fourier 1987).
\end{proof}

The root-of-unity case is the setting relevant to conformal blocks.

\begin{proposition}[Modular tensor category at a root of unity]
\label{prop:qgf-mtc-root-of-unity}
\ClaimStatusProvedElsewhere
Let $q = \exp(\pi i / (d \cdot (k + h^\vee)))$ with $k$ a positive integer, $h^\vee$ the dual Coxeter number, and $d$ the lacing number. The category $\Rep_q(\frakg)$ is \emph{not} semisimple, but the semisimplification (quotient by the tensor ideal of negligible morphisms, equivalently by modules of quantum dimension zero) is a modular tensor category $\mathcal{MTC}_q(\frakg)$ in the sense of Turaev.
\end{proposition}

\begin{proof}[Attribution]
Andersen, ``Tensor products of quantized tilting modules'' (Comm.\ Math.\ Phys.\ 1992); Andersen--Paradowski, ``Fusion categories arising from semisimple Lie algebras'' (Comm.\ Math.\ Phys.\ 1995). The Lusztig divided-power integral form is used implicitly (Lusztig, \emph{Introduction to Quantum Groups}, 1993).
\end{proof}

\begin{theorem}[Kazhdan--Lusztig equivalence]
\label{thm:qgf-kazhdan-lusztig}
\ClaimStatusProvedElsewhere
For $k$ a positive integer and $q = \exp(\pi i / (d(k + h^\vee)))$, there is an equivalence of modular tensor categories
\begin{equation}
 \mathcal{MTC}_q(\frakg) \ \simeq \ \mathcal{O}_k^{\mathrm{int}}(\hat{\frakg}),
 \label{eq:qgf-KL}
\end{equation}
between the semisimplified quantum group category and the category $\mathcal{O}_k^{\mathrm{int}}(\hat{\frakg})$ of integrable highest-weight modules over the affine Kac--Moody algebra $\hat{\frakg}$ at level $k$, with its Knizhnik--Zamolodchikov braiding.
\end{theorem}

\begin{proof}[Attribution]
Kazhdan--Lusztig, ``Tensor structures arising from affine Lie algebras'' (I--IV, J.\ Amer.\ Math.\ Soc.\ 1993--1994); Finkelberg, ``An equivalence of fusion categories'' (GAFA 1996), closing the positive-level case. The equivalence requires $q$ at a root of unity; at generic $q$ the two sides are no longer equivalent (the affine side does not even make sense as a finite modular category).
\end{proof}

\begin{conjecture}[Two-parameter Kazhdan--Lusztig equivalence, toroidal $\mathfrak{sl}_2$]
\label{conj:kazhdan-lusztig-toroidal-sl2}
\ClaimStatusConjectured
Let $\mathfrak{sl}_2^{\hat{\hat{\ }}}$ denote the toroidal (double-affine) Lie algebra associated with $\mathfrak{sl}_2$, and let $U_{q,t}(\mathfrak{sl}_2^{\hat{\hat{\ }}})$ be the two-parameter quantum toroidal algebra of Ding--Iohara--Miki type at rank one. For parameters $(q, t) \in (\C^*)^2$ of the form
\[
 (q, t) \;=\; \bigl(e^{i\pi k/(k+2)},\ e^{i\pi k'/(k'+2)}\bigr),
 \qquad k, k' \in \Q \setminus \Z_{\le -2},
\]
restricted to the off-resonance locus (excluding $k, k' \in \Z_{\le -2}$, where the specialisation is ill-defined), there exists a braided monoidal equivalence
\begin{equation}
 \Rep_{q,t}\bigl(U_{q,t}(\mathfrak{sl}_2^{\hat{\hat{\ }}})\bigr)
 \ \simeq\ \cO_{k,k'}\bigl(\mathfrak{sl}_2^{\hat{\hat{\ }}}\bigr),
 \label{eq:kl-toroidal-sl2}
\end{equation}
between (i) the category of finite-dimensional (admissible) representations of $U_{q,t}(\mathfrak{sl}_2^{\hat{\hat{\ }}})$ with its universal $R$-matrix braiding, and (ii) a suitably-defined double-level category $\cO_{k,k'}(\mathfrak{sl}_2^{\hat{\hat{\ }}})$ of admissible highest-weight $\mathfrak{sl}_2^{\hat{\hat{\ }}}$-modules at horizontal level $k$ and vertical level $k'$, equipped with the two-parameter Knizhnik--Zamolodchikov--Bernard braiding on the torus. The equivalence restricts, along either degeneration $t \to 1$ or $q \to 1$, to the one-parameter affine Kazhdan--Lusztig equivalence of Theorem~\ref{thm:qgf-kazhdan-lusztig} at $\frakg = \mathfrak{sl}_2$.
\end{conjecture}

\begin{remark}[Partial evidence; status of the two sides]
\label{rem:kl-toroidal-sl2-evidence}
The evidence for Conjecture~\ref{conj:kazhdan-lusztig-toroidal-sl2} splits across algebra and category sides.

\emph{Algebra side (reducible).} The quantum toroidal algebra $U_{q,t}(\mathfrak{sl}_2^{\hat{\hat{\ }}})$ admits three isomorphic presentations whose equivalences are established: (a)~a Drinfeld-type generator-and-relation presentation (Ding--Iohara 1997, $\mathfrak{sl}_2$ specialisation); (b)~a shuffle-algebra presentation on $\Q(q, t)[z^{\pm 1}]$ with the Feigin--Hashizume--Hoshino--Shiraishi--Yanagida symmetric wheel conditions (FHHSY, J.\ Math.\ Phys.\ 2009); (c)~an elliptic-Hall / Ding--Iohara--Miki Fock presentation (Schiffmann--Vasserot, Duke 2013, at $\mathfrak{gl}_1$, extended to $\mathfrak{sl}_2$ via Feigin--Tsymbaliuk). Miki (J.\ Math.\ Phys.\ 2007) supplies an $S_3$-automorphism exchanging the horizontal, vertical, and diagonal axes of the two-parameter spectral torus. On the Drinfeld-double level, the positive half $U^+_{q,t}$ coincides with the shuffle algebra and, with its natural coproduct, lifts to a bialgebra with a universal $R$-matrix; this produces the braided monoidal structure on $\Rep_{q,t}$ in \eqref{eq:kl-toroidal-sl2}. The $\mathfrak{sl}_2$ case descends from the Feigin--Jimbo--Miwa--Mukhin treatment of higher-rank quantum toroidal $\mathfrak{gl}_n$ (J.\ Phys.\ A 2015) by specialisation, with Soibelman's cohomological Hall algebra picture and Negu\c{t}'s shuffle presentation (IMRN 2014) providing further structural reductions.

\emph{Category side (genuinely open).} The right-hand side $\cO_{k,k'}(\mathfrak{sl}_2^{\hat{\hat{\ }}})$ of \eqref{eq:kl-toroidal-sl2} has no construction comparable to Finkelberg's affine fusion at one level. Two candidate constructions are (i)~admissible modules over the toroidal Kac--Moody vertex algebra with fusion via surface operators on a torus, and (ii)~a modular functor for the double-affine Knizhnik--Zamolodchikov--Bernard equation on a $T^2$-worldsheet. Neither has been shown to produce a modular-tensor-category structure; the associativity constraint for fusion of admissible modules at double level is the explicit gap, and this is why \eqref{eq:kl-toroidal-sl2} cannot presently be stated as a theorem.

\emph{One-parameter degeneration check.} Under $t \to 1$, the Ding--Iohara--Miki shuffle algebra degenerates to the quantum affine shuffle algebra of Feigin--Odesskii type, $\Rep_{q,t}$ degenerates to $\Rep_q(\widehat{\mathfrak{sl}_2})$ at level $k$, and the right-hand side degenerates to $\cO_k^{\mathrm{int}}(\widehat{\mathfrak{sl}_2})$. On this degeneration, Equation~\eqref{eq:kl-toroidal-sl2} becomes the $\frakg = \mathfrak{sl}_2$ case of Theorem~\ref{thm:qgf-kazhdan-lusztig}, which provides a codimension-one consistency check.
\end{remark}

\begin{proposition}[Algebra side of toroidal Kazhdan--Lusztig, $\mathfrak{sl}_2$, formal disk]
\label{prop:kl-toroidal-sl2-algebra-side}
\ClaimStatusProvedElsewhere
At formal-disk level, the algebra $U_{q,t}(\mathfrak{sl}_2^{\hat{\hat{\ }}})$ is isomorphic to the Ding--Iohara--Miki shuffle algebra $\mathrm{Sh}_{q,t}(\mathfrak{sl}_2)$ of Feigin--Hashizume--Hoshino--Shiraishi--Yanagida, and carries a natural bialgebra structure with a universal $R$-matrix acting on Fock representations, compatible with the $S_3$-automorphism of Miki (2007) that permutes the three spectral axes of the two-parameter torus.
\end{proposition}

\begin{proof}[Attribution]
Ding--Iohara 1997 (Drinfeld-type presentation); Feigin--Hashizume--Hoshino--Shiraishi--Yanagida 2009 (shuffle presentation and bialgebra structure for $\mathfrak{sl}_2$, Theorem~3.1 and Section~5); Miki 2007 (the $S_3$-automorphism); Schiffmann--Vasserot 2013 (elliptic Hall / Fock realisation and universal $R$-matrix at $\mathfrak{gl}_1$, extended to $\mathfrak{sl}_2$ via Feigin--Tsymbaliuk); Negu\c{t} 2014 (shuffle presentation in the higher-rank framework, specialising at rank one). The composition $\text{Ding--Iohara (Drinfeld)} \cong \text{FHHSY (shuffle)} \cong \text{Schiffmann--Vasserot / Feigin--Tsymbaliuk (elliptic-Hall Fock)}$, glued structurally by the Miki $S_3$-automorphism, assembles the three presentations into a single bialgebra with universal $R$-matrix.
\end{proof}

\section{The four regimes}
\label{sec:qgf-four-regimes}
% ============================================================

The quantum group associated to a simple Lie algebra $\frakg$ admits four incarnations, indexed by the geometry of the spectral-parameter curve. Each regime produces a different algebra with a different $R$-matrix; the bar-cobar machine of Vol~I applies uniformly to all four, and the CY programme of this volume recovers each regime from a geometric input.

\begin{definition}[The four quantum group regimes]
\label{def:qgf-four-regimes}
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \emph{Rational: the Yangian $Y_\hbar(\frakg)$.} Spectral parameter $u = z_1 - z_2 \in \C$ (additive). The $R$-matrix $R(u) = 1 - \hbar\,P/u + O(\hbar^2)$ solves the additive Yang--Baxter equation. The underlying curve is $\bP^1$ (or $\C$). In the classical limit, $r(u) = P/u$ is the rational $r$-matrix. The Yangian is the $\Eone$-ordered bar construction of the affine chiral algebra at generic level (Vol~I, Face~F5; Vol~II, DK bridge).

 \item \emph{Trigonometric: the quantum loop algebra $U_q(L\frakg) \cong U_q(\hat{\frakg})$ at level zero.} Spectral parameter $z = e^{2\pi i u / \beta} \in \C^*$ (multiplicative). The $R$-matrix is the image of the universal $\cR$ of Proposition~\ref{prop:qgf-parametric-ybe} under evaluation representations. The underlying curve is $\C^*$. Sections~\ref{sec:qgf-quantized-enveloping}--\ref{sec:qgf-rep-categories} treat this regime.

 \item \emph{Elliptic: the elliptic quantum group $E_{q,p}(\frakg)$.} Spectral parameter $u \in E_\tau$ (elliptic curve). The Belavin $R$-matrix is the unique (up to gauge) solution of the QYBE with doubly-periodic meromorphic dependence on the spectral parameter. The propagator becomes $d\!\log\theta_1(z \mid \tau)$, and the Arnold relation lifts to the Fay trisecant identity. See Chapter~\ref{chap:toroidal-elliptic} for the bar-cobar treatment.

 \item \emph{Toroidal: the quantum toroidal algebra $U_{q,t}(\hat{\hat{\frakg}})$.} Two spectral parameters $(z_1, z_2) \in \C^* \times \C^*$ (double loop). This regime arises from the Omega-background geometry $\R_t \times \C^2$ (Costello's 5d noncommutative Chern--Simons theory) and is the habitat of the affine Yangian $Y(\widehat{\fgl}_1)$ central to toric CY$_3$ computations. The Koszul dual is conjectured to be $U_{q^{-1}, t^{-1}}(\hat{\hat{\frakg}})$ (parameter inversion; Table~\ref{tab:rat-trig-ell}).
\end{enumerate}
\end{definition}

\begin{remark}[Regime vs.\ Lie type]
\label{rem:qgf-regime-vs-type}
The four regimes are orthogonal to the choice of $\frakg$: each simple Lie algebra $\frakg$ has a rational, trigonometric, elliptic, and toroidal quantum group. The CY programme predicts that the regime is determined by the geometry of the ambient space of line operators: a CY$_3$ with line operators supported on a rational curve produces a Yangian; on a cylinder, a quantum loop algebra; on an elliptic curve, an elliptic quantum group; on a torus fibration, a toroidal algebra.
\end{remark}

\section{Connection to Conjecture CY-C}
\label{sec:qgf-cy-c-connection}
% ============================================================

The Vol~III main conjecture CY-C upgrades Theorem~\ref{thm:qgf-kazhdan-lusztig} from affine Kac--Moody data to CY-category data. We recall the precise statement and scope.

\begin{conjecture}[CY-C: quantum group realization]
\label{conj:qgf-cy-c}
\ClaimStatusConjectured
Let $\cC$ be a smooth proper CY-$d$ category ($d = 2$ or $d = 3$), with $A_\cC = \Phi(\cC)$ defined by Theorem~CY-A$_2$ when $d = 2$ and by Theorem~CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}) when $d = 3$. There exists a parameter $q \in \C^*$ (determined by $\kappa_{\mathrm{ch}}(A_\cC)$) and a Hopf algebra in a braided category, $C(\cC, q)$, such that there is an equivalence of modular tensor categories
\begin{equation}
 \Rep^{\mathrm{fd}}(C(\cC, q)) \ \simeq \ \mathcal{MTC}(\Phi(\cC)),
 \label{eq:qgf-cy-c}
\end{equation}
where $\mathcal{MTC}(\Phi(\cC)) := \Rep^{\Etwo}(A_\cC)^{\mathrm{ss}}$ is the (semisimplified) conformal-block category of $A_\cC$. When $\cC = D^b(\mathrm{Coh}(\mathrm{pt}/G))$ for $G$ a simple algebraic group, one recovers $C(\cC, q) = U_q(\mathrm{Lie}\, G)$ and Theorem~\ref{thm:qgf-kazhdan-lusztig}.
\end{conjecture}

\begin{remark}[Scope by CY dimension]
\label{rem:qgf-cy-c-scope}
Conjecture~\ref{conj:qgf-cy-c} is stated as a single conjecture but splits by dimension:
\begin{itemize}
 \item $d = 2$ (K$3$ and abelian surface cases): the correspondence $\Phi_2$ is constructed (Theorem~CY-A$_2$), so $A_\cC$ is defined, and CY-C reduces to a semisimplification plus reconstruction statement. The BKM route (Borcherds--Gritsenko--Nikulin) provides partial evidence; see Chapter~\ref{ch:k3-times-e} and the toroidal elliptic computation in Chapter~\ref{chap:toroidal-elliptic}. At $d = 2$: $\kappa_{\mathrm{cat}}(\mathrm{K3}) = \chi(\cO_{\mathrm{K3}}) = 2$ (manifold-topological) and the algebraisation invariant $\kappa_{\mathrm{ch}}(\Phi(D^b\Coh(\mathrm{K3}))) = 2$ coincide at the K3 input, by Theorem~CY-A$_2$. The Borcherds weight $\kappa_{\mathrm{BKM}}$ is a distinct invariant attached to the CY$_3$ $K3\times E$ (weight $5$ of $\Delta_5$, per Chapter~\ref{ch:k3-times-e}); it lives in a different dimension and a different construction, and the numerical coincidence $\kappa_{\mathrm{BKM}} = \kappa_{\mathrm{ch}} + \chi(\cO_{\mathrm{K3}}) = 3 + 2 = 5$ at $N=1$ is an $N=1$-specific arithmetic that fails at $N \geq 2$.
 \item $d = 3$: the correspondence $\Phi_3$ is constructed (Theorem~CY-A$_3$, Theorem~\ref{thm:cy-to-chiral-d3}). Conjecture~\ref{conj:qgf-cy-c} at $d = 3$ remains conjectural: CY-A$_3$ constructs the chiral algebra $A_\cC$, but the existence of the quantum vertex chiral group $C(\cC, q)$ is a separate conjecture. The only end-to-end verified case is $\C^3$, where $C(\C^3, q)$ is the affine Yangian $Y^+(\widehat{\mathfrak{gl}}_1)$ (Schiffmann--Vasserot, Kontsevich--Soibelman).
\end{itemize}
In neither case is $C(\cC, q)$ constructed for a general CY category; Conjecture~\ref{conj:qgf-cy-c} is the Vol~III programme, not a theorem.
\end{remark}

\begin{remark}[Mediation by quantum chiral algebras]
\label{rem:qgf-mediation}
The bridge between $U_q(\frakg)$-style data and the chiral algebra side is the quantum chiral algebra formalism of Chapter~\ref{ch:quantum-chiral-algebras}. A quantum chiral algebra is an $\Etwo$-chiral algebra $A$ equipped with a spectral-parameter $R$-matrix $R(z) \in A \otimes A \otimes k((z))$ satisfying the parametric Yang--Baxter equation~\eqref{eq:qgf-parametric-ybe}. Its category of representations $\Rep^{\Etwo}(A)$ is braided monoidal, and the Drinfeld center construction of Chapter~\ref{ch:drinfeld-center} passes from $\Eone$-data on $A$ to $\Etwo$-data on $\Rep(A)$. Conjecture~\ref{conj:qgf-cy-c} asserts that this braided category is the representation category of a Hopf-algebra-like object $C(\cC, q)$, which for $\cC$ of ``affine'' type reduces to a quantum group.
\end{remark}

\begin{remark}[CoHA, line operators, and the positive half]
\label{rem:qgf-coha-line-ops}
For a CY$_3$ manifold $X$ with gauge group $G$, the quantum group $U_q(G)$ controls the category of line operators in the associated 3d $\cN = 2$ theory (Costello--Gaiotto). The cohomological Hall algebra $\CoHA(Q_X, W_X)$ of the quiver-with-potential datum provides the \emph{positive half} $U_q^+(G)$: its multiplication is the Hall product, not a vertex algebra structure. The full quantum group is recovered by the Drinfeld double construction $D(\CoHA) \twoheadrightarrow U_q(G)$, which adjoins the Cartan and negative generators. This passage is constructed for toric CY$_3$ without compact $4$-cycles (Schiffmann--Vasserot, Rapcak--Soibelman--Yang--Zhao); for general CY$_3$, it is part of Conjecture~\ref{conj:qgf-cy-c}.
\end{remark}

\section{CY-C at the abelian level: K3}
\label{sec:qgf-cy-c-k3-abelian}
% ============================================================

This section identifies $C(\frakg, q)$ for $\text{K3}$ at the abelian ($\mathfrak{gl}_1$) level. Three independent \emph{constructions} (of which only Route~A uses the CY-to-chiral correspondence~$\Phi_2$) approach the same algebra; their conjectural convergence is the content of CY-C.

\begin{conjecture}[CY-C for K3 at the abelian level]
\label{conj:cy-c-k3-abelian}
\ClaimStatusConjectured
Let $\cC = D^b(\mathrm{Coh}(\mathrm{K3}))$. At the abelian ($\fgl_1$) level, the quantum group of Conjecture~\ref{conj:qgf-cy-c} is the Drinfeld double of the positive half of the K3 Yangian:
\begin{equation}
  C(\frakg_{\mathrm{K3}}, q) \ = \ D\bigl(Y^+(\frakg_{\mathrm{K3}})\bigr),
  \label{eq:cy-c-k3-abelian}
\end{equation}
where $Y^+(\frakg_{\mathrm{K3}})$ is the positive half of the rank-$24$ Heisenberg-type Yangian with structure function
\begin{equation}
  g_{\mathrm{K3}}(z) = \prod_{i=1}^{24} \frac{z - h_i}{z + h_i},
  \label{eq:cy-c-structure-fn}
\end{equation}
and the parameters $h_1, \ldots, h_{24}$ live in the complexified Mukai lattice $\Lambda_{\mathrm{Muk}} \otimes \C$, with the normalisation $\sum h_i = 0$ (the traceless slice, chosen so that the overall translation of the spectral parameter acts trivially; the condition is a choice of origin rather than a constraint forced by the CY$_2$ structure).
\end{conjecture}

\begin{remark}[Three routes to $C(\frakg_{\mathrm{K3}}, q)$]
\label{rem:cy-c-three-routes}
The identification~\eqref{eq:cy-c-k3-abelian} is supported by three independent constructions:
\begin{enumerate}[label=\textup{(\Alph*)}]
  \item \emph{Chiral route.} $\Phi(D^b(\mathrm{K3})) = H_{\mathrm{Muk}}$ (the Mukai Heisenberg) by Theorem~CY-A$_2$. The $\Eone$-ordered bar complex $B^{\mathrm{ord}}(H_{\mathrm{Muk}})$ produces $Y^+(\frakg_{\mathrm{K3}})$; taking the Drinfeld double gives $C(\frakg, q)$. Conditional on CY-A$_2$ (proved at $d = 2$).
  \item \emph{BFN route.} The Braverman--Finkelberg--Nakajima quantized Coulomb branch $\cA_\hbar(\mathrm{K3})$ for the 3d $\cN = 4$ theory on K3 (Braverman--Finkelberg--Nakajima, arXiv:1601.03586, 1604.03625). For ADE quiver gauge theories, the identification $\cA_\hbar = Y(\frakg_{\mathrm{ADE}})$ is a theorem (BFN 2016). For K3, the identification $\cA_\hbar(\mathrm{K3}) = Y(\frakg_{\mathrm{K3}})$ is conjectural. This route is independent of CY-A.
  \item \emph{MO route, scope-restricted to ADE/Kummer locus.} The Maulik--Okounkov stable envelope construction (arXiv:1211.1287) produces an $R$-matrix $R_{\mathrm{MO}}(z)$ on $K_T(\mathrm{Hilb}^n(\mathrm{K3} \times E))$ \emph{at the ADE/Kummer orbifold locus of K3 moduli only} (Lemma~\ref{lem:mo-bypass-local-to-global} in \texttt{drinfeld\_center.tex}: for generic K3, $\mathrm{Aut}(\mathrm{K3} \times E)^\circ$ is the elliptic curve $E$, not an algebraic torus, so the MO chamber criterion $\dim T \geq 2$ fails; further sharpened by Lemma~\ref{lem:no-Gm-on-E}: the elliptic curve $E$ contains no $\mathbb{G}_m$-subgroup). The FRT reconstruction (Faddeev--Reshetikhin--Takhtajan 1989) applied to $R_{\mathrm{MO}}$ produces a bialgebra $A(R_{\mathrm{MO}})$ at the chosen toric chart. Route~(C) identifies $A(R_{\mathrm{MO}}) = C(\frakg_{\mathrm{K3}}, q)|_{\text{ADE/Kummer}}$ at the ADE/Kummer locus only; the global identification $A(R_{\mathrm{MO}}) = C(\frakg_{\mathrm{K3}}, q)$ across all of K3 moduli requires the additional cocycle assembly described in Remark~\ref{rem:k3-fibration-rescue} of \texttt{drinfeld\_center.tex}.
\end{enumerate}
All three routes produce a rank-$24$ algebra with the same classical limit $\mathrm{Drin}(H_{\mathrm{Muk}})$ (dimension $49 = 24 + 24 + 1$), the same structure function~\eqref{eq:cy-c-structure-fn}, and the same Fock-space dimensions $\dim F_n = p_{24}(n)$ (coefficients of $1/\eta(q)^{24}$). Engine: \texttt{cy\_c\_quantum\_group\_k3} (104 tests).
\end{remark}

\begin{conjecture}[Rep equivalence for K3]
\label{conj:cy-c-k3-rep}
\ClaimStatusConjectured
Under Conjecture~\ref{conj:cy-c-k3-abelian}, the BZFN theorem (Ben-Zvi--Francis--Nadler, arXiv:0805.0157) yields
\begin{equation}
  \Rep^{\mathrm{fd}}\bigl(C(\frakg_{\mathrm{K3}}, q)\bigr)
  \ = \ \cZ\bigl(\Rep^{\Eone}(Y^+(\frakg_{\mathrm{K3}}))\bigr)
  \ = \ \Rep^{\Etwo}\bigl(Y(\frakg_{\mathrm{K3}})\bigr),
  \label{eq:cy-c-k3-rep}
\end{equation}
where $\cZ$ denotes the Drinfeld center. The braiding on $\Rep^{\Etwo}(Y(\frakg_{\mathrm{K3}}))$ is determined by the $R$-matrix~\eqref{eq:cy-c-structure-fn}. Four structural checks (verified computationally, 104 tests):
\begin{enumerate}[label=\textup{(V\arabic*)}]
  \item \emph{Simple objects:} evaluation modules $V_u^{(a)}$ parametrised by $(u, a) \in \C \times \{1, \ldots, 24\}$.
  \item \emph{Braiding:} $R$-matrix matches the MO stable envelope $R$-matrix at all test points. Unitarity $R(z)R(-z) = \mathrm{Id}$ verified.
  \item \emph{Fusion rules:} $V_u^{(a)} \otimes V_v^{(a)}$ is reducible iff $u - v = \pm h_a$.
  \item \emph{Ribbon twist:} $\theta(V_u^{(a)}) = g_a(0)^{-1} = -1$. The $-1$ twist is a fermionic $\Z/2$-grading on $V_u^{(a)}$; physically (heuristic, Costello--Gaiotto), this matches the half-hypermultiplet BPS sector of the associated 3d $\cN=4$ theory.
\end{enumerate}
\end{conjecture}

\begin{remark}[$R$-matrix is the MO $R$-matrix]
\label{rem:cy-c-rmatrix-mo}
At charge~$1$, the $R$-matrix of $C(\frakg_{\mathrm{K3}}, q)$ is the $24 \times 24$ diagonal matrix $R(z) = \mathrm{diag}(g_a(z))_{a=1}^{24}$ where $g_a(z) = (z - h_a)/(z + h_a)$. This is precisely the Maulik--Okounkov stable envelope $R$-matrix on $K_T(\mathrm{Hilb}^1(\mathrm{K3} \times E))$ (Proposition~\ref{prop:mo-rmatrix-charge2}). At charge~$2$, the eigenvalues on the $324$-dimensional Fock space $F_2$ are given by the box-content formula $R_\lambda(z) = \prod_{s \in \lambda} g_{\mathrm{K3}}(z + c(s))$. The identification is tautological at the FRT level: the MO $R$-matrix is the \emph{input} to the FRT reconstruction. The nontrivial content is that the resulting FRT algebra has the structure of a Yangian.
\end{remark}

\begin{remark}[Root of unity and the CY Kazhdan--Lusztig programme]
\label{rem:cy-c-root-of-unity}
At generic $q$ (equivalently generic $\epsilon$), $\Rep(C(\frakg_{\mathrm{K3}}, q))$ is semisimple with infinitely many simple objects, hence not modular. At a root of unity $q = e^{2\pi i/N}$, the level-$N$ alcove produces $24 N$ simple objects, and the S-matrix is
\[
  S_{(u,a),(v,b)} = \exp\!\bigl(2\pi i \,\omega^{ab}(u - v)/N\bigr),
\]
where $\omega^{ab}$ is the inverse Mukai pairing. For the abelian ($\fgl_1$) theory, this S-matrix is \emph{degenerate}: each of $24$ diagonal blocks is a rank-$1$ circulant. Non-degeneracy (and hence modularity) requires the non-abelian K3 Yangian. The non-degenerate MTC is the CY analogue of the Kazhdan--Lusztig equivalence (Theorem~\ref{thm:qgf-kazhdan-lusztig}).
\end{remark}

\begin{remark}[Koszul conductor]
\label{rem:cy-c-koszul-conductor}
The Koszul dual $C(\frakg_{\mathrm{K3}}, q)^!$ has inverted parameters $h_i \mapsto -h_i$, giving dual structure function $g^!(z) = 1/g_{\mathrm{K3}}(z)$. The Koszul conductor is $K = 0$ (free-field/Kac--Moody branch), so $\kappa_{\mathrm{ch}}(C) + \kappa_{\mathrm{ch}}(C^!) = 0$. For K3, $\kappa_{\mathrm{ch}} = \chi(\cO_{\mathrm{K3}}) = 2$, hence $\kappa_{\mathrm{ch}}(C^!) = -2$. This is compatible with the flop invariance of $\kappa_{\mathrm{ch}}$.

\end{remark}
