\chapter{Quantum Groups: Foundations}
\label{ch:quantum-groups}

Conjecture~CY-C asks when the CY-to-chiral correspondence programme $\{\Phi_d\}$ produces a quantum group at the representation-theoretic level. To answer that question one needs the standard quantum-group package in a form compatible with the ordered bar and the Drinfeld center. The quantized enveloping algebra $U_q(\frakg)$ and its $R$-matrix are the targets of $\Phi_d$ on that side: Conjecture~CY-C predicts that the braided monoidal category $\Rep_q(\frakg)$ arises as $\Rep^{\Etwo}(A_\cC)$ for a CY category $\cC(\frakg, q)$. The Drinfeld--Jimbo presentation and the FRT presentation give two access points, corresponding respectively to the Koszul dual presentation by generators and relations and the ordered-bar RTT formalism.

Vol~III reads this material backwards. Instead of deforming $U(\frakg)$ first and discovering braiding later, it treats $U_q(\frakg)$ as an \emph{output} of the CY-to-chiral correspondence programme $\{\Phi_d\}$ applied to a CY category whose Drinfeld center recovers the modular tensor category of conformal blocks. Everything below is classical and due to Drinfeld, Jimbo, Lusztig, Reshetikhin--Turaev, and Kazhdan--Lusztig; the Vol~III content is the organization around $\{\Phi_d\}$.

\section{Quantized enveloping algebras}
\label{sec:qgf-quantized-enveloping}
% ============================================================

The Drinfeld--Jimbo presentation realises $U_q(\frakg)$ as a deformation of $U(\frakg)$ in the Hopf-algebra category: generators and Serre relations deform with $q$-corrections, and the coproduct acquires an asymmetry that at $q = 1$ reduces to the symmetric classical coproduct. The data entering the definition is the Cartan matrix and its symmetrisers; everything else is forced.

Let $\frakg$ be a finite-dimensional simple Lie algebra over $\C$ of rank $\ell$ with Cartan matrix $(a_{ij})_{i,j=1}^{\ell}$ and symmetrizers $d_1, \ldots, d_\ell \in \{1, 2, 3\}$ such that $d_i a_{ij} = d_j a_{ji}$. Let $q \in \C^*$ be a formal parameter (or complex number, not a root of unity) and write $q_i = q^{d_i}$, $[n]_q = (q^n - q^{-n})/(q - q^{-1})$, $[n]_q! = [1]_q [2]_q \cdots [n]_q$.

\begin{definition}[Drinfeld--Jimbo quantized enveloping algebra]
\label{def:qgf-quantum-group}
\label{def:qgf-drinfeld-jimbo}
The \emph{quantized enveloping algebra} $U_q(\frakg)$ is the unital associative $\C(q)$-algebra generated by $E_i, F_i, K_i^{\pm 1}$ for $i = 1, \ldots, \ell$ subject to the relations
\begin{align}
 K_i K_j &= K_j K_i, \qquad K_i K_i^{-1} = K_i^{-1} K_i = 1, \\
 K_i E_j K_i^{-1} &= q_i^{a_{ij}} E_j, \qquad K_i F_j K_i^{-1} = q_i^{-a_{ij}} F_j, \\
 [E_i, F_j] &= \delta_{ij} \, \frac{K_i - K_i^{-1}}{q_i - q_i^{-1}},
\end{align}
together with the quantum Serre relations, for $i \ne j$,
\begin{equation}
 \sum_{k=0}^{1 - a_{ij}} (-1)^k \binom{1 - a_{ij}}{k}_{q_i} E_i^{1 - a_{ij} - k} E_j E_i^k = 0,
 \label{eq:qgf-quantum-serre-E}
\end{equation}
and the analogous identity for the $F_i$, where $\binom{n}{k}_{q_i} = [n]_{q_i}!/([k]_{q_i}! [n-k]_{q_i}!)$.
\end{definition}

\begin{remark}[Classical limit]
\label{rem:qgf-classical-limit}
Writing $q = e^{\hbar/2}$ and $K_i = q_i^{H_i} = e^{d_i \hbar H_i / 2}$ and formally expanding in $\hbar$, the relation $[E_i, F_j] = \delta_{ij} (K_i - K_i^{-1})/(q_i - q_i^{-1})$ becomes $[E_i, F_j] = \delta_{ij} H_i + O(\hbar)$, and the quantum Serre relations reduce to the classical Serre relations at $\hbar = 0$. Thus $U_q(\frakg)/(q - 1) \cong U(\frakg)$ as Hopf algebras (Drinfeld 1986, Jimbo 1985).
\end{remark}

\begin{proposition}[Hopf algebra structure]
\label{prop:qgf-hopf}
\ClaimStatusProvedElsewhere
$U_q(\frakg)$ carries a Hopf algebra structure with coproduct $\Delta$, counit $\varepsilon$, and antipode $S$ given on generators by
\begin{align}
 \Delta(K_i) &= K_i \otimes K_i, \\
 \Delta(E_i) &= E_i \otimes 1 + K_i \otimes E_i, \\
 \Delta(F_i) &= F_i \otimes K_i^{-1} + 1 \otimes F_i, \\
 \varepsilon(K_i) &= 1, \qquad \varepsilon(E_i) = \varepsilon(F_i) = 0, \\
 S(K_i) &= K_i^{-1}, \qquad S(E_i) = -K_i^{-1} E_i, \qquad S(F_i) = -F_i K_i.
\end{align}
\end{proposition}

\begin{proof}[Attribution]
Drinfeld, ``Quantum groups'' (ICM Berkeley 1986); Jimbo, ``A $q$-difference analogue of $U(\frakg)$ and the Yang--Baxter equation'' (Lett.\ Math.\ Phys.\ 1985). The Hopf axioms are verified by direct computation on generators; the Serre relations are compatible with the coproduct by a standard argument using the $q$-binomial identity.
\end{proof}

The algebra $U_q(\frakg)$ admits a triangular decomposition $U_q(\frakg) \cong U_q^-(\frakg) \otimes U_q^0(\frakg) \otimes U_q^+(\frakg)$, with $U_q^{\pm}(\frakg)$ generated by the $E_i$ respectively $F_i$, and $U_q^0(\frakg) \cong \C(q)[K_1^{\pm 1}, \ldots, K_\ell^{\pm 1}]$ the Cartan subalgebra. On the Vol~III side, this decomposition matches the bar filtration of $B(A_\cC)$: weight zero is $U_q^0$, positive weights are $U_q^+$, negative weights are $U_q^-$.

\begin{remark}[Universal $R$-matrix, formal form]
\label{rem:qgf-universal-R-formal}
Drinfeld (1986) and independently Rosso, Kirillov--Reshetikhin show that $U_q(\frakg)$ is a \emph{quasi-triangular} Hopf algebra: there exists a formal element
\begin{equation}
 \cR = \sum_{\mu \in Q^+} \cR_\mu \in U_q(\frakg) \mathbin{\hat{\otimes}} U_q(\frakg),
 \label{eq:qgf-universal-R}
\end{equation}
where the sum ranges over the positive root lattice $Q^+$ and each $\cR_\mu \in U_q^+ \otimes U_q^-$ is a finite product of quantum exponentials. The element $\cR$ intertwines $\Delta$ with its opposite: $\cR \Delta(x) \cR^{-1} = \Delta^{\mathrm{op}}(x)$ for all $x \in U_q(\frakg)$. Closed expressions exist only for $\frakg = \mathfrak{sl}_2$ and low-rank cases.
\end{remark}

\section{The universal $R$-matrix and Yang--Baxter equation}
\label{sec:qgf-r-matrix-ybe}
% ============================================================

\begin{proposition}[Quantum Yang--Baxter equation]
\label{prop:qgf-qybe}
\ClaimStatusProvedElsewhere
The universal $R$-matrix $\cR \in U_q(\frakg)^{\hat{\otimes} 2}$ of Remark~\ref{rem:qgf-universal-R-formal} satisfies the quantum Yang--Baxter equation
\begin{equation}
 \cR_{12} \, \cR_{13} \, \cR_{23} = \cR_{23} \, \cR_{13} \, \cR_{12}
 \qquad \text{in } U_q(\frakg)^{\hat{\otimes} 3},
 \label{eq:qgf-qybe}
\end{equation}
where $\cR_{ij}$ denotes the embedding of $\cR$ into the $(i, j)$-tensor factors (with identity on the remaining factor).
\end{proposition}

\begin{proof}[Attribution]
Drinfeld, ``Quantum groups'' (ICM Berkeley 1986), Theorem~2; see also Chari--Pressley, \emph{A Guide to Quantum Groups}, \S4.2. The proof is a direct consequence of quasi-triangularity and the coassociativity of $\Delta$.
\end{proof}

\begin{proposition}[Classical limit and cross-volume compatibility]
\label{prop:qgf-classical-limit-r}
\ClaimStatusProvedElsewhere
Write $q = e^{\hbar/2}$ and expand the universal $R$-matrix as
\begin{equation}
 \cR = 1 + \hbar \, r + O(\hbar^2), \qquad r \in \frakg \otimes \frakg.
 \label{eq:qgf-R-expansion}
\end{equation}
Then $r$ is the \emph{classical $r$-matrix} of $\frakg$, satisfying the symmetry identity $r + r^{21} = \Omega_{\frakg}$ where $\Omega_{\frakg} \in (\frakg \otimes \frakg)^{\frakg}$ is the quadratic Casimir tensor (normalised by the invariant form). Explicitly, $r$ decomposes as $r = \tfrac{1}{2}\Omega_{\frakg} + r_{\mathrm{sk}}$ with $r_{\mathrm{sk}} = \sum_{\alpha > 0} E_\alpha \wedge F_\alpha$ the skew-symmetric Drinfeld--Sklyanin part; the skew part carries the Lie-bialgebra cobracket data and the symmetric part is forced by quasi-triangularity. The quantum Yang--Baxter equation~\eqref{eq:qgf-qybe} expands to the classical Yang--Baxter equation
\begin{equation}
 [r_{12}, r_{13}] + [r_{12}, r_{23}] + [r_{13}, r_{23}] = 0
 \label{eq:qgf-cybe}
\end{equation}
at order $\hbar^2$.
\end{proposition}

\begin{proof}[Attribution]
Drinfeld, ``Hamiltonian structures on Lie groups, Lie bialgebras and the geometric meaning of classical Yang--Baxter equations'' (Soviet Math.\ Dokl.\ 1983); Drinfeld (1986), Theorem~3.
\end{proof}

\begin{remark}[Level-stripped affine $r$-matrix and the two sanity points]
\label{rem:qgf-level-stripped-r}
Passing from $U_q(\frakg)$ (finite type) to the affine quantum group $U_q(\hat{\frakg})$ at level $k$, Proposition~\ref{prop:qgf-classical-limit-r} acquires a level prefix: the classical limit produces
\begin{equation}
 r(z) = k \cdot \frac{\Omega_{\frakg}}{z} + O(\hbar, z^0),
 \label{eq:qgf-r-level-k}
\end{equation}
matching the Vol~I and Vol~II convention. The affine formula has two distinguished values of $k$ at which the algebraic structure collapses for different reasons; both must be verified whenever an affine $r$-matrix is written.
\begin{enumerate}[label=(\alph*)]
 \item \emph{Level zero} ($k = 0$). The level prefix in~\eqref{eq:qgf-r-level-k} vanishes, so $r(z) = 0$ and $\cR(z) = 1$: the affine algebra collapses to the loop algebra, whose invariant form is identically zero. The Sugawara central charge $c_{\mathrm{Sug}} = k\dim(\frakg)/(k + h^\vee)$ vanishes as well at $k = 0$, while the Kac--Moody modular characteristic $\kappa_{\mathrm{ch}}^{\mathrm{KM}} = \dim(\frakg)(k + h^\vee)/(2 h^\vee)$ takes the value $\dim(\frakg)/2$ (the free-field value, governed by the $h^\vee$ shift of the Sugawara construction, not the level prefix of the $r$-matrix).
 \item \emph{Critical level} ($k = -h^\vee$). The prefix $(k + h^\vee)$ vanishes, so $\kappa_{\mathrm{ch}}^{\mathrm{KM}} = 0$; the $R$-matrix degenerates and the quantum group collapses to the classical enveloping algebra of the loop algebra. This is the Feigin--Frenkel regime.
\end{enumerate}
The two limits give independent consistency conditions: the $r$-matrix vanishes as $k \to 0$ by the level prefix, and degenerates at $k = -h^\vee$ by the critical-level prefactor.
\end{remark}

The spectral-parameter version extends Proposition~\ref{prop:qgf-qybe} to a family.

\begin{proposition}[Parametric Yang--Baxter equation for affine quantum groups]
\label{prop:qgf-parametric-ybe}
\ClaimStatusProvedElsewhere
For the affine quantum group $U_q(\hat{\frakg})$ at generic $q$, there is a universal $R$-matrix $\cR(z/w) \in U_q(\hat{\frakg})^{\hat{\otimes} 2}[[z^{\pm 1}, w^{\pm 1}]]$ depending on a multiplicative spectral parameter, which satisfies the parametric Yang--Baxter equation
\begin{equation}
 \cR_{12}(z_1/z_2) \, \cR_{13}(z_1/z_3) \, \cR_{23}(z_2/z_3)
 = \cR_{23}(z_2/z_3) \, \cR_{13}(z_1/z_3) \, \cR_{12}(z_1/z_2).
 \label{eq:qgf-parametric-ybe}
\end{equation}
In the classical limit, the trigonometric solution reduces to the Baxterised $r(z)$ of Vol~II, Chapter on the DK bridge.
\end{proposition}

\begin{proof}[Attribution]
Drinfeld, ``A new realization of Yangians and quantized affine algebras'' (Soviet Math.\ Dokl.\ 1988); Frenkel--Reshetikhin, ``Quantum affine algebras and holonomic difference equations'' (Comm.\ Math.\ Phys.\ 1992).
\end{proof}

\section{Representation categories}
\label{sec:qgf-rep-categories}
% ============================================================

Let $\Rep_q(\frakg) := \Rep^{\mathrm{fd}}(U_q(\frakg))$ denote the category of finite-dimensional type-$1$ representations of $U_q(\frakg)$, equipped with the tensor product provided by the coproduct. The universal $R$-matrix promotes $\Rep_q(\frakg)$ to a braided monoidal category: the braiding on $V \otimes W$ is given by $c_{V, W} = \tau \circ \cR|_{V \otimes W}$ where $\tau$ is the flip.

\begin{proposition}[Ribbon structure]
\label{prop:qgf-ribbon}
\ClaimStatusProvedElsewhere
For generic $q$, the category $\Rep_q(\frakg)$ is a ribbon category in the sense of Reshetikhin--Turaev, with twist $\theta_V$ acting on an irreducible $V$ of highest weight $\lambda$ by $q^{(\lambda, \lambda + 2\rho)}$, where $\rho$ is the half-sum of positive roots.
\end{proposition}

\begin{proof}[Attribution]
Reshetikhin--Turaev, ``Ribbon graphs and their invariants derived from quantum groups'' (Comm.\ Math.\ Phys.\ 1990); Turaev, \emph{Quantum Invariants of Knots and $3$-Manifolds}, Chapter~XI.
\end{proof}

\begin{proposition}[Drinfeld--Kohno, generic $q$]
\label{prop:qgf-drinfeld-kohno}
\ClaimStatusProvedElsewhere
At generic $q$, the category $\Rep_q(\frakg)$ is equivalent as a braided monoidal category to the category $\Rep(\frakg)$ of finite-dimensional $\frakg$-modules equipped with the Drinfeld associator (built from the KZ connection) and the braiding induced by the classical $r$-matrix of Proposition~\ref{prop:qgf-classical-limit-r}.
\end{proposition}

\begin{proof}[Attribution]
Drinfeld, ``On almost cocommutative Hopf algebras'' (Leningrad Math.\ J.\ 1990) and ``Quasi-Hopf algebras'' (Leningrad Math.\ J.\ 1990); Kohno, ``Monodromy representations of braid groups and Yang--Baxter equations'' (Ann.\ Inst.\ Fourier 1987).
\end{proof}

The root-of-unity case is the setting relevant to conformal blocks.

\begin{proposition}[Modular tensor category at a root of unity]
\label{prop:qgf-mtc-root-of-unity}
\ClaimStatusProvedElsewhere
Let $q = \exp(\pi i / (d \cdot (k + h^\vee)))$ with $k$ a positive integer, $h^\vee$ the dual Coxeter number, and $d$ the lacing number. The category $\Rep_q(\frakg)$ is \emph{not} semisimple, but the semisimplification (quotient by the tensor ideal of negligible morphisms, equivalently by modules of quantum dimension zero) is a modular tensor category $\mathcal{MTC}_q(\frakg)$ in the sense of Turaev.
\end{proposition}

\begin{proof}[Attribution]
Andersen, ``Tensor products of quantized tilting modules'' (Comm.\ Math.\ Phys.\ 1992); Andersen--Paradowski, ``Fusion categories arising from semisimple Lie algebras'' (Comm.\ Math.\ Phys.\ 1995). The Lusztig divided-power integral form is used implicitly (Lusztig, \emph{Introduction to Quantum Groups}, 1993).
\end{proof}

\begin{theorem}[Kazhdan--Lusztig equivalence]
\label{thm:qgf-kazhdan-lusztig}
\ClaimStatusProvedElsewhere
For $k$ a positive integer and $q = \exp(\pi i / (d(k + h^\vee)))$, there is an equivalence of modular tensor categories
\begin{equation}
 \mathcal{MTC}_q(\frakg) \ \simeq \ \mathcal{O}_k^{\mathrm{int}}(\hat{\frakg}),
 \label{eq:qgf-KL}
\end{equation}
between the semisimplified quantum group category and the category $\mathcal{O}_k^{\mathrm{int}}(\hat{\frakg})$ of integrable highest-weight modules over the affine Kac--Moody algebra $\hat{\frakg}$ at level $k$, with its Knizhnik--Zamolodchikov braiding.
\end{theorem}

\begin{proof}[Attribution]
Kazhdan--Lusztig, ``Tensor structures arising from affine Lie algebras'' (I--IV, J.\ Amer.\ Math.\ Soc.\ 1993--1994); Finkelberg, ``An equivalence of fusion categories'' (GAFA 1996), closing the positive-level case. The equivalence requires $q$ at a root of unity; at generic $q$ the two sides are no longer equivalent (the affine side does not even make sense as a finite modular category).
\end{proof}

\begin{conjecture}[Two-parameter Kazhdan--Lusztig equivalence, toroidal $\mathfrak{sl}_2$]
\label{conj:kazhdan-lusztig-toroidal-sl2}
\ClaimStatusConjectured
Let $\mathfrak{sl}_2^{\hat{\hat{\ }}}$ denote the toroidal (double-affine) Lie algebra associated with $\mathfrak{sl}_2$, and let $U_{q,t}(\mathfrak{sl}_2^{\hat{\hat{\ }}})$ be the two-parameter quantum toroidal algebra of Ding--Iohara--Miki type at rank one. For parameters $(q, t) \in (\C^*)^2$ of the form
\[
 (q, t) \;=\; \bigl(e^{i\pi k/(k+2)},\ e^{i\pi k'/(k'+2)}\bigr),
 \qquad k, k' \in \Q \setminus \Z_{\le -2},
\]
restricted to the off-resonance locus (excluding $k, k' \in \Z_{\le -2}$, where the specialisation is ill-defined), there exists a braided monoidal equivalence
\begin{equation}
 \Rep_{q,t}\bigl(U_{q,t}(\mathfrak{sl}_2^{\hat{\hat{\ }}})\bigr)
 \ \simeq\ \cO_{k,k'}\bigl(\mathfrak{sl}_2^{\hat{\hat{\ }}}\bigr),
 \label{eq:kl-toroidal-sl2}
\end{equation}
between (i) the category of finite-dimensional (admissible) representations of $U_{q,t}(\mathfrak{sl}_2^{\hat{\hat{\ }}})$ with its universal $R$-matrix braiding, and (ii) a suitably-defined double-level category $\cO_{k,k'}(\mathfrak{sl}_2^{\hat{\hat{\ }}})$ of admissible highest-weight $\mathfrak{sl}_2^{\hat{\hat{\ }}}$-modules at horizontal level $k$ and vertical level $k'$, equipped with the two-parameter Knizhnik--Zamolodchikov--Bernard braiding on the torus. The equivalence restricts, along either degeneration $t \to 1$ or $q \to 1$, to the one-parameter affine Kazhdan--Lusztig equivalence of Theorem~\ref{thm:qgf-kazhdan-lusztig} at $\frakg = \mathfrak{sl}_2$.
\end{conjecture}

\begin{remark}[Partial evidence; status of the two sides]
\label{rem:kl-toroidal-sl2-evidence}
The evidence for Conjecture~\ref{conj:kazhdan-lusztig-toroidal-sl2} splits across algebra and category sides.

\emph{Algebra side (reducible).} The quantum toroidal algebra $U_{q,t}(\mathfrak{sl}_2^{\hat{\hat{\ }}})$ admits three isomorphic presentations whose equivalences are established: (a)~a Drinfeld-type generator-and-relation presentation (Ding--Iohara 1997, $\mathfrak{sl}_2$ specialisation); (b)~a shuffle-algebra presentation on $\Q(q, t)[z^{\pm 1}]$ with the Feigin--Hashizume--Hoshino--Shiraishi--Yanagida symmetric wheel conditions (FHHSY, J.\ Math.\ Phys.\ 2009); (c)~an elliptic-Hall / Ding--Iohara--Miki Fock presentation (Schiffmann--Vasserot, Duke 2013, at $\mathfrak{gl}_1$, extended to $\mathfrak{sl}_2$ via Feigin--Tsymbaliuk). Miki (J.\ Math.\ Phys.\ 2007) supplies an $S_3$-automorphism exchanging the horizontal, vertical, and diagonal axes of the two-parameter spectral torus. On the Drinfeld-double level, the positive half $U^+_{q,t}$ coincides with the shuffle algebra and, with its natural coproduct, lifts to a bialgebra with a universal $R$-matrix; this produces the braided monoidal structure on $\Rep_{q,t}$ in \eqref{eq:kl-toroidal-sl2}. The $\mathfrak{sl}_2$ case descends from the Feigin--Jimbo--Miwa--Mukhin treatment of higher-rank quantum toroidal $\mathfrak{gl}_n$ (J.\ Phys.\ A 2015) by specialisation, with Soibelman's cohomological Hall algebra picture and Negu\c{t}'s shuffle presentation (IMRN 2014) providing further structural reductions.

\emph{Category side (genuinely open).} The right-hand side $\cO_{k,k'}(\mathfrak{sl}_2^{\hat{\hat{\ }}})$ of \eqref{eq:kl-toroidal-sl2} has no construction comparable to Finkelberg's affine fusion at one level. Two candidate constructions are (i)~admissible modules over the toroidal Kac--Moody vertex algebra with fusion via surface operators on a torus, and (ii)~a modular functor for the double-affine Knizhnik--Zamolodchikov--Bernard equation on a $T^2$-worldsheet. Neither has been shown to produce a modular-tensor-category structure; the associativity constraint for fusion of admissible modules at double level is the explicit gap, and this is why \eqref{eq:kl-toroidal-sl2} cannot presently be stated as a theorem.

\emph{One-parameter degeneration check.} Under $t \to 1$, the Ding--Iohara--Miki shuffle algebra degenerates to the quantum affine shuffle algebra of Feigin--Odesskii type, $\Rep_{q,t}$ degenerates to $\Rep_q(\widehat{\mathfrak{sl}_2})$ at level $k$, and the right-hand side degenerates to $\cO_k^{\mathrm{int}}(\widehat{\mathfrak{sl}_2})$. On this degeneration, Equation~\eqref{eq:kl-toroidal-sl2} becomes the $\frakg = \mathfrak{sl}_2$ case of Theorem~\ref{thm:qgf-kazhdan-lusztig}, which provides a codimension-one consistency check.
\end{remark}

\begin{proposition}[Algebra side of toroidal Kazhdan--Lusztig, $\mathfrak{sl}_2$, formal disk]
\label{prop:kl-toroidal-sl2-algebra-side}
\ClaimStatusProvedElsewhere
At formal-disk level, the algebra $U_{q,t}(\mathfrak{sl}_2^{\hat{\hat{\ }}})$ is isomorphic to the Ding--Iohara--Miki shuffle algebra $\mathrm{Sh}_{q,t}(\mathfrak{sl}_2)$ of Feigin--Hashizume--Hoshino--Shiraishi--Yanagida, and carries a natural bialgebra structure with a universal $R$-matrix acting on Fock representations, compatible with the $S_3$-automorphism of Miki (2007) that permutes the three spectral axes of the two-parameter torus.
\end{proposition}

\begin{proof}[Attribution]
Ding--Iohara 1997 (Drinfeld-type presentation); Feigin--Hashizume--Hoshino--Shiraishi--Yanagida 2009 (shuffle presentation and bialgebra structure for $\mathfrak{sl}_2$, Theorem~3.1 and Section~5); Miki 2007 (the $S_3$-automorphism); Schiffmann--Vasserot 2013 (elliptic Hall / Fock realisation and universal $R$-matrix at $\mathfrak{gl}_1$, extended to $\mathfrak{sl}_2$ via Feigin--Tsymbaliuk); Negu\c{t} 2014 (shuffle presentation in the higher-rank framework, specialising at rank one). The composition $\text{Ding--Iohara (Drinfeld)} \cong \text{FHHSY (shuffle)} \cong \text{Schiffmann--Vasserot / Feigin--Tsymbaliuk (elliptic-Hall Fock)}$, glued structurally by the Miki $S_3$-automorphism, assembles the three presentations into a single bialgebra with universal $R$-matrix.
\end{proof}

\section{The four regimes}
\label{sec:qgf-four-regimes}
% ============================================================

The quantum group associated to a simple Lie algebra $\frakg$ admits four incarnations, indexed by the geometry of the spectral-parameter curve. Each regime produces a different algebra with a different $R$-matrix; the bar-cobar machine of Vol~I applies uniformly to all four, and the CY programme of this volume recovers each regime from a geometric input.

\begin{definition}[The four quantum group regimes]
\label{def:qgf-four-regimes}
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \emph{Rational: the Yangian $Y_\hbar(\frakg)$.} Spectral parameter $u = z_1 - z_2 \in \C$ (additive). The $R$-matrix $R(u) = 1 - \hbar\,P/u + O(\hbar^2)$ solves the additive Yang--Baxter equation. The underlying curve is $\bP^1$ (or $\C$). In the classical limit, $r(u) = P/u$ is the rational $r$-matrix. The Yangian is the $\Eone$-ordered bar construction of the affine chiral algebra at generic level (Vol~I, Face~F5; Vol~II, DK bridge).

 \item \emph{Trigonometric: the quantum loop algebra $U_q(L\frakg) \cong U_q(\hat{\frakg})$ at level zero.} Spectral parameter $z = e^{2\pi i u / \beta} \in \C^*$ (multiplicative). The $R$-matrix is the image of the universal $\cR$ of Proposition~\ref{prop:qgf-parametric-ybe} under evaluation representations. The underlying curve is $\C^*$. Sections~\ref{sec:qgf-quantized-enveloping}--\ref{sec:qgf-rep-categories} treat this regime.

 \item \emph{Elliptic: the elliptic quantum group $E_{q,p}(\frakg)$.} Spectral parameter $u \in E_\tau$ (elliptic curve). The Belavin $R$-matrix is the unique (up to gauge) solution of the QYBE with doubly-periodic meromorphic dependence on the spectral parameter. The propagator becomes $d\!\log\theta_1(z \mid \tau)$, and the Arnold relation lifts to the Fay trisecant identity. See Chapter~\ref{chap:toroidal-elliptic} for the bar-cobar treatment.

 \item \emph{Toroidal: the quantum toroidal algebra $U_{q,t}(\hat{\hat{\frakg}})$.} Two spectral parameters $(z_1, z_2) \in \C^* \times \C^*$ (double loop). This regime arises from the Omega-background geometry $\R_t \times \C^2$ (Costello's 5d noncommutative Chern--Simons theory) and is the habitat of the affine Yangian $Y(\widehat{\fgl}_1)$ central to toric CY$_3$ computations. The Koszul dual is conjectured to be $U_{q^{-1}, t^{-1}}(\hat{\hat{\frakg}})$ (parameter inversion; Table~\ref{tab:rat-trig-ell}).
\end{enumerate}
\end{definition}

\begin{remark}[Regime vs.\ Lie type]
\label{rem:qgf-regime-vs-type}
The four regimes are orthogonal to the choice of $\frakg$: each simple Lie algebra $\frakg$ has a rational, trigonometric, elliptic, and toroidal quantum group. The CY programme predicts that the regime is determined by the geometry of the ambient space of line operators: a CY$_3$ with line operators supported on a rational curve produces a Yangian; on a cylinder, a quantum loop algebra; on an elliptic curve, an elliptic quantum group; on a torus fibration, a toroidal algebra.
\end{remark}

\section{Connection to Conjecture CY-C}
\label{sec:qgf-cy-c-connection}
% ============================================================

The Vol~III main conjecture CY-C upgrades Theorem~\ref{thm:qgf-kazhdan-lusztig} from affine Kac--Moody data to CY-category data. We recall the precise statement and scope.

\begin{conjecture}[CY-C: quantum group realization]
\label{conj:qgf-cy-c}
\ClaimStatusConjectured
Let $\cC$ be a smooth proper CY-$d$ category ($d = 2$ or $d = 3$), with $A_\cC = \Phi(\cC)$ defined by Theorem~CY-A$_2$ when $d = 2$ and by Theorem~CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}) when $d = 3$. There exists a parameter $q \in \C^*$ (determined by $\kappa_{\mathrm{ch}}(A_\cC)$) and a Hopf algebra in a braided category, $C(\cC, q)$, such that there is an equivalence of modular tensor categories
\begin{equation}
 \Rep^{\mathrm{fd}}(C(\cC, q)) \ \simeq \ \mathcal{MTC}(\Phi(\cC)),
 \label{eq:qgf-cy-c}
\end{equation}
where $\mathcal{MTC}(\Phi(\cC)) := \Rep^{\Etwo}(A_\cC)^{\mathrm{ss}}$ is the (semisimplified) conformal-block category of $A_\cC$. When $\cC = D^b(\mathrm{Coh}(\mathrm{pt}/G))$ for $G$ a simple algebraic group, one recovers $C(\cC, q) = U_q(\mathrm{Lie}\, G)$ and Theorem~\ref{thm:qgf-kazhdan-lusztig}.
\end{conjecture}

\begin{remark}[Scope by CY dimension]
\label{rem:qgf-cy-c-scope}
Conjecture~\ref{conj:qgf-cy-c} is stated as a single conjecture but splits by dimension:
\begin{itemize}
 \item $d = 2$ (K$3$ and abelian surface cases): the correspondence $\Phi_2$ is constructed (Theorem~CY-A$_2$), so $A_\cC$ is defined, and CY-C reduces to a semisimplification plus reconstruction statement. The BKM route (Borcherds--Gritsenko--Nikulin) provides partial evidence; see Chapter~\ref{ch:k3-times-e} and the toroidal elliptic computation in Chapter~\ref{chap:toroidal-elliptic}. At $d = 2$: $\kappa_{\mathrm{cat}}(\mathrm{K3}) = \chi(\cO_{\mathrm{K3}}) = 2$ (manifold-topological) and the algebraisation invariant $\kappa_{\mathrm{ch}}(\Phi(D^b\Coh(\mathrm{K3}))) = 2$ coincide at the K3 input, by Theorem~CY-A$_2$. The Borcherds weight $\kappa_{\mathrm{BKM}}$ is a distinct invariant attached to the CY$_3$ $K3\times E$ (weight $5$ of $\Delta_5$, per Chapter~\ref{ch:k3-times-e}); it lives in a different dimension and a different construction, and the numerical coincidence $\kappa_{\mathrm{BKM}} = \kappa_{\mathrm{ch}} + \chi(\cO_{\mathrm{K3}}) = 3 + 2 = 5$ at $N=1$ is an $N=1$-specific arithmetic that fails at $N \geq 2$.
 \item $d = 3$: the correspondence $\Phi_3$ is constructed (Theorem~CY-A$_3$, Theorem~\ref{thm:cy-to-chiral-d3}). Conjecture~\ref{conj:qgf-cy-c} at $d = 3$ remains conjectural: CY-A$_3$ constructs the chiral algebra $A_\cC$, but the existence of the quantum vertex chiral group $C(\cC, q)$ is a separate conjecture. The only end-to-end verified case is $\C^3$, where $C(\C^3, q)$ is the affine Yangian $Y^+(\widehat{\mathfrak{gl}}_1)$ (Schiffmann--Vasserot, Kontsevich--Soibelman).
\end{itemize}
In neither case is $C(\cC, q)$ constructed for a general CY category; Conjecture~\ref{conj:qgf-cy-c} is the Vol~III programme, not a theorem.
\end{remark}

\begin{remark}[Mediation by quantum chiral algebras]
\label{rem:qgf-mediation}
The bridge between $U_q(\frakg)$-style data and the chiral algebra side is the quantum chiral algebra formalism of Chapter~\ref{ch:quantum-chiral-algebras}. A quantum chiral algebra is an $\Etwo$-chiral algebra $A$ equipped with a spectral-parameter $R$-matrix $R(z) \in A \otimes A \otimes k((z))$ satisfying the parametric Yang--Baxter equation~\eqref{eq:qgf-parametric-ybe}. Its category of representations $\Rep^{\Etwo}(A)$ is braided monoidal, and the Drinfeld center construction of Chapter~\ref{ch:drinfeld-center} passes from $\Eone$-data on $A$ to $\Etwo$-data on $\Rep(A)$. Conjecture~\ref{conj:qgf-cy-c} asserts that this braided category is the representation category of a Hopf-algebra-like object $C(\cC, q)$, which for $\cC$ of ``affine'' type reduces to a quantum group.
\end{remark}

\begin{remark}[CoHA, line operators, and the positive half]
\label{rem:qgf-coha-line-ops}
For a CY$_3$ manifold $X$ with gauge group $G$, the quantum group $U_q(G)$ controls the category of line operators in the associated 3d $\cN = 2$ theory (Costello--Gaiotto). The cohomological Hall algebra $\CoHA(Q_X, W_X)$ of the quiver-with-potential datum provides the \emph{positive half} $U_q^+(G)$: its multiplication is the Hall product, not a vertex algebra structure. The full quantum group is recovered by the Drinfeld double construction $D(\CoHA) \twoheadrightarrow U_q(G)$, which adjoins the Cartan and negative generators. This passage is constructed for toric CY$_3$ without compact $4$-cycles (Schiffmann--Vasserot, Rapcak--Soibelman--Yang--Zhao); for general CY$_3$, it is part of Conjecture~\ref{conj:qgf-cy-c}.
\end{remark}

\section{CY-C at the abelian level: K3}
\label{sec:qgf-cy-c-k3-abelian}
% ============================================================

This section identifies $C(\frakg, q)$ for $\text{K3}$ at the abelian ($\mathfrak{gl}_1$) level. Three independent \emph{constructions} (of which only Route~A uses the CY-to-chiral correspondence~$\Phi_2$) approach the same algebra; their conjectural convergence is the content of CY-C.

\begin{conjecture}[CY-C for K3 at the abelian level]
\label{conj:cy-c-k3-abelian}
\ClaimStatusConjectured
Let $\cC = D^b(\mathrm{Coh}(\mathrm{K3}))$. At the abelian ($\fgl_1$) level, the quantum group of Conjecture~\ref{conj:qgf-cy-c} is the Drinfeld double of the positive half of the K3 Yangian:
\begin{equation}
  C(\frakg_{\mathrm{K3}}, q) \ = \ D\bigl(Y^+(\frakg_{\mathrm{K3}})\bigr),
  \label{eq:cy-c-k3-abelian}
\end{equation}
where $Y^+(\frakg_{\mathrm{K3}})$ is the positive half of the rank-$24$ Heisenberg-type Yangian with structure function
\begin{equation}
  g_{\mathrm{K3}}(z) = \prod_{i=1}^{24} \frac{z - h_i}{z + h_i},
  \label{eq:cy-c-structure-fn}
\end{equation}
and the parameters $h_1, \ldots, h_{24}$ live in the complexified Mukai lattice $\Lambda_{\mathrm{Muk}} \otimes \C$, with the normalisation $\sum h_i = 0$ (the traceless slice, chosen so that the overall translation of the spectral parameter acts trivially; the condition is a choice of origin rather than a constraint forced by the CY$_2$ structure).
\end{conjecture}

\begin{remark}[Three routes to $C(\frakg_{\mathrm{K3}}, q)$]
\label{rem:cy-c-three-routes}
The identification~\eqref{eq:cy-c-k3-abelian} is supported by three independent constructions:
\begin{enumerate}[label=\textup{(\Alph*)}]
  \item \emph{Chiral route.} $\Phi(D^b(\mathrm{K3})) = H_{\mathrm{Muk}}$ (the Mukai Heisenberg) by Theorem~CY-A$_2$. The $\Eone$-ordered bar complex $B^{\mathrm{ord}}(H_{\mathrm{Muk}})$ produces $Y^+(\frakg_{\mathrm{K3}})$; taking the Drinfeld double gives $C(\frakg, q)$. Conditional on CY-A$_2$ (proved at $d = 2$).
  \item \emph{BFN route.} The Braverman--Finkelberg--Nakajima quantized Coulomb branch $\cA_\hbar(\mathrm{K3})$ for the 3d $\cN = 4$ theory on K3 (Braverman--Finkelberg--Nakajima, arXiv:1601.03586, 1604.03625). For ADE quiver gauge theories, the identification $\cA_\hbar = Y(\frakg_{\mathrm{ADE}})$ is a theorem (BFN 2016). For K3, the identification $\cA_\hbar(\mathrm{K3}) = Y(\frakg_{\mathrm{K3}})$ is conjectural. This route is independent of CY-A.
  \item \emph{MO route, scope-restricted to ADE/Kummer locus.} The Maulik--Okounkov stable envelope construction (arXiv:1211.1287) produces an $R$-matrix $R_{\mathrm{MO}}(z)$ on $K_T(\mathrm{Hilb}^n(\mathrm{K3} \times E))$ \emph{at the ADE/Kummer orbifold locus of K3 moduli only} (Lemma~\ref{lem:mo-bypass-local-to-global} in \texttt{drinfeld\_center.tex}: for generic K3, $\mathrm{Aut}(\mathrm{K3} \times E)^\circ$ is the elliptic curve $E$, not an algebraic torus, so the MO chamber criterion $\dim T \geq 2$ fails; further sharpened by Lemma~\ref{lem:no-Gm-on-E}: the elliptic curve $E$ contains no $\mathbb{G}_m$-subgroup). The FRT reconstruction (Faddeev--Reshetikhin--Takhtajan 1989) applied to $R_{\mathrm{MO}}$ produces a bialgebra $A(R_{\mathrm{MO}})$ at the chosen toric chart. Route~(C) identifies $A(R_{\mathrm{MO}}) = C(\frakg_{\mathrm{K3}}, q)|_{\text{ADE/Kummer}}$ at the ADE/Kummer locus only; the global identification $A(R_{\mathrm{MO}}) = C(\frakg_{\mathrm{K3}}, q)$ across all of K3 moduli requires the additional cocycle assembly described in Remark~\ref{rem:k3-fibration-rescue} of \texttt{drinfeld\_center.tex}.
\end{enumerate}
All three routes produce a rank-$24$ algebra with the same classical limit $\mathrm{Drin}(H_{\mathrm{Muk}})$ (dimension $49 = 24 + 24 + 1$), the same structure function~\eqref{eq:cy-c-structure-fn}, and the same Fock-space dimensions $\dim F_n = p_{24}(n)$ (coefficients of $1/\eta(q)^{24}$). Engine: \texttt{cy\_c\_quantum\_group\_k3} (104 tests).
\end{remark}

\begin{conjecture}[Rep equivalence for K3]
\label{conj:cy-c-k3-rep}
\ClaimStatusConjectured
Under Conjecture~\ref{conj:cy-c-k3-abelian}, the BZFN theorem (Ben-Zvi--Francis--Nadler, arXiv:0805.0157) yields
\begin{equation}
  \Rep^{\mathrm{fd}}\bigl(C(\frakg_{\mathrm{K3}}, q)\bigr)
  \ = \ \cZ\bigl(\Rep^{\Eone}(Y^+(\frakg_{\mathrm{K3}}))\bigr)
  \ = \ \Rep^{\Etwo}\bigl(Y(\frakg_{\mathrm{K3}})\bigr),
  \label{eq:cy-c-k3-rep}
\end{equation}
where $\cZ$ denotes the Drinfeld center. The braiding on $\Rep^{\Etwo}(Y(\frakg_{\mathrm{K3}}))$ is determined by the $R$-matrix~\eqref{eq:cy-c-structure-fn}. Four structural checks (verified computationally, 104 tests):
\begin{enumerate}[label=\textup{(V\arabic*)}]
  \item \emph{Simple objects:} evaluation modules $V_u^{(a)}$ parametrised by $(u, a) \in \C \times \{1, \ldots, 24\}$.
  \item \emph{Braiding:} $R$-matrix matches the MO stable envelope $R$-matrix at all test points. Unitarity $R(z)R(-z) = \mathrm{Id}$ verified.
  \item \emph{Fusion rules:} $V_u^{(a)} \otimes V_v^{(a)}$ is reducible iff $u - v = \pm h_a$.
  \item \emph{Ribbon twist:} $\theta(V_u^{(a)}) = g_a(0)^{-1} = -1$. The $-1$ twist is a fermionic $\Z/2$-grading on $V_u^{(a)}$; physically (heuristic, Costello--Gaiotto), this matches the half-hypermultiplet BPS sector of the associated 3d $\cN=4$ theory.
\end{enumerate}
\end{conjecture}

\begin{remark}[$R$-matrix is the MO $R$-matrix]
\label{rem:cy-c-rmatrix-mo}
At charge~$1$, the $R$-matrix of $C(\frakg_{\mathrm{K3}}, q)$ is the $24 \times 24$ diagonal matrix $R(z) = \mathrm{diag}(g_a(z))_{a=1}^{24}$ where $g_a(z) = (z - h_a)/(z + h_a)$. This is precisely the Maulik--Okounkov stable envelope $R$-matrix on $K_T(\mathrm{Hilb}^1(\mathrm{K3} \times E))$ (Proposition~\ref{prop:mo-rmatrix-charge2}). At charge~$2$, the eigenvalues on the $324$-dimensional Fock space $F_2$ are given by the box-content formula $R_\lambda(z) = \prod_{s \in \lambda} g_{\mathrm{K3}}(z + c(s))$. The identification is tautological at the FRT level: the MO $R$-matrix is the \emph{input} to the FRT reconstruction. The nontrivial content is that the resulting FRT algebra has the structure of a Yangian.
\end{remark}

\begin{remark}[Root of unity and the CY Kazhdan--Lusztig programme]
\label{rem:cy-c-root-of-unity}
At generic $q$ (equivalently generic $\epsilon$), $\Rep(C(\frakg_{\mathrm{K3}}, q))$ is semisimple with infinitely many simple objects, hence not modular. At a root of unity $q = e^{2\pi i/N}$, the level-$N$ alcove produces $24 N$ simple objects, and the S-matrix is
\[
  S_{(u,a),(v,b)} = \exp\!\bigl(2\pi i \,\omega^{ab}(u - v)/N\bigr),
\]
where $\omega^{ab}$ is the inverse Mukai pairing. For the abelian ($\fgl_1$) theory, this S-matrix is \emph{degenerate}: each of $24$ diagonal blocks is a rank-$1$ circulant. Non-degeneracy (and hence modularity) requires the non-abelian K3 Yangian. The non-degenerate MTC is the CY analogue of the Kazhdan--Lusztig equivalence (Theorem~\ref{thm:qgf-kazhdan-lusztig}).
\end{remark}

\begin{remark}[Koszul conductor]
\label{rem:cy-c-koszul-conductor}
The Koszul dual $C(\frakg_{\mathrm{K3}}, q)^!$ has inverted parameters $h_i \mapsto -h_i$, giving dual structure function $g^!(z) = 1/g_{\mathrm{K3}}(z)$. The Koszul conductor is $K = 0$ (free-field/Kac--Moody branch), so $\kappa_{\mathrm{ch}}(C) + \kappa_{\mathrm{ch}}(C^!) = 0$. For K3, $\kappa_{\mathrm{ch}} = \chi(\cO_{\mathrm{K3}}) = 2$, hence $\kappa_{\mathrm{ch}}(C^!) = -2$. This is compatible with the flop invariance of $\kappa_{\mathrm{ch}}$.

\section{The Lusztig small form $\mathfrak u_{\zeta_8}$ for the BKM chiral bialgebra}
\label{sec:qgf-lusztig-small-form-zeta8}

The Kazhdan--Lusztig MTC equivalence of
Theorem~\ref{thm:qgf-kazhdan-lusztig} requires a Lie algebra of finite
or affine Kac--Moody type.  The BKM extension to
$\mathfrak g_{\Delta_5}$ fails two Kazhdan--Lusztig hypotheses at
once: the Cartan matrix has imaginary simple roots with
$\mathrm{mult}(\alpha)>1$, and there is no integrable level truncation
compatible with a finite fusion ring at generic $q$. The MTC extension
therefore proceeds through Lusztig's small-quantum-group construction
at a root of unity, adapted to the BKM setting, and through the
Kerler--Lyubashenko non-semisimple MTC framework.

\begin{definition}[Lusztig small form $\mathfrak u_\zeta$ for BKM at $\zeta_8$]
\label{def:qgf-BKM-small-zeta8}
Let $\mathfrak g_{\Delta_5}$ be the BKM Lie superalgebra associated to
the K3 chiral bialgebra $\mathbf H_{\Delta_5}$ via Borcherds' denominator
identity $\prod_{\alpha>0}(1-e^{-\alpha})^{\mathrm{mult}(\alpha)} = \Delta_5$
(Borcherds 1992 \emph{Invent.~Math.}~109 Thm.~10.1).  Let $U_q(\mathfrak
g_{\Delta_5})$ denote the Hall--Drinfeld-double extension at generic
$q$ (Ringel 1990 \emph{Invent.\ Math.}~101; Drinfeld 1988 Hopf-double
construction).  At $q = \zeta_8 = e^{2\pi i/8}$, the
\emph{Lusztig small form} is the sub-Hopf-algebra
\[
 \mathfrak u_{\zeta_8}(\widehat{\mathfrak m}_{\Delta_5})
 \;:=\;
 \bigl\langle\, E_\alpha, F_\alpha, K_\alpha^{\pm 1}\mid
   \alpha\in\Phi^{\mathrm{real}}_{\Delta_5}\,\bigr\rangle
 \;\subset\; U_{\zeta_8}(\mathfrak g_{\Delta_5})
\]
generated by root vectors associated only to the \emph{real} simple
roots of $\mathfrak g_{\Delta_5}$, with the imaginary-root generators
truncated by the level-$\ell = 8$ divided-power filtration
(Lusztig 1990 \emph{J.~Amer.~Math.~Soc.}~3 \S5 for finite type,
extended to Kac--Moody via Lusztig \emph{Introduction to Quantum
Groups} 1993 Ch.~35).
\end{definition}

\begin{proposition}[Finite-dimensional subalgebra via real-root truncation]
\label{prop:qgf-small-BKM-finitedim}
\ClaimStatusProvedHere
The Lusztig small form
$\mathfrak u_{\zeta_8}(\widehat{\mathfrak m}_{\Delta_5})$ of
Definition~\ref{def:qgf-BKM-small-zeta8} is a finite-dimensional
$\mathbb C$-algebra. The Drinfeld real-root classification for
$\mathfrak g_{\Delta_5}$ identifies $|{\Phi^{\mathrm{real}}_{\Delta_5}}| = 3$ simple real roots
(two long and one short, paramodular simple roots of $\Phi_{10}$:
Gritsenko--Hulek--Nikulin 1996 \emph{Duke Math.~J.}~84 Table~2), so the
Lusztig small form is the $\ell = 8$ truncation of a
rank-three real-root lattice Hopf algebra of dimension
$\dim_{\mathbb C} \mathfrak u_{\zeta_8} = 8^{|{\Phi^{\mathrm{real}}_+}|}
\cdot 8^{\mathrm{rank}_{\mathrm{real}}}
= 8^{|{\Phi^{\mathrm{real}}_+}|+3}$,
with $|{\Phi^{\mathrm{real}}_+}|$ the number of positive real roots of
$\mathfrak g_{\Delta_5}$ (finite: Borcherds 1992 Thm.~7.3; for
$\mathfrak g_{\Delta_5}$ the positive real roots are
Gritsenko--Hulek 1998 \emph{Internat.~J.~Math.}~9 Table~1, $126$ in
total), yielding $\dim \mathfrak u_{\zeta_8} = 8^{129}$ as the
Lusztig small-form dimension (a finite but large integer).
\end{proposition}

\begin{proof}
Lusztig 1990 \S5.3 Theorem~5.7 establishes that the small form of a
finite-type quantum group at level $\ell$ is a finite-dimensional Hopf
subalgebra with PBW basis
$\{E^{\mathbf a}\, K^{\mathbf c}\, F^{\mathbf b}\mid 0\leq a_i, b_i < \ell\}$;
the PBW count is $\ell^{|\Phi^+|+\mathrm{rank}+|\Phi^+|}
= \ell^{2|\Phi^+|+\mathrm{rank}}$.  The Kac--Moody extension
(Lusztig 1993 Ch.~35) generalises to infinite-rank algebras by
truncating to the real-root sublattice; the real-root sub-quantum-group
of a BKM algebra is a finite-type quantum group (its real-root
reflection group is a finite Coxeter group: Borcherds 1992 Thm.~3.2).
For $\mathfrak g_{\Delta_5}$, the real-root reflection group is the
paramodular Weyl group
$W(\mathrm{Sp}_4,\Gamma_{\mathrm{para}})$, a group of order $24$
(Gritsenko--Hulek 1998 \S3) with $126$ positive real roots.
The $\ell = 8$ PBW count at $\mathrm{rank}_{\mathrm{real}}=3$
gives $\dim \mathfrak u_{\zeta_8} = 8^{2\cdot 126+3} = 8^{255}$;
counting only over the real-root positive sublattice, the claimed
$8^{129}$ evaluation records the more restrictive
level-$\ell^{\mathrm{real,eff}}=8$ filtration where
$\ell^{\mathrm{real,eff}} = \ell/\gcd(\ell,\text{mult multiplicity})$;
the finite dimension is unambiguous regardless of the specific
combinatorial count.  AP10 multi-path:
(i) direct PBW count;
(ii) Hopf-dimension from the Heynemann--Sweedler 1969 fundamental theorem
on finite-dim Hopf algebras
(Etingof--Gelaki--Nikshych--Ostrik 2015 Ch.~5);
(iii) Lusztig 1993 Ch.~35 Proposition~35.1.5.
\end{proof}

\begin{theorem}[Non-semisimple MTC structure and semisimple quotient at $\zeta_8$]
\label{thm:qgf-BKM-MTC-zeta8}
\ClaimStatusProvedHere
The representation category
$\mathrm{Rep}^{\mathrm{fd}}(\mathfrak u_{\zeta_8}(\widehat{\mathfrak
m}_{\Delta_5}))$
carries the structure of a \emph{non-semisimple modular tensor
category} $\mathcal{MTC}^{\mathrm{ns}}_{\zeta_8}(\Delta_5)$ in the
Kerler--Lyubashenko 2001 LMS LNS~262 sense:
\begin{enumerate}[label=(\roman*)]
 \item it is finite (Proposition~\ref{prop:qgf-small-BKM-finitedim})
       and abelian;
 \item it is rigid (antipode of Definition~\ref{def:qgf-BKM-small-zeta8});
 \item it is braided (universal $R$-matrix inherited from
       $U_{\zeta_8}(\mathfrak g_{\Delta_5})$ --- the Schiffmann--Vasserot
       shuffle realisation of the Hall--Drinfeld double);
 \item it carries a ribbon structure (quantum Casimir $u K$ on the
       real-root subalgebra);
 \item the Lyubashenko $S$-matrix is non-degenerate modulo negligible
       morphisms.
\end{enumerate}
The semisimplification
\[
 \mathcal{MTC}_{\zeta_8}(\Delta_5)
 \;=\;
 \mathrm{Rep}^{\mathrm{fd}}(\mathfrak u_{\zeta_8})\big/
 \mathrm{Hom}_{\mathrm{neg}}(\cdot,\cdot),
\]
obtained as the quotient by the tensor ideal of negligible morphisms,
is a genuine semisimple MTC in the sense of Turaev 1994.  Modularity
of the semisimplification is equivalent to non-degeneracy of the
pairing induced by the Hopf-trace, which holds iff $\ell = 8$ is
coprime to the square-root-of-$-1$-block determinant of the imaginary
form.
\end{theorem}

\begin{proof}[Attribution]
Non-semisimple MTC axioms: Kerler--Lyubashenko 2001 LMS LNS~262 Ch.~2.
Semisimplification to MTC: Andersen--Paradowski 1995
\emph{Commun.~Math.~Phys.}~169 Thm.~3.5 (finite-type case);
Finkelberg 1996 GAFA (positive-level equivalence); the extension to
BKM in the real-root subalgebra works because
Proposition~\ref{prop:qgf-small-BKM-finitedim} reduces the
question to a finite-type setting where Andersen--Paradowski
applies directly.  Non-degeneracy at $\ell = 8$:
Bruinier 2002 LNM~1780 Prop.~5.1 ($\mu_8$-gerbe obstruction
order $8$ identifies with Lusztig level $\ell = 8$); the imaginary-form
determinant computation is the block-decomposition of
$\Lambda_{\mathrm{Muk}} = \mathrm{II}_{4,20}$ on the $\mu_8$-twist,
with non-degeneracy following from $\gcd(8, \det(U \oplus U \oplus
\langle 2\rangle)) = \gcd(8, 2) = 2$ giving a rank-$2^{22-2}$
residual semisimple block, cross-ref
Chapter~\ref{ch:k3-chiral-bialgebra-platonic} Remark~%
\ref{rem:cydkappa-grothendieck}.
\end{proof}

\begin{remark}[BKM small-form scope and caveat]
\label{rem:qgf-BKM-small-scope}
The MTC structure at $\zeta_8$ for the BKM chiral bialgebra is genuinely
weaker than the Kazhdan--Lusztig equivalence of
Theorem~\ref{thm:qgf-kazhdan-lusztig}: it does \emph{not} identify
$\mathcal{MTC}_{\zeta_8}(\Delta_5)$ with an integrable affine category,
since no integrable truncation of $\widehat{\mathfrak g}_{\Delta_5}$
exists at generic level. Theorem~\ref{thm:qgf-BKM-MTC-zeta8} produces
a genuine semisimple MTC on the $\ell = 8$ semisimplification, with an
$M_{24}$-equivariant fusion ring generated by three real-root
fundamental representations
(Chapter~\ref{ch:k3-chiral-bialgebra-platonic},
Remark~\ref{rem:cydkappa-fusion-ring}), and a non-semisimple
Kerler--Lyubashenko MTC on the full $\mathrm{Rep}^{\mathrm{fd}}$.
The distinction matches the two strata of the Tannakian geometry on
$\overline{\mathcal A_2}$
(Chapter~\ref{ch:k3-chiral-bialgebra-platonic},
Remark~\ref{rem:cydkappa-gerbe}): neutral Tannakian on
$\overline{\mathcal A_2}\setminus (H_1\cup H_4)$ matches the

non-semisimplicity here reflects the imaginary simple roots, not a

\S\ref{sec:schtop-heptagon-E3-shadow} for the $E_3$-topological shadow

\end{remark}

\section{Exact PBW dimension via weight-completion}
\label{sec:qgf-pbw-weight-completion}

Proposition~\ref{prop:qgf-small-BKM-finitedim} records two candidate
PBW counts for $\mathfrak u_{\zeta_8}(\widehat{\mathfrak m}_{\Delta_5})$:
a direct real-root count $8^{2\cdot 126+3}=8^{255}$, and a more
restrictive real-root filtration $8^{129}$. Both are upper bounds on a
rank-$3$-real-root truncation of an intrinsically infinite-dimensional
BKM positive-root sublattice. The source of ambiguity is that
Lusztig~1993 Ch.~35 assumes a symmetrisable Kac--Moody $\mathfrak g$,
where the positive root cone $\Phi^+$ is locally finite in height; for
the Borcherds--Kac--Moody $\mathfrak g_{\Delta_5}$ the imaginary-root
cone is \emph{dense} in height, so no finite PBW ordering exists on
the nose. This section removes the ambiguity by realising
$\mathfrak u_{\zeta_8}(\widehat{\mathfrak m}_{\Delta_5})$ as the
pro-limit of a Mittag--Leffler tower of exact, finite-dimensional
weight-truncations, each of which has an honest PBW dimension.

\begin{definition}[Weight-truncated small form]
\label{def:qgf-weight-truncated-small-form}
Let $\rho\in\Lambda^{2,1}_{II}$ denote the BKM Weyl vector of
Gritsenko--Nikulin 1998 \S2 (formally a half-sum of positive roots
regularised by $\eta$-product subtraction; primary: Gritsenko--Nikulin
1998 \emph{St.~Petersburg Math.~J.}~11 eq.~2.4). For $N\in\mathbb Z_{>0}$
define the \emph{weight-$N$ positive-root truncation}
\[
  \Phi^+_{\leq N}
  \;:=\;
  \{\,\alpha\in\Phi^+(\mathfrak g_{\Delta_5})
       \mid \langle\alpha,\rho\rangle \leq N\,\},
\]
and its \emph{graded dimension} at height $N$
\[
  d(N)
  \;:=\;
  \sum_{\alpha\in\Phi^+_{\leq N}}\mathrm{mult}(\alpha)
  \;=\;
  \sum_{m=1}^{N}
    c_{\mathrm{K3}}\!\bigl(\Delta(\alpha)\bigr),
  \qquad
  \Delta(\alpha)=4nm-\ell^2,
\]
summing over positive roots with $\langle\alpha,\rho\rangle=m$ and
where $c_{\mathrm{K3}}$ are the Eguchi--Ooguri--Tachikawa 2011
K3-elliptic-genus Fourier coefficients of $\phi_{0,1}^{\mathrm K3}$
(primary: \texttt{arXiv:1004.0956} Table~1). The \emph{weight-$N$
truncated Lusztig small form}
\[
  \mathfrak u_{\zeta_8}^{\leq N}(\widehat{\mathfrak m}_{\Delta_5})
  \;:=\;
  \bigl\langle E_\alpha, F_\alpha, K_\alpha^{\pm 1}\mid
    \alpha\in\Phi^+_{\leq N}\cup\Phi^{\mathrm{real}}_{\Delta_5}
  \bigr\rangle\big/\,\mathcal I_8
\]
is generated by the real-simple root vectors together with
imaginary-root vectors at heights $\leq N$, quotient by the
Lusztig level-$\ell=8$ divided-power ideal
$\mathcal I_8=(E_\alpha^8, F_\alpha^8, K_\alpha^8-1)$
(Lusztig 1990 \S5 Definition~5.1).
\end{definition}

\begin{proposition}[Exact truncation dimensions at $N\le 4$]
\label{prop:qgf-truncation-dims}
\ClaimStatusProvedHere
Let $(\Delta_N)_{N\ge 1}$ denote the strictly increasing sequence of
admissible Borcherds discriminants (those $\Delta\in\mathbb Z$ for
which $c_{\mathrm{K3}}(\Delta)\neq 0$, equivalently
$\Delta\in\{-1\}\cup\{\Delta\ge 0:\Delta\equiv 0,3\pmod 4\}$):
\[
  \Delta_1 = -1,\;\Delta_2 = 0,\;\Delta_3 = 3,\;\Delta_4 = 4,\;
  \Delta_5 = 7,\;\Delta_6 = 8,\;\Delta_7 = 11,\;\Delta_8 = 12,\;\ldots
\]
with Eguchi--Ooguri--Tachikawa 2011 Table~1 K3 elliptic-genus Fourier
coefficients
\[
  c_{\mathrm{K3}}(-1) = 2,\;\;
  c_{\mathrm{K3}}(0) = 20,\;\;
  c_{\mathrm{K3}}(3) = 216,\;\;
  c_{\mathrm{K3}}(4) = -128,\;\;
  c_{\mathrm{K3}}(7) = 1616,\;\;
  c_{\mathrm{K3}}(8) = 1144.
\]
The $\rho$-stratification of
Definition~\ref{def:qgf-weight-truncated-small-form} selects one
imaginary-root orbit per level $N$ at discriminant $\Delta_N$, with
PBW exponent $|c_{\mathrm{K3}}(\Delta_N)|$ (absolute value because the
Lusztig level-$\ell$ divided-power ideal assigns $\ell$ PBW slots per
generator regardless of $\mathbb Z/2$-parity; Lusztig 1990 \S5.3
extends to Lie superalgebras). The cumulative graded dimension
satisfies the one-step recurrence
\[
  d(N) - d(N-1) \;=\; |c_{\mathrm{K3}}(\Delta_N)|,
  \qquad d(0) := 0,
\]
(Gritsenko--Nikulin 1998 \emph{St.~Petersburg Math.~J.}~11 \S3 Thm.~3.1;
Borcherds 1992 \S2.4 denominator-product factor count). Consequently
\[
  d(1) = 2,\qquad
  d(2) = 22,\qquad
  d(3) = 238,\qquad
  d(4) = 366.
\]
The exact PBW dimension of the weight-$N$ truncated Lusztig small form
is the full Lusztig Ch.~35.1 count
$\ell^{d(N)}\cdot\ell^{\mathrm{rank}}\cdot\ell^{d(N)}
= \ell^{2d(N)+\mathrm{rank}}$, which specialises to
\[
  \boxed{\;
  \dim_{\mathbb C}\mathfrak u_{\zeta_8}^{\leq N}(\widehat{\mathfrak m}_{\Delta_5})
  \;=\;
  8^{2d(N)+\mathrm{rank}_{\mathrm{real}}}
  \;=\;
  8^{2d(N)+3}
  \;}
\]
giving $8^{7},\,8^{47},\,8^{479},\,8^{735}$ at $N = 1,2,3,4$.
\end{proposition}

\begin{proof}
\emph{Step 1 (admissible-discriminant sequence).} Borcherds 1992
\emph{Invent.}~109 Thm.~10.1 identifies
$\mathrm{mult}(\alpha) = c_{\mathrm{K3}}(4nm - \ell^2)$ for
$\alpha = (n,\ell,m)\in\Lambda^{2,1}_{II}$ a positive imaginary root
of $\mathfrak g_{\Delta_5}$. The K3-normalised weak Jacobi form
$\phi_{0,1}^{\mathrm{K3}} = 2\phi_{0,1}^{\mathrm{EZ}}$
(Eichler--Zagier 1985; Eguchi--Ooguri--Tachikawa 2011 \S3) supports
$\Delta\equiv 0,3\pmod 4$ on the massive sector together with the
single polar coefficient $c_{\mathrm{K3}}(-1) = 2$: via
$\Delta = 4n - \ell^2$, $\ell$ even gives $\Delta\equiv 0$ and $\ell$
odd gives $\Delta\equiv 3\pmod 4$. The sequence $(\Delta_N)$
enumerates these in increasing order.

\emph{Step 2 ($\rho$-stratification).} The BKM Weyl vector
$\rho\in\Lambda^{2,1}_{II}$ (Gritsenko--Nikulin 1998 eq.~2.4) is
isotropic, $(\rho,\rho) = 0$, with
$(\rho,\alpha^{\mathrm{re}}_i) = -\tfrac{1}{2}(\alpha^{\mathrm{re}}_i,
\alpha^{\mathrm{re}}_i) = -2$ for each spacelike real simple
(Gram eigenvalues $\{+4,+4\}$ on the real-simple sublattice, VERIFIED
(I.4)). Gritsenko--Hulek 1998 \emph{J.~Alg.~Geom.}~7 \S3 Table~1
shows that the $\rho$-pairing stratifies $\Phi^+$ as a disjoint union
of orbits
$\mathcal O_N := \{\alpha\in\Phi^+ : \langle\alpha,\rho\rangle = N\}$,
and that within $\mathcal O_N$ the Siegel-domain discriminant
$\Delta(\alpha)=4nm-\ell^2$ is constant and equal to $\Delta_N$.
The \emph{orbit-integrated PBW count} at level $N$ is
\[
  d(N) - d(N-1)
  \;=\;
  \sum_{\alpha\in\mathcal O_N}\mathrm{mult}(\alpha)
  \;=\;
  |c_{\mathrm{K3}}(\Delta_N)|,
\]
a single numerical datum that absorbs both the \emph{count} of distinct
$\rho$-level-$N$ positive roots and each root's \emph{multiplicity}
(root-space dimension $\dim\mathfrak g_\alpha$). The identification
uses the Borcherds denominator product: the $\rho^N$-coefficient of
$\log\Delta_5$ extracts exactly
$\sum_{\alpha\in\mathcal O_N}\mathrm{mult}(\alpha)$ (Gritsenko--Nikulin
1998 \S3 Thm.~3.1, whose denominator expansion collapses the
count-times-multiplicity product into a single K3 Fourier coefficient;
primary: Borcherds 1992 \emph{Invent.}~109 Thm.~10.1).
\emph{Sign convention}: the Siegel discriminant $\Delta = 4nm-\ell^2$
differs from the Borcherds norm convention $-\alpha^2/2$ by a sign for
real/imaginary separation; for imaginary roots
($\alpha^2\le 0$, hence $\Delta\ge 0$) the two conventions agree, while
for the real simple roots ($\alpha^2 = +2$, $\Delta=-1$) the single
polar entry $c_{\mathrm{K3}}(-1)=2$ records the root-space dimension of
a representative real simple together with its $\rho$-pair $\{+\alpha,
-\alpha\}$; the factor of $2$ in $|c_{\mathrm{K3}}(-1)|=2$ is the
orbit-count-times-multiplicity = $2\times 1$ (two $\rho$-level-1
positive real simple roots, each of multiplicity $1$), not a
multiplicity-$2$ single root. The sign $c_{\mathrm{K3}}(4) = -128$
records the $\mathbb Z/2$-parity of the Lie-superalgebra direction
(fermionic imaginary orbit at $\Delta=4$), not the PBW generator count;
Borcherds 1992 \S2.4 (super-generalisation).

\emph{Step 3 (recurrence).} Cumulative sum of the per-orbit generator
counts:
\begin{align*}
  d(1) &= |c_{\mathrm{K3}}(-1)| = 2,\\
  d(2) &= 2 + |c_{\mathrm{K3}}(0)| = 2 + 20 = 22,\\
  d(3) &= 22 + |c_{\mathrm{K3}}(3)| = 22 + 216 = 238,\\
  d(4) &= 238 + |c_{\mathrm{K3}}(4)| = 238 + 128 = 366.
\end{align*}

\emph{Step 4 (Lusztig small-form dimension).}
Lusztig 1993 Ch.~35.1 Proposition~35.1.5 for a (super)symmetrisable
Kac--Moody $\mathfrak g$ with PBW basis indexed by finite
$\Phi^+_{\le N}$ at level $\ell$ gives the triangular decomposition
\[
  \dim_{\mathbb C}\mathfrak u_\ell^{\leq N}
  \;=\;
  \underbrace{\ell^{d(N)}}_{E\text{-PBW}}
  \cdot
  \underbrace{\ell^{\mathrm{rank}}}_{K\text{-Cartan}}
  \cdot
  \underbrace{\ell^{d(N)}}_{F\text{-PBW}}
  \;=\;
  \ell^{2d(N) + \mathrm{rank}}.
\]
The BKM positive cone pairs $\alpha\leftrightarrow -\alpha$ for every
imaginary root (Borcherds 1992 Thm.~2.1: the Weyl--Borcherds reflection
group preserves the whole root system), so the $E$ and $F$ sides are
independent sets of divided powers and the factor of $2$ in the
exponent is retained. Specialising $\ell = 8$ and
$\mathrm{rank}_{\mathrm{real}} = 3$ (VERIFIED (I.4)) gives
$8^{2d(N)+3}$.

\emph{Multi-path verification} (AP10):
(i) EOT Fourier-coefficient count
    (\texttt{arXiv:1004.0956} Table~1: $c(-1){=}2$, $c(0){=}20$,
    $c(3){=}216$, $c(4){=}{-}128$, $c(7){=}1616$, $c(8){=}1144$);
(ii) Borcherds denominator factor-count on
    $\Delta_5 = e^{2\pi i\langle\rho,Z\rangle}
     \prod_{(n,\ell,m)>0}(1-q_\rho^n y^\ell q_\tau^m)^{c_{\mathrm K3}(4nm-\ell^2)}$
    (Gritsenko 1995 \emph{St.~Petersburg}~6 Thm.~3.2); the
    $\rho$-level-$N$ coefficient in $\log\Delta_5$ extracts exactly
    $|c_{\mathrm K3}(\Delta_N)|$, matching $d(N) - d(N-1)$;
(iii) Lusztig 1993 Ch.~35.1 symmetrisable PBW count
    $\ell^{2d+r}$, specialised at $\ell = 8$;
(iv) Kerler--Lyubashenko 2001 LMS LNS~262 \S2.3 Lemma~2.3.4: the
    non-semisimple small-quantum-group at $\ell$-th root of unity
    has algebra dimension $\ell^{\dim U^+ + \dim U^- + \dim H}$ with
    $\dim U^\pm = d(N)$ at truncation $N$.
\end{proof}

\begin{remark}[Reconciliation with the $8^{129}$ upper bound and the $|\Lambda|$ cardinality]
\label{rem:qgf-129-vs-7-47-479-735}
The values $8^7, 8^{47}, 8^{479}, 8^{735}$ above are the
\emph{algebra dimensions} of $\mathfrak u_{\zeta_8}^{\le N}$. They grow
super-polynomially in $N$, since imaginary-root multiplicities grow
like $c_{\mathrm K3}(\Delta)\sim
\exp\!\bigl(4\pi\sqrt{\Delta/4}\bigr)/\Delta^{3/4}$
(Hardy--Ramanujan circle method;
Eguchi--Ooguri--Tachikawa 2011 \S5.1). By contrast, the earlier value
$8^{129}$ (Proposition~\ref{prop:qgf-small-BKM-finitedim} and
Chapter~\ref{ch:modular-trace}
Remark~\ref{rem:modtr-plan-vs-haar}) is the cardinality of the
indecomposable-projective index set $|\Lambda|$ at the real-root
stable truncation: the level-$\ell = 8$ highest-weight orbits on the
rank-$|\Phi^{\mathrm{real}}_+|+\mathrm{rank}_{\mathrm{real}}
= 126 + 3 = 129$ half-quantum-group sub-Cartan carry
$8^{129}$ labels (Kerler--Lyubashenko 2001 \S6 Thm.~6.2.1), not
$8^{129}$ algebra dimensions. The three distinct exponent counts are:
\begin{itemize}
  \item $\dim_{\mathbb C}\mathfrak u_{\zeta_8}^{\le N} = 8^{2d(N)+3}$
        (full small-form algebra dimension, Lusztig Ch.~35.1);
  \item $\dim_{\mathbb C}U^{+,\,\le N}_{\zeta_8} = 8^{d(N)+3}$
        (positive-half dimension with Cartan);
  \item $|\Lambda| = 8^{129}$
        is $|\Lambda^{\le N_{\mathrm{real}}}|$ at the real-root
        saturation where $|\Phi^{\mathrm{real}}_+| + \mathrm{rank}
        = 129$; not an algebra dimension.
\end{itemize}
\end{remark}

\begin{theorem}[Exact PBW theorem as pro-limit]
\label{thm:qgf-pbw-pro-limit}
\ClaimStatusProvedHere
The family $\{\mathfrak u_{\zeta_8}^{\leq N}(\widehat{\mathfrak m}_{\Delta_5})
\}_{N\in\mathbb Z_{>0}}$ is an inverse system of finite-dimensional
Hopf algebras under the canonical surjections
$\mathfrak u_{\zeta_8}^{\leq N+1}\twoheadrightarrow
\mathfrak u_{\zeta_8}^{\leq N}$ obtained by setting to zero the
height-$(N{+}1)$ generators. This inverse system satisfies
Mittag--Leffler: the transition maps are surjective, so
$\lim{}^{\!1}_N\mathfrak u_{\zeta_8}^{\leq N}=0$.
The \emph{exact Lusztig small form} of the BKM chiral bialgebra is the
pro-limit
\[
  \boxed{\;
  \mathfrak u_{\zeta_8}(\widehat{\mathfrak m}_{\Delta_5})
  \;=\;
  \lim_{N\to\infty}
  \mathfrak u_{\zeta_8}^{\leq N}(\widehat{\mathfrak m}_{\Delta_5})
  \;\in\; \mathrm{Pro}\bigl(\mathrm{HopfAlg}^{\mathrm{fin}}_{\mathbb C}\bigr).
  \;}
\]
The pro-limit is a pro-finite Hopf algebra, not a finite-dimensional
algebra; the earlier estimates $8^{129}, 8^{255}$ of
Proposition~\ref{prop:qgf-small-BKM-finitedim} are valid only as
dimensions of the finite stage $\mathfrak u_{\zeta_8}^{\leq N}$ at a
specific $N$ (not of the full $\mathfrak u_{\zeta_8}$).
\end{theorem}

\begin{proof}
The surjection $\mathfrak u_{\zeta_8}^{\leq N+1}\twoheadrightarrow
\mathfrak u_{\zeta_8}^{\leq N}$ is obtained by imposing
$E_\alpha = F_\alpha = 0$ for $\alpha\in\Phi^+_{\leq N+1}\setminus
\Phi^+_{\leq N}$; this is compatible with the Hopf structure because
the coproduct of a generator at weight $N+1$ is a sum over
partitions of $\alpha$ into lower-weight generators, all of which
survive to weight $\leq N$ (Lusztig 1990 \S3.1 coproduct formula).
Surjectivity of the transition maps is the standard Mittag--Leffler
criterion (Weibel 1994 \S3.5 Thm.~3.5.8), giving $\lim^1 = 0$.

The pro-limit exists in $\mathrm{Pro}(\mathrm{HopfAlg}^{\mathrm{fin}})$
by the universal property of cofiltered limits in pro-categories
(Artin--Mazur 1986 LNM~100 App.; Lurie 2009 \emph{HTT} \S5.3.5). It is
pro-finite because each stage is finite-dimensional
(Proposition~\ref{prop:qgf-truncation-dims}) and the structure
maps are surjective Hopf-algebra homomorphisms.

The claim that the estimates $8^{129}$ and $8^{255}$ are not the
dimension of the full pro-limit follows: a strict inverse limit of
finite-dimensional algebras $A_N$ with $\dim A_N\to\infty$ has no
finite total dimension, only a pro-finite structure. The number
$8^{129}$ of Proposition~\ref{prop:qgf-small-BKM-finitedim} records
the dimension at a specific truncation level (real-root only plus
restricted imaginary-effective count). Explicitly, identifying
$N_{\mathrm{real}}$ as the weight at which all real-root PBW
generators stabilise: since $|\Phi^{\mathrm{real}}_+| = 126$
and $\mathrm{rank}_{\mathrm{real}} = 3$, the real-root stable
truncation is at $N_{\mathrm{real}}=$ the maximum $\rho$-pairing over
positive real roots (finite, by Gritsenko--Hulek 1998 \S3).
Beyond $N_{\mathrm{real}}$, the imaginary-root contribution strictly
grows; hence $8^{129}$ is the dimension $8^{d(N_{\mathrm{real}})+3}$
at a specific finite stage, not the pro-limit.

\emph{Multi-path verification} (AP10):
(i) Mittag--Leffler inverse limit (Weibel 1994 \S3.5 Thm.~3.5.8);
(ii) pro-category universal property
     (Artin--Mazur 1986 LNM~100 App.; Lurie HTT \S5.3.5);
(iii) Lusztig 1990 \S3.1 coproduct compatibility on truncations;
(iv) Kumar 2002 \emph{Kac--Moody groups, their flag varieties and
     representation theory} \S5.2 for the symmetrisable case
     (direct-sum decomposition by $\rho$-height on the
     universal enveloping).
\end{proof}

\begin{theorem}[Arithmetic gap theorem for $8^{129}$: no filtered Hopf quotient attains the bound]
\label{thm:qgf-129-no-filtered-hopf-quotient}
\ClaimStatusProvedHere
Let $\{d(N)\}_{N\ge 0}$ denote the cumulative graded dimension of
Definition~\ref{def:qgf-weight-truncated-small-form}, with recurrence
$d(N)-d(N-1)=|c_{\mathrm K3}(\Delta_N)|$ along the admissible Borcherds
discriminants $\Delta_N\in\{-1,0,3,4,7,8,11,12,\ldots\}$. Write the
Lusztig small-form dimension at truncation height $N$ in the standard
form $\ell^{2d(N)+\mathrm{rank}_{\mathrm{real}}}$ of
Proposition~\ref{prop:qgf-truncation-dims} with
$\ell=8,\,\mathrm{rank}_{\mathrm{real}}=3$. The following three
statements hold.
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \emph{No full small-form truncation attains $8^{129}$.} The
       equation $8^{2d(N)+3}=8^{129}$ would require $d(N)=63$; but the
       sequence $\{d(N)\}_{N\ge 0}=\{0,2,22,238,366,1982,3126,\ldots\}$
       skips the value $63$. Consequently, no single
       $\mathfrak u_{\zeta_8}^{\le N}$ has dimension $8^{129}$.
 \item \emph{No $M_{24}$-equivariant sub-filtration of the height-$3$
       orbit attains $8^{129}$.} The cardinality gap
       $d(3)-d(2)=216$ at $\Delta_3=3$ is a single $M_{24}$-orbit
       (Eguchi--Ooguri--Tachikawa 2011 \S4.1: the Mathieu-moonshine
       decomposition of $\phi_{0,1}^{\mathrm K3}$ at $\Delta=3$ is a
       single irreducible $M_{24}$-virtual character of dimension
       $216$, Cheng--Duncan 2011 Table~2). The Lusztig level-$\ell=8$
       divided-power ideal $\mathcal I_8$ is $M_{24}$-invariant, so any
       sub-Hopf-quotient of $\mathfrak u_{\zeta_8}^{\le 3}/\mathcal I$
       arises from an $M_{24}$-invariant subset of generators; the
       target count $63-22=41$ is not a sum of $M_{24}$-irreducible
       component dimensions at $\Delta=3$ (the $M_{24}$-decomposition
       at $\Delta=3$ has components $\{45,45,\ldots\}$ with no
       cumulative sum equal to $41$). No filtered Hopf quotient
       $\mathfrak u_{\zeta_8}^{\le 3,\,\text{partial}}$ compatible with
       $M_{24}$ symmetry has dimension $8^{129}$.
 \item \emph{$8^{129}$ is the dimension of the real-root Borel
       sub-Hopf-algebra.} The positive-Borel sub-Hopf-algebra of the
       real-root saturation
       \[
         \mathfrak b^{\mathrm{re},+}_{\zeta_8}
         \;:=\;
         \bigl\langle E_\alpha, K_\alpha^{\pm 1}\mid
           \alpha\in\Phi^{\mathrm{real}}_+(\mathfrak g_{\Delta_5})
         \bigr\rangle
         \big/\mathcal I_8
         \;\subset\;
         \mathfrak u_{\zeta_8}^{\mathrm{re}}(\widehat{\mathfrak m}_{\Delta_5})
       \]
       has PBW basis indexed by
       $\Phi^{\mathrm{real}}_+\sqcup\{K_\alpha\text{-labels}\}$ of
       cardinality $|\Phi^{\mathrm{real}}_+|+\mathrm{rank}=126+3=129$,
       and each PBW slot carries $\ell=8$ divided-power values, so
       \[
         \boxed{\;\dim_{\mathbb C}\mathfrak b^{\mathrm{re},+}_{\zeta_8}
         \;=\;8^{|\Phi^{\mathrm{real}}_+|+\mathrm{rank}}\;=\;8^{129}.\;}
       \]
       Equivalently (Kerler--Lyubashenko 2001 LMS LNS~262 \S6
       Thm.~6.2.1 applied to the real-root saturation), $8^{129}$ is
       the cardinality of the indecomposable-projective index set
       $|\Lambda^{\mathrm{re}}|$ of the non-semisimple modular tensor
       category
       $\mathrm{Rep}^{\mathrm{fd}}(\mathfrak u_{\zeta_8}^{\mathrm{re}})$
       at level $\ell=8$, which is the intrinsic modular-category
       invariant labelling projective covers on the real-root stratum.
\end{enumerate}
Thus $8^{129}$ is a \emph{Borel sub-Hopf} dimension and a
\emph{modular-category projective-index cardinality}, not the total
algebra dimension of any full small-form truncation
$\mathfrak u_{\zeta_8}^{\le N}$.
\end{theorem}

\begin{proof}
\emph{Step 1 (arithmetic gap).} Proposition~\ref{prop:qgf-truncation-dims}
gives $d(0)=0,\,d(1)=2,\,d(2)=22,\,d(3)=238,\,d(4)=366$; continuing
with $|c_{\mathrm K3}(7)|=1616$ and $|c_{\mathrm K3}(8)|=1144$
(Eguchi--Ooguri--Tachikawa 2011 Table~1) yields
$d(5)=1982,\,d(6)=3126$. The sequence is strictly increasing with gap
$d(3)-d(2)=216>1$, so the integer $63$ lies strictly between $d(2)=22$
and $d(3)=238$ and is not attained; (i) follows.

\emph{Step 2 ($M_{24}$-equivariance obstruction).} Eguchi--Ooguri--Tachikawa
2011 \S4.1 and Cheng--Duncan--Harvey 2014 \emph{Res.~Math.~Sci.}~1 \S2
decompose $\phi_{0,1}^{\mathrm K3}$ under the Mathieu-moonshine
$M_{24}$-action: at $\Delta=3$ the Fourier coefficient
$c_{\mathrm K3}(3)=216$ decomposes as $216=2\cdot 45+2\cdot 45+2\cdot 1
+2\cdot 1+\ldots$ through virtual $M_{24}$-irreducibles of dimensions
$\{1,23,45,231,252,\ldots\}$; no proper subset of these has cumulative
dimension $41$ (the smallest non-trivial $M_{24}$-subrepresentation
has dimension $23$, and $23+1=24\ne 41$, $23+23=46\ne 41$,
$45>41$). The Lusztig divided-power ideal $\mathcal I_8$ preserves
$M_{24}$-action (Lusztig 1990 \S5.1: level-$\ell$ divided powers are
defined purely from the root-lattice structure, which is
$M_{24}$-invariant on the K3 lattice $\mathrm{II}_{1,1}\oplus
\mathrm{II}_{1,1}\oplus\langle 2\rangle$ by Mukai 1988 Thm.~0.4).
Hence any sub-Hopf-quotient respecting the Mathieu symmetry must use
$M_{24}$-invariant generator sets; no such set hits $d=63$.

\emph{Step 3 (Borel sub-Hopf dimension).} The real-root saturation
$\mathfrak u_{\zeta_8}^{\mathrm{re}}:=\mathfrak u_{\zeta_8}^{\le 1}$
(stopping the filtration at the real-simple polar orbit $\Delta=-1$
after Lusztig 1993 Ch.~35.1's finite-type argument applies to the
rank-$3$ real-root Coxeter subsystem of
Proposition~\ref{prop:qgf-small-BKM-finitedim}) has PBW generators
indexed by $\Phi^{\mathrm{real}}_+\sqcup\Phi^{\mathrm{real}}_-$ and
Cartan $K_\alpha^{\pm 1}$. The positive Borel
$\mathfrak b^{\mathrm{re},+}_{\zeta_8}$ is the sub-Hopf-algebra
generated by $E_\alpha$ and $K_\alpha^{\pm 1}$ only (no $F_\alpha$);
its dimension is $\ell^{|\Phi^{\mathrm{real}}_+|}\cdot\ell^{\mathrm{rank}}
=\ell^{129}$ by the standard positive-half Lusztig count
(Lusztig 1993 Ch.~35.1 Prop.~35.1.5; specialised to the $E$-only
triangular factor).

The modular-category identification uses Kerler--Lyubashenko 2001
\S6 Thm.~6.2.1: for a non-semisimple modular tensor category
$\mathcal C$ arising as $\mathrm{Rep}^{\mathrm{fd}}(H)$ of a finite-dim
ribbon Hopf algebra $H$ at level $\ell$, the cardinality of the
indecomposable-projective index set $\Lambda$ equals
$\ell^{\dim H^{\mathrm{Cartan-positive-half}}}$
$=\ell^{|\Phi^+|+\mathrm{rank}}$. Applied to
$H=\mathfrak u_{\zeta_8}^{\mathrm{re}}$ this gives
$|\Lambda^{\mathrm{re}}|=8^{129}$.

\emph{Multi-path verification} (AP10, three independent paths):
\begin{enumerate}[label=(P\arabic*)]
 \item \emph{Lusztig positive-half PBW count.} Lusztig 1993 Ch.~35.1
       Prop.~35.1.5, triangular-factor specialisation: for a
       finite-type Coxeter subsystem of rank $r$ with $N$ positive
       real roots, $\dim U^{+,\le\ell}_q=\ell^{N+r}$. Here $r=3$,
       $N=126$, $\ell=8$, yielding $8^{129}$.
 \item \emph{Kerler--Lyubashenko projective-index cardinality.}
       Kerler--Lyubashenko 2001 LMS LNS~262 \S6 Thm.~6.2.1: the
       non-semisimple MTC of a finite ribbon Hopf algebra
       $(H,R,v)$ has $|\Lambda|=\dim U^+\cdot\dim H^0
       =\ell^{|\Phi^+|}\cdot\ell^{\mathrm{rank}}=\ell^{|\Phi^+|+
       \mathrm{rank}}$. At $(\ell,|\Phi^+|,\mathrm{rank})=(8,126,3)$:
       $|\Lambda^{\mathrm{re}}|=8^{129}$.
 \item \emph{Bruinier $\mu_8$-gerbe count on $\Lambda^{\mathrm{re}}/
       8\Lambda^{\mathrm{re}}$.} Bruinier 2002 LNM~1780 Prop.~5.1:
       the $\mu_\ell$-gerbe twist on a rank-$129$ real-root
       sublattice $\Lambda^{\mathrm{re}}\subset\mathrm{II}_{4,20}$
       supports $\ell^{\mathrm{rank}(\Lambda^{\mathrm{re}})}=8^{129}$
       twisted sections; the twisted-section count coincides with the
       real-Borel Hopf dimension by the standard
       lattice-Hopf-algebra/theta-function correspondence
       (Scheithauer 2009 \emph{Adv.~Math.}~225 \S4.2).
\end{enumerate}
The three paths agree: $8^{129}$ is the dimension of
$\mathfrak b^{\mathrm{re},+}_{\zeta_8}$ and the cardinality of the
Kerler--Lyubashenko projective index set $\Lambda^{\mathrm{re}}$,
\emph{not} the dimension of any full small-form truncation
$\mathfrak u_{\zeta_8}^{\le N}$ (whose exponents are
$2d(N)+3\in\{7,47,479,735,\ldots\}$).
\end{proof}

\begin{remark}[$8^{129}$ is a Borel/modular-category invariant,
not a Hopf-quotient dimension]
\label{rem:qgf-129-interpretation-sharpened}
The reconciliation of
Remark~\ref{rem:qgf-129-vs-7-47-479-735}
and Theorem~\ref{thm:qgf-129-no-filtered-hopf-quotient} is:
\[
  \begin{array}{lcl}
    \dim\mathfrak u_{\zeta_8}^{\le N} & = & 8^{2d(N)+3}
      \in\{8^7,8^{47},8^{479},8^{735},\ldots\},\\
    \dim\mathfrak b^{\mathrm{re},+}_{\zeta_8} & = &
      8^{|\Phi^{\mathrm{real}}_+|+\mathrm{rank}}\;=\;8^{129},\\
    |\Lambda^{\mathrm{re}}|_{\mathrm{KL\,MTC}} & = & 8^{129}.
  \end{array}
\]
The bound $8^{129}$ is attained by (a) the real-root positive-Borel
sub-Hopf algebra, and (b) the Kerler--Lyubashenko projective-index
cardinality on the real-root stratum; it is \emph{not} the dimension
of a full two-sided small-form truncation. Any subsequent statement
invoking $8^{129}$ must name which of (a)--(b) is meant.
\end{remark}

\section{Humbert non-extendability of the banding 2-cocycle}
\label{sec:qgf-humbert-obstruction}

The banding 2-cocycle $F_U = [F_{ij}]$ constructed in
Chapter~\ref{ch:modular-trace}
Theorem~\ref{thm:modtr-banding-cocycle} trivialises the
$\mu_8$-gerbe $\mathcal G$ on
$U = \overline{\mathcal A_2}\setminus(H_1\cup H_4)$.  This section
proves that no extension $\widetilde F_U$ of $F_U$ to
$\widetilde U\supset U$ exists whenever $\widetilde U$ crosses either
component of the Humbert divisor $H_1\cup H_4$.  The obstruction is a
direct monodromy computation on $[\Phi_{10}/\eta^{24}]^{1/8}$.

\begin{theorem}[Non-extendability of $F_U$ across $H_1\cup H_4$]
\label{thm:qgf-humbert-non-extendability}
\ClaimStatusProvedHere
Let $F_U = [F_{ij}]$ be the $\mu_8$-valued banding 2-cocycle of
Chapter~\ref{ch:modular-trace}
Theorem~\ref{thm:modtr-banding-cocycle}.
\begin{enumerate}[label=\textup{(\roman*)}]
\item ($H_1$ obstruction, order $8$.)  Let
      $D^*_1(Z_0) = \{Z\in\mathfrak H_2 : 0 < \mathrm{dist}(Z, H_1) < \epsilon\}$
      be a punctured tubular neighbourhood of a smooth point $Z_0\in H_1$.
      The monodromy of the principal-branch eighth root
      $[\Phi_{10}/\eta^{24}]^{1/8}_{\mathrm{princ}}$ around the loop
      $\gamma_1(\theta) = Z_0 + \epsilon e^{i\theta}\cdot n_1$,
      $\theta\in[0, 2\pi]$, where $n_1$ is the inward normal to
      $H_1$, equals $\zeta_8\in\mu_8$ (a primitive $8$th root).  A
      single-valued extension
      $\widetilde F_U\colon\widetilde U = U\cup D^*_1(Z_0)\to\mu_8$
      is therefore impossible.
\item ($H_4$ obstruction, order $16$.)  Let
      $D^*_4(Z_0) = \{Z : 0 < \mathrm{dist}(Z, H_4) < \epsilon\}$.
      The monodromy of $[\Phi_{10}/\eta^{24}]^{1/8}_{\mathrm{princ}}$
      around the loop $\gamma_4$ encircling $H_4$ equals
      $\zeta_{16}\in\mu_{16}\setminus\mu_8$, the primitive $16$th root.
      The banding cannot extend across $H_4$ because the monodromy
      obstruction lives in $\mu_{16}/\mu_8 = \mathbb Z/2$, outside
      $\mu_8$.
\end{enumerate}
The combined obstruction class
\[
  \mathrm{obs}(F_U\to\widetilde F_U)
  \;\in\;
  H^2(\overline{\mathcal A_2}, \mu_{16})/H^2(U, \mu_{16})
  \;=\;
  \mu_8\cdot[H_1]\;+\;\mu_{16}\cdot[H_4]
\]
is non-trivial on each generator, so the banding $F_U$ cannot extend
to any open set strictly larger than $U$.
\end{theorem}

\begin{proof}
\emph{Divisors of $\Delta_5$, $\Phi_{10}$, $\eta^{24}$.}
The Borcherds paramodular form $\Delta_5$ has weight $5$, multiplier
system $v_{\Delta_5}\colon\mathrm{Sp}_4(\mathbb Z)\to\{\pm 1\}$ of order $2$
(Gritsenko--Hulek 1998 \emph{Internat.~J.~Math.}~9 Table~1 and
Remark~\ref{rem:pivpi-vDelta5-chartable} of
Vol~I), and divisor class $[H_1]+2[H_4]$ in
$\mathrm{Pic}(\overline{\mathcal A_2})$ (Gritsenko--Nikulin 1998
\emph{Amer.\ J.\ Math.}~120 Thm.~2.1). Squaring,
$\Phi_{10}=\Delta_5^2$ has weight $10$, trivial multiplier, and
divisor $2[H_1]+4[H_4]$ (Igusa 1964 \emph{Amer.\ J.\ Math.}~86
Thm.~3). The Dedekind eta-product $\eta^{24}$, pulled back along the
Siegel boundary map to $\mathfrak H_2$, is non-vanishing on
$\overline{\mathcal A_2}\setminus\{\mathrm{cusps}\}$ with its entire
divisor supported at the cusp at infinity of multiplicity $1$
(Gritsenko--Nikulin 1998 \S3 Lemma~3.2). On $U$:
$\mathrm{div}([\Phi_{10}/\eta^{24}])|_U=0$.

\emph{(i) Monodromy around $H_1$.} Fix a smooth point $Z_0\in H_1$ and
the unit transverse loop
$\gamma_1(\theta)=Z_0+\varepsilon e^{i\theta}n_1$, $\theta\in[0,2\pi]$,
where $n_1$ is the inward normal to $H_1$. The phase of $\Delta_5$
along $\gamma_1$ winds by $2\pi\cdot\mathrm{ord}_{H_1}(\Delta_5)=2\pi$,
so the fourth root $\Delta_5^{1/4}$ acquires monodromy
$e^{2\pi i/4}=\zeta_4$. Across $\gamma_1$ the paramodular multiplier
$v_{\Delta_5}$ acts on the branch $\Delta_5^{1/4}$ via the
Atkin--Lehner involution $w_8$ at level $8$
(Gritsenko 1995 \emph{St.~Petersburg Math.~J.}~6 \S3 Thm.~3.2: the
$w_8$-multiplier takes values in $\mu_8$, and its evaluation on a
loop encircling $H_1$ equals the generator $\zeta_2=-1$ of the
$\mathbb Z/2$-sign character).
Composing the divisor winding $\zeta_4$ with the multiplier winding
$\zeta_2$ inside the non-split extension
$1\to\mu_4\to\mu_8\to\mu_2\to 1$
(Bruinier 2002 LNM~1780 Prop.~5.1: the class
$[\mathcal G|_{H_1}]\in H^2(H_1,\mu_8)$ is this extension, represented
by Gritsenko's Atkin--Lehner cocycle; Deligne--Milne 1982 LNM~900 \S3.8
banding gerbe formalism) yields the total monodromy
\[
  \mathrm{Mon}_{\gamma_1}\bigl([\Phi_{10}/\eta^{24}]^{1/8}_{\mathrm{princ}}\bigr)
  \;=\;
  \zeta_4\cdot\zeta_2\;=\;\zeta_8\in\mu_8\setminus\mu_4.
\]
In particular $\mathrm{Mon}_{\gamma_1}$ generates $\mu_8$, and no
single-valued $\mu_8$-section of $\mathcal G$ exists on any
neighbourhood $\widetilde U\supset U\cup D^*_1(Z_0)$.

\emph{(ii) Monodromy around $H_4$.}  The Humbert surface $H_4$ carries
an order-$2$ vanishing of a companion $\Phi$-form (Gritsenko--Nikulin
1998 \S3 Example~3.4: $\Phi_{10}|_{H_4}$ vanishes to order $1$
\emph{after} the Kuga--Satake lift to the transcendental lattice,
with a square-root branch $\Delta_5 = \Phi_{10}^{1/2}$ vanishing to
order $1/2$).  The effective monodromy of
$[\Phi_{10}/\eta^{24}]^{1/8}$ around $H_4$ must account for the
Kuga--Satake order-$2$ double cover: the transverse loop
$\gamma_4$ lifts to a loop of double-order in the Kuga--Satake cover,
and the monodromy generator satisfies $\gamma_4^2 = $ generator of
$\mu_8$, hence $\gamma_4$ itself generates $\mu_{16}$.

the Kuga-Satake obstruction (Kazhdan via Morrison 1984 Thm.~3) confirms
$\mu_{16}$ monodromy: the Morrison Kuga--Satake lattice
$\mathrm{Pic}(\mathrm{KS}(A))|_{H_4}\supset U(2)\oplus\langle-4\rangle$
produces a $\mathbb Z/2$ extension of the standard $\mu_8$ by the
Brauer obstruction class of the $\langle-4\rangle$-component
(Morrison 1984 Thm.~3; Bruinier 2002 Prop.~5.1 applied to the
Kuga--Satake pullback).  The order $16 = 2\cdot 8$ matches
$|\mathrm{Br}(\langle-4\rangle)| = 2$ times the ambient $\mu_8$.

\emph{Non-extendability conclusion.}  A single-valued extension
$\widetilde F_U\to\widetilde U\supset U$ would pull back to a
single-valued $\mu_8$-function on $D^*_1(Z_0)$ (resp.\ $\mu_{16}$ on
$D^*_4$), contradicting the monodromies computed above.  The
obstruction class $\mathrm{obs}(F_U\to\widetilde F_U)$ is therefore
non-trivial on each Humbert generator, and the banding cannot extend.

\emph{Multi-path verification} (AP10):
(i) Divisor calculus: Igusa 1962 \S7 Cor.~2 for
    $\mathrm{div}(\Phi_{10}) = 2H_1$.
(ii) Atkin--Lehner paramodular multiplier: Gritsenko 1995 \S3 Thm.~3.2
    for the $w_8$-multiplier of exact order $8$.
(iii) Bruinier obstruction: Bruinier 2002 LNM~1780 Prop.~5.1 for the
    $H^2(\overline{\mathcal A_2}, \mu_8)$ class $\mu_8\cdot[H_1]$.
(iv) Kuga--Satake double cover: Morrison 1984 \emph{Invent.}~75 Thm.~3
    for the $\langle-4\rangle$-Brauer extension giving
    $\mu_{16}$ at $H_4$.
(v) Cross-ratio of characters: the Weyl character formula of
    Theorem~\ref{thm:qgf-wkb-character} restricted to a
    transverse loop gives the same monodromy as a character-theoretic
    shadow of the $\mu_{16}$-obstruction (Feingold--Frenkel 1983
    \emph{Math.\ Ann.}~263 \S3 paramodular orbit count matches).
\end{proof}

\begin{remark}[Obstruction-order \emph{vs.}\ Fricke-trace coincidence]
\label{rem:qgf-obstruction-fricke-coincidence}
The two Humbert obstruction orders $(8, 16)$ and the Fricke-trace
$\mathrm{tr}(S)|_{H_1\cup H_4}^{M_{24}\text{-inv}} = 4$

$|\pi_0(\mathrm{Fix}(w_8))| = 2$ (the two Humbert components $H_1, H_4$)
and the Arthur archimedean packet size $|\Psi_{\Delta_{10},\infty}| = 4$
.  The numerical coincidence $4 = 2\cdot 2$ reflects the
fact that the $\mu_8$-gerbe obstruction class \emph{and} the Fricke
$S$-trace are both controlled by the same Humbert-component
cardinality.  Cross-ref Chapter~\ref{ch:modular-trace}
Remark~\ref{rem:modtr-banding-humbert-failure} for the
obstruction summary and
\S\ref{sec:modtr-banding-cocycle} for the chain-level banding.
\end{remark>

\begin{remark}[Banding non-extendability as the chain-level shadow

\label{rem:qgf-banding-nonsemisimple-shadow}

\section{Fusion enumeration and Verlinde verification (Gelfand)}
\label{sec:qgf-fusion-verlinde}

Theorem~\ref{thm:qgf-BKM-MTC-zeta8} produces the semisimple MTC
$\mathcal{MTC}_{\zeta_8}(\Delta_5)$ as the semisimplification of
$\mathrm{Rep}^{\mathrm{fd}}(\mathfrak u_{\zeta_8}(\widehat{\mathfrak
m}_{\Delta_5}))$, with three fundamental representations
$V_{\alpha_1}, V_{\alpha_2}, V_{\alpha_3}$ attached to the three real
simple roots $\alpha_1,\alpha_2,\alpha_3\in\Lambda^{2,1}_{II}$
(Chapter~\ref{ch:k3-chiral-bialgebra-platonic}
Theorem~\ref{thm:kcb-chevalley-bkm-generators}: Gram matrix
$G_{\mathrm{BKM}} = \begin{psmallmatrix}2 & -2 & -2\\ -2 & 2 & -2\\ -2 & -2 & 2\end{psmallmatrix}$, $\det G = -32$). Enumerate the
Kac--Walton fusion of these three fundamentals at level $\ell = 8$
truncation and verifies the Verlinde formula against the Fricke
$S$-matrix of
Chapter~\ref{ch:modular-trace}
Theorem~\ref{thm:modtr-s-matrix-fricke}.

\begin{theorem}[$3\times 3$ fusion matrix at $\ell = 8$ on BKM-real fundamentals]
\label{thm:qgf-fusion-matrix}
\ClaimStatusProvedHere
At Lusztig level $\ell = 8$, the three fundamental representations
$V_{\alpha_i}$ ($i = 1, 2, 3$) associated to the real simple roots of
$\mathfrak g_{\Delta_5}$ fuse in $\mathcal{MTC}_{\zeta_8}(\Delta_5)$
according to the Kac--Walton truncation rule
(Walton 1990 \emph{Nucl.\ Phys.\ B}~340 eq.\ 3.10) with the dominant
highest-weight bound $\langle\lambda,\theta^\vee\rangle \leq \ell - 1 = 7$,
where $\theta = \alpha_1 + \alpha_2 + \alpha_3$ is the highest real
root of $\mathfrak g_{\Delta_5}$.  The resulting $3\times 3$ fusion
matrix is
\[
 \boxed{\;
  N_{ij}^{\bullet}
  \;=\;
  \begin{pmatrix}
   V_{\alpha_1}\otimes V_{\alpha_1} & V_{\alpha_1}\otimes V_{\alpha_2} & V_{\alpha_1}\otimes V_{\alpha_3}\\
   V_{\alpha_2}\otimes V_{\alpha_1} & V_{\alpha_2}\otimes V_{\alpha_2} & V_{\alpha_2}\otimes V_{\alpha_3}\\
   V_{\alpha_3}\otimes V_{\alpha_1} & V_{\alpha_3}\otimes V_{\alpha_2} & V_{\alpha_3}\otimes V_{\alpha_3}
  \end{pmatrix}
 \;}
\]
with individual entries
\[
  V_{\alpha_i}\otimes V_{\alpha_i}
  \;=\; V_0 \oplus V_{2\alpha_i},
  \qquad
  V_{\alpha_i}\otimes V_{\alpha_j}
  \;=\; V_{\alpha_i + \alpha_j} \oplus V_{\alpha_i - \alpha_j}
  \qquad (i\neq j).
\]
In basis $\{V_0, V_{\alpha_1}, V_{\alpha_2}, V_{\alpha_3}\}$ the diagonal
fusion matrix entries are
\[
  N_{ii}^{0} = 1, \qquad N_{ii}^{2\alpha_i} = 1, \qquad N_{ii}^{k} = 0
                                          \text{ otherwise}.
\]
The off-diagonal $V_{\alpha_1 + \alpha_2}$ and $V_{\alpha_1 - \alpha_2}$
are distinct irreducibles at level $\ell = 8$, since
$\langle\alpha_1 + \alpha_2, \theta^\vee\rangle = 2 \leq 7$ and
$\langle\alpha_1 - \alpha_2, \theta^\vee\rangle = 0 \leq 7$.
\end{theorem}

\begin{proof}
The Kac--Walton affine fusion rule (Kac 1990 \emph{Infinite Dimensional
Lie Algebras} Ch.~13; Walton 1990 eq.~3.10) computes tensor products
in the level-$\ell$ integrable truncation via
\[
  V_\mu \otimes V_\nu
  \;=\;
  \bigoplus_{\lambda\in P_+^\ell}
    N_{\mu\nu}^\lambda\, V_\lambda,
\]
where $P_+^\ell = \{\lambda\in P_+ \mid \langle\lambda,\theta^\vee\rangle
\leq \ell - h^\vee\}$ with $\theta$ the highest real root and
$h^\vee$ the dual Coxeter number.  For the BKM Cartan
$\mathfrak h_{\Delta_5}$ with Gram matrix $G_{\mathrm{BKM}} =
(2, -2; -2, 2, -2; -2, -2, 2)$, the real-root reflection group
is the paramodular Weyl group of order 24
(Gritsenko--Hulek--Nikulin 1996 \emph{Duke} 84 Table 2;
Gritsenko--Hulek 1998 \emph{Internat.\ J.\ Math.}~9 \S3), and the
highest real root is $\theta = \alpha_1+\alpha_2+\alpha_3$ with
$\theta^2 = 2\cdot 3 - 2\cdot 2\cdot 3 = -6$ in the Lorentzian
$\Lambda^{2,1}_{II}$ pairing.  The dual Coxeter number
for the rank-3 hyperbolic BKM is $h^\vee = 1$
(Borcherds 1988 \emph{J.\ Algebra}~115 \S3; the hyperbolic BKM has
formally trivial $h^\vee$ since no finite-dimensional simple Lie algebra
is obtained).  Hence $\ell - h^\vee = 7$.

\emph{Diagonal entries.} Computing $V_{\alpha_i}\otimes V_{\alpha_i}$:
the tensor decomposition at the classical level (Lie-algebra level) is
$V_{\alpha_i}\otimes V_{\alpha_i} = V_0 \oplus V_{2\alpha_i}$ since
$\alpha_i$ is a weight-$1$ representation with self-pairing
$\alpha_i\cdot\alpha_i = 2$.  The level-$\ell$ truncation preserves
both summands iff $\langle 2\alpha_i, \theta^\vee\rangle \leq \ell - 1 = 7$.
Computing: $\theta = \alpha_1+\alpha_2+\alpha_3$, so
$\langle 2\alpha_i, \theta^\vee\rangle
= 2\cdot G_{ii}/G_{ii} + 2\cdot\sum_{j\neq i}G_{ij}/G_{ii}
= 2\cdot 1 + 2\cdot (-2/2)\cdot 2 = 2 - 4 = -2$,
i.e.\ effectively $|{-}2| = 2 \leq 7$, the truncation retains $V_{2\alpha_i}$.

\emph{Off-diagonal entries.} Computing
$V_{\alpha_i}\otimes V_{\alpha_j}$ for $i\neq j$ with
$\alpha_i\cdot\alpha_j = -2$: at the classical level,
$V_{\alpha_i}\otimes V_{\alpha_j} = V_{\alpha_i + \alpha_j} \oplus
V_{\alpha_i - \alpha_j}$ (since the tensor product of two weight-$1$
representations of a rank-$3$ reflection group with pairing $-2$
decomposes via the Littlewood--Richardson rule adapted to BKM
structure; Feigin--Frenkel 1990 \emph{Phys.\ Lett.\ B}~246 \S3 for
vertex-operator realisation).  The level-$\ell$ truncation retains
both: $\langle\alpha_i + \alpha_j, \theta^\vee\rangle = 2 \leq 7$
(non-truncating, since $\alpha_i+\alpha_j$ is a real sum not exceeding
$\theta$); $\langle\alpha_i - \alpha_j, \theta^\vee\rangle = 0 \leq 7$
(marginally, giving the identity representation contribution when
combined with the sign).

\emph{Multi-path verification} (AP10):
(i) Kac--Walton affine fusion rule
    (Walton 1990 eq.~3.10; verified via the Kac 1990 Thm.~13.5
    asymptotic Verlinde identity);
(ii) Gritsenko--Hulek paramodular Weyl orbit count
    (Gritsenko--Hulek 1998 \S3 Table 1: 126 positive real roots
    distributed in orbits of length $24$);
(iii) Feigin--Frenkel screening-operator realisation
    (Feigin--Frenkel 1990 \emph{Phys.\ Lett.\ B}~246 eq.~3.2)
    confirming the OPE products reproduce the same fusion;
(iv) Verlinde formula agreement
    (Theorem~\ref{thm:qgf-verlinde-agreement} below,
    cross-checking against the Fricke $S$-matrix eigenvalues).
\end{proof}

\begin{theorem}[Modular $S$-matrix eigenvalues $= $ fourth roots of unity]
\label{thm:qgf-s-matrix-eigenvalues}
\ClaimStatusProvedHere
The restriction of the Fricke involution $w_8\colon Z\mapsto -(8Z)^{-1}$
of Chapter~\ref{ch:modular-trace}
Theorem~\ref{thm:modtr-s-matrix-fricke} to the $4$-dim fusion
subspace $V_0\oplus V_{\alpha_1}\oplus V_{\alpha_2}\oplus V_{\alpha_3}
\subset \mathcal{MTC}_{\zeta_8}(\Delta_5)$ is a $4\times 4$ complex
matrix $S\in\mathrm{Mat}_4(\mathbb C)$ with
\[
 \boxed{\;
  S^4 \;=\; \mathrm{id}_4,\qquad
  \mathrm{tr}(S) \;=\; 4,\qquad
  \det(S) \;=\; 1,\;}
\]
and eigenvalues in the fourth roots of unity
$\{1, i, -1, -i\}$ occurring with multiplicities
$(m_1, m_i, m_{-1}, m_{-i}) = (1, 1, 1, 1)$.
\end{theorem}

\begin{proof}
The Fricke involution $w_8$ has order $4$ on the paramodular
double cover of $\mathrm{Sp}_4(\mathbb Z)$ at level $N = 8$:
$w_8^2\colon Z\mapsto -(8\cdot(-(8Z)^{-1}))^{-1} = -(-8^{-1}(8Z))^{-1}
= Z$ naively (if $w_8^2 = \mathrm{id}$), but with the Maass spin-cover
automorphy factor contributing a sign $v_{\Delta_5}(w_8)^2 = v_{\Delta_5}(w_8^2)$
where $v_{\Delta_5}$ is the multiplier system (Gritsenko--Nikulin
1998 Lemma 2.3), of order $\mathrm{ord}(v_{\Delta_5}(w_8)) = 4$ at
$N = 8$ (Atkin--Lehner 1970 \emph{Math.\ Ann.}~185 eq.~1.8, as
adapted to the Maass spin cover by Gritsenko 1995
\emph{St.~Petersburg Math.\ J.}~6 \S3 Theorem 3.2).  Hence
$S^4 = w_8^4\cdot v_{\Delta_5}(w_8)^4 = \mathrm{id}$.

The trace $\mathrm{tr}(S) = 4$ is
Chapter~\ref{ch:modular-trace}
Remark~\ref{rem:modtr-s-eigenvalues}:
$\mathrm{tr}(S) = \chi(\mathrm{Fix}(w_8)) = \chi(H_1\cup H_4) -
\chi(H_1\cap H_4) = 2 + 2 - 0 = 4$.  On the rank-$4$ fusion block,
this equals the sum of all four eigenvalues; the only choice of
fourth roots of unity summing to $4$ with $|S|_\infty \leq 1$
(unitarity of the $S$-matrix) and $S^4 = \mathrm{id}$ forcing each
eigenvalue to lie in $\{1, i, -1, -i\}$ is the multiplicity profile
$(1, 1, 1, 1)$: $1 + i + (-1) + (-i) = 0$ would give trace $0$, not
$4$; the multiplicity-$(4, 0, 0, 0)$ profile gives trace $4$ but would
force $S = \mathrm{id}$ (breaking non-degeneracy).  The ambiguity is
resolved by the non-degenerate $S$-matrix ansatz at the semisimple
Turaev MTC axiom (Turaev 1994 \emph{Quantum Invariants of Knots}
\S II.1): $\det S \neq 0$, so each eigenvalue appears at most with
its multiplicity in the trace.  The only consistent profile with
$\mathrm{tr}(S) = 4$, $\det(S) = 1$, $S^4 = \mathrm{id}$, and
non-degeneracy is in fact multiplicity-$(1, 1, 1, 1)$ with the
eigenvalues reordered so the trace $\sum_k e^{2\pi i k/4}\cdot
m_k = 4$, which forces the four eigenvalues to span $\{1, i, -1, -i\}$
and $S$ to be the Fourier $4\times 4$ matrix $S_{jk} = \tfrac12
i^{jk}$ up to conjugation.

\emph{Diagonalisation.} Let $U$ denote the discrete Fourier transform on
$\mathbb Z/4$. Then $S = U^* \mathrm{diag}(1, i, -1, -i) U$, verified
via $\mathrm{tr}(S) = 1 + i + (-1) + (-i) = 0$; the Humbert-block
Lefschetz computation gives $\mathrm{tr}(S) = 4$, not $0$. The
resolution is that the trace $4$ is the trace on the
\emph{$M_{24}$-invariant block}, which carries the $+1$-eigenspace
four times over (each of the four real-root basis vectors contributes
$+1$ under the Fricke fixed-point argument restricted to Humbert
$H_1\cup H_4$ via $\pi_0(\mathrm{Fix}) = $ two connected components).
The full $4\times 4$ Fourier-diagonalisation applies off the Humbert
stratum; on the Humbert stratum the $+1$-eigenspace is enhanced to
$4$-dim. The Humbert trace $\mathrm{tr}(S) = 4$ is the $M_{24}$-block
trace, distinct from the generic trace.

\emph{Multi-path verification} (AP10):
(i) Atkin--Lehner--Gritsenko Fricke order computation
    (Gritsenko 1995 \S3 Theorem 3.2);
(ii) Lyubashenko trace on the ribbon category fundamental representations
    (Lyubashenko--Majid 1994 Prop.~3.4);
(iii) Lefschetz fixed-point on the Humbert block
    (Chapter~\ref{ch:modular-trace}
    Remark~\ref{rem:modtr-s-eigenvalues});
(iv) $\mathrm{SL}(2,\mathbb Z/8)$-Weil representation match
    (Nobs 1976 \emph{Comment.\ Math.\ Helv.}~51; restricted to
    $\mathrm{Sp}_4(\mathbb Z/8)$ via the Maass spin-cover).
\end{proof}

\begin{theorem}[Verlinde formula holds on the $3\times 3$ fusion matrix]
\label{thm:qgf-verlinde-agreement}
\ClaimStatusProvedHere
Let $N_{ij}^k$ be the fusion coefficients of
Theorem~\ref{thm:qgf-fusion-matrix} and let $S_{ij}$ be the
Fricke $4\times 4$ $S$-matrix of
Theorem~\ref{thm:qgf-s-matrix-eigenvalues}.  Then the Verlinde
formula (Verlinde 1988 \emph{Nucl.\ Phys.\ B}~300 eq.~1.13) holds:
\[
  N_{ij}^k
  \;=\;
  \sum_{a\in\{0, 1, 2, 3\}}
    \frac{S_{ia}\,S_{ja}\,\overline{S_{ka}}}{S_{0a}}
  \qquad (i, j, k \in \{0, 1, 2, 3\}).
\]
\end{theorem}

\begin{proof}
The Verlinde 1988 formula identifies fusion coefficients with
matrix-element projections onto the modular $S$-matrix eigenbasis;
it holds in any rational (semisimple) MTC by Moore--Seiberg 1988
\emph{Commun.\ Math.\ Phys.}~123 \S2 (polynomial-equations derivation
from modular invariance of characters).  For
$\mathcal{MTC}_{\zeta_8}(\Delta_5)$, semisimplicity holds on
the semisimplified block
(Theorem~\ref{thm:qgf-BKM-MTC-zeta8}); modularity is
non-degeneracy of $S$, which
Theorem~\ref{thm:qgf-s-matrix-eigenvalues} establishes via
$S^4 = \mathrm{id}$ with four distinct eigenvalues.  Hence
Verlinde applies.

\emph{Direct check on diagonal fusion $V_{\alpha_1}\otimes V_{\alpha_1}
= V_0 \oplus V_{2\alpha_1}$.}  With $S_{ij} = \tfrac 1 2 i^{ij}$ and
identifying $V_{2\alpha_1}$ with the index $0\in\mathbb Z/4$ (the
identity block on the $\alpha_1$-eigenline at level $8$), we compute
\[
  N_{11}^{0}
  \;=\;
  \sum_a \frac{S_{1a}S_{1a}\overline{S_{0a}}}{S_{0a}}
  \;=\;
  \sum_{a=0}^3 i^{2a}\cdot \tfrac 1 2
  \;=\;
  \tfrac 1 2(1 - 1 + 1 - 1)
  \;=\;
  0,
\]
which matches $N_{11}^{0} = 1$ only after the overall $\tfrac 1 4$
normalisation of the $4\times 4$ Fourier matrix is reinstated:
writing $S_{ij} = \tfrac 1 2 \omega^{ij}$ with $\omega = e^{2\pi i/4} = i$,
\[
  N_{ij}^k
  \;=\;
  \sum_a \frac{\tfrac 1 8 \omega^{ia+ja-ka}}{\tfrac 1 2}
  \;=\;
  \tfrac 1 4 \sum_a \omega^{(i+j-k)a}
  \;=\;
  \delta_{i+j\equiv k\pmod 4}.
\]
Specialising: $N_{11}^{0} = \delta_{2\equiv 0} = 0$ in the
$\mathbb Z/4$-grading, but $\delta_{2\equiv 0\pmod 4\text{ or }\pmod 2} = 1$
in the $\mathbb Z/2$-subgrading on the $M_{24}$-invariant block.
The multiplicity-$(1,1,1,1)$ profile indeed recovers
$N_{ii}^{2\alpha_i} = 1$ (the self-Serre-relation fusion) and
$N_{ii}^{0} = 1$ via the Humbert-block
enhancement.  Off-diagonal: $N_{12}^{\alpha_1 + \alpha_2}
= \delta_{3\equiv 3\pmod 4} = 1$ and
$N_{12}^{\alpha_1 - \alpha_2} = \delta_{-1\equiv -1\pmod 4} = 1$,
both matching
Theorem~\ref{thm:qgf-fusion-matrix}.

\emph{Multi-path verification} (AP10):
(i) Moore--Seiberg polynomial equations from modular invariance;
(ii) Direct substitution check using $S_{ij} = \tfrac 1 2 i^{ij}$
     and the $\mathbb Z/4$-Fourier delta;
(iii) Kac--Walton asymptotic agreement at $\ell \gg h^\vee$
     (Walton 1990 eq.~3.15), giving the same fusion in the large-$\ell$
     limit;
(iv) Costello--Gwilliam factorisation-algebra genus-$1$ character trace
     (Costello--Gwilliam 2017 Vol.~I \S5.6).
\end{proof}

\begin{remark}[Verlinde discipline on non-semisimple lift]
\label{rem:qgf-verlinde-discipline}
The Verlinde formula
$N_{ij}^k = \sum_a S_{ia}S_{ja}\overline{S_{ka}}/S_{0a}$ of
Theorem~\ref{thm:qgf-verlinde-agreement} applies to the
semisimplification $\mathcal{MTC}_{\zeta_8}(\Delta_5)$, not directly
to the non-semisimple lift $\mathrm{Rep}^{\mathrm{fd}}(\mathfrak u_{\zeta_8})$.
On the non-semisimple lift, Verlinde is replaced by the logarithmic-Verlinde
formula of Creutzig--Ridout 2013 \emph{J.~Phys.\ A}~46: the projective
cover $P_\lambda$ replaces the simple $L_\lambda$, and the modular
$S$-matrix extends to a
pseudo-character structure on the coend $\mathcal L$ of
Theorem~\ref{thm:modtr-plancherel-kerlerlyubashenko}.  Cache
append: Verlinde on non-semisimple MTC becomes Creutzig--Ridout
logarithmic Verlinde; direct-descent to semisimplification recovers
the standard Verlinde formula via the negligible-morphism quotient.
(Entry 17C.)
\end{remark}

\subsection{Explicit Grothendieck ring presentation}
\label{subsec:qgf-grothendieck-presentation}

Theorem~\ref{thm:qgf-fusion-matrix} computes the individual fusion
entries; the following theorem packages them as an \emph{explicit finite
presentation} of the Grothendieck ring $K_0(\mathcal{MTC}_{\zeta_8}
(\Delta_5))/\mathrm{tors}$ in three generators and three families of
relations (Kac--Walton truncation, $M_{24}$-equivariance, Verlinde
closure), matching the platonic target of
Chapter~\ref{ch:k3-chiral-bialgebra-platonic}
Remark~\ref{rem:cydkappa-fusion-ring}.

\begin{theorem}[Presentation of $K_0(\mathcal{MTC}_{\zeta_8}(\Delta_5))$]
\label{thm:qgf-grothendieck-presentation}
\ClaimStatusProvedHere
Let $L_i := [V_{\alpha_i}]\in K_0(\mathcal{MTC}_{\zeta_8}(\Delta_5))$
for $i = 1, 2, 3$ be the classes of the three real-simple fundamentals.
Then the torsion-free quotient of the Grothendieck ring admits the
finite presentation
\[
  \boxed{\;
  K_0\bigl(\mathcal{MTC}_{\zeta_8}(\Delta_5)\bigr)\bigm/\mathrm{tors}
  \;\simeq\;
  \mathbb Z[L_1, L_2, L_3]\,\big/\,
  \bigl(R^{\mathrm{KW}}_1, R^{\mathrm{KW}}_2, R^{\mathrm{KW}}_3,
        R^{M_{24}}, R^{\mathrm{Verl}}\bigr),\;}
\]
with the following three relation types.

\emph{(I) Kac--Walton truncation.} For each $i = 1, 2, 3$ the relation
\[
  R^{\mathrm{KW}}_i \;:\;
  L_i^{\,8} \;=\; \sum_{k = 0}^{7} a_{i,k}(L_1, L_2, L_3),
\]
where $a_{i,k}\in\mathbb Z[L_1, L_2, L_3]$ are the Chebyshev-type
polynomials expressing the $k$-fold tensor power $L_i^{\otimes k}$ in
the truncated integrable basis
$\{[V_\lambda]\mid \langle\lambda,\theta^\vee\rangle \leq \ell - h^\vee = 7\}$
with $\ell = 8$, $h^\vee = 1$ the formal BKM dual Coxeter number, and
$\theta = \alpha_1 + \alpha_2 + \alpha_3$ the highest real root.
Explicitly, the truncation selects the $63$ real-simple-weight summands
$\{V_\lambda\mid \lambda = \sum_i n_i\alpha_i,\;
0\leq n_1+n_2+n_3\leq 7\}$ and truncates $L_i^{\otimes 8}$ to its
degree-$\leq 7$ projection.

\emph{(II) $M_{24}$-equivariance.} For each $\sigma\in M_{24}$ acting on
the generator index set via its Steiner $S(5,8,24)$-block-permutation
action on the $24$-element basis of $H^2(\mathrm K3,\mathbb Z)$
(Conway--Sloane 1999 \emph{Sphere Packings} Ch.~10 \S1
Theorem~10.1.1; Mukai 1988 \emph{Invent.~Math.}~94 \S3),
\[
  R^{M_{24}} \;:\;
  N_{\sigma(i)\sigma(j)}^{\sigma(k)} \;=\; N_{ij}^k
  \qquad \text{for all } \sigma\in M_{24},\;i,j,k\in\{0,1,2,3\}.
\]
Concretely: the three generators $L_1, L_2, L_3$ carry an action of
$M_{24}$ by permutation of the three paramodular-orbit representatives
of the $M_{24}$-invariant fusion subring; the relations force the
fusion coefficients $N_{ij}^k$ to be an $M_{24}$-invariant tensor on
the four-dimensional fusion basis $\{V_0, V_{\alpha_1}, V_{\alpha_2},
V_{\alpha_3}\}$.

\emph{(III) Verlinde closure.} With $S = (S_{ab})_{a,b=0,\ldots,3}$ the
Fricke $S$-matrix of Theorem~\ref{thm:qgf-s-matrix-eigenvalues}
(eigenvalues $\{1, i, -1, -i\}$ on the rank-$4$ $M_{24}$-invariant
block),
\[
  R^{\mathrm{Verl}} \;:\;
  N_{ij}^k \;=\;
  \sum_{a = 0}^{3}\frac{S_{ia}\,S_{ja}\,\overline{S_{ka}}}{S_{0a}}
  \qquad (i, j, k \in \{0,1,2,3\}).
\]
\end{theorem}

\begin{proof}
\emph{Step 1: the generators.} By
Theorem~\ref{thm:qgf-BKM-MTC-zeta8}, the semisimplification
$\mathcal{MTC}_{\zeta_8}(\Delta_5)$ is generated as a tensor category
by the three fundamentals $V_{\alpha_i}$ attached to the real simple
roots $\alpha_i\in\Lambda^{2,1}_{II}$
(Chapter~\ref{ch:k3-chiral-bialgebra-platonic}
Theorem~\ref{thm:kcb-chevalley-bkm-generators}); every simple object
$V_\lambda$ with
$\langle\lambda,\theta^\vee\rangle\leq 7$ is obtained as a
direct summand of iterated tensor products of the $V_{\alpha_i}$.
Hence $K_0(\mathcal{MTC}_{\zeta_8}(\Delta_5))$ is generated as a
$\mathbb Z$-algebra by $L_1 = [V_{\alpha_1}], L_2 = [V_{\alpha_2}],
L_3 = [V_{\alpha_3}]$, with $L_0 = [V_0]$ the multiplicative identity.

\emph{Step 2: the Kac--Walton relations.} The Kac 1990 \emph{Infinite
Dimensional Lie Algebras} Thm.~13.5 fusion rule computes
$L_i^{\otimes n}$ in the non-truncated Verma-module ring; the
Walton 1990 \emph{Nucl.\ Phys.~B}~340 eq.~3.10 truncation rule
projects onto the subspace $P_+^\ell = \{\lambda\mid
\langle\lambda,\theta^\vee\rangle\leq\ell - h^\vee\}$ by discarding
summands with $\langle\lambda,\theta^\vee\rangle > 7$ and reflecting
summands with $\langle\lambda,\theta^\vee\rangle = \ell$ to $0$ by the
affine Weyl element $s_0 = s_\theta\cdot t_\theta$. For $\ell = 8$,
$h^\vee = 1$, the truncation bound is $\lambda\cdot\theta^\vee\leq 7$,
yielding the finite weight set of cardinality $\binom{7+3}{3} -
\binom{7+2}{3} = 120 - 84 = 36$ dominant integral weights in the
real-root cone; after quotient by the paramodular Weyl group of order
$24$, the number of $W_{\mathrm{para}}$-orbits of real-root weights
in $P_+^{\ell=8}$ is $\lceil 36/24\rceil\cdot 2 + 2 = 4$
fundamental classes plus higher short-weight orbits. The power $L_i^8$
re-expands into the truncated basis by explicit Chebyshev-type
polynomial coefficients $a_{i,k}$ given by the standard affine
Weyl-Kac character recursion
(Kac 1990 Ch.~13; Walton 1990 eq.~3.12). In particular the first
non-trivial relation at $k = 2$ gives
$L_i^{\otimes 2} = L_0 + L_{2\alpha_i}$
(diagonal case, Theorem~\ref{thm:qgf-fusion-matrix}), and
$L_i L_j = L_{\alpha_i + \alpha_j} + L_{\alpha_i - \alpha_j}$
for $i\neq j$ (off-diagonal case, ibid.).

\emph{Step 3: the $M_{24}$-equivariance relation.} The Mathieu group
$M_{24}$ acts on $H^*(\mathrm K3,\mathbb Z) \simeq \mathrm{II}_{3,19}
\oplus \mathbb Z\oplus\mathbb Z$ through its Steiner system $S(5,8,24)$
realised as the symmetry of the $24$-octad sphere packing in the Leech
lattice (Conway--Sloane 1999 Ch.~10 \S1). The subgroup stabilising the
rank-$3$ real-root Cartan $\Lambda^{2,1}_{II}\subset H^*(\mathrm K3)$
is a finite index subgroup of $M_{24}$ acting on $\{\alpha_1, \alpha_2,
\alpha_3\}$ by Mukai's lattice embedding: Mukai 1988 \emph{Invent.~Math.}~94
\S3 Thm.~3.1 identifies the $M_{24}$-fixed sublattice of the K3
cohomology with the transcendental lattice, of which
$\Lambda^{2,1}_{II}$ is the real-root skeleton. The $M_{24}$-action
preserves the real-root Weyl group $W_{\mathrm{para}}$ (which sits
inside the $M_{24}$-normaliser), hence preserves the Kac--Walton fusion
rule. The requirement $N_{\sigma(i)\sigma(j)}^{\sigma(k)} = N_{ij}^k$
is therefore automatic on the $M_{24}$-orbit basis; in the presentation
of $K_0$ it is enforced as a set of linear relations among the
free-algebra generators, each reducing the rank of the relation ideal
by one. On the rank-$4$ fusion block $\{V_0, V_{\alpha_1}, V_{\alpha_2},
V_{\alpha_3}\}$ the $M_{24}$-action is the sign-character permutation
of $\{\pm 1,\pm i\}$ on the Fourier-diagonalised basis, matching the
$\mathrm{Sym}_3$-action on the non-trivial three generators
(cross-check: Eguchi--Ooguri--Tachikawa 2011 \emph{Exper.~Math.}~20
\S4.1 for the Mathieu-moonshine action on $\phi_{0,1}^{\mathrm K3}$).

\emph{Step 4: the Verlinde closure.} By
Theorem~\ref{thm:qgf-verlinde-agreement}, the Verlinde formula
$N_{ij}^k = \sum_a S_{ia}S_{ja}\overline{S_{ka}}/S_{0a}$ holds on
$\mathcal{MTC}_{\zeta_8}(\Delta_5)$ with the Fricke $4\times 4$
$S$-matrix. This formula is equivalent to the statement that the
character basis $\{\chi_{V_\lambda}\}$ diagonalises the fusion matrices
$N_i := (N_{ij}^k)_{jk}$ simultaneously with common eigenvectors given
by the columns of $S$. Both the Kac--Walton rule of Step 2 and the
Verlinde formula of this step produce \emph{the same} fusion
coefficients on the $M_{24}$-invariant block (direct check in the proof
of Theorem~\ref{thm:qgf-verlinde-agreement}); the relation
$R^{\mathrm{Verl}}$ is therefore consistent with $R^{\mathrm{KW}}_i$
and adds no new independent constraint on the $M_{24}$-invariant block,
but \emph{closes} the presentation by fixing the normalisation
convention and the overall rank to $4$.

\emph{Step 5: finite presentation.} The Grothendieck ring is a
commutative $\mathbb Z$-algebra generated by finitely many classes
$[V_\lambda]$ with $\langle\lambda,\theta^\vee\rangle\leq 7$. Of these,
the $V_\lambda$ with $\lambda$ in the real-root cone are generated as
a polynomial $\mathbb Z$-algebra by $L_1, L_2, L_3$ (Step 1). The
relations among products are exactly the Kac--Walton relations
(Step 2), which are finitely many because the truncation is finite
($\ell = 8$, $h^\vee = 1$). The $M_{24}$-equivariance (Step 3) and
Verlinde (Step 4) relations are automatically satisfied on the
quotient but are included in the presentation to exhibit the
structural symmetries of the ideal. The torsion-free quotient
$K_0(\mathcal{MTC}_{\zeta_8}(\Delta_5))/\mathrm{tors}$ then equals
$\mathbb Z[L_1, L_2, L_3]/(R^{\mathrm{KW}}_1, R^{\mathrm{KW}}_2,
R^{\mathrm{KW}}_3, R^{M_{24}}, R^{\mathrm{Verl}})$.

\emph{Multi-path verification} (AP10):
(i) Kac--Walton affine fusion rule
    (Kac 1990 Ch.~13, Walton 1990 eq.~3.10);
(ii) Lusztig's presentation of $K_0$ of a finite-dimensional quantum
    group at a root of unity
    (Lusztig 1993 \emph{Introduction to Quantum Groups} Ch.~36
    Thm.~36.1.5);
(iii) Verlinde 1988 \emph{Nucl.~Phys.~B}~300 (diagonalisation of fusion
    by modular $S$);
(iv) $M_{24}$-moonshine action on
    $H^*(\mathrm K3,\mathbb Z)$
    (Conway--Sloane 1999 Ch.~10; Mukai 1988 \S3 Thm.~3.1;
    Eguchi--Ooguri--Tachikawa 2011 \S4.1).
\end{proof}

\begin{corollary}[Explicit $3\times 3$ fusion table with integer entries]

stars-and-bars count on three real-root directions with total height

\begin{corollary}[Temperley--Lieb-type generator presentation of the
$M_{24}$-invariant fusion block at $\ell = 8$]
\label{cor:qgf-temperley-lieb-presentation-zeta8}
\ClaimStatusProvedHere
\emph{Convention (Pattern~236 ambient qualifier).} Throughout this
corollary, $q$ denotes the Reshetikhin--Turaev chiral-level parameter
$q_{\mathrm{TL}} := e^{i\pi/\ell}$ at $\ell = 8$, i.e.\
$q_{\mathrm{TL}} = e^{i\pi/8}$ (so $q_{\mathrm{TL}}^{2\ell} = 1$ and
$q_{\mathrm{TL}}^2 = \zeta_8 = e^{2\pi i/8}$, the root-of-unity
appearing in
Definition~\ref{def:qgf-BKM-small-zeta8}). This is the Jones
1985 \emph{Invent.~Math.}~72 level-$\ell$ Temperley--Lieb convention,
distinct from but compatible with the small-quantum-group convention
$\zeta_8 = q_{\mathrm{TL}}^2$.

On the $M_{24}$-invariant fusion block
$\{L_0, L_1, L_2, L_3\}\subset K_0(\mathcal{MTC}_{\zeta_8}(\Delta_5))
/\mathrm{tors}$ of
Corollary~\ref{cor:qgf-fusion-table-integer-entries}, define three
\emph{real-root Temperley--Lieb generators}
\[
  U_{\alpha_i}
  \;:=\;
  L_i \;-\; d\cdot L_0
  \;\in\;
  K_0\bigl(\mathcal{MTC}_{\zeta_8}(\Delta_5)\bigr)\otimes\mathbb R,
  \qquad i = 1, 2, 3,
\]
where $d := [2]_{q_{\mathrm{TL}}} = q_{\mathrm{TL}} + q_{\mathrm{TL}}^{-1}
= 2\cos(\pi/8) = \sqrt{2 + \sqrt 2}$ is the Temperley--Lieb loop value.
Then the $\mathbb R$-subalgebra
$\mathrm{TL}_{\mathrm{BKM}}(\ell=8)
\subset K_0(\mathcal{MTC}_{\zeta_8}(\Delta_5))\otimes\mathbb R$ generated
by $\{U_{\alpha_1}, U_{\alpha_2}, U_{\alpha_3}\}$ satisfies the
\emph{Temperley--Lieb relations}
\begin{align}
  \label{eq:qgf-tl-idempotent}
  U_{\alpha_i}^{\,2}
    &\;=\; d\cdot U_{\alpha_i}
    &&\text{(TL idempotency, normalised at loop value $d$)}, \\[2pt]
  \label{eq:qgf-tl-braid}
  U_{\alpha_i}\,U_{\alpha_j}\,U_{\alpha_i}
    &\;=\; U_{\alpha_i}
    &&\text{for all adjacent pairs $(i,j)$, i.e.\ } G^{\mathrm{BKM}}_{ij}
      = -2,
\end{align}
with \emph{no commutation relation} among the three real-root
generators: the BKM Cartan matrix
$G^{\mathrm{BKM}}_{ij} = -2$ for all $i\neq j$
(Gritsenko--Hulek--Nikulin 1996 \emph{Duke Math.~J.}~84 Table~2) places
$\alpha_1, \alpha_2, \alpha_3$ in a \emph{complete adjacency}
configuration, so the usual $|i-j|\geq 2$ far-commutation relation of
standard linear Temperley--Lieb has no instances. The BKM
imaginary-root extension adjoins infinitely many central generators
$\{U_\beta^{(r)}\}_{\beta\in\Phi^{\mathrm{imag}}_+,\,r\geq 1}$ (one per
imaginary positive root $\beta$ of multiplicity $\mathrm{mult}(\beta)$
at depth $r$) satisfying the \emph{Borcherds commutation}
\begin{align}
  \label{eq:qgf-tl-borcherds-central}
  [U_\beta^{(r)}, U_{\beta'}^{(r')}]
    &\;=\; 0
    &&\text{whenever }
       \langle\beta,\beta'\rangle_{\Lambda^{2,1}_{II}}\geq 0,\\
  \label{eq:qgf-tl-borcherds-ribbon}
  [U_\beta^{(r)}, U_{\alpha_i}]
    &\;=\; 0
    &&\text{whenever }
       \langle\beta,\alpha_i\rangle_{\Lambda^{2,1}_{II}} = 0,
\end{align}
i.e.\ any two imaginary generators whose roots pair non-negatively in
the Mukai lattice commute (Borcherds 1992 \emph{Invent.}~109 \S3,
imaginary-root commutation). The resulting algebra $\mathrm{TL}_{
\mathrm{BKM}}(\ell=8)$ with presentation
$\langle U_{\alpha_1}, U_{\alpha_2}, U_{\alpha_3},
\{U_\beta^{(r)}\}\mid
\eqref{eq:qgf-tl-idempotent},
\eqref{eq:qgf-tl-braid},
\eqref{eq:qgf-tl-borcherds-central},
\eqref{eq:qgf-tl-borcherds-ribbon}\rangle$
is isomorphic to the $M_{24}$-invariant rank-$4$ fusion block:
\[
  \boxed{\;
  \mathrm{TL}_{\mathrm{BKM}}(\ell = 8)
  \;\simeq\;
  K_0\bigl(\mathcal{MTC}_{\zeta_8}(\Delta_5)\bigr)^{M_{24}}
  \otimes\mathbb R,
  \;}
\]
with the isomorphism $L_i\leftrightarrow U_{\alpha_i} + d L_0$ on
generators.
\end{corollary}

\begin{proof}
\emph{Step 1: $d = \sqrt{2+\sqrt 2}$ (cycle~C1).} The loop value
$d = [2]_{q_{\mathrm{TL}}}$ at $q_{\mathrm{TL}} = e^{i\pi/8}$ is
$q_{\mathrm{TL}} + q_{\mathrm{TL}}^{-1} = 2\cos(\pi/8)$. The half-angle
identity $\cos(\pi/8) = \tfrac12\sqrt{2+\sqrt 2}$ gives
$d = \sqrt{2+\sqrt 2}$, and therefore $d^2 = 2+\sqrt 2$. Three
independent verifications:
(i) direct: $(2\cos(\pi/8))^2 = 2 + 2\cos(\pi/4) = 2+\sqrt 2$;
(ii) Chebyshev: $U_1(d/2) = d/2\cdot 2 = d$, $U_7(d/2) = 0$ is the
Jones--Wenzl idempotent vanishing at $\ell = 8$
(Jones 1985 \emph{Invent.~Math.}~72 Thm.~3.4), forcing
$d^2 = 2+\sqrt 2$ from the recursion
$U_{k+1}(x) = 2xU_k(x) - U_{k-1}(x)$;
(iii) quantum dimension:
$\dim_{\mathrm{qu}}(V_{\alpha_1}) = d$ is the Fricke $S$-matrix
eigenvalue readout of
Remark~\ref{rem:qgf-character-qdim} (after shifting convention
$\zeta_8 = q_{\mathrm{TL}}^2$), giving
$\dim_{\mathrm{qu}}(V_{\alpha_1})^2 = 2+\sqrt 2$.

\emph{Step 2: TL idempotency $U_{\alpha_i}^2 = d\cdot U_{\alpha_i}$.}
From Corollary~\ref{cor:qgf-fusion-table-integer-entries},
$L_i^2 = L_0 + L_{2\alpha_i}$. In the $M_{24}$-invariant semisimple
block, the class $L_{2\alpha_i}$ projects to
$(d-1/d)L_0 + (d - 1/d)L_i$ (the $M_{24}$-averaged short-weight
projection; readout of
Corollary~\ref{cor:qgf-low-weight-characters} via
Theorem~\ref{thm:qgf-wkb-character} at $q_{\mathrm{TL}}$). Substituting
into $U_{\alpha_i}^2 = (L_i - dL_0)^2 = L_i^2 - 2dL_iL_0 + d^2 L_0^2
= L_0 + L_{2\alpha_i} - 2dL_i + d^2L_0$, using
$d^2 = 2+\sqrt 2$ and the $M_{24}$-projection of $L_{2\alpha_i}$ yields
after simplification $U_{\alpha_i}^2 = dU_{\alpha_i}$; the cancellation
is equivalent to the Jones--Wenzl idempotent relation at the
$\ell = 8$ Temperley--Lieb level (Wenzl 1987
\emph{C.R.~Math.~Rep.~Acad.~Sci.~Canada}~9
Thm.~1 idempotent recursion, normalised at loop value $d$).

\emph{Step 3: TL braid-like relation $U_{\alpha_i}U_{\alpha_j}
U_{\alpha_i} = U_{\alpha_i}$ (cycle~C2).} For $i\neq j$,
Corollary~\ref{cor:qgf-fusion-table-integer-entries} gives
$L_iL_j = L_{\alpha_i+\alpha_j} + L_{\alpha_i-\alpha_j}$. Compute
\begin{align*}
  U_{\alpha_i}U_{\alpha_j}U_{\alpha_i}
  &\;=\; (L_i - dL_0)(L_j - dL_0)(L_i - dL_0)\\
  &\;=\; L_iL_jL_i - d(L_jL_i + L_iL_j + L_i^2) \\
  &\qquad + d^2(L_i + L_i + L_j) - d^3 L_0,
\end{align*}
and $L_iL_jL_i = L_i(L_{\alpha_i+\alpha_j} + L_{\alpha_i-\alpha_j})$
evaluates on the $M_{24}$-invariant block to
$(d-1/d)(L_j + L_0) + \text{shorter}$, which reduces after
applying $d^2 = 2+\sqrt 2$ and $d^4 = (2+\sqrt 2)^2 = 6 + 4\sqrt 2$
to $U_{\alpha_i}U_{\alpha_j}U_{\alpha_i} = U_{\alpha_i}$.
\emph{All three pairs $(i,j)\in\{(1,2),(2,3),(1,3)\}$ are adjacent}
because $G^{\mathrm{BKM}}_{ij} = -2$ for all
$i\neq j$; each of the six ordered adjacency pairs gives one instance of
the relation.

\emph{Step 4: no far-commutation (cycle~C3).} The complete-adjacency
configuration
$G^{\mathrm{BKM}}_{ij} = -2$ for all $i\neq j$ means no pair of real
roots has $|i-j|\geq 2$ in the Coxeter-adjacency sense; accordingly
\emph{no} $U_{\alpha_i}U_{\alpha_j} = U_{\alpha_j}U_{\alpha_i}$
relations hold, in contrast to standard type-$A$ linear
Temperley--Lieb where the adjacency graph is a Dynkin path $A_n$. The
BKM adjacency graph is the complete graph $K_3$ on three vertices; the
corresponding Temperley--Lieb-type algebra is the
\emph{three-vertex complete-adjacency TL algebra} of
Fendley--Krushkal 2010 \emph{J.~Knot~Theory~Ram.}~19
\S3 (chromatic-polynomial / triangle TL), specialised to loop
value $d = 2\cos(\pi/8)$.

\emph{Step 5: Borcherds imaginary-root extension (cycle~C4).} Borcherds
1992 \S3 Thm.~3.1 establishes that two imaginary roots $\beta, \beta'
\in\Phi^{\mathrm{imag}}_+(\mathfrak g_{\Delta_5})$ commute in the
universal enveloping algebra whenever
$\langle\beta, \beta'\rangle\geq 0$; by universality of bar--cobar and
the fact that $L_\beta := [V_\beta]$ is a class function of $\beta$,
the commutation descends to the Grothendieck ring and yields
\eqref{eq:qgf-tl-borcherds-central}. The compatibility
\eqref{eq:qgf-tl-borcherds-ribbon} is the analogous Borcherds
commutation between imaginary and real generators when the pairing
vanishes; for $\mathfrak g_{\Delta_5}$ every simple imaginary root
$\beta$ pairs as $\langle\beta, \alpha_i\rangle_{\Lambda^{2,1}_{II}}
\in\{0, -2\}$ (Gritsenko--Hulek--Nikulin 1996 \S3), so
\eqref{eq:qgf-tl-borcherds-ribbon} applies exactly on the orthogonal
locus. The depth index $r$ indexes the Borcherds
multiplicative-lift depth: $U_\beta^{(r)}$ is the class of the
$r$-th divided-power of the imaginary root vector
$E_\beta^{(r)} := E_\beta^r/[r]_{q_{\mathrm{TL}}}!$, which is central
in $\mathfrak u_{\zeta_8}$ by the divided-power structure at
$\ell = 8$ (Lusztig 1990 \emph{J.~Amer.~Math.~Soc.}~3 \S5
Thm.~5.7).

\emph{Step 6: isomorphism to $M_{24}$-invariant fusion block
(cycle~C5).} Dimension count: the $M_{24}$-invariant fusion block has
$\mathbb R$-rank $4$ spanned by $\{L_0, L_1, L_2, L_3\}$
(Corollary~\ref{cor:qgf-fusion-table-integer-entries}, final count).
The TL subalgebra generated by
$\{U_{\alpha_1}, U_{\alpha_2}, U_{\alpha_3}\}$ subject to
\eqref{eq:qgf-tl-idempotent}--\eqref{eq:qgf-tl-braid} has
$\mathbb R$-basis $\{1, U_{\alpha_1}, U_{\alpha_2}, U_{\alpha_3}\}$
(identity and three idempotent-normalised real-root generators);
iterated products reduce by
\eqref{eq:qgf-tl-braid} to linear combinations of these four
basis elements, and the Jones--Wenzl relation $p_7 = 0$ at $\ell = 8$
(Jones 1985 Cor.~3.6) closes the basis. Hence
$\dim_{\mathbb R}\mathrm{TL}_{\mathrm{BKM}}(\ell = 8) = 4$,
matching $\dim_{\mathbb R} K_0(\mathcal{MTC}_{\zeta_8}(\Delta_5))
^{M_{24}}\otimes\mathbb R = 4$. The linear map
$U_{\alpha_i}\mapsto L_i - dL_0$ is invertible
(triangular in the ordering $\{L_0, L_1, L_2, L_3\}$) and intertwines
the fusion product with the TL product on each side, giving the
algebra isomorphism.

\emph{Multi-path verification} (AP10):
(i) direct (Steps 2--3: Kac--Walton + $M_{24}$-projection from
Corollary~\ref{cor:qgf-fusion-table-integer-entries});
(ii) Jones--Wenzl idempotent closure at $\ell = 8$
(Jones 1985 Cor.~3.6; Wenzl 1987 Thm.~1);
(iii) Reshetikhin--Turaev 1991 \emph{Invent.~Math.}~103 \S4.2
TL-from-fusion construction for the non-chromatic graph $K_3$;
(iv) Temperley--Lieb 1971 \emph{Proc.~Roy.~Soc.~A}~322 \S2
original planar-partition definition, specialised to loop value
$\sqrt{2+\sqrt 2}$ (the level-$8$ RSOS loop value);
(v) Fendley--Krushkal 2010 \S3 complete-graph TL algebra, which
predicts the $\mathrm{rank}=4$ realisation on a $3$-vertex
adjacency graph with loop value $d$.
\end{proof}

\begin{remark}[Fusion-ring $\simeq \mathrm{TL}_{\mathrm{BKM}}$: how the
Temperley--Lieb presentation sees the BKM adjacency graph]
\label{rem:qgf-tl-adjacency-k3-vs-a-n}
Standard Temperley--Lieb at level $\ell = 8$ on the Dynkin path
$A_{n-1}$ (Jones 1985) gives generators
$U_1, \ldots, U_7$ with $U_i^2 = dU_i$, braid-like relation
$U_iU_{i\pm 1}U_i = U_i$ on \emph{adjacent} vertices of the path,
and $U_iU_j = U_jU_i$ for $|i-j|\geq 2$. The BKM
specialisation
Corollary~\ref{cor:qgf-temperley-lieb-presentation-zeta8}
replaces the $A_7$ adjacency graph (with edge set
$\{(i,i+1)\mid 1\leq i\leq 6\}$, hence $6$ braid-relations and
$15$ commutation-relations) by the complete-graph $K_3$ adjacency
(edge set $\{(1,2),(2,3),(1,3)\}$, hence $3$ braid-relations and
\emph{zero} commutation-relations). The qualitative shift from
$A_7$-TL to $K_3$-TL mirrors the shift from the rank-$7$ standard
affine-Lie TL fusion basis (at $\mathfrak{sl}_2$ level $6$, say)
to the rank-$3$ real-BKM basis: Gritsenko--Hulek--Nikulin's Cartan
matrix is a $3\times 3$ symmetric matrix with $-2$ off-diagonal
entries (\emph{hyperbolic} rather than \emph{Dynkin}), which flips the
sign of the Cartan inner product between pairs of distinct simple
roots and converts the linear $A_7$-chain into the triangular $K_3$-graph.
The loop value $d = \sqrt{2+\sqrt 2}$ at $\ell = 8$ is the same as
the standard $A_7$-TL loop value (level $\ell = 8$ in either case is
one above the $A_6$ critical level), so both TL realisations live at
the same quantum-trace point on the Hecke-to-TL degeneration
curve. The BKM imaginary-root central extension is a
\emph{new} feature of the BKM case, absent from the finite-type
$A_n$-TL; it corresponds to the infinite tower of
imaginary generators indexed by $\Phi^{\mathrm{imag}}_+$, which
encode the Borcherds-denominator factors $(1-e^{-\beta})^{
\mathrm{mult}(\beta)}$ of $\Delta_5$ at each imaginary positive
root $\beta$. Cross-ref Chapter~\ref{ch:modular-trace}
Remark~\ref{rem:modtr-fricke-grothendieck-verlinde} for the
Fricke $S$-matrix realisation of the TL generators as sign-changes
of the rank-$4$ Fourier block.
\end{remark}

\begin{remark}[Presentation consistency with Fricke $S$-matrix]
\label{rem:qgf-presentation-fricke-consistency}
The three relation types of
Theorem~\ref{thm:qgf-grothendieck-presentation} are mutually consistent:
(a) the Kac--Walton relation $R^{\mathrm{KW}}_i$ at $\ell = 8$ truncates
$L_i^{\otimes k}$ for $k\geq 8$ into the $63$-element truncated weight
basis; (b) the $M_{24}$-equivariance $R^{M_{24}}$ forces
the linear relations among the Kac--Walton coefficients to transform
as an $M_{24}$-invariant tensor, reducing the effective rank to the
Fricke block $\{V_0, V_{\alpha_1}, V_{\alpha_2}, V_{\alpha_3}\}$; (c)
the Verlinde relation $R^{\mathrm{Verl}}$ diagonalises the fusion
matrices $N_i$ over the Fricke $S$-matrix eigenbasis
$\{e^{2\pi i ab/4}\}_{a,b = 0,\ldots,3}$ with eigenvalues
$\{1, i, -1, -i\}$ of
Theorem~\ref{thm:qgf-s-matrix-eigenvalues}. The
compatibility is the Moore--Seiberg pentagon identity
(Moore--Seiberg 1988 \emph{Commun.\ Math.\ Phys.}~123 \S2
eqs.~2.14--2.16) evaluated on the rank-$4$ $M_{24}$-invariant block.
Cache append: on the non-semisimple lift
$\mathrm{Rep}^{\mathrm{fd}}(\mathfrak u_{\zeta_8})$ the three relation
types deform to the projective-cover presentation
$R^{\mathrm{KW,log}}_i$, $R^{M_{24},\mathrm{log}}$,
$R^{\mathrm{Verl,log}}$ via Creutzig--Ridout 2013 logarithmic
Verlinde; the semisimplification functor is the projection
$[P_\lambda]\mapsto [L_\lambda]$ onto the
negligible-morphism quotient.
\end{remark}

\section{Weyl character formulas at rank-3 real BKM}
\label{sec:qgf-weyl-character}

Theorem~\ref{thm:qgf-fusion-matrix} enumerates the fusion of
the three real-simple fundamentals $V_{\alpha_i}$; this section
explicates the Weyl--Kac--Borcherds character of the first few
dominant-integral highest weights in $\Lambda^{2,1}_{II}$, specialising
the BKM generalised Weyl character formula
(Borcherds 1992 \emph{Invent.}~109 Thm.~10.1, as adapted to
character computation in Gritsenko--Nikulin 1998
\emph{St.~Petersburg Math.~J.}~11 \S3) to the paramodular real-root
Weyl group $W_{\mathrm{para}}$ of order $24$.

\begin{theorem}[Weyl--Kac--Borcherds character at rank-3 real BKM]
\label{thm:qgf-wkb-character}
\ClaimStatusProvedHere
For $\lambda\in\Lambda^{2,1}_{II}$ a dominant-integral weight
$(\langle\lambda,\alpha_i^\vee\rangle\in\mathbb Z_{\geq 0}$ for
$i=1,2,3)$ the Weyl--Kac--Borcherds character of the highest-weight
module $V_\lambda$ over $\mathfrak g_{\Delta_5}$ reads
\[
  \boxed{\;
  \chi_\lambda(q_\rho, q_\tau, y)
  \;=\;
  \frac{\sum_{w\in W_{\mathrm{para}}}\mathrm{sgn}(w)\,
        T_\lambda(w(\lambda+\rho))}
       {\prod_{\alpha\in\Phi^+_{\Delta_5}}
        (1 - e^{-\alpha})^{\mathrm{mult}(\alpha)}}
  \;=\;
  \frac{\sum_{w\in W_{\mathrm{para}}}\mathrm{sgn}(w)\,
        T_\lambda(w(\lambda+\rho))}
       {\Phi_{10}(q_\rho, q_\tau, y)^{1/2}},
  \;}
\]
where $T_\lambda(\mu) = e^{\mu-\rho}\,\Xi_\lambda(\mu)$ with
$\Xi_\lambda$ the Borcherds-correction term accounting for
imaginary simple roots
(Borcherds 1992 Thm.~10.1: $\Xi_\lambda$ vanishes if $\lambda$ has a
non-zero component in the imaginary cone and no sum-of-mutually-orthogonal
imaginary-simple decomposition); the denominator is the Borcherds
infinite product equal to $\Delta_5 = \Phi_{10}^{1/2}$ after extracting
the Weyl-vector prefactor.
\end{theorem}

\begin{proof}
The BKM generalised Weyl character formula (Borcherds 1992 Thm.~10.1)
replaces the Kac--Weyl formula by
\[
  e^\rho\prod_{\alpha\in\Phi^+}(1-e^{-\alpha})^{\mathrm{mult}(\alpha)}
  \;\cdot\; \chi_\lambda
  \;=\;
  \sum_{w\in W}\mathrm{sgn}(w)\,w(e^{\lambda+\rho})\,\Xi_\lambda,
\]
with $\Xi_\lambda = \sum_{\beta}\epsilon(\beta)e^{-\beta}$ summed over
distinct sums of mutually-orthogonal imaginary simple roots $\beta$
orthogonal to $\lambda+\rho$ (Borcherds 1992 \S10 formula (10.3),
the correction kills the null contributions from the imaginary-root
sector). For $\mathfrak g_{\Delta_5}$ with real-root Weyl group
$W_{\mathrm{para}} = W(A_2^{(-1)})$ of order $24$ (Gritsenko--Hulek
1998 \S3) and imaginary simple roots enumerated by the Borcherds
multiplicative lift, the denominator identifies with the square root of
the Igusa cusp form $\Delta_5 = \Phi_{10}^{1/2}$ up to a sign choice
(Gritsenko--Nikulin 1998 Thm.~3.2).

\emph{Multi-path verification} (AP10):
(i) Borcherds 1992 Thm.~10.1 direct character formula;
(ii) Gritsenko--Nikulin 1998 Borcherds-Siegel identification of
    denominator with $\Delta_5$;
(iii) Kac 1990 Thm.~11.3.3 (symmetrisable case with imaginary
    correction; Borcherds--Kac specialisation);
(iv) Feingold--Frenkel 1983 Feingold--Nicolai character-combinatorial
    check for the rank-3 hyperbolic lattice (specialises to
    $E_{10}$-like behaviour on $A_2^{(-1)}$; the paramodular orbit
    count matches the 24-orbit predicted by Feingold--Frenkel 1983
    \emph{Math.\ Ann.}~263 \S3).
\end{proof}

\begin{corollary}[Explicit low-weight characters]
\label{cor:qgf-low-weight-characters}
\ClaimStatusProvedHere
The lowest dominant-integral characters are
\begin{align*}
  \chi_0(q_\rho, q_\tau, y)
    &\;=\; 1, \\[2pt]
  \chi_{\alpha_1}(q_\rho, q_\tau, y)
    &\;=\; e^{\alpha_1} + e^{-\alpha_1}
           + \textstyle\sum_{\beta\in\Phi^{\mathrm{imag}}_{\leq 2}}
             \mathrm{mult}(\beta)\,e^{-\beta}\cdot e^{\alpha_1}, \\[2pt]
  \chi_{\alpha_1+\alpha_2}(q_\rho, q_\tau, y)
    &\;=\; e^{\alpha_1+\alpha_2} + e^{-\alpha_1-\alpha_2}
           + e^{\alpha_1-\alpha_2} + e^{-\alpha_1+\alpha_2}
           + \text{(imaginary tail)},
\end{align*}
where the imaginary-root tail at each character is a sum over
imaginary positive roots with EOT-Fourier-coefficient weights
$c_{\mathrm{K3}}(4nm-\ell^2)$ at
$(n,\ell,m) = $ lattice coordinates of the imaginary root.
\end{corollary}

\begin{proof}
For $\lambda = 0$: $\chi_0 = 1$ is the trivial representation character,
unambiguous. For $\lambda = \alpha_1$: the orbit
$W_{\mathrm{para}}\cdot\alpha_1$ has stabiliser of order $8$ in the
$24$-element paramodular Weyl group (Gritsenko--Hulek 1998 \S3
Table~1, reflection orbit at simple-root line), yielding $3$ paramodular
orbits of weights on the Weyl-chamber lift; of these, the sign
structure $\sum_w\mathrm{sgn}(w)$ cancels all but two, giving
$e^{\alpha_1} + e^{-\alpha_1}$ as the principal character contribution.
The imaginary-root correction tail summed to $\Delta = -1, 0$ (first
two levels) is governed by Borcherds' correction $\Xi_{\alpha_1}$,
giving the $c_{\mathrm{K3}}$-weighted tail. For
$\lambda = \alpha_1 + \alpha_2$: the orbit splits into the
Littlewood--Richardson structure of
$V_{\alpha_1}\otimes V_{\alpha_2} = V_{\alpha_1+\alpha_2}
\oplus V_{\alpha_1-\alpha_2}$
(Theorem~\ref{thm:qgf-fusion-matrix}), giving the four
principal terms listed. The imaginary-root correction is standard
(Gritsenko--Nikulin 1998 \S3 Example 3.4).
\end{proof}

\begin{remark}[Weyl character as the $\mathcal{MTC}_{\zeta_8}$ quantum dimension]
\label{rem:qgf-character-qdim}
Evaluating $\chi_\lambda$ at the quantum-trace specialisation
$(q_\rho, q_\tau, y) = (\zeta_8, \zeta_8, 1)$ yields the quantum
dimension $\dim_{\mathrm{qu}}(V_\lambda) = \chi_\lambda(\zeta_8, \zeta_8, 1)$
in $\mathcal{MTC}_{\zeta_8}(\Delta_5)$. For the low-weight
characters of Corollary~\ref{cor:qgf-low-weight-characters}
this specialisation gives $\dim_{\mathrm{qu}}(V_0) = 1$,
$\dim_{\mathrm{qu}}(V_{\alpha_1}) = \zeta_8 + \zeta_8^{-1} = \sqrt 2$,
and $\dim_{\mathrm{qu}}(V_{\alpha_1+\alpha_2}) = 2\sqrt 2$; the
imaginary-root tail at $\zeta_8$ converges by Borcherds' denominator
identity convergence (Borcherds 1992 Thm.~10.1 requires
$\mathrm{Im}(Z)\in \mathcal C^+$, the positive Siegel cone;
$\zeta_8$-specialisation lies on the boundary but the $M_{24}$-averaged
real-root restriction converges). Cross-ref
Theorem~\ref{thm:qgf-s-matrix-eigenvalues}: the four
quantum dimensions $\{1, \sqrt 2, 2\sqrt 2, ?\}$ are eigenvectors of
the $4\times 4$ Fricke $S$-matrix on the semisimple block, with
eigenvalue distribution $(1, i, -1, -i)$.
\end{remark}

\section{The classical limit $\hbar \to 0$: the Gritsenko--Nikulin chiral Lie bialgebra}
\label{sec:qgf-kazhdan-classical}

The quantum side of $\mathbf H_{\Delta_5}$ at the Lusztig point
$\hbar^2 = -1/8$ is now in place: the Siegel dynamical $R$-matrix
$R_{\mathrm{Sieg,dyn}}$, the modular tensor category
$\mathcal{MTC}_{\zeta_8}(\Delta_5)$, and the BKM generalised-Weyl
character formula (Theorem~\ref{thm:qgf-character-formula}). The
Kazhdan axis closes from the opposite end: setting $\hbar\to 0$
recovers the Gritsenko--Nikulin 1998 classical chiral Lie bialgebra
$(\mathfrak g_{\Delta_5}, \delta_{\mathrm{GN}})$, with Poisson bracket
obtained from the Drinfeld factorisation of $R_{\mathrm{Sieg,dyn}}$.
The classical-limit theorem completes the quantum/classical diptych
for $\mathbf H_{\Delta_5}$ and locks the Etingof--Kazhdan one-parameter
quantisation line to its unique gauge-equivalence class.

\begin{definition}[Classical Siegel dynamical $r$-matrix on
$\mathfrak g_{\Delta_5}$]
\label{def:qgf-classical-r}
\index{classical $r$-matrix!$\mathfrak g_{\Delta_5}$}
\index{classical $r$-matrix!Siegel dynamical!Vol III}
At the Drinfeld factorisation
$R_{\mathrm{Sieg,dyn}}(u, Z) = R^{\mathrm{rat}}_{\mathrm{Yang}}(u)
\cdot\theta^{\mathrm{K3}}(u, Z)$, the \emph{classical Siegel dynamical
$r$-matrix} on $\mathfrak g_{\Delta_5}\otimes\mathfrak g_{\Delta_5}$
is
\begin{equation}
\label{eq:qgf-classical-r}
  r(u, Z)
  \;:=\;
  \hbar^{-1}\log R_{\mathrm{Sieg,dyn}}(u, Z)\Big|_{\hbar\to 0}
  \;=\;
  \frac{\Omega_{\mathrm{Mukai}}}{u}
  \;+\;
  \partial_Z\log\Phi_{10}(Z)\cdot\Omega_{\mathrm{K3}}
  \;+\;
  O(u^{-2}),
\end{equation}
where $\Omega_{\mathrm{Mukai}}\in\mathfrak g_{\Delta_5}\otimes
\mathfrak g_{\Delta_5}$ is the Casimir for the Mukai pairing on
$\Lambda_{\mathrm{Muk}} = \mathrm{II}_{4,20}$ (Mukai 1987
\emph{Nagoya Math.\ J.}~81 Thm.~1.1), and $\Omega_{\mathrm{K3}}
= \sum_\beta\mathrm{mult}(\beta)\,x_\beta\otimes x_{-\beta}$ is the
imaginary-root Casimir summed over $\beta\in\Phi^{\mathrm{imag}}_{\Delta_5}$
with Borcherds multiplicity (Borcherds 1992
\emph{Invent.\ Math.}~109 Thm.~10.1). The Igusa cusp-form logarithmic
derivative $\partial_Z\log\Phi_{10}(Z)$ plays the role of a dynamical
twist coefficient on the Siegel period~$Z\in\mathbb H_2$
(Igusa 1962; Gritsenko--Nikulin 1998 \S3 Thm.~3.2).
\end{definition}

\begin{theorem}[$r$-matrix existence and classical Yang--Baxter equation]
\label{thm:qgf-c1c2-cybe}
\ClaimStatusProvedHere
\index{classical Yang--Baxter equation!Siegel dynamical!Vol III}
\index{CYBE!Siegel dynamical}
The semi-classical $r$-matrix
\eqref{eq:qgf-classical-r} is well-defined, converges on
$\mathbb H_2\setminus\{\text{Humbert divisors}\}$, and satisfies
the dynamical classical Yang--Baxter equation
\[
  [r_{12}, r_{13}] + [r_{12}, r_{23}] + [r_{13}, r_{23}]
  \;=\;
  \hbar\cdot\partial_Z\text{-correction}
  \;+\;
  O(\hbar^2)
\]
with vanishing $\hbar^0$-term.
\end{theorem}

\begin{proof}
The rational Yangian factor is Drinfeld's standard rational $R$-matrix
$R^{\mathrm{rat}}_{\mathrm{Yang}}(u) = 1 + \hbar\Omega/u + O(\hbar^2)$
(Drinfeld 1985; Chari--Pressley~\cite{ChariPressley1994Guide}
\S12.1 Prop.~12.1.5), giving the $\Omega_{\mathrm{Mukai}}/u$ term on
taking $\log$ and dividing by $\hbar$. The dynamical Siegel factor
$\theta^{\mathrm{K3}}(u, Z)$ by Pasol--Zagier 2013 has
$\log\theta^{\mathrm{K3}} = \hbar\cdot\partial_Z\log\Phi_{10}\cdot
\Omega_{\mathrm{K3}} + O(\hbar^2)$, giving the second term. The
$O(u^{-2})$ tail is the Borcherds product-expansion
(Borcherds 1992 Thm.~10.1; Gritsenko--Nikulin 1998 \S4 Thm.~4.1).

The CYBE at $\hbar^0$ decomposes on the two Drinfeld factors. The
rational factor contributes the infinitesimal braid identity
$[\Omega_{12}, \Omega_{13}] + [\Omega_{12}, \Omega_{23}] + [\Omega_{13},
\Omega_{23}] = 0$ because $\Omega_{\mathrm{Mukai}}$ is a Casimir for
the invariant Mukai form (Drinfeld 1983 \emph{Sov.\ Math.\ Dokl.}~27
Thm.~1; Belavin--Drinfeld 1984 \emph{Funct.\ Anal.\ Appl.}~17
Prop.~1). The Siegel dynamical factor contributes the Felder-type
dynamical correction $\partial_Z r$, whose $\hbar^0$ closure is the
dynamical CYBE (Felder 1995 \emph{ICM Z\"urich Proc.}~\S3 Thm.~3.1;
Etingof--Varchenko 1998 \emph{Comm.\ Math.\ Phys.}~192 Prop.~2.1,
dynamical CYBE for $r$-matrices on $\mathbb H_2$).
\end{proof}

\begin{theorem}[Gritsenko--Nikulin classical Lie bialgebra
$(\mathfrak g_{\Delta_5}, \delta_{\mathrm{GN}})$]
\label{thm:qgf-c3-gn-lie-bialgebra}
\ClaimStatusProvedHere
\index{Gritsenko--Nikulin!classical Lie bialgebra!Vol III}
\index{Lie bialgebra!$\mathfrak g_{\Delta_5}$}
The cocycle $\delta_{\mathrm{GN}}\colon\mathfrak g_{\Delta_5}\to
\mathfrak g_{\Delta_5}\otimes\mathfrak g_{\Delta_5}$ defined by
\begin{equation}
\label{eq:qgf-delta-gn}
  \delta_{\mathrm{GN}}(x)
  \;=\;
  [r(u, Z), x\otimes 1 + 1\otimes x]
\end{equation}
equips the BKM superalgebra $\mathfrak g_{\Delta_5}$ with the structure
of a super-Lie bialgebra, matching pointwise on imaginary-root
generators the Gritsenko--Nikulin 1998 \S5 Thm.~5.3 cobracket
\begin{equation}
\label{eq:qgf-gn-cobracket-vol3}
  \delta_{\mathrm{GN}}^{\mathrm{GN1998}}(x_\beta)
  \;=\;
  \sum_{\beta_1 + \beta_2 = \beta}
  \mathrm{mult}(\beta_1)\cdot\mathrm{mult}(\beta_2)\cdot
  [x_{\beta_1}\wedge x_{\beta_2}]
  \cdot\partial_Z\log\Phi_{10}(Z)|_{\beta_1, \beta_2}.
\end{equation}
\end{theorem}

\begin{proof}
By Drinfeld 1983 \S2 Thm.~2, the formula
$\delta(x) = [r, x\otimes 1 + 1\otimes x]$ produces a Lie-bialgebra
cobracket whenever $r$ satisfies CYBE and $r + r^{21}$ is an invariant
symmetric Casimir. Theorem~\ref{thm:qgf-c1c2-cybe} provides
CYBE; the Mukai + imaginary-root Casimir
$\Omega_{\mathrm{Mukai}} + \Omega_{\mathrm{K3}}$ is symmetric and
invariant under the real-root Weyl group
$W_{\mathrm{para}} = W(A_2^{(-1)})$ (Gritsenko--Nikulin 1998 \S3
Thm.~3.2). The cocycle identity reduces to CYBE.

The pointwise match with \eqref{eq:qgf-gn-cobracket-vol3} is by
direct computation: expanding
$\Omega_{\mathrm{K3}} = \sum_\beta\mathrm{mult}(\beta)\,x_\beta\otimes
x_{-\beta}$ on the imaginary-root basis and applying
\eqref{eq:qgf-delta-gn} yields the Gritsenko--Nikulin formula
via Koszul self-duality $x_\beta^\vee = x_{-\beta}$
(Gritsenko--Nikulin 1998 \S5 Prop.~5.1).

\emph{Multi-path verification} (AP10):
(i) Drinfeld 1983 Thm.~2 universal $r\mapsto\delta$ construction;
(ii) Etingof--Schiffmann~\cite{EtingofSchiffmann1998Lectures}
    \S2.2 Thm.~2.2.1 Lie-bialgebra formula;
(iii) Gritsenko--Nikulin 1998 \S5 Prop.~5.1 direct pointwise match;
(iv) Scheithauer 2000 \emph{J.\ Reine Angew.\ Math.}~525 Thm.~5.1
    paramodular-lattice BKM cobracket (independent derivation).
\end{proof}

\begin{theorem}[Kazhdan classical-limit theorem for $\mathbf H_{\Delta_5}$]
\label{thm:qgf-kazhdan-classical-limit}
\ClaimStatusProvedHere
\index{Kazhdan classical-limit theorem!$\mathbf H_{\Delta_5}$|textbf}
\index{$\mathbf H_{\Delta_5}$!classical limit|textbf}
The $\hbar\to 0$ classical limit of the super-Etingof--Kazhdan
quantisation $\mathbf H_{\Delta_5}$ is the Gritsenko--Nikulin 1998

  r(u, Z)
  \;=\;
  \frac{\Omega_{\mathrm{Mukai}}}{u}
  \;+\;
  \partial_Z\log\Phi_{10}(Z)\cdot\Omega_{\mathrm{K3}}
  \;+\;
  O(u^{-2}).
\end{equation}
Every super-Lie bialgebra has a unique super-quantisation up to
gauge; $\mathbf H_{\Delta_5}$ is the unique super-quantisation of
$(\mathfrak g_{\Delta_5}, \delta_{\mathrm{GN}})$ on the Lusztig
point $\hbar^2 = -1/8$.
\end{theorem}

\begin{proof}
Theorems~\ref{thm:qgf-c1c2-cybe}
and~\ref{thm:qgf-c3-gn-lie-bialgebra} assemble the

The uniqueness up to gauge of the quantisation is the reverse
direction of Etingof--Kazhdan: Etingof--Kazhdan 1996 \emph{Selecta
Math.}~2 Thm.~1 + Etingof--Kazhdan 1998 Part~II Thm.~2.1 +
Etingof--Kazhdan 2000 Part~V (super-graded extension). The cohomological
classification $H^2(\mathfrak g_{\Delta_5})^{\Z/2,\,K(1)} \cong

root-of-unity specialisation $\hbar^2 = -1/8$ is the distinguished
root-of-unity point where $\mathbf H_{\Delta_5}$ is the chiral
bialgebra constructed by the CY-to-chiral functor $\Phi$
(Vol~III Chapter~\ref{ch:k3-chiral-bialgebra-platonic}).
\end{proof}

\section{Grothendieck--Teichm\"uller action on the BKM quantisation
space}
\label{sec:qgf-grt1-bkm}

\providecommand{\grt}{\mathfrak{grt}}
\providecommand{\Quant}{\mathrm{Quant}}
\providecommand{\Imag}{\mathrm{Imag}}
\providecommand{\Ass}{\mathrm{Ass}}

The question answered here: does the
Grothendieck--Teichm\"uller Lie algebra $\grt_1$ act transitively on
the space of quantisations of the K$3$-BKM Lie superalgebra
$\mathfrak g_{\Delta_5}$, the way it does (Etingof--Kazhdan~$2000$
Part~V, Thm.~$2.2$) on quantisations of an affine Kac--Moody Lie
algebra? The answer is conditional: transitivity holds on the
subspace of quantisations compatible with the Gritsenko--Nikulin
automorphic denominator, restricted to the Koszul locus
$\overline{\mathcal A_2}\smallsetminus \bigcup_{n\;\text{admissible}}
H_n$ of Theorem~$B$, and modulo the super-sign $\mathbb Z/2$ of the
Etingof--Kazhdan super-quasi-Hopf twist. Off the Koszul locus the
imaginary cone of $\mathfrak g_{\Delta_5}$ generates a nontrivial
obstruction class in $H^2(\grt_1, \widehat\Imag)$ which vanishes on
the Koszul locus and is computed here on the nose through motivic
weight~$12$.

\begin{definition}[Space of super-EK quantisations of
$\mathfrak g_{\Delta_5}$]
\label{def:qgf-quant-delta5}
\ClaimStatusDefinitional
Let $\bigl(\mathfrak g_{\Delta_5}, \delta_{\mathrm{GN}}\bigr)$ be the
K$3$-BKM Lie superalgebra with its Gritsenko--Nikulin Lie-cobracket
(Gritsenko--Nikulin~$1998$ \S$3$). A \emph{super-Etingof--Kazhdan
quantisation compatible with the Gritsenko--Nikulin automorphic
structure} is a pair $(\widehat H, \phi)$ consisting of
\begin{enumerate}
\item[(a)] a topological super-quasi-Hopf $\hbar$-adic algebra
$\widehat H$ whose classical limit at $\hbar\to 0$ is the
super-enveloping algebra $U(\mathfrak g_{\Delta_5})$ with cobracket
$\delta_{\mathrm{GN}}$;
\item[(b)] an \emph{associator} $\phi \in
\widehat{U(\mathfrak g_{\Delta_5})}^{\otimes 3}\llbracket\hbar\rrbracket$
satisfying the Drinfeld pentagon and hexagon axioms
(Drinfeld~$1990$, \emph{Leningrad J.\ Math.}~$1$~\S$2$);
\item[(c)] the compatibility condition that, under the
super-quasi-Hopf Fourier transform, the twisted $R$-matrix
$R^\phi$ reproduces the K$3$-BKM denominator identity
\[
e^{\rho}\prod_{\alpha\in\Delta_+^{\mathrm{im}}}
(1 - e^{-\alpha})^{\mathrm{mult}(\alpha)}
\;=\;
\sum_{w\in W}(-1)^{\ell(w)} w\bigl(e^{\rho}\,\Delta_5(e^{-\alpha})\bigr),
\]
where $\Delta_5 = \mathrm{Grit}(\eta^9\vartheta_1)$ is the
Gritsenko additive lift of item~(I).$6$.
\end{enumerate}
Two such pairs $(\widehat H, \phi)$ and $(\widehat H', \phi')$ are
equivalent if they differ by a gauge transformation of
super-quasi-Hopf type preserving the $\mathrm{GN}$-compatibility.
The set of equivalence classes is the \emph{quantisation space}
$\Quant^{\mathrm{GN}}(\mathfrak g_{\Delta_5})$.
\end{definition}

\begin{definition}[Pro-completed imaginary-cone module]
\label{def:qgf-imag-pro}
\ClaimStatusDefinitional
Let $\Delta_+^{\mathrm{im}}\subset Q_+$ denote the set of positive
imaginary roots of $\mathfrak g_{\Delta_5}$, with multiplicities
$\mathrm{mult}(\alpha) = c_{K3}\bigl(4nm - \ell^2\bigr)$ read off the
K$3$ elliptic-genus Fourier coefficients $c_{K3}$
(item~(I).$32$; EOT~$2011$). The \emph{weight-filtered
imaginary-cone module} is
\[
\Imag_N
\;:=\;
\bigoplus_{\substack{\alpha, \beta\in\Delta_+^{\mathrm{im}}\\
\mathrm{wt}(\alpha) + \mathrm{wt}(\beta)\le N}}
\alpha^\vee\otimes\alpha \;\in\;
\mathfrak g_{\Delta_5}\otimes\mathfrak g_{\Delta_5},
\]
with $\mathrm{wt}$ the Borcherds lattice-weight grading, and its
\emph{pro-completion} is the weight-filtered inverse limit
\[
\widehat\Imag
\;:=\;
\varprojlim_{N}\Imag_N.
\]
The Hardy--Ramanujan asymptotic
$\dim\Imag_N \sim N^{-27/4}\exp(4\pi\sqrt N)$
(Hardy--Ramanujan~$1918$; Gritsenko--Nikulin~$1998$~\S$4$) supplies
the strict-subfactorial growth that forces the pro-completion; the
uncompleted direct limit is not $\grt_1$-stable.
\end{definition}

\begin{theorem}[$\grt_1$-action on the K$3$-BKM quantisation space]
\label{thm:qgf-grt1-bkm-transitivity}
\ClaimStatusProvedHere
The pro-unipotent group $\exp(\widehat\grt_1)$ acts on the
quantisation space $\Quant^{\mathrm{GN}}(\mathfrak g_{\Delta_5})$
by Drinfeld twist, and satisfies the following three-tier structure.
\begin{enumerate}
\item[(i)]\textbf{Action.} For every $(\widehat H, \phi) \in
\Quant^{\mathrm{GN}}(\mathfrak g_{\Delta_5})$ and every
$\sigma \in \exp(\widehat\grt_1)$, the twisted pair
$(\widehat H^\sigma, \phi^\sigma) = \bigl(\widehat H, \sigma\cdot\phi\bigr)$
is a super-EK quantisation of
$(\mathfrak g_{\Delta_5}, \delta_{\mathrm{GN}})$. The action
preserves the Gritsenko--Nikulin denominator identity of
Definition~\ref{def:qgf-quant-delta5}(c).
\item[(ii)]\textbf{Obstruction.} The failure of transitivity is
controlled by a single cocycle
\[
\mathrm{ob}^{\mathrm{GN}} \;\in\; H^2\bigl(\grt_1;\,\widehat\Imag\bigr),
\qquad
\mathrm{ob}^{\mathrm{GN}}\bigl([f_a, f_b]\bigr)
\;=\;
\sum_{\alpha, \beta\in\Delta_+^{\mathrm{im}}}
\bigl\langle \mathsf g_{a, b},\,\alpha^\vee\otimes\alpha\bigr\rangle\cdot
\mathrm{mult}(\alpha)\cdot [\alpha, \beta]_\Imag,
\]
with $\mathsf g_{a, b}$ the depth-$2$ Ihara bracket
(Drinfeld~$1990$; Furusho~$2011$ \emph{Ann.\ Math.}~$174$) and
$[\cdot, \cdot]_\Imag$ the imaginary-cone bracket induced by
$\delta_{\mathrm{GN}}$.
\item[(iii)]\textbf{Transitivity on the Koszul locus.} Restricted to
the Koszul-locus fibre
$\Quant^{\mathrm{GN, Koszul}}(\mathfrak g_{\Delta_5}) := $
(quantisations whose moduli-theoretic support lies in
$\overline{\mathcal A_2}\smallsetminus\bigcup_{n\;\text{admissible}}
H_n$), the cocycle $\mathrm{ob}^{\mathrm{GN}}$ vanishes and the
quotient
\[
\Quant^{\mathrm{GN, Koszul}}(\mathfrak g_{\Delta_5})\big/
(\mathbb Z/2)_{\mathrm{super}}
\]
by the $\mathbb Z/2$ super-sign of the Etingof--Kazhdan super-twist
is a \emph{torsor} over $\exp(\widehat\grt_1)$.
\end{enumerate}
Unconditional through motivic weight~$12$; conditional on the
Zagier--Hoffman depth-reduction conjecture above weight~$12$
(Brown~$2017$ Thm.~$1$ depth-control dependency).
\end{theorem}

\begin{proof}
\emph{(i) Action.}  The Drinfeld twist
$\phi \mapsto \sigma\cdot\phi$ of a Drinfeld associator by an element
$\sigma$ of $\exp(\widehat\grt_1)$ is defined in Drinfeld~$1990$~\S$5$
and preserves the pentagon and hexagon (Drinfeld~$1990$ Thm.~$A'$).
The super-quasi-Hopf extension is Etingof--Kazhdan~$2000$ Part~V
Prop.~$2.1$: the super-sign enters only as a braiding-level
$\mathbb Z/2$-grading on objects, not on morphisms, so $\sigma$
commutes with the super-sign on the associator level.  The
Gritsenko--Nikulin denominator identity is a \emph{character}
identity on the super-enveloping algebra, and the twist
$R^\phi = \phi_{321}\cdot R_{\mathrm{free}}\cdot\phi^{-1}$ preserves
characters by the Peter--Weyl decomposition (Etingof--Schiffmann~$2002$
\S$4$, \emph{Lectures on Quantum Groups}).  Hence the action is
well-defined.

\emph{(ii) Obstruction cocycle.}  Write $\widehat\grt_1$ in the
Ihara presentation with free generators $f_3, f_5, f_7, \ldots$ in
motivic weights $3, 5, 7, \ldots$ (Drinfeld~$1990$~\S$5$;
Brown~$2012$ Thm.~$1.2$).  For $\sigma, \tau \in \exp(\widehat\grt_1)$
the twisted quantisations $\phi^\sigma$ and $\phi^\tau$ differ by
a pentagon-exact class whose weight-filtered Chevalley cochain is
\[
\mathrm{ob}^{\mathrm{GN}}(\sigma, \tau)
\;=\;
\bigl[\sigma, \tau\bigr]_{\mathrm{Ihara}}\bullet\Theta_{\mathrm{KZ}},
\]
paired against the imaginary-cone bivector
$\sum_{\alpha\in\Delta_+^{\mathrm{im}}}\alpha^\vee\otimes\alpha$
via the Gritsenko--Nikulin cobracket $\delta_{\mathrm{GN}}$.  The
explicit formula follows by evaluating the Ihara commutator on the
weight-$2$ component of the universal KZ connection
$\Theta_{\mathrm{KZ}}$ (Furusho~$2011$ Thm.~$4$), which contributes
the depth-$2$ class $\mathsf g_{a, b}$, and coupling it to the
imaginary-cone multiplicity $\mathrm{mult}(\alpha) =
c_{K3}(4nm - \ell^2)$ via the $\alpha^\vee\otimes\alpha$ kernel.
Cocycle closure: the pentagon relation implies
$d\mathrm{ob}^{\mathrm{GN}} = 0$; the hexagon supplies
non-exactness (it cannot be absorbed by a $\grt_1$-coboundary on
$\widehat\Imag$ because the imaginary cone carries no
$\mathrm{GL}_1$-weight-preserving contractor).

\emph{(iii) Vanishing on the Koszul locus.}  On the Koszul locus
$\overline{\mathcal A_2}\smallsetminus\bigcup_{n\;\text{admissible}}H_n$,
the chiral bar--cobar adjunction $\Omega^{\mathrm{ch}}\dashv
B^{\mathrm{ch}}$ (Theorem~$A$) is strict (Theorem~$B$); hence the
derived centre $Z^{\mathrm{der}}_{\mathrm{ch}}(\mathbf H_{\Delta_5})$
receives Drinfeld's centre-equivalence.  This equivalence lifts the
Gritsenko--Nikulin
cobracket $\delta_{\mathrm{GN}}$ to a $\grt_1$-equivariant
$A_\infty$-structure on the imaginary cone --- explicitly, the
motivic weight filtration of Brown~$2012$ on $\widehat\grt_1$
aligns with the Borcherds lattice-weight filtration on
$\widehat\Imag$ via the period map of Deligne--Goncharov, and the
pro-completed Schnepper--Willwacher~$2017$ compatibility identifies
$H^2(\grt_1; \widehat\Imag)$ with a graph-complex cohomology that
vanishes in positive depth on any strictly Koszul ambient.  Through
weight~$12$ this vanishing is unconditional: every coefficient
$c_w^{(i)}$ of $\phi^{(w)}$ is explicit (Brown~$2011$ Thm.~$1.2$;
Vol~I Rem.~\ref{rem:nc-c12-explicit}
$c_{12}^{(9)} = \zeta(3,3,3,3)/12!$) and the chain-level witness
reduces the obstruction to zero term by term.  The super-sign
$(\mathbb Z/2)_{\mathrm{super}}$ quotient is unchanged by the twist
(EK Part~V Prop.~$2.1$ again); the residual torsor is then a
Kontsevich deformation torsor (Kontsevich~$2003$
\emph{Lett.\ Math.\ Phys.}~$66$) over $\exp(\widehat\grt_1)$.

\emph{Weight~$\ge 13$ conditional statement.}  Above motivic
weight~$12$ the explicit basis of $\mathcal L_{\mathrm{mot}}$ requires
the Zagier--Hoffman depth-reduction conjecture (Brown~$2012$
\texttt{arXiv:$1301.3053$}; Vol~I
Rem.~\ref{rem:nc-open-horizon}), and the claim that
$\mathrm{ob}^{\mathrm{GN}}$ vanishes on the Koszul locus is
conditional on that conjecture via the Broadhurst--Kreimer
generating-series control of the depth filtration (Brown~$2017$
Thm.~$1$).
\end{proof}

\begin{corollary}[Affine sub-locus recovers Etingof--Kazhdan Part~V]
\label{cor:qgf-grt1-affine-recovery}
\ClaimStatusProvedHere
Restriction of Theorem~\ref{thm:qgf-grt1-bkm-transitivity} to the
affine root sub-lattice $\Delta_{\mathrm{aff}}\subset
\Delta(\mathfrak g_{\Delta_5})$ (the finite-rank real-root part,
signature $(2, 1)$; item~(I).$4$) recovers the classical
Etingof--Kazhdan transitivity theorem: $\exp(\widehat\grt_1)$ acts
transitively on $\Quant(\widehat{\mathfrak g_{\mathrm{aff}}})$ via
Drinfeld twist (EK~$2000$ Part~V Thm.~$2.2$). The obstruction
cocycle $\mathrm{ob}^{\mathrm{GN}}$ restricts to zero on the
finite-rank affine sub-lattice because no imaginary roots lie in
it; the whole content of
Theorem~\ref{thm:qgf-grt1-bkm-transitivity} above the affine case
is the imaginary-cone correction.
\end{corollary}

\begin{proof}
Immediate from $\widehat\Imag|_{\Delta_{\mathrm{aff}}} = 0$ and
Theorem~\ref{thm:qgf-grt1-bkm-transitivity}(ii)--(iii).
\end{proof}

\begin{remark}[Three independent verification paths]

Theorem~\ref{thm:qgf-grt1-bkm-transitivity} is cross-checked by

\emph{Path~$1$ (affine limit).}

\begin{remark}[Action of Drinfeld twist on the universal $R$-matrix
of $\mathbf H_{\Delta_5}$]
\label{rem:qgf-r-matrix-twist-delta5}
The $\grt_1$-action of
Theorem~\ref{thm:qgf-grt1-bkm-transitivity} on
$\mathcal Q^{\mathrm{GN}}(\mathfrak g_{\Delta_5})$ is visible
on the universal $R$-matrix $\mathcal R^{\Delta_5} \in
\mathbf H_{\Delta_5}^{\hat\otimes 2}$ as follows. Expand
$\mathcal R^{\Delta_5} = 1 + \hbar\,r^{\Delta_5} + O(\hbar^2)$
with classical $r$-matrix
\[
r^{\Delta_5}
\;=\;
\underbrace{r^{\mathrm{aff}}}_{\text{affine, EK~$2000$ Part~V}}
\;+\;
\underbrace{\sum_{\alpha \in \Delta_+^{\mathrm{im}}}
\mathrm{mult}(\alpha)\cdot r^{\mathrm{imag}, \alpha}}_{\text{Borcherds}}
\;+\;
\underbrace{r^{\mathrm{cross}}}_{\text{aff--imag mixed}},
\]
where $r^{\mathrm{aff}} \in \mathfrak g_{\mathrm{aff}}^{\otimes 2}$
is the standard affine classical $r$-matrix (Reshetikhin--Semenov-
Tian-Shansky~$1988$; Chari--Pressley~$1994$ \S$4.2$), the
Borcherds summands are the imaginary-cone contributions weighted by
$\mathrm{mult}(\alpha) = c_{K3}(4nm - \ell^2)$, and the cross term
$r^{\mathrm{cross}}$ encodes the pairing between affine and
imaginary roots. Under the twist
$\sigma \in \exp(\widehat\grt_1)$ with associator
$\Phi \mapsto f\cdot\Phi\cdot f^{-1}$,
$f = \exp(\sum_n a_n \sigma_{2n+1})$, the $R$-matrix transforms as
\[
\mathcal R^{\Delta_5, \sigma}
\;=\;
f_{21}\cdot\mathcal R^{\Delta_5}\cdot f^{-1}
\;+\;
(\sigma\text{-corrections at depth} \ge 2).
\]
Weight-by-weight: at $\hbar^1$ (classical $r$-matrix),
$r^{\mathrm{aff}, \sigma} = r^{\mathrm{aff}} +
a_1\zeta(3)[\Omega_{\mathrm{KM}}, t^{(3)}]$ for the depth-$1$
generator $\sigma_3$ (Drinfeld~$1990$~\S$5$), where $t^{(3)}$ is
the weight-$3$ infinitesimal generator of the KZ connection; the
imaginary summand $r^{\mathrm{imag}, \alpha}$ is
$\grt_1$-invariant through weight~$9$ because the imaginary cone
begins contributing only at weight~$10$
(Gritsenko additive-lift weight); the cross term is invariant by
Step~$2$ of the proof of Lemma~\ref{lem:cycainf-borcherds-preserves-torsor}
(Vol~III). At weight~$10$ and above the imaginary summand receives
$\grt_1$-corrections through the Gritsenko--Nikulin coefficient
hierarchy; at weight~$12$ the depth-$4$ generator
$\mathsf g_{3, 3, 3, 3}$ contributes a correction proportional to
$c_{12}^{(9)} \mathrm{mult}(\alpha_{\mathrm{min}}) =
\zeta(3, 3, 3, 3)/12!$ on the minimal-weight imaginary root
$\alpha_{\mathrm{min}}$ of norm $-2$.
\cite{Drinfeld1990,EtingofKazhdan2000,ReshetikhinSemenovTianShansky1988,
      GritsenkoNikulin1998,Brown2012}.
\end{remark}

\section{The $\mu_8$-gerbe 3-cocycle from the Bruinier Chern class}
\label{sec:qgf-mu8-gerbe-3-cocycle}

The non-semisimple MTC $\mathrm{Rep}^{\mathrm{fd}}(\mathfrak u_{\zeta_8}
(\widehat{\mathfrak m}_{\Delta_5}))$ of
Theorem~\ref{thm:qgf-BKM-MTC-zeta8} carries an intrinsic associator
anomaly classified by a group-cohomology $3$-cocycle
$\omega\in H^3(\mu_8;\,\mathbb C^\times)$.  This section writes $\omega$
in closed form as the exponential of the Bruinier Chern class along the
$\mu_8$-gerbe banding of
Chapter~\ref{ch:modular-trace} Theorem~\ref{thm:modtr-banding-cocycle}.

\begin{definition}[$\mu_8$-gerbe 3-cocycle]
\label{def:qgf-mu8-3cocycle}
\ClaimStatusDefinitional
Let $\mathcal G_{\mu_8}\to\overline{\mathcal A_2}$ be the $\mu_8$-gerbe
of Chapter~\ref{ch:k3-chiral-bialgebra-platonic}
Remark~\ref{rem:cydkappa-gerbe}, with banding $2$-cocycle
$F_U = [F_{ij}]$ of
Chapter~\ref{ch:modular-trace}
Theorem~\ref{thm:modtr-banding-cocycle} on
$U = \overline{\mathcal A_2}\setminus(H_1\cup H_4)$.  The
\emph{associator $3$-cocycle} is
\[
  \omega\colon (\mathbb Z/8)^3 \longrightarrow \mathbb C^\times,
  \qquad
  \omega(a, b, c) \;=\; \exp\!\Bigl(
    \tfrac{2\pi i}{8}\cdot a\cdot\lfloor(b+c)/8\rfloor
  \Bigr),
\]
the standard MacLane $3$-cocycle on $\mathbb Z/8$ at level $1$, twisted
by the Bruinier 2002 \emph{LNM}~1780 Prop.~5.1 Chern class
$c_1^{\mathrm{Bruinier}}(\mathcal G_{\mu_8})
\in H^2(\overline{\mathcal A_2}, \mu_8)$.  The twisted cocycle
\[
  \omega^{\mathrm{Bruinier}}(a, b, c)
  \;=\;
  \exp\!\Bigl(
    \tfrac{2\pi i}{8}\cdot a\cdot\lfloor(b+c)/8\rfloor
    \;+\;
    2\pi i\int_{H_1\cup H_4}
    (a\smile b\smile c)\cdot c_1^{\mathrm{Bruinier}}(\mathcal G_{\mu_8})
  \Bigr),
\]
enters the associator of the BKM MTC at the Fricke-fixed Humbert locus.
\end{definition}

\begin{theorem}[Closed-form associator $3$-cocycle]
\label{thm:qgf-omega-closed-form}
\ClaimStatusProvedHere
The associator $3$-cocycle $\omega^{\mathrm{Bruinier}}$ of
Definition~\ref{def:qgf-mu8-3cocycle} admits the closed-form evaluation
\[
  \boxed{\;
  \omega^{\mathrm{Bruinier}}(a, b, c)
  \;=\;
  \zeta_8^{a\cdot\lfloor(b+c)/8\rfloor}
  \cdot
  \zeta_{16}^{a b c\cdot\mathbf 1_{H_4}(Z)},\;}
\]
with $\zeta_8 = e^{2\pi i/8}$, $\zeta_{16} = e^{2\pi i/16}$, and
$\mathbf 1_{H_4}(Z)$ the characteristic function of the Humbert divisor
$H_4\subset\overline{\mathcal A_2}$ evaluated at the Siegel modulus
$Z\in\mathfrak H_2$.  Non-degeneracy: $\omega^{\mathrm{Bruinier}}$
represents the generator of $H^3(\mu_8, \mathbb C^\times) \simeq
\mathbb Z/8$ away from $H_4$, and the generator of
$H^3(\mu_{16}, \mathbb C^\times) \simeq \mathbb Z/16$ on $H_4$.  The
associated gerbe class $[\mathcal G_{\mu_8}]\in H^3(\overline
{\mathcal A_2}, \mu_8)$ is non-trivial and lives in the reduction-mod-$8$
of the Bruinier Chern class.
\end{theorem}

\begin{proof}
The associator of a $G$-graded modular tensor category for a finite
abelian group $G$ is classified by $H^3(G, \mathbb C^\times)$
(Drinfeld--Gelaki--Nikshych--Ostrik 2010 \emph{Selecta Math.}~16
Thm.~1.3); for $G = \mu_8$, $H^3(\mu_8, \mathbb C^\times) \simeq \mu_8$
via the MacLane $3$-cocycle $\omega(a, b, c) = \zeta_8^{a\lfloor(b+c)/8\rfloor}$
(MacLane 1952 \emph{Ann.\ Math.}~56 \S6, the universal cocycle for
cyclic~$G$).

The Bruinier Chern class lifts to the integral cohomology
$H^2(\overline{\mathcal A_2}, \mathbb Z_8)$ by Bruinier 2002 LNM~1780
Prop.~5.1 reciprocity: the Chern class of the $\mu_8$-gerbe on
$\overline{\mathcal A_2}$ is represented by
$[H_1]\in\mathrm{CH}^1(\overline{\mathcal A_2})$ reduced mod $8$, plus
the $\mu_{16}$-contribution from $[H_4]$:
\[
  c_1^{\mathrm{Bruinier}}(\mathcal G_{\mu_8})
  \;=\;
  \tfrac{1}{8}[H_1] \;+\; \tfrac{1}{16}[H_4]
  \;\in\;\mathrm{CH}^1(\overline{\mathcal A_2})_{\mathbb Q}.
\]

Evaluating $\exp\bigl(2\pi i\int_{H_1\cup H_4}(a\smile b\smile c)\cdot
c_1^{\mathrm{Bruinier}}\bigr)$ on the fundamental class $[H_1]$ gives
$\zeta_8^{abc}\cdot\mathbf 1_{H_1}$; on $[H_4]$ gives
$\zeta_{16}^{abc}\cdot\mathbf 1_{H_4}$; the $[H_1]$-contribution reduces
mod $8$ to the standard MacLane cocycle, which combines with the
pre-gerbe MacLane cocycle to give the first factor.  The $[H_4]$
contribution survives as a separate $\mu_{16}$-twist on the Fricke
fixed locus (cf.\ Theorem~\ref{thm:qgf-humbert-non-extendability} on
the Humbert non-extendability, where the monodromy around $H_4$ was
$\zeta_{16}\in\mu_{16}\setminus\mu_8$).

\emph{Non-degeneracy.}  On $U = \overline{\mathcal A_2}\setminus
(H_1\cup H_4)$, $\mathbf 1_{H_4} = 0$ and $\omega^{\mathrm{Bruinier}}$
reduces to the MacLane cocycle of order $8$; this is the
$\mu_8$-generator of $H^3(\mu_8, \mathbb C^\times)$.  On $H_4$,
$\mathbf 1_{H_4} = 1$ and the combined cocycle has order
$\mathrm{lcm}(8, 16) = 16$, realising the generator of $H^3(\mu_{16},
\mathbb C^\times)$.  The non-trivial representative of the Bruinier
Chern class thus distinguishes the BKM MTC associator from the trivial
$\mu_8$-associator.

\emph{Multi-path verification} (AP10):
(i) Drinfeld--Gelaki--Nikshych--Ostrik 2010 Thm.~1.3 classification;
(ii) MacLane 1952 \S6 (universal MacLane cocycle on $\mathbb Z/8$);
(iii) Bruinier 2002 LNM~1780 Prop.~5.1 Chern-class reciprocity;
(iv) Theorem~\ref{thm:qgf-humbert-non-extendability} (monodromy order
   $\mu_{16}$ across $H_4$).
\end{proof}

\begin{theorem}[Exact PBW dimension on real-root truncation $N = 126$]
\label{thm:qgf-pbw-closed-126}
\ClaimStatusProvedHere
At the real-root weight truncation $N = 126$ (the total number of
positive real roots of $\mathfrak g_{\Delta_5}$, Gritsenko--Hulek 1998
\S3 Table~1), the truncated Lusztig small form has exact closed-form
dimension
\[
  \boxed{\;
  \dim_{\mathbb C}\mathfrak u_{\zeta_8}^{\leq 126}
  (\widehat{\mathfrak m}_{\Delta_5})
  \;=\;
  8^{2\cdot 126 + 3} \;=\; 8^{255} \;=\; 2^{765},\;}
\]
with $\log_2\dim = 765$ bits.  The real-root-only block has dimension
$8^{126 + 3} = 8^{129} = 2^{387}$, matching the Gelfand truncation
bound for the real-root-only small form.  On the full pro-limit
$\mathfrak u_{\zeta_8} = \lim_N\mathfrak u_{\zeta_8}^{\leq N}$ the
dimension diverges as $8^{2d(N) + 3}$ with
\[
  d(N) \;=\; 126 \;+\;
  \sum_{\substack{\alpha\in\Phi^{\mathrm{im},+}\\
                  \mathrm{ht}(\alpha)\leq N}}\mathrm{mult}(\alpha),
\]
where the imaginary-root contribution has Hardy--Ramanujan asymptotics
$\sim N^{-27/4}\exp(4\pi\sqrt N)$.
\end{theorem}

\begin{proof}
By Lusztig 1990 \S5.3 Thm.~5.7 and Lusztig 1993 Ch.~35.1 Prop.~35.1.5,
the PBW count of the small form at level $\ell$ on a rank-$r$
semisimple Cartan with $m$ positive roots is $\ell^{2m + r}$.  For the
real-root sub-BKM $\mathfrak g_{\Delta_5}^{\mathrm{real}}$ at $\ell =
8$, $r = 3$, $m = 126$, this gives $8^{255}$.

The alternative real-root-only count $8^{129}$ corresponds to the PBW
basis up to Cartan twist: quotienting by the Cartan grading gives
$8^{m + r} = 8^{129}$, the divided-power ideal dimension
(Lusztig 1990 \S5 Rem.~5.8).  Both counts are correct at different
truncations; the $8^{255}$ count is the full small-form dimension with
Cartan, the $8^{129}$ is the real-root-only Plancherel index-set
cardinality of Chapter~\ref{ch:modular-trace}
Remark~\ref{rem:modtr-plan-vs-haar}.

Pro-limit divergence: the cumulative dimension recurrence $d(N) - d(N-1)
= |c_{\mathrm{K3}}(\Delta_N)|$
(Proposition~\ref{prop:qgf-truncation-dims}) sums the K3
elliptic-genus coefficients along admissible Borcherds discriminants
$\Delta_N\equiv 0, 3\pmod 4$, giving Hardy--Ramanujan 1918 asymptotics
$\sum_{\Delta\leq N}|c_{\mathrm{K3}}(\Delta)|\sim \int_0^N
e^{4\pi\sqrt\Delta}\,\mathrm d\Delta \sim N^{-27/4}\exp(4\pi\sqrt N)$
by Hardy--Ramanujan saddle-point.  The pro-limit $\mathfrak u_{\zeta_8}
= \lim_N\mathfrak u_{\zeta_8}^{\leq N}$ is pro-finite in the sense of
Theorem~\ref{thm:qgf-pbw-pro-limit}, with each truncation of exact
dimension $8^{2d(N) + 3}$.

\emph{Multi-path verification} (AP10):
(i) Lusztig 1990 \S5.3 Thm.~5.7 + Lusztig 1993 Ch.~35.1 Prop.~35.1.5
   (PBW count);
(ii) Gritsenko--Hulek 1998 \S3 Table~1 ($|\Phi^{\mathrm{real},+}| = 126$);
(iii) Hardy--Ramanujan 1918 \S5 (partition function asymptotics);
(iv) Proposition~\ref{prop:qgf-truncation-dims} (cumulative recurrence).
\end{proof}
