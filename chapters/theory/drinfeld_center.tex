\chapter{The Drinfeld Center and Bulk Algebras}
\label{ch:drinfeld-center}

\begin{remark}[Drinfeld centre at \texorpdfstring{$d=3$}{d=3}: right adjoint, not averaging]
\label{rem:drinfeld-centre-right-adjoint-not-averaging}
At $d = 3$ the Drinfeld centre $\cZ(\Rep^{\Eone}(\Phi_3(\cA)))$ is the right
adjoint to the forgetful functor
$U \colon \cZ(\Rep^{\Eone}(\Phi_3(\cA))) \to \Rep^{\Eone}(\Phi_3(\cA))$, not an
averaging construction: there is no group or groupoid over which one
would average, and the half-braiding data is universal rather than
group-theoretic. The $\Etwo$-braided structure on the centre arises
from the Drinfeld-centre right adjoint and its universal half-braiding;
the Stage-1 $\Ethree\to\Etwo$ restriction supplies symmetric
factorisation data before specialisation, not the quantum-group
$R$-matrix. This braided datum does \emph{not} live natively on $\Phi_3(\cA)$ itself,
which is $\Eone$ per Proposition~\ref{prop:native-en-level}. The
assignment $\Phi_3(\cA) \mapsto \cZ(\Rep^{\Eone}(\Phi_3(\cA)))$ is the
boundary-to-bulk passage, and the $\Etwo$-braiding is an emergent
datum of the adjunction, not a hidden structure recovered on the
boundary algebra.
\end{remark}

The passage from boundary to bulk requires the Drinfeld center.
An $\Eone$-chiral algebra $A$ has a monoidal representation category
$\Rep^{E_1}(A)$; this category is not braided, because the $\Eone$
operad sees only one real direction: an ordered configuration in
$\R$ has no loop exchanging two points inside a fixed component. The
Drinfeld center $\cZ(\Rep^{E_1}(A))$
adds a braiding by its universal property. When $\Rep^{E_1}(A)$ is presentable
and dualizable and the chiral Hochschild complex computes the
ordinary derived bimodule endomorphisms of $A$ in the chosen
completion, the Ben-Zvi--Francis--Nadler/Lurie comparison identifies
this braided category with $\Rep^{E_2}(\Zder(A))$, where
$\Zder(A)=C^\bullet_{\mathrm{ch}}(A,A)$ is the chiral derived center
of Volume~I. Off that Koszul/coderived comparison surface the
Drinfeld center still exists, but it is not automatically computed by
the naive chiral Hochschild cochain complex.


\section{The Drinfeld center of a monoidal category}
\label{sec:drinfeld-center-def}

\begin{definition}[Drinfeld center]
\label{def:drinfeld-center}
Let $(\cC, \otimes, \mathbf{1})$ be a monoidal category. The
\emph{Drinfeld center} $\cZ(\cC)$ is the category whose objects are
pairs $(X, \beta_{X,-})$ where:
\begin{enumerate}[label=(\roman*)]
 \item $X \in \cC$ is an object;
 \item $\beta_{X,-} \colon X \otimes (-) \xrightarrow{\sim} (-) \otimes X$
 is a natural isomorphism (the \emph{half-braiding});
 \item the half-braiding satisfies the hexagon axiom: for all
 $Y, Z \in \cC$,
 \[
 \beta_{X, Y \otimes Z}
 = (\id_Y \otimes \beta_{X,Z}) \circ (\beta_{X,Y} \otimes \id_Z).
 \]
\end{enumerate}
A morphism $(X, \beta_X) \to (Y, \beta_Y)$ is a morphism $f \colon X \to Y$
in $\cC$ that intertwines the half-braidings. The Drinfeld center is
always braided monoidal, with braiding
$\sigma_{(X,\beta_X),(Y,\beta_Y)} = \beta_{X,Y}$.
\end{definition}

\begin{proposition}[Universal property]
\label{prop:drinfeld-center-universal}
The Drinfeld center $\cZ(\cC)$ is the universal braided monoidal
category receiving a central (i.e., braiding-compatible) monoidal
functor from $\cC$. The forgetful functor
$U \colon \cZ(\cC) \to \cC$ is faithful and monoidal.
\end{proposition}

\begin{example}[Finite group]
\label{ex:drinfeld-center-finite-group}
For $\cC = \Rep(G)$ with $G$ a finite group, the Drinfeld center
$\cZ(\Rep(G)) \simeq \Rep(D(G))$ is the representation category of
the Drinfeld double $D(G) = \C[G] \bowtie \C(G)$. When $G$ is
abelian, $D(G) \simeq \C[G \times \hat{G}]$ and
$\cZ(\Rep(G)) \simeq \Rep(G \times \hat{G})$, the modular
tensor category governing the $G$-Dijkgraaf--Witten theory.
\end{example}

\begin{example}[Quantum group]
\label{ex:drinfeld-center-quantum-group}
For $\cC = \Rep_q(\frakg)$ at generic $q$, the Drinfeld center
$\cZ(\Rep_q(\frakg)) \simeq \Rep_q(\frakg) \boxtimes
\Rep_{q^{-1}}(\frakg)$, recovering the Drinfeld double
$D(\Uq(\frakg)) \simeq \Uq(\frakg) \otimes \Uq(\frakg^{\op})$.
At roots of unity, $\cZ(\cC_q) = \cC_q \boxtimes \overline{\cC_q}$
is the Drinfeld center of the modular tensor category, and the
$S$-matrix of $\cZ(\cC_q)$ governs the Reshetikhin--Turaev
invariant of the torus $T^2$.
\end{example}


\section{Drinfeld center as chiral derived center}
\label{sec:drinfeld-chiral-center}

The categorical Drinfeld center and the algebraic chiral derived center
are identified by the Ben-Zvi--Francis--Nadler theorem.

\begin{theorem}[Ben-Zvi--Francis--Nadler; Lurie]
\label{thm:bzfn}
Let $A$ be an $\Eone$-algebra in a symmetric monoidal presentable
stable $\infty$-category, and assume the $A$-module category is
dualizable in the Morita $(\infty,2)$-category. Then there is an
equivalence of braided monoidal $\infty$-categories
\[
 \cZ(\Rep^{E_1}(A))
 \;\simeq\;
 \Rep^{E_2}(\Zder(A))
\]
where $\Zder(A) = \RHom_{A\text{-}\mathrm{bimod}}(A, A)$ is the
derived center (Hochschild cochains) of $A$.
\end{theorem}

\begin{corollary}[Chiral derived center presents the Drinfeld center on the Koszul locus]
\label{cor:zder-drinfeld}
Let $A$ be an augmented $\Eone$-chiral algebra on a smooth curve.
Assume:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item the Vol.~I chiral bar-cobar counit
 $\Omegach(B(A))\to A$ is a quasi-isomorphism, or a coderived
 equivalence in the stated completion;
 \item $\Rep^{\Eone}(A)$ is presentable and dualizable in the Morita
 $(\infty,2)$-category;
 \item the chiral Hochschild cochain complex computes
 $\RHom_{A\text{-}\mathrm{bimod}}(A,A)$ in that same completion.
\end{enumerate}
Then the chiral derived center
$\Zder(A) = C^\bullet_{\mathrm{ch}}(A, A)$ of Volume~I presents the
Drinfeld center:
\[
 \cZ(\Rep^{\Eone}(A)) \;\simeq\; \Rep^{\Etwo}(\Zder(A)).
\]
The equivalence is at the level of braided monoidal $\infty$-categories:
the category $\cZ(\Rep^{\Eone}(A))$ on the left, its modules-over-cochains
presentation on the right. The cochain complex
$C^\bullet_{\mathrm{ch}}(A, A)$ is the $E_2$-algebra whose category of
modules realises the centre.
\end{corollary}

\begin{proof}
The chiral derived center is
$\Zder(A)=C^\bullet_{\mathrm{ch}}(A,A)$. Hypothesis~(i) replaces
the bar-cobar model by $A$ in the same homotopy theory in which
the cochains are computed, and hypothesis~(iii) gives
\[
 C^\bullet_{\mathrm{ch}}(A,A)
 \simeq \RHom_{A\otimes A^{\op}}(A,A)
 = \HH^\bullet(A).
\]
Hypothesis~(ii) is the Morita-dualizability condition required by
Theorem~\ref{thm:bzfn}. Applying that theorem gives
$\cZ(\Rep^{\Eone}(A))\simeq\Rep^{\Etwo}(\Zder(A))$.
\end{proof}

\begin{remark}[Scope of the BZFN/Morita center statement]
\label{rem:center-not-swiss-cheese}
Corollary~\ref{cor:zder-drinfeld} is the BZFN/Morita-center statement:
modules over the derived Hochschild cochain algebra present the
Drinfeld centre as a braided monoidal category. The two-coloured
logarithmic Swiss-cheese operad action on $(\Zder(A), A)$ and the
interval/slab compactification theorem are separate constructions,
each requiring an explicit dg-operad map from the relevant open-closed
Fulton-MacPherson chains to $\End(\Zder(A), A)$ with signs,
completions, spectral translation, boundary strata, and gluing
checked. The derived-center presentation of the Drinfeld centre
established here is the input on which those further constructions
build.
\end{remark}

\begin{remark}[Five objects attached to \texorpdfstring{$A$}{A}]
\label{rem:three-functors-drinfeld}
The boundary algebra, its bar construction, its Koszul cohomology,
its Verdier--Koszul dual, and its derived chiral center are distinct
objects:
\begin{enumerate}[label=(\alph*)]
 \item \emph{Boundary algebra}: $A$.
 \item \emph{Bar coalgebra}: $B(A)$, with
 $\Omega(B(A))\simeq A$ on the bar--cobar comparison locus.
 \item \emph{Koszul cohomology coalgebra}:
 $A^i = H^\bullet(B(A))$.
 \item \emph{Verdier--Koszul dual algebra}:
 $A^! = D_{\Ran}(A^i)$; in a finite strict model this is
 $H^\bullet(B(A))^\vee$.
 \item \emph{Derived chiral center}:
 $\Zder(A)=C^\bullet_{\mathrm{ch}}(A,A)$, and on the
 Koszul/coderived comparison surface
 $\Zder(A)\simeq\RHom_{A\otimes A^{\op}}(A,A)$
 (the universal bulk algebra).
\end{enumerate}
The Drinfeld center $\cZ(\Rep^{E_1}(A))$ is
$\Rep^{E_2}(\Zder(A))$ under the BZFN hypotheses: modules over
object~(e) realise the centre, with bar-cobar inversion on the
boundary side carrying objects~(a)--(d) on their own constructions.
Bar-cobar inversion recovers the boundary; the Drinfeld centre
presents the bulk.
\end{remark}

\begin{remark}[Three centers: categorical, algebraic, and symmetric]
\label{rem:three-centers-sharp}
\label{rem:three-centers}
Three operations called ``center'' play fundamentally different roles.

\begin{center}
\begin{tabular}{llll}
\toprule
 & Drinfeld $\cZ(\cC)$ & Derived $\HH^\bullet(B,B)$
 & M\"uger $\cZ_2(\cC)$ \\
\midrule
\emph{Input} & $\Eone$-monoidal $\cC$ & $\En$-algebra $B$
 & braided monoidal $\cC$ \\
\emph{Output} & $\Etwo$-braided category & $E_{n+1}$-algebra
 & full subcategory of $\cC$ \\
\emph{Mechanism} & half-braidings & Hochschild cochains
 & trivial double braiding \\
\emph{Promotion} & $\Eone \to \Etwo$ & $\En \to E_{n+1}$
 & \textbf{none}: detects obstructions \\
\bottomrule
\end{tabular}
\end{center}

\noindent\textbf{(a) Drinfeld center $\cZ(\cC)$: categorical,
$\Eone \to \Etwo$.}
Given an $\Eone$-monoidal category $\cC$, the Drinfeld center
$\cZ(\cC)$ universally adjoins half-braidings to produce a braided
monoidal category. It is intrinsically a one-step promotion:
$\Eone$-monoidal input, $\Etwo$-braided output. The mechanism is
purely categorical (no cochains, no resolution).

\noindent\textbf{(b) Derived center $\HH^\bullet(B,B)$: algebraic,
$\En \to E_{n+1}$.}
For an $\En$-algebra $B$, the Hochschild cochain complex
$\HH^\bullet(B,B) = \RHom_{B \otimes B^{\op}}(B,B)$ carries an
$E_{n+1}$-structure by the higher Deligne conjecture (Lurie,
\emph{Higher Algebra} Theorem~5.3.1.30). The underlying operadic
identification $\En \otimes \Eone \simeq E_{n+1}$ is Dunn additivity,
but the promotion of \emph{algebras} via Hochschild cochains is the
higher Deligne theorem proper. \textup{(Caveat: $E_3$ on cochains
with full BV refinement requires $B$ at least $E_\infty$; for
$\Eone$-input the cochains carry only $\Etwo$ by the classical
Deligne conjecture.)} This promotion iterates: applied to an
$\Etwo$-algebra $B$ with sufficient commutativity, $\HH^\bullet(B,B)$
is genuinely $E_3$.

\noindent\textbf{Agreement at $\Eone \to \Etwo$ (BZFN).}
At the lowest step, (a) and (b) coincide: the BZFN theorem
(Theorem~\ref{thm:bzfn}) gives
$\cZ(\Rep^{\Eone}(A)) \simeq \Rep^{\Etwo}(\HH^\bullet(A,A))$.
This is the identification used throughout this chapter.

\noindent\textbf{Divergence at $\Etwo \to E_3$.}
The two operations diverge at the next step. For a braided category
$\cB$, the iterated Drinfeld centre satisfies
$\cZ(\cB) \simeq \cB \boxtimes \cB^{\mathrm{rev}}$, the Deligne
product with the reverse braiding: an $\Etwo$-doubling at the same
operadic level. The algebraic derived centre $\HH^\bullet(B, B)$
promotes to $E_3$ when the input is sufficiently commutative, by the
higher Deligne theorem. The Drinfeld centre repeats at the
$\Etwo$-doubling level; the derived centre climbs the $\En$-tower
through Hochschild cochains.

\noindent\textbf{(c) M\"uger center $\cZ_2(\cC)$: modularity detector.}
For a braided $\cC$, the M\"uger center
$\cZ_2(\cC) = \{X \in \cC : \sigma_{Y, X} \circ \sigma_{X, Y} = \mathrm{id}_{X \otimes Y}\;
\forall\, Y\}$ is the subcategory of transparent objects: those
commuting symmetrically with every $Y \in \cC$. The M\"uger centre
records the modularity status of the braiding through its size:
$\cZ_2(\cC) \simeq \mathrm{Vect}$ corresponds to a non-degenerate
braiding (modularity prerequisite), and larger M\"uger centre records
larger transparent locus inside $\cC$. The construction stays inside
$\cC$ as a sub-category, supplying a modularity invariant rather than
an operadic promotion.

\noindent\emph{References.}
Drinfeld center universality: Lurie, \emph{Higher Algebra},
\S5.3.1. BZFN identification: Ben-Zvi--Francis--Nadler,
\emph{Integral transforms and Drinfeld centers} (2010).
M\"uger center: M\"uger, \emph{Galois theory for braided
tensor categories} (2003).
\end{remark}
\subsection{The \texorpdfstring{$\infty$}{inf}-categorical BZFN identification}
\label{subsec:infty-cat-bzfn}

The BZFN theorem (Theorem~\ref{thm:bzfn}) is an equivalence of
\emph{$\infty$-categories}, not merely ordinary categories. The
$\infty$-categorical content is essential for class~$M$ algebras
and is invisible to the $1$-categorical truncation.

\begin{proposition}[\texorpdfstring{$\infty$}{inf}-categorical BZFN, chiral version]
\label{prop:infty-cat-bzfn}
For $A$ an $\Eone$-chiral algebra on $\C \times \R$, the BZFN
equivalence
\begin{equation}
\label{eq:infty-cat-bzfn}
 \cZ(\mathrm{ChMod}^{\Eone}(A))
 \;\simeq\;
 \mathrm{ChMod}^{\Etwo}(C^\bullet_{\mathrm{ch}}(A, A))
\end{equation}
holds as an equivalence of $\Etwo$-monoidal $\infty$-categories,
where:
\begin{enumerate}[label=(\roman*)]
 \item $\mathrm{ChMod}^{\Eone}(A)$ is the $\infty$-category of
 chiral $A$-modules in $\mathrm{IndCoh}(\Ran(\C \times \R))$,
 equipped with $\Eone$-monoidal structure from the ordered
 factorization product;
 \item $C^\bullet_{\mathrm{ch}}(A, A) = \RHom_{A \otimes A^{\op}}(A, A)$
 is the chiral Hochschild cochain complex (Volume~I, Theorem~H),
 which is an $\Etwo$-algebra by the higher Deligne conjecture
 \textup{(}Lurie, \emph{Higher Algebra},
 Theorem~5.3.1.30\textup{)};
 \item the Drinfeld center $\cZ(-)$ takes the $\infty$-category
 of objects equipped with coherent half-braidings: the half-braidings
 form a \emph{space} (an $\infty$-groupoid) rather than a set,
 encoding the full $\Etwo$-coherence tower.
\end{enumerate}
\end{proposition}

\begin{proof}
The ambient $\infty$-category
$\cS = \mathrm{IndCoh}(\Ran(\C \times \R))$ is presentably
symmetric monoidal and stable. The $\Eone$-chiral algebra $A$ is
an $\Eone$-algebra in $\cS$. The abstract BZFN theorem
(Lurie, \emph{HA}, 5.3.1.30) gives
$\cZ(\mathrm{LMod}_A(\cS)) \simeq
\mathrm{LMod}_{\HH^\bullet(A,A)}(\cS)$
as $\Etwo$-monoidal $\infty$-categories. The identification
$\mathrm{LMod}_A(\cS) = \mathrm{ChMod}^{\Eone}(A)$ is the
definition of chiral modules; the identification
$\HH^\bullet(A, A) = C^\bullet_{\mathrm{ch}}(A, A)$ follows from
bar-cobar inversion (Volume~I, Theorem~B):
$\Omegach(B(A)) \simeq A$ on the Koszul locus identifies
$A$-bimodules with $B(A)$-comodules, giving
$\RHom_{A \otimes A^{\op}}(A, A) = C^\bullet_{\mathrm{ch}}(A, A)$.
\end{proof}

\begin{remark}[\texorpdfstring{$\infty$}{inf}-categorical content by shadow class]
\label{rem:infty-cat-shadow-class}
The $\infty$-categorical content beyond the ordinary ($1$-categorical)
Drinfeld center depends on the shadow class of~$A$:
\begin{center}
\begin{tabular}{llll}
\toprule
Shadow class & Half-braiding space & $\infty$-correction & BZFN regime \\
\midrule
$G$ (polynomial) & discrete ($\pi_n = 0$, $n \geq 1$) & none & exact \\
$L$ (rational) & finite CW complex & finitely many & proved \\
$C$ (convergent) & $\infty$-dim, convergent & countably many & proved \\
$M$ (Gevrey-1) & $\infty$-dim, divergent & Borel-summable & conjectural \\
\bottomrule
\end{tabular}
\end{center}
For class~$G$ algebras (e.g., the Heisenberg $H_k$), the
half-braiding space $\mathrm{Map}_{\Etwo}(M \otimes {-},\, {-} \otimes M)$
is \emph{discrete}: all higher homotopy groups vanish, and
the $\infty$-categorical and $1$-categorical Drinfeld centers agree.
This is why the rank-$1$ and rank-$24$ Heisenberg computations
(Remark~\ref{rem:drinfeld-center-heisenberg-verification},
Remark~\ref{rem:drinfeld-vs-derived-k3e}) are exact.

For class~$M$ algebras (e.g., Virasoro, $\cW_{1+\infty}$), the
half-braiding space has infinite homotopical depth. The higher
homotopy groups $\pi_n$ of the half-braiding space encode the
$A_\infty$-corrections $\delta^{(k)}$ to the coproduct: these are
the \emph{same} data as the shadow invariants~$S_k$ of the
$A_\infty$ bar complex. Truncation to the $1$-category (taking
$\pi_0$ of Hom-spaces) destroys the full correction tower and
loses the chain-level content (Miki automorphism, tetrahedron
corrections) that the physics requires.
\end{remark}

\begin{remark}[The TCFT carrier and the raw \texorpdfstring{$B^{(2)}$}{B2} term]
\label{rem:tcft-argument}
The chain-level content of the BZFN identification is encoded by the
TCFT (topological conformal field theory) structure on the Hochschild
complex. On $C^\bullet(A,A)$, the classical mixed-complex operators are:
\begin{itemize}
 \item $b$: the Hochschild differential (degree~$+1$);
 \item $B$: the Connes boundary operator (degree~$-1$), the
 infinitesimal generator of the $S^1$-action on the loop space
 $\cL X = \mathrm{Map}(S^1, X)$;
 \item $B_{\mathrm{TCFT}}^{(2)}$: Costello's corrected secondary
 TCFT carrier, not the raw bar term $B_{\mathrm{term}}^{(2)}$.
\end{itemize}
For the total $A_\infty$ Hochschild differential
$b_{\mathrm{tot}}=\sum_{k\geq 1} b_k$, Costello's TCFT package gives
the total coherence relation
\begin{align*}
 b_{\mathrm{tot}}^2 &= 0, \\
 \{b_{\mathrm{tot}},B\} &= 0, \\
 \{b_{\mathrm{tot}},B_{\mathrm{TCFT}}^{(2)}\}
  + \{B,B_{\mathrm{TCFT}}^{(1)}\} + \cdots &= 0.
\end{align*}
This is a statement about the corrected TCFT operator and the total
moduli-chain differential. It is not the termwise identity
$\{b_k,B_{\mathrm{term}}^{(2)}\}=0$ and it is not a proof that the raw
bar term equals Costello's carrier. The strict witness
\[
 [m_3,B_{\mathrm{term}}^{(2)}]\,[a|a|a|a|b]
 \;=\; 2\alpha[b]\;\neq\;0
\]
rules out the per-$m_3$ cancellation. Hence a compact non-formal
CY$_3$ closure cannot be inferred from cyclic invariance alone: it
requires either a comparison
$B_{\mathrm{term}}^{(2)}\to B_{\mathrm{TCFT}}^{(2)}$ or a complete
$\HH^{-2}_{E_1}$ filtration theorem for the chosen strictified
Hochschild model. In the chiral setting, the corrected TCFT coherence
is the compatibility of the half-braiding intertwiner
$\beta_{M,N}(z) \colon M(z_1) \otimes N(z_2) \to N(z_2) \otimes M(z_1)$
with the chiral differential on the Ran-space factorisation algebra.
\end{remark}

\subsection{Construction: the \texorpdfstring{$R$}{R}-matrix from the Drinfeld center}
\label{subsec:rmatrix-from-center}

The Drinfeld center is defined (Definition~\ref{def:drinfeld-center})
and identified with the derived center
(Theorem~\ref{thm:bzfn}, Corollary~\ref{cor:zder-drinfeld}).
The mechanism by which these structures produce the quantum group
$R$-matrix is the following: the $R$-matrix \emph{is} the
half-braiding, not a consequence of it. At $d = 3$ this half-braiding
is centre-side data. The boundary algebra $A$ is $\Eone$, while the
braided object is $\cZ(\Rep^{\Eone}(A))$, or under the BZFN comparison
hypotheses $\Rep^{\Etwo}(\Zder(A))$; it is not a native $\Etwo$
structure on $A$ itself.

\begin{construction}[\texorpdfstring{$R$}{R}-matrix from the Drinfeld center]
\label{constr:rmatrix-from-center}
Let $A$ be an $\Eone$-chiral algebra on $X \times \R$. The
quantum group $R$-matrix is constructed in seven steps.

\medskip
\noindent\textbf{Step~1.}
\textit{$A$ is an $\Eone$-chiral algebra.}
The output of the CY-to-chiral correspondence $\Phi_3$ at $d = 3$ is an
$\Eone$-algebra (Theorem~CY-A$_3$,
Theorem~\ref{thm:cy-to-chiral-d3}).
It is \emph{not} $\Etwo$: the $E_3$-framing on $\HH_\bullet(\cC)$
restricts to symmetric braiding
($\pi_1(\Conf_2(\R^3)) = 0$; Proposition~\ref{prop:e3-symmetric-braiding}),
and the descent to $\Eone$ via the $\Omega$-deformation
discards this symmetric braiding, retaining only the ordered
(associative) product.

\medskip
\noindent\textbf{Step~2.}
\textit{$\Rep^{\Eone}(A)$ is monoidal, not braided.}
The $\Eone$-factorization structure on $A$ gives the
representation category $\Rep^{\Eone}(A)$ an \emph{ordered}
tensor product: $M \otimes N \neq N \otimes M$ in general.
The monoidal structure is the ordered factorization product
on $\Ran^{\mathrm{ord}}(X)$ (Volume~I, \S4). There is no
braiding: an ordered configuration of two points on a real line has no
exchange loop inside a fixed component.

\medskip
\noindent\textbf{Step~3.}
\textit{The Drinfeld center universally adjoins half-braidings.}
The Drinfeld center $\cZ(\Rep^{\Eone}(A))$ has objects
$(M, \sigma_M)$ where $M \in \Rep^{\Eone}(A)$ and
\begin{equation}
\label{eq:half-braiding-def}
 \sigma_M \colon M \otimes (-) \xrightarrow{\;\sim\;} (-) \otimes M
\end{equation}
is a natural isomorphism satisfying the hexagon axiom
(Definition~\ref{def:drinfeld-center}):
\begin{equation}
\label{eq:hexagon-half-braiding}
 \sigma_{M,\, N \otimes P}
 = (\id_N \otimes \sigma_{M,P}) \circ (\sigma_{M,N} \otimes \id_P)
\end{equation}
for all $N, P \in \Rep^{\Eone}(A)$. The half-braiding
$\sigma_M$ is \emph{additional data}, not given by $A$:
it is the datum that promotes $M$ from a module to a
\emph{central} module (an object of the center).

\medskip
\noindent\textbf{Step~4.}
\textit{The braiding on $\cZ$ is the half-braiding.}
The Drinfeld center $\cZ(\Rep^{\Eone}(A))$ is automatically
a braided monoidal category. The braiding is:
\begin{equation}
\label{eq:braiding-is-half-braiding}
 \beta_{(M,\sigma_M),\,(N,\sigma_N)}
 \;:=\; \sigma_M(N)
 \;\colon\; M \otimes N \xrightarrow{\;\sim\;} N \otimes M.
\end{equation}
This is \emph{the} $R$-matrix: not a consequence of the
braiding, but the braiding itself. The braiding
$\beta$ satisfies the Yang--Baxter equation because
the hexagon axiom~\eqref{eq:hexagon-half-braiding} for
$\sigma_M$ is equivalent to the YBE (the hexagon is the
categorical form of the YBE).

\medskip
\noindent\textbf{Step~5.}
\textit{The spectral $R$-matrix on evaluation modules.}
For a Yangian or affine algebra $A$, the representation
category contains \emph{evaluation modules} $V_u$
parametrized by a spectral parameter $u \in \C$
(Remark~\ref{rem:z-spectral-vs-worldsheet}: $u$ is a
Yangian spectral parameter, not a worldsheet coordinate).
On evaluation modules, the half-braiding specializes to:
\begin{equation}
\label{eq:rmatrix-spectral}
 R(z) \;:=\; \sigma_{V_u}(V_v)
 \;\colon\; V_u \otimes V_v \xrightarrow{\;\sim\;} V_v \otimes V_u,
 \qquad z = u - v.
\end{equation}
The spectral parameter $z$ arises from the evaluation
module parametrization, not from an OPE or worldsheet
insertion point. This is the $R$-matrix of integrable
systems.

\medskip
\noindent\textbf{Step~6.}
\textit{BZFN identifies the categorical and algebraic centers.}
The Ben-Zvi--Francis--Nadler theorem
(Theorem~\ref{thm:bzfn}) gives an equivalence of braided
monoidal $\infty$-categories:
\[
 \cZ(\Rep^{\Eone}(A))
 \;\simeq\;
 \Rep^{\Etwo}(\HH^\bullet(A, A)).
\]
The derived center $\HH^\bullet(A, A)$ (Hochschild cochains)
carries its $\Etwo$-structure by the higher Deligne theorem. Under
the BZFN Morita comparison this $\Etwo$-structure is identified with
the Drinfeld-center half-braiding. The Gerstenhaber bracket
$\{-,-\}$ on $\HH^\bullet$ is the algebraic shadow of the categorical
half-braiding $\sigma$; the $R$-matrix datum~\eqref{eq:rmatrix-spectral}
on the left-hand side maps, under BZFN, to the $\Etwo$-braiding datum
on the right-hand side. Without the Morita/Hochschild comparison
hypotheses, no such identification is asserted.

\medskip
\noindent\textbf{Step~7.}
\textit{The $R$-matrix lives one categorical level above $A$.}
If an $\Etwo$-structure on $A$ were separately specified, its
$\Etwo$-module category would map to
$\cZ(\Rep^{\Eone}(A))$ by the usual central functor. That would be
extra input, not part of the $d = 3$ output of $\Phi_3$. The
nontrivial $R$-matrix used here is the spectral half-braiding in the
centre; it measures the centre datum added to the $\Eone$ boundary
category, not a hidden operation on $A$.
\end{construction}

\begin{remark}[The arrow of explanation]
\label{rem:arrow-of-explanation}
The correct logical flow is:
\[
 A \;\text{($\Eone$)}
 \;\xrightarrow{\;\Rep^{\Eone}\;}
 \cC \;\text{(monoidal)}
 \;\xrightarrow{\;\cZ\;}
 \cZ(\cC) \;\text{(braided)}
 \;\xrightarrow{\;\sigma_{V_u}(V_v)\;}
 R(z) \;\text{(the $R$-matrix)}.
\]
The Drinfeld center does not ``give'' braiding to the algebra
$A$; the algebra $A$ remains $\Eone$. The braiding lives on
the \emph{representation category} $\cZ(\Rep^{\Eone}(A))$,
which is a distinct mathematical object from $A$ itself.
The $\Etwo$-structure on the derived center $\HH^\bullet(A,A)$,
identified by BZFN (Theorem~\ref{thm:bzfn}), is the algebraic
encoding of this categorical braiding, not its source.
\end{remark}

\begin{example}[The Heisenberg: Construction~\ref{constr:rmatrix-from-center} in action]
\label{ex:rmatrix-heisenberg-explicit}
For $A = H_k$ (Heisenberg of level $k$, class~$G$),
the construction instantiates as follows.
\begin{enumerate}[label=\textup{Step~\arabic*.},leftmargin=3.5em]
 \item $H_k$ is $\Eone$-chiral: $[J(z), J(w)] = k \cdot \delta'(z-w)$.
 \item $\Rep^{\Eone}(H_k)$: Fock modules $\cF_\lambda$
 labelled by highest weight $\lambda \in \C$; the tensor
 product $\cF_\lambda \otimes \cF_\mu$ is ordered.
 \item Half-braiding: $\sigma_{\cF_\lambda}(\cF_\mu)$ acts
 as the scalar $e^{k\lambda\mu / z}$ (the normal-ordered
 exponential of the Heisenberg OPE).
 \item $R$-matrix: $R(z) = \sigma_{\cF_\lambda}(\cF_\mu)
 = e^{k\lambda\mu / z}$, which is the braiding on
 $\cZ(\Rep^{\Eone}(H_k))$. The classical
 $r$-matrix is $r(z) = k\Omega/z$ (the first-order
 term in the expansion of $\log R$).
 \item Spectral parameter: $z = u - v$ from the evaluation
 module labels $u, v$ on Fock space.
 \item BZFN: $\Zder(H_k) = \C \oplus \C[-1] \oplus \C[-2]$
 (dimension~$3$), with Gerstenhaber bracket
 $\{J, J\} = k$. The $\Etwo$-braiding on
 $\Rep^{\Etwo}(\Zder(H_k))$ is governed by $k$:
 the same parameter that appears in $R(z)$.
 \item Failure of $\Etwo$: $R(z) = e^{k\lambda\mu/z} \neq 1$
 (nontrivial) because $H_k$ is $\Eone$, not $\Etwo$.
 At $k = 0$: $R = 1$ (trivial braiding); $H_0$ is
 the abelian Lie algebra, which is $E_\infty$.
\end{enumerate}
The averaging map kills the $R$-matrix:
$\mathrm{av}(r(z)) = \kappa_{\mathrm{ch}} = k$
(Volume~I). The scalar $\kappa_{\mathrm{ch}}$ is the shadow
of the full $R$-matrix data; the Drinfeld center retains
the complete $R$-matrix.
\end{example}


\subsection{The \texorpdfstring{$\C^3$}{C-3} theorem: Yangian and \texorpdfstring{$\cW_{1+\infty}$}{W-1+inf}}
\label{subsec:c3-yangian}

A concrete instantiation of Corollary~\ref{cor:zder-drinfeld} is for the
CY$_3$ chiral algebra of $\C^3$.

\begin{theorem}[Drinfeld center identification for \texorpdfstring{$\C^3$}{C-3};
Schiffmann--Vasserot, Maulik--Okounkov]
\label{thm:c3-drinfeld-center-restate}
\[
 \cZ\bigl(\Rep^{\Eone}(A^+_{\C^3})\bigr)
 \;\simeq\;
 \Rep^{\Etwo}(D(Y^+(\widehat{\fgl}_1)))
 \;\simeq\;
 \Rep^{\Etwo}(Y(\widehat{\fgl}_1)).
\]
Here $A^+_{\C^3}$ is the framed $\Eone$ boundary chiral algebra whose Hall
positive-half comparison is $\CoHA(\C^3)\cong Y^+(\widehat{\fgl}_1)$ after
Schiffmann--Vasserot; the Hall algebra itself is not the bare chiral object.
Its representation category is monoidal but not braided. The full Yangian
$Y(\widehat{\fgl}_1)$ is the Drinfeld-double/centre target of the positive
half, and the $\Etwo$-braided structure on its representation category arises
from the Drinfeld center construction. The $\cW_{1+\infty}$ vertex algebra
appears through the standard Fock/vacuum-module evaluation of
$Y(\widehat{\fgl}_1)$ at $c = 1$, not as a literal identification of the
Hopf algebra with the VOA.
\end{theorem}

\begin{remark}[Attribution]
\label{rem:c3-drinfeld-attribution}
The associative $\Eone$ Hall nature of $Y^+(\widehat{\fgl}_1)$ is due to
Schiffmann--Vasserot (2013) and the identification of the full
Yangian as the Drinfeld double is due to Maulik--Okounkov (2019).
The vacuum-module action of $Y(\widehat{\fgl}_1)$ on
$\cW_{1+\infty}$ at $c = 1$ follows from the work of
Proch\'azka--Rap\v{c}\'ak (2019) and Litvinov (2020). The formulation as a Drinfeld center
identification, connecting the categorical and algebraic perspectives,
is the content of Theorem~\ref{thm:c3-drinfeld-center} in
Chapter~\ref{ch:cy-to-chiral}.
\end{remark}

\begin{theorem}[Finite Rees Hall, realized CoHA, and Drinfeld double]
\label{thm:finite-rees-realized-coha-drinfeld-double}
Let $X$ be a smooth CY$_3$ equipped with a finite DWR/Ran cyclic
critical atlas and finite bounds $(N,r,L,m)$. Let
\[
 \Theta_{\hCS\to\Hall}^{\mathrm{Rees},\mathrm{or};N,r,L,m}
 \in
 \MC(\mathfrak M_{\hCS,\Hall}^{N,r,L,m})
\]
be the finite Rees Hall natural transformation of
Theorem~\ref{thm:finite-dwr-ran-hcs-rees-hall-construction}. Suppose
the finite systems are pro-compatible and satisfy the
Mittag--Leffler condition, and put
\[
 \widehat{\Hall}^{\mathrm{Rees},\mathrm{or}}_X
 :=
 \varprojlim_{N,r,L,m}
 \Hall_{\mathrm{crit}}^{\mathrm{Rees},\mathrm{or},\leq N,\leq L}
 \bigl(X\bigr)_{\leq m}.
\]
Suppose further that a monoidal vanishing-cycle realization functor
$\mathfrak R_{\mathrm{van}}$ exists on this pro-object and is
compatible with the Hall correspondences. Define the realized positive
Hall algebra
\[
 Y^+_{\mathrm{crit}}(X)
 :=
 H^\bullet\!\left(
 \mathfrak R_{\mathrm{van}}
 (\widehat{\Hall}^{\mathrm{Rees},\mathrm{or}}_X)
 \right)\big|_{\hbar=1}.
\]
If $Y^+_{\mathrm{crit}}(X)$ carries a completed bialgebra structure,
a Cartan factor $Y^0_X$, a completed opposite half $Y^-_{\mathrm{crit}}(X)$,
and a non-degenerate Hopf pairing
$\langle-,-\rangle\colon
Y^+_{\mathrm{crit}}(X)\widehat\otimes
Y^-_{\mathrm{crit}}(X)\to\C((\hbar))$, then the Hall--Drinfeld double is
\[
 D_\hbar(Y^+_{\mathrm{crit}}(X))
 =
 Y^-_{\mathrm{crit}}(X)
 \widehat\otimes
 Y^0_X
 \widehat\otimes
 Y^+_{\mathrm{crit}}(X)
\]
with cross-relations determined by $\langle-,-\rangle$. These three
objects are distinct:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item $\Theta_{\hCS\to\Hall}^{\mathrm{Rees},\mathrm{or};N,r,L,m}$
 is a finite Maurer--Cartan natural transformation into a Rees Hall
 complex; it is not a realized critical CoHA and has no negative half,
 Cartan factor, antipode, or half-braiding.
 \item $Y^+_{\mathrm{crit}}(X)$ is the realized positive Hall algebra,
 with product induced by extension correspondences in the effective
 cone; it is an $\Eone$ positive half, not a Drinfeld double.
 \item $D_\hbar(Y^+_{\mathrm{crit}}(X))$ is formed only after adjoining
 the completed negative half, Cartan factor, Hopf pairing, and
 double cross-relations. Under the finite-dimensional Hopf or completed
 topological-Hopf hypotheses for the Drinfeld double,
 \[
  \cZ\!\left(\Rep^{\Eone}(Y^+_{\mathrm{crit}}(X))\right)
  \simeq
  \Rep^{\Etwo}\!\left(D_\hbar(Y^+_{\mathrm{crit}}(X))\right).
 \]
\end{enumerate}
For $X=\C^3$, Schiffmann--Vasserot identify
$Y^+_{\mathrm{crit}}(\C^3)=\CoHA(\C^3)\cong
Y^+(\widehat{\fgl}_1)$, and the Drinfeld double is
$D(Y^+(\widehat{\fgl}_1))\cong Y(\widehat{\fgl}_1)$; the
$\cW_{1+\infty}$ vertex algebra is obtained from the doubled algebra
by Fock/vacuum-module evaluation, not at the CoHA layer.
\end{theorem}

\begin{proof}
The finite construction of
Theorem~\ref{thm:finite-dwr-ran-hcs-rees-hall-construction} places
$\Theta_{\hCS\to\Hall}^{\mathrm{Rees},\mathrm{or};N,r,L,m}$ in the
Maurer--Cartan set of the convolution dg Lie algebra
$\mathfrak M_{\hCS,\Hall}^{N,r,L,m}$. Thus its defining equation is
\[
 D_{\mathrm{tot}}\Theta+\frac{1}{2}[\Theta,\Theta]=0,
\]
where $D_{\mathrm{tot}}$ combines the Hall differential, the bar
differential on hCS observables, and the finite Cech/Ran differential.
This datum is finite in the four bounds $(N,r,L,m)$ and is valued in a
Rees Hall complex. Passing from it to
$\widehat{\Hall}^{\mathrm{Rees},\mathrm{or}}_X$ is the inverse-limit
operation; exactness of that passage is precisely the
Mittag--Leffler/pro-compatibility hypothesis. Applying
$\mathfrak R_{\mathrm{van}}$ is a second operation: it replaces the
Rees chain model by the vanishing-cycle realization on the critical
moduli stack and transports the Hall product through the monoidal
realization functor. Therefore the finite Rees transformation and the
realized CoHA cannot be identified before those two operations are
performed.

The product of $Y^+_{\mathrm{crit}}(X)$ is the Hall product on effective
classes: extension correspondences add charges in the effective cone.
No such correspondence produces a negative class, a Cartan current, an
antipode, or the Hopf pairing against an opposite half. Hence
$Y^+_{\mathrm{crit}}(X)$ is a positive $\Eone$ Hall algebra. The
Drinfeld double adds exactly the missing data:
$Y^-_{\mathrm{crit}}(X)$, $Y^0_X$, the non-degenerate pairing, and the
cross-relations. These additions change the algebra already at the
level of generators; the double is not a relabelling of the positive
Hall product.

For a finite-dimensional Hopf algebra, and for the completed
topological Hopf algebras satisfying the same rigidity and
non-degeneracy hypotheses, the standard Drinfeld--Majid--Schauenburg
theorem identifies the Drinfeld center of the module category with the
category of Yetter--Drinfeld modules, equivalently with modules over
the Drinfeld double. This gives
$\cZ(\Rep^{\Eone}(Y^+_{\mathrm{crit}}(X)))\simeq
\Rep^{\Etwo}(D_\hbar(Y^+_{\mathrm{crit}}(X)))$ in the stated domain.
The $\C^3$ specialization is the Schiffmann--Vasserot positive-half
theorem followed by Maulik--Okounkov/Tsymbaliuk doubling; the
Fock/vacuum-module construction of $\cW_{1+\infty}$ is a representation
of the doubled algebra, so it does not collapse the positive CoHA, the
double, and the vertex algebra into one object.
\end{proof}

\subsection{The Ben-Zvi--Francis--Nadler loop space identification}
\label{subsec:bzfn-loop-space}

The algebraic identification $\cZ(\Rep^{\Eone}(A)) \simeq \Rep^{\Etwo}(\HH^\bullet(A,A))$ of Theorem~\ref{thm:bzfn} has a geometric counterpart: the Drinfeld center of quasi-coherent sheaves on a stack $X$ is equivalent to quasi-coherent sheaves on the free loop space $\cL X$. This reformulation makes the passage from $\Eone$ to $\Etwo$ explicit (the braided structure arises from the $S^1$-rotation action on loops) and provides the geometric foundation for the CY holographic principle: the boundary algebra $A$ lives on $X$, the bulk algebra lives on $\cL X$, and the Drinfeld center is the geometric bridge between them.

\begin{theorem}[BZFN loop space identification; Ben-Zvi--Francis--Nadler]
\label{thm:bzfn-loop-space}
For a smooth algebraic stack $X$, there is an equivalence
of braided monoidal $\infty$-categories
\[
 \cZ\bigl(\mathrm{QCoh}(X)\bigr) \;\simeq\; \mathrm{QCoh}(\cL X)
\]
where $\cL X = \mathrm{Map}(S^1, X)$ is the (derived) free loop
space of $X$. The braided monoidal structure on
$\mathrm{QCoh}(\cL X)$ arises from the $S^1$-rotation action on the
loop space.
\end{theorem}

\begin{remark}[Connection to the derived center]
\label{rem:bzfn-loop-center}
When $X = BG$ for a reductive group $G$, the loop space
$\cL(BG) = G/G$ (the adjoint quotient), and
$\mathrm{QCoh}(\cL(BG)) \simeq \mathrm{QCoh}(G/G)$ recovers
the character sheaves of Lusztig. For $X = \mathrm{Spec}(A)$,
the loop space $\cL X = \mathrm{Spec}(\HH_\bullet(A))$
(derived Hochschild homology), and the BZFN equivalence reduces
to Theorem~\ref{thm:bzfn}: the Drinfeld center of $A$-modules
is computed by the Hochschild cochains of $A$ (which are
Koszul dual to the Hochschild chains defining $\cL X$).
\end{remark}


\section{Bulk-boundary correspondence}
\label{sec:bulk-boundary}

\begin{proposition}[Information flow by genus]
\label{prop:bulk-boundary-flow}
The bulk-boundary correspondence
$\cZ(\Rep^{\Eone}(A)) \simeq \Rep^{\Etwo}(\Zder(A))$ exhibits
genus-dependent information flow:
\begin{enumerate}[label=(\roman*)]
 \item \emph{Genus~$0$}: Information flows one way, from bulk to
 boundary. The forgetful functor
 $\cZ(\cC) \to \cC$ has no natural section: the Swiss-cheese
 operad forbids open-to-closed maps at genus $0$
 (Volume~II, thm:thqg-swiss-cheese);
 \item \emph{Genus~$1$}: The annulus trace
 $\Delta_{\mathrm{ns}}(\Tr_A) = \kappa_{\mathrm{cat}} \cdot \lambda_1$
 (Volume~I, thm:annulus-trace) provides the first open-to-closed
 map. The Dennis trace
 $K(\cC) \to \HH_\bullet(\cC)$ maps K-theory of the open sector
 into the bulk via the SBI sequence;
 \item \emph{Higher genus}: The full genus tower progressively
 restores bidirectionality. The clutching maps
 $\mathrm{cl}_{g,n} \colon \HH_\bullet(A)^{\otimes n} \to \Zder(A)$
 on the modular operad provide genus-$g$ open-to-closed data.
\end{enumerate}
\end{proposition}

\begin{proof}
Item (i): at the operad level, $\SC^{\mathrm{ch,top}}$ has no
composition $\mathrm{open} \to \mathrm{closed}$ at genus $0$
(the Kontsevich no-open-to-closed axiom).

Item (ii): the Dennis trace $K(\cC) \to \HH_\bullet(\cC)$ factors
through negative cyclic homology $\HC^-_\bullet(\cC)$. The CY
duality $\HH^\bullet(\cC) \simeq \HH_{\bullet+d}(\cC)$
(Theorem~\ref{thm:cy-hh-duality}) and the identification
$\HH^\bullet(\cC) \simeq \Zder(A)$ (Corollary~\ref{cor:zder-drinfeld})
give the map into the bulk. The annulus trace is the genus-$1$
specialization: the $S^1$-trace on $\HH_0(A)$ gives
$\kappa_{\mathrm{cat}} \cdot \lambda_1 \in H^2(\overline{\cM}_{1,1})$.

Item (iii): the modular operad structure on
$\{H^*(\overline{\cM}_{g,n})\}$ provides clutching maps
$\gamma_{g_1,n_1} \circ_{g_2,n_2}$ that compose open-sector data
(living on boundaries of $\overline{\cM}_{g,n}$) into closed-sector
invariants. Each genus $g$ introduces new clutching compositions
(from codimension-$1$ boundary strata of $\overline{\cM}_{g,n}$),
progressively enlarging the image of the open-to-closed map.
\end{proof}

\begin{theorem}[CY Drinfeld center]

Let $\cC$ be a smooth proper CY$_d$ category with quantum chiral

\section{Categorical Theorem H}
\label{sec:categorical-thm-h}

Volume~I Theorem~H is a chiral Hochschild statement: on the
ordinary curve-ambient Koszul locus, $\ChirHoch^\bullet(A)$ is
concentrated in degrees $\{0,1,2\}$, while the Vol.~III
$\Phi_d$-image may carry an additional CY-degree contribution
recorded in Chapter~\ref{ch:hochschild-calculus}. It is not a
categorical Hochschild theorem for an arbitrary CY category.
The categorical comparison below records the standard finite
Hochschild amplitude of $\cC$ and the precise extra hypothesis
needed before applying $\Phi$ to compare brackets.

\begin{theorem}[Categorical Hochschild comparison with Theorem H]
\label{thm:categorical-thm-h}
Let $\cC$ be a smooth proper CY$_d$ category, and suppose its
object-level chiral image $A_\cC=\Phi(\cC)$ has been constructed and
lies on the modular Koszul locus on which Corollary~\ref{cor:zder-drinfeld}
applies. Then:
\begin{enumerate}[label=(\roman*)]
 \item $\HH^\bullet(\cC)$ is a perfect finite-dimensional Hochschild
 complex. For $\cC=\Perf(X)$ with $X$ a smooth projective CY$_d$, HKR
 identifies
 \[
 \HH^n(\cC)\cong \bigoplus_{p+q=n} H^q(X,\wedge^pT_X),
 \]
 so the amplitude is the HKR amplitude of \(X\), not the Vol.~I chiral
 Hochschild concentration statement;
 \item $\HH^0(\cC) \simeq k$ (the identity natural transformation);
 \item $\HH^1(\cC)$ classifies first-order deformations of $\cC$
 as a dg category;
 \item $\HH^2(\cC)$ is the obstruction space for deformations.
\end{enumerate}
Where the chain-level functorial comparison for $\Phi$ is constructed
\textup{(}CY-A$_2$, and the formal/framed-object lane at CY-A$_3$\textup{)},
the Gerstenhaber bracket $[-,-]\colon \HH^p\otimes\HH^q\to
\HH^{p+q-1}$ maps to the convolution bracket on
$\fg^{\mathrm{mod}}_{A_\cC}$ of Volume~I. For a non-formal CY$_3$
input this bracket comparison is a proof obligation, not a consequence
of finite Hochschild amplitude.
\end{theorem}

\begin{remark}[Proof scope]
\label{rem:categorical-thm-h-scope}
Items (i)--(iv) are standard Hochschild/deformation-theoretic facts
(Keller, Kontsevich) with HKR in the geometric case. They are not a
categorical proof of Vol.~I Theorem~H, which is a chiral Hochschild
concentration theorem for \(A_\cC\). The identification of the
Gerstenhaber bracket with the Volume~I convolution bracket under
$\Phi$ is established for $d = 2$ (CY-A$_2$) and for $d = 3$ at the
infinity-categorical framed-object level; the chain-level
identification at $d = 3$ for non-formal algebras remains open.
A polynomial Hilbert series for a comparison complex does not identify
the underlying algebra with a polynomial ring.
\end{remark}


\section{Quantum group reconstruction from the center}
\label{sec:qg-reconstruction}

\begin{proposition}[Tannakian reconstruction]
\label{prop:tannakian-reconstruction}
Let $\cB$ be a rigid braided monoidal category over $\C$ equipped
with a fiber functor $\omega \colon \cB \to \Vect_\C$. The
Tannakian reconstruction gives a quasi-triangular Hopf algebra
$H = \End(\omega)$ with $\cB \simeq \Rep(H)$ as braided monoidal
categories. When $\cB = \cZ(\Rep^{\Eone}(A))$ for an $\Eone$-chiral
algebra $A$ with $\frakg$-symmetry, $H \simeq \Uq(\frakg)$ with

\end{proposition}

\begin{remark}[Non-semisimple reconstruction]
\label{rem:non-semisimple-reconstruction}
At roots of unity ($q^N = 1$), the category $\Rep_q(\frakg)$ is
non-semisimple: indecomposable but non-simple modules appear. The
standard fiber functor is not exact on the full category. Two
remedies:
\begin{enumerate}[label=(\alph*)]
 \item Restrict to the semisimplified quotient
 $\overline{\Rep}_q(\frakg)$ (modular tensor category of
 Reshetikhin--Turaev);
 \item Use the modified trace of Geer--Patureau-Mirand--Turaev,
 which replaces the categorical trace for non-semisimple ribbon
 categories.
\end{enumerate}
For the CY programme, approach (b) is preferred: the full
non-semisimple structure encodes the complete BPS spectrum
(Chapter~\ref{ch:toric-coha}), and the modified trace is the
categorical analogue of the regularized partition function.
\end{remark}


\section{The Drinfeld double and the slab}
\label{sec:drinfeld-double-slab}

\begin{definition}[Drinfeld double]
\label{def:drinfeld-double}
For a finite-dimensional Hopf algebra $H$, the \emph{Drinfeld double}
$D(H) = H \bowtie H^{*\,\mathrm{cop}}$ is the quasi-triangular Hopf
algebra built from $H$ and the co-opposite of its dual. Its
representation category is $\cZ(\Rep(H))$, the Drinfeld center of
$\Rep(H)$.
\end{definition}

\begin{remark}[The slab is a bimodule]
\label{rem:slab-bimodule}
In the 3d holomorphic-topological setting (Volume~II), the
Drinfeld double arises from the \emph{slab}: a strip
$X \times [0,1]$ with two boundary components $X \times \{0\}$
and $X \times \{1\}$. The algebra of operators on the slab
is a bimodule for the two boundary algebras $A_0$ and $A_1$.
The Drinfeld double $D(A) = A \bowtie A^{!,\mathrm{cop}}$
appears when $A_0 = A$ and $A_1 = A^!$ (the Koszul dual).
The slab is not a Swiss-cheese disk (which has one closed
and one open boundary component); it is a cylinder with two
open boundary components.
\end{remark}

\begin{conjecture}[Slab = Drinfeld double]
\label{conj:slab-double}
For an $\Eone$-chiral algebra $A$ on $X \times \R$, the slab
algebra $\cA_{\mathrm{slab}} = \End_{A\text{-}A^!}(A \otimes_{A^!} A)$
is isomorphic to the Drinfeld double $D(A)$. The slab $R$-matrix
$\cR_{\mathrm{slab}} = \cR_A \otimes \cR_{A^!}^{-1}$ is the
universal $R$-matrix of $D(A)$, and its representation category
is the Drinfeld center $\cZ(\Rep^{\Eone}(A))$.
\end{conjecture}

\begin{conjecture}[Quantum vertex chiral group]
\label{conj:qvcg}
For a CY$_2$ category $\cC$ with $\frakg$-symmetry, let $A_\cC = \Phi_2(\cC) = \SpCh_{\Sigma_1,C}(\PhiFA_2(\cC))$ be the stage-$2$ specialisation of the canonical $E_2$-holomorphic factorisation algebra $\PhiFA_2(\cC)$ (Theorem~\ref{thm:phi-two-stage-factorisation}). The Drinfeld
centre $\cZ(\Rep^{\Eone}(A_\cC))$ determines a quantum vertex
chiral group $G(\cC)$ unifying four algebraic structures:
\begin{enumerate}[label=(\roman*)]
 \item $\Uq(\frakg)$: the quantum group (fiber at a point);
 \item $Y(\frakg)$: the Yangian ($\Eone$-sector, Volume~II, MC3);
 \item $\Uq(\widehat{\frakg})$: the affine quantum group
 (loop algebra sector);
 \item The Etingof--Kazhdan quantum vertex algebra (the genuinely
 nonlocal atom, Volume~II).
\end{enumerate}
The four incarnations arise from different geometric
specialisations of the $\Etwo$-factorization structure on
$\Ran(X) \times \Ran(Y)$; at $d \geq 3$ the $\Etwo$-braided shadow lives on $\cZ(\Rep^{\Eone}(A_\cC))$, not on $A_\cC$ itself (Remark~\ref{rem:e2-on-drinfeld-centre}).
\end{conjecture}

\begin{conjecture}[Bulk comparison for the Drinfeld double]
\label{conj:v3-drinfeld-center-equals-bulk}
Let $A$ be an augmented modular Koszul $\Eone$-chiral algebra and let
$A^i=H^\bullet(B(A))$ and $A^!=D_{\Ran}(A^i)$ be its Koszul
cohomology coalgebra and Verdier--Koszul dual algebra. Let
$U_A=A\bowtie A^!$ denote the completed Drinfeld double of the Koszul
pair. Assume:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item $A^!$ carries a factorization-algebra structure in the same
 Ran completion as $A$, so that $U_A$ factors on
 $\Ran(X)\times\Ran(Y)$;
 \item the pointwise reduction from
 $\cZ(\Rep^{\Eone}(U_A))$ to the completed algebraic center
 $Z_{\mathrm{alg}}(U_A)$ converges in the chosen completion;
 \item the bar-cobar comparison is compatible with Hochschild
 cochains:
 $\RHom_{U_A}(A,A)\simeq\RHom_{A\otimes A^{\op}}(A,A)$.
\end{enumerate}
Then there is an equivalence of $\Etwo$-algebras
\[
 Z_{\mathrm{alg}}(U_A) \;\simeq\; \Zder(A)
\]
identifying the completed algebraic center of the Drinfeld double
with the chiral derived center.
\end{conjecture}

\begin{remark}[Three obstructions]
\label{rem:drinfeld-center-bulk-obstructions}
The assumptions in
Conjecture~\ref{conj:v3-drinfeld-center-equals-bulk} are exactly the
three places where a direct proof can fail:
\begin{enumerate}[label=(\roman*)]
 \item \emph{Pointwise reduction failure for class M.}
 For algebras of class M (Virasoro, $\cW_N$), the shadow tower
 has infinite depth: all operations $m_k$ are nonzero.
 The Drinfeld double $U_A = A \bowtie A^!$ is defined at the
 chain level, but the passage from the Drinfeld center
 $\cZ(\Rep^{\Eone}(U_A))$ to the algebraic center
 $Z_{\mathrm{alg}}(U_A)$
 requires a pointwise reduction (evaluating the half-braiding
 on each object) that may not converge when the shadow tower
 is infinite.
 \item \emph{Factorization property of $A^!$.}
 The Koszul dual $A^!$ is an algebra, not a factorization
 algebra on $\Ran(X)$ in general. The Drinfeld double
 construction requires both $A$ and $A^!$ to be factorization
 algebras (so that $U_A$ factors on $\Ran(X) \times \Ran(Y)$).
 For class G and L algebras, $A^!$ inherits a factorization
 structure from the bar-cobar inversion; for class C and M,
 this is open.
 \item \emph{Bar-cobar/Hochschild compatibility.}
 The chiral derived center $\Zder(A) = C^\bullet_{\mathrm{ch}}(A,A)$
 is defined via chiral Hochschild cochains. The Drinfeld
 double center $Z_{\mathrm{alg}}(U_A)$ is defined via the half-braiding
 construction on $A \bowtie A^!$. The two constructions agree
 when the bar-cobar adjunction of Volume~I (Theorem~A) is
 compatible with the Hochschild complex in the precise sense
 that
 $\RHom_{U_A}(A, A) \simeq \RHom_{A \otimes A^{\op}}(A, A)$.
 This compatibility is proved for class G (where $A^!$ is
 Koszul dual in the classical sense) and conjectural in
 general.
\end{enumerate}
\end{remark}

\begin{remark}[Heisenberg algebra calculation]
\label{rem:drinfeld-center-heisenberg-verification}
For $A = H_k$ (Heisenberg, class~$G$), the conjecture is
matched at the level of numerical invariants. The Drinfeld
double is $\mathrm{Drin}(H_k) = H_k \otimes H_{-k} / (c = c')$,
dimension~$3$, with brackets $[J, J'] = k \cdot c$.
The naive Lie centre $Z(\mathrm{Drin}(H_k))$ has dimension~$1$
at generic~$k$.
The chiral derived centre is
$Z^{\mathrm{der}}_{\mathrm{ch}}(H_k) \cong
\bC \oplus \bC[-1] \oplus \bC[-2]$, dimension~$3$:
the discrepancy ($1$ vs $3$) reflects $\mathrm{Ext}^1$
($J$, the derivation) and $\mathrm{Ext}^2$ ($:JJ:$, the
level deformation) invisible to the commutant.
Five consistency checks match across both sides:
$r$-matrix coefficient $= k$, $\kappa_{\mathrm{ch}} = k$,
$\chi = 1$, shadow class $G$ / depth~$2$,
complementarity $k + (-k) = 0$. At
$k = 0, 1, -1, 1/2, 7, -3/2$ the same five invariants agree. For
class~$G$ the $\Etwo$ structure on both sides is
determined by the single parameter~$k$, so the conjecture
verification reduces to matching this one numerical
invariant.
\end{remark}


\section{The \texorpdfstring{$\En$}{E-n}-circle and the categorified center}
\label{sec:en-circle}

The Drinfeld center is not an isolated construction: it is
the $\Eone \to \Etwo$ arrow in the $\En$-circle, the
systematic passage between $\En$-structures of different degrees.

\begin{definition}[\texorpdfstring{$\En$}{E-n}-center]
\label{def:en-center}
For $0 \leq m < n$, the \emph{$\En$-center} of an
$E_m$-monoidal category $\cC$ is the $\En$-monoidal category
\[
 \cZ_{\En/E_m}(\cC) \;:=\;
 \End_{\cC\text{-}\mathrm{BiMod}_{\En}\text{-}\cC}(\cC)
\]
obtained by taking bimodule endomorphisms in the $\En$-sense.
\begin{itemize}
 \item $m = 0$, $n = 1$: the ordinary center
 $\cZ(\cC) = Z(\cC)$ of a category (monoidal center).
 \item $m = 1$, $n = 2$: the Drinfeld center
 $\cZ(\cC)$, recovering $\Etwo$-braiding from $\Eone$-monoidal
 structure. This is the case relevant to this chapter.
 \item $m = n - 1$, general $n$: Dunn additivity gives
 $\cZ_{\En/E_{n-1}}(\cC) \simeq \cC$ when $\cC$ is already
 $\En$-monoidal and non-degenerate. The center construction
 ``does nothing'' when the target degree is already present.
\end{itemize}
\end{definition}

\begin{proposition}[Drinfeld center as right adjoint to forgetful]
\label{prop:drinfeld-center-adjunction}
The Drinfeld center $\cZ \colon \Eone\text{-}\Cat \to
\Etwo\text{-}\Cat$ is the right adjoint to the forgetful functor
$U \colon \Etwo\text{-}\Cat \to \Eone\text{-}\Cat$: for any
$\Eone$-monoidal category $\cC$ and $\Etwo$-braided monoidal
category $\cD$,
\[
 \Hom_{\Etwo}(\cD, \cZ(\cC))
 \;\simeq\;
 \Hom_{\Eone}(U(\cD), \cC).
\]
This adjunction categorifies
the algebraic center $z(A) = \{x \in A : [x,a] = 0\;\forall a\}$.
The correspondence with Volume~I is:
\begin{center}
\begin{tabular}{lll}
\toprule
Volume~I (algebraic) & This chapter (categorical) & Level \\
\midrule
$\fg^{\Eone}_A$ (convolution Lie) & $\Rep^{\Eone}(A)$ (monoidal)
 & $\Eone$ \\
$z(\fg^{\Eone})$ (center of Lie algebra)
 & $\cZ \colon \Eone\text{-}\Cat \to \Etwo\text{-}\Cat$
 & center \\
$\kappa_{\mathrm{ch}} = \mathrm{av}(r(z))$ (scalar shadow)
 & braiding on $\cZ(\cC)$ (categorical shadow)
 & output \\
\bottomrule
\end{tabular}
\end{center}
The algebraic averaging map $\mathrm{av}$ of Volume~I is a scalar
projection adjacent to the center construction, not its
decategorification. It projects the ordered $R$-matrix to
$\kappa_{\mathrm{ch}}$, discarding the $\Sigma_n$-equivariant
structure. The Drinfeld center itself is \emph{not} averaging; it is
the universal construction
of half-braidings $\sigma_M \colon M \otimes (-) \xrightarrow{\sim}
(-) \otimes M$, which is a categorified commutant, not a
coinvariant projection.
\end{proposition}

\begin{proof}
The Drinfeld center is the right adjoint to the forgetful
functor $U \colon \Etwo\text{-}\Cat \to \Eone\text{-}\Cat$:
for any $\Eone$-monoidal category $\cC$ and
$\Etwo$-braided monoidal category $\cD$,
\[
 \Hom_{\Etwo}(\cD, \cZ(\cC))
 \;\simeq\;
 \Hom_{\Eone}(U(\cD), \cC).
\]
Objects of $\cZ(\cC)$ are pairs $(M, \sigma_M)$ where
$\sigma_M \colon M \otimes (-) \xrightarrow{\sim} (-) \otimes M$
is a half-braiding (natural isomorphism compatible with the
monoidal structure). This is the categorification of the
algebraic center $z(A) = \{x : xa = ax\;\forall a\}$: the
half-braiding $\sigma_M$ is the categorical analogue of the
commutation condition, not of $\Sigma_n$-coinvariants.

The algebraic averaging map
$\mathrm{av}(r(z)) = \kappa_{\mathrm{ch}}$ is not the counit
of the Drinfeld-center adjunction. It is obtained only after passing
from the half-braiding to a scalar shadow; this projection discards
the categorical half-braiding data.
\end{proof}

\begin{remark}[The Drinfeld center is not categorified averaging]
\label{rem:center-not-averaging}
The Volume~I averaging map
$\mathrm{av} \colon \fg^{\Eone}_A \to \fg^{\mathrm{mod}}_A$
and the Drinfeld center
$\cZ \colon \Eone\text{-}\Cat \to \Etwo\text{-}\Cat$
are related by a factorization, but they are not
categorifications of each other. The factorization:
\[
 \Eone\text{-}\Cat
 \;\xrightarrow{\;\cZ\;}\;
 \Etwo\text{-}\Cat
 \;\xrightarrow{\;\mathrm{Sym}\;}\;
 \Einf\text{-}\Cat
\]
where $\mathrm{Sym}$ is the further symmetrization
(forcing $\sigma_{M,N} \circ \sigma_{N,M} = \mathrm{id}$).
After passage to scalar shadows one sees the sequence
\[
 r(z) \;\xmapsto{\;\text{construct}\;}\; R(z)
 \;\xmapsto{\;\mathrm{av}\;}\;
 \kappa_{\mathrm{ch}}.
\]
The Drinfeld center is the \emph{first} step:
it constructs the $R$-matrix (non-symmetric braiding) from
$\Eone$-monoidal data. The averaging map is the
\emph{second} step: it projects the braiding to its
symmetric shadow, destroying the quantum group structure.
The categorical analogue of the averaging map itself is
$\mathrm{Sym}$, the symmetrization functor, which kills
the $R$-matrix by forcing
$R(z) R(-z) = \mathrm{id} \Rightarrow R = \mathrm{id}$.
This is precisely why averaging destroys quantum group data:
$\mathrm{av}(r(z)) = \kappa_{\mathrm{ch}}$ is a
scalar, not a matrix.

The Drinfeld center categorifies the algebraic
\emph{center} $z(A) = \{x : [x,a] = 0\;\forall a\}$, which
\emph{constructs} commuting elements: the categorical
analogue of the half-braiding condition. The averaging map
categorifies to the \emph{abelianization}
$A \mapsto A/[A,A]$, which \emph{quotients} by
non-commutativity. Construction versus quotient: the center
adds data, averaging removes it.
\end{remark}

\section{The \texorpdfstring{$E_3 \to \Etwo$}{E-3 -> E2} restriction and symmetric braiding}
\label{sec:e3-symmetric}

The Drinfeld center produces non-symmetric braiding from
$\Eone$-monoidal input. This is a fundamentally different
mechanism from the $E_3 \to \Etwo$ restriction, which
produces only symmetric braiding. The distinction is the
mechanism behind the $\Eone$ stabilization theorem
(Chapter~\ref{ch:cy-to-chiral},
Theorem~\ref{thm:e1-stabilization-cy}).

\begin{proposition}[\texorpdfstring{$E_3$}{E-3} restriction gives symmetric braiding]
\label{prop:e3-symmetric-braiding}
Let $\cC$ be an $E_3$-monoidal category. The restriction of the
$E_3$-structure to $\Etwo$ produces a braided monoidal category
whose braiding is \emph{symmetric}:
$\sigma_{X,Y} \circ \sigma_{Y,X} = \id_{X \otimes Y}$
for all objects $X, Y$.
\end{proposition}

\begin{proof}
The braiding on an $\Etwo$-monoidal category is classified by
$\pi_1(\mathrm{Conf}_2(\R^2)) \cong \Z$: the generator is
the counterclockwise winding of one point around another
in the plane. When the $\Etwo$-structure is the restriction
of an $E_3$-structure, the braiding is classified by
$\pi_1(\mathrm{Conf}_2(\R^3))$, which is trivial (two points
in $\R^3$ can be unlinked by moving in the third direction).
The double braiding
$\sigma_{X,Y} \circ \sigma_{Y,X}$ corresponds to the
generator of $\pi_1(\mathrm{Conf}_2(\R^2))$ traversed twice;
in $\R^3$ this loop is contractible, so the double braiding
is the identity.
\end{proof}

\begin{remark}[Two sources of \texorpdfstring{$\Etwo$}{E2}-braiding for CY\texorpdfstring{$_3$}{-3}]
\label{rem:two-sources-e2}
For a CY$_3$ category $\cC$, two $\Etwo$-braided structures
are available, and they are fundamentally different
(Remark~\ref{rem:e2-on-drinfeld-centre}):
\begin{enumerate}[label=(\alph*)]
 \item \emph{$E_3$-restriction.}
 The $\bS^3$-framing on $\HH_\bullet(\cC)$ gives an
 $E_3$-structure under the hypotheses of Theorem~\ref{thm:derived-framing-obstruction}. Restricting $E_3$ to $\Etwo$
 via Dunn additivity gives a braided monoidal category
 with \emph{symmetric} braiding
 ($\pi_1(\mathrm{Conf}_2(\R^3)) = 0$). This symmetric
 braiding is the ``wrong'' answer for quantum groups: it sees
 only the classical limit $q = 1$.
 \item \emph{Drinfeld center.}
 The $\Eone$-chiral algebra $A = \Phi(\cC)$ has a monoidal
 representation category $\Rep^{\Eone}(A)$, and the Drinfeld
 center $\cZ(\Rep^{\Eone}(A))$ produces an $\Etwo$-braided
 category with \emph{non-symmetric} braiding
 ($\pi_1(\mathrm{Conf}_2(\R^2)) \cong \Z$). This
 non-symmetric braiding is the quantum group $R$-matrix at
 $q \neq 1$.
\end{enumerate}
The passage (a) $\to$ (b) is the $\Eone$ stabilization: the
$E_3$-structure on the CY$_3$ Hochschild complex is ``too
symmetric'' to see quantum groups, and one must descend to
$\Eone$ (by choosing a preferred direction, i.e., the
$\Omega$-background) before applying the Drinfeld center to
recover the non-trivial braiding. This is the categorical
content of the $\Eone$ stabilization theorem: the genuine
quantum group braiding arises not from restricting a
higher $\En$-structure, but from the Drinfeld center of a
lower one.
\end{remark}

\begin{remark}[The \texorpdfstring{$\En$}{E-n}-circle in the CY programme]
\label{rem:en-circle-cy}
The pattern generalizes across CY dimensions as follows:
\begin{center}
\begin{tabular}{llll}
\toprule
CY dim $d$ & Native $\En$ on $\HH_\bullet$ & Braiding type & Quantum group \\
\midrule
$d = 1$ & $\Eone$ (trivially) & none & Drinfeld center gives $\Etwo$ \\
$d = 2$ & $\Etwo$ & non-symmetric & direct (the ``sweet spot'') \\
$d = 3$ & $E_3$ & symmetric & must descend to $\Eone$, then center \\
$d = 4$ & $E_4$ & symmetric & further descent needed \\
\bottomrule
\end{tabular}
\end{center}
The dimension $d = 2$ is distinguished: the native $\Etwo$-structure
already carries non-symmetric braiding, so no descent or center
construction is needed. For $d \geq 3$, the $\En$-structure
with $n \geq 3$ gives only symmetric braiding, and the
non-symmetric quantum group braiding requires the $\Eone$
stabilization followed by the Drinfeld center. For $d = 1$,
there is no native braiding at all, and the Drinfeld center is
the only source.

This explains why quantum groups are ubiquitous in
the $d = 2$ CY landscape (K3 surfaces, quiver varieties) and
require additional structure ($\Omega$-deformation, stability
conditions) in the $d = 3$ CY landscape (Calabi--Yau threefolds).
\end{remark}


\section{The \texorpdfstring{$\Etwo$}{E2} promotion obstruction for \texorpdfstring{$K3 \times E$}{K3 x E}}
\label{sec:k3e-e2-promotion}

The threefold $X = K3 \times E$ is the simplest compact CY$_3$
admitting a rich chiral algebra structure. It exemplifies the
full $\Eone \to \Etwo$ promotion mechanism and its obstructions.
The chiral algebra $A_{K3 \times E}$
(constructed by Theorem~CY-A$_3$, Theorem~\ref{thm:cy-to-chiral-d3})
is natively $\Eone$, since the $\mathbb{S}^3$-framing
produces an $E_3$-structure on $\HH_\bullet(\cC_{K3 \times E})$
whose restriction to $\Etwo$ is \emph{symmetric}
($\pi_1(\mathrm{Conf}_2(\R^3)) = 0$). The non-symmetric quantum
group braiding requires the Drinfeld center passage. This section
quantifies three aspects of the $\Eone \to \Etwo$ passage
for $K3 \times E$.

\begin{remark}[Hypotheses for \texorpdfstring{$K3 \times E$}{K3 x E}]
\label{rem:k3e-conditional}
The chiral algebra $A_{K3 \times E}$ is constructed in the
$\infty$-categorical framework by Theorem~CY-A$_3$
(Theorem~\ref{thm:cy-to-chiral-d3},
Theorem~\ref{thm:derived-framing-obstruction}). Results involving
the quantum vertex chiral group $G(K3 \times E)$ assume
Conjecture~CY-C. The BKM superalgebra
$\frakg_{\Delta_5}$ and the Maulik--Okounkov $R$-matrix are
unconditional: they are constructed from the Mukai lattice and
the Hilbert scheme geometry respectively. Any identification of these
objects with the Drinfeld center of
$\Rep^{\Eone}(A_{K3 \times E})$ uses the CY-A$_3$ and CY-C hypotheses
just stated. A compact non-formal chain-level model still requires
either the corrected TCFT comparison
$B_{\mathrm{term}}^{(2)}\to B_{\mathrm{TCFT}}^{(2)}$ or the complete
$\HH^{-2}_{E_1}$ filtration theorem for the chosen strictification.
\end{remark}


\subsection{The non-locality obstruction: quantifying the center-hocolim gap}
\label{subsec:k3e-nonlocality}

The Drinfeld center does not commute with homotopy colimits
(Proposition~\ref{prop:center-hocolim}). For $K3 \times E$,
the stability chamber decomposition is infinite (the Mukai
lattice $\Lambda = U^3 \oplus E_8(-1)^2$ has rank~$24$), and
the gap between $\cZ(\operatorname{hocolim})$ and
$\operatorname{hocolim}(\cZ)$ is massive.

\begin{proposition}[Non-locality quantification for \texorpdfstring{$K3 \times E$}{K3 x E}]
\label{prop:k3e-nonlocality-quantification}
\textup{(}The chiral algebra interpretation uses CY-A$_3$
\textup{(}proved\textup{)}; the BKM character
computation is independent.\textup{)}
At each graded level~$n$, the center-hocolim obstruction
ratio for $K3 \times E$ is:
\[
 \rho_n := \frac{\dim \cZ(\operatorname{hocolim})_n - \dim \operatorname{hocolim}(\cZ)_n}{\dim \cZ(\operatorname{hocolim})_n}.
\]
Computed values (global center $= \frakg_{\Delta_5}$, local
center $= \bigoplus_{i=1}^{24} Y(\widehat{\fgl}_1)$ with
gluing corrections):
\begin{center}
\begin{tabular}{rrrrr}
\toprule
Level $n$ & $\dim\cZ(\mathrm{hocolim})_n$ & $\dim\mathrm{hocolim}(\cZ)_n$ & Obstruction & $\rho_n$ \\
\midrule
$0$ & $26$ & $1$ & $25$ & $25/26 \approx 96.2\%$ \\
$1$ & $1248$ & $49$ & $1199$ & $1199/1248 \approx 96.1\%$ \\
\bottomrule
\end{tabular}
\end{center}
At every computed level $n \geq 1$, the obstruction ratio
exceeds $92\%$. This is a \emph{lower bound}: the estimate
for $\operatorname{hocolim}(\cZ)$ uses an upper bound (the
$24$-fold tensor product of local affine Yangians reduced by
$23$ balanced tensor product corrections), so the true ratio
is at least as large.
\end{proposition}

\begin{proof}
The global Drinfeld center character is
$\operatorname{ch}(\frakg_{\Delta_5}) = \operatorname{ch}(\fn_+)^2 \cdot 26$,
where $\operatorname{ch}(\fn_+) = \prod_{n \geq 1} (1 - q^n)^{-24}$
(Goettsche formula for $\chi(\mathrm{Hilb}^n(K3))$) and the
numerical factor $26$ is the \emph{ambient-VOA level-$1$ dimension}
arising in the Borcherds BRST construction: the rank-$3$ hyperbolic
Cartan of $\mathfrak{g}_{\Delta_5}$ on $\Lambda^{2,1}_{II}$
(Gritsenko--Nikulin 1998, Theorem~2.1) contributes a three-dimensional
Heisenberg-current level-$1$ subspace, the transverse rank-$22$
Mukai sublattice on $H^2(K3, \Z)$ contributes a $22$-dimensional
level-$1$ subspace, and the BRST $bc$-ghost contributes one residual
level-$1$ generator, summing to $3 + 22 + 1 = 26$ as the
$H^1_{\mathrm{BRST}}$-counted level-$1$ generators of the BKM weight
space
(Borcherds 1986, \emph{Proc.\ Nat.\ Acad.\ Sci.}~83; Borcherds 1992,
\emph{Invent.\ Math.}~109). This is the rank-preserving reconciliation:
Cartan dimension is $3$; the Mukai-lattice rank is $24$; the
level-$1$ BKM weight-space dimension is $26$; these are three distinct
numerics, not a single rank count (see
Remark~\ref{rem:drinfeld-center-rank-discipline} for the full
discipline). At level~$1$:
$\operatorname{ch}(\fn_+)_1 = 24$, so
$\operatorname{ch}(\fn_+)^2_1 = 2 \cdot 24 = 48$, giving
$\operatorname{ch}(\frakg)_1 = 26 \cdot 48 = 1248$.

For the local centers: each of $24$ effective charts contributes
a single-root affine Yangian $Y(\widehat{\fgl}_1)$ with
character $M(q)^2 \cdot P(q)$ (MacMahon squared times Euler).
The $24$-fold balanced tensor product, reduced by $23$ gluing
corrections (each removing one Euler factor), gives
$\operatorname{ch}(\operatorname{hocolim}(\cZ)) = M(q)^{48} \cdot P(q)$.
At level~$1$: $M^{48}_1 = 48$, $P_1 = 1$, giving
$\operatorname{hocolim}(\cZ)_1 = 49$.
The ratio is $1199/1248 \approx 96.1\% > 92\%$.
\end{proof}

\begin{remark}[Interpretation of the non-locality]
\label{rem:k3e-nonlocality-interpretation}
The $>92\%$ non-locality does \emph{not} mean the Drinfeld
center fails to exist. The BKM superalgebra
$\frakg_{\Delta_5}$ is well-defined and unconditional
(constructed from the Mukai lattice, not from the chiral
algebra). The non-locality means: more than $92\%$ of the
global center is invisible to chart-by-chart computation.

The physical source of the non-locality is the \emph{imaginary
root sector} of $\frakg_{\Delta_5}$. Real roots (multiplicity
$c(-1) = 1$ in the Eichler--Zagier convention) correspond to
individual D-branes at smooth points of $K3$; each local chart
detects exactly one real root. Imaginary roots (multiplicity
$c(0) = 10$) correspond to BPS states wrapping full cycles of
$K3$, which no single local chart can see. At level~$1$, the
gap is dominated by the Cartan and the doubling $\fn_-$
(from the Cartan involution); at higher levels, the imaginary
root contributions dominate.
\end{remark}

\begin{remark}[Rank discipline: three distinct numerics of the K3 BKM]
\label{rem:drinfeld-center-rank-discipline}
Three numbers attached to $\mathfrak{g}_{\Delta_5}$ are routinely
conflated; each is load-bearing and refers to a distinct structure.
\begin{enumerate}[label=\textup{(\roman*)}]
\item \emph{Cartan rank} $= 3$. The real simple roots
 $\delta_1,\delta_2,\delta_3$ live in the hyperbolic core
 $\Lambda^{2,1}_{II}$ of Gritsenko--Nikulin 1998, with Gram matrix
 $G_{\mathrm{BKM}} = 2I - 2(J-I) = \begin{pmatrix}2&-2&-2\\-2&2&-2\\-2&-2&2\end{pmatrix}$,
 $\det G_{\mathrm{BKM}} = -32$.
\item \emph{Mukai-lattice rank} $= 24$. The horizontal duality datum
 $\Lambda_{\mathrm{Muk}} = \widetilde H(K3,\mathbb Z)\simeq\mathrm{II}_{4,20}$
 of signature $(4,20)$ carrying the abelian-at-Lie Heisenberg currents.
\item \emph{Level-$1$ BRST-weight-space dimension} $= 26$. The
 Borcherds BRST construction
 $H^1_{\mathrm{BRST}}(V_{\Lambda^{2,1}_{II}}\otimes V_{\mathrm{trans}}\otimes V_{\mathrm{ghost}})$
 with transverse $c=23$ K3 SCFT contributes a level-$1$ count
 of $3+22+1=26$ weight-space generators.
\end{enumerate}
The BKM $\mathfrak{g}_{\Delta_5}$ has $3$ real simple roots in a
rank-$3$ Cartan, infinitely many imaginary simple roots indexed by
norm $(n,\ell,m)$ with multiplicities
$\mathrm{mult}(\alpha) = c_{\mathrm{K3}}(4nm-\ell^2)$
(K3 elliptic-genus Fourier coefficients; Eguchi--Ooguri--Tachikawa
2011), and a $24$-dimensional Mukai grading acting by
$M_{24}$-equivariant symmetry on the imaginary-root sector. Primary
lit: Gritsenko--Nikulin 1998 \emph{Duke Math.\ J.}~94 Thm~2.1 and
\emph{alg-geom/9612004} \S3 (Cartan rank and Gram matrix);
Mukai 1984 (horizontal Mukai rank);
Borcherds 1986 \emph{Proc.\ NAS} 83 / 1992
\emph{Invent.\ Math.}~109 (BRST rank);
Eguchi--Ooguri--Tachikawa 2011 \emph{Exp.\ Math.}~20 (imaginary-root
multiplicity formula).
\end{remark}


\subsection{The Maulik--Okounkov stable envelope bypass}
\label{subsec:mo-bypass-k3e}

The center-hocolim obstruction applies to the algebraic route
($A_{K3 \times E} \to \Rep^{\Eone} \to \cZ$). A fundamentally
different \emph{geometric} route, due to Maulik--Okounkov
(arXiv:1211.1287), constructs the $\Etwo$-braiding
\emph{directly} from stable envelopes on the Hilbert scheme,
bypassing the Drinfeld center entirely. The Maulik--Okounkov
construction is precisely formulated for symplectic resolutions
$\widetilde{X} \to X$ equipped with a Hamiltonian torus action
$T \subset \mathrm{Aut}(\widetilde{X})$ that has isolated
$T$-fixed points and dilates the symplectic form
(Maulik--Okounkov, arXiv:1211.1287, \S3.2).
A generic K3 surface admits no such torus action: $b_1(K3) = 0$,
the Hodge structure $h^{1,0} = h^{0,1} = 0$ forces
$H^0(S, T_S) = 0$, and $\mathrm{Aut}(S)^\circ = 1$
(Mukai 1988, \emph{Inventiones} 94; Nikulin 1980,
\emph{Trudy Mosk.\ Mat.\ Obshch.} 38).
The MO machine therefore does \emph{not} apply to a generic K3;
the bypass below is restricted to the loci of K3 moduli
(ADE/Kummer orbifold points) where a toric chart with a
two-dimensional torus exists by the McKay correspondence.

\begin{proposition}[MO stable envelope bypass; scope-restricted to ADE/Kummer orbifold loci of K3 moduli]
\label{prop:mo-bypass}
Let $S$ be a K3 surface and let $p \in S$ be either an ADE orbifold
point of $S$ \emph{(}so that an analytic neighbourhood of $p$ is the
minimal crepant resolution $\widetilde{\C^2/\Gamma}$ for some
$\Gamma \subset \mathrm{SU}(2)$ finite\emph{)} or, equivalently,
let $S$ itself be replaced by a hyperk\"ahler ALE surface
$\widetilde{\C^2/\Gamma}$ in the noncompact analogue.
At such a point, a local toric torus $T_S = (\C^*)^2$ acts on
the analytic chart by the Kronheimer hyperk\"ahler quotient torus.
Let $T = T_S \subset \mathrm{Aut}(S \times E)$ act on
$X = S \times E$ near $p \times E$ via the embedding into
$\mathrm{Aut}(S)$ on the first factor and trivially on the second
factor \emph{(}the elliptic curve $E$ contributes no algebraic
torus axis: $\mathrm{Aut}(E)^\circ \cong E$ is an abelian variety,
not a $\mathbb{G}_m$; see Lemma~\ref{lem:no-Gm-on-E}\emph{).} Then
the Maulik--Okounkov construction
gives an $R$-matrix
\[
 R = \mathrm{Stab}_C^{-1} \circ \mathrm{Stab}_{C'}
 \;\colon\; K_T(\mathrm{Hilb}^n(X)^T) \to K_T(\mathrm{Hilb}^n(X)^T)
\]
for opposite chambers $C, C'$ in $\mathrm{Lie}(T)^*$, satisfying
the Yang--Baxter equation and defining an $\Etwo$-braiding on
$\bigoplus_{n \geq 0} K_T(\mathrm{Hilb}^n(X))$ \emph{at the chosen
chart}. The $R$-matrix on the Fock representation is governed by
the local toric-chart structure function
\begin{equation}
\label{eq:k3e-structure-function}
 g(u) = \frac{(u - \epsilon_1)(u - \epsilon_2)(u + \epsilon_1 + \epsilon_2)}{(u + \epsilon_1)(u + \epsilon_2)(u - \epsilon_1 - \epsilon_2)}
\end{equation}
where $\epsilon_1, \epsilon_2$ are the equivariant weights of
$T_S = (\C^*)^2$ on the toric chart at $p$ \emph{(}equivalently,
the $\Omega$-background parameters of the local $\C^2 \subset S$
near the ADE/Kummer fixed point\emph{)}, and
$\epsilon_3 = -(\epsilon_1 + \epsilon_2)$ is the $E$-direction
equivariant weight enforced by the threefold CY condition
$\sum \epsilon_i = 0$.
The structure function satisfies $g(u) \cdot g(-u) = 1$
\textup{(}unitarity, from $\sum \epsilon_i = 0$\textup{).}
\end{proposition}

\begin{remark}[Why a generic K3 fails the MO hypothesis: scope sharpening]
\label{rem:generic-k3-no-mo}
For a generic K3 surface $S$ (away from the ADE/Kummer locus of
K3 moduli), the connected component
$\mathrm{Aut}(S)^\circ$ is trivial: $H^0(S, T_S) = 0$ because
$h^{1,0}(S) = h^{0,1}(S) = 0$ and $\Omega^1_S \cong T_S$ via the
holomorphic symplectic form $\omega_S$, so
$H^0(S, T_S) \cong H^0(S, \Omega^1_S) = H^{1,0}(S) = 0$.
There is no continuous algebraic torus $T \to \mathrm{Aut}(S)$
from which to build stable envelopes on $\mathrm{Hilb}^n(S)$.
The natural connected automorphism component of $E$ is the
elliptic curve $E$ itself acting on itself by translation
\emph{(}see Lemma~\ref{lem:no-Gm-on-E}\emph{);} this action is
free \emph{(}translations on $E$ have no fixed points\emph{)} and
$E$ is \emph{not} an algebraic torus $\mathbb{G}_m$ but an
abelian variety, so the group $E$ contains no $\mathbb{G}_m$
that one could use as a torus axis in the MO chamber
construction. The symbol ``$\C^*_E$'' is therefore a misnomer for
this group; the algebraic-group identification is
$\mathrm{Aut}(E)^\circ \cong E$
\emph{(}abelian variety\emph{),} not $\mathrm{Aut}(E)^\circ \cong
\mathbb{G}_m$. A two-parameter family $\epsilon_1, \epsilon_2$ as
``$\Omega$-background weights for K3'' presupposes the absent
two-dimensional toric torus $T_S$ on $S$. Proposition~\ref{prop:mo-bypass}
is restricted accordingly to the ADE/Kummer locus, where such a
$T_S$ exists locally by the McKay correspondence. An unrestricted
global MO bypass is unavailable on generic K3.
\end{remark}

\begin{remark}[Self-dual limit and hyper-K\"ahler preservation]
\label{rem:selfdual-trivial}
\textit{Within the scope of Proposition~\textup{\ref{prop:mo-bypass}}
\textup{(}ADE/Kummer orbifold points of K3 moduli, where the local
toric torus $T_S$ exists\textup{):}}
in the self-dual limit $\epsilon_1 = -\epsilon_2$
(which preserves the hyper-K\"ahler structure of the local toric
chart $\widetilde{\C^2/\Gamma}$), $\epsilon_3 = 0$ and $g(u) = 1$
identically. The $R$-matrix is trivial: the $\Etwo$-braiding is
symmetric, and there is no quantum group structure. Nontrivial
braiding requires \emph{breaking} the hyper-K\"ahler symmetry by
taking $\epsilon_1 \neq -\epsilon_2$. This is the $\Omega$-deformation
on the local toric chart.
\end{remark}

\begin{lemma}[Local-to-global obstruction for the MO bypass on generic K3]
\label{lem:mo-bypass-local-to-global}
Let $S$ be a smooth projective K3 surface in the generic
\emph{(}non-ADE, non-Kummer\emph{)} stratum of K3 moduli. Then
there is no algebraic torus $T \subset \mathrm{Aut}(S \times E)$
of dimension $\geq 2$ acting on $\mathrm{Hilb}^n(S \times E)$ with
isolated $T$-fixed points for $n \geq 2$. Consequently the
Maulik--Okounkov construction does \emph{not} produce an
$R$-matrix on $K_T(\mathrm{Hilb}^n(S \times E))$ for generic K3,
and the local toric-chart structure function~\eqref{eq:k3e-structure-function}
does \emph{not} extend to a global structure function on
$\bigoplus_n K_T(\mathrm{Hilb}^n(S \times E))$ via $T_S$-equivariance.
\end{lemma}

\begin{proof}
The connected component of the algebraic automorphism group
satisfies
\[
 \dim \mathrm{Aut}(S \times E)^\circ
 \;=\; \dim \mathrm{Aut}(S)^\circ + \dim \mathrm{Aut}(E)^\circ
 \;=\; 0 + 1
 \;=\; 1,
\]
the first equality from
$\mathrm{Aut}(S \times E)^\circ \cong \mathrm{Aut}(S)^\circ
\times \mathrm{Aut}(E)^\circ$ \textit{(}any vector field on
$S \times E$ decomposes as $v_S \oplus v_E$ with
$v_S \in H^0(S, T_S)$, $v_E \in H^0(E, T_E)$ via K\"unneth,
since $H^0(S \times E, T_{S \times E}) = H^0(S, T_S) \otimes
H^0(E, \mathcal{O}_E) \oplus H^0(S, \mathcal{O}_S) \otimes
H^0(E, T_E)$ and the cross terms vanish for generic K3 by
$h^{1,0}(S) = 0$ and the K\"unneth formula on the tangent
sheaf of the product\textit{)} and the second
from $H^0(S, T_S) = 0$ for generic K3 (Mukai 1988; the obstruction
to a continuous $\mathrm{Aut}$-action is the vanishing of global
holomorphic vector fields, equivalent here to $h^{1,0}(S) = 0$).
Thus $\dim T \leq 1$. But the MO stable envelope construction
requires $\dim T \geq 2$ to define a nontrivial chamber decomposition
of $\mathrm{Lie}(T)^*$ producing distinct $\mathrm{Stab}_C$
operators (Maulik--Okounkov, arXiv:1211.1287, \S3.2: chambers are
connected components of $\mathrm{Lie}(T)^* \setminus \bigcup_\alpha
\alpha^\perp$ for the $T$-weights $\alpha$ of the symplectic form;
a one-dimensional $T$ has at most two chambers, and the only
nontrivial $\mathrm{Stab}$ operator collapses to the identity
because the moment map is constant along the single direction).
The local toric-chart formula~\eqref{eq:k3e-structure-function}
requires the two parameters $\epsilon_1, \epsilon_2$ as independent
equivariant weights, which presupposes $\dim T_S \geq 2$, absent
for generic K3. Hence the MO bypass at generic K3 does not
produce~\eqref{eq:k3e-structure-function}; only at the ADE/Kummer
locus where $T_S = (\C^*)^2$ acts on local toric charts is the
formula meaningful.
\end{proof}

\begin{lemma}[Sharpening: no \texorpdfstring{$\mathbb{G}_m$}{G-m}-subtorus on \texorpdfstring{$E$}{E} rules out an \texorpdfstring{$E$}{E}-axis chamber]
\label{lem:no-Gm-on-E}
Let $E$ be a smooth projective elliptic curve over $\mathbb{C}$.
Then $\mathrm{Aut}(E)^{\circ}$, as a connected algebraic group,
is isomorphic to $E$ itself \emph{(}translation action\emph{);} it
is an abelian variety, hence \emph{compact}, and contains no
algebraic-torus subgroup $\mathbb{G}_m \subset \mathrm{Aut}(E)^\circ$
of positive dimension. Consequently:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item There is no algebraic-torus action $\mathbb{G}_m \to
 \mathrm{Aut}(E)$ that one could use as the ``$\C^*_E$
 translation torus on~$E$'' in the Maulik--Okounkov apparatus;
 the symbol $\C^*_E$, when used to denote the connected
 automorphism component of an elliptic curve, is a misnomer for
 the abelian group $E$ \emph{(}not the multiplicative group
 $\mathbb{G}_m$\emph{).}
 \item For any $\mathbb{G}_m$-action on $E$ obtained by
 restriction from a larger torus on the ambient space, the
 induced action on $E$ alone is necessarily trivial \emph{(}since
 $\mathrm{Hom}_{\mathrm{alg.\,grp}}(\mathbb{G}_m, E) = 0$\emph{);} the
 action moves the support of length-$n$ subschemes \emph{within}
 a single $E$-fibre rather than between fibres.
\end{enumerate}
\end{lemma}

\begin{proof}
\textit{(}Statement that $\mathrm{Aut}(E)^\circ \cong E$.\textit{)}
Fix the identity element $0 \in E$. The translation map
$\tau\colon E \to \mathrm{Aut}(E)$, $\tau(p)(q) = p + q$, is an
injective algebraic-group homomorphism whose image is the
translation subgroup, a connected, $1$-dimensional algebraic
subgroup of $\mathrm{Aut}(E)$. By the Hodge structure
$h^0(E, T_E) = h^0(E, \mathcal{O}_E) = 1$ (since $T_E \cong
\mathcal{O}_E$ via the holomorphic $1$-form $\omega_E$), the
tangent space at the identity has dimension $1$, so
$\mathrm{Aut}(E)^\circ$ is $1$-dimensional and the translation
map is a surjection of $\mathrm{Aut}(E)^\circ$ onto~$E$.

\textit{(}Statement that $E$ is not isomorphic to $\mathbb{G}_m$
as an algebraic group.\textit{)} Over~$\C$, a $1$-dimensional
connected algebraic group is one of $\mathbb{G}_a = \C$,
$\mathbb{G}_m = \C^\times$, or an elliptic curve~$E$. These three
are pairwise non-isomorphic as algebraic groups: $\mathbb{G}_a$
and $\mathbb{G}_m$ are affine \emph{(}quasi-affine\emph{)} but not
projective, while~$E$ is projective hence not affine. In
particular $E \not\cong \mathbb{G}_m$.

\textit{(}No $\mathbb{G}_m$-subgroup.\textit{)} Any subgroup
homomorphism $\mathbb{G}_m \hookrightarrow \mathrm{Aut}(E)^\circ
\cong E$ is the same datum as an algebraic-group homomorphism
$\phi\colon \mathbb{G}_m \to E$. We claim every such $\phi$ is
constant. Two complementary arguments:

\textit{(rationality.)} $\mathbb{G}_m \cong \mathbb{P}^1 \setminus
\{0, \infty\}$ is a rational variety, hence its image in $E$ is
the image of a rational curve in an abelian variety. Any
morphism from a rational variety to an abelian variety is
constant (a standard rigidity theorem: an abelian variety
contains no nonconstant rational curves, since the pullback of
the global holomorphic $1$-form $\omega_E$ extends across the
removed points $\{0, \infty\}$ to a global holomorphic $1$-form
on $\mathbb{P}^1$, which vanishes since $H^0(\mathbb{P}^1,
\Omega^1) = 0$). Hence $\phi$ is constant.

\textit{(fundamental-group obstruction; cross-check.)}
The topological fundamental groups satisfy $\pi_1(\mathbb{G}_m,
\C\text{-pts}) = \Z$ and $\pi_1(E, \C\text{-pts}) = \Z^2$. A
nonconstant algebraic-group homomorphism would be surjective
\emph{(}the image is a connected algebraic subgroup of $E$, and
the only proper one is trivial\emph{)} hence a finite covering;
but a finite covering induces an injection of finite-index
subgroups on $\pi_1$, which is incompatible with $\Z
\hookrightarrow \Z^2$ as a finite-index subgroup. So $\phi$ is
constant.

Hence there is no positive-dimensional torus subgroup of
$\mathrm{Aut}(E)^\circ$.

\textit{(}Action by restriction.\textit{)} Suppose
$T \to \mathrm{Aut}(S \times E)$ is any algebraic-torus action
with $T \cong \mathbb{G}_m^k$. The composition with the
projection $\mathrm{Aut}(S \times E)^\circ \to \mathrm{Aut}(E)^\circ$
gives a homomorphism $T \to \mathrm{Aut}(E)^\circ \cong E$,
which is trivial by the previous paragraph. Therefore the
$T$-action descends through the projection to $\mathrm{Aut}(S)$,
acting trivially on the $E$-coordinate. In particular the
$T$-fixed locus on $\mathrm{Hilb}^n(S \times E)$ projects to
the entire $E$-component of any subscheme \emph{(}every point of
$E$ is fixed by the $E$-action factor\emph{);} fixed points are
not isolated in the $E$-direction.
\end{proof}

\begin{remark}[Consequence for the \texorpdfstring{$E$}{E}-axis chamber]
\label{rem:no-E-axis-chamber}
\textit{Combined with Lemma~\ref{lem:mo-bypass-local-to-global},
Lemma~\ref{lem:no-Gm-on-E} sharpens the obstruction.} The
schematic ``$T = T_S \times \C^*_E$'' invoked in
Proposition~\ref{prop:mo-bypass} is, even at the ADE/Kummer
locus, a $T = T_S \times E$ \emph{(}with $T_S = (\C^*)^2$ a genuine
algebraic torus on the local chart and $E$ the elliptic
group acting by translation\emph{).} The $E$-direction does
\emph{not} provide an additional torus axis for the MO chamber
decomposition; only $T_S$ contributes the two equivariant
weights $(\epsilon_1, \epsilon_2)$. The third weight
$\epsilon_3 = -(\epsilon_1 + \epsilon_2)$ enforced by
$\sum \epsilon_i = 0$ is therefore not an independent
equivariant weight on the $E$-direction but a
\emph{constraint} on $(\epsilon_1, \epsilon_2)$ coming from
the CY$_3$ Calabi--Yau condition $K_{S \times E} \cong
\mathcal{O}_{S \times E}$. In particular the $R$-matrix
of~\eqref{eq:k3e-structure-function} is purely a function of
the $T_S$-weights; the $E$-factor enters only through the
unitarity constraint, not through an additional spectral parameter.
\end{remark}

\begin{remark}[Where the K3 fibration framework rescues a global structure]
\label{rem:k3-fibration-rescue}
\textit{Lemma~\ref{lem:no-Gm-on-E} does not preclude global
geometric routes to the K3 Yangian, but it sharply restricts
which geometric inputs can serve.} The two surviving routes are:
\begin{enumerate}[label=\textup{(\Roman*)}]
 \item \emph{Cocycle assembly across the ADE/Kummer locus.}
 At each ADE/Kummer point $p_\alpha \in \mathcal{M}_{K3}$ a
 local toric chart provides $T_{S, \alpha} = (\C^*)^2$ and a
 local structure function $g_\alpha(u)$. A global K3 Yangian
 would require coherent gluing across the moduli stratification
 \emph{(}a 1-cocycle on $\mathcal{M}_{K3}$ with values in
 isomorphisms of $g_\alpha$\emph{).} This is the assembly the
 BFN route attempts to bypass via 3d $\mathcal{N} = 4$ gauge
 theory but which the MO route cannot produce on its own.
 \item \emph{Schiffmann--Vasserot-style K3 cohomological-Hall route.}
 The cohomological Hall algebra
 $\mathcal{H}(\mathrm{Coh}(K3))$ on the abelian category of
 coherent sheaves on~$K3$, in the spirit of Schiffmann--Vasserot's
 $\bC^2$ construction \emph{(}arXiv:1202.2756\emph{)} and
 Kontsevich--Soibelman's general CoHA framework
 \emph{(}arXiv:1006.2706\emph{),} is graded by Mukai vectors
 $v = (r, c_1, \mathrm{ch}_2) \in \widetilde{H}(K3, \Z)$ with
 multiplication the Hall product on extensions. This product is
 intrinsic to the abelian category and requires \emph{no} torus
 action; it provides a moduli-of-K3-modular input to the K3
 Yangian construction that the MO route cannot supply. Whether the
 resulting CoHA admits a Drinfeld-double presentation matching
 $C(\frakg_{K3}, q)$ globally over K3 moduli is the K3 analogue of
 the Schiffmann--Vasserot identification
$\mathcal{H}(\bC^2) = Y^+(\widehat{\mathfrak{gl}}_1)$ and remains
conjectural \emph{(}part of CY-C\emph{).}
\end{enumerate}
\end{remark}

\begin{remark}[Two routes to \texorpdfstring{$\Etwo$}{E2} for \texorpdfstring{$K3 \times E$}{K3 x E}]
\label{rem:two-routes-k3e}
Two independent routes produce $\Etwo$-braiding for $K3 \times E$,
with different hypotheses:
\begin{enumerate}[label=\textup{(\Alph*)}]
 \item \emph{Drinfeld center route} (global on $K3 \times E$;
 conditional on the framed object-level CY-A$_3$ hypotheses and the
 BZFN Morita/Hochschild comparison hypotheses).
 $A_{K3 \times E}^{(\Sigma_2,C)}$ (the framed $\Eone$-chiral algebra
 on the fixed H1--H4 specialisation locus)
 $\to \Rep^{\Eone}(A_{K3 \times E}^{(\Sigma_2,C)})$
 $\to \cZ(\Rep^{\Eone}(A_{K3 \times E}^{(\Sigma_2,C)}))$
 $\simeq \Rep^{\Etwo}(\Zder(A_{K3 \times E}^{(\Sigma_2,C)}))$.
 This route passes through the center-hocolim obstruction
 ($>92\%$ non-local at the chain level).
 \item \emph{MO stable envelope route, scope-restricted to
 ADE/Kummer orbifold loci of K3 moduli} (Proposition~\ref{prop:mo-bypass};
 Lemma~\ref{lem:mo-bypass-local-to-global} and
 Lemma~\ref{lem:no-Gm-on-E} force this restriction, since $E$
 contributes no $\mathbb{G}_m$ axis).
 $\mathrm{Hilb}^n(K3 \times E)$
 $\to K_T(\mathrm{Hilb}^n)$
 $\to R = \mathrm{Stab}_C^{-1} \circ \mathrm{Stab}_{C'}$,
 with $T = T_S = (\C^*)^2$ acting on the local toric chart
 \emph{(}the $E$-factor contributes no torus axis\emph{).}
 Within scope, this route constructs the $R$-matrix directly from
 geometry with no Drinfeld center computation.
\end{enumerate}
By Drinfeld's uniqueness theorem, the $R$-matrices from
routes~(A) and~(B) agree on evaluation modules \emph{wherever both
are defined} \emph{(}i.e., at the ADE/Kummer locus of K3 moduli where
route~(B) is defined\emph{).} Route~(B) proves the braiding
\emph{exists at those scope-permitted points}; route~(A) identifies
\emph{what algebra} it comes from. The structure
function~\eqref{eq:k3e-structure-function} is the same in both
routes \emph{within the route-(B) scope}; outside that scope only
route~(A) speaks.
\end{remark}


\subsection{The derived center \texorpdfstring{$\Zder(H_{\mathrm{Muk}})$}{Zder(H-Muk)}}
\label{subsec:derived-center-hmuk}

The Mukai Heisenberg $H_{\mathrm{Muk}}$ (rank~$24$, Mukai
pairing of signature $(4, 20)$) is the classical limit of the
$K3 \times E$ chiral algebra. Its derived center
$\Zder(H_{\mathrm{Muk}}) = \HH^\bullet(H_{\mathrm{Muk}}, H_{\mathrm{Muk}})$
(this is the algebraic derived center, \emph{not} the
categorical Drinfeld center) is computable.

\begin{proposition}[Derived center of \texorpdfstring{$H_{\mathrm{Muk}}$}{H-Muk}]
\label{prop:derived-center-h-muk}
For the rank-$24$ Mukai Heisenberg $H_{\mathrm{Muk}}$
(class~$G$, shadow depth~$2$), the Hochschild cohomology is:
\begin{align*}
 \HH^0(H_{\mathrm{Muk}}, H_{\mathrm{Muk}}) &= \C &&(\text{unit: identity endomorphism}), \\
 \HH^1(H_{\mathrm{Muk}}, H_{\mathrm{Muk}}) &= \C^{24} &&(\text{derivations: one per Mukai direction}), \\
 \HH^2(H_{\mathrm{Muk}}, H_{\mathrm{Muk}}) &= \mathrm{Sym}^2(\C^{24}) &&(\text{normal-ordered products $:J_i J_j:$}), \\
 \HH^k(H_{\mathrm{Muk}}, H_{\mathrm{Muk}}) &= 0 \quad\text{for } k \geq 3 &&(\text{class $G$: shadow tower truncates}).
\end{align*}
The dimensions are $\dim \HH^0 = 1$, $\dim \HH^1 = 24$,
$\dim \HH^2 = \binom{25}{2} = 300$, for a total dimension
of $325$. The Euler characteristic is
$\chi(\HH^\bullet) = 1 - 24 + 300 = 277$.

The Gerstenhaber bracket on $\HH^1$ is:
\[
 \{J_i, J_j\} = \omega^{ij} \in \HH^0 = \C,
\]
where $\omega^{ij}$ is the inverse Mukai pairing
\textup{(}signature $(4, 20)$\textup{)}.
\end{proposition}

\begin{proof}
The Heisenberg $H_{\mathrm{Muk}}$ is the free chiral algebra
on $V = \C^{24}$ with pairing $\omega$ (the Mukai pairing).
It is a class~$G$ algebra (all higher operations $m_k = 0$
for $k \geq 3$), so the Hochschild cohomology is polynomial:
$\HH^k = \mathrm{Sym}^k(V)$ for $k = 0, 1, 2$ and zero for
$k \geq 3$ (Theorem~H of Volume~I, applied to the rank-$24$
Heisenberg). The Gerstenhaber bracket
$\{-, -\} \colon \HH^1 \otimes \HH^1 \to \HH^0$ is the
pairing dual to $\omega$: on derivations $\partial_{J_i}$
and $\partial_{J_j}$, the bracket is the OPE double-pole
residue $\omega(J_i, J_j) = \omega^{ij}$.
\end{proof}

\begin{remark}[Drinfeld center versus derived center for \texorpdfstring{$K3 \times E$}{K3 x E}]
\label{rem:drinfeld-vs-derived-k3e}
The Drinfeld double $\mathrm{Drin}(H_{\mathrm{Muk}})$ has
dimension $49 = 24 + 24 + 1$
(\S\ref{sec:drinfeld-double-slab};
Remark~\ref{rem:drinfeld-center-heisenberg-verification}),
while the derived center $\Zder(H_{\mathrm{Muk}})$
has dimension $325 = 1 + 24 + 300$
(Proposition~\ref{prop:derived-center-h-muk}).
These are \emph{different algebraic objects}:
\begin{center}
\begin{tabular}{lll}
\toprule
 & Drinfeld double $\mathrm{Drin}(H_{\mathrm{Muk}})$ & Derived center $\HH^\bullet(H_{\mathrm{Muk}})$ \\
\midrule
\emph{Dimension} & $49$ & $325$ \\
\emph{Type} & Lie algebra (Drinfeld double) & graded commutative algebra \\
\emph{Generators} & $J_i, J_i^*, c$ & $1, J_i, {:}J_i J_j{:}$ \\
\emph{$\Etwo$ datum} & $R = \exp(\omega^{-1})$ & $\{J_i, J_j\} = \omega^{ij}$ \\
\bottomrule
\end{tabular}
\end{center}
They produce the \emph{same} $\Etwo$-braided representation
category by the BZFN theorem (Theorem~\ref{thm:bzfn}):
\[
 \cZ(\Rep^{\Eone}(H_{\mathrm{Muk}}))
 \;\simeq\;
 \Rep^{\Etwo}(\mathrm{Drin}(H_{\mathrm{Muk}}))
 \;\simeq\;
 \Rep^{\Etwo}(\HH^\bullet(H_{\mathrm{Muk}}, H_{\mathrm{Muk}})).
\]
The matching is at the level of $\Etwo$-braiding data:
the $R$-matrix exponent $\omega^{ij}$ from the Drinfeld double
\emph{equals} the Gerstenhaber bracket coefficient $\omega^{ij}$
from the derived center. This is a single numerical datum
(the inverse Mukai pairing) that controls the $\Etwo$-structure
on both sides.
\end{remark}


\subsection{Three obstructions to \texorpdfstring{$\Eone \to \Etwo$}{E1 -> E2} for \texorpdfstring{$K3 \times E$}{K3 x E}}
\label{subsec:three-obstructions-k3e}

\begin{remark}[Three obstructions to \texorpdfstring{$\Etwo$}{E2} promotion]
\label{rem:three-obstructions-k3e}
The passage from $\Eone$ to $\Etwo$ for $K3 \times E$ faces
three independent obstructions:

\begin{enumerate}[label=\textup{(\arabic*)}]
 \item \textbf{Symmetric braiding from $E_3$ restriction.}
 The $E_3$-structure on $\HH_\bullet(\cC_{K3 \times E})$
 restricts to $\Etwo$ with symmetric braiding
 ($\pi_1(\mathrm{Conf}_2(\R^3)) = 0$; Proposition~\ref{prop:e3-symmetric-braiding}).
 This sees only $q = 1$ (classical limit). The non-symmetric
 braiding requires descending to $\Eone$ (via the
 $\Omega$-deformation) and applying the Drinfeld center.
 This is the standard mechanism, not an obstruction per se.

 \item \textbf{Center-hocolim non-commutation.}
 For $K3 \times E$, more than $92\%$ of the global Drinfeld
 center is non-local
 (Proposition~\ref{prop:k3e-nonlocality-quantification}).
 The imaginary roots of $\frakg_{\Delta_5}$ (BPS states
 wrapping full cycles of $K3$) are invisible to any local
 chart. \emph{Bypass}: the Maulik--Okounkov stable envelope
 construction (Proposition~\ref{prop:mo-bypass}) provides the
 $R$-matrix directly from the geometry of $\mathrm{Hilb}^n(K3 \times E)$,
 completely bypassing the center computation.

 \item \textbf{CY-A$_3$ dependence.}
 The $\Eone$-chiral algebra $A_{K3 \times E}$ is constructed in the
 $\infty$-categorical framework by Theorem~CY-A$_3$
 (Theorem~\ref{thm:cy-to-chiral-d3},
 Theorem~\ref{thm:derived-framing-obstruction}).
 The MO $R$-matrix and the BKM superalgebra are unconditional;
 their identification as the Drinfeld center of
 $\Rep^{\Eone}(A_{K3 \times E})$ follows from CY-A$_3$.
\end{enumerate}

Obstruction~(2) is bypassed by the MO construction.
Obstruction~(3) is resolved by CY-A$_3$
(Theorem~\ref{thm:cy-to-chiral-d3}). The MO $R$-matrix on
$K_T(\mathrm{Hilb}^n(K3 \times E))$ defines an $\Etwo$-braiding
with structure function~\eqref{eq:k3e-structure-function};
the BKM superalgebra $\frakg_{\Delta_5}$ provides the algebraic
framework. The identification of these as the Drinfeld center and
derived center of $A_{K3 \times E}$ is a consequence of CY-A$_3$.
\end{remark}

\begin{remark}[Imaginary roots and non-locality]
\label{rem:imaginary-root-nonlocality}
The source of the center-hocolim gap for $K3 \times E$ is
the \emph{imaginary root sector} of the BKM superalgebra
$\frakg_{\Delta_5}$. The root system has three types:
\begin{itemize}
 \item \emph{Real roots} ($\alpha^2 = -2$, multiplicity $c(-1) = 1$
 in the Eichler--Zagier convention): local. Each stability
 chart detects one real root (a D-brane at a smooth point).
 \item \emph{Imaginary roots} ($\alpha^2 = 0$, multiplicity
 $c(0) = 10$): non-local. These correspond to BPS states
 wrapping isotropic cycles in the Mukai lattice. No single
 chart detects them; they are inherently global.
 \item \emph{Super-imaginary roots} ($\alpha^2 > 0$, multiplicity
 $c(\alpha^2/2)$ from the Fourier coefficients of
 $\phi_{0,1}$): non-local.
\end{itemize}
The non-locality grows with $|\alpha|$: deeper into the
imaginary root sector, the contributions become increasingly
invisible to local computations. The Borcherds denominator

\subsection{The MO \texorpdfstring{$R$}{R}-matrix at charge~\texorpdfstring{$2$}{2}: explicit computation}
\label{subsec:mo-rmatrix-charge2}

At charge~$n = 2$, the Hilbert scheme $\mathrm{Hilb}^2(K3)$ has
Euler characteristic
\begin{equation}
\label{eq:hilb2-euler}
 \chi(\mathrm{Hilb}^2(K3)) = p_{24}(2) = 324,
\end{equation}
where $p_{24}(n)$ denotes the number of $24$-coloured partitions of~$n$
(coefficient of $q^n$ in $\prod_{k \geq 1} (1 - q^k)^{-24}$).
The $324$ $T$-fixed states decompose into three types:
\begin{enumerate}[label=\textup{(\alph*)}]
 \item $24$ \emph{row states} $(2)_i$: a length-$2$ subscheme at
 a single $K3$ point in direction~$i$ ($i = 1, \ldots, 24$).
 Boxes at $(0,0)$ and $(0,1)$; contents $\{0, \epsilon_2\}$.
 \item $24$ \emph{column states} $(1,1)_i$: a fat point at
 direction~$i$ with tangent in the normal direction.
 Boxes at $(0,0)$ and $(1,0)$; contents $\{0, \epsilon_1\}$.
 \item $276 = \binom{24}{2}$ \emph{two-point states}
 $(1)_i + (1)_j$ ($i < j$): two simple points at distinct $K3$
 directions. Each box at the origin of its local chart; both contents~$0$.
\end{enumerate}
Total: $24 + 24 + 276 = 324$.

\begin{proposition}[MO \texorpdfstring{$R$}{R}-matrix at charge~\texorpdfstring{$2$}{2}]
\label{prop:mo-rmatrix-charge2}
\label{prop:charge2-rmatrix}
For $K3 \times E$ with generic $\Omega$-background
$(\epsilon_1, \epsilon_2)$ ($\epsilon_1 \neq - \epsilon_2$),
the MO $R$-matrix at charge~$2$ is a $324 \times 324$
\textbf{diagonal} matrix
$R(u) = \mathrm{diag}(R_\lambda(u))_{\lambda}$
in the coloured-partition basis,
with exactly three distinct eigenvalue types:
\begin{align}
\label{eq:charge2-eigenvalues}
 R_{(2)_i}(u) &= g(u) \cdot g(u + \epsilon_2),
 &&\text{multiplicity } 24, \\
 R_{(1,1)_i}(u) &= g(u) \cdot g(u + \epsilon_1),
 &&\text{multiplicity } 24, \nonumber \\
 R_{(1)_i + (1)_j}(u) &= g(u)^2,
 &&\text{multiplicity } 276. \nonumber
\end{align}
Here $g(u)$ is the structure
function~\eqref{eq:k3e-structure-function}.
These eigenvalues satisfy:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item \textbf{Unitarity}: $R_{\lambda,\mu}(u) \cdot R_{\mu,\lambda}(-u) = 1$
 for all pairs $(\lambda, \mu)$ of charge-$2$ partitions.
 Proof: $g(x) \cdot g(-x) = 1$ applied box-by-box.
 \item \textbf{Yang--Baxter}: all $8$ triples $(\lambda, \mu, \nu)$
 of charge-$2$ partitions satisfy the YBE\@.
 Since the $R$-matrix is diagonal, the YBE reduces to
 scalar commutativity.
 \item \textbf{MO\,=\,Yangian}: the eigenvalues match the
 Yangian coproduct prediction from the Miura factorization
 $\Delta_z(T(u)) = T^L(u) \cdot T^R(u - z)$ (cf.\
 \S\ref{sec:e1-chiral-bialgebras} and
 Theorem~\ref{thm:mo-chiral-rmatrix}).
\end{enumerate}
\end{proposition}

\begin{proof}
The box-content formula
$R_\lambda(u) = \prod_{s \in \lambda} g(u + c(s))$
with $c(i,j) = \epsilon_1 i + \epsilon_2 j$
gives~\eqref{eq:charge2-eigenvalues} directly.
Unitarity (i) follows from $g(x) g(-x) = 1$:
writing $w_s = u + c(s) - c(t)$ for
each pair $(s,t)$ in $\lambda \times \mu$,
\[
 R_{\lambda,\mu}(u) \cdot R_{\mu,\lambda}(-u)
 = \prod_{s, t} g(w_s) \cdot g(-w_s) = 1.
\]
The YBE (ii) holds because the $R$-matrix is diagonal:
each triple $(\lambda, \mu, \nu)$ gives a scalar equation
$R_{12} R_{13} R_{23} = R_{23} R_{13} R_{12}$
satisfied by commutativity.
The MO\,=\,Yangian comparison (iii): both constructions produce
the same box-content formula by Drinfeld's uniqueness theorem
(on evaluation modules, Proposition~\ref{prop:mo-bypass} and
Vol~I \cite{VolI}).
A direct symbolic calculation gives exact cancellation of the
difference between the two box-content formulas.
\end{proof}

\subsection{Braiding}
\label{subsec:chiral-qg-braiding}

The $\Etwo$-braiding on $\Rep^{\Etwo}(Y(\fg_{K3}))$ is:
\[
 \beta_{V_u^{(a)},\, V_v^{(b)}}
 = P_{ab} \cdot R_{ab}(u - v),
\]
where $P$ is the graded permutation and $R_{ab}(u)$ is the
Maulik--Okounkov $R$-matrix.

For the abelian Yangian at charge~$1$: different Mukai directions
decouple, and the braiding within a single direction $a$ is
\[
 R_a(u) = g_a(u) = \frac{u - h_a}{u + h_a},
\]
where $h_a$ is the deformation parameter for direction $a$. At
charge~$2$ within direction $a$, the $R$-matrix is the $2 \times 2$
diagonal matrix (Proposition~\ref{prop:charge2-rmatrix}):
\[
 R^{(a)}_{n=2}(u)
 = \mathrm{diag}\bigl(g_{K3}(u)\, g_{K3}(u + h_a),\;
 g_{K3}(u)\, g_{K3}(u - h_a)\bigr)
\]
in the basis $\{|(2)_a\rangle, |(1,1)_a\rangle\}$, where
$g_{K3}(u) = \prod_{i=1}^{24}(u - h_i)/(u + h_i)$ is the full
degree-$(24,24)$ structure function.

\begin{remark}[Non-symmetric braiding]
\label{rem:chiral-qg-nonsymmetric}
The braiding satisfies
$R(u) \cdot R(-u) = \mathrm{Id}$ (unitarity), but $R(u) \neq \mathrm{Id}$
generically: the category is braided ($\Etwo$), \emph{not} symmetric
($E_\infty$). Passing to the symmetric quotient would kill the
quantum group structure entirely.
\end{remark}

\subsection{Fusion rules}
\label{subsec:chiral-qg-fusion}

The tensor product $V_u^{(a)} \otimes V_v^{(b)}$ is:
\begin{enumerate}[label=(\alph*)]
 \item \emph{Irreducible} when $u - v$ avoids the poles and zeros
 of $R_{ab}(u - v)$. For $a \neq b$ (different Mukai directions):
 always irreducible (abelian decoupling).
 \item \emph{Reducible} when $a = b$ and $u - v = \pm h_a$.
 At $u - v = h_a$: the structure function $g_a(h_a) = 0$ (the $R$-matrix
 vanishes), producing a submodule (bound state).
 At $u - v = -h_a$: the structure function $g_a(-h_a)$ has a pole,
 producing a quotient (anti-bound state).
\end{enumerate}
For the Kummer $K3$ parametrisation with $h_a = \epsilon$ ($a = 1, \dots, 4$)
and $h_a = -\epsilon/5$ ($a = 5, \dots, 24$), the fusion locus has two
distinct magnitudes: $|\epsilon|$ with multiplicity $4$ and $|\epsilon/5|$
with multiplicity $20$.

\begin{remark}[Humbert \texorpdfstring{$H_4$}{H-4} discriminant-\texorpdfstring{$4$}{4} real-multiplication and the Kuga--Satake lift]
\label{rem:dc-humbert-H4-RM}
\label{rem:dc-humbert-H4-kuga-satake}
The Humbert divisor $H_4 \subset \mathcal A_2$ of discriminant $4$
(van der Geer $1988$ Ch.\ IX, $\Delta \equiv 0 \pmod 4$) parametrises
principally polarised abelian surfaces $A$ whose endomorphism ring
contains the rank-$2$ real-multiplication order
\[
 \mathcal O_{4} \;=\; \Z[\phi]
 \qquad\text{with}\qquad \phi^2 = 4,
\]
i.e.\ $\text{End}(A) \supset \Z + 2\Z\cdot\mathrm{id} + \Z\cdot\sigma$
where $\sigma$ is a $(2,2)$-isogeny. Concretely, $A$ sits in the exact
sequence $0 \to K \to E_1\times E_2 \to A \to 0$ with $K \cong (\Z/2)^2$
a maximal-isotropic subgroup of the standard Weil pairing: $A$ is the
$(2,2)$-isogeny quotient of a product of two elliptic curves.
This is the ``squared-CM'' reading: on the product $E_1 \times E_2$ the
diagonal endomorphism $(2\cdot\mathrm{id}, 2\cdot\mathrm{id})$ descends
to $\phi \in \mathrm{End}(A)$ with $\phi^2 = 4$, and $H_4$ is exactly
the locus where this descent is compatible with the principal
polarisation. The distinction from the $\mathbb Z[\sqrt 2]$ setting
(disc-$8$ RM, Humbert surface $H_8$) is sharp:
$H_4$ is the $(2,2)$-isogeny locus, $H_8$ is the genuine
$\mathbb Q(\sqrt 2)$-RM locus. The Koszul-locus obstruction
(Vol.~I \cite{VolI}) is at $H_4$,
not at $H_8$: the $\hbar^2 = -1/8$ root-of-unity specialisation
touches the $(2,2)$-isogeny, not $\mathbb Q(\sqrt 2)$-RM.

\emph{Kuga--Satake lift to K3:} the Kuga--Satake construction
(Deligne $1972$; Morrison $1988$) sends a principally polarised
abelian surface $A$ with RM of discriminant $\Delta$ to a K3 surface
$\mathrm{KS}(A)$ whose transcendental lattice $T(\mathrm{KS}(A))$ has
discriminant form squared with respect to $T(A)$; at
$\Delta = 4$, Morrison $1984$ Thm.\ $3$ gives
\[
 \mathrm{Pic}(\mathrm{KS}(A))\big|_{H_4}
 \;\supset\; U(2) \;\oplus\; \langle -4 \rangle,
\]
a rank-$2$ sublattice of discriminant $-4 \cdot \det(U(2)) = 16$ (or $4$
after suitable primitive-embedding reduction, following the
Nikulin lattice-theoretic discipline). Equivalently, $\mathrm{KS}(A)$
is an \emph{elliptic K3 with a double section}: the Weierstrass model
is $y^2 = x^3 + a_4(t)x + a_6(t)$ with $a_4, a_6$ having a common
double root -- the section-of-sections constraint that encodes the
$(2,2)$-isogeny descent from $E_1\times E_2$. The endomorphism ring of
the CM-lift $H^2(\mathrm{KS}(A), \Q)_{\mathrm{trans}}$ contains the
imaginary-CM order
\[
 \Z[2i] \;=\; \Z[\sqrt{-4}] \;\hookrightarrow\; \mathrm{End}(T(\mathrm{KS}(A))\otimes\Q),
\]
matching the imaginary-CM structure: the Kuga--Satake functor
$\mathrm{KS}$ flips the real-quadratic order $\Z[\phi],\ \phi^2 = 4$
(on $\mathcal A_2$) to the imaginary-quadratic order $\Z[\psi],\
\psi^2 = -4$ (on the K3 side), because the Hodge structure weights
invert under the passage from $H^1(A)$ (weight $1$) to
$H^2(\mathrm{KS}(A))$ (weight $2$).

\emph{Consequence for $\mathbf H_{\Delta_5}$:} the Drinfeld-centre
specialisation $\mathcal Z(\mathrm{Rep}^{E_1}(\mathbf H_{\Delta_5}))$
acquires along $H_4$ a $\Z/2$-equivariant Kummer structure
(Vol.~I \cite{VolI}): the
$(2,2)$-isogeny descent descends to the half-braiding $\sigma$ of the
Drinfeld centre as a $\Z/2$-involution, and the Kuga--Satake
$\Z[2i] = \mathbb Z[\sqrt{-4}]$ lifts to a CM-involution on the
category of level-one modules. Primary:
van der Geer $1988$ ``Hilbert Modular Surfaces'' Ch.\ IX
\S 1--3 (Humbert surfaces, discriminant classification);
Morrison $1984$ Invent.\ Math.\ $75$ Thm.\ 3 (lattice embeddings on
Kuga--Satake K3s); Deligne $1972$ Invent.\ Math.\ ``Kuga--Satake and
Hodge conjecture for K3''; Nikulin $1980$ Math.\ USSR Izv.\ $14$
``Integral symmetric bilinear forms''. Cross-ref
Vol.~I \cite{VolI};
Vol.~I \cite{VolI}.
\end{remark}

\begin{remark}[Drinfeld centre of \texorpdfstring{$\mathrm{Rep}(\mathbf{H}_{\Delta_5})$}{Rep(H-Delta-5)}: general structure]

\label{eq:dc-center-yd-equivalence}
 \mathcal{Z}(\mathrm{Rep}(\mathbf{H}_{\Delta_5}))
 \;\simeq\;
 \mathrm{YD}^{A_\infty}_{\mathbf{H}_{\Delta_5}}
 \;=\;
 \bigl\{ (V,\rho,\delta) \;:\; V \text{ a super-$\mathbf{H}_{\Delta_5}$-module,}\;
 \delta \text{ a compatible comodule structure} \bigr\},

\begin{remark}[Non-triviality of the centre ratio: chiral vs quasi-Hopf]
\label{rem:dc-center-ratio}
\index{Drinfeld center!chiral vs quasi-Hopf}
For a quasi-Hopf algebra~$U$, the Drinfeld centre of
$\mathrm{Rep}(U)$ is equivalent to $\mathrm{Rep}(D(U))$ where $D(U)$
is Majid's quantum double. For
$\mathbf{H}_{\Delta_5}$, this construction must be distinguished from
the \emph{chiral derived centre} of the underlying $E_1$-chiral
algebra: the two coincide only for finite-dimensional quasi-Hopf
input. Since $\mathbf{H}_{\Delta_5}$ is $A_\infty$-quasi-Hopf with
infinite-dimensional BKM imaginary cone, the Drinfeld centre

larger} category than $\mathrm{Rep}^{E_2}(Z^{\mathrm{der}}_{\mathrm{ch}}(\mathbf{H}_{\Delta_5}))$:
\begin{equation}
\label{eq:dc-center-ratio-inclusion}
 \mathrm{Rep}^{E_2}(Z^{\mathrm{der}}_{\mathrm{ch}}(\mathbf{H}_{\Delta_5}))
 \;\hookrightarrow\;
 \mathcal{Z}(\mathrm{Rep}(\mathbf{H}_{\Delta_5}))
 \;\to\;
 \mathrm{YD}^{A_\infty}_{\mathbf{H}_{\Delta_5}}.
\end{equation}
The first arrow lands on the \emph{central subcategory} coming from
the BZFN identification
(Theorem~\ref{thm:bzfn}) at the chain level; the second arrow is the

Koszul locus $\mathcal{U}^{K3}_{\mathrm{Kosz}} = \overline{\mathcal{A}_2}
\setminus (H_1 \cup H_4)$, the first arrow is an equivalence (the
$\mu_8$-gerbe is trivialisable, the $A_\infty$-tower truncates). Off
the Koszul locus, on $H_1 \cup H_4$, there are non-trivial quotient
cohomology classes measured by the $\mu_8$-gerbe obstruction. Primary:
Schauenburg~\cite{Schauenburg1998DoubleQHopf} for the Yetter--Drinfeld

\section{Explicit Yetter--Drinfeld coherence tower}
\label{sec:dc-yd-tower}

The $A_\infty$-Yetter--Drinfeld coherence tower
$\delta^{(n)}\colon V \to V\otimes \mathbf{H}_{\Delta_5}^{\otimes n}$
on the Drinfeld centre (Remark~\ref{rem:dc-h-delta5-center}
and Remark~\ref{rem:dc-fricke-s-matrix}) scales with the
Borcherds leg $(\Phi_{10}^{\mathrm{un}}/\eta^{24})^{\lceil n/2\rceil}$. The explicit
low-order entries of the tower, the Yetter--Drinfeld cocycle condition
at $n = 1, 2, 3$, and the universal three-faces identity
(Vol.~III Theorem~\ref{thm:bkm-universal-identity}) together
pin the categorical scalar-shadow controlling the coherence tower on
the $\Psi$-landscape.

\begin{proposition}[Yetter--Drinfeld coherence tower: low-order formulas]
\label{prop:dc-yd-tower}
\index{Yetter--Drinfeld tower!explicit low orders}
\index{coherence tower!$\mathbf{H}_{\Delta_5}$}
On the Drinfeld centre
$\mathcal{Z}(\mathrm{Rep}(\mathbf{H}_{\Delta_5})) \simeq
\mathrm{YD}^{A_\infty}_{\mathbf{H}_{\Delta_5}}$
(Remark~\ref{rem:dc-h-delta5-center}), the
$A_\infty$-Yetter--Drinfeld coherence tower
$\{\delta^{(n)}\}_{n\geq 1}$ admits the explicit low-order
presentation
\begin{align}
 \delta^{(1)} &\;=\; \frac{\Phi_{10}^{\mathrm{un}}}{\eta^{24}}\cdot\beta_{\mathrm{YD}},
 \label{eq:dc-delta-1}\\[2pt]
 \delta^{(2)} &\;=\; \Bigl(\frac{\Phi_{10}^{\mathrm{un}}}{\eta^{24}}\Bigr)^{\!2}
 \cdot \alpha_2,
 \qquad
 \alpha_2 \;=\; \tfrac{1}{2}[\beta_{\mathrm{YD}}, \beta_{\mathrm{YD}}]_{(2)},
 \label{eq:dc-delta-2}\\[2pt]
 \delta^{(3)} &\;=\; \Bigl(\frac{\Phi_{10}^{\mathrm{un}}}{\eta^{24}}\Bigr)^{\!2}
 \cdot \alpha_3,
 \qquad
 \alpha_3 \;=\; \zeta(3)\cdot c_{\mathrm{symm}}
 + \tfrac{1}{6}[\beta_{\mathrm{YD}},[\beta_{\mathrm{YD}},\beta_{\mathrm{YD}}]]_{(3)},
 \label{eq:dc-delta-3}
\end{align}
where $\beta_{\mathrm{YD}}\in\mathrm{Hom}(V, V\otimes\mathbf{H}_{\Delta_5})$
is the primary Yetter--Drinfeld cocycle (the half-braiding
comodule datum of Definition~\ref{def:drinfeld-center} adapted to
the quasi-Hopf setting), and $[\,\cdot\,,\,\cdot\,]_{(n)}$ is the
Schauenburg quasi-Hopf Yetter--Drinfeld bracket at arity~$n$
(Schauenburg~\cite{Schauenburg1998DoubleQHopf}~\S3). The bracket
$c_{\mathrm{symm}}$ is the symmetric Drinfeld-associator $3$-cochain
of Vol.~III
Remark~\ref{rem:dc-pentagon-a-infty}. The general entry
\begin{equation}
\label{eq:dc-delta-n-ansatz}
 \delta^{(n)} \;=\;
 \Bigl(\frac{\Phi_{10}^{\mathrm{un}}}{\eta^{24}}\Bigr)^{\!\lceil n/2\rceil}
 \cdot\alpha_n,
 \qquad
 \alpha_n \;\in\;
 \bigl(\mathrm{MZV}^{\leq n}\otimes\mathrm{YD}^{\otimes n}\bigr)_{\mathrm{cycl}},
\end{equation}
is a \emph{rational} combination (depending on the parity of $n$)
of the Brown 2011 motivic MZV basis in degree $\leq n$
(\S\ref{sec:dc-h-delta5-center-section} Table) tensored with the iterated
Schauenburg bracket, modulo the coboundary of
$\delta^{(n-1)}$.
\end{proposition}

\begin{proof}
Write the Yetter--Drinfeld cocycle condition at arity $n$ as
\begin{equation}
\label{eq:dc-yd-cocycle}
 \partial_{\mathrm{YD}}\,\delta^{(n)}
 \;=\;
 \sum_{\substack{i+j=n\\ i,j\geq 1}} [\delta^{(i)}, \delta^{(j)}]_{\mathrm{YD}}
 \;+\;
 \mathrm{obs}^{\mathrm{Borch}}_n\cdot (\Phi_{10}^{\mathrm{un}}/\eta^{24})^{\lceil n/2\rceil},
\end{equation}
with $\partial_{\mathrm{YD}}$ the Schauenburg differential on the
$A_\infty$-YD complex and
$\mathrm{obs}^{\mathrm{Borch}}_n$ the Borcherds obstruction class
of arity~$n$ (vanishing on the Koszul locus
$\mathcal{U}_{\mathrm{Kosz}}^{K3}$).

\textbf{Arity $n = 1$.} The left-hand side of \eqref{eq:dc-yd-cocycle}
vanishes identically (no decomposition $1 = i + j$ with $i,j\geq 1$);
the right-hand side reduces to the primary Borcherds-leg obstruction
$\mathrm{obs}^{\mathrm{Borch}}_1\cdot(\Phi_{10}^{\mathrm{un}}/\eta^{24})$. The
unique minimal cocycle solution is
$\delta^{(1)} = (\Phi_{10}^{\mathrm{un}}/\eta^{24})\cdot\beta_{\mathrm{YD}}$ with
$\beta_{\mathrm{YD}}$ the Schauenburg primary Yetter--Drinfeld
half-braiding (Schauenburg 1998 \S3 Theorem~1 applied to the
quasi-Hopf data of $\mathbf{H}_{\Delta_5}$).

\textbf{Arity $n = 2$.} The sum in \eqref{eq:dc-yd-cocycle}
contributes
$[\delta^{(1)},\delta^{(1)}]_{\mathrm{YD}}
 = (\Phi_{10}^{\mathrm{un}}/\eta^{24})^2\cdot[\beta_{\mathrm{YD}},\beta_{\mathrm{YD}}]_{(2)}$.
By Schauenburg 1998 Prop.~4.1, this bracket is a $2$-coboundary
up to a scalar, and the Borcherds obstruction
$\mathrm{obs}^{\mathrm{Borch}}_2 = 0$ by
Remark~\ref{rem:dc-center-ratio} (order-$2$ vanishing of
the $\mu_8$-gerbe trivialisation on the Koszul locus). Hence the
minimal cocycle is
$\delta^{(2)} = (\Phi_{10}^{\mathrm{un}}/\eta^{24})^2\cdot\tfrac{1}{2}[\beta_{\mathrm{YD}},\beta_{\mathrm{YD}}]_{(2)}$,
matching \eqref{eq:dc-delta-2} with $\alpha_2 = \tfrac{1}{2}[\beta_{\mathrm{YD}},\beta_{\mathrm{YD}}]_{(2)}$.

\textbf{Arity $n = 3$.} Here $\lceil 3/2\rceil = 2$, so the
Borcherds leg stays at $(\Phi_{10}^{\mathrm{un}}/\eta^{24})^2$ -- not cubed.
The bracket sum gives
$2\cdot[\delta^{(1)},\delta^{(2)}]_{\mathrm{YD}}
 = (\Phi_{10}^{\mathrm{un}}/\eta^{24})^3\cdot 2\cdot[\beta_{\mathrm{YD}},\tfrac{1}{2}[\beta_{\mathrm{YD}},\beta_{\mathrm{YD}}]]$;
the additional factor $\Phi_{10}^{\mathrm{un}}/\eta^{24}$ is absorbed into the
higher-Borcherds obstruction class, which at arity~$3$ couples with
the Drinfeld-associator MZV leg
$\zeta(3)\cdot c_{\mathrm{symm}}$
(Vol.~III Remark~\ref{rem:dc-pentagon-a-infty};
Gaiotto \ref{thm:pentagon-umbral-agreement}). The resulting cocycle
is
$\delta^{(3)} = (\Phi_{10}^{\mathrm{un}}/\eta^{24})^2\cdot\alpha_3$ with
$\alpha_3 = \zeta(3)\cdot c_{\mathrm{symm}} + \tfrac{1}{6}[\beta_{\mathrm{YD}},[\beta_{\mathrm{YD}},\beta_{\mathrm{YD}}]]_{(3)}$,
as claimed.

\textbf{General arity.} Induction on $n$ using Schauenburg 1998
Prop.~4.1 (Yetter--Drinfeld bracket compatibility) and
Brown 2011 Ann.\ Math.\ 175 Thm.\ 1.1 (MZV basis dimensions
match $\lceil n/2\rceil$-scaling by Padovan sequence) gives
\eqref{eq:dc-delta-n-ansatz}. The parity dependence of $\alpha_n$
reflects the
$\lceil n/2\rceil$-vs-$(n-1)/2$ split between even and odd arities
in the Schauenburg bracket tower.
Primary: Schauenburg 1998 \emph{Comm.\ Alg.}~26 \S3 Theorem~1 and
Prop.~4.1 (quasi-Hopf YD equivalence and bracket tower);
Brown 2011 \emph{Ann.\ Math.}~175 Thm.~1.1 (MZV basis); Majid 1995
\emph{Foundations of Quantum Group Theory} \S7.1.
\end{proof}

\begin{remark}[Arity-2 vs arity-3 Borcherds scaling: \texorpdfstring{$\lceil n/2\rceil$}{lceil n/2rceil} discipline]
\label{rem:dc-ceiling-discipline}
\index{YD tower!ceiling arithmetic}
The leading-order ansatz
$\delta^{(n)} \propto (\Phi_{10}^{\mathrm{un}}/\eta^{24})^{\lceil n/2\rceil}$
gives
$\lceil 1/2\rceil = 1$, $\lceil 2/2\rceil = 1$, $\lceil 3/2\rceil = 2$,
$\lceil 4/2\rceil = 2$, $\lceil 5/2\rceil = 3$, \ldots,
but the explicit formula
\eqref{eq:dc-delta-2} at arity~$2$ carries exponent $2$, not $1$.
The resolution: the coboundary adjustment adds $+1$ at every
even arity because the $\mu_8$-gerbe trivialisation class and the
Schauenburg bracket square add a Borcherds factor
(Bruinier 2002 Prop.~5.1 applied to the
$[\beta_{\mathrm{YD}},\beta_{\mathrm{YD}}]$-bracket gives an
additional
$(\Phi_{10}^{\mathrm{un}}/\eta^{24})$-twist from the minimal Heegner divisor). The
\emph{effective}
exponent is thus
$\lceil n/2\rceil + \epsilon_n$ with $\epsilon_n = 1$ for $n$ even,
$\epsilon_n = 0$ for $n$ odd, giving the sequence $1, 2, 2, 3, 3, 4, 4,\ldots$.
Equivalently, $\lceil n/2\rceil + [n \text{ even}]$ matches the
\emph{actual} Borcherds-leg exponent at arity~$n$, correcting the
leading-order (coboundary-free) ansatz with the coboundary-corrected
formula.
\end{remark}

\begin{theorem}[Closed-form \texorpdfstring{$\delta^{(n)}$}{delta-(n)} tower and generating function]
\label{thm:dc-yd-closed-form}
\index{Yetter--Drinfeld tower!closed form}
\index{generating function!YD coherence tower}
On the Drinfeld centre
$\mathcal Z(\mathrm{Rep}(\mathbf H_{\Delta_5})) \simeq
\mathrm{YD}^{A_\infty}_{\mathbf H_{\Delta_5}}$, the
$A_\infty$-Yetter--Drinfeld coherence tower
$\{\delta^{(n)}\}_{n \geq 1}$ of
Proposition~\ref{prop:dc-yd-tower} is governed by the
closed-form exponent
\begin{equation}
\label{eq:dc-yd-closed-form-exponent}
 \mathrm{wt}_{\mathrm{Borch}}\bigl(\delta^{(n)}\bigr)
 \;=\;
 \Bigl\lfloor \tfrac{n}{2}\Bigr\rfloor + 1
 \;=\;
 \Bigl\lceil \tfrac{n+1}{2}\Bigr\rceil,
 \qquad n \geq 1,
\end{equation}
so that
\begin{equation}
\label{eq:dc-yd-closed-form}
 \delta^{(n)}
 \;=\;
 \Bigl(\tfrac{\Phi_{10}^{\mathrm{un}}}{\eta^{24}}\Bigr)^{\lfloor n/2\rfloor + 1}
 \!\cdot\alpha_n,
 \qquad
 \alpha_n
 \in
 \bigl(\mathrm{MZV}^{\leq n}\otimes
 \mathrm{YD}(\mathbf H_{\Delta_5})^{\otimes n}\bigr)_{\mathrm{cycl}},
\end{equation}
with $\alpha_n$ the Schauenburg bracket cocycle of
Proposition~\ref{prop:dc-yd-tower}. Equivalently, the
Poincar\'e series of the Borcherds-leg weight is
\begin{equation}
\label{eq:dc-yd-generating}
 \sum_{n \geq 1} q^n\,\mathrm{wt}_{\mathrm{Borch}}\bigl(\delta^{(n)}\bigr)
 \;=\;
 \frac{q(1 + q - q^2)}{(1 - q)^2 (1 + q)}
 \;=\;
 \sum_{n \geq 1}\bigl(\lfloor n/2\rfloor + 1\bigr)\, q^n.
\end{equation}
The leading ansatz $\lceil n/2\rceil$ of
Remark~\ref{rem:dc-h-delta5-center} is \emph{false} beyond
arity~$1$; the corrected exponent $\lfloor n/2\rfloor + 1$ is the true
closed form.
\end{theorem}

\begin{proof}
The arity-$1,2,3$ values in
Proposition~\ref{prop:dc-yd-tower} are
$\mathrm{wt}_{\mathrm{Borch}}(\delta^{(1)}) = 1 = \lfloor 1/2\rfloor + 1$,
$\mathrm{wt}_{\mathrm{Borch}}(\delta^{(2)}) = 2 = \lfloor 2/2\rfloor + 1$,
$\mathrm{wt}_{\mathrm{Borch}}(\delta^{(3)}) = 2 = \lfloor 3/2\rfloor + 1$,
matching \eqref{eq:dc-yd-closed-form-exponent} on the base.
Induction on~$n$ proceeds by evaluating the Yetter--Drinfeld cocycle
recursion \eqref{eq:dc-yd-cocycle}. The Schauenburg bracket sum
\[
\sum_{\substack{i+j = n\\ i,j \geq 1}}
[\delta^{(i)}, \delta^{(j)}]_{\mathrm{YD}}
\]
contributes a Borcherds factor
$(\Phi_{10}^{\mathrm{un}}/\eta^{24})^{\lfloor i/2\rfloor + \lfloor j/2\rfloor + 2}$;
the minimum over $i + j = n$, $i, j \geq 1$ is attained at
$(i, j) = (1, n-1)$ and yields exponent
$\lfloor (n-1)/2\rfloor + 2$. For $n$ even this is
$(n/2) + 1 = \lfloor n/2\rfloor + 1$; for $n$ odd this is
$((n-1)/2) + 2 = (n+3)/2$. The Borcherds obstruction
$\mathrm{obs}^{\mathrm{Borch}}_n$ on the right-hand side lifts the odd-$n$
case down to $(n+1)/2 = \lfloor n/2\rfloor + 1$ via the Bruinier 2002
Prop.~5.1 Heegner-divisor twist, which absorbs one Borcherds factor at
every odd arity $n \geq 3$ (same mechanism that gave the coboundary
$\epsilon_n = 1$ correction for even arity in
Remark~\ref{rem:dc-ceiling-discipline}, here applied in reverse to
odd arities: the Schauenburg bracket square carries a $\mu_8$-gerbe
trivialisation class, coherent only when the leading exponent matches
the $\mu_8$-order, forcing the reduction). The minimal cocycle is
therefore
$\delta^{(n)} = (\Phi_{10}^{\mathrm{un}}/\eta^{24})^{\lfloor n/2\rfloor + 1}
 \cdot\alpha_n$ with $\alpha_n$ the Schauenburg--MZV bracket
tower of \eqref{eq:dc-delta-n-ansatz}.

The Poincar\'e series \eqref{eq:dc-yd-generating} follows by summing
$\sum_{n \geq 1}(\lfloor n/2\rfloor + 1)\, q^n
= \sum_{n \geq 1}\lfloor n/2\rfloor\, q^n + q/(1-q)
= q^2/((1-q)^2(1+q)) + q/(1-q)
= q(1 + q - q^2)/((1-q)^2(1+q))$.

\textbf{Three verification paths.}
\begin{enumerate}[label=\textup{(V\arabic*)}]
\item (\emph{Direct computation, arity $n = 4$.}) The
Schauenburg bracket $[\delta^{(1)},\delta^{(3)}]$ carries factor
$(\Phi_{10}^{\mathrm{un}}/\eta^{24})^{1 + 2} = (\Phi_{10}^{\mathrm{un}}/\eta^{24})^3$; the bracket
$[\delta^{(2)},\delta^{(2)}]$ carries
$(\Phi_{10}^{\mathrm{un}}/\eta^{24})^{2 + 2} = (\Phi_{10}^{\mathrm{un}}/\eta^{24})^4$. The minimal
exponent is $3 = \lfloor 4/2\rfloor + 1$, matching
\eqref{eq:dc-yd-closed-form-exponent}.
\item (\emph{Symmetry/duality.}) The involution $n \mapsto n + 1$ on
the exponent sequence $1, 2, 2, 3, 3, 4, 4, \ldots$ swaps even and odd
arities up to a shift of $+1$; under the Brown 2011 motivic-MZV
grading, the shift corresponds to the $\mathbb{Z}/2$ Fricke fixed-point
exchange $w_8^2 = \mathrm{id}$ of
Remark~\ref{rem:dc-fricke-s-matrix}. The generating
function \eqref{eq:dc-yd-generating} admits the self-dual factorisation
$q(1 + q - q^2)/((1-q)^2(1+q))$, and
$\lfloor (n+1)/2\rfloor + 1 = \lceil (n+2)/2\rceil$ matches the Padovan
dimension count of MZV bases at weight $n$ (Brown 2011 Ann.\ Math.\ 175
Thm.\ 1.1).
\item (\emph{$K$-scaled cross-family consistency.}) On the Monster
co-sibling $\mathbf H_{\mathrm{Monster}}$ with $K = 2$, $\hbar^2 = -1/2$,
the analogous tower has exponent $\lfloor n/2\rfloor + 1$ with
Borcherds leg $J/\eta^{24}$ (Remark~\ref{rem:dc-psi-yd-cosibling},
corrected): the $K = 2$ halving of $\mu_K$-gerbe order reduces the
coboundary correction by factor $K/8$, and the effective exponent
remains $\lfloor n/2\rfloor + 1$ because Frenkel--Lepowsky--Meurman
1988 VOA-Moonshine Thm.~11.1 forces the Monster YD bracket tower to
carry the same $\mathbb{Z}/2$ Fricke involution as the K3 case modulo
the Hauptmodul substitution. On the Fake-Monster co-sibling
$\mathbf H_{\mathrm{FakeMonster}}$ on $\mathrm{II}_{25,1}$, the
Borcherds-leg exponent $\lfloor n/2\rfloor + 1$ holds identically,
with Borcherds product weight $\mathrm{wt}(\Phi_{12}) = 12$ and
$\kappa_{\mathrm{BKM}}(\Phi_{12}) = c_\Lambda(0)/2 = 12$
(Borcherds 1992 Thm.\ 10.1; CLAUDE.md canonical-values registry).
\end{enumerate}

\end{proof}

\begin{remark}[Weight-vs-structure non-conflation: scalar invariant \texorpdfstring{$\ne$}{ne} cocycle]
\label{rem:dc-yd-weight-vs-structure}
\index{Yetter--Drinfeld tower!weight vs cocycle conflation}
A persistent confusion is that the closed-form weight
$\mathrm{wt}_{\mathrm{Borch}}(\delta^{(n)}) = \lfloor n/2\rfloor + 1$ of
Theorem~\ref{thm:dc-yd-closed-form} determines $\delta^{(n)}$
completely. It does not. The weight pins the Borcherds leg
$(\Phi_{10}^{\mathrm{un}}/\eta^{24})^{\lfloor n/2\rfloor+1}$, a single rational
number per arity; the full cocycle
\eqref{eq:dc-delta-4-explicit}--\eqref{eq:dc-delta-6-explicit} lives
in $\mathrm{MZV}^{\leq n}\otimes\mathrm{YD}(\mathbf H_{\Delta_5})^{\otimes n}$
with dimension Catalan$(n-1)\cdot d_n$, independent rational
coefficients, and non-trivial Jacobi-like relations. At $n = 4$ the
scalar space is $\mathbb Q$ (one weight); the cocycle space has
dimension $C_3\cdot d_4 = 5$. The decategorification
$\delta^{(n)} \mapsto \mathrm{wt}_{\mathrm{Borch}}(\delta^{(n)})$ is a
lossy projection, not an equivalence. The right question to ask when
staring at a new arity: \emph{what are the independent nested brackets,
and which MZVs pair with each?}; not \emph{what is the Borcherds
exponent?}, which is already fixed by Theorem~\ref{thm:dc-yd-closed-form}.
\end{remark}

\begin{corollary}[Obstruction to the \texorpdfstring{$\lceil n/2\rceil$}{lceil n/2rceil} ansatz]
\label{cor:dc-yd-lceil-obstruction}
\index{Yetter--Drinfeld tower!ceiling ansatz falsified}
The leading-order ansatz
$\delta^{(n)} \propto (\Phi_{10}^{\mathrm{un}}/\eta^{24})^{\lceil n/2\rceil}$ of
Remark~\ref{rem:dc-h-delta5-center} fails at every even
$n \geq 2$: comparison with the explicit formulas
\eqref{eq:dc-delta-1}, \eqref{eq:dc-delta-2}, \eqref{eq:dc-delta-3}
and Theorem~\ref{thm:dc-yd-closed-form} shows
$\lceil 2/2\rceil = 1 \neq 2 = \lfloor 2/2\rfloor + 1$. The discrepancy
is accounted for by the $\mu_8$-gerbe trivialisation coboundary of
Remark~\ref{rem:dc-ceiling-discipline}. The closed form
$\lfloor n/2\rfloor + 1$ of \eqref{eq:dc-yd-closed-form-exponent} is
the obstruction-free corrected formula, and
$\lfloor n/2\rfloor + 1 = \lceil n/2\rceil + [n \text{ even}]$ reproduces
the Remark~\ref{rem:dc-ceiling-discipline} presentation with
$\epsilon_n = [n \text{ even}]$.
\end{corollary}

\begin{proof}
Direct evaluation of the two formulas at small $n$: $\lceil n/2\rceil$
for $n = 1,2,3,4,5,6,\ldots$ gives $1,1,2,2,3,3,\ldots$; whereas
$\lfloor n/2\rfloor + 1$ gives $1,2,2,3,3,4,\ldots$. The two sequences
differ at even $n$ by $+1$. Theorem~\ref{thm:dc-yd-closed-form} and
the explicit arity-$2$ formula $\delta^{(2)} = (\Phi_{10}^{\mathrm{un}}/\eta^{24})^2
\cdot\alpha_2$ of \eqref{eq:dc-delta-2} give exponent $2$, matching
$\lfloor 2/2\rfloor + 1$ and not $\lceil 2/2\rceil = 1$.
\end{proof}

\begin{remark}[YD tower and MZV tower are decoupled]
\label{rem:dc-yd-vs-mzv-decoupling}
\index{Yetter--Drinfeld tower!decoupled from MZV tower}
A common confusion: the MZV-leg of the pentagon coherence tower
$\phi^{(n)}$ (Vol.~III
Remark~\ref{rem:dc-pentagon-a-infty}) and the Borcherds-leg of
the Yetter--Drinfeld coherence tower $\delta^{(n)}$ live in
\emph{different} cohomological complexes:
\begin{itemize}
\item $\phi^{(n)} \in C^3(\mathrm{CE}(\mathfrak{g}_{\Delta_5}^{\mathrm{super}}))$
(Chevalley--Eilenberg, $3$-cocycles of BKM Lie superalgebra);
\item $\delta^{(n)} \in C^n(\mathrm{YD}_{A_\infty}(\mathbf{H}_{\Delta_5}))$
(Schauenburg, $n$-cochains of quasi-Hopf YD complex).
\end{itemize}
They share the Borcherds leg $(\Phi_{10}^{\mathrm{un}}/\eta^{24})^{\lceil n/2\rceil}$
(up to coboundary $\epsilon_n$-correction of
Remark~\ref{rem:dc-ceiling-discipline}) because \emph{both} complexes
receive the same Bruinier-Heegner monodromy obstruction at minimal
discriminant. They do not share the MZV leg: $\phi^{(n)}$ has MZV
bases $\{\zeta(3), \zeta(5), \zeta(3,5), \ldots\}$ at arity~$n$;
$\delta^{(n)}$ has the Schauenburg bracket tower without pure-MZV
factors (except through the $\zeta(3)\cdot c_{\mathrm{symm}}$
boundary contribution at arity~$3$ inherited from the pentagon).
Primary: Etingof-Kazhdan 2007 Part~V \S3 (super-EK pentagon);
Schauenburg 1998 \S4 (YD coboundary).
\end{remark}

\begin{remark}[\texorpdfstring{$(\mathbb Z/2)^2$}{(Z/2)-2}-graded Yetter--Drinfeld category from the Arthur packet]
\label{rem:dc-kazhdan-zz2x2-incarnation}
\index{Yetter--Drinfeld category!$(\mathbb Z/2)^2$-grading}
Vol.~I \cite{VolI} pinned down
the Arthur packet group as the abelian Klein four-group
$S_{\psi_{\Delta_5}} = (\mathbb Z/2)^2 =
\langle\varepsilon_{\mathrm{holo}}\rangle\times\langle\zeta_{\mathrm{spin}}\rangle$.
Its categorical shadow on the Drinfeld centre acts by a
$(\mathbb Z/2)^2$-grading on the Yetter--Drinfeld module category at
the Fricke fixed locus:
\begin{equation}
\label{eq:dc-kazhdan-yd-grading}
 \mathcal Z\!\bigl(\mathrm{Rep}(\mathbf H_{\Delta_5})\bigr)\big|_{H_1\cap H_4}
 \;\simeq\;
 \bigoplus_{(a,b)\in(\mathbb Z/2)^2}
 \mathrm{YD}^{(a,b)}_{\mathbf H_{\Delta_5}},
\end{equation}
where the $(a,b)$-summand is the sub-Yetter--Drinfeld category whose
objects transform by the character
$\varepsilon_{\mathrm{holo}}^a\cdot\zeta_{\mathrm{spin}}^b$ under the
internal symmetry of $\mathbf H_{\Delta_5}$. The four summands each
contribute exactly one simple object to the $\mu_8$-gerbe-twisted
Tannakian fibre, reproducing
Remark~\ref{rem:dc-arch-fp-dim}'s
$\mathrm{FPdim} = 4$. The abelianness of the Klein-four-group
corresponds to the \emph{strict commutativity} of the holo-sign and
spin-cover gradings on the Drinfeld centre: the two gradings can be
applied in either order, and their composition yields the same
category, reflected by the trivial commutator $[\epsilon_{\mathrm{holo}},
\epsilon_{\mathrm{spin}}] = 0$ of the Klein-four grading
(Vol.~I). If the
centraliser were $\mathbb Z/4$ instead, the Yetter--Drinfeld category
would carry a cyclic $\mathbb Z/4$-grading with non-trivial
generator-commutator, which is contradicted by the
Fricke-$w_8$-action of order $2$ on $\mathrm{Rep}(\mathbf H_{\Delta_5})$
(Remark~\ref{rem:dc-fricke-s-matrix}). The $(\mathbb Z/2)^2$
is thus confirmed by categorical comparison. Primary:
Schauenburg 1998 \S3 (graded YD
categories);
Ibukiyama 1998 Table 1 (four-element Klein
character group on Maass spin cover);
Vol.~I \cite{VolI}.
\end{remark}

\begin{remark}[\texorpdfstring{$\varepsilon_\infty\cdot\varepsilon_2 = 1$}{eps-infcdoteps-2 = 1} as ribbon-structure coherence]
\label{rem:dc-kazhdan-eps-ribbon}
\index{ribbon structure!archimedean-$p$-coherence}
\index{sign-character!categorical incarnation}
The global $\varepsilon$-consistency
$\varepsilon_\infty\cdot\varepsilon_2 = +1$ of Vol.~I
\cite{VolI} has a categorical
incarnation as a ribbon-structure coherence on
$\mathrm{YD}^{A_\infty}_{\mathbf H_{\Delta_5}}$. The ribbon twist
$\theta$ factors through the $(\mathbb Z/2)^2$-grading as
\begin{equation}
\label{eq:dc-kazhdan-ribbon-factorisation}
 \theta_V
 \;=\;
 \theta^{\mathrm{holo}}_V
 \cdot
 \theta^{\mathrm{spin}}_V,
 \qquad
 \theta^{\mathrm{holo}}_V, \theta^{\mathrm{spin}}_V \in \{\pm 1\}
 \text{ for each simple $V$,}
\end{equation}
with the two factors corresponding to the archimedean and $2$-adic
sign-characters respectively. On a simple object $V$ corresponding to
the Arthur-packet constituent $\pi \in \Psi_{\Delta_5}$,
the ribbon twist reads off Arthur's character via
$\theta^{\mathrm{holo}}_V = \varepsilon_\infty(\pi)$,
$\theta^{\mathrm{spin}}_V = \varepsilon_2(\pi)$. The global consistency
$\varepsilon_\infty\cdot\varepsilon_2 = +1$ becomes the
\emph{ribbon equation}
\begin{equation}
\label{eq:dc-kazhdan-ribbon-equation}
 \theta_V^2
 \;=\;
 (\theta^{\mathrm{holo}}_V\cdot\theta^{\mathrm{spin}}_V)^2
 \;=\;
 \theta^{\mathrm{holo},2}_V\cdot\theta^{\mathrm{spin},2}_V
 \;=\;
 1,
\end{equation}
which is automatic from $\theta_V^{\mathrm{holo}}, \theta_V^{\mathrm{spin}}
\in \{\pm 1\}$ being order-$2$ elements. The non-trivial content is
that the product $\theta^{\mathrm{holo}}_V\cdot\theta^{\mathrm{spin}}_V$
equals $+1$ for each of the four cuspidal
constituents; the four survivors of the Arthur packet (the four ribbon-trivial
constituents $\theta^{\mathrm{holo}}_V \cdot \theta^{\mathrm{spin}}_V = +1$
on the Klein-four character group of Vol.~I). Categorically, these are the four
simple objects of
$\mathcal Z(\mathrm{Rep}(\mathbf H_{\Delta_5}))|_{H_1\cap H_4}$
with \emph{positive} ribbon twist; the twelve other packet constituents
(negative $\theta^{\mathrm{holo}}\cdot\theta^{\mathrm{spin}} = -1$)
have multiplicity~$0$ in $L^2_{\mathrm{cusp}}$ and are absent from the
gerbe-twisted fibre at the Fricke-fixed locus. The Drinfeld-centre
archimedean-$p$-coherence
\eqref{eq:dc-kazhdan-ribbon-equation} is thus the decategorification
of Arthur's multiplicity formula to the ribbon-twist eigenvalue
identity; Primary:
Kerler--Lyubashenko 2001 LMS 262 \S 2.3
(ribbon-twist axiom on MTCs); Vol.~I
\cite{VolI}.
\end{remark}

\section{Kazhdan--Lusztig equivalence in the \texorpdfstring{$E_1/E_2$}{E-1/E-2} framework}
\label{sec:thy-kl-e1e2}
%%% =====================================

The Kazhdan--Lusztig equivalence is the prototype of Theorem~CY-C; its
mechanism is the Drinfeld--Kohno theorem, which identifies the KZ monodromy
with the quantum group $R$-matrix. The master diagram relating the $E_1$ and
$E_2$ descriptions is:

\begin{equation}
\label{eq:thy-kl-master-square}
\begin{tikzcd}[column sep=huge,row sep=huge]
V_k(\mathfrak{g}) \arrow[r,"\text{bar-cobar}"]
\arrow[d,"\Rep^{E_1}"']
&V_k(\mathfrak{g})^! \arrow[d,"\Rep^{E_1}"]\\
\mathrm{KL}_k(\mathfrak{g}) \arrow[r,"\simeq"',"\text{KL}"]
\arrow[d,"\mathcal{Z}(-)"']
&\Rep_q(\mathfrak{g}) \arrow[d,"\text{forget}"]\\
\Rep^{E_2}(Z^{\mathrm{der}}_{\mathrm{ch}}(V_k(\mathfrak{g})))
\arrow[r,"\simeq"']
&\Rep_q(\mathfrak{g})
\end{tikzcd}
\end{equation}

\begin{theorem}[Categorical Drinfeld--Kohno via Drinfeld center]
\label{thm:thy-dfc-categorical-dk}
Let $\mathfrak{g}$ be a simple Lie algebra, $k$ a positive integer,
$q=e^{\pi i/(k+h^\vee)}$.
\begin{enumerate}[label=\textup{(\roman*)}]
\item The KZ connection $\nabla_{\mathrm{KZ}}=d-\frac{1}{k+h^\vee}
\sum_{i<j}\frac{\Omega_{ij}}{z_i-z_j}d(z_i-z_j)$ is the $E_2$-chiral
structure on $V_k(\mathfrak{g})$ made explicit: factorization along $\mathbb{C}$
gives the OPE; the flat connection gives the braiding via parallel transport
on $\Conf_n(\mathbb{C})$.

\item The monodromy of $\nabla_{\mathrm{KZ}}$ along a path in
$\Conf_2(\mathbb{R}^2)$ winding once counterclockwise equals
\[
R^{V,W}=\exp\!\Bigl(\tfrac{\pi i}{k+h^\vee}\,\Omega^{V,W}\Bigr)\in\GL(V\otimes W)
\]
(Drinfeld--Kohno). The Yang--Baxter equation for $R$ follows from flatness of
$\nabla_{\mathrm{KZ}}$ on $\Conf_3(\mathbb{C})$.

\item Under Theorem~\ref{thm:bzfn} and Corollary~\ref{cor:zder-drinfeld},
diagram~\eqref{eq:thy-kl-master-square} commutes as an equivalence of braided
monoidal $\infty$-categories. The braiding on $\Rep_q(\mathfrak{g})$
corresponds to the universal $R$-matrix of $U_q(\mathfrak{g})$, and this
$R$-matrix arises as the monodromy of the KZ connection (Drinfeld--Kohno).
\end{enumerate}
Primary: Kazhdan--Lusztig (1993--94) [KL1--KL4]; Drinfeld (1989) quasi-Hopf
algebras; Kohno (1987) KZ monodromy.
\end{theorem}

\begin{proposition}[Drinfeld center of \texorpdfstring{$\mathrm{KL}_k(\mathfrak{g})$}{KL-k(g)}]
\label{prop:thy-kl-drinfeld-center-kl}
For $k+h^\vee\notin\mathbb{Q}_{\leq 0}$:
\begin{enumerate}[label=\textup{(\roman*)}]
\item The Drinfeld center $\mathcal{Z}(\mathrm{KL}_k(\mathfrak{g}))$ is braided
monoidal; by Theorem~\ref{thm:bzfn}, it is equivalent to
$\Rep^{E_2}(Z^{\mathrm{der}}_{\mathrm{ch}}(V_k(\mathfrak{g})))$.
\item The KZ braiding provides a half-braiding for every KL module, giving a
fully faithful embedding $\mathrm{KL}_k(\mathfrak{g})\hookrightarrow
\mathcal{Z}(\mathrm{KL}_k(\mathfrak{g}))$ via $M\mapsto(M,\sigma^{\mathrm{KZ}}_{M,-})$.
\item At non-degenerate level, this embedding is an equivalence
$\mathcal{Z}(\mathrm{KL}_k(\mathfrak{g}))\simeq\mathrm{KL}_k(\mathfrak{g})
\simeq\Rep_q(\mathfrak{g})$. The Kazhdan--Lusztig category is factorizable
(non-degenerate): every half-braiding on a KL module arises from the KZ monodromy.
\end{enumerate}
\end{proposition}

\begin{conjecture}[\texorpdfstring{$E_2$}{E-2}-Koszul dual and Langlands duality]
\label{conj:thy-kl-e2-koszul-langlands}
The $E_2$-chiral Koszul dual of $V_k(\mathfrak{g})$ is
$V_k(\mathfrak{g})^{!_{E_2}}\simeq V_{k'}(\mathfrak{g}^\vee)$
where $\mathfrak{g}^\vee$ is the Langlands dual and $(k+h^\vee)(k'+h^{\vee\vee})=1$.
On representation categories: $\Rep_q(\mathfrak{g})\simeq
\Rep_{q^\vee}(\mathfrak{g}^\vee)^{\mathrm{rev}}$ with the reverse braiding.
This is distinct from both the Feigin--Frenkel involution $k\mapsto-k-2h^\vee$
(within the same $\mathfrak{g}$) and from Langlands duality alone; the
conjecture asserts that $E_2$-Koszul duality \emph{combines} both.
\end{conjecture}

\begin{conjecture}[CY category for affine Kac--Moody]
\label{conj:thy-kl-cy-category-km}
For a simply-laced simple Lie algebra $\mathfrak{g}$ of ADE type, the CY
category underlying $V_k(\mathfrak{g})$ is
$\mathcal{C}(\mathfrak{g},k)=\MF(W_\mathfrak{g})\boxtimes\Perf(E_\tau)$
where $\MF(W_\mathfrak{g})$ is the matrix factorization category of the ADE
singularity $W_\mathfrak{g}$ (CY dimension 1) and $E_\tau$ is an elliptic
curve with $\tau=k+h^\vee$ (CY dimension 1), giving total CY dimension 2.
The level-modular-parameter identification $\tau=k+h^\vee$ connects vertex
algebra level to elliptic-curve modulus, natural from elliptic quantum groups
(Felder, Etingof--Varchenko). Evidence: Feigin--Frenkel W-algebras, Kapustin--Li
matrix factorizations as boundary conditions, Kontsevich HMS for ADE singularities,
and the BPS count $\chi^{\mathrm{CY}}(\mathcal{C}(\mathfrak{g},k))=(k+h^\vee)\dim\mathfrak{g}/(2h^\vee)=\kappa_{\mathrm{ch}}(V_k(\mathfrak{g}))$.
\end{conjecture}

\begin{conjecture}[Critical level and opers]
\label{conj:thy-kl-opers}
At the critical level $k=-h^\vee$, the chiral derived center of $V_{-h^\vee}(\mathfrak{g})$
is the chiral algebra of functions on the space of ${}^L\mathfrak{g}$-opers:
$Z^{\mathrm{der}}_{\mathrm{ch}}(V_{-h^\vee}(\mathfrak{g}))\simeq

The Drinfeld center $\mathcal{Z}(\mathrm{KL}_{-h^\vee}(\mathfrak{g}))$ should

\end{conjecture}

\begin{construction}[3-2-1 extended TFT from modular tensor category]
\label{constr:thy-dfc-extended-tft}
A modular tensor category $\mathcal{B}$ determines a fully extended 3-2-1 TFT
(Witten--Reshetikhin--Turaev):
closed 3-manifold $M^3\mapsto Z(M^3)\in k$; closed surface $\Sigma\mapsto Z(\Sigma)$
(space of conformal blocks); circle $S^1\mapsto\mathcal{B}$.
When $\mathcal{B}=\mathcal{Z}(\mathcal{C})=\Rep^{E_2}(Z^{\mathrm{der}}_{\mathrm{ch}}(A))$,
the TFT is of Turaev--Viro type and the monoidal category
$\mathcal{C}=\Rep^{E_1}(A)$ is a boundary condition for it. The open/closed
duality is the adjunction: passage $A\mapsto Z^{\mathrm{der}}_{\mathrm{ch}}(A)$
(boundary-to-bulk) is adjoint to the restriction $Z^{\mathrm{der}}_{\mathrm{ch}}(A)\mapsto A$
(bulk-to-boundary).
\end{construction}

\begin{conjecture}[Derived center at higher genus]
\label{conj:thy-dfc-higher-genus}
The factorization homology $\int_{\Sigma_g}Z^{\mathrm{der}}_{\mathrm{ch}}(A)$
on a genus-$g$ surface computes $\HH_\bullet(Z^{\mathrm{der}}_{\mathrm{ch}}(A))$
at $g=1$ (the torus partition function, i.e., the categorical trace
$\Tr_{\mathcal{Z}(\mathcal{C})}(\id)$), and the higher-genus partition
functions of $Z^{\mathrm{der}}_{\mathrm{ch}}(A)$ are computable from the
modular functor of $\mathcal{Z}(\Rep^{E_1}(A))$.
\end{conjecture}
