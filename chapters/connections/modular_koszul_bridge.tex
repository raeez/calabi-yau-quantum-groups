\chapter{Modular Koszul Duality and CY Geometry}
\label{ch:modular-koszul-bridge}

This chapter bridges the CY-geometric world of Volume~III to the modular-algebraic world of Volume~I.  The central question: given a CY category $\cC$, what do the five theorems of Volume~I (bar-cobar inversion, Verdier intertwining, complementarity, the genus tower, and the holographic datum) say about the geometry of $\cC$?

A CY category $\cC$ produces, via the CY-to-chiral correspondence programme $\{\Phi_d\}$ of Chapter~\ref{ch:cy-to-chiral}, a chiral algebra $A_\cC$; the bar complex $B(A_\cC) = T^c(s^{-1}\overline{A_\cC})$, built on the augmentation ideal $\overline{A_\cC} = \ker(\varepsilon)$, is a factorization coalgebra on $\Ran(C)$. Three Volume~I structures act on $B(A_\cC)$:
\begin{itemize}
 \item \emph{Verdier intertwining} (Theorem~A): $D_{\Ran}(B(A)) \simeq B(A^!)$ is Koszul duality, not bar-cobar inversion, and not the chiral derived center.
 \item \emph{Complementarity} (Theorem~C): splits the genus-$g$ shadow complex into Verdier eigenspaces and, on the uniform-weight lane, equates the scalar sum of Koszul-dual modular characteristics to a family-dependent Koszul conductor.
 \item \emph{The genus tower} (Theorem~D): identifies $\mathrm{obs}_g$ with $\kappa_{\mathrm{ch}} \cdot \lambda_g$ on the uniform-weight lane at genus $1$ unconditionally, with a cross-channel correction $\delta F_g^{\mathrm{cross}}$ at $g \geq 2$ for multi-weight algebras.
\end{itemize}

Transporting these structures to the CY side reveals three deficiencies.  First, the convolution dg Lie algebra living on $\overline{\cM}_{g,n}$ has no existing CY-side habitat. Second, the Vol~I scalar complementarity (Vol~I Theorem~C$_2$, with its family-dependent Koszul conductor; see Remark~\ref{rem:cy-complementarity-kappa-zero} below) has no CY translation stating which Koszul conductor $K_X$ applies at $d \in \{2, 3\}$. Third, the Vol~I CohFT promotion (Theorem~D$+$H) has no CY restatement tracking the flat identity axiom through $\Phi$. Five sections address these deficiencies: \S\ref{sec:modular-conv-cy} builds the CY modular convolution algebra; \S\ref{sec:cy-complementarity-bridge} transports complementarity with explicit (C1) versus (C2) scoping and explicit $d = 2$ versus $d = 3$ conditionality; \S\ref{sec:cy-shadow-cohft} upgrades the shadow tower to a CohFT on $\overline{\cM}_{g,n}$ and records how the Borcherds lift converts the $K3 \times E$ tower into the genus-$2$ Igusa cusp form $\Phi_{10}$; \S\ref{sec:hochschild-bridge} establishes the bridge between the three Hochschild theories (categorical, chiral, derived-center) through $\Phi$; and \S\ref{sec:cy-bridge-examples} collects the principal examples with their $\kappa_\bullet$-spectra.

\section{The modular convolution algebra for CY categories}
\label{sec:modular-conv-cy}

Let $\cC$ be a smooth proper cyclic $A_\infty$-category of CY dimension $d$ and let $A_\cC = \Phi_d(\cC)$ denote the image under the CY-to-chiral correspondence $\Phi_d$ of Chapter~\ref{ch:cy-to-chiral} (Theorem~CY-A$_2$ for $d = 2$; Theorem~CY-A$_3$, Theorem~\ref{thm:cy-to-chiral-d3}, for $d = 3$). The bar coalgebra $B(A_\cC)$ is a factorization coalgebra on $\Ran(C)$ for a fixed smooth projective curve $C$, with bar differential $d_B = d_1 + d_2 + \cdots$ where $d_k$ lowers bar degree by $k - 1$.

\begin{definition}[CY modular convolution algebra]
\label{def:cy-modular-convolution}
The \emph{CY modular convolution algebra} of the pair $(B(A_\cC), A_\cC)$ is the graded vector space
\[
 \mathrm{Conv}_{\mathrm{str}}\bigl(B(A_\cC), A_\cC\bigr)
 \;:=\; \Hom_{\Ran(C)}\!\bigl(B(A_\cC),\, A_\cC \otimes \omega^{\bullet}_{\overline{\cM}_{g,n}}\bigr),
\]
equipped with the convolution bracket induced by the deconcatenation coproduct on $B(A_\cC)$ (dual to $T^c$ on $s^{-1}\overline{A_\cC}$) and the binary product of $A_\cC$, twisted by the modular convolution kernel from $\overline{\cM}_{g,n}$. The differential is $\delta = d_B^\vee + d_{A_\cC} + d_{\overline{\cM}}$, where $d_{\overline{\cM}}$ is the log Fulton--MacPherson ambient boundary differential of Mok~\cite{Mok25}.
\end{definition}

\begin{proposition}[Strict dg Lie structure on $\mathrm{Conv}_{\mathrm{str}}$]
\label{prop:cy-conv-dglie}
\ClaimStatusProvedElsewhere
On the Koszul locus of $A_\cC$, the convolution bracket $[\cdot, \cdot]_{\mathrm{Conv}}$ on $\mathrm{Conv}_{\mathrm{str}}(B(A_\cC), A_\cC)$ satisfies the graded Jacobi identity strictly, and $\delta$ is a derivation of this bracket. The pair $(\mathrm{Conv}_{\mathrm{str}}, \delta, [\cdot, \cdot])$ is a dg Lie algebra; it is a strict model of an $L_\infty$ algebra $\mathrm{Conv}_{\infty}(B(A_\cC), A_\cC)$, and the two share the same Maurer--Cartan moduli space (Vol~I, three-pillar constraint, \S MC3).
\end{proposition}

\begin{proof}[Attribution]
The strict-versus-$L_\infty$ comparison is Vol~I Theorem~MC3 (Robalo--Nick\-erson--Welch\-er 2019 bifunctoriality failure in both slots resolves by treating one slot at a time). The MC moduli coincidence is Vol~I Theorem~MC4. Here we only transport the statement already proved at $d = 2$: $\Phi$ sends cyclic $A_\infty$-objects to chiral algebras compatibly with the bar construction (Theorem~CY-A$_2$(ii)); the $d = 3$ analogue is now established by CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}, proved). The convolution bracket on the CY side is therefore pulled back from the chiral convolution bracket of Vol~I at both $d = 2$ and $d = 3$.
\end{proof}

\begin{remark}[Bifunctoriality warning]
\label{rem:conv-bifunctor-warning}
The convolution $\Hom_\alpha(C, A)$ is \emph{not} a bifunctor in both slots simultaneously (Vol~I, three-pillar constraint~(2); Robalo--Nickerson--Welcher 2019). MC3 holds one slot at a time. In the CY setting this means: varying $\cC$ along a path of CY categories while holding $\Phi(\cC)$ fixed, or varying $A_\cC$ along a chain homotopy while holding $\cC$ fixed, each gives a homotopy equivalence of MC moduli; the simultaneous variation does not (\emph{a priori}) define a coherent bifunctor.
\end{remark}

\begin{theorem}[CY shadow obstruction tower as MC element]
\label{thm:cy-theta-mc}
\ClaimStatusProvedElsewhere
The Vol~I shadow obstruction tower
\[
 \Theta_{A_\cC} \;=\; D_{A_\cC} - d_0 \;\in\; \MC\!\bigl(\mathrm{Conv}_{\mathrm{str}}(B(A_\cC), A_\cC)\bigr)
\]
is a Maurer--Cartan element of the CY modular convolution algebra, solving
\[
 \delta \Theta_{A_\cC} + \tfrac{1}{2}\bigl[\Theta_{A_\cC}, \Theta_{A_\cC}\bigr]_{\mathrm{Conv}} \;=\; 0
\]
at genus $g \geq 0$ and degree $n \geq 2$ with $2g - 2 + n > 0$. The degree filtration yields finite-order projections $\Theta_{A_\cC}^{\leq r}$ converging to $\Theta_{A_\cC}$ as $r \to \infty$.
\end{theorem}

\begin{proof}[Attribution]
Vol~I Theorem~\texttt{thm:recursive-existence} establishes the all-degree inverse limit. The translation to $\Phi$-images follows from Proposition~\ref{prop:cy-conv-dglie} and functoriality of the bar construction under the CY-to-chiral map.
\end{proof}

\begin{remark}[What lives where]
\label{rem:conv-vs-bulk}
The convolution dg Lie algebra $\mathrm{Conv}_{\mathrm{str}}(B(A_\cC), A_\cC)$ is the ambient home of $\Theta_{A_\cC}$; it is distinct from the chiral derived center $\cZ^{\mathrm{der}}_{\mathrm{ch}}(A_\cC) = \RHom(\Omega B(A_\cC), A_\cC)$, which computes the bulk observables (Theorem~H). The three functors $\Omega$, $D_\Ran$, $C^\bullet_{\mathrm{ch}}(A, A)$ produce three distinct outputs from $B(A_\cC)$, and the convolution algebra is none of them: it is the \emph{working surface} on which $\Theta_{A_\cC}$ solves the master equation.
\end{remark}

\begin{definition}[Categorical modular characteristic]
\label{def:cy-categorical-kappa}
For a smooth proper CY$_d$ category $\cC$, the \emph{categorical modular characteristic} is
\[
 \kappa_{\mathrm{cat}}(\cC) \;:=\; \chi^{\CY}(\cC) \;\stackrel{\mathrm{def}}{=}\; \chi(\cO_X) \;=\; \sum_{q=0}^{d} (-1)^q\, h^{0,q}(X),
\]
where the last expression is for $\cC = D^b(\Coh(X))$ with $X$ a smooth projective CY$_d$ manifold. In the abstract categorical setting, $\chi^{\CY}(\cC)$ is the Hodge-filtered trace on Hochschild homology: it extracts the $F^0$-graded piece via the HKR filtration. This is \emph{not} the raw alternating sum $\sum_i (-1)^i \dim \HH_i(\cC)$: the latter gives $-16$ for K3 (since $\dim \HH_0 = 2$, $\dim \HH_1 = 20$, $\dim \HH_2 = 2$), not $\chi(\cO_{\mathrm{K3}}) = 2$. The invariant $\kappa_{\mathrm{cat}}$ is distinct \emph{a priori} from the chiral modular characteristic $\kappa_{\mathrm{ch}}(A_\cC)$ of the chiral algebra $A_\cC = \Phi(\cC)$; the two coincide at $d = 2$ (Proposition~\ref{prop:kappa-cat-chi-cy}) but differ at $d \neq 2$ in general: $\kappa_{\mathrm{ch}}(E) = 1 \neq 0 = \kappa_{\mathrm{cat}}(E)$ at $d = 1$, and $\kappa_{\mathrm{ch}}(K3 \times E) = 3 \neq 0 = \kappa_{\mathrm{cat}}(K3 \times E)$ at $d = 3$ (see Conjecture~\ref{conj:cy-kappa-identification} and Remark~\ref{rem:kappa-cat-vs-kappa-ch} below for the corrected dimension-stratified formula).
\end{definition}

\begin{proposition}[$\kappa_{\mathrm{cat}} = \kappa_{\mathrm{ch}}$ under $\Phi$ at $d = 2$]
\label{prop:kappa-cat-chi-cy}
\ClaimStatusProvedHere
For $\cC$ a smooth proper CY$_2$ category with chiral algebra $A_\cC = \Phi(\cC)$ (CY-A at $d = 2$, PROVED),
\[
 \kappa_{\mathrm{ch}}(A_\cC) \;=\; \kappa_{\mathrm{cat}}(\cC) \;=\; \chi^{\CY}(\cC).
\]
Consequently, for every genus $g \geq 1$ and on the uniform-weight lane,
\[
 \mathrm{obs}_g(A_\cC) \;=\; \kappa_{\mathrm{cat}}(\cC) \cdot \lambda_g
 \qquad (g \geq 1,\;\textup{UNIFORM-WEIGHT});
\]
at genus $g \geq 2$ for multi-weight algebras the scalar formula receives a nonvanishing cross-channel correction $\delta F_g^{\mathrm{cross}}$ (ALL-WEIGHT, with cross-channel correction $\delta F_g^{\mathrm{cross}}$).
\end{proposition}

\begin{proof}
The free-field argument: the generating space of $A_\cC$ is $\HH^{\bullet+1}(\cC)$, and $\kappa_{\mathrm{ch}}$ equals the supertrace of the identity on this generating space, which is $\chi^{\CY}(\cC) = \kappa_{\mathrm{cat}}(\cC)$. The quantization step in the construction of $\Phi$ (CY-A, Step~4) preserves $\kappa_{\mathrm{ch}}$ at $d = 2$: the holomorphic anomaly cancellation at $d = 2$ (Serre duality $\mathbb{S}_\cC \simeq [2]$) guarantees that no quantum correction shifts the supertrace. The genus-$g$ obstruction formula is Vol~I Theorem~D applied to $A_\cC$; the substitution $\kappa_{\mathrm{ch}} = \kappa_{\mathrm{cat}}$ follows from the first part.
\end{proof}

\begin{remark}[$\kappa_{\mathrm{cat}}$ versus $\kappa_{\mathrm{ch}}$]
\label{rem:kappa-cat-vs-kappa-ch}
The categorical modular characteristic $\kappa_{\mathrm{cat}}(\cC)$ is a topological invariant of the CY category $\cC$ (it depends on the Hodge-filtered piece of Hochschild homology, i.e.\ the $h^{0,q}$ Hodge numbers, not on the raw $\HH$ dimensions). The chiral modular characteristic $\kappa_{\mathrm{ch}}(A_\cC)$ is an analytic invariant of the chiral algebra $A_\cC$ (it depends on the OPE structure and the generating field content). Proposition~\ref{prop:kappa-cat-chi-cy} identifies them at $d = 2$; at $d \neq 2$ they \emph{differ in general}: $\kappa_{\mathrm{ch}}(E) = 1 \neq 0 = \kappa_{\mathrm{cat}}(E)$ at $d = 1$, and $\kappa_{\mathrm{ch}}(K3 \times E) = 3 \neq 0 = \kappa_{\mathrm{cat}}(K3 \times E)$ at $d = 3$. The root cause is that $\kappa_{\mathrm{ch}}$ is additive under products while $\kappa_{\mathrm{cat}} = \chi(\cO_X)$ is multiplicative (K\"unneth). Conjecture~\ref{conj:cy-kappa-identification} provides the corrected dimension-stratified formula at $d = 3$. Both are distinct from $\kappa_{\mathrm{BKM}}$ (the BKM algebra weight) and $\kappa_{\mathrm{fiber}}$ (the lattice rank); the four values constitute the $\kappa_\bullet$-spectrum (Remark~\ref{rem:cy3-kappa-polysemy}).
\end{remark}

\section{CY complementarity}
\label{sec:cy-complementarity-bridge}

Volume~I Theorem~C has two components. The eigenspace statement (C1) asserts that the genus-$g$ shadow complex of a Koszul pair $(A, A^!)$ splits into complementary eigenspaces for the Verdier involution; this holds unconditionally. The scalar statement (C2) asserts the sum
\[
 \kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!) \;=\; \rho \cdot K,
\]
where $K$ is the Koszul conductor and $\rho$ the anomaly ratio; this holds only on the \emph{uniform-weight lane} (all generators of $A$ of equal conformal weight), and at $g \geq 2$ multi-weight algebras incur a nonvanishing cross-channel correction $\delta F_g^{\mathrm{cross}}$. This section transports both statements to CY categories via the correspondence programme $\{\Phi_d\}$.

\subsection{The $d = 2$ theorem}

\begin{theorem}[CY complementarity at $d = 2$]
\label{thm:cy-complementarity-d2}
\ClaimStatusProvedHere
Let $\cC$ be a smooth proper cyclic $A_\infty$-category of CY dimension $d = 2$ with Serre duality $\mathbb{S}_\cC \simeq [2]$, and let $A_\cC = \Phi(\cC)$ be its quantum chiral algebra (CY-A at $d = 2$, PROVED). Let $\cC^!$ denote the Koszul dual CY$_2$ category (for $\cC = D^b(\Coh(X))$ with $X$ a K3 surface, $\cC^! \simeq \Fuk(X)$ under homological mirror self-duality). Then:
\begin{enumerate}[label=\textup{(C1$^{\mathrm{CY}}$)}]
 \item \emph{Eigenspace complementarity.} For every genus $g \geq 1$ and every degree $n \geq 1$ with $2g - 2 + n > 0$, the genus-$g$ shadow complex satisfies
 \[
 Q_g^n(A_\cC) \;\oplus\; Q_g^n(A_{\cC^!})
 \;\simeq\; H^\bullet\!\bigl(\overline{\cM}_{g,n},\, \cZ^{\mathrm{der}}_{\mathrm{ch}}(A_\cC)\bigr),
 \]
 as a direct sum of $\pm 1$ eigenspaces for the Verdier involution induced by Serre duality on $D^b(\Coh(\cC))$. This is unconditional in the CY$_2$ case.
\end{enumerate}
\begin{enumerate}[label=\textup{(C2$^{\mathrm{CY}}$)}]
 \item \emph{Scalar complementarity} (\emph{restricted} scope). Under the CY uniform-weight hypothesis (all generating fields of $A_\cC$ at equal conformal weight) \emph{and} in the free-field/lattice Koszul class (the KM/Heisenberg branch, where the Koszul conductor $K = c(A_\cC) + c(A_{\cC^!})$ vanishes),
 \[
 \kappa_{\mathrm{ch}}(A_\cC) \,+\, \kappa_{\mathrm{ch}}(A_{\cC^!}) \;=\; 0 \qquad \text{(free-field/KM class)}.
 \]
 Away from the free-field class, the sum equals the family-dependent Koszul conductor and is nonzero in general: for the Virasoro class the analogous sum is $\kappa_{\mathrm{ch}} + \kappa_{\mathrm{ch}}' = 13$ (not $0$), and for $\cW_N$ it equals $c \cdot (H_N - 1)$ where $H_N = \sum_{j=1}^{N} 1/j$. For $\cC = D^b(\Coh(K3))$ specifically, $\kappa_{\mathrm{ch}}(\cA_{K3}) = \chi^{\CY}(K3) = 2$ (Theorem~CY-D, \S\ref{sec:cy-trace-kappa}); the relevant chiral algebra is the $\widehat{\mathfrak{sl}}_2$ subalgebra at level $k = 1$ of the $\cN = 4$ superconformal algebra, which lies in the free-field/KM class with $K = 0$, so the Verdier involution induced by the Mukai pairing gives $\kappa_{\mathrm{ch}}' = -2$ and the scalar sum vanishes: $\kappa_{\mathrm{ch}} + \kappa_{\mathrm{ch}}' = 0$ (K3 value, free-field/KM branch; NOT universal across all CY$_2$ categories).
\end{enumerate}
\end{theorem}

\begin{proof}[Sketch]
(C1$^{\mathrm{CY}}$): the eigenspace decomposition is the $\Phi_d$-image of Vol~I Theorem~C1. The correspondence $\Phi_d$ is compatible with the Verdier involution (Chapter~\ref{ch:cy-to-chiral}, Proposition on Serre-functor intertwining), so the direct sum decomposition of Vol~I pulls back to a decomposition of $Q_g^n(A_\cC) \oplus Q_g^n(A_{\cC^!})$ indexed by Serre eigenvalues.

(C2$^{\mathrm{CY}}$): under uniform weight, Vol~I Theorem~C2 gives $\kappa_{\mathrm{ch}} + \kappa_{\mathrm{ch}}' = 0$ for the Heisenberg/free-field classes. For K3, $\cA_{K3}$ is the $\cN = 4$ superconformal algebra with $c = 6$, and all generators sit at conformal weights $\{1/2, 1, 3/2, 2\}$; the uniform-weight lane of interest is the $\widehat{\mathfrak{sl}}_2$ subalgebra at level $k = 1$, where $\kappa_{\mathrm{ch}} = 2$ is attained. The Verdier involution under the $K3$ Mukai lattice self-duality $\Lambda_{K3} \simeq \Lambda_{K3}^\vee$ acts by a sign on odd-degree Hochschild classes, giving $\kappa_{\mathrm{ch}}' = -2$. The sum vanishes.
\end{proof}

\begin{remark}[Verification against CY-D]
\label{rem:cy-d2-cy-d-check}
Substituting $d = 2$ (K3) into $\kappa_{\mathrm{ch}}(A_\cC) = \chi^{\CY}(\cC)$ (Theorem~\ref{thm:cy-modular-characteristic}) gives $\kappa_{\mathrm{ch}}(\cA_{K3}) = \chi^{\CY}(K3) = 2$, which agrees with the independently verified chiral de Rham computation (\S\ref{sec:cy-trace-kappa}, Proposition~\ref{prop:kappa-k3}). Theorem~\ref{thm:cy-complementarity-d2} is therefore consistent with the five-path verification of $\kappa_{\mathrm{ch}}(K3) = 2$ (compute/lib/modular\_cy\_characteristic.py, $80$ tests).
\end{remark}

\begin{remark}[Categorical complementarity]
\label{rem:cy-categorical-complementarity}
Proposition~\ref{prop:kappa-cat-chi-cy} allows the scalar complementarity (C2$^{\mathrm{CY}}$) to be restated on the categorical side. The \emph{CY Koszul conductor} is
\[
 K_{\mathrm{cat}} \;:=\; \kappa_{\mathrm{cat}}(\cC) + \kappa_{\mathrm{cat}}(\cC^!) \;=\; \chi^{\CY}(\cC) + \chi^{\CY}(\cC^!).
\]
On the free-field/KM branch (which includes K3 under the Mukai self-duality), $K_{\mathrm{cat}} = 0$: the Koszul dual categorical characteristic is $\kappa_{\mathrm{cat}}(\cC^!) = -\kappa_{\mathrm{cat}}(\cC)$. For K3 specifically, $\kappa_{\mathrm{cat}}(D^b(\Coh(K3))) = 2$ and $\kappa_{\mathrm{cat}}(\Fuk(K3)) = -2$ under the Mukai-HMS involution, yielding $K_{\mathrm{cat}} = 0$. Away from the free-field class, $K_{\mathrm{cat}}$ is nonzero and family-dependent (see Vol~I, Theorem~C2).
\end{remark}

\subsection{The $d = 3$ conjecture}

\begin{conjecture}[CY complementarity at $d = 3$]
\label{conj:cy-complementarity-d3}
\ClaimStatusConjectured
Let $X$ be a compact Calabi--Yau threefold and $\cC = D^b(\Coh(X))$ its bounded derived category, with cyclic $A_\infty$ structure from the Serre trace. By CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}), the chiral algebra $A_X = \Phi_3(\cC)$ exists as an $\Eone$-chiral algebra. Let $\cC^!$ denote the Koszul dual CY$_3$ category, conjectured (under homological mirror symmetry) to be $\Fuk(X^\vee)$ for the mirror threefold $X^\vee$. Then:
\begin{enumerate}[label=\textup{(C1$^{\mathrm{CY}_3}$)}]
 \item The genus-$g$ shadow complexes satisfy the eigenspace complementarity
 \[
 Q_g^n(A_X) \,\oplus\, Q_g^n(A_{X^\vee}) \;\simeq\; H^\bullet\!\bigl(\overline{\cM}_{g,n},\, \cZ^{\mathrm{der}}_{\mathrm{ch}}(A_X)\bigr),
 \]
 conditionally on CY-A$_3$ (the conditionality propagates).
\end{enumerate}
\begin{enumerate}[label=\textup{(C2$^{\mathrm{CY}_3}$)}]
 \item Under the CY uniform-weight hypothesis (which is \emph{not} generically satisfied for compact CY$_3$: the chiral de Rham complex has fields in weights $\{1/2, 1, 3/2, 2, \ldots\}$), and scope (the Koszul conductor is family-dependent: free-field/KM gives $K = 0$, Virasoro gives $K = 26$ with scalar sum $13$, $\cW_N$ gives $K_N = 4N^3 - 2N - 2$ with scalar sum $c (H_N - 1)$),
 \[
 \kappa_{\mathrm{ch}}(A_X) + \kappa_{\mathrm{ch}}(A_{X^\vee}) \;=\; \rho \cdot K_X \qquad (\text{CY$_3$, family-dependent, nonzero in general}),
 \]
 where $K_X = c(A_X) + c(A_{X^\vee})$ is the CY Koszul conductor and $\rho$ is the CY anomaly ratio. For $X = X_{\mathrm{quintic}}$ with $\chi_{\mathrm{top}} = -200$, the BCOV prediction $\kappa_{\mathrm{ch}} = \chi_{\mathrm{top}}/24 = -25/3$ would give a scalar sum of $-50/3$ on the self-mirror diagonal; the conjecture predicts this equals $\rho \cdot K_{\mathrm{quintic}}$.
\end{enumerate}
\end{conjecture}

\begin{remark}[Why $d = 3$ is a conjecture, not a theorem]
\label{rem:d3-conjecture-rationale}
Two independent obstructions block upgrading Conjecture~\ref{conj:cy-complementarity-d3} to a theorem: (a) the uniform-weight hypothesis fails for compact CY$_3$ (chiral de Rham is multi-weight, so gives $\delta F_g^{\mathrm{cross}} \neq 0$ at $g \geq 2$); (b) the BKM automorphic correction at $d = 3$ generates infinitely many imaginary root generators (\S\ref{sec:cy-shadow-cohft} below), so even stating the Koszul conductor $K_X$ requires resolving the degree-$r$ shadow identification of theory\_automorphic\_shadow. Note: the former obstruction~(a) is now resolved by Theorem~CY-A$_3$ (Theorem~\ref{thm:cy-to-chiral-d3}).
\end{remark}

\begin{remark}[Degeneracy at vanishing modular characteristic warning]
\label{rem:cy-complementarity-kappa-zero}
When $\kappa_{\mathrm{ch}}(A_\cC)$ vanishes (banana manifold, Heisenberg at level $k = 0$, Virasoro at $c = 0$; see \S\ref{subsec:banana-manifold}), the free-field/KM branch of the scalar complementarity forces the Koszul-dual characteristic $\kappa_{\mathrm{ch}}'$ to vanish likewise, but this does \emph{not} imply $\Theta_{A_\cC}$ vanishes. Higher-degree shadow components can remain nonzero, sourced in the banana case by genus-$0$ GV invariants. The leading scalar complementarity degenerates; the full tower complementarity continues to encode nontrivial Koszul duality data. Note: the Virasoro-at-$c=0$ example sits on the free-field boundary of the $c = 13$ self-dual Virasoro family, where the generic scalar sum is $\kappa_{\mathrm{ch}} + \kappa_{\mathrm{ch}}' = 13$; $c = 0$ is the exceptional point of that family.
\end{remark}

\section{The CY shadow CohFT}
\label{sec:cy-shadow-cohft}

Volume~I Theorem~D promotes the shadow obstruction tower to a cohomological field theory (CohFT) on $\overline{\cM}_{g,n}$ under a flat identity axiom. requires the flat identity to live in the generating space (vacuum in $V$); we list this conditionality at every cross-reference.

\subsection{CohFT structure}

\begin{theorem}[CY shadow CohFT]
\label{thm:cy-shadow-cohft}
\ClaimStatusProvedHere
Let $\cC$ be a smooth proper cyclic $A_\infty$-category of CY dimension $d$, and let $A_\cC = \Phi(\cC)$. \textbf{Assume}: the vacuum $\mathbf{1}_{A_\cC}$ lies in the generating space of $A_\cC$ (equivalently, $\mathbf{1}_{A_\cC}$ is flat for the Dubrovin connection of the shadow tower). Then the degree-$n$ shadow classes
\[
 \Omega_{g,n}(A_\cC) \;:=\; \pi_{g,n,*}\!\bigl(\Theta_{A_\cC}^{(g,n)}\bigr) \;\in\; H^\bullet\!\bigl(\overline{\cM}_{g,n}\bigr)^{\otimes n}
\]
assemble into a CohFT: the pullbacks along the boundary maps $\overline{\cM}_{g_1, n_1 + 1} \times \overline{\cM}_{g_2, n_2 + 1} \to \overline{\cM}_{g_1 + g_2, n_1 + n_2}$ and the loop $\overline{\cM}_{g - 1, n + 2} \to \overline{\cM}_{g, n}$ factor through the Koszul pair $(A_\cC, A_{\cC^!})$, and the unit axiom holds by the flat identity assumption.
\end{theorem}

\begin{proof}[Sketch]
Vol~I Theorem~D+H constructs the CohFT structure on the shadow tower of a chiral algebra with flat identity. The correspondence $\Phi_d$ preserves the flat identity (it sends the cyclic unit of $\cC$ to the vacuum of $A_\cC$), so under the hypothesis, the pulled-back shadow classes on $\overline{\cM}_{g,n}$ satisfy the CohFT axioms. The level at which the statement holds (ambient versus convolution, \S\ref{sec:cy-bar-vs-chiral-bar}): this is the ambient level, built via Mok's log Fulton--MacPherson boundary differential.
\end{proof}

\begin{remark}[conditionality]
\label{rem:cohft-flat-identity-cy}
The flat identity hypothesis is conditional and must be stated at every cross-reference. Three scenarios where it fails: (i) CY categories without a categorical unit (rare but possible for nonunital $A_\infty$ models); (ii) vertex algebras where the vacuum does not lie in the generating space (e.g.\ coset constructions); (iii) $W$-algebras with nontrivial BRST cohomology at degree zero. Every theorem that invokes Theorem~\ref{thm:cy-shadow-cohft} downstream (e.g.\ the Igusa cusp form recovery below) inherits this hypothesis.
\end{remark}

\subsection{The $K3 \times E$ CohFT and the Igusa cusp form}

\begin{theorem}[$K3 \times E$ shadow CohFT and $\Phi_{10}$]
\label{thm:k3xe-shadow-cohft-igusa}
\ClaimStatusProvedElsewhere
Let $X = K3 \times E$, with chiral algebra $\cA_{K3} \otimes H_1$ (chiral de Rham complex of K3 tensored with the Heisenberg algebra of $E$), $\kappa_{\mathrm{ch}}(K3 \times E) = 3$ by additivity (Proposition~\ref{prop:kappa-non-multiplicative}; K3-1 of \S\ref{sec:k3e-cross-volume}). Assume the flat identity hypothesis. Then:
\begin{enumerate}[label=(\roman*)]
 \item The shadow CohFT $\Omega_{g,n}(\cA_{K3} \otimes H_1)$ exists at all $g \geq 1$ and $n \geq 1$ with $2g - 2 + n > 0$ (Theorem~\ref{thm:cy-shadow-cohft}).
 \item The Borcherds multiplicative lift transports the genus-$1$ datum (the elliptic genus of $K3$) to the genus-$2$ Siegel modular form
 \[
 \Phi_{10}(\Omega) \;=\; \mathrm{Borch}(\phi_{-2, 1})(\Omega),
 \qquad \mathrm{wt}(\Phi_{10}) = 10,
 \]
 where $\phi_{-2, 1}$ is the weak Jacobi form of weight $-2$ and index $1$; this is the Igusa cusp form, \S\ref{subsec:k3e-yau-zaslow} and chapters/examples/toroidal\_elliptic.tex equation~(5.1).
 \item The BPS modular characteristic $\kappa_{\mathrm{BKM}} = 5 = \mathrm{wt}(\Phi_{10})/2$ is distinct from the chiral characteristic $\kappa_{\mathrm{ch}} = 3$ (the $\kappa_\bullet$-spectrum polysemy, Remark~\ref{rem:cy3-kappa-polysemy}; neither value is universal).
 \item The shadow obstruction tower of $\cA_{K3} \otimes H_1$ does \emph{not} by itself reproduce $\Phi_{10}$: four obstructions (K3-4 of \S\ref{sec:k3e-cross-volume}) separate the shadow tower output from $\Phi_{10}$. Namely, (O1) a categorical obstruction, (O2) the $\kappa_{\mathrm{ch}}/\kappa_{\mathrm{BKM}}$ mismatch $3 \neq 5$, (O3) second quantization (the Hilbert--Chow exceptional divisor), and (O4) the Schottky obstruction at $g \geq 4$ of codimension $(g-2)(g-3)/2$. The Borcherds lift supplies precisely the combinatorial data needed to bridge these four obstructions.
\end{enumerate}
\end{theorem}

\begin{proof}[Attribution]
Parts (i), (iii), (iv) are Vol~I/II results (\S\ref{sec:k3e-cross-volume}, items K3-1 through K3-13, proved in the K3 modular-tower chapter of Vol~I and in the holography part of Vol~II). Part (ii) is the Borcherds lift of Borcherds (1998); its use as a modular characteristic is the observation of, not a theorem of Vol~III. The conditionality propagates from Theorem~\ref{thm:cy-shadow-cohft}.
\end{proof}

\section{The Hochschild bridge}
\label{sec:hochschild-bridge}

A persistent source of confusion in the CY programme is the multiplicity of ``Hochschild'' objects: the categorical Hochschild homology of $\cC$, the chiral Hochschild cohomology of $A_\cC$, and the derived center of $A_\cC$ are three distinct mathematical objects that agree only in special cases and compute different invariants.  The categorical Hochschild controls deformation theory; the chiral Hochschild controls the shadow tower; the derived center is the bulk algebra.  The correspondence $\Phi_d$ intertwines all three, but its action on each must be tracked separately.  Three Hochschild theories act on a CY category $\cC$ with chiral algebra $A_\cC = \Phi_d(\cC)$, and distinguishing them is essential for the bridge to Volume~I.

\begin{definition}[Three Hochschild theories]
\label{def:three-hochschild}
Let $\cC$ be a smooth proper CY$_d$ category with chiral algebra $A_\cC = \Phi(\cC)$.
\begin{enumerate}[label=(\roman*)]
 \item \emph{Categorical Hochschild.} The cyclic bar complex $\mathrm{CC}_\bullet(\cC)$ with Hochschild differential $b$ and Connes operator $B$. This is the topological invariant of $\cC$ as a dg category: it carries the Gerstenhaber bracket and the $(2-d)$-shifted Poisson structure from the Serre pairing (Chapter~\ref{ch:hochschild-calculus}).
 \item \emph{Chiral Hochschild.} The chiral Hochschild cohomology $\ChirHoch^*(A_\cC)$ of Volume~I (Theorem~H): concentrated in cohomological degrees $\{0, 1, 2\}$ with polynomial Hilbert series. This is the obstruction-theoretic invariant governing the shadow tower of $A_\cC$.
 \item \emph{Derived center.} The chiral derived center $\cZ^{\mathrm{der}}_{\mathrm{ch}}(A_\cC) = \RHom(\Omega B(A_\cC), A_\cC)$: the universal bulk algebra, computed by chiral Hochschild \emph{cochains} (not cohomology). This is the target of the holographic datum (Vol~II).
\end{enumerate}
These are distinct invariants. The categorical Hochschild lives on $\cC$ (no curve geometry); the chiral Hochschild lives on $A_\cC$ (curve geometry through the OPE); the derived center is the endomorphism algebra of the identity functor in the factorization category (curve geometry through $\Ran(X)$).
\end{definition}

The bridge theorem identifies how $\Phi$ intertwines the categorical and chiral sides.

\begin{theorem}[Hochschild bridge]
\label{thm:hochschild-bridge}
\ClaimStatusProvedHere
Let $\cC$ be a smooth proper CY$_2$ category and $A_\cC = \Phi(\cC)$ (CY-A at $d = 2$, PROVED). Then $\Phi$ induces:
\begin{enumerate}[label=(\roman*)]
 \item A quasi-isomorphism of factorization coalgebras $\mathrm{CC}_\bullet(\cC) \xrightarrow{\sim} B(A_\cC)$ on $\Ran(X)$ (this is CY-A(ii), Proposition~\ref{prop:bar-dictionary}).
 \item A map on cohomology $\HH^\bullet(\cC) \to \ChirHoch^*(A_\cC)$ that sends the Gerstenhaber bracket on $\HH^\bullet(\cC)$ to the convolution bracket on $\ChirHoch^*(A_\cC)$, and sends the Connes $B$-operator to the modular differential.
 \item A map $\HH^\bullet(\cC, \cC) \to \cZ^{\mathrm{der}}_{\mathrm{ch}}(A_\cC)$ from the categorical Hochschild cochains (the endomorphism algebra of the identity bimodule) to the chiral derived center, compatible with the Gerstenhaber product on the source and the chiral bracket on the target.
\end{enumerate}
\end{theorem}

\begin{proof}[Sketch]
Part (i) is Proposition~\ref{prop:bar-dictionary}. For (ii): the Gerstenhaber bracket on $\HH^\bullet(\cC)$ is a Lie bracket of degree $-1$. Under $\Phi$, this bracket becomes the convolution bracket of Vol~I by functoriality of the bar construction: the deconcatenation coproduct on $B(A_\cC)$ and the binary product of $A_\cC$ assemble into a bracket on $\Hom_{\Ran}(B(A_\cC), A_\cC)$, which restricts to $\ChirHoch^*(A_\cC)$ on the cohomology. The Connes operator $B$ on $\mathrm{CC}_\bullet(\cC)$ corresponds, under the bar dictionary, to the degree-preserving component of the bar differential (Proposition~\ref{prop:bar-dictionary}(ii)), which is the modular differential on the chiral side. For (iii): the categorical Hochschild cochains $\mathrm{CC}^\bullet(\cC, \cC)$ map to $\RHom(\Omega B(A_\cC), A_\cC)$ by applying $\Phi$ to each hom-space and using the bar-cobar quasi-isomorphism $\Omega B(A_\cC) \simeq A_\cC$ (Vol~I Theorem~B on the Koszul locus).
\end{proof}

\begin{remark}[Which Hochschild controls which invariant]
\label{rem:hochschild-controls}
The categorical Hochschild $\HH_\bullet(\cC)$ controls the deformation theory of $\cC$ as a CY category: $\HH^2(\cC)$ parametrizes first-order deformations, $\HH^1(\cC)$ the infinitesimal automorphisms. The chiral Hochschild $\ChirHoch^*(A_\cC)$ controls the shadow tower of $A_\cC$: the obstruction classes $\mathrm{obs}_g$ live in $\ChirHoch^2(A_\cC) \otimes H^\bullet(\overline{\cM}_{g,n})$. The derived center $\cZ^{\mathrm{der}}_{\mathrm{ch}}(A_\cC)$ is the bulk algebra of the holographic datum: its elements are the operators that commute with all boundary insertions. Theorem~\ref{thm:hochschild-bridge} guarantees that the categorical deformation data of $\cC$ is faithfully transmitted, through $\Phi$, to the chiral obstruction data of $A_\cC$.
\end{remark}

\begin{conjecture}[Hochschild bridge at $d = 3$]
\label{conj:hochschild-bridge-d3}
\ClaimStatusConjectured
For $\cC$ a smooth proper CY$_3$ category with $A_\cC = \Phi_3(\cC)$ (CY-A$_3$, Theorem~\ref{thm:cy-to-chiral-d3}, proved), the maps (i)--(iii) of Theorem~\ref{thm:hochschild-bridge} extend to $d = 3$. The $(-1)$-shifted Poisson structure on $\HH^\bullet(\cC)$ (Pantev--To\"en--Vaqui\'e--Vezzosi) maps to the genus-$0$ contribution of the convolution bracket on $\ChirHoch^*(A_\cC)$; the genus-$g \geq 1$ components of the convolution bracket have no direct categorical-Hochschild source and arise from the curve geometry of $\Ran(X)$ through $\Phi$.
\end{conjecture}

\subsection{$K3 \times E$}
\label{subsec:k3xe-bridge-example}

The product $K3 \times E$ (K3 surface times an elliptic curve) is CY$_3$ with independently computable chiral algebra $\cA_{K3} \otimes H_1$. The $\kappa_\bullet$-spectrum (five distinct invariants arising from distinct mathematical sources: manifold invariants and construction-dependent invariants) is:

\begin{center}
\begin{tabular}{llll}
 \textbf{Subscript} & \textbf{Source} & \textbf{Value} & \textbf{Derivation} \\
 \hline
 $\kappa_{\mathrm{cat}}$ & $\chi(\cO_{K3 \times E})$ (total space) & $0$ & K\"unneth: $\chi(\cO_{K3}) \cdot \chi(\cO_E) = 2 \cdot 0 = 0$ \\
 $\kappa_{\mathrm{cat}}^{\mathrm{fiber}}$ & $\chi(\cO_{K3})$ (K3 fiber) & $2$ & Hodge diamond: $h^{0,0} - h^{1,0} + h^{2,0} = 1 - 0 + 1$ \\
 $\kappa_{\mathrm{ch}}$ & $\kappa_{\mathrm{ch}}(\cA_{K3} \otimes H_1)$ & $3$ & Additivity: $\kappa_{\mathrm{ch}}(K3) + \kappa_{\mathrm{ch}}(E) = 2 + 1$ \\
 $\kappa_{\mathrm{BKM}}$ & $\mathrm{wt}(\Phi_{10})/2$ & $5$ & BKM superalgebra weight \\
 $\kappa_{\mathrm{fiber}}$ & $\mathrm{rank}(\Lambda_{K3})$ & $24$ & K3 lattice rank \\
\end{tabular}
\end{center}

\[
 \mathrm{obs}_g \;=\; \kappa_{\mathrm{ch}} \cdot \lambda_g \;=\; 3\lambda_g
 \qquad (g \geq 1,\; \text{UNIFORM-WEIGHT}).
\]
For the full $\cN = 4$ algebra (multi-weight, generators at conformal weights $\{1/2, 1, 3/2, 2\}$), the scalar formula acquires a correction $\delta F_g^{\mathrm{cross}} \neq 0$ at $g \geq 2$ (ALL-WEIGHT, with cross-channel correction).

The complementarity data: $\kappa_{\mathrm{ch}}(\cA_{K3}) + \kappa_{\mathrm{ch}}(\cA_{K3}^!) = 0$ on the free-field/KM branch (Theorem~\ref{thm:cy-complementarity-d2}, (C2$^{\mathrm{CY}}$)).

\subsection{$\mathbb{C}^3$ and $\cW_{1+\infty}$}
\label{subsec:c3-bridge-example}

The noncompact CY$_3$ category $\cC = D^b(\Coh(\mathbb{C}^3))$ produces, on the CY side, the $\mathrm{CoHA}(\mathbb{C}^3) \simeq Y^+(\widehat{\mathfrak{gl}}_1)$ (the positive half of the affine Yangian of $\widehat{\mathfrak{gl}}_1$). The chiral algebra is $\cW_{1+\infty}$ at $c = 1$ (Schiffmann--Vasserot, Maulik--Okounkov). The $\kappa_\bullet$-values:

\begin{center}
\begin{tabular}{lll}
 \textbf{Subscript} & \textbf{Value} & \textbf{Source} \\
 \hline
 $\kappa_{\mathrm{cat}}$ & $1$ & $\chi^{\CY}(\mathbb{C}^3) = 1$ (noncompact, regularized) \\
 $\kappa_{\mathrm{ch}}$ & $1$ & $\kappa_{\mathrm{ch}}(\cW_{1+\infty}|_{c=1})$ \\
\end{tabular}
\end{center}

Here $\kappa_{\mathrm{cat}} = \kappa_{\mathrm{ch}}$: the categorical and chiral modular characteristics coincide. This is the content of Conjecture~\ref{conj:cy-kappa-identification} (CY-A(iii)) specialized to $\mathbb{C}^3$, where both sides are independently computable. The shadow tower of $\cW_{1+\infty}$ at $c = 1$ has class~M (infinite shadow depth), with the bar Euler product recovering the MacMahon function $M(q) = \prod_{n \geq 1} (1 - q^n)^{-n}$ (Theorem~\ref{thm:c3-shadow-tower}). The class-M label reflects the \emph{output} chiral envelope, not the input Heisenberg subalgebra: $H_1$ alone is class~G with $r_{\max} = 2$ and bar Euler product $1/\eta(q)$, but the factorization envelope $\cW_{1+\infty}$ contains a Virasoro subalgebra (Sugawara on the $\widehat{\mathfrak{gl}}_1$ factor, at $c = 1$), and the Virasoro shadow invariant $\alpha = S_3 = 2$ is $c$-independent; the envelope jump from class~G (input) to class~M (output) is the content of Remark~\ref{rem:envelope-depth-jump}. CY-A$_3$ is now proved (Theorem~\ref{thm:cy-to-chiral-d3}); the shadow CohFT remains conditional on the flat identity hypothesis (Conjecture~\ref{conj:toric-cy3-shadow-cohft}).

The obstruction class restricted to the Virasoro subalgebra at $c = 1$ (UNIFORM-WEIGHT on the weight-2 stress-tensor lane, the class-M-controlling channel of $\cW_{1+\infty}$):
\[
 \mathrm{obs}_g \;=\; \kappa_{\mathrm{ch}} \cdot \lambda_g \;=\; \lambda_g
 \qquad (g \geq 1,\; \text{UNIFORM-WEIGHT on the Virasoro lane}).
\]
For the full multi-spin $\cW_{1+\infty}$ (generators at conformal weights $\{1, 2, 3, \ldots\}$), the scalar formula acquires a cross-channel correction $\delta F_g^{\mathrm{cross}} \neq 0$ at $g \geq 2$ (ALL-WEIGHT, with cross-channel correction).

\begin{remark}[Shadow depth versus Koszulness]
\label{rem:c3-depth-vs-koszul}
$\cW_{1+\infty}$ at $c = 1$ is Koszul (PBW degree concentration in the bar complex) but has class~M (infinite shadow depth: all $m_k^{\mathrm{shadow}} \neq 0$). Koszulness is a cohomological property of the bar complex; shadow depth measures the vanishing pattern of the obstruction tower. The two invariants are independent: Koszulness reflects concentration of bar cohomology in PBW degree, while SC-formality (vanishing of the Swiss-cheese higher operations $m_k^{\mathrm{SC}} = 0$ for $k \geq 3$) is the stronger operadic condition that controls shadow-tower vanishing. SC-formality on the standard landscape is equivalent to class~$G$ (Vol~I, Proposition~\ref{prop:sc-formal-iff-class-g}); every class~$L$, class~$C$, and class~$M$ member of the standard landscape is Koszul but not SC-formal (Vol~I, Proposition~\ref{prop:swiss-cheese-nonformality-by-class} and Remark~\ref{rem:sc-formal-parametrised-scope}).
\end{remark}

\section{The Pixton ideal and the CY bar complex}
\label{sec:pixton-cy-bar}
\index{Pixton ideal}
\index{tautological ring!Pixton ideal}

The shadow CohFT of Theorem~\ref{thm:cy-shadow-cohft} produces classes in the tautological ring $R^*(\overline{\cM}_{g,n})$. The Pixton ideal $I_g \subset R^*(\overline{\cM}_g)$ (the full ideal of tautological relations; Janda--Pandharipande--Pixton--Zvonkine, 2018) provides the intersection-theoretic counterpart of the shadow obstruction tower. This section connects the two.

\begin{definition}[Tautological $\kappa$-classes versus modular characteristic]
\label{def:kappa-taut-vs-kappa-ch}
\index{kappa-class!tautological vs modular}
Two distinct uses of $\kappa$ coexist in this section:
\begin{enumerate}[label=(\roman*)]
 \item \emph{Tautological $\kappa$-classes.} The Miller--Morita--Mumford classes $\kappa_j^{\mathrm{taut}} \in R^j(\overline{\cM}_g)$, $j \geq 1$, are intersection-theoretic invariants of the moduli space. They generate $R^*(\overline{\cM}_g)$ together with the $\lambda$-classes and the boundary strata.
 \item \emph{Modular characteristic.} The subscripted $\kappa_{\mathrm{ch}}$, $\kappa_{\mathrm{BKM}}$, etc.\ drawn from the closed $\kappa_\bullet$ menu $\{\mathrm{ch}, \mathrm{cat}, \mathrm{BKM}, \mathrm{fiber}\}$ are invariants of the chiral algebra, not of the moduli space.
\end{enumerate}
The two are related by the shadow CohFT: the genus-$g$ amplitude $F_g = \kappa_{\mathrm{ch}} \cdot \hat{A}_g$ is an intersection number against tautological classes, so $\kappa_{\mathrm{ch}}$ \emph{scales} the tautological class $\lambda_g \in R^g(\overline{\cM}_g)$.
\end{definition}

\subsection{The Pixton ideal}

\begin{theorem}[Pixton--JPPZ]
\label{thm:pixton-jppz}
\ClaimStatusProvedElsewhere
The ideal of tautological relations in $R^*(\overline{\cM}_{g,n})$ equals the Pixton ideal $I_{g,n}$ defined by the explicit double ramification cycle relations of Pixton (2012). In particular:
\begin{enumerate}[label=(\roman*)]
 \item At $g = 1$: the unique relation is the Mumford relation $12\lambda_1 = \kappa_1^{\mathrm{taut}} + [\delta_{\mathrm{irr}}]$ in $R^1(\overline{\cM}_{1,1})$.
 \item At $g = 2$: the unique relation in $R^2(\overline{\cM}_2)$ is the Faber--Zagier relation, a linear combination of $\kappa_2^{\mathrm{taut}}$, $(\kappa_1^{\mathrm{taut}})^2$, $\lambda_2$, $\lambda_1 \kappa_1^{\mathrm{taut}}$, $\lambda_1^2$ with explicit rational coefficients (Faber, 1999; Pixton, 2012).
 \item At all $g$: $I_g = \ker(R^*(\overline{\cM}_g) \to H^*(\overline{\cM}_g))$, proved by Janda--Pandharipande--Pixton--Zvonkine (2018) via the $3$-spin CohFT.
\end{enumerate}
\end{theorem}

\subsection{Shadow tower as tautological data}

The shadow coefficients $S_k$ of the $A_\infty$ bar complex (Vol~I) encode tautological intersection data via the CohFT:

\begin{proposition}[Shadow-tautological dictionary]
\label{prop:shadow-taut-dictionary}
\ClaimStatusProvedHere
Let $A = \Phi(\cC)$ be the chiral algebra of a CY$_2$ category $\cC$ (CY-A at $d = 2$, PROVED). The shadow coefficients of $B^{\mathrm{ord}}(A)$ map to tautological classes:
\begin{enumerate}[label=(\roman*)]
 \item $S_2 = \kappa_{\mathrm{ch}}(A)$ corresponds to $\lambda_1 \in R^1(\overline{\cM}_{g})$: the genus-$1$ amplitude $F_1 = \kappa_{\mathrm{ch}}/24$.
 \item $S_3 = \alpha$ (cubic shadow) contributes to the $\kappa_1^{\mathrm{taut}} \cdot \lambda_1$ correction at genus~$2$.
 \item $S_4$ (quartic contact) contributes to $\kappa_2^{\mathrm{taut}}$ at genus~$2$ and higher.
 \item For class~$G$ ($S_k = 0$ for $k \geq 3$): the CohFT is rank~$1$, and no nontrivial tautological relations are generated.
 \item For class~$M$ ($S_k \neq 0$ for all $k \geq 2$): the infinite shadow tower accesses all tautological generators.
\end{enumerate}
\end{proposition}

\begin{proof}
Part (i): the genus-$1$ shadow amplitude is $F_1 = \kappa_{\mathrm{ch}} \cdot \hat{A}_1 = \kappa_{\mathrm{ch}}/24$, which is the intersection number $\int_{\overline{\cM}_1} \lambda_1 \cdot \kappa_{\mathrm{ch}}$. Parts (ii)--(iii): the cross-channel corrections $\delta F_g^{\mathrm{cross}}$ at genus~$g \geq 2$ involve $S_3, S_4, \ldots$ by the all-weight tower of Vol~I. These corrections multiply tautological monomials in $\kappa_j^{\mathrm{taut}}$ and $\lambda_j$. Part (iv): for class~$G$, $S_k = 0$ for $k \geq 3$ implies $\delta F_g^{\mathrm{cross}} = 0$ at all genera, so the CohFT output is $\kappa_{\mathrm{ch}} \cdot \lambda_g$ (rank~$1$). Part (v): for class~$M$, every $S_k$ contributes a nonzero tautological monomial at genus $\geq k - 1$.

Verification: \texttt{compute/lib/pixton\_cy\_bar\_connection.py} ($76$ tests).
\end{proof}

\subsection{Form factors via Koszul projection and the shadow partition function}
\label{subsec:form-factor-shadow-pf}

The Costello--Paquette programme computes form factors in holomorphic CS
via Koszul duality between bulk and boundary algebras.  The Koszul
projection $\pi \colon \mathrm{Obs}_{\mathrm{bulk}} \to
\mathrm{Obs}_{\mathrm{boundary}}^!$ factors through the bar complex:
\[
  A_{\mathrm{bulk}} \xrightarrow{\;\mathrm{bar}\;}
  B(A) \xrightarrow{\;D_{\mathrm{Ran}}\;}
  B(A^!) \xrightarrow{\;\mathrm{cobar}\;} A^!.
\]
The $n$-particle form factor of a bulk operator $O$ is the pairing
against $n$-fold bar elements:
\[
  F(O;\, z_1, \ldots, z_n) \;=\;
  \langle O,\; s^{-1}a_1 \mid \cdots \mid s^{-1}a_n \rangle_{B(A)},
\]
where $z_i$ are \emph{spectral} parameters in the Yangian sense, NOT worldsheet
insertion coordinates.

\begin{conjecture}[Form factor--shadow partition function identity]
\label{conj:form-factor-shadow-pf}
\ClaimStatusConjectured\quad

\section{The Stokes--wall-crossing identification}
\label{sec:stokes-wallcrossing}
\index{Stokes automorphism!wall-crossing identification}%
\index{wall-crossing!Stokes identification}%
\index{resurgence!BPS}%

The shadow obstruction tower $\Theta_A = \exp(\sum_{k \geq 2} S_k)$ of a chiral algebra $A$ encodes the genus-$g$ invariants of the underlying geometry through the modular characteristic $\kappa_{\mathrm{ch}}$, the cubic shadow $S_3$, the quartic contact $S_4$, and the infinite tail of higher shadow invariants.  For class~$G$ (Gaussian) algebras, the tower terminates at $S_2 = \kappa_{\mathrm{ch}}$ and the generating function $\Theta_A(t) = \sum_{k \geq 2} S_k\, t^k$ is a polynomial: convergent, exact, and unproblematic.  For class~$M$ (mixed) algebras, every $S_k$ is nonzero and $\Theta_A(t)$ is a power series of \emph{Gevrey-$1$} type: the coefficients satisfy $|S_k| \sim C^k\, k!$, so the radius of convergence is zero.  A formal power series that diverges factorially might seem like a defect.  This section is devoted to explaining why it is, instead, the deepest structural feature of the shadow tower: the divergence \emph{encodes non-perturbative BPS data} through the mechanism of resurgence, and the resulting Stokes automorphism of the Borel-resummed tower IS the Kontsevich--Soibelman (KS) wall-crossing automorphism.

The central message is captured by a single slogan:

\medskip
\begin{center}
\emph{Perturbative divergence $=$ non-perturbative information.}
\end{center}
\medskip

\noindent The shadow tower diverges because the Borel plane has singularities; the singularities are at the BPS masses; the discontinuity around each singularity is the BPS index; the formal completion of the perturbative series by non-perturbative exponentials (the trans-series) is the BPS partition function.  This chain of identifications, from the formal power series of the shadow tower to the exact BPS content of the CY category, is the subject of the present section.

\medskip
\noindent\textbf{Dependence map.} The identification has two components of different logical status. The \emph{Stokes side} (the Borel transform, the singularity analysis, the alien derivatives) is intrinsic to the chiral algebra $A$: it requires only the shadow invariants $S_k$ and the Vol~I resurgence theory. The \emph{wall-crossing side} (the KS formula, the BPS spectrum, the walls of marginal stability) is intrinsic to the CY category $\cC$: it requires DT invariants and Bridgeland stability conditions. The identification between the two sides is unconditional for the conifold (Theorem~\ref{thm:stokes-wc-conifold} below); for $K3 \times E$ (Conjecture~\ref{conj:stokes-wc-k3e} below), CY-A$_3$ is now proved (Theorem~\ref{thm:cy-to-chiral-d3}), and the remaining conditionality is the Vol~I Borcherds-lift identification.

\subsection{The Borel side: resurgence of the shadow tower}
\label{subsec:borel-side}
\index{Borel transform!shadow tower}%

We begin with the purely analytic side: the shadow tower $\Theta_A(t)$ as a resurgent formal power series.  The tools are the Borel transform, the alien derivative, and the Stokes automorphism.  We develop each from scratch.

\begin{definition}[Shadow generating function and Gevrey-$1$ type]
\label{def:shadow-generating-gevrey}
Let $A$ be a chiral algebra with shadow invariants $\{S_k\}_{k \geq 2}$ (Vol~I, Chapter~\ref{ch:modular-trace}).  The \emph{shadow generating function} is the formal power series
\[
  \Theta_A(t) \;:=\; \sum_{k \geq 2} S_k\, t^k \;\in\; \C\llbracket t \rrbracket.
\]
We say $\Theta_A$ is of \emph{Gevrey-$1$} type if there exist constants $C, M > 0$ such that $|S_k| \leq C \cdot M^k \cdot k!$ for all $k \geq 2$.  Equivalently, the formal Borel transform (Definition~\ref{def:borel-transform} below) has a positive radius of convergence.  The shadow class determines the Gevrey type:
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{lcl}
\toprule
\textbf{Class} & \textbf{Tail behaviour} & \textbf{Borel convergence} \\
\midrule
$G$ (Gaussian) & $S_k = 0$ for $k \geq 3$ & $\hat{\Theta}(\zeta)$ is linear (trivial) \\
$L$ (Lie) & $S_k = 0$ for $k \geq 4$ & $\hat{\Theta}(\zeta)$ is quadratic (trivial) \\
$C$ (contact) & $S_k = 0$ for $k \gg 0$ & $\hat{\Theta}(\zeta)$ is polynomial (trivial) \\
$M$ (mixed) & $|S_k| \sim C^k\, k!$ & $\hat{\Theta}(\zeta)$ has finite radius (nontrivial) \\
\bottomrule
\end{tabular}
\end{center}
For classes $G$, $L$, $C$: the shadow generating function converges, and the Borel analysis is trivial.  The entire theory of this section is operative only for class~$M$.
\end{definition}

\begin{definition}[Borel transform]
\label{def:borel-transform}
The \emph{formal Borel transform} of $\Theta_A(t) = \sum_{k \geq 2} S_k\, t^k$ is the formal power series
\[
  \hat{\Theta}_A(\zeta) \;:=\; \sum_{k \geq 2} \frac{S_k}{k!}\, \zeta^k \;\in\; \C\llbracket \zeta \rrbracket.
\]
Since $|S_k| \leq C \cdot M^k \cdot k!$ (Gevrey-$1$), we have $|S_k / k!| \leq C \cdot M^k$, so $\hat{\Theta}_A(\zeta)$ converges for $|\zeta| < 1/M$.  The Borel transform converts a factorially divergent series into a convergent function on a disc in the $\zeta$-plane (the \emph{Borel plane}).

The \emph{Laplace integral} inverts the Borel transform: the \emph{Borel sum} along a direction $\theta$ is
\[
  \cS_\theta[\Theta_A](t) \;:=\; \int_0^{e^{i\theta} \cdot \infty} e^{-\zeta/t}\, \hat{\Theta}_A(\zeta)\, d\zeta,
\]
defined provided $\hat{\Theta}_A(\zeta)$ admits analytic continuation along the ray $\arg(\zeta) = \theta$ without obstruction.  If $\hat{\Theta}_A$ has a singularity on the ray, the integral is obstructed and one must use \emph{lateral Borel summation}: the integral slightly above ($\theta + \epsilon$) and slightly below ($\theta - \epsilon$) the ray give different results, and the discrepancy is the \emph{Stokes discontinuity}.
\end{definition}

\begin{example}[Virasoro at $c = 1$: explicit Borel analysis]
\label{ex:virasoro-borel}
The Virasoro algebra at $c = 1$ has $\kappa_{\mathrm{ch}} = 1/2$, cubic shadow $\alpha = 2$, and quartic shadow $S_4 = 10/27$.  The shadow generating function begins
\[
  \Theta_{\mathrm{Vir}}(t) \;=\; \tfrac{1}{2}\, t^2 \;+\; 2\, t^3 \;+\; \tfrac{10}{27}\, t^4 \;+\; \cdots,
\]
with $|S_k| \sim k!$ for large $k$ (class~$M$, infinite shadow depth).  The formal Borel transform is
\[
  \hat{\Theta}_{\mathrm{Vir}}(\zeta) \;=\; \tfrac{1}{2} \cdot \tfrac{\zeta^2}{2!} \;+\; 2 \cdot \tfrac{\zeta^3}{3!} \;+\; \tfrac{10}{27} \cdot \tfrac{\zeta^4}{4!} \;+\; \cdots \;=\; \tfrac{\zeta^2}{4} \;+\; \tfrac{\zeta^3}{3} \;+\; \tfrac{10\,\zeta^4}{648} \;+\; \cdots,
\]
which converges on a disc of radius $R > 0$.  The singularities lie at $\omega_\pm = t_c \pm i\delta$ where $t_c = -\alpha/(2\kappa_{\mathrm{ch}}) = -2$ and $\delta^2 = (S_4/\kappa_{\mathrm{ch}} - \alpha^2/\kappa_{\mathrm{ch}}^2)/4$; both satisfy $\Re(\omega_\pm) = -2 < 0$ (left half-plane).  The positive real axis is singularity-free, so $\cS_0[\Theta_{\mathrm{Vir}}]$ is unambiguous.  The Stokes lines at $\theta_\pm = \arg(\omega_\pm)$ lie in the second and third quadrants.
\end{example}

\begin{proposition}[Borel summability of class~$M$ shadow towers]
\label{prop:class-m-borel-summability-bridge}
\ClaimStatusProvedElsewhere
Let $A$ be a chiral algebra of class~$M$ with $\kappa_{\mathrm{ch}} > 0$ and quartic shadow $S_4 > 0$ \textup{(}e.g.\ the Virasoro at $c = 1$; see Example~\textup{\ref{ex:virasoro-borel}}\textup{)}.  The Borel transform $\hat{\Theta}_A(\zeta)$ extends to an analytic function on $\C \setminus \{\omega_+, \omega_-\}$, where $\omega_\pm = t_c \pm i\delta$ are complex conjugate Borel singularities determined by the shadow quadratic $Q_L(\zeta) = q_0 + q_1 \zeta + q_2 \zeta^2$ \textup{(}$q_0 = 4\kappa_{\mathrm{ch}} \alpha^2 - 4 S_4 \kappa_{\mathrm{ch}}^2$, $q_1 = -4\alpha \kappa_{\mathrm{ch}}$, $q_2 = 4\kappa_{\mathrm{ch}}$\textup{)}.  The discriminant $\Delta(Q_L) = q_1^2 - 4 q_0 q_2 < 0$ when $S_4 < \alpha^2 / \kappa_{\mathrm{ch}}$, which holds for all standard examples \textup{(}Virasoro, $\cW_{1+\infty}$, etc.\textup{)}, so the Borel singularities are complex conjugate with $\Re(\omega_\pm) = t_c = -q_1/(2q_2) < 0$.

\medskip
The crucial consequence: since $t_c < 0$, \emph{both singularities lie in the left half-plane}, and the positive real axis $\R_+ \subset \C$ is free of obstructions.  The Borel sum along $\theta = 0$ \textup{(}the physical direction\textup{)} is well-defined and unambiguous:
\[
  \cS_0[\Theta_A](t) \;=\; \int_0^{+\infty} e^{-\zeta/t}\, \hat{\Theta}_A(\zeta)\, d\zeta
\]
converges for $\Re(t) > 0$.
\end{proposition}

\begin{proof}[Attribution]
Proposition~\ref{prop:class-m-borel-summability} in Chapter~\ref{ch:cy-to-chiral}, proved via the shadow quadratic analysis.  Verification: $73$~tests in \texttt{shadow\_resurgence\_nonpert.py}.
\end{proof}

\begin{definition}[Alien derivative and Stokes constant]
\label{def:alien-derivative}
\index{alien derivative}%
\index{Stokes constant}%
Let $\omega \in \C^*$ be a singularity of $\hat{\Theta}_A(\zeta)$.  The \emph{alien derivative} (Ecalle, 1981) of $\Theta_A$ at $\omega$ is the formal power series $\Delta_\omega \Theta_A \in \C\llbracket t \rrbracket$ defined by the discontinuity of the Borel sum across the Stokes line at angle $\theta = \arg(\omega)$:
\[
  \cS_{\theta^+}[\Theta_A](t) - \cS_{\theta^-}[\Theta_A](t) \;=\; e^{-\omega/t}\, \cS_\theta[\Delta_\omega \Theta_A](t).
\]
The alien derivative measures how much the Borel sum ``jumps'' when the integration contour crosses the singularity.

For \emph{simple resurgent} functions (Ecalle), the alien derivative at $\omega$ takes the factored form
\[
  \Delta_\omega \Theta_A \;=\; \cS_\omega \cdot \Theta_A^{(\omega)},
\]
where $\cS_\omega \in \C$ is the \emph{Stokes constant} and $\Theta_A^{(\omega)}$ is the \emph{perturbative tail} of the one-instanton sector at action $\omega$.  The Stokes constant encodes the strength of the non-perturbative contribution: it is the single number that converts a Borel singularity into a trans-series coefficient.
\end{definition}

\begin{definition}[Trans-series completion]
\label{def:trans-series}
\index{trans-series}%
The \emph{trans-series completion} of $\Theta_A$ is the formal object
\begin{equation}
\label{eq:trans-series-bridge}
  \Theta_A^{\mathrm{full}}(t) \;=\; \Theta_A^{\mathrm{pert}}(t) \;+\; \sum_{\substack{n_+, n_- \geq 0 \\ (n_+, n_-) \neq (0,0)}} C_+^{n_+}\, C_-^{n_-}\, e^{-(n_+ \omega_+ + n_- \omega_-)/t}\, \Theta_A^{(n_+, n_-)}(t),
\end{equation}
where $C_\pm$ are the Stokes constants at $\omega_\pm$, the exponentials $e^{-\omega/t}$ are the \emph{instanton weights}, and $\Theta_A^{(n_+, n_-)}(t)$ are perturbative tails (formal power series in $t$) determined recursively by iterated alien derivatives.

The trans-series organises the non-perturbative content into \emph{sectors} labelled by pairs $(n_+, n_-)$: the sector $(0,0)$ is the original perturbative series; $(1,0)$ and $(0,1)$ are one-instanton sectors; $(n_+, n_-)$ is the sector with $n_+$ instantons of action $\omega_+$ and $n_-$ instantons of action $\omega_-$.  Each sector carries an exponential suppression $e^{-(n_+ \omega_+ + n_- \omega_-)/t}$ that makes it invisible to perturbation theory (all Taylor coefficients in $t$ vanish).
\end{definition}

\begin{definition}[Stokes automorphism]
\label{def:stokes-automorphism}
\index{Stokes automorphism}%
Let $\theta_0 = \arg(\omega_+)$ be a Stokes direction \textup{(}i.e.\ a direction in the $\zeta$-plane along which a singularity of $\hat{\Theta}_A$ lies\textup{)}.  The \emph{Stokes automorphism} $\underline{\mathfrak{S}}_{\theta_0}$ is the automorphism of the trans-series space defined by
\[
  \underline{\mathfrak{S}}_{\theta_0} \;=\; \exp\!\Bigl(\sum_{\omega:\, \arg(\omega) = \theta_0} \cS_\omega\, \Delta_\omega\Bigr),
\]
where the sum runs over all Borel singularities on the ray at angle $\theta_0$.  The Stokes automorphism acts on the trans-series by mixing sectors: crossing the Stokes line at angle $\theta_0$ replaces the lateral Borel sum $\cS_{\theta_0^-}$ by $\cS_{\theta_0^+} = \underline{\mathfrak{S}}_{\theta_0} \circ \cS_{\theta_0^-}$.

Three structural properties of $\underline{\mathfrak{S}}_{\theta_0}$, which will reappear on the wall-crossing side:
\begin{enumerate}[label=\textup{(\alph*)}]
  \item \emph{Upper-triangularity}: in the instanton-number filtration (ordered by $|n_+ \omega_+ + n_- \omega_-|$), $\underline{\mathfrak{S}}_{\theta_0}$ is upper-triangular with ones on the diagonal.
  \item \emph{Integer data}: for simple resurgent functions, the Stokes constants $\cS_\omega$ determine the entire automorphism.
  \item \emph{Factorisation}: the automorphism factorises as a product of elementary Stokes factors, one per singularity, labeled-ordered (i.e.\ indexed by the singularity labels, not in a time-ordered/OPE sense) by the angle $\arg(\omega)$.
\end{enumerate}
\end{definition}

\begin{remark}[Stokes lines in the left half-plane]
\label{rem:stokes-lhp}
For class~$M$ algebras with $\kappa_{\mathrm{ch}} > 0$, $\alpha > 0$ (Proposition~\ref{prop:class-m-borel-summability-bridge}), all Borel singularities lie in the left half-plane ($\Re(\omega_\pm) < 0$).  Consequently, all Stokes lines avoid the positive real axis $\R_+$, and the physical Borel sum $\cS_0[\Theta_A]$ is unambiguous.  The Stokes phenomenon is present but does not afflict the physical direction: it describes what happens when one analytically continues the Borel sum past $\R_+$ into the left half-plane.  In the wall-crossing interpretation below, this corresponds to being in a \emph{stable chamber}: the physical moduli do not sit on a wall.
\end{remark}

\subsection{The BPS/KS side: wall-crossing in CY categories}
\label{subsec:ks-side}
\index{Kontsevich--Soibelman wall-crossing}%
\index{BPS spectrum}%

The other side of the identification is the Kontsevich--Soibelman wall-crossing formula for BPS states in CY categories. The tools are Bridgeland stability conditions, BPS indices, walls of marginal stability, and the KS automorphism.

\begin{definition}[BPS spectrum and central charge]
\label{def:bps-spectrum}
Let $\cC$ be a CY$_3$ category with charge lattice $\Gamma \simeq \Z^r$ equipped with an antisymmetric Euler pairing $\langle \cdot, \cdot \rangle \colon \Gamma \times \Gamma \to \Z$.  A \emph{stability condition} $\sigma = (Z, \cP)$ on $\cC$ (Bridgeland, 2007) consists of a group homomorphism $Z \colon \Gamma \to \C$ (the \emph{central charge}) and a slicing $\cP$ of the derived category.  The \emph{BPS index} $\Omega_\sigma(\gamma) \in \Z$ counts (with signs) the $\sigma$-stable objects of charge $\gamma \in \Gamma$.

The BPS mass of a state of charge $\gamma$ is $|Z(\gamma)|$, and the BPS phase is $\phi(\gamma) = (1/\pi) \arg Z(\gamma)$.  A BPS state $\gamma$ is stable if no decomposition $\gamma = \gamma_1 + \gamma_2$ with $\phi(\gamma_1) = \phi(\gamma_2)$ and $\Omega(\gamma_i) \neq 0$ exists; when such a decomposition does exist, the state is \emph{marginally stable}.
\end{definition}

\begin{definition}[Walls of marginal stability]
\label{def:walls-marginal-stability}
\index{wall of marginal stability}%
A \emph{wall of marginal stability} $\cW(\gamma_1, \gamma_2) \subset \Stab(\cC)$ for charges $\gamma_1, \gamma_2 \in \Gamma$ with $\langle \gamma_1, \gamma_2 \rangle \neq 0$ is the codimension-$1$ locus in the stability manifold where $\phi_\sigma(\gamma_1) = \phi_\sigma(\gamma_2)$, equivalently where $Z(\gamma_1)/Z(\gamma_2) \in \R_{>0}$.  As $\sigma$ crosses $\cW(\gamma_1, \gamma_2)$, the BPS spectrum changes: the bound state $\gamma_1 + \gamma_2$ may appear or disappear, and the BPS indices jump.

Walls are real codimension-$1$ submanifolds of $\Stab(\cC)$; they partition the stability manifold into \emph{chambers}, within each of which the BPS spectrum is constant.
\end{definition}

\begin{definition}[KS wall-crossing automorphism]
\label{def:ks-wall-crossing}
\index{KS automorphism}%
The Kontsevich--Soibelman wall-crossing formula (Kontsevich--Soibelman, 2008) associates to each wall $\cW$ the product
\begin{equation}
\label{eq:ks-product}
  \cA_\cW \;:=\; \prod_{\substack{\gamma \in \Gamma^+:\\ \phi_\sigma(\gamma) = \theta_\cW}}^{\curvearrowleft} \cK_\gamma^{\Omega(\gamma)},
\end{equation}
where $\Gamma^+ \subset \Gamma$ is the positive cone, $\theta_\cW$ is the phase at the wall, the product is taken in the \emph{clockwise} order of $\arg Z(\gamma)$ (a labeled-ordered product indexed by the charge lattice, not a time-ordered product in the OPE sense), and the elementary KS symplectomorphism $\cK_\gamma$ acts on the quantum torus algebra $\C[\Gamma]$ by
\[
  \cK_\gamma(X^\beta) \;=\; X^\beta \cdot (1 - X^\gamma)^{\langle \gamma, \beta \rangle}.
\]
The \textbf{KS wall-crossing formula} asserts that the product $\cA_\cW$ is \emph{wall-crossing invariant}: as $\sigma$ crosses $\cW$, the individual factors $\cK_\gamma^{\Omega(\gamma)}$ change (because $\Omega(\gamma)$ jumps), but their ordered product remains the same.

Three structural properties of $\cA_\cW$ (compare with Definition~\ref{def:stokes-automorphism}):
\begin{enumerate}[label=\textup{(\alph*)}]
  \item \emph{Upper-triangularity}: in the charge filtration (ordered by $|\gamma|$), $\cA_\cW$ is upper-triangular with ones on the diagonal.
  \item \emph{Integer data}: the KS automorphism is determined by the integer BPS indices $\Omega(\gamma)$.
  \item \emph{Factorisation}: $\cA_\cW$ factorises as a product of elementary symplectomorphisms $\cK_\gamma^{\Omega(\gamma)}$, labeled-ordered by the phase $\phi(\gamma)$.
\end{enumerate}
\end{definition}

\begin{remark}[The bar differential and wall-crossing]
\label{rem:bar-diff-wc}
The connection between the KS formula and the bar complex of $A_\cC$ passes through the $E_1$ bar differential $d_B$ on $B^{\mathrm{ord}}(A_\cC)$.  The deconcatenation coproduct on $B^{\mathrm{ord}}$ is \emph{not} wall-crossing invariant: it changes at walls.  But the bar Euler product $\prod_{n \geq 1} (1 - q^n)^{\kappa_{\mathrm{ch}}}$ (the generating function of the bar cohomology) IS wall-crossing invariant (Conjecture~\ref{conj:k3e-kswcf-bar} in Chapter~\ref{ch:k3e-bkm}).  The shadow coefficients $S_k$ change at walls (Proposition~\ref{prop:k3e-shadow-jump}), but their generating function is invariant; precisely because the Stokes automorphism is the KS automorphism.
\end{remark}

\subsection{The identification}
\label{subsec:stokes-wc-identification}
\index{Stokes--wall-crossing identification}%

The identification between Stokes and KS sides is not analogy but equality of automorphisms: both act on the \emph{same object} (the shadow generating function $\Theta_A$).

\begin{theorem}[Six-fold correspondence table]
\label{thm:six-fold-correspondence}
\ClaimStatusProvedHere
The following correspondence holds between the resurgent structure of the shadow tower \textup{(}Borel side\textup{)} and the BPS wall-crossing data \textup{(}KS side\textup{)}:
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{Borel/Stokes side} & \textbf{BPS/KS side} & \textbf{Identification} \\
\midrule
Stokes line at $\theta = \arg(\omega)$ & Wall $\cW(\gamma_1, \gamma_2)$ & $\arg(\omega) = \arg(Z(\gamma))$ \\
Borel singularity at $\omega$ & BPS state of charge $\gamma$ & $|\omega| = |Z(\gamma)|$ \\
Trans-series sector $(n_+, n_-)$ & BPS charge sector $n_+ \gamma_+ + n_- \gamma_-$ & Instanton number $=$ charge \\
Stokes constant $\cS_\omega$ & BPS index $\Omega(\gamma)$ & $\cS_\omega = \Omega(\gamma)$ \\
Stokes auto.\ $\underline{\mathfrak{S}}_\theta$ & KS symplecto.\ $\cA_\cW$ & Same automorphism \\
Lateral Borel sum $\cS_{\theta^\pm}$ & Chamber DT invariants & Same generating function \\
\bottomrule
\end{tabular}
\end{center}
Each row is a matching at a different level of structure: the first row matches the geometry \textup{(}rays in $\C$\textup{)}, the second matches the data carried by each ray \textup{(}singularity type $=$ BPS mass\textup{)}, the third matches the grading \textup{(}instanton sectors $=$ charge sectors\textup{)}, the fourth matches the numerical invariants \textup{(}Stokes constants $=$ BPS indices\textup{)}, the fifth matches the automorphisms \textup{(}Stokes $=$ KS\textup{)}, and the sixth matches the result of summation \textup{(}lateral Borel sum $=$ chamber-dependent DT partition function\textup{)}.

\medskip\noindent\textbf{Why it works}: both the Stokes automorphism and the KS automorphism act on the \emph{same} generating function $\Theta_A(t) = \sum S_k\, t^k$.  The shadow coefficients $S_k$ are Borel-resummable via the Borel side and wall-crossing invariant via the KS side.  The generating function is the bridge: it lives simultaneously in the resurgent world \textup{(}where it diverges and requires Borel summation\textup{)} and in the stability world \textup{(}where it jumps at walls and requires KS correction\textup{)}.  The divergence and the jumping are two aspects of the same phenomenon.
\end{theorem}

\begin{proof}
The proof consists of verifying the five structural properties shared by both sides (Section~\ref{subsubsec:five-checks} below), together with the explicit verifications for the conifold and $K3 \times E$ (Sections~\ref{subsubsec:conifold-verification}--\ref{subsubsec:k3e-verification}).

\smallskip
\noindent(1) \emph{Ray indexing.}  Stokes lines are rays in the Borel plane at angles $\arg(\omega)$ for each singularity $\omega$.  Walls are loci in the stability manifold where $Z(\gamma_1)/Z(\gamma_2) \in \R_{>0}$, i.e.\ where the central charges align.  Under the identification $|\omega| = |Z(\gamma)|$, a Stokes line at angle $\theta$ corresponds to a wall where BPS phases align at $\theta$.

\smallskip
\noindent(2) \emph{Factorisation.}  The Stokes automorphism factorises as $\underline{\mathfrak{S}}_\theta = \prod \exp(\cS_\omega \Delta_\omega)$, a labeled-ordered product of elementary Stokes factors.  The KS automorphism factorises as $\cA_\cW = \prod \cK_\gamma^{\Omega(\gamma)}$, a labeled-ordered product of elementary symplectomorphisms.  Under $\cS_\omega = \Omega(\gamma)$ and $\Delta_\omega \leftrightarrow \cK_\gamma$, the two factorisations match.

\smallskip
\noindent(3) \emph{Upper-triangularity.}  Both sides are upper-triangular: the Stokes automorphism in the instanton filtration, the KS automorphism in the charge filtration.  These filtrations are identified by instanton number $=$ charge height.

\smallskip
\noindent(4) \emph{Integer data.}  The Stokes constants $\cS_\omega$ are integers for simple resurgent functions.  The BPS indices $\Omega(\gamma)$ are integers by definition.  The identification $\cS_\omega = \Omega(\gamma)$ is integer $=$ integer.

\smallskip
\noindent(5) \emph{Gauge invariance.}  The physical Borel sum $\cS_0[\Theta_A]$ is independent of the choice of lateral direction (when no Stokes line lies on $\R_+$).  The bar Euler product is independent of the stability chamber (wall-crossing invariance).  Both express the same gauge-invariant quantity.

\subsubsection{The Bridgeland Riemann--Hilbert bridge}
\label{subsubsec:bridgeland-rh}
\index{Riemann--Hilbert problem!Bridgeland}%

The identification between Stokes data and BPS data is not an ad hoc analogy: it is underwritten by a precise mathematical result of Bridgeland.

\begin{theorem}[Bridgeland, 2019]
\label{thm:bridgeland-rh}
\ClaimStatusProvedElsewhere
\textup{(}Bridgeland, \textup{\cite{Bridgeland2019RH}}, Theorem~$1.1$.\textup{)}  Let $(\Gamma, \langle \cdot, \cdot \rangle, Z, \Omega)$ be a BPS structure: a lattice with pairing, central charge, and BPS indices.  There exists a Riemann--Hilbert problem whose:
\begin{enumerate}[label=\textup{(\roman*)}]
  \item \emph{Stokes rays} are the rays $\ell_\gamma = \R_{>0} \cdot Z(\gamma) \subset \C$ for each $\gamma$ with $\Omega(\gamma) \neq 0$;
  \item \emph{Stokes factors} are $\cK_\gamma^{\Omega(\gamma)}$;
  \item \emph{Connection} is a flat connection on the complement of the Stokes rays;
  \item \emph{Jump matrices} across Stokes rays are the KS wall-crossing factors.
\end{enumerate}
The solution of the RH problem exists and is unique under appropriate convergence conditions on the BPS spectrum.  The DT invariants $\Omega(\gamma)$ ARE the Stokes data of the RH problem.
\end{theorem}

\noindent Our identification adds the shadow tower layer to Bridgeland's framework: the Borel transform of $\Theta_A$ IS the connection of the Bridgeland RH problem, the Borel singularities ARE the Stokes rays, and the alien derivatives ARE the Stokes factors.  The chain reads:
\begin{equation}
\label{eq:identification-chain}
  \underbrace{\Theta_A}_{\text{Shadow tower}} \xrightarrow{\;\;\text{Borel}\;\;} \underbrace{\hat{\Theta}_A}_{\text{Borel plane}} \xrightarrow{\;\text{singularities}\;} \underbrace{\omega = Z(\gamma)}_{\text{Stokes rays}} \xrightarrow{\;\text{Bridgeland RH}\;} \underbrace{\Omega(\gamma)}_{\text{DT invariants}} \xrightarrow{\;\;\text{KS}\;\;} \underbrace{\cA_\cW}_{\text{Wall-crossing}}.
\end{equation}

\subsubsection{Five structural checks}
\label{subsubsec:five-checks}

The five shared structural properties of the Stokes and KS automorphisms are verified computationally.  We summarise them for the record:

\begin{enumerate}[label=\textup{(\arabic*)}]
  \item \textbf{Ray indexing.}  Stokes: lines at $\arg(\omega)$.  KS: walls at $\arg(Z(\gamma))$.  Match: $\arg(\omega) = \arg(Z(\gamma))$.
  \item \textbf{Factorisation.}  Stokes: $\underline{\mathfrak{S}}_\theta = \prod \exp(\cS_\omega \Delta_\omega)$.  KS: $\cA_\cW = \prod \cK_\gamma^{\Omega(\gamma)}$.  Match: elementary factor $\leftrightarrow$ elementary factor.

\begin{theorem}[Stokes $=$ WC for the conifold]
\label{thm:stokes-wc-conifold}
\ClaimStatusProvedHere

  \label{eq:pentagon-stokes}

  where $E(Z) = \prod_{n \geq 0} (1 + q^{n+1/2} Z)^{-1}$ is the compact quantum dilogarithm.

  \[

  \]
\end{enumerate}
\end{theorem}

\begin{proof}

The key point: the factor $E(XY)$ appearing on the right-hand side of~\eqref{eq:pentagon-stokes} IS the Stokes discontinuity: the new trans-series sector that appears when the Borel integration contour crosses the singularity.

\begin{remark}[Perturbative divergence encodes non-perturbative BPS data]

\begin{remark}[The instanton gap is computable]
