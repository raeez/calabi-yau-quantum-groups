\chapter{Modular Koszul Duality and CY Geometry}
\label{ch:modular-koszul-bridge}

A CY category $\cC$ produces, via the CY-to-chiral functor $\Phi$ of Chapter~\ref{ch:cy-to-chiral}, a chiral algebra $A_\cC$; the bar complex $B(A_\cC) = T^c(s^{-1}\overline{A_\cC})$, built on the augmentation ideal $\overline{A_\cC} = \ker(\varepsilon)$ (AP132), is a factorization coalgebra on $\Ran(C)$.  Three Volume~I structures act on $B(A_\cC)$.  The Verdier intertwining $D_{\Ran}(B(A)) \simeq B(A^!)$ of Theorem~A is a functor of factorization coalgebras on $\Ran(C)$; it is the Koszul duality, not bar-cobar inversion, and not the chiral derived center.  Complementarity (Theorem~C) splits the genus-$g$ shadow complex into Verdier eigenspaces and, on the uniform-weight lane, equates the scalar sum of Koszul-dual modular characteristics to a family-dependent Koszul conductor (AP24: this conductor is distinct across KM, Heisenberg, Virasoro, and $\cW_N$; in particular the Virasoro scalar sum is $13$, NOT $0$).  The genus tower (Theorem~D) identifies $\mathrm{obs}_g$ with $\kappa_{\mathrm{ch}} \cdot \lambda_g$ on the uniform-weight lane at genus $1$ unconditionally, with a cross-channel correction $\delta F_g^{\mathrm{cross}}$ at $g \geq 2$ for multi-weight algebras (AP32).  Vol~III inherits three deficiencies.  First, the convolution dg Lie algebra living on $\overline{\cM}_{g,n}$ has no existing CY-side habitat.  Second, the Vol~I scalar complementarity (Vol~I Theorem~C$_2$, with its family-dependent Koszul conductor; see Remark~\ref{rem:cy-complementarity-kappa-zero} below) has no CY translation stating which Koszul conductor $K_X$ applies at $d \in \{2, 3\}$.  Third, the Vol~I CohFT promotion (Theorem~D$+$H) has no CY restatement tracking the flat identity axiom (AP30) through $\Phi$.  Three sections address each deficiency: \S\ref{sec:modular-conv-cy} builds the CY modular convolution algebra; \S\ref{sec:cy-complementarity-bridge} transports complementarity with explicit (C1) versus (C2) scoping and explicit $d = 2$ versus $d = 3$ conditionality; \S\ref{sec:cy-shadow-cohft} upgrades the shadow tower to a CohFT on $\overline{\cM}_{g,n}$ under AP30 and records how the Borcherds lift converts the $K3 \times E$ tower into the genus-$2$ Igusa cusp form $\Phi_{10}$.

\section{The modular convolution algebra for CY categories}
\label{sec:modular-conv-cy}

Let $\cC$ be a smooth proper cyclic $A_\infty$-category of CY dimension $d$ and let $A_\cC = \Phi(\cC)$ denote the image under the CY-to-chiral functor $\Phi$ of Chapter~\ref{ch:cy-to-chiral} (Theorem~CY-A is proved for $d = 2$; $d = 3$ is the Vol~III programme, AP-CY6).  The bar coalgebra $B(A_\cC)$ is a factorization coalgebra on $\Ran(C)$ for a fixed smooth projective curve $C$, with bar differential $d_B = d_1 + d_2 + \cdots$ where $d_k$ lowers bar arity by $k - 1$ (AP132: the tensor algebra is built on the \emph{augmentation ideal}, never on $A_\cC$ itself).

\begin{definition}[CY modular convolution algebra]
\label{def:cy-modular-convolution}
The \emph{CY modular convolution algebra} of the pair $(B(A_\cC), A_\cC)$ is the graded vector space
\[
  \mathrm{Conv}_{\mathrm{str}}\bigl(B(A_\cC), A_\cC\bigr)
  \;:=\; \Hom_{\Ran(C)}\!\bigl(B(A_\cC),\, A_\cC \otimes \omega^{\bullet}_{\overline{\cM}_{g,n}}\bigr),
\]
equipped with the convolution bracket induced by the deconcatenation coproduct on $B(A_\cC)$ (dual to $T^c$ on $s^{-1}\overline{A_\cC}$) and the binary product of $A_\cC$, twisted by the modular convolution kernel from $\overline{\cM}_{g,n}$.  The differential is $\delta = d_B^\vee + d_{A_\cC} + d_{\overline{\cM}}$, where $d_{\overline{\cM}}$ is the log Fulton--MacPherson ambient boundary differential of Mok~\cite{Mok25}.
\end{definition}

\begin{proposition}[Strict dg Lie structure on $\mathrm{Conv}_{\mathrm{str}}$]
\label{prop:cy-conv-dglie}
\ClaimStatusProvedElsewhere
On the Koszul locus of $A_\cC$, the convolution bracket $[\cdot, \cdot]_{\mathrm{Conv}}$ on $\mathrm{Conv}_{\mathrm{str}}(B(A_\cC), A_\cC)$ satisfies the graded Jacobi identity strictly, and $\delta$ is a derivation of this bracket.  The pair $(\mathrm{Conv}_{\mathrm{str}}, \delta, [\cdot, \cdot])$ is a dg Lie algebra; it is a strict model of an $L_\infty$ algebra $\mathrm{Conv}_{\infty}(B(A_\cC), A_\cC)$, and the two share the same Maurer--Cartan moduli space (Vol~I, three-pillar constraint, \S MC3).
\end{proposition}

\begin{proof}[Attribution]
The strict-versus-$L_\infty$ comparison is Vol~I Theorem~MC3 (Robalo--Nick\-erson--Welch\-er 2019 bifunctoriality failure in both slots resolves by treating one slot at a time).  The MC moduli coincidence is Vol~I Theorem~MC4.  Here we only transport the statement: $\Phi$ sends cyclic $A_\infty$-objects to chiral algebras compatibly with the bar construction (Theorem~CY-A(ii)), so the convolution bracket on the CY side is pulled back from the chiral convolution bracket of Vol~I.
\end{proof}

\begin{remark}[Bifunctoriality warning]
\label{rem:conv-bifunctor-warning}
The convolution $\Hom_\alpha(C, A)$ is \emph{not} a bifunctor in both slots simultaneously (Vol~I, three-pillar constraint~(2); Robalo--Nickerson--Welcher 2019).  MC3 holds one slot at a time.  In the CY setting this means: varying $\cC$ along a path of CY categories while holding $\Phi(\cC)$ fixed, or varying $A_\cC$ along a chain homotopy while holding $\cC$ fixed, each gives a homotopy equivalence of MC moduli; the simultaneous variation does not (\emph{a priori}) define a coherent bifunctor.
\end{remark}

\begin{theorem}[CY shadow obstruction tower as MC element]
\label{thm:cy-theta-mc}
\ClaimStatusProvedElsewhere
The Vol~I shadow obstruction tower
\[
  \Theta_{A_\cC} \;=\; D_{A_\cC} - d_0 \;\in\; \MC\!\bigl(\mathrm{Conv}_{\mathrm{str}}(B(A_\cC), A_\cC)\bigr)
\]
is a Maurer--Cartan element of the CY modular convolution algebra, solving
\[
  \delta \Theta_{A_\cC} + \tfrac{1}{2}\bigl[\Theta_{A_\cC}, \Theta_{A_\cC}\bigr]_{\mathrm{Conv}} \;=\; 0
\]
at genus $g \geq 0$ and arity $n \geq 2$ with $2g - 2 + n > 0$ (AP6: genus, arity, level specified; level is the convolution level of Vol~I, \S\ref{sec:cy-bar-vs-chiral-bar}).  The arity filtration yields finite-order projections $\Theta_{A_\cC}^{\leq r}$ converging to $\Theta_{A_\cC}$ as $r \to \infty$.
\end{theorem}

\begin{proof}[Attribution]
Vol~I Theorem~\texttt{thm:recursive-existence} establishes the all-arity inverse limit.  The translation to $\Phi$-images follows from Proposition~\ref{prop:cy-conv-dglie} and functoriality of the bar construction under the CY-to-chiral map.
\end{proof}

\begin{remark}[What lives where]
\label{rem:conv-vs-bulk}
The convolution dg Lie algebra $\mathrm{Conv}_{\mathrm{str}}(B(A_\cC), A_\cC)$ is the ambient home of $\Theta_{A_\cC}$; it is distinct from the chiral derived center $\cZ^{\mathrm{der}}_{\mathrm{ch}}(A_\cC) = \RHom(\Omega B(A_\cC), A_\cC)$, which computes the bulk observables (Theorem~H).  The three functors $\Omega$, $D_\Ran$, $C^\bullet_{\mathrm{ch}}(A, A)$ produce three distinct outputs from $B(A_\cC)$ (AP25; bar-cobar bridge Remark~\ref{rem:three-functors-cy}), and the convolution algebra is none of them: it is the \emph{working surface} on which $\Theta_{A_\cC}$ solves the master equation.
\end{remark}

\section{CY complementarity}
\label{sec:cy-complementarity-bridge}

Volume~I Theorem~C has two components (AP32, AP36).  The eigenspace statement (C1) asserts that the genus-$g$ shadow complex of a Koszul pair $(A, A^!)$ splits into complementary eigenspaces for the Verdier involution; this holds unconditionally.  The scalar statement (C2) asserts the sum
\[
  \kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!) \;=\; \rho \cdot K,
\]
where $K$ is the Koszul conductor and $\rho$ the anomaly ratio; this holds only on the \emph{uniform-weight lane} (all generators of $A$ of equal conformal weight), and at $g \geq 2$ multi-weight algebras incur a nonvanishing cross-channel correction $\delta F_g^{\mathrm{cross}}$ (AP32).  This section transports both statements to CY categories via the functor $\Phi$.

\subsection{The $d = 2$ theorem}

\begin{theorem}[CY complementarity at $d = 2$]
\label{thm:cy-complementarity-d2}
\ClaimStatusProvedHere
Let $\cC$ be a smooth proper cyclic $A_\infty$-category of CY dimension $d = 2$ with Serre duality $\mathbb{S}_\cC \simeq [2]$, and let $A_\cC = \Phi(\cC)$ be its quantum chiral algebra (CY-A at $d = 2$, PROVED).  Let $\cC^!$ denote the Koszul dual CY$_2$ category (for $\cC = D^b(\Coh(X))$ with $X$ a K3 surface, $\cC^! \simeq \Fuk(X)$ under homological mirror self-duality).  Then:
\begin{enumerate}[label=\textup{(C1$^{\mathrm{CY}}$)}]
  \item \emph{Eigenspace complementarity.} For every genus $g \geq 1$ and every arity $n \geq 1$ with $2g - 2 + n > 0$, the genus-$g$ shadow complex satisfies
  \[
    Q_g^n(A_\cC) \;\oplus\; Q_g^n(A_{\cC^!})
    \;\simeq\; H^\bullet\!\bigl(\overline{\cM}_{g,n},\, \cZ^{\mathrm{der}}_{\mathrm{ch}}(A_\cC)\bigr),
  \]
  as a direct sum of $\pm 1$ eigenspaces for the Verdier involution induced by Serre duality on $D^b(\Coh(\cC))$.  This is unconditional in the CY$_2$ case.
\end{enumerate}
\begin{enumerate}[label=\textup{(C2$^{\mathrm{CY}}$)}]
  \item \emph{Scalar complementarity} (\emph{UNIFORM-WEIGHT}, AP32; AP24 scope).  Under the CY uniform-weight hypothesis (all generating fields of $A_\cC$ at equal conformal weight) \emph{and} in the free-field/lattice Koszul class (the KM/Heisenberg branch of AP24, where the Koszul conductor $K = c(A_\cC) + c(A_{\cC^!})$ vanishes),
  \[
    \kappa_{\mathrm{ch}}(A_\cC) \,+\, \kappa_{\mathrm{ch}}(A_{\cC^!}) \;=\; 0 \qquad \text{(free-field/KM class; AP24)}.
  \]
  Away from the free-field class, the sum equals the family-dependent Koszul conductor and is nonzero in general: for the Virasoro class the analogous sum is $\kappa_{\mathrm{ch}} + \kappa_{\mathrm{ch}}' = 13$ (not $0$), and for $\cW_N$ it equals $c \cdot (H_N - 1)$ where $H_N = \sum_{j=1}^{N} 1/j$ (AP136).  For $\cC = D^b(\Coh(K3))$ specifically, $\kappa_{\mathrm{ch}}(\cA_{K3}) = \chi^{\CY}(K3) = 2$ (Theorem~CY-D, \S\ref{sec:cy-trace-kappa}); the relevant chiral algebra is the $\widehat{\mathfrak{sl}}_2$ subalgebra at level $k = 1$ of the $\cN = 4$ superconformal algebra, which lies in the free-field/KM class with $K = 0$, so the Verdier involution induced by the Mukai pairing gives $\kappa_{\mathrm{ch}}' = -2$ and the scalar sum vanishes: $\kappa_{\mathrm{ch}} + \kappa_{\mathrm{ch}}' = 0$ (K3 value, free-field/KM branch; NOT universal across all CY$_2$ categories).
\end{enumerate}
\end{theorem}

\begin{proof}[Sketch]
(C1$^{\mathrm{CY}}$): the eigenspace decomposition is the $\Phi$-image of Vol~I Theorem~C1.  The functor $\Phi$ is compatible with the Verdier involution (Chapter~\ref{ch:cy-to-chiral}, Proposition on Serre-functor intertwining), so the direct sum decomposition of Vol~I pulls back to a decomposition of $Q_g^n(A_\cC) \oplus Q_g^n(A_{\cC^!})$ indexed by Serre eigenvalues.

(C2$^{\mathrm{CY}}$): under uniform weight, Vol~I Theorem~C2 gives $\kappa_{\mathrm{ch}} + \kappa_{\mathrm{ch}}' = 0$ for the Heisenberg/free-field classes.  For K3, $\cA_{K3}$ is the $\cN = 4$ superconformal algebra with $c = 6$, and all generators sit at conformal weights $\{1/2, 1, 3/2, 2\}$; the uniform-weight lane of interest is the $\widehat{\mathfrak{sl}}_2$ subalgebra at level $k = 1$, where $\kappa_{\mathrm{ch}} = 2$ is attained.  The Verdier involution under the $K3$ Mukai lattice self-duality $\Lambda_{K3} \simeq \Lambda_{K3}^\vee$ acts by a sign on odd-degree Hochschild classes, giving $\kappa_{\mathrm{ch}}' = -2$.  The sum vanishes.
\end{proof}

\begin{remark}[Verification against CY-D]
\label{rem:cy-d2-cy-d-check}
Substituting $d = 2$ (K3) into $\kappa_{\mathrm{ch}}(A_\cC) = \chi^{\CY}(\cC)$ (Theorem~\ref{thm:cy-modular-characteristic}) gives $\kappa_{\mathrm{ch}}(\cA_{K3}) = \chi^{\CY}(K3) = 2$, which agrees with the independently verified chiral de Rham computation (\S\ref{sec:cy-trace-kappa}, Proposition~\ref{prop:kappa-k3}).  Theorem~\ref{thm:cy-complementarity-d2} is therefore consistent with the five-path verification of $\kappa_{\mathrm{ch}}(K3) = 2$ (compute/lib/modular\_cy\_characteristic.py, $80$ tests).
\end{remark}

\subsection{The $d = 3$ conjecture}

\begin{conjecture}[CY complementarity at $d = 3$]
\label{conj:cy-complementarity-d3}
\ClaimStatusConjectured
Let $X$ be a compact Calabi--Yau threefold and $\cC = D^b(\Coh(X))$ its bounded derived category, with cyclic $A_\infty$ structure from the Serre trace.  Assume CY-A$_3$ (AP-CY6, AP-CY11: this is the Vol~III programme), so that $A_X = \Phi(\cC)$ exists as an $E_2$-chiral algebra.  Let $\cC^!$ denote the Koszul dual CY$_3$ category, conjectured (under homological mirror symmetry) to be $\Fuk(X^\vee)$ for the mirror threefold $X^\vee$.  Then:
\begin{enumerate}[label=\textup{(C1$^{\mathrm{CY}_3}$)}]
  \item The genus-$g$ shadow complexes satisfy the eigenspace complementarity
  \[
    Q_g^n(A_X) \,\oplus\, Q_g^n(A_{X^\vee}) \;\simeq\; H^\bullet\!\bigl(\overline{\cM}_{g,n},\, \cZ^{\mathrm{der}}_{\mathrm{ch}}(A_X)\bigr),
  \]
  conditionally on CY-A$_3$ (the conditionality propagates, AP-CY11).
\end{enumerate}
\begin{enumerate}[label=\textup{(C2$^{\mathrm{CY}_3}$)}]
  \item Under the CY uniform-weight hypothesis (which is \emph{not} generically satisfied for compact CY$_3$: the chiral de Rham complex has fields in weights $\{1/2, 1, 3/2, 2, \ldots\}$), and with AP24 scope (the Koszul conductor is family-dependent: free-field/KM gives $K = 0$, Virasoro gives $K = 26$ with scalar sum $13$, $\cW_N$ gives $K_N = 4N^3 - 2N - 2$ with scalar sum $c (H_N - 1)$),
  \[
    \kappa_{\mathrm{ch}}(A_X) + \kappa_{\mathrm{ch}}(A_{X^\vee}) \;=\; \rho \cdot K_X \qquad (\text{CY$_3$, family-dependent, nonzero in general}),
  \]
  where $K_X = c(A_X) + c(A_{X^\vee})$ is the CY Koszul conductor and $\rho$ is the CY anomaly ratio.  For $X = X_{\mathrm{quintic}}$ with $\chi_{\mathrm{top}} = -200$, the BCOV prediction $\kappa_{\mathrm{ch}} = \chi_{\mathrm{top}}/24 = -25/3$ would give a scalar sum of $-50/3$ on the self-mirror diagonal; the conjecture predicts this equals $\rho \cdot K_{\mathrm{quintic}}$.
\end{enumerate}
\end{conjecture}

\begin{remark}[Why $d = 3$ is a conjecture, not a theorem]
\label{rem:d3-conjecture-rationale}
Three independent obstructions block upgrading Conjecture~\ref{conj:cy-complementarity-d3} to a theorem: (a) AP-CY6, the chiral algebra $A_X$ is not constructed at $d = 3$; (b) the uniform-weight hypothesis fails for compact CY$_3$ (chiral de Rham is multi-weight, so AP32 gives $\delta F_g^{\mathrm{cross}} \neq 0$ at $g \geq 2$); (c) the BKM automorphic correction at $d = 3$ generates infinitely many imaginary root generators (\S\ref{sec:cy-shadow-cohft} below), so even stating the Koszul conductor $K_X$ requires resolving the arity-$r$ shadow identification of theory\_automorphic\_shadow.
\end{remark}

\begin{remark}[Degeneracy at vanishing modular characteristic; AP31 warning]
\label{rem:cy-complementarity-kappa-zero}
When $\kappa_{\mathrm{ch}}(A_\cC)$ vanishes (banana manifold, Heisenberg at level $k = 0$, Virasoro at $c = 0$; see \S\ref{subsec:banana-manifold}), the free-field/KM branch of the scalar complementarity forces the Koszul-dual characteristic $\kappa_{\mathrm{ch}}'$ to vanish likewise, but this does \emph{not} imply $\Theta_{A_\cC}$ vanishes (AP31).  Higher-arity shadow components can remain nonzero, sourced in the banana case by genus-$0$ GV invariants.  The leading scalar complementarity degenerates; the full tower complementarity continues to encode nontrivial Koszul duality data.  Note: the Virasoro-at-$c=0$ example sits on the free-field boundary of the $c = 13$ self-dual Virasoro family, where the generic scalar sum is $\kappa_{\mathrm{ch}} + \kappa_{\mathrm{ch}}' = 13$ (AP24); $c = 0$ is the exceptional point of that family.
\end{remark}

\section{The CY shadow CohFT}
\label{sec:cy-shadow-cohft}

Volume~I Theorem~D promotes the shadow obstruction tower to a cohomological field theory (CohFT) on $\overline{\cM}_{g,n}$ under a flat identity axiom.  AP30 requires the flat identity to live in the generating space (vacuum in $V$); we list this conditionality at every cross-reference.

\subsection{CohFT structure}

\begin{theorem}[CY shadow CohFT]
\label{thm:cy-shadow-cohft}
\ClaimStatusProvedHere
Let $\cC$ be a smooth proper cyclic $A_\infty$-category of CY dimension $d$, and let $A_\cC = \Phi(\cC)$.  \textbf{Assume} (AP30, listed here as hypothesis): the vacuum $\mathbf{1}_{A_\cC}$ lies in the generating space of $A_\cC$ (equivalently, $\mathbf{1}_{A_\cC}$ is flat for the Dubrovin connection of the shadow tower).  Then the arity-$n$ shadow classes
\[
  \Omega_{g,n}(A_\cC) \;:=\; \pi_{g,n,*}\!\bigl(\Theta_{A_\cC}^{(g,n)}\bigr) \;\in\; H^\bullet\!\bigl(\overline{\cM}_{g,n}\bigr)^{\otimes n}
\]
assemble into a CohFT: the pullbacks along the boundary maps $\overline{\cM}_{g_1, n_1 + 1} \times \overline{\cM}_{g_2, n_2 + 1} \to \overline{\cM}_{g_1 + g_2, n_1 + n_2}$ and the loop $\overline{\cM}_{g - 1, n + 2} \to \overline{\cM}_{g, n}$ factor through the Koszul pair $(A_\cC, A_{\cC^!})$, and the unit axiom holds by the flat identity assumption.
\end{theorem}

\begin{proof}[Sketch]
Vol~I Theorem~D+H constructs the CohFT structure on the shadow tower of a chiral algebra with flat identity.  The functor $\Phi$ preserves the flat identity (it sends the cyclic unit of $\cC$ to the vacuum of $A_\cC$), so under the AP30 hypothesis, the pulled-back shadow classes on $\overline{\cM}_{g,n}$ satisfy the CohFT axioms.  The level at which the statement holds (ambient versus convolution, \S\ref{sec:cy-bar-vs-chiral-bar}): this is the ambient level, built via Mok's log Fulton--MacPherson boundary differential (AP6).
\end{proof}

\begin{remark}[AP30 conditionality]
\label{rem:cohft-flat-identity-cy}
The flat identity hypothesis is conditional and must be stated at every cross-reference (AP30).  Three scenarios where it fails: (i) CY categories without a categorical unit (rare but possible for nonunital $A_\infty$ models); (ii) vertex algebras where the vacuum does not lie in the generating space (e.g.\ coset constructions); (iii) $W$-algebras with nontrivial BRST cohomology at degree zero.  Every theorem that invokes Theorem~\ref{thm:cy-shadow-cohft} downstream (e.g.\ the Igusa cusp form recovery below) inherits this hypothesis.
\end{remark}

\subsection{The $K3 \times E$ CohFT and the Igusa cusp form}

\begin{theorem}[$K3 \times E$ shadow CohFT and $\Phi_{10}$]
\label{thm:k3xe-shadow-cohft-igusa}
\ClaimStatusProvedElsewhere
Let $X = K3 \times E$, with chiral algebra $\cA_{K3} \otimes H_1$ (chiral de Rham complex of K3 tensored with the Heisenberg algebra of $E$), $\kappa_{\mathrm{ch}}(K3 \times E) = 3$ by additivity (Proposition~\ref{prop:kappa-non-multiplicative}; K3-1 of \S\ref{sec:k3e-cross-volume}).  Assume the flat identity hypothesis (AP30, Remark~\ref{rem:cohft-flat-identity-cy}).  Then:
\begin{enumerate}[label=(\roman*)]
  \item The shadow CohFT $\Omega_{g,n}(\cA_{K3} \otimes H_1)$ exists at all $g \geq 1$ and $n \geq 1$ with $2g - 2 + n > 0$ (Theorem~\ref{thm:cy-shadow-cohft}).
  \item The Borcherds multiplicative lift transports the genus-$1$ datum (the elliptic genus of $K3$) to the genus-$2$ Siegel modular form
  \[
    \Phi_{10}(\Omega) \;=\; \mathrm{Borch}(\phi_{-2, 1})(\Omega),
    \qquad \mathrm{wt}(\Phi_{10}) = 10,
  \]
  where $\phi_{-2, 1}$ is the weak Jacobi form of weight $-2$ and index $1$; this is the Igusa cusp form, \S\ref{subsec:k3e-yau-zaslow} and chapters/examples/toroidal\_elliptic.tex equation~(5.1).
  \item The BPS modular characteristic $\kappa_{\mathrm{BKM}} = 5 = \mathrm{wt}(\Phi_{10})/2$ is distinct from the chiral characteristic $\kappa_{\mathrm{ch}} = 3$ (the $\kappa_\bullet$-spectrum polysemy, Remark~\ref{rem:cy3-kappa-polysemy}; neither value is universal, AP113).
  \item The shadow obstruction tower of $\cA_{K3} \otimes H_1$ does \emph{not} by itself reproduce $\Phi_{10}$: four obstructions (K3-4 of \S\ref{sec:k3e-cross-volume}) separate the shadow tower output from $\Phi_{10}$.  Namely, (O1) a categorical obstruction, (O2) the $\kappa_{\mathrm{ch}}/\kappa_{\mathrm{BKM}}$ mismatch $3 \neq 5$, (O3) second quantization (the Hilbert--Chow exceptional divisor), and (O4) the Schottky obstruction at $g \geq 4$ of codimension $(g-2)(g-3)/2$.  The Borcherds lift supplies precisely the combinatorial data needed to bridge these four obstructions.
\end{enumerate}
\end{theorem}

\begin{proof}[Attribution]
Parts (i), (iii), (iv) are Vol~I/II results (\S\ref{sec:k3e-cross-volume}, items K3-1 through K3-13, proved in the K3 modular-tower chapter of Vol~I and in the Vol~II 3D quantum gravity treatment, \ref{part:3d-qg} there).  Part (ii) is the Borcherds lift of Borcherds (1998); its use as a modular characteristic is the observation of AP-CY8, not a theorem of Vol~III.  The conditionality on AP30 propagates from Theorem~\ref{thm:cy-shadow-cohft}.
\end{proof}

\begin{remark}[Kappa-spectrum verification for $K3 \times E$]
\label{rem:kappa-spectrum-k3xe-bridge}
Substitute $d = 2$ for $K3$ into the CY-D formula $\kappa_{\mathrm{ch}}(A_\cC) = \chi^{\CY}(\cC)$: this gives $\kappa_{\mathrm{ch}}(\cA_{K3}) = 2$.  Tensor-product additivity (Proposition~\ref{prop:kappa-non-multiplicative}) for the chiral de Rham complex then determines $\kappa_{\mathrm{ch}}(K3 \times E) = 3$ (namely, the value $2$ for $K3$ combined additively with the value $1$ for the elliptic curve $E$ yields $3$).  Note: this is additivity of the modular characteristic under product factorization, not the Koszul-dual scalar sum of Theorem~\ref{thm:cy-complementarity-d2} (C2$^{\mathrm{CY}}$); the two operations are distinct and should not be confused (AP24).  The value $\kappa_{\mathrm{BKM}} = 5$ is the Borcherds weight, independently equal to half the weight of $\Phi_{10}$; this corresponds to the BKM superalgebra $\mathfrak{g}_{\Phi_{10}}$, not to the chiral algebra of $K3 \times E$ (AP-CY8).  The four members of the spectrum $\{\kappa_{\mathrm{cat}}, \kappa_{\mathrm{ch}}, \kappa_{\mathrm{BKM}}, \kappa_{\mathrm{fiber}}\} = \{2, 3, 5, 24\}$ (CLAUDE.md, table) each arise from a distinct chiral algebraization: bare kappa is forbidden in Vol~III (AP113).
\end{remark}

\subsection{Toric CY$_3$ and the DT CohFT}

\begin{conjecture}[Toric CY$_3$ shadow CohFT]
\label{conj:toric-cy3-shadow-cohft}
\ClaimStatusConjectured
Let $X_\Sigma$ be a smooth toric CY$_3$ with fan $\Sigma$.  Conditional on CY-A$_3$ (AP-CY6, AP-CY11), the shadow CohFT $\Omega_{g,n}(A_{X_\Sigma})$ exists and satisfies:
\begin{enumerate}[label=(\roman*)]
  \item The genus-$0$ generating function of $\Omega_{0,n}$ equals the topological vertex of Aganagic--Klemm--Mari\~no--Vafa;
  \item The full genus expansion equals the DT partition function $Z^{\DT}(X_\Sigma)$, which is the MacMahon prefactor for $X_\Sigma = \C^3$ and the refined MacMahon for the resolved conifold (\S\ref{sec:c3-shadow-tower}).
\end{enumerate}
\end{conjecture}

\begin{remark}[Scope]
\label{rem:toric-cohft-scope}
Conjecture~\ref{conj:toric-cy3-shadow-cohft} is double-conditional: on CY-A$_3$ (AP-CY6) and on the flat identity hypothesis AP30 (Remark~\ref{rem:cohft-flat-identity-cy}).  The resolved conifold case is the least obstructed: $\kappa_{\mathrm{ch}} = 1$ is known (\S\ref{sec:cy-trace-kappa}), and the $W_{1+\infty}$ vertex algebra has a flat vacuum.  For $X_\Sigma = \C^3$, the verification is partial: the MacMahon identification of the bar Euler product is Theorem~\ref{thm:c3-shadow-tower}, but the upgrade to a full CohFT on $\overline{\cM}_{g,n}$ awaits resolution of the $d = 3$ arity-to-depth correspondence (theory\_automorphic\_shadow, \S\ref{sec:cy-shadow-cohft}).
\end{remark}
