\section{CY bar complex vs chiral bar complex}
\label{sec:cy-bar-vs-chiral-bar}

Let $\cC$ be a smooth proper CY$_3$ category with Serre functor $\mathbb{S}_\cC \simeq [3]$, and let $A = A_\cC$ denote the chiral algebra produced by the CY-to-chiral functor $\Phi$ (Theorem~CY-A$_2$ for $d = 2$; Theorem~CY-A$_3$, Theorem~\ref{thm:cy-to-chiral-d3}, for $d = 3$). The cyclic bar complex $\mathrm{CC}_\bullet(\cC)$ and the chiral bar complex $B(A)$ are related by a canonical quasi-isomorphism
\[
 \mathrm{CC}_\bullet(\cC) \;\simeq\; B(A_\cC)
 \tag{CY-A(ii)}
\]
as factorization coalgebras on $\Ran(X)$.

The precise dictionary is as follows.

\begin{proposition}[Bar complex dictionary]
\label{prop:bar-dictionary}
Under the identification CY-A(ii):
\begin{enumerate}[label=(\roman*)]
 \item The cyclic differential $b + B$ on $\mathrm{CC}_\bullet(\cC)$ corresponds to the bar differential $d_B$ on $B(A)$.
 \item The Connes operator $B \colon \HH_n(\cC) \to \HH_{n+1}(\cC)$ corresponds to the degree-preserving component of $d_B$ (the internal differential), while the Hochschild differential $b$ corresponds to the degree-lowering component (the bar contraction).
 \item The $S^1$-action on $\mathrm{CC}_\bullet(\cC)$ (the cyclic rotation) corresponds to the factorization coalgebra structure on $B(A)$, with the cocomposition maps $\Delta_\Gamma$ indexed by stable graphs $\Gamma$.
 \item The Hodge filtration on $\HH_*(\cC)$ corresponds to the degree filtration on $B(A)$: the degree-$r$ piece $B^{(r)}(A)$ captures Hochschild chains of tensor length $\leq r$.
\end{enumerate}
\ClaimStatusProvedHere
\end{proposition}

\begin{remark}[Three functors, three outputs]
\label{rem:three-functors-cy}
The bar complex $B(A)$ is a factorization \emph{coalgebra}. Three distinct functors produce three distinct objects from $B(A)$:
\begin{enumerate}[label=(\arabic*)]
 \item $\Omega(B(A)) \simeq A$ recovers the original algebra (bar-cobar inversion, Theorem~B of Volume~I).
 \item $D_{\Ran}(B(A)) \simeq B(A^!)$ is the Verdier dual, a factorization \emph{algebra} identified with the bar of the Koszul dual $A^!$ (Theorem~A, Convention~conv:bar-coalgebra-identity).
 \item $\cZ^{\mathrm{der}}_{\mathrm{ch}}(A) = \RHom(\Omega(B(A)), A)$ is the chiral derived center, computing the universal bulk (Theorem~H).
\end{enumerate}
These are not the same operation, and the CY cyclic bar complex $\mathrm{CC}_\bullet(\cC)$ corresponds to $B(A)$ itself, not to the derived center and not to the cobar.
\end{remark}

\section{The shadow tower of $\mathbb{C}^3$: MacMahon meets Koszul}
\label{sec:c3-shadow-tower}

The affine threefold $\mathbb{C}^3$ is the simplest case where both sides of the CY-to-chiral correspondence are explicit.

\begin{theorem}[Shadow tower of $\mathbb{C}^3$]
\label{thm:c3-shadow-tower}
\ClaimStatusProvedHere
The shadow obstruction tower of $W_{1+\infty}$ at $c = 1$ satisfies:
\begin{enumerate}[label=(\roman*)]
 \item The bar Euler product equals the inverse MacMahon function:
 \[
 \sum_{n \geq 0} \dim B^{(n)}_{\Einf}(A) \cdot q^n
 \;=\;
 \frac{1}{M(q)},
 \qquad
 M(q) = \prod_{n \geq 1} \frac{1}{(1-q^n)^n}.
 \]
 \item The modular characteristic is $\kappa_{\mathrm{ch}} = 1$, verified by five independent paths:
 \begin{enumerate}[label=(\alph*)]
 \item genus-$1$ free energy: $F_1 = \kappa_{\mathrm{ch}} / 24 = 1/24$;
 \item degree-$0$ MacMahon exponent: $\log M(q) = \sum \sigma_2(k) q^k / k$ matches $\kappa_{\mathrm{ch}} = 1$;
 \item channel decomposition: $\sum_{s \geq 1} \kappa_s^{\mathrm{eff}} = \sum_{s \geq 1} s \cdot (1/s) = \sum 1 = +\infty$ (at finite $N$, $\kappa_{\mathrm{ch}}(W_N) = c(H_N - 1)$; the total $\kappa_{\mathrm{ch}}(W_{1+\infty}, c=1) = \kappa_{\mathrm{ch}}(H_1) = 1$);
 \item Hodge-theoretic: via the categorical trace $\chi^{\CY}(\mathrm{Perf}(\mathbb{C}^3))$;
 \item complementarity: $\kappa_{\mathrm{ch}}(\mathbb{C}^3) + \kappa_{\mathrm{ch}}(\text{dual}) = 0$ (free-field anti-symmetry).
 \end{enumerate}
 \item Per-channel modular characteristics: for the spin-$s$ channel of $W_{1+\infty}$,
 \[
 \kappa_s = \frac{c}{s} = \frac{1}{s},
 \qquad
 \kappa_s^{\mathrm{eff}} = s \cdot \kappa_s = 1 \quad \text{(constant per MacMahon level)}.
 \]
 This constancy is the shadow-DT Rosetta stone: each energy level contributes equally to the modular characteristic, and the MacMahon function $M(q) = \prod (1 - q^n)^{-n}$ is the partition function of a system with effective $\kappa_{\mathrm{ch}} = 1$ per energy level.
\end{enumerate}
\end{theorem}

Verification: \texttt{compute/lib/c3\_shadow\_tower.py} (73 tests), \texttt{compute/lib/macmahon\_shadow\_decomposition.py}.

\begin{remark}[Envelope creates infinite depth from abelian input]
\label{rem:envelope-depth-jump}
The Heisenberg algebra $H_1$ (a single free boson) has shadow depth class~G with $r_{\max} = 2$: the tower terminates at degree~$2$, and $\kappa_{\mathrm{ch}} = 1$ is the only nonvanishing shadow invariant.

The vertex algebra $W_{1+\infty}$ at $c = 1$, the factorization envelope of the Lie conformal algebra of polyvector fields on $\mathbb{C}^3$, has shadow depth class~M with $r_{\max} = \infty$. The spin-$2$ channel (Virasoro at $c = 1$) already has infinite shadow depth, with $\kappa_2 = 1/2$, $\alpha_2 = 2$, $S_4 = 10/27$.

This discrepancy between input (class~G, depth~$2$) and output (class~M, depth~$\infty$) reflects the non-conservativity of the factorization envelope on shadow invariants. Normal ordering, Wick contractions, and OPE singular terms generate higher-degree shadows invisible at the $E_1$-level. Each stage of the operad filtration $E_1 \to E_2 \to E_\infty$ introduces genuinely new invariants.
\end{remark}

\section{Open string field theory: Koszul duality at chain level}
\label{sec:open-sft-koszul}

For $\mathbb{C}^3$, the cyclic $A_\infty$-algebra is the exterior algebra $A = \bigwedge^*(V)$ with $V = \mathbb{C}^3$, equipped with the CY$_3$ trace $\langle a, b \rangle = \Tr(a \wedge b)$ (projection to top degree).

\begin{theorem}[Chain-level Koszul duality for $\mathbb{C}^3$]
\label{thm:c3-koszul-chain-level}
\ClaimStatusProvedHere
The Koszul duality pair for $A = \bigwedge^*(\mathbb{C}^3)$ satisfies:
\begin{enumerate}[label=(\roman*)]
 \item $A^! = \Sym^*(\mathbb{C}^{3,*})$, the symmetric algebra on the dual vector space. This is the $\mathrm{Com}^! = \mathrm{Lie}$ incarnation at the chain level: the exterior algebra (free $\mathrm{Com}$-algebra on $V$) is Koszul dual to the symmetric algebra (free $\mathrm{Lie}$-algebra envelope on $V^*$).
 \item The bar complex $B(\bigwedge^* V)$ has cohomology concentrated in bar degree~$1$:
 \[
 H^*(B(\bigwedge^* V)) \;\simeq\; \Sym^*(V^*[-1]),
 \]
 confirming that $A$ is Koszul (chirally and classically).
 \item The bar-cobar round-trip recovers $A$ itself:
 \[
 \Omega(B(\bigwedge^* V)) \;\xrightarrow{\;\sim\;}\; \bigwedge^* V.
 \]
\end{enumerate}
\end{theorem}

\section{From shuffle product to $\lambda$-bracket: the seven-step chain}
\label{sec:shuffle-to-lambda}

The Schiffmann--Vasserot shuffle presentation of $Y^+(\widehat{\fgl}_1)$ does \emph{not} directly produce the $\lambda$-bracket of $R_{\mathbb{C}^3}$. Seven comparison steps are required.

\begin{construction}[The seven-step chain]
\label{constr:seven-step-chain}
The chain connecting the shuffle algebra to the chiral algebra:
\begin{enumerate}
 \item \textbf{Shuffle algebra} $\mathrm{Sh}(h_1, h_2, h_3)$: the associative algebra with generators $e(z)$ at each spectral parameter $z$ and multiplication by the rational shuffle kernel $\zeta(z_i - z_j) = \prod_a (z_i - z_j + h_a)/(z_i - z_j - h_a)$.
 \item \textbf{Affine Yangian} $Y(\widehat{\fgl}_1)$: the $\RTT$-presentation with generators $e_j, f_j, \psi_j$ satisfying the affine Yangian relations. Schiffmann--Vasserot proved $\mathrm{Sh}^+ \simeq Y^+(\widehat{\fgl}_1)$.
 \item \textbf{$W_{1+\infty}$ modes}: the Feigin--Frenkel realization of $W_{1+\infty}$ as the limit $\lim_{N \to \infty} W_N$, with explicit mode algebra generators $W^s_n$ for spin $s$ and mode $n$.
 \item \textbf{OPE modes}: the vertex algebra OPE $W^s(z) W^{s'}(w) \sim \sum_n C^{ss'}_n(W)(z-w)^{-n-1}$, extracting the singular coefficients $C^{ss'}_n$.
 \item \textbf{$\lambda$-bracket}: $\{W^s_\lambda W^{s'}\} = \sum_n \lambda^{(n)} C^{ss'}_n(W)$, where $\lambda^{(n)} = \lambda^n / n!$ is the divided power.
 \item \textbf{Lie conformal algebra} $R_{\mathbb{C}^3}$: the $\mathrm{GL}(3)$-invariant subalgebra of the Schouten--Nijenhuis Lie conformal algebra on polyvector fields $\mathrm{PV}^*(\mathbb{C}^3)$.
 \item \textbf{Factorization envelope} $U^{\mathrm{ch}}(R_{\mathbb{C}^3})$: the enveloping vertex algebra, which by the Nishinaka--Vicedo construction recovers $W_{1+\infty}$.
\end{enumerate}
Steps 1$\to$2 is Schiffmann--Vasserot. Steps 2$\to$3 is Prochazka--Rapcak. Steps 3$\to$4 is the standard mode-to-OPE dictionary. Step 4$\to$5 requires the divided-power convention. Steps 5$\to$6 is the Kontsevich--Soibelman identification of $R_{\mathbb{C}^3}$ as the $\mathrm{GL}(3)$-invariant part of $\mathrm{PV}^*(\mathbb{C}^3)$ with the Schouten--Nijenhuis bracket. Step 6$\to$7 is the factorization envelope.

The composition $1 \to 7$ is the full CY-to-chiral functor for $\mathbb{C}^3$.
\end{construction}

Verification: \texttt{compute/lib/c3\_lie\_conformal.py}, \texttt{compute/lib/c3\_envelope\_comparison.py} (52 tests).

\begin{warning}[Shuffle product $\neq$ $\lambda$-bracket]
\label{warn:shuffle-neq-lambda}
The shuffle product on $\mathrm{Sh}(h_1, h_2, h_3)$ is an associative product on symmetric functions of the spectral parameters, while the $\lambda$-bracket is a Lie conformal bracket on fields. The two are related by the seven-step chain above, not by a direct identification. In particular, the shuffle kernel $\zeta(z)$ is \emph{not} the OPE kernel: the OPE has poles at $z = 0$ (collision singularities), while the shuffle kernel has poles at $z = \pm h_a$ (Yangian structure function poles). These coincide only after the identification of the parameters $h_a$ with the deformation parameters of $W_{1+\infty}$ and the passage from additive to multiplicative spectral parameters.
\end{warning}

\section{Koszul duality: CY vs chiral}
\label{sec:koszul-cy-chiral}

The chiral Koszul duality of Volume~I (Theorems~A and~B) descends to CY Koszul duality.

\begin{conjecture}[CY Koszul duality dictionary]
\label{conj:cy-koszul-dictionary}
\ClaimStatusConjectured
For a CY$_3$ category $\cC$ with chiral algebra $A = A_\cC$:
\begin{enumerate}[label=(\roman*)]
 \item The chiral Koszul dual $A^!$ corresponds to the \emph{mirror} CY category:
 \[
 A_\cC^! \;\simeq\; A_{\cC^!}
 \]
 where $\cC^!$ is the Koszul dual CY category (for $\cC = D^b(\Coh(X))$ of a CY manifold $X$, this is $\cC^! \simeq \Fuk(X^\vee)$ under homological mirror symmetry).
 \item The complementarity theorem (Theorem~C) gives:
 \[
 Q_g(A_\cC) + Q_g(A_{\cC^!}) = H^*(\overline{\cM}_g, \cZ(A_\cC))
 \]
 where $\cZ(A_\cC) = \cZ^{\mathrm{der}}_{\mathrm{ch}}(A_\cC)$ is the chiral derived center.
 \item The modular characteristics satisfy a family-dependent Koszul conductor relation:
 \[
 \kappa_{\mathrm{ch}}(A_\cC) + \kappa_{\mathrm{ch}}(A_{\cC^!}) = \rho \cdot K
 \]
 where $K$ is the Koszul conductor. For the free-field/KM class (including K3 under Mukai self-duality), $K = 0$ and the sum vanishes. For the Virasoro class, $K = 13$ (the self-dual point $c = 13$). The conductor is family-dependent, not universal.
\end{enumerate}
\end{conjecture}

\section{The five theorems in the CY setting}
\label{sec:five-theorems-cy}

We summarize the status of the five main theorems of Volume~I when specialized to chiral algebras arising from CY categories.

\begin{theorem*}[The five theorems for CY chiral algebras]
Let $A = A_\cC$ be the chiral algebra of a CY$_3$ category $\cC$. Then:
\begin{description}
 \item[Theorem A (adjunction).] The bar-cobar adjunction $B \dashv \Omega$ restricts to CY chiral algebras: $B(A_\cC)$ is a factorization coalgebra on $\Ran(X)$, and $D_{\Ran}(B(A_\cC)) \simeq B(A_{\cC^!})$. The CY identification CY-A(ii) gives $\mathrm{CC}_\bullet(\cC) \simeq B(A_\cC)$.

 \item[Theorem B (inversion).] Bar-cobar inversion $\Omega(B(A_\cC)) \xrightarrow{\sim} A_\cC$ holds on the Koszul locus. For CY categories, chirally Koszul is equivalent to the formality of $\mathrm{CC}_\bullet(\cC)$ as a dg coalgebra.

 \item[Theorem C (complementarity).] The CY Euler characteristic $\chi(\cC)$ splits into complementary halves: $Q_g(\cC) + Q_g(\cC^!)$ recovers the full Hochschild cohomology.

 \item[Theorem D (modular characteristic).] The genus-$g$ obstruction is $\mathrm{obs}_g(A_\cC) = \kappa_{\mathrm{ch}}(A_\cC) \cdot \lambda_g^{\mathrm{FP}}$ on the uniform-weight lane. At $d = 2$ with $h^{1,0} = 0$, $\kappa_{\mathrm{ch}}(A_\cC) = \chi(\cO_X)$ (PROVED; Proposition~\ref{prop:cy-kappa-d2}); at $d = 3$, the formula is dimension-stratified: $\kappa_{\mathrm{ch}} = \chi_{\mathrm{top}}/24$ is the BCOV \emph{prediction} (conjectural; note $\chi_{\mathrm{top}}/24 \neq \chi(\cO_X)$ in general since $\chi(\cO_X) = 0$ for all CY$_3$ by Serre duality, AP-CY34), $\kappa_{\mathrm{ch}} = \kappa_{\mathrm{ch}}(A_S) + \kappa_{\mathrm{ch}}(A_E)$ for products (additivity), and $\kappa_{\mathrm{ch}} \neq \chi(\cO_X)$ in general (Conjecture~\ref{conj:cy-kappa-identification}; Theorem~\ref{thm:chi-neq-kappa}).

 \item[Theorem H (Hochschild).] $\ChirHoch^*(A_\cC)$ is polynomial in degrees $\{0, 1, 2\}$, and the CY Hochschild calculus of $\cC$ (HKR decomposition, Connes operator, categorical Hodge structure) is faithfully reflected in $\ChirHoch^*(A_\cC)$.
\end{description}
\end{theorem*}

\subsection{The banana manifold: $\kappa_{\mathrm{ch}} = 0$ with nontrivial tower}
\label{subsec:banana-manifold}

The banana manifold $X_{\mathrm{ban}}$ (Schoen's CY$_3$, the fiber product of two rational elliptic surfaces over $\mathbb{P}^1$) has Hodge numbers $h^{1,1} = h^{2,1} = 19$ and Euler characteristic $\chi = 0$.

\begin{proposition}[Shadow tower of the banana manifold]
\label{prop:banana-shadow}
The shadow obstruction tower of $X_{\mathrm{ban}}$ has:
\begin{enumerate}[label=(\roman*)]
 \item $\kappa_{\mathrm{ch}} = \chi/24 = 0$. The degree-$2$ scalar shadow \emph{vanishes}.
 \item $S_4 = -44$ (quartic shadow from the genus-$0$ GV invariants $n^0_{d_1, d_2} = -2$ of the banana curve classes).
 \item The shadow tower starts at degree~$4$, not degree~$2$: the cubic shadow is invisible when $\kappa_{\mathrm{ch}} = 0$ (it enters as $\kappa_{\mathrm{ch}} \cdot \alpha$, which vanishes).
 \item The critical discriminant $\Delta = 8 \kappa_{\mathrm{ch}} S_4 = 0$ (since $\kappa_{\mathrm{ch}} = 0$). The standard single-line classification (G/L/C/M) does not directly apply.
 \item Shadow depth class: M (infinite tower) shifted to degree~$4$. The BKY partition function involves infinitely many banana-curve BPS states, generating an infinite shadow tower whose leading term is quartic.
\end{enumerate}
\ClaimStatusProvedHere
\end{proposition}

\begin{remark}[$\kappa_{\mathrm{ch}} = 0$ does not imply $\Theta_A = 0$]
\label{rem:banana-ap31}
The banana manifold is the prototypical example: $\kappa_{\mathrm{ch}} = 0$ does \emph{not} imply $\Theta_A = 0$. The vanishing of $\kappa_{\mathrm{ch}}$ means the bar complex is uncurved ($d^2 = 0$ at the leading scalar level), but the higher-degree components of $\Theta_A$ (the quartic shadow $Q$ and beyond) are nonvanishing, sourced by the instanton contributions (genus-$0$ GV invariants of the banana curves).

Three CY$_3$ examples with $\kappa_{\mathrm{ch}} = 0$ show qualitatively different behaviour:
\begin{itemize}
 \item Heisenberg at $k = 0$: $\kappa_{\mathrm{ch}} = 0$, trivially uncurved, class~G.
 \item Virasoro at $c = 0$: $\kappa_{\mathrm{ch}} = 0$, but higher shadows from contact terms (class~M).
 \item Banana manifold: $\kappa_{\mathrm{ch}} = 0$, but higher shadows from \emph{instantons} (class~M, shifted to degree~$4$).
\end{itemize}
\end{remark}

\subsection{Enriques $\times\, E$: the $\mathbb{Z}_2$ quotient of K3$\,\times\, E$}
\label{subsec:enriques-times-e}

Let $S_{\mathrm{Enr}} = \mathrm{K3} / \iota$ be an Enriques surface ($\iota$ a fixed-point-free involution), and $E$ an elliptic curve. The product $S_{\mathrm{Enr}} \times E$ is a generalized CY$_3$ (the canonical bundle $K_{S_{\mathrm{Enr}} \times E}$ is $2$-torsion, so $h^{3,0} = 0$). Nevertheless, the shadow tower machinery applies to the associated vertex algebra, and the Borcherds lift on $O(2,10)$ provides the automorphic structure.

\begin{theorem}[Shadow tower of Enriques $\times\, E$]
\label{thm:enriques-shadow}
\ClaimStatusProvedHere
The shadow data of Enriques $\times\, E$ is:
\begin{enumerate}[label=(\roman*)]
 \item $\kappa_{\mathrm{BKM}}(\mathrm{Enr} \times E) = 4$, the weight of the Enriques Borcherds product $\Psi$ on $O(2, 10)$ \textup{(}Allcock \textup{2000)}.
 \item Root multiplicities come from the Enriques elliptic genus $\phi_{\mathrm{Enr}} = \tfrac{1}{2} \phi_{0,1}$: the Fourier coefficients are $f_{\mathrm{Enr}}(n, l) = f_{\mathrm{K3}}(n, l) / 2$ \textup{(}all integers since $f_{\mathrm{K3}}$ has even coefficients\textup{)}.
 \item The ratio $\kappa_{\mathrm{BKM}}(\mathrm{K3} \times E) / \kappa_{\mathrm{BKM}}(\mathrm{Enr} \times E) = 5/4$.
 \item Shadow depth class: M \textup{(}infinite tower\textup{)}, driven by the infinite BKM root spectrum.
\end{enumerate}
\end{theorem}

\begin{remark}[The $\mathbb{Z}_2$ quotient on $\kappa_{\mathrm{BKM}}$]
\label{rem:z2-quotient-kappa}
The passage from K3$\,\times\, E$ ($\kappa_{\mathrm{BKM}} = 5$) to Enriques$\,\times\, E$ ($\kappa_{\mathrm{BKM}} = 4$) under the $\mathbb{Z}_2$ quotient does \emph{not} halve $\kappa_{\mathrm{BKM}}$. Rather, the weight of the Borcherds product drops by~$1$: the $O(2,10)$ lattice for the Enriques surface has a different constant term in the Borcherds input form than the $O(3,19)$ lattice for K3, and the weight formula gives $8/2 = 4$ instead of $10/2 = 5$. The ratio $5/4$ is not a simple topological quotient; it reflects the lattice-theoretic structure of the Borcherds lift.
\end{remark}

\subsection{The quintic threefold: non-integer $\kappa_{\mathrm{ch}}$ and BKM obstruction}
\label{subsec:quintic}

The quintic hypersurface $Q \subset \mathbb{P}^4$ has $h^{1,1} = 1$, $h^{2,1} = 101$, $\chi = -200$.

\begin{proposition}[Quintic shadow tower]
\label{prop:quintic-shadow}
The shadow data of the quintic is:
\begin{enumerate}[label=(\roman*)]
 \item $\kappa_{\mathrm{ch}} = \chi/24 = -25/3$ \textup{(}non-integer\textup{)}.
 \item The non-integrality of $\kappa_{\mathrm{ch}}$ obstructs a naive BKM superalgebra: BKM root multiplicities must be integers, and the standard denominator-identity machinery requires integral Borcherds lift weight. The quintic has no natural Borcherds--Kac--Moody structure.
 \item The shadow metric $Q_L(t) = 4\kappa_{\mathrm{ch}}^2 + 12\kappa_{\mathrm{ch}}\alpha t + (9\alpha^2 + 16\kappa_{\mathrm{ch}} S_4) t^2$ is well-defined for fractional $\kappa_{\mathrm{ch}}$. With $\kappa_{\mathrm{ch}} = -25/3$: $Q_L(0) = 2500/9 > 0$.
 \item The GW invariants are known to high degree:
 \[
 n^0_1 = 2875, \quad n^0_2 = 609250, \quad n^0_3 = 317206375, \quad n^0_4 = 242467530000.
 \]
 These are BPS state counts via the Gopakumar--Vafa formula.
 \item Shadow depth class: M \textup{(}infinite tower\textup{)}. The quintic has an infinite BPS spectrum with curves of all degrees and genera; the shadow tower is controlled by the BCOV holomorphic anomaly equation.
\end{enumerate}
\ClaimStatusProvedHere
\end{proposition}

\begin{remark}[$\chi/24$ integrality survey]
\label{rem:chi-24-integrality}
The quantity $\chi/24$ is an integer if and only if $12 \mid (h^{1,1} - h^{2,1})$. Among standard complete intersections:
\begin{itemize}
 \item $\mathbb{P}^5[3,3]$: $\chi = -144$, $\kappa_{\mathrm{ch}} = -6$ (integer).
 \item $\mathbb{P}^5[2,4]$: $\chi = -176$, $\kappa_{\mathrm{ch}} = -22/3$ (non-integer).
 \item $\mathbb{P}^7[2,2,2,2]$: $\chi = -128$, $\kappa_{\mathrm{ch}} = -16/3$ (non-integer).
 \item Quintic: $\chi = -200$, $\kappa_{\mathrm{ch}} = -25/3$ (non-integer).
\end{itemize}
Non-integrality of $\kappa_{\mathrm{ch}}$ is the generic situation for compact CY$_3$ manifolds. A BKM structure requires additional input beyond the BCOV prediction.
\end{remark}

\begin{proposition}[Quintic shadow coefficients from Gromov--Witten data]
\label{prop:quintic-shadow-coefficients}
\ClaimStatusProvedHere{By CY-A$_3$ (Theorem~\textup{\ref{thm:cy-to-chiral-d3}}, proved).}
At the large complex structure limit \textup{(}LCS degeneration type\textup{)}, the first shadow coefficients of the quintic chiral algebra are:
\begin{enumerate}[label=(\roman*)]
 \item $S_2 = \kappa_{\mathrm{ch}} = -25/3$. The scalar shadow, from the genus-$1$ BCOV constant-map contribution $F_1 = \kappa_{\mathrm{ch}}/24 = -25/72$.
 \item $S_3 = \alpha = -3/5$ \textup{(}classical\textup{)}. The cubic shadow, from the Yukawa coupling: $\alpha = C_{111}/\kappa_{\mathrm{ch}} = 5/(-25/3) = -3/5$, where $C_{111} = H^3 = 5$ is the triple intersection number. At the quantum level, $\alpha$ receives instanton corrections of the form $\alpha(q) = -3/5 + \sum_{d \geq 1} \alpha_d \, q^d$, where $\alpha_1 = n^0_1 / \kappa_{\mathrm{ch}} = 2875/(-25/3) = -345$.
 \item $S_4 = 0$ \textup{(}classical\textup{)}. The quartic shadow vanishes classically because $\partial^4 F_0^{\mathrm{cl}} / \partial t^4 = 0$ for the cubic prepotential. Instanton corrections are nonzero: the leading correction is proportional to $n^0_1 \cdot 1^4 / \kappa_{\mathrm{ch}} = -345$. The non-vanishing of $S_4$ after instanton corrections promotes the shadow class from~$\mathbf{L}$ to~$\mathbf{M}$.
\end{enumerate}
The classical shadow tower \textup{(}$S_4 = 0$\textup{)} is class~$\mathbf{L}$ \textup{(}depth~$3$\textup{)}: the shadow metric $Q_L(t) = (-50/3 - 9t/5)^2$ is a perfect square with discriminant $\Delta = 0$. Instanton corrections destroy the perfect-square structure, promoting to class~$\mathbf{M}$ \textup{(}infinite depth\textup{)}.
\end{proposition}

\begin{proof}

The cubic shadow $S_3 = \alpha$ is the normalised Yukawa coupling: the classical prepotential $F_0^{\mathrm{cl}} = (5/6) t^3$ gives $C_{111} = \partial^3 F_0^{\mathrm{cl}} / \partial t^3 = 5$. The classical tower terminates at depth~$3$. For the instanton promotion: $C(q) = 5 + \sum_{d \geq 1} \sum_{d'|d} n^0_{d'} {d'}^3 \, q^d$ has nonzero coefficients at all positive degrees (since $n^0_1 = 2875 \neq 0$), so the instanton-corrected $S_4 \neq 0$, giving $\Delta = 8 \kappa_{\mathrm{ch}} S_4 \neq 0$ and class~$\mathbf{M}$.
\end{proof}

\subsection{Crystal melting: combinatorial vs bar-algebraic degree}
\label{subsec:crystal-melting}

The MacMahon function $M(q)$ counts plane partitions (3D Young diagrams), and the crystal melting model of Okounkov--Reshetikhin--Vafa interprets the DT partition function of $\mathbb{C}^3$ as a statistical-mechanical partition function on the crystal lattice $\mathbb{Z}^3_{\geq 0}$.

\begin{warning}[Combinatorial degree $\neq$ bar degree]
\label{warn:crystal-bar-degree}
The crystal melting ``degree'' (the number of boxes removed from the corner of the crystal) does \emph{not} match the bar complex degree (the tensor length in $B^{(n)}(A)$).

The crystal partition function organizes by the number of boxes:
\[
 M(q) = \sum_{\pi \in \mathcal{PP}} q^{|\pi|}, \qquad |\pi| = \text{number of boxes in plane partition } \pi.
\]
The bar complex organizes by tensor length:
\[
 B(A) = \bigoplus_{n \geq 1} \bar{A}^{\otimes n}[n], \qquad \bar{A}=\ker(\varepsilon),\quad \text{degree } n = \text{number of tensor factors}.
\]

These two gradings are \emph{not} compatible: a plane partition with $k$ boxes does not correspond to a bar element of degree~$k$. Rather, the crystal melting expansion is an exponential reorganization of the bar complex: $\log M(q) = \sum_k \sigma_2(k) q^k / k$ is the free energy, while the bar complex detects the individual OPE modes.

The fundamental mismatch: crystal melting is a \emph{tensor-algebraic} construction (counting 3D diagrams in $\mathbb{Z}^3$), while the bar complex is a \emph{factorization-algebraic} construction (organized by collision patterns on $\Ran(X)$). The two agree in the graded \emph{character} (both reproduce $M(q)$ or its inverse) but not in the \emph{degree structure}.
\end{warning}

\section{Form factors via Koszul duality and the shadow partition function}
\label{sec:form-factors-shadow-pf}

The Costello--Paquette programme computes form factors in holomorphic Chern--Simons theory via Koszul duality between bulk and boundary algebras. The Koszul projection $\pi \colon \Obs_{\mathrm{bulk}} \to \Obs_{\mathrm{bdry}}^!$ factors through the bar complex $B(A)$, and the $n$-particle form factor
\[
 F(\cO;\, z_1, \ldots, z_n) \;=\; \langle \mathrm{vac} \,|\, \cO(0) \,|\, z_1, \ldots, z_n \rangle
\]
is computed by pairing the bulk operator~$\cO$ against $n$-fold bar elements:
\[
 F(\cO;\, z_1, \ldots, z_n)
 \;=\; \bigl\langle \cO,\; s^{-1}a_1 \,|\, \cdots \,|\, s^{-1}a_n \bigr\rangle_{B(A)}.
\]
Here $z$ is the Yangian spectral parameter (not a worldsheet coordinate; the distinction is essential, cf.\ Remark~\ref{rem:z-spectral-vs-worldsheet} in the E$_1$-chiral bialgebra chapter).

\begin{conjecture}[Form factor--shadow PF identity]
\label{conj:ff-shadow-identity}
\ClaimStatusConjectured
For a chiral algebra $A$ arising from a CY category $\cC$ via the functor~$\Phi$, the generating function of integrated form factors equals the shadow partition function:
\[
 \sum_{n \geq 2} \frac{1}{n!} \int_{\cM_{0,n}} F(\cO;\, z_1, \ldots, z_n)\, dz_1 \cdots dz_n
 \;=\; \log \Theta_A(q),
\]
where $\Theta_A = \prod_{k \geq 1} (1 - q^k)^{a_k}$ is the bar Euler product of Volume~I. The integrated $n$-particle form factor equals the shadow invariant at degree~$n$:
\[
 \frac{1}{n!} \int_{\cM_{0,n}} F(\cO;\, z_1, \ldots, z_n)
 \;=\; S_n(A).
\]
\end{conjecture}

The identity is realised differently at each shadow depth class.

\begin{proposition}[Form factor truncation by shadow class]
\label{prop:ff-truncation-by-class}
\ClaimStatusProvedHere
The $n$-particle form factor $F(\cO;\, z_1, \ldots, z_n)$ truncates according to the shadow class of~$A$:
\begin{enumerate}[label=(\roman*)]
 \item \textbf{Class~G} \textup{(}Heisenberg, depth~$2$\textup{)}: only the $2$-particle form factor is nonzero. The Koszul projection $\pi$ is a quasi-isomorphism, and $F(\cO;\, z_1, z_2) = \kappa_{\mathrm{ch}} / (z_1 - z_2)^2$ \textup{(}Wick contraction\textup{)}. All higher form factors vanish: the bar complex is formal, and $B^{\mathrm{der}}(A) = B(A)$.

 \item \textbf{Class~L} \textup{(}Kac--Moody, depth~$3$\textup{)}: the $3$-particle form factor is the first loop correction, with integrated value $S_3 = \alpha$ \textup{(}the cubic shadow\textup{)}. The Koszul projection carries a single $A_\infty$-correction from $m_3$. All form factors with $n \geq 4$ vanish.

 \item \textbf{Class~C} \textup{(}$\beta\gamma$, finite depth\textup{)}: finitely many form factors are nonzero. The correction series converges, and $\pi$ carries finitely many $A_\infty$-corrections.

 \item \textbf{Class~M} \textup{(}Virasoro, infinite depth\textup{)}: \emph{all} form factors are nonzero. The Koszul projection $\pi$ carries infinitely many $A_\infty$-corrections. The integrated form factor series $\sum S_n$ is Gevrey-$1$ divergent and Borel summable. For the spin-$2$ channel of $W_{1+\infty}$ at $c = 1$: $S_2 = 1/2$, $S_3 = 2$, $S_4 = 10/27$ \textup{(}matching the $A_\infty$-bar computation of \S\ref{sec:c3-shadow-tower}\textup{)}.
\end{enumerate}
\end{proposition}

\begin{proof}
The shadow class determines the depth of the $A_\infty$-tower on $B(A)$, hence the maximal degree of nonvanishing bar elements contributing to $\langle \cO, s^{-1}a_1 | \cdots | s^{-1}a_n \rangle$. For class~G, the bar complex is formal ($m_k = 0$ for $k \geq 3$), so only the binary pairing ($n = 2$) contributes. For class~L, $m_3 \neq 0$ but $m_k = 0$ for $k \geq 4$, so only $n \leq 3$ contributes. For class~M, all $m_k$ are nonzero, giving contributions at every degree. The Gevrey-$1$ growth for class~M follows from the Gevrey-$1$ growth of the shadow invariants (Vol~I, Theorem~D). The integrated values $S_n$ match the shadow invariants by the comparison of the moduli-space integral $\int_{\cM_{0,n}}$ with the degree-$n$ projection of the shadow tower (the Costello--Gwilliam factorisation-algebra integration).
\end{proof}

\begin{remark}[Koszul conductor and form factor normalisation]
\label{rem:conductor-ff-normalisation}

\begin{enumerate}[label=(\alph*)]
 \item Heisenberg: $K = 0$. The bulk and boundary form factors are \emph{anti-symmetric}: $S_2(A) + S_2(A^!) = 0$.
 \item Kac--Moody: $K = 0$. Anti-symmetric, with Feigin--Frenkel involution $k \mapsto -k - 2h^\vee$.
 \item Virasoro: $K = 13$. The conductor is nonzero: $c/2 + (26-c)/2 = 13$ for all~$c$. The form factor normalisation absorbs the conductor through the modular correction $\delta F_g^{\mathrm{cross}}$.
\end{enumerate}
\end{remark}

\section{Categorical S-duality from the bar complex}
\label{sec:categorical-s-duality}

S-duality in physics ($\tau \mapsto -1/\tau$) exchanges strong and weak coupling.  The bar-cobar adjunction of Volume~I provides a precise categorical incarnation: the exchange $B \dashv \Omega$ between factorisation algebras and coalgebras \emph{is} the categorical S-duality.

\subsection{The bar-cobar adjunction as S-duality}
\label{subsec:bar-cobar-s-duality}

\begin{proposition}[Bar-cobar = categorical S-duality]
\label{prop:bar-cobar-s-duality}
\ClaimStatusProvedHere
Let $A$ be an $\Eone$-chiral algebra with coupling parameter $g_s$ controlling the bar differential $d_{\mathrm{bar}} = g_s \cdot \mu$ (where $\mu$ is the chiral bracket).  The bar-cobar adjunction
\[
  B \colon \Eone\text{-}\mathrm{ChirAlg} \rightleftarrows \Eone\text{-}\mathrm{ChirCoalg} \cocolon \Omega
\]
exchanges the coupling:
\begin{enumerate}[label=\textup{(\roman*)}]
  \item $B_{g_s}(A)$ carries differential $d = g_s \cdot \mu$;
  \item $\Omega_{1/g_s}(B(A))$ carries differential $d = (1/g_s) \cdot \Delta$;
  \item The bar-cobar inversion $\Omega(B(A)) \simeq A$ on the Koszul locus (Vol~I, Theorem~B) realises the exchange $g_s \leftrightarrow 1/g_s$.
\end{enumerate}
The exchange is an involution: $S^2 = \mathrm{id}$ (applying bar-cobar twice recovers~$A$).  The self-dual point $g_s = 1$ is the point where $B(A)$ and $\Omega(B(A))$ carry isomorphic differentials.
\end{proposition}

\begin{proof}
Items~(i) and~(ii) are the definitions of the bar and cobar differentials respectively.  Item~(iii) is the content of Vol~I Theorem~B restricted to the Koszul locus: $\Omega \circ B \simeq \mathrm{id}$.  The involution property $S^2 = \mathrm{id}$ follows from the unit-counit identities of the adjunction.
\end{proof}

\subsection{The Koszul conductor as S-duality invariant}
\label{subsec:conductor-s-invariant}

\begin{theorem}[Conductor invariance under S-duality]
\label{thm:conductor-s-invariant}
\ClaimStatusProvedHere
The Koszul conductor $K = \kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!)$ is invariant under the S-duality exchange $A \leftrightarrow A^!$:
\[
  K(A) = K(A^!).
\]
\end{theorem}

\begin{proof}
By definition, $K(A) = \kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!)$ and $K(A^!) = \kappa_{\mathrm{ch}}(A^!) + \kappa_{\mathrm{ch}}(A^{!!})$.  On the Koszul locus, bar-cobar inversion gives $A^{!!} \simeq A$, so $\kappa_{\mathrm{ch}}(A^{!!}) = \kappa_{\mathrm{ch}}(A)$ and $K(A^!) = \kappa_{\mathrm{ch}}(A^!) + \kappa_{\mathrm{ch}}(A) = K(A)$.
\end{proof}
