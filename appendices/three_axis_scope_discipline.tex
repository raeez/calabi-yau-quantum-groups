\chapter{The Three-Axis Comparison Calculus}
\label{app:three-axis-scope-discipline}

\providecommand{\Zder}{Z^{\mathrm{der}}_{\mathrm{ch}}}
\providecommand{\GRTone}{\mathrm{GRT}_1(\mathbb{Q})}

An assertion in the Calabi--Yau-to-chiral correspondence carries a
coordinate
\[
 (\ell,\gamma,\mathfrak a),
\]
where \(\ell\) is the vertical level, \(\gamma\) is the chart datum, and
\(\mathfrak a\) is the ambient category.  Equality of mathematical
objects is asserted only inside a fixed coordinate.  Passage between
different coordinates is a comparison morphism, with its hypotheses part
of the morphism.

\section{The Vertical Coordinate}
\label{sec:tas-vertical}

The vertical coordinate takes five values:
\begin{align*}
0 &\quad \cC,\qquad \mathrm{CY}_d\text{-category with cyclic }\Ainf
      \text{-data},\\
1 &\quad \PhiFA_d(\cC)\in E_d\text{-}\HolFA(X),\\
2 &\quad A_X^{(\Sigma_{d-1},C)}
      =\SpCh_{\Sigma_{d-1},C}\PhiFA_d(\cC)
      \in \mathrm{ChirAlg}^{E_{n(d)}}(C),\\
3 &\quad \Zder(A_X),\quad Y^+(X),\quad
      G(X)=D(Y^+(X)),\\
4 &\quad \kappa_{\mathrm{BKM}}(\Phi_N)=c_N(0)/2 .
\end{align*}

The comparison morphisms are:
\begin{align*}
0\longrightarrow 1 &: \quad
  \text{Kontsevich--Tamarkin }E_d\text{-formality and
  Costello--Gwilliam--Li holomorphic locality},\\
1\longrightarrow 2 &: \quad
  \SpCh_{\Sigma_{d-1},C}(-)=
  \int_{\Sigma_{d-1}}(-)\big|_C,\\
2\longrightarrow 3 &: \quad
  \Zder(A)=\mathrm{ChirHoch}^{\bullet}(A,A)
  =\RHom(\Omega B(A),A),\\
3\longrightarrow 4 &: \quad
  \mathrm{Tr}_{\mathrm{cent}}\text{ followed by the Borcherds
  denominator identity.}
\end{align*}
The first arrow is canonical up to the \(\GRTone\)-torsor.  The second
arrow depends on \((\Sigma_{d-1},C)\).  The third arrow uses the bar
resolution.  The fourth arrow is a scalar trace.

Before chart specialisation the shifted-symplectic and Hochschild
levels are
\[
(d,\mathrm{PTVV\ shift},\mathrm{function\ bracket},\mathrm{Hochschild})
\in\{(2,0,\deg 0,E_2),(3,-1,\deg 1,E_3),(4,-2,\deg 2,E_4),(5,-3,\deg 3,E_5)\}.
\]
After \(\SpCh_{\Sigma_{d-1},C}\), \(A\) is \(E_1\) for \(d\geq3\).
The \(E_2\)-braiding then lives on \(\mathcal Z(\Rep^{E_1}(A))\), not
on \(A\) itself.

\section{The Chart Coordinate}
\label{sec:tas-horizontal}

At level \(2\) the chart is
\[
\gamma=(\mathfrak s,\Sigma_{d-1},C,b,\mathfrak w),
\]
where \(\mathfrak s\) is the equivariance stratum, \(b\) is the boundary
vacuum, and \(\mathfrak w\) is the admissibility window.  The strata used
in this volume are:
\begin{itemize}[nosep]
\item toric \(T^d\): \(\mathbb C^3\), local \(\mathbb P^2\), resolved
  conifold;
\item reduced \(\mathbb C^\times+\Aut(X)\): Nikulin or Kummer K3 loci,
  \(K3\times E\), abelian threefolds, K3 fibrations with specified
  automorphism;
\item orbifold inertia \(I(X/G)\): McKay \(\mathbb C^d/\Gamma\),
  \(\Gamma\subset\mathrm{SU}(d)\), and Mathieu-type siblings;
\item lattice-polarised period domains: Borcherds lifts on
  signature \((2,n)\) lattices, Gritsenko \(\Delta_5\), Igusa
  \(\Phi_{10}\).
\end{itemize}

For \(K3\times E\) the five scalars
\[
\bigl\{\kappa_{\mathrm{cat}},\,
\kappa_{\mathrm{ch}}^{\mathrm{Hodge}},\,
\kappa_{\mathrm{ch}}^{\mathrm{Heis}},\,
\kappa_{\mathrm{BKM}}(\Delta_5),\,
\kappa_{\mathrm{fiber}}\bigr\}
=\{0,0,3,5,24\}
\]
come from five constructions at one chart point.  They are not the five
archetype values
\[
K^{\kappa_{\mathrm{ch}}}
=\kappa_{\mathrm{ch}}+\kappa_{\mathrm{ch}}^!
\in\{0,8,13,250/3,25/3\}
\]
on the \(\mathsf G/\mathsf L/\mathsf C/\mathsf M/\mathsf B\) landmark.
The two enumerations meet only at the Mukai-doubling
\(\mathsf B\)-row.

At level \(4\) the chart is the Borcherds input \(\Phi_N\).  On the CHL
ladder \(N\in\{1,2,3,4,6\}\),
\[
\kappa_{\mathrm{BKM}}(\Phi_N)=c_N(0)/2 .
\]
The paramodular input \(\Delta_5\) gives
\(\kappa_{\mathrm{BKM}}(\Delta_5)=5\).  The Fake-Monster input
\(\Phi_{12}\) gives \(\kappa_{\mathrm{BKM}}(\Phi_{12})=12\).

\section{The Ambient Coordinate}
\label{sec:tas-ambient}

The ambient coordinate records the category in which the proof is
formed.  The ambients used here are ordinary chain complexes,
weight-completed complexes, pro-objects, \(J\)-adic completions,
Heisenberg--Siegel sewing, formal-local geometry,
global-with-descent geometry, and derived \(\infty\)-categories.

An identity in one ambient need not survive in another.  The class
\(\mathcal M\) \(E_3\) bar has finite cohomological value \(6^g\) for
\(g\leq3\) in the weight-completed ambient, whereas the ordinary
chain-level complex is infinite.  The implication
\[
W_\infty[\lambda]\Longrightarrow E_\infty
\]
holds only in the admissible window cut out by Proch\'azka triangular
truncation, Creutzig--Kanade--Linshaw parafermions,
Pope--Romans--Shen--Bakas structure constants, and Yamada's weight
condition.

\section{Comparison Lemmas}
\label{sec:tas-examples}

\begin{lemma}[The \(d=3\) chiral output]
For a CY\(_3\)-category satisfying the hypotheses of
Theorem~\ref{thm:cy-to-chiral-d3} and a fixed admissible chart
\((\Sigma_2,C,b,\mathfrak w)\), the expression
\[
\Phi^{(\Sigma_2,C)}_3(\cC)
=\SpCh_{\Sigma_2,C}\PhiFA_3(\cC)
\]
is an \(E_1\)-chiral algebra on \(C\).  The functorial construction is
\(\PhiFA_3\) at level \(1\); the displayed algebra is the chart
specialisation at level \(2\).
\end{lemma}

\begin{lemma}[Bar and bulk]
For an \(E_1\)-chiral algebra \(A\),
\[
\Zder(A)=\mathrm{ChirHoch}^{\bullet}(A,A)=\RHom(\Omega B(A),A).
\]
The bar complex \(B(A)\) is the level-\(2\) twisting object.  The derived
chiral centre is the level-\(3\) bulk algebra.
\end{lemma}

\begin{lemma}[Borcherds weight]
Let \(\Phi_N\) be a Borcherds input in the CHL ladder
\(N\in\{1,2,3,4,6\}\), and let \(c_N(0)\) denote its constant Fourier
coefficient.  Then
\[
\kappa_{\mathrm{BKM}}(\Phi_N)=c_N(0)/2 .
\]
The identity is level \(4\).  It is not an additive formula involving
\(\kappa_{\mathrm{ch}}^{\mathrm{Hodge}}\) and a fibre Euler
characteristic.
\end{lemma}

\begin{lemma}[Mukai orthogonality]
The Mukai pairing on \(K3\) is symmetric on both Hodge-parity pieces.
The structure-preserving finite Lie algebra is therefore
\(\mathfrak{so}(4,20)\).  The Hodge-parity extension
\(\mathfrak{so}(4\mid20)\) is non-Kac.  Kac's
\(\mathfrak{osp}(4\mid20)\) requires a symplectic form on the odd part
and is not the K3 Yangian envelope.
\end{lemma}

\begin{conjecture}[Chain fusion]
\ClaimStatusConjectured
For a CY\(_d\)-category \(\cC_X\) and an admissible chart
\((\Sigma_{d-1},C,b,\mathfrak w)\), the Stage-\(2\) algebra
\[
A_X^{(\Sigma_{d-1},C)}=\Phi_d^{(\Sigma_{d-1},C)}(\cC_X)
\]
equals the boundary algebra \(A_{b(X,\Sigma,C)}\) of the open
factorisation dg-category on \((C,D_C,\tau_C)\), where \(D_C\) records
the orbifold points, fibration punctures, and conifold points inherited
from \(X\).  The toric and local CY$_3$ charts provide constructed
model comparisons; the \(K3\times E\) comparison remains conditional on
the framed \(d=3\) assignment and compact Hall--Borcherds bridge data.
\end{conjecture}

\section{Non-Isomorphisms}
\label{sec:tas-collapses}

The following pairs have comparison morphisms; they are not identical
objects.
\begin{enumerate}[label=\textup{(\roman*)}, leftmargin=*]
\item \(B(A)\) and \(\Zder(A)\).  The comparison is
  \(\Zder(A)=\RHom(\Omega B(A),A)\).
\item \(\PhiFA_d(\cC)\) and
  \(\Phi_d^{(\Sigma_{d-1},C)}(\cC)\).  The comparison is
  \(\SpCh_{\Sigma_{d-1},C}\).
\item A Borcherds product and its operator algebra.  The comparison is
  the centre trace.
\item \(Y^+(X)\) and \(G(X)\).  The comparison is the Drinfeld double
  \(G(X)=D(Y^+(X))\).
\item \(\CoHA(\mathbb C^3)\) and \(\mathcal W_{1+\infty}\).  The
  comparison chain is
  \[
  \CoHA(\mathbb C^3)=Y^+(\widehat{\mathfrak{gl}}_1)
  \hookrightarrow Y(\widehat{\mathfrak{gl}}_1)
  \xrightarrow{\mathrm{ev}_\lambda}
  \End(\mathcal W_{1+\infty}[\lambda]\text{-vac}).
  \]
\item A toric CY\(_3\) statement and a compact non-toric CY\(_3\)
  statement.  The comparison requires orientation, Hall descent,
  radical descent, Serre/PBW equality, Green-adjoint coproduct,
  primitive-centre reduction, associator cohomology, parity,
  separated completion, inverse-limit exactness, and Heegner divisor
  comparison.
\item A weight-completed identity and an ordinary chain identity.  The
  comparison is the completion morphism and its convergence theorem.
\item Three-dimensional Chern--Simons theory and six-dimensional
  holomorphic Chern--Simons theory.  The latter has one-loop anomaly
  governed by the cubic symmetric Casimir \(d^{abc}\) and the quartic
  functional \(\int_X\mathrm{Tr}_{\mathrm{ad}}(A(F_A)^3)\).
\end{enumerate}

\begin{proposition}[Coordinate criterion]
Let \(S\) and \(T\) be assertions in the correspondence, with
coordinates \((\ell_S,\gamma_S,\mathfrak a_S)\) and
\((\ell_T,\gamma_T,\mathfrak a_T)\).  If \(S\) is used to prove \(T\),
then either the coordinates agree, or the proof factors through a
specified comparison morphism from
\((\ell_S,\gamma_S,\mathfrak a_S)\) to
\((\ell_T,\gamma_T,\mathfrak a_T)\).  The hypotheses of this comparison
morphism are hypotheses of the proof of \(T\).
\end{proposition}

\begin{proof}
The coordinates are part of the type of an assertion.  If the
coordinates agree, the proof is internal to one category of objects.  If
one coordinate changes, the proof must include the functor, trace,
completion, specialisation, or comparison morphism that changes that
coordinate.  Omitting that morphism asserts equality in two different
categories.  The hypotheses needed to construct the comparison morphism
are therefore hypotheses of the transported assertion.
\end{proof}
