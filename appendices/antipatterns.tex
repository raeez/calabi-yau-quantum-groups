\chapter{Anti-Pattern Catalogue}
\label{app:antipatterns}

This appendix collects all CY-specific anti-patterns (AP-CY1 through
AP-CY33) and cross-programme anti-patterns (AP150--AP157, FM24) into a
reference table.  Each entry records the failure mode, its severity, and
the counter-measure.  These patterns were identified through systematic
error archaeology across 100+ commits.

\section{Severity levels}
\label{sec:ap-severity}

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
\textbf{Level} & \textbf{Meaning} & \textbf{Action} \\
\midrule
\textsc{Critical} & Theorem status wrong (conjecture $\to$ theorem) & Immediate fix; audit all instances \\
\textsc{High} & Numerical or structural error propagates & Fix before next build \\
\textsc{Medium} & Convention clash or ambiguity & Fix in current session \\
\textsc{Low} & Cosmetic or cross-reference staleness & Fix in batch \\
\bottomrule
\end{tabular}
\end{center}

{%

\toprule
\textbf{ID} & \textbf{Name} & \textbf{Severity} & \textbf{Description and counter} \\
\midrule
\endhead
AP-CY1 & CY dim $\neq$ cpx dim & \textsc{High} &
  $\Fuk(X)$ and $D^b(\Coh(X))$ are CY$_n$ where $n$ is the \emph{complex}
  dimension, not the real dimension $2n$.
  \textbf{Counter}: always state ``CY$_d$ with $d = \dim_\C X$''. \\
\midrule
AP-CY2 & CY trace target & \textsc{High} &
  The CY trace lives in $\HC^-_d(\cC)$ (negative cyclic homology), not just
  $\HH_d \to k$.  The negative cyclic refinement is essential for the
  $\Sd$-framing.
  \textbf{Counter}: always write $\HC^-_d$, never bare $\HH_d \to k$. \\
\midrule
AP-CY3 & $\Etwo \neq$ commutative & \textsc{High} &
  $\Etwo$ braiding is \emph{not} symmetric.  $\Etwo \to \Einf$ loses all
  quantum group structure.
  \textbf{Counter}: never write ``commutative'' for $\Etwo$; write ``braided''. \\
\midrule
AP-CY4 & Drinfeld $\neq$ derived center & \textsc{High} &
  $Z(\cC)$ (monoidal center via half-braidings) $\neq$
  $\Zder(A)$ (Hochschild cochains).  The relationship: Drinfeld center
  categorifies the derived center.
  \textbf{Counter}: always specify which center. \\
\midrule
AP-CY5 & Root-of-unity req. & \textsc{Medium} &
  Kazhdan--Lusztig equivalence requires $q$ a root of unity.  At generic $q$,
  $\Rep_q(\frakg)$ is semisimple.
  \textbf{Counter}: state the $q$-specialization explicitly. \\
\midrule
AP-CY6 & $A_X$ at $d=3$ & \textsc{Critical} &
  $A_X$ for CY3 does \emph{not} exist --- it IS the $d=3$ programme.
  Results depending on $A_X$ at $d=3$ must use \verb|\begin{conjecture}|
  and \verb|\ClaimStatusConditional|, naming CY-A$_3$.
  \textbf{Counter}: decision tree HZ3-1. \\
\midrule
AP-CY7 & CoHA $\neq$ $\Eone$-chiral & \textsc{High} &
  CoHA is associative (Hall product), not a chiral algebra.
  ``$\Eone$-sector of $G(X)$'' assumes $G(X)$ exists.
  \textbf{Counter}: connection is via the functor $\Phi$, not identification. \\
\midrule
AP-CY8 & Borcherds $\neq$ bar Euler & \textsc{High} &
  The identification $\Phi_{10} = $ bar Euler product is an
  \emph{observation} (for K3$\times E$), not a theorem.  Conditional on
  CY-A$_2$ and Vol~I Borcherds-lift identification.
  \textbf{Counter}: cite both CY-A and Vol~I anchor. \\
\bottomrule
\end{longtable}

\endhead
AP-CY9 & Jacobi discriminant & \textsc{High} &
  For $\phi_{k,m}$ of index $m$, only discriminants $D$ with $D \equiv 0$ or
  $3 \pmod{4}$ (at $m=1$) can appear.  Also $c(-1) = 2$ for $\phi_{0,1}$ in

  \textbf{Counter}: verify discriminant constraint before filling tables. \\

AP-CY10 & Flop $\neq$ Koszul dual & \textsc{High} &
  Birational flop $X \dashrightarrow X^+$ preserves
  $\kappa_{\mathrm{ch}}$.  Koszul dual $A^!$ satisfies
  $\kappa_{\mathrm{ch}}(A) + \kappa_{\mathrm{ch}}(A^!) = \rho_K$.
  Flop exchanges chambers; Koszul exchanges algebra/coalgebra.
  \textbf{Counter}: $\kappa_{\mathrm{ch}}(A_X) = \kappa_{\mathrm{ch}}(A_{X^+})$ for flop. \\
\midrule
AP-CY11 & Conditional transitivity & \textsc{Critical} &
  If Result B depends on Result A which depends on CY-A$_3$, then B is
  \emph{also} conditional on CY-A$_3$.  Conditionality propagates.
  \textbf{Counter}: use \verb|\ClaimStatusConditional| with full chain. \\
\midrule
AP-CY12 & Shadow class computation & \textsc{High} &
  G/L/C/M must be computed from the full shadow tower, not from generator
  counting or non-formality ($m_3 \neq 0$) alone.
  local $\bP^2$ is class M (infinite depth), not class L.
  \textbf{Counter}: always compute the full tower. \\
\midrule
AP-CY13 & Stale Part references & \textsc{Low} &
  After any Part restructuring, grep all three volumes for stale
  \texttt{Part\textasciitilde{}[IVXL]} references.
  \textbf{Counter}: use \verb|\ref{part:...}| exclusively. \\
\bottomrule
\end{longtable}

  Any statement whose proof chain passes through $G(X)$ at $d=3$,

  $\Sp_4$ quotient by $\pm I_4$ ($4 \times 4$), not $\pm I_5$.

  \textbf{Counter}: verify matrix dimensions match group rank. \\
\midrule
AP-CY17 & MF CY dimension & \textsc{High} &
  For $W\colon \bA^n \to \bA^1$, $\MF(W)$ is CY$_{n-2}$, not CY$_{n-1}$.
  ADE in 2 variables: CY$_0$.  Need 4 variables for CY$_2$, 5 for CY$_3$.
  \textbf{Counter}: verify $n - 2$ against desired CY dimension. \\
\midrule
AP-CY18 & Lattice theta series & \textsc{Medium} &
  Leech theta: minimum norm${}^2 = 4$, first correction at $q^2$ not $q^1$.
  Never conflate $j(\tau)$ coefficients with $V_\Lambda$ character.
  \textbf{Counter}: verify by direct computation. \\
\midrule
AP-CY19 & $\hat{A}$-genus halving & \textsc{High} &
  $\hat{A}(x) = \frac{x/2}{\sinh(x/2)}$; convergence radius $= 2\pi$
  (first pole of $\sin(x/2)$ at $x = 2\pi$).  Dropping the $/2$ gives
  spurious radius $\pi$.
  \textbf{Counter}: always include the $/2$ in the argument. \\
\midrule
AP-CY20 & Normal bundle $\neq$ spectral & \textsc{High} &
  The $\Z \times \Z$ grading from $N_{C/Y}$ connects to $(q,t)$ through
  the $\Omega$-background, not directly.
  \textbf{Counter}: name the intermediary mechanism (equivariant localization). \\

  Algebraic $E_3$ (Deligne) vs topological $E_3$ (configuration space).

  When $\Phi$ recovers a known invariant, the invariant is not new, only
