\chapter{Conventions}
\label{app:conventions}

This appendix collects the conventions in force throughout this volume and
records the bridges to the conventions of Volumes~I and~II.

\section{Grading conventions}
\label{sec:grading-conventions}

All grading is \textbf{cohomological}: differentials have degree $+1$. The
bar construction uses desuspension (matching Volumes~I--II).

\section{Sign conventions}
\label{sec:sign-conventions}

Signs follow the Koszul rule throughout. For the curved $\Ainf$-structure:
$\mu_1^2(a) = [\mu_0, a]$ (commutator, minus sign).

\section{Categorical conventions}
\label{sec:categorical-conventions}

\begin{itemize}
 \item $\DGCat$ denotes the $\infty$-category of small dg categories over $k$.
 \item $\CY_d\text{-}\Cat$ denotes the $\infty$-category of smooth
 $d$-dimensional CY categories.
 \item Tensor products of dg categories are derived unless stated otherwise.
 \item ``Equivalence'' means quasi-equivalence of dg categories
 (equivalence of associated $\infty$-categories).
\end{itemize}

\section{Operadic conventions}
\label{sec:operadic-conventions}

\begin{itemize}
 \item $E_n$ denotes the little $n$-disks operad (Boardman--Vogt).
 \item $\FM_k(\mathbb{C})$ denotes the Fulton--MacPherson compactification of
 $\mathrm{Conf}_k(\mathbb{C})$. $\FM$ is a blowup along diagonals,
 \emph{not} the complement of the diagonal.
 \item $\SC^{\mathrm{ch,top}}$ denotes the Swiss-cheese operad from
 Volume~II.
\end{itemize}

\section{Bar complex and Koszul duality conventions}
\label{sec:bar-koszul-conventions}

\begin{itemize}
 \item $B(A)$ is the bar complex, a factorization \emph{coalgebra} on $\Ran(X)$.
 \item $\Omega(B(A)) \simeq A$ is bar-cobar \emph{inversion}: it recovers the
 original algebra (Vol~I, Theorem~B). This is NOT Koszul duality.
 \item $A^! = (H^*(B(A)))^\vee$ is the Koszul dual \emph{algebra} (linear
 dual of bar cohomology).
 \item $D_{\Ran}(B(A)) \simeq B(A^!)$ is the Verdier dual
 \textup{(}Vol~I, Theorem~A; convention
 \texttt{conv:bar-coalgebra-identity}\textup{).}
 \item The collision $r$-matrix
 $r(z) = \mathrm{Res}^{\mathrm{coll}}_{0,2}(\Theta_A)$ has pole orders
 \emph{one less} than the OPE, because the bar construction extracts
 residues along $d\log(z_i - z_j)$, which absorbs one power of
 $(z - w)$ (Vol~I).
\end{itemize}


\section{The \texorpdfstring{$\kappa_\bullet$}{kappa-bullet}-spectrum: subscripts and cross-volume reference}
\label{sec:kappa-spectrum}

A Calabi--Yau manifold $X$ gives rise to multiple chiral algebraisations,
each carrying its own modular trace. Within this volume kappa notation
is never used without a subscript; it is always marked by one of four
indices. Let $\cA_X = \Phi_d(D^b(\Coh(X))) = \SpCh_{\Sigma_{d-1},C}(\PhiFA_d(D^b(\Coh(X))))$
denote the chiral algebra attached to $X$ by the two-stage CY-to-chiral
functor (stage-1: $\PhiFA_d$ assembles the $E_d$-holomorphic FA on $X$;
stage-2: $\SpCh_{\Sigma_{d-1},C}$ specialises to the reference curve $C$)
and let $\cG_{\Lambda}$ denote
the Borcherds--Kac--Moody superalgebra attached to a Borcherds-admissible
lattice~$\Lambda$.

\begin{itemize}
\item $\kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(\cA_X)$ is the Hodge modular trace of the
chiral algebraisation: the Hodge supertrace component of the shadow tower
 attached to $\cA_X$ (Vol~I, Section on the shadow tower). At
 $d = 2$, $h^{1,0} = 0$, $\kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(\cA_{K3}) = 2 =
 \chi(\mathcal{O}_{K3})$ (Proposition on $\mathrm{CY}_2$,
 $h^{1,0} = 0$).
 \item $\kappa_{\mathrm{cat}}(X) = \chi(\mathcal{O}_X)$ is the holomorphic
 Euler characteristic of $X$, a topological invariant of the manifold.
 \item $\kappa_{\mathrm{BKM}}(\cG_{\Lambda}) = c_{\Lambda}(0)/2$ is the
 Borcherds weight of the BKM superalgebra (Borcherds 1995, \emph{Invent.\ Math.}\ 120;
 Gritsenko 1999 universal-weight identity).
 \item $\kappa_{\mathrm{fiber}}(X)$ is the Mukai-lattice rank of the K3
 fibre when $X$ is K3-fibered (24 for the Mukai $(4, 20)$ lattice).
\end{itemize}

\paragraph{Manifold versus algebraisation invariants.}
$\kappa_{\mathrm{cat}}$ and $\kappa_{\mathrm{fiber}}$ are topological
invariants of the manifold and cannot vary between algebraisations.
$\kappa_{\mathrm{ch}}$ and $\kappa_{\mathrm{BKM}}$ depend on the
algebraisation: distinct algebraisations of the same manifold can produce
different values. At K$3 \times E$ the spectrum carries five values
$\{0, 0, 3, 5, 24\}$ from five constructions:
$\kappa_{\mathrm{cat}}(K3 \times E) = \chi(\cO_{K3})\cdot\chi(\cO_E) = 2 \cdot 0 = 0$
\textup{(}K\"unneth, total
space\textup{)},
$\kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(\cA_{K3 \times E}) =
 \sum_q(-1)^q h^{0,q}(K3 \times E) = 0$ \textup{(}Hodge supertrace\textup{)},
$\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(\cA_{K3 \times E}) = 3$ \textup{(}Heisenberg-level; cf.\ Remark~\ref{rem:beauville-kappa-formula-subscript-split}\textup{)},
$\kappa_{\mathrm{BKM}}(\cG_{\Delta_5}) = 5$ \textup{(}paramodular $\Delta_5$ input\textup{)},
$\kappa_{\mathrm{fiber}}(K3 \times E) = 24$ \textup{(}Mukai-lattice rank of the K3
fibre\textup{).} The number $\kappa_{\mathrm{cat}}(K3) = 2$ is the K3-fibre
holomorphic Euler witness; it is not a total-space spectrum value.

\paragraph{Cross-volume reference.}
\begin{tabular}{l|l|l|l}
Symbol & Volume & Definition & First appearance \\ \hline
unsubscripted Virasoro conductor symbol & I & Vir conductor: $K_{\mathrm{Vir}} = 26$ & Vol~I conductor chapter \\
$K(A)$ & I & Koszul conductor & Vol~I Theorem on conductor \\
$\kappa_{\mathrm{ch}}$ & III & shadow $r=2$ coefficient & Vol~I shadow tower \\
$\kappa_{\mathrm{cat}}$ & III & $\chi(\mathcal{O}_X)$ & Vol~III \texttt{cy\_d\_kappa\_stratification} \\
$\kappa_{\mathrm{BKM}}$ & III & $c_{\Lambda}(0)/2$ & Vol~III \texttt{kappa\_bkm\_universal} \\
$\kappa_{\mathrm{fiber}}$ & III & K3 fibre Mukai rank & Vol~III \texttt{k3e\_bkm\_chapter} \\
\end{tabular}

The unsubscripted Vol~I Virasoro conductor notation at $c = 13$ gives
$K_{\mathrm{Vir}} = 26$. Vol~III uses only subscripted kappa-symbols.
When a Vol~I formula uses the unsubscripted conductor notation, the
Vol~III correspondent is $\kappa_{\mathrm{ch}}$ unless the
context names a different algebraisation.


\section{The \texorpdfstring{$q$}{q}-convention bridge: \texorpdfstring{$q_{\mathrm{KL}}^2 = q_{\mathrm{DK}}$}{q-KL-2 = q-DK}}
\label{sec:q-convention-bridge}

Two parameter-conventions for quantum groups appear across the three
volumes:
\begin{itemize}
 \item Kazhdan--Lusztig: $q_{\mathrm{KL}} = e^{\pi i / r}$ at $r$-th root of
 unity \textup{(}or formally $q_{\mathrm{KL}} = e^{\pi i \tau}$
 parameterised by the modular variable\textup{).}
 \item Drinfeld--Kohno: $q_{\mathrm{DK}} = e^{2 \pi i / r}$ \textup{(}or
 $q_{\mathrm{DK}} = e^{2 \pi i \tau}$\textup{).}
\end{itemize}

The two conventions are related by the universal identity
\[
 \boxed{\;q_{\mathrm{KL}}^2 \;=\; q_{\mathrm{DK}}.\;}
\]

\paragraph{Origin of the factor 2.}
The KL convention is the natural one for the Kazhdan--Lusztig equivalence
$\Rep_q(\fg) \simeq \widehat{\fg}_k\text{-mod}$ at the affine level
$k = -h^{\vee} + r$ (level shift), where $r$ is the order of the root of
unity. The DK convention is the natural one for the Drinfeld--Kohno
monodromy of the Knizhnik--Zamolodchikov equation, where the half-period
shift in the spectral variable is absorbed into the $q$-parameter. The
square relation reflects the level-doubling under the KL equivalence.

\paragraph{Bridge as KZ cocycle.}
The identity $q_{\mathrm{KL}}^2 = q_{\mathrm{DK}}$ is the universal
Kazhdan--Lusztig $\to$ Drinfeld--Kohno cocycle: it intertwines the two
braided tensor structures on $\Rep_q$ and the trivial change of variable
$\tau \mapsto 2 \tau$ on the upper half plane. Volumes~I--II use the DK
convention; this volume uses the KL convention except where explicitly noted
in the K3-Yangian and CY-C chapters.

When translating a Vol~I formula
$F(q_{\mathrm{DK}})$ to a Vol~III statement $G(q_{\mathrm{KL}})$, the
substitution rule is $G(q_{\mathrm{KL}}) = F(q_{\mathrm{KL}}^2)$.

\begin{remark}[Vol~I canonical reference]
\label{rem:q-bridge-v3-vol1-pointer}
The canonical six-clause statement of the bridge is the Vol~I appendix
\cite{VolI}
(the $q$-convention bridge theorem). The Vol~II companion is
Vol~I appendix \cite{VolI},
which supplies the nine-$\hbar$ zoo. The Vol~III convention applies the bridge
at four representative inheritance sites: the K3 Yangian comparison
(with parameter $q_{\mathrm{KL}}$), the
Maulik--Okounkov stable envelope R-matrix
(trace-form r-matrix; bridge clause~\textup{(iii)} converts to KZ form), the
Costello 5d holomorphic Chern--Simons comparison (DK convention with BV
deformation parameter $\hbar = 2 \pi i / (k + h^{\vee})$), and the
Betti--Finkelberg--Nakajima Coulomb branch construction (inherits from Vol~II
folded BCFG framework). The Chriss--Ginzburg structural reading
interprets the squaring as the index-$2$
$\mathbb{Z}/2$-cover $B_2 \to S_2$ of the Artin braid group on two strands.
\end{remark}


\section{The \texorpdfstring{$\hbar$}{hbar}-convention bridge}
\label{sec:hbar-bridge}

Two $\hbar$ conventions appear across the volumes:
\begin{itemize}
 \item $\hbar_{\mathrm{ch}} = \log q$ (chiral / Yangian convention,
 Vol~I and Vol~III chiral chapters).
 \item $\hbar_{\mathrm{geom}} = (\log q)/(2 \pi i)$ (geometric / KZ
 convention, Vol~II and Vol~III KZ chapters).
\end{itemize}

The bridge is
\[
 \boxed{\;\hbar_{\mathrm{ch}} \;=\; (2 \pi i) \cdot \hbar_{\mathrm{geom}}.\;}
\]

\paragraph{Origin of the factor \texorpdfstring{$2 \pi i$}{2 pi i}.}
The factor $(2 \pi i)^{-1}$ is the KZ-monodromy normalisation: the
Drinfeld associator in $\hbar_{\mathrm{geom}}$ has the natural scaling of an
algebraic-geometric $\hbar$ \textup{(}as in
$q = \exp(2 \pi i \tau)$\textup{);} the chiral $\hbar_{\mathrm{ch}}$ has the
natural scaling of a vertex-algebra deformation parameter
\textup{(}as in OPE coefficients $a(z) b(w) \sim \hbar c(w)/(z - w) +
\cdots$\textup{).}

When two $\hbar$-conventions coexist in a single chapter, the bridge above
must be invoked explicitly. Failure to bridge them produces spurious
factors of $(2 \pi i)^n$ in $n$-loop expressions.

\paragraph{The single-volume rule.}
Within any single chapter, exactly one $\hbar$ is in scope. When a chapter
imports content from another volume, the bridge is invoked at the import
step.


\section{The \texorpdfstring{$\phi_{0,1}$}{phi-0,1} normalisation: K3 elliptic genus and
 \texorpdfstring{$\kappa_{\mathrm{ch}}$}{kappa-ch}}
\label{sec:phi01-normalisation}

The weak Jacobi form $\phi_{0,1}(\tau, z)$ of weight zero and index one
admits two normalisations in the literature:
\begin{itemize}
 \item Gritsenko--Nikulin (EZ-convention): $c(-1) = 1$. Standard reference:
 Eichler--Zagier, \emph{The theory of Jacobi forms}.
 \item K3 elliptic genus (this volume's convention): $c(-1) = 2$.
\end{itemize}

The two normalisations are related by the universal identity
\[
 \boxed{\;\mathrm{ell}(K3, \tau, z) \;=\; 2 \cdot \phi_{0,1}^{\mathrm{GN}}(\tau, z),\;}
\]
i.e.\ the K3 elliptic genus is exactly twice the Gritsenko--Nikulin
normalisation of $\phi_{0,1}$.

\paragraph{The factor 2 is \texorpdfstring{$\kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(K3)$}{kappa-ch-Hodge(K3)}.}
The factor of $2$ on the right-hand side is precisely the
Hodge chiral invariant $\kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(\cA_{K3}) = 2$
of the K3 chiral algebra (Section~\ref{sec:kappa-spectrum}). The Mukai-rank identity
$\chi_{\mathrm{top}}(K3) = 24 = 2 c(0) + 4 c(-1)$ holds in both
normalisations: in the GN convention the right-hand side is
$2 \cdot 10 + 4 \cdot 1 = 24$; in the K3-elliptic-genus convention it is
$2 \cdot 20 + 4 \cdot 2 = 48 = 2 \times 24$, with the
factor \(\kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(K3)=2\) absorbed into both terms.

\paragraph{The single-convention rule.}
Within any single computation, exactly one convention is in scope.
Comparisons between conventions invoke the bridge above explicitly.


\section{Igusa and Oberdieck--Pandharipande normalisation}
\label{sec:igusa-op-normalisation}

Three automorphic products are separated throughout this volume:
\[
 \Delta_5,\qquad \Phi_{10}^{\mathrm{un}} = \Delta_5^2,\qquad
 \Phi_{12}.
\]
Here $\Delta_5$ is the weight-$5$ paramodular K3 denominator,
$\Phi_{10}^{\mathrm{un}}$ is the unnormalised Igusa square, and
$\Phi_{12}$ is the Fake-Monster denominator.  They are related objects,
not interchangeable normalisations of a single symbol.

\paragraph{The OP reduced-DT scalar normalisation.}
The Oberdieck--Pandharipande reduced-DT scalar uses the monic Borcherds
product
\[
 D_5 = 64^{-1}\Delta_5,
 \qquad
 \Phi_{10}^{\mathrm{OP}} = D_5^2 = 4096^{-1}\Delta_5^2.
\]
Consequently
\[
 Z_{\mathrm{OP/DT}}^{K3\times E}
 = -(\Phi_{10}^{\mathrm{OP}})^{-1}
 = -D_5^{-2}
 = -4096\,\Delta_5^{-2}.
\]
A formula written as $-\Delta_5^{-2}$ is therefore only an
unnormalised shorthand if the factor $4096$ is restored before it is
used as an OP/DT scalar.


\section{The Hochschild trinity: geometric, algebraic, bigraded}
\label{sec:hochschild-trinity}

Three Hochschild constructions appear across the three volumes:
\begin{itemize}
 \item $C^{*}_{\mathrm{ch,geom}}(A)$: the geometric chiral Hochschild complex,
 constructed via the Beilinson--Drinfeld factorisation-homology
 construction on the curve $C$ (Vol~II).
 \item $C^{*}_{\mathrm{ch,alg}}(A)$: the algebraic chiral Hochschild complex,
 constructed via the cyclic bar complex with the algebraic colimit
 definition (Vol~III bar chapter).
 \item $\RHH_{\mathrm{ch}}(A) = A \otimes^{\bL}_{A \otimes A^{\op}} A$:
 the bigraded Hochschild complex via derived tensor product, the
 universal definition.
\end{itemize}

\paragraph{The trinity theorem (proved on logarithmic chiral algebras of
finite type).}
For a logarithmic chiral algebra $A$ of finite type \textup{(}admitting
a finite-rank graded presentation\textup{),} the three constructions are
quasi-isomorphic:
\[
 C^{*}_{\mathrm{ch,geom}}(A) \;\simeq\; C^{*}_{\mathrm{ch,alg}}(A) \;\simeq\; \RHH_{\mathrm{ch}}(A).
\]

\paragraph{Scope and counter-examples.}
The trinity holds on the logarithmic-finite-type subcategory. Beyond this
scope:
\begin{itemize}
 \item For non-logarithmic chiral algebras (e.g.\ at the critical level
 $k = -h^{\vee}$), $C^{*}_{\mathrm{ch,geom}}$ and
 $C^{*}_{\mathrm{ch,alg}}$ disagree on Feigin--Frenkel torsion.
 \item For chiral algebras of infinite graded rank (e.g.\ universal vertex
 envelopes of infinite-rank Lie conformal algebras), $\RHH_{\mathrm{ch}}$
 requires a derived completion that the geometric construction does
 not perform.
\end{itemize}

\paragraph{Naming discipline.}
Bare ``Hochschild'' is forbidden in chain-level arguments. Specify
$C^{*}_{\mathrm{ch,geom}}$, $C^{*}_{\mathrm{ch,alg}}$, or
$\RHH_{\mathrm{ch}}$ explicitly. When stated as
$\HH_{\mathrm{ch}}^*(A)$ the convention is $\RHH_{\mathrm{ch}}$ unless
modified.


\section{The B-cycle sign convention}
\label{sec:b-cycle-sign}

For genus $g \geq 1$ Riemann surfaces, the B-cycle integral involves factors
of $i$ (from the imaginary part of the period matrix). The convention in
force throughout this volume:
\[
 \boxed{\;i^2 = -1,\quad |q| < 1,\quad \mathrm{Im}(\tau) > 0.\;}
\]

\paragraph{B-cycle sign convention.}
After any computation involving B-cycle integrals, require that
$|q| < 1$ on the resulting $q$-expansion. If $|q| = 1$ appears, the cause
is invariably an $i^2$ sign error in the B-cycle setup that propagates
through the Riemann bilinear identity. A computation that produces
$|q| = 1$ is incorrect; trace back to the B-cycle integral and require that
$\mathrm{Im}(\tau) > 0$ is preserved by every transformation in the chain.

The $i^2 = -1$ convention is universal across the three volumes. No
chapter uses $i^2 = 1$; if it appears, it is an error.


\section{The Mukai pairing and the intersection form}
\label{sec:mukai-vs-intersection}

Two pairings on the cohomology of K3 appear across the volumes:
\begin{itemize}
 \item The intersection form $\langle \cdot, \cdot \rangle_{\mathrm{int}}$
 on $H^{\mathrm{even}}(K3, \bbZ)$ has signature $(3, 19)$.
 \item The Mukai pairing $\langle \cdot, \cdot \rangle_{\mathrm{Muk}}$ on
 $H^{*}(K3, \bbZ) = H^0 \oplus H^2 \oplus H^4$ has signature
 $(4, 20)$.
\end{itemize}

The two pairings are related by the universal identity
\[
 \boxed{\;\langle (r_0, c, r_4), (r_0', c', r_4') \rangle_{\mathrm{Muk}}
 \;=\; -\, r_0 r_4' - r_0' r_4 + \langle c, c' \rangle_{\mathrm{int}},\;}
\]
i.e.\ the Mukai pairing extends the intersection form by adding the
hyperbolic plane $\bbU = \begin{pmatrix} 0 & -1 \\ -1 & 0 \end{pmatrix}$
on the rank-two summand $H^0 \oplus H^4$. The signature change
$(3, 19) \to (4, 20)$ adds one positive and one negative direction from
$\bbU$.

\paragraph{Origin of the sign.}
The minus sign on $r_0 r_4'$ comes from the Mukai vector convention
$v(\cE) = (\mathrm{rk}(\cE), c_1(\cE), \chi(\cE) - \mathrm{rk}(\cE))$ for
$\cE \in D^b(\Coh(K3))$, where $\chi(\cE) - \mathrm{rk}(\cE)$ is the
\emph{shifted} Euler characteristic. The Mukai vector pairs with itself
giving the holomorphic Euler characteristic:
$\langle v(\cE), v(\cF) \rangle_{\mathrm{Muk}} = -\chi(\cE^\vee \otimes \cF)$.

\paragraph{Use across the volumes.}
The K3-Yangian, the Borcherds lift, and the BKM denominator-identity
analysis all use the Mukai pairing of signature $(4, 20)$ \textup{(}this
volume\textup{).} The classical algebraic-geometric statements about
$D^b(\Coh(K3))$ (Bondal--Orlov classification, derived equivalences) use
the intersection form of signature $(3, 19)$ \textup{(}standard
references\textup{).} When translating between the two, invoke the bridge
above.


\section{The Hodge-filtered supertrace}
\label{sec:hodge-filtered-supertrace}

Throughout the CY-D dimension stratification chapter, the Hodge chiral trace
$\kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(\cA_X)$ for a compact CY$_d$ manifold $X$ is computed
via the Hodge-filtered supertrace
\[
 \kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(\cA_X) \;=\; \sum_{q = 0}^{d} (-1)^q\, h^{0, q}(X).
\]
This is the holomorphic Euler characteristic $\chi(\mathcal{O}_X)$
written as a supertrace on the Hodge column.

For $d = 2$, $h^{1,0} = 0$, the supertrace gives
$\kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(\cA_X) = h^{0,0} - h^{0,1} + h^{0,2} =
1 - 0 + 1 = 2 = \chi(\mathcal{O}_X)$ for K3. For $d = 3$ with $X = K3 \times E$,
the Hodge column is $(1,1,1,1)$ and the supertrace is
$1 - 1 + 1 - 1 = 0 = \chi(\mathcal{O}_{K3 \times E})$. For odd $d$, Serre pairwise
cancellation forces $\kappa_{\mathrm{ch}}^{\mathrm{Hodge}} = 0$ on every compact CY$_d$
\textup{(}Section on the even-$d$/odd-$d$ dichotomy in the CY-D
stratification chapter\textup{).}


\section{The \texorpdfstring{$V_4$}{V-4} Klein-four convention}
\label{sec:V4-convention}

The Klein-four group $V_4 = \langle \epsilon_{\mathrm{wt}},
\epsilon_{\mathrm{par}} \rangle$ acts throughout the K3-Yangian chapter,
where $\epsilon_{\mathrm{wt}}$ is the worldsheet $\bbZ/2$ involution and
$\epsilon_{\mathrm{par}}$ is the Mukai-parity $\bbZ/2$ involution.

\paragraph{\texorpdfstring{$V_4$}{V-4}-coordinate convention.}
A bigraded Lefschetz matrix $M \in \bbZ[V_4]$ is written
$M = (m_{++}, m_{+-}, m_{-+}, m_{--})$ where the four entries are the
projections onto the four $V_4$-characters
$\chi_{\pm \pm}$. The character convention:
$\chi_{ab}(\epsilon_{\mathrm{wt}}^s \epsilon_{\mathrm{par}}^t) = (-1)^{a s + b t}$
with $a, b, s, t \in \bbF_2$ and $+ \mapsto 0$, $- \mapsto 1$.

\paragraph{Antipodal involution.}
The antipodal involution $\sigma_{\mathrm{tot}}^* =
\epsilon_{\mathrm{wt}} \epsilon_{\mathrm{par}}$ acts on $V_4$-coordinate
tuples by
\[
 \sigma_{\mathrm{tot}}^* (m_{++}, m_{+-}, m_{-+}, m_{--}) \;=\; (m_{--}, m_{-+}, m_{+-}, m_{++}).
\]
In Fourier coordinates $\sigma_{\mathrm{tot}}^*$ acts by multiplication by
$(1, -1, -1, 1)$.

\paragraph{\texorpdfstring{$V_4$}{V-4} Fourier transform.}
The Fourier transform on $\bbZ[V_4]$ is the matrix
\[
 \hat M(\chi_{ab}) \;=\; \sum_{s, t} (-1)^{a s + b t} M(g_{st}).
\]
Convolution in $\bbZ[V_4]$ is pointwise multiplication in Fourier
coordinates.


\section{The shadow tower and the modular trace}
\label{sec:shadow-tower-convention}

The shadow tower $\{S_k\}_{k \geq 2}$ of a chiral algebra $A$ is the sequence
of higher-arity Connes--Tsygan trace defects, with $S_2 = \kappa_{\mathrm{ch}}$
the modular trace. The shadow tower is computed recursively from the
$\Ainf$-coproduct corrections of \cite{VolI}. For a compact CY input,
the Hodge branch of this scalar is
$\kappa_{\mathrm{ch}}^{\mathrm{Hodge}}(\cA_X)$, the supertrace of
Section~\ref{sec:hodge-filtered-supertrace}.

\paragraph{Shadow-tower identification.}
The first shadow tower coefficient $S_2(A)$ equals
$\kappa_{\mathrm{ch}}(A)$. Higher coefficients $S_k$ for $k \geq 3$ encode
the $\Ainf$-deformation data; for the K3 chiral algebra these are computed
through $S_8$ in \cite{VolI}.


\section{Cross-volume citation discipline}
\label{sec:cross-volume-citation}

When citing a result from another volume, use the explicit reference form
\[
 \text{Vol~I (resp.\ Vol~II) Theorem (or Section, Lemma, etc.) ...}
\]
rather than implicit references.

\paragraph{Concordance is not proof.}
Agreement between Volume~III and another repository, including
\texttt{\string~/igusa-cusp-form}, is a consistency constraint. It does
not prove an Igusa, Borcherds, OP/DT, Hall, or CoHA assertion. A crossing
claim must name one of four proof inputs: a direct product expansion, an
executable normalisation check, a primary theorem with the convention
conversion written out, or a counterexample to the transported claim.
