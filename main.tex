{\scshape Abstract}
\end{center}

\medskip

\noindent\itshape
Volumes~I--II assume a chiral algebra $\cA$ on a curve~$X$
and extract its modular invariants: the ordered bar
complex $B^{\mathrm{ord}}(\cA) = T^c(s^{-1}\bar\cA)$, the
Maurer--Cartan element $\Theta_\cA$, the modular
characteristic $\kappa_{\mathrm{ch}}(\cA)$, the
$\SC^{\mathrm{ch,top}}$ structure on the derived centre.
Volume~III supplies the algebra: the symmetric
monoidal functor family $\{\Phi_d\}_{d \geq 1}$,
\[
 \Phi_d \colon \CY_d\text{-}\Cat \longrightarrow
 \En\text{-}\mathrm{ChirAlg}(\cM_d),
 \qquad
 n = \begin{cases}\infty, & d = 1, \\ 2, & d = 2, \\
 1, & d \ge 3,\end{cases}
\]
characterised by four universal properties:
(U1)~\emph{Hochschild pullback},
$B^{\mathrm{ord}}(\Phi_d(\cC)) \simeq \mathrm{CC}_\bullet(\cC)$
as factorisation coalgebras;
(U2)~\emph{CY-morphism functoriality}, wall-crossing in
$\cC$ pulls to $R$-matrix gauge transformations on $\Phi_d(\cC)$;
(U3)~\emph{Drinfeld-centre compatibility},
$\cZ(\Rep^{\Eone}(\Phi_d(\cC))) \simeq
\Rep^{\Etwo}(\Zder(\Phi_d(\cC)))$ at $d \ge 3$;
(U4)~\emph{standard-input recovery},
$\Phi_1(\Coh(E)) = $ lattice VOA at $d = 1$,
$\Phi_2(D^b(K3)) = $ Mukai Heisenberg at $d = 2$,
$\Phi_3(\CoHA(\C^3)) = Y^+(\widehat{\fgl}_1)$ at $d = 3$.
The four-step recipe (cyclic $\Ainf \to$ Lie conformal
$\to$ factorisation envelope $\to$ $\mathbb{S}^d$-enhanced
quantisation) proves existence and uniqueness.
\textbf{At $d \ge 3$, $\Phi_d(\cC)$ is $\Eone$-chiral; the
$\Etwo$-braiding lives on $\cZ(\Rep^{\Eone}(\Phi_d(\cC)))$
only}, never on the chiral algebra itself
($\pi_1(\Conf_2(\R^d)) = 0$ for $d \ge 3$).

\textbf{CY-A} (proved, all $d$). $\Phi_d$ exists at every $d$.
At $d = 2$ via Kontsevich--Vlassopoulos $\mathbb{S}^2$-framing
(CY-A$_2$). At $d = 3$ via the $\infty$-categorical vanishing
$\HH^{-2}_{\Eone}(\cC) = 0$ and Goodwillie contractibility of
the $E_3$-lifting space (CY-A$_3$,
Theorem~\ref{thm:derived-framing-obstruction}). At $d \ge 4$
via $\Eone$-stabilisation
(Theorem~\ref{thm:e1-stabilization-cy}); the effective framing
obstruction $\Obs_{\mathrm{eff}}(d)$ is the colimit of the complex
tower $\pi_d(BU) = \mathrm{KU}^{-d}$ (2-periodic: $\Z$ at even $d$,
$0$ at odd $d$) and the symplectic refinement
$\pi_d(B\Sp) \subset \pi_d(BO) = \mathrm{KO}^{-d}$ (8-periodic, real
Bott). $\Obs_{\mathrm{eff}}(d) = 0$ at $d \bmod 8 \in \{1, 3, 7\}$,
is $\Z$-valued at every even $d$ (from $\pi_d(BU)$), and is
$\Z/2$-valued at $d \equiv 5 \pmod 8$ (from the Sp-refinement, since
$\pi_d(BU) = 0$ at odd $d$).

\textbf{CY-B} (proved, $d = 3$). $\Eone$-Koszul duality on
$\cA = \Phi_3(\cC)$ induces an $\Etwo$-braided equivalence on
$\cZ(\Rep^{\Eone}(\cA))$ via the Verdier spectral functor,
identified page-by-page on the bar spectral sequence.

\textbf{CY-D} (proved, $d = 2$; programme at $d \ge 3$).
At $d = 2$,
$\kappa_{\mathrm{ch}}(\Phi_2(\cC)) = \chi^{\CY}(\cC)$ via
Serre duality $\mathbb{S}_\cC = [2]$, which kills the
one-loop correction. At $d \ge 3$ the topological identity
$\kappa_{\mathrm{ch}} = \chi(\cO_X)$ fails: Serre's pairing
$h^{0,q} = h^{0,d-q}$ forces $\chi(\cO_X) = 0$ at every odd
$d$, while $\kappa_{\mathrm{ch}}^{\mathrm{Heis}}$ remains nonzero by
additivity under products
($\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(E) = 1$,
$\kappa_{\mathrm{ch}}^{\mathrm{Heis}}(K3 \times E) = 3$).
Theorem~\ref{thm:kappa-stratification-by-d} records a
Beauville--Bogomolov tri-stratum at $d \ge 3$ (odd-$d$ Serre
stratum, strict-CY even-$d$, holomorphic-symplectic) with
distinct mechanisms on each branch.

\textbf{CY-C} (conjectural). The quantum vertex chiral group
$G(X)$ is unconstructed in general. The abelian level
$C(\frakg, q) = D(Y^+(\frakg_{K3}))$ is explicit. For $K3 \times E$,
the original six-way isomorphism is \emph{falsified}: at the
generator level the routes stratify by lattice rank
$\rho^{R_i} \in \{3, 12, 24\}$
(Theorem~\ref{thm:cy-c-six-routes-generator-level-convergence}),
not by $\kappa_{\mathrm{ch}}$, which is route-independent as
a categorical invariant of $\Phi_3(\cC)$. The five non-source
routes assemble into
a pentagon colimit over named intertwiners
$\beta_{13}, \beta_{34}, \beta_{45}, \beta_{56}, \beta_{61}$
(Proposition~\ref{prop:cy-c-pentagon-colimit}); $R_2$ (Borcherds
branch) is the source. The six routes are six \emph{different
constructions witnessing the same $\Phi_3$-output}, not six
$\Phi_3$-applications.

\textbf{Mukai Lagrangian, two faces.}
$\Lambda_X \subset H^{\mathrm{even}}(X, \Q)$ with
$\langle v, w \rangle_{\mathrm{Muk}}
 = -\!\int_X v^\vee \cdot w \cdot \mathrm{td}(X)$
carries a structural redundancy. \emph{Arithmetic face}: the
Borcherds singular theta lift targets $\Lambda_X$ with weight
$c(0)/2$ (Theorem~\ref{thm:borcherds-lift-universal}).
\emph{Lie face}: $\CoHA(Q_X, W_X) \cong Y^+(\widehat{\frakg}_X)$
(Schiffmann--Vasserot, Rap\v{c}\'ak--Soibelman--Yang--Zhao) has
$\Lambda_X$ as weight lattice, with universal $R$-matrix from
Maulik--Okounkov stable envelopes on $\mathrm{Hilb}^n(X)$.
Both encode the BPS generating series on $\Lambda_X$. The
functor family $\{\Phi_d\}_{d \geq 1}$ is the arithmetic-to-Lie bridge; the Borcherds
lift is the Lie-to-arithmetic bridge.

\textbf{K3 Yangian (the crown).} The crown jewel is
Part~\ref{part:k3-yangian}: $Y(\mathfrak{g}_{K3})$ with $24$
generators, Mukai-signature Serre relations, and degree-$(24, 24)$
structure function. First nontrivial quantum group constructed
from CY geometry via $\Phi_2$. Conjectural envelope
$Y_{\osp(4 \mid 20)}$ (the orthosymplectic super-Yangian preserving
the Mukai orthogonal form of signature $(4, 20)$; the Mukai form is
symmetric indefinite, not $\Z/2$-super-graded, so the correct home
is $\osp$, not $\fgl$).
At $K3 \times E$ the chiral-Borcherds chain produces the Igusa
denominator $\Phi_{10} = \Delta_5^2$, and the
$\kappa_\bullet$-spectrum
$\{\kappa_{\mathrm{cat}} = 0, \kappa_{\mathrm{ch}} = 3,
\kappa_{\mathrm{BKM}} = 5, \kappa_{\mathrm{fiber}} = 24\}$
records four invariants from four distinct constructions.

\textbf{Borcherds-weight universal} for K3-fibered Class~A.
$\kappa_{\mathrm{BKM}} = c_N(0)/2$ universal across the eight
diagonal $\Z/N\Z$-orbifolds and the STU model with $N \in \{1,
\ldots, 6\}$ (Proposition~\ref{prop:bkm-weight-universal}).
\emph{Undefined for Class~B} (quintic, $\C^3$, conifold,
local $\bP^2$): replacement is $\kappa_{\mathrm{BCOV}}
= \chi(X)/24$ and shadow depth (conditional on CY-A). The
naive decomposition $\kappa_{\mathrm{BKM}}
= \kappa_{\mathrm{ch}} + \chi(\cO_{\mathrm{fiber}})$
is the $N = 1$ K3$\times E$ numerical coincidence; it FAILS
for all $N \geq 2$
(Remark~\ref{rem:bkm-decomposition-adversarial}).

\textbf{Universal trace identity (cross-volume bridge).}
The Vol~I conductor $K(\cA) = -c_{\mathrm{ghost}}(\BRST(\cA))$
and the Vol~III Borcherds weight $\kappa_{\mathrm{BKM}}(X)
= c_N(0)/2$ are evaluations of one universal trace formula,
mediated by $\kappa_{\mathrm{ch}}$ and bridged through $\{\Phi_d\}_{d \geq 1}$
(per-family verified for K3-fibered Class~A; conjectural at
programme level).

\textbf{$d = 5$.} $\Phi_5$ proved at
$X = K3 \times K3 \times E$ via K\"unneth on the Hodge diamond
$h^{0,q}(X) = (1, 1, 2, 2, 1, 1)$; the $\Z/2$-gerbe obstruction
vanishes by Whitney $+$ Wu
(Construction~\ref{constr:phi-5-family}).

\textbf{Compact quintic (refutation).}
Strict formality of $A_{\mathrm{BVDB}}
= \End^\bullet(\bigoplus_{i=0}^4 \cO_{X_5}(i))$ is refuted:
Yukawa coupling $Y_3 = H^3 = 5$ is the non-zero
Kodaira--Spencer $m_3$. Correct statement: curved formality
with Yukawa as BCOV datum.

Thirteen structural results beyond the functor.
(1)~\emph{Stokes $=$ wall-crossing}: Borel singularities of the
shadow tower are walls of marginal stability; trans-series
sectors are BPS sectors.
(2)~\emph{Non-abelian MTC from 6d geometry}: at level $N = 3$,
the K3 quantum group produces a non-abelian modular tensor
category with $3200$ modules, quantum dimension
$d_1 = \sqrt{2}$, and Ising fusion: the first such category
from six-dimensional geometry.
(3)~\emph{Chiral volume conjecture}:
$\lim_{N \to \infty} (1/N) \log |Z_N| = (1/2\pi)\, |AJ(C)|$,
the Abel--Jacobi period.
(4)~\emph{BCOV $=$ shadow tower}: genus-$g$ holomorphic anomaly
amplitudes equal the integrated degree-$(g + 1)$ shadow
invariants.
(5)~\emph{Mock modular K3} proved at $d = 2$ in four steps:
shadow $24\,\eta^3$, mock theta transform, Zwegers completion,
Borcherds lift.
(6)~\emph{Beauville--Bogomolov stratification} of $\kappa_{\mathrm{ch}}$
at $d \geq 3$: odd-$d$ Serre vanishing, strict-CY even-$d$,
holomorphic-symplectic. The naive topological formula
$\kappa_{\mathrm{ch}} = \chi(\cO_X)$ fails on products
(additivity vs Künneth vanishing); the Hodge-supertrace
formulation holds only in the restricted scope where
categorical $\kappa_{\mathrm{ch}}$ and $\chi(\cO_X)$ coincide.
(7)~\emph{Chiral Khovanov complex} (conjectural):
$\mathrm{CKh}_{2,n}$ has dimension~$2^g$ with Mukai bigrading.
(8)~\emph{Niemeier rigidity}:
$24 = \mathrm{rank}(\widetilde\Lambda) =
\#\{\text{Niemeier lattices}\}$, a structural identity, not a
coincidence.
(9)~\emph{Categorical DT}:
$D^b(B_n(H_{\mathrm{Muk}})) \simeq D^b(\mathrm{Hilb}^n(K3))$.
(10)~\emph{ZTE correction matrix $T$} computed explicitly
(exact rational, persymmetric).
(11)~Shadow tower computed through $m_{10}$.
(12)~\emph{No native CY$_4$ Yangian}: the first Pontryagin
class $p_1 \in \pi_4(BU) = \Z$ is the obstruction \emph{group},
blocking the $E_4$ structure.
(13)~\emph{$E_n$-hierarchy}: $E_3$-bar cohomology is
$(1 + t)^{3g} = 2^{3g}$ for shadow classes~$\mathbf{L}$,
$\mathbf{C}$ and $6^g$ for class~$\mathbf{M}$.

\part{Foundations: CY Categories and Cyclic $\Ainf$}
\label{part:cy-categories}

\noindent
The point of departure is the categorical datum: a Calabi--Yau category
$\cC$ of dimension~$d$, equipped with a cyclic $\Ainf$-structure encoding
the non-degenerate trace $\Tr \colon \HH_\bullet(\cC) \to k[-d]$.
This part establishes the language in which the rest of the volume is
written.  Everything here is $d$-independent and unconditional: the
definitions and constructions apply equally to CY surfaces, threefolds,
and higher-dimensional categories.  The reader who has worked through
Volumes~I and~II will recognize the bar-cobar machine and the modular
characteristic $\kappa_{\mathrm{ch}}$ as the output of a chiral algebra;
the present volume constructs the \emph{input} to that machine, and
this first part prepares the categorical raw material from which the
input is fashioned.

Chapter~\ref{ch:introduction} introduces the central question (how
does CY geometry produce chiral algebras?), states the four main
theorems CY-A through CY-D with their current status, and provides a
guide for the reader with three recommended paths through the volume.
Chapter~\ref{ch:cy-categories} develops CY categories: smooth proper
dg categories with Serre functors, the CY pairing, and the four
standard families (derived categories, Fukaya categories, matrix
factorizations, and quantum group representations).
Chapter~\ref{ch:cyclic-ainf} constructs cyclic $\Ainf$-structures and
the Connes $B$-operator: the CY trace lives in negative cyclic
homology $\HC^-_d(\cC)$, not merely as a map $\HH_d \to k$, and the
$\mathbb{S}^d$-framing that governs the operadic level of the output
is encoded in this cyclic refinement.  Chapter~\ref{ch:hochschild-calculus}
builds the Hochschild calculus (homology, cohomology, the
Gerstenhaber bracket, and the Lie conformal algebra
$\mathfrak{L}_\cC$) that mediates between the categorical CY
condition and the factorization envelope of
Part~\ref{part:bridge}.

The key theorem of this part is structural rather than dramatic: it
is the identification of the Hochschild complex $\HH_\bullet(\cC)$
as the correct receptacle for the CY data, carrying both the
$\mathbb{S}^d$-action from the trace and the Gerstenhaber bracket
from the $\Ainf$-structure.  These two pieces of data (the framing
and the bracket) are precisely what the functor family~$\{\Phi_d\}_{d \geq 1}$ of
Part~\ref{part:bridge} consumes.

\part{The CY-to-Chiral Functor}
\label{part:bridge}

\noindent
The CY-to-chiral correspondence is a family $\{\Phi_d\}_{d \geq 1}$:
\[
 \Phi_d \colon \CY_d\text{-}\Cat
 \longrightarrow
 \En\text{-}\mathrm{ChirAlg}(X),
 \qquad
 n = \begin{cases} 2 & d = 2, \\ 1 & d \ge 3. \end{cases}
\]
Input: a smooth proper CY-category $\cC$ with cyclic $\Ainf$-structure
and CY trace $\Tr \colon \HC^-_d(\cC) \to k$. Output: a chiral algebra
$A_\cC$ on a smooth curve $X$. Four steps.

\begin{enumerate}[label=\textup{(\arabic*)}]
\item $\mathfrak{L}_\cC := \bigl(\HH_{\bullet+1}(\cC),\ \{\cdot_\lambda \cdot\},\ \partial\bigr)$:
Lie conformal algebra with $\lambda$-bracket from the Gerstenhaber
bracket on $\HH^\bullet$, translation $\partial$ from the $S^1$-action.
\item $A_\cC^{(0)} := U^{\mathrm{fact}}_X\bigl(\mathfrak{L}_\cC\bigr)$:
factorization envelope (Beilinson--Drinfeld).
\item $\mathbb{S}^d$-framing:
Kontsevich--Vlassopoulos enhancement of the factorization structure.
At $d = 2$: $\Etwo$-chiral. At $d \ge 3$: $\Eone$-stabilized.
\item Quantization: Maurer--Cartan closure.
\end{enumerate}

\smallskip
\noindent\textbf{CY-A (proved at every $d$).}
\begin{itemize}[leftmargin=1.5em,itemsep=1pt]
\item[($d = 2$)]
$\mathbb{S}^2$-framing (Kontsevich--Vlassopoulos) $+$ factorization
envelope (Costello--Gwilliam) produce an $\Etwo$-chiral algebra.
\item[($d = 3$)]
$\infty$-categorical: $\HH^{-2}_{E_1}(\cC) = 0$ by unit-connectedness,
the space of $E_3$-liftings is contractible, all Goodwillie layers
vanish. Chain-level $[m_3, B^{(2)}] \ne 0$ is \emph{not} an obstruction.
\item[($d \ge 4$)]
$\Eone$-stabilization
(Theorem~\ref{thm:e1-stabilization-cy}):
$\pi_d(BU) \ne 0$ at $d$ even prevents additional $\Eone$-factors;
the output is $\Eone$.
\end{itemize}

Chapter~\ref{ch:cy-to-chiral} proves CY-A at every $d$.
Chapter~\ref{ch:m3-b2-saga} recounts the $[m_3, B^{(2)}]$ obstruction
analysis at the chain level (three retracted proofs, four

$\{b, B^{(2)}\} = 0$ on the total differential with cross-arity

Chapter~\ref{ch:quantum-chiral-algebras} provides a second,

\item[(H1)] product $\mu \colon A \otimes A \to A$ associative,
\item[(H2)] spectral coproduct $\Delta_z \colon A \to A \otimes A$

\item[(H4)] strong unitarity $R_{12}(z) R_{21}(-z) = \mathrm{id}$,
\item[(H5)] antipode $S \colon A \to A^{\mathrm{op}}$ from Verdier
duality $A^! = D_{\Ran}(B^{\mathrm{ord}}(A))$.
\end{itemize}

\part{Frontiers}
\label{part:frontier}

\noindent\textbf{The structural residue.}
CY-A is the existence direction: it closes in every dimension,
including CY-A$_3$ at the $\infty$-categorical level, with
chain-level $\Ainf$-compatible $S^3$-framing for non-formal CY$_3$
(quintic, generic hypersurface) retained in the pro-object and
J-adic ambients of the raw bar complex but open at the direct-sum
chain level. What CY-A does not touch is structure: the Langlands
kernel attached to $\Phi$, the closure of the Zamolodchikov
tetrahedron at $O(\kappa_{\mathrm{ch}}^2)$, the $\Sp_4(\Z)$
covariance of the $E_3$ $S$-matrix, and the non-abelian K3 Yangian
extending the abelian case. These four problems are the minimal
set of structural obstructions whose resolution would promote
CY-A from an existence theorem to a classification; the present
Part is their catalogue.

\smallskip
\noindent\textbf{Front~1: nonabelian $Y(\mathfrak{g}_{K3})$.}
Abelian: $24$ Heisenberg generators, Mukai-signature Serre (PROVED).
Nonabelian extension: matrix Miura transform on
$V_k(\widehat{\fsl}_2)$; $\fsl_2$ Serre constraint
\[
 P_2(D)
 \;=\;

 \;=\;
 0,

 D = 3,
\]
($P_2 = 0$ at leading order proved; exact-vanishing conjectural).
The super-Yangian candidate sitting over the Mukai lattice

$Y_{\mathfrak{osp}(4|20)}$, not $Y(\mathfrak{gl}(4|20))$: the Mukai
form is symmetric indefinite (orthogonal), not $\Z/2$-super-graded,

$\mathfrak{so}(4) \oplus \mathfrak{sp}(20)$ and odd part
